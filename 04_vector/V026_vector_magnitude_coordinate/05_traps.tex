\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忘记根号）}}]
			错误地将模长公式写为 $|\vec{a}| = x^2+y^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（牢记根号）：}}]
			模长公式必须为 $|\vec{a}| = \sqrt{x^2+y^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式印象浅）：}}]
			只记住了平方和，忽略了开方步骤。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（模长当绝对值和）}}]
			错误地认为 $|\vec{a}| = |x|+|y|$。
		\item[\textbf{\textcolor{CTrap}{正确理解（平方和开方）：}}]
			模长公式是基于勾股定理的平方和再开方，不能拆成绝对值相加。
		\item[\textbf{\textcolor{CTrap}{错因分析（类比错误）：}}]
			受到实数绝对值影响，将二维长度误认为横纵坐标的线性组合。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忘记平方运算）}}]
			计算 $|\overrightarrow{AB}|$ 时写为 $\sqrt{(x_2-x_1)+(y_2-y_1)}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（先平方再求和）：}}]
			应先分别计算坐标差的平方，再求和，公式为 $\sqrt{(x_2-x_1)^2+(y_2-y_1)^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆公式）：}}]
			误将距离公式与平均公式或加法混合公式混淆。
	\end{description}

\end{trapbox}
