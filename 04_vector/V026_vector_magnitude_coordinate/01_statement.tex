		\item[\textbf{\textcolor{CStatement}{条件 1（坐标分量已知）：}}]
			向量 $\vec{a}$ 的坐标分量 $x$ 和 $y$ 已知。
		\item[\textbf{\textcolor{CStatement}{条件 2（起点终点已知）：}}]
			起点 $A(x_1,y_1)$ 和终点 $B(x_2,y_2)$ 的坐标已知。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（坐标模长公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			向量模长等于横纵坐标平方和的开方。
			\[
				|\vec{a}| = \sqrt{x^{2} + y^{2}}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（两点距离公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			模长等于终点减起点坐标差平方和的开方。
			\[
				|\overrightarrow{AB}| = \sqrt{(x_2-x_1)^{2} + (y_2-y_1)^{2}}.
			\]
	\end{description}
