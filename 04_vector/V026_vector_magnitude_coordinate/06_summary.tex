\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{向量模长等于坐标平方和的算术平方根，也等于两点间距离。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（向量坐标已知）：} 已知向量 $\vec{a}$ 的坐标分量 $(x,y)$。
		\item \textbf{条件 2（起点终点已知）：} 已知起点 $A(x_1,y_1)$ 和终点 $B(x_2,y_2)$ 的坐标。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（遗忘根号）：} 直接取平方和作为模长。
		\item \textbf{检查项（坐标差平方）：} 注意先计算坐标差的平方再求和。
	\end{itemize}

\end{summarybox}
