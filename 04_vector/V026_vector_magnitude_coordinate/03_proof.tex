\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	通过构造直角三角形，利用勾股定理推导两点间的距离公式，从而得到向量模长公式。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（向量坐标差）：}]
			设 $A(x_1,y_1)$、$B(x_2,y_2)$，则 $\overrightarrow{AB}=(x_2-x_1,y_2-y_1)$。
			\[
				\overrightarrow{AB} = (x_2-x_1, y_2-y_1).
			\]
			\par\textbf{理由：} 向量坐标等于终点坐标减去起点坐标。
		\item[\textbf{步骤二（直角边长度）：}]
			过点 $A$ 作 $x$ 轴平行线、过点 $B$ 作 $y$ 轴平行线，交于点 $C(x_2,y_1)$。
			\[
				AC = |x_2-x_1|,
			\]
			\[
				BC = |y_2-y_1|.
			\]
			\par\textbf{理由：} 平行于坐标轴的线段长度由对应坐标差决定。
		\item[\textbf{步骤三（勾股定理）：}]
			在直角三角形 $ACB$ 中，斜边 $AB$ 满足：
			\[
				\begin{aligned}
					|AB|^2 &= AC^2 + BC^2 \\ &= (x_2-x_1)^2 + (y_2-y_1)^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 直角三角形斜边平方等于两直角边平方和。
		\item[\textbf{步骤四（算术平方根）：}]
			模长取正值，得
			\[
				|\overrightarrow{AB}| = \sqrt{(x_2-x_1)^2 + (y_2-y_1)^2}.
			\]
			\par\textbf{理由：} 长度非负，开方后取算术平方根。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{当起点为原点 $O(0,0)$，终点为 $A(x,y)$ 时，即得向量的模长公式：$|\vec{a}| = \sqrt{x^2+y^2}$。}
\end{proofbox}
