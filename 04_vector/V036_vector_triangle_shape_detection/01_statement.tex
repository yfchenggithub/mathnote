\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（不共线三点）：}}]
			平面内不共线的三点 $A,B,C$，构成三角形 $ABC$。
		\item[\textbf{\textcolor{CStatement}{条件2（边向量记号）：}}]
			记 $\overrightarrow{AB},\overrightarrow{AC},\overrightarrow{BC}$ 为此三角形的边向量。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（直角判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若两邻边向量的数量积为零，则该夹角为直角。
			\[
				\overrightarrow{AB}\cdot\overrightarrow{AC}=0 \iff \angle A = 90^\circ
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（等腰判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若两邻边向量的模相等，则以这两边为腰的三角形为等腰三角形。
			\[
				|\overrightarrow{AB}| = |\overrightarrow{AC}| \iff AB = AC
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（等边判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若三边向量的模均相等，则该三角形为等边三角形。
			\[
				|\overrightarrow{AB}| = |\overrightarrow{AC}| = |\overrightarrow{BC}| \iff AB = AC = BC
			\]
	\end{description}

\end{statementbox}
