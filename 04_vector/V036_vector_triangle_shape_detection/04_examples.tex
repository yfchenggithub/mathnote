\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 顶点 $A(1,2),\ B(4,5),\ C(2,1)$，判断该三角形的形状。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求向量坐标）：}]
			$\overrightarrow{AB} = (4-1,5-2) = (3,3)$，$\overrightarrow{AC} = (2-1,1-2) = (1,-1)$。
			\par\textbf{理由：}终点坐标减去起点坐标得到向量坐标。
		\item[\textbf{第二步（计算数量积）：}]
			$\overrightarrow{AB}\cdot\overrightarrow{AC} = 3\times1 + 3\times(-1) = 0$。
			\par\textbf{理由：}利用平面向量数量积的坐标公式 $x_1x_2+y_1y_2$。
		\item[\textbf{第三步（判定形状）：}]
			由 $\overrightarrow{AB}\cdot\overrightarrow{AC}=0$ 知 $\angle A = 90^\circ$，故 $\triangle ABC$ 为直角三角形。
			\par\textbf{理由：}向量数量积为零等价于两向量垂直，即夹角为 $90^\circ$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\triangle ABC$ 是直角三角形，直角位于顶点 $A$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$\overrightarrow{AB}\cdot\overrightarrow{AC}=0$ 且 $|\overrightarrow{AB}|=|\overrightarrow{AC}|$，请判断 $\triangle ABC$ 的形状。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（垂直判定）：}]
			$\overrightarrow{AB}\cdot\overrightarrow{AC}=0$。
			\par\textbf{理由：}根据直角判定结论，该条件等价于 $\angle A=90^\circ$，三角形有一个直角。
		\item[\textbf{第二步（等腰判定）：}]
			$|\overrightarrow{AB}|=|\overrightarrow{AC}|$。
			\par\textbf{理由：}根据等腰判定定理，该条件推出 $AB=AC$，三角形是等腰三角形。
		\item[\textbf{第三步（综合形状）：}]
			结合以上两点，$\triangle ABC$ 中既有直角又有两腰相等，故为等腰直角三角形。
			\[
				\angle A=90^\circ,\; AB=AC.
			\]
			\par\textbf{理由：}等腰直角三角形的定义：一个直角且两直角边相等（即两腰相等）。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\triangle ABC$ 是以 $A$ 为直角顶点的等腰直角三角形。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(0,0), B(4,0), C(2,2\sqrt{3})$，判断 $\triangle ABC$ 的形状。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求向量坐标）：}]
			$\overrightarrow{AB}=(4,0)$，$\overrightarrow{AC}=(2,2\sqrt{3})$，$\overrightarrow{BC}=(-2,2\sqrt{3})$。
			\par\textbf{理由：}坐标相减。
		\item[\textbf{第二步（计算模长）：}]
			$|\overrightarrow{AB}| = \sqrt{4^2+0^2}=4$，$|\overrightarrow{AC}| = \sqrt{2^2+(2\sqrt{3})^2}=4$，$|\overrightarrow{BC}| = \sqrt{(-2)^2+(2\sqrt{3})^2}=4$。
			\par\textbf{理由：}模长公式 $\sqrt{x^2+y^2}$。
		\item[\textbf{第三步（判定形状）：}]
			由于 $|\overrightarrow{AB}| = |\overrightarrow{AC}| = |\overrightarrow{BC}|$，根据等边判定条件，$\triangle ABC$ 为等边三角形。
			\par\textbf{理由：}三边向量的模全相等等价于三边相等，即等边三角形。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\triangle ABC$ 是等边三角形。}
\end{examplebox}
