\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{点 $P$ 到直线 $l$ 的距离等于向量 $\vec{AP}$ 与方向向量 $\vec{v}$ 的叉积模除以 $\vec{v}$ 的模。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（点与方向）：} 已知直线 $l$ 上一点 $A$ 和方向向量 $\vec{v}\neq\mathbf{0}$。
		\item \textbf{条件2（坐标数据）：} 点 $A$、$P$ 和 $\vec{v}$ 的坐标已知，可计算 $\vec{AP}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（叉积与绝对值）：} 分子必须取绝对值，叉积公式勿混淆。
		\item \textbf{检查项（向量非零）：} 使用前确认 $\vec{v}\neq\mathbf{0}$，否则分母为零无意义。
	\end{itemize}
\end{summarybox}
