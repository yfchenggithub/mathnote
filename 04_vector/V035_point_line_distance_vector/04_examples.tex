\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l$ 过点 $A(1,2)$，方向向量 $\vec{v}=(2,1)$，求点 $P(4,6)$ 到直线 $l$ 的距离 $d$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（向量坐标）：}] 计算 $\vec{AP}=(4-1,\;6-2)=(3,4)$。
			\par\textbf{理由：} 根据坐标差定义。

		\item[\textbf{第二步（叉积计算）：}] 叉积 $\vec{AP}\times\vec{v}=3\times1-4\times2=3-8=-5$，取绝对值 $\lvert\vec{AP}\times\vec{v}\rvert=5$。
			\par\textbf{理由：} 平面向量叉积公式。

		\item[\textbf{第三步（距离求解）：}] 计算 $d = \dfrac{\lvert\vec{AP}\times\vec{v}\rvert}{\lvert\vec{v}\rvert} = \dfrac{5}{\sqrt{2^2+1^2}} = \dfrac{5}{\sqrt5} = \sqrt5$。
			\par\textbf{理由：} 代入向量法距离公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boldsymbol{d=\sqrt5}$。整个计算仅需坐标差与叉积，无需解垂足。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l$ 的方程为 $2x+3y-6=0$，求点 $P(1,2)$ 到 $l$ 的距离。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（取点方向）：}] 从直线方程取点 $A(0,2)$，方向向量 $\vec{v}=(3,-2)$（因为法向量为 $(2,3)$，方向向量与法向量垂直）。
			\par\textbf{理由：} 直线方向向量可由法向量旋转得，取一点便于计算。

		\item[\textbf{第二步（向量坐标）：}] 计算 $\vec{AP}=(1-0,\;2-2)=(1,0)$。
			\par\textbf{理由：} 坐标差。

		\item[\textbf{第三步（距离求解）：}] 叉积 $\vec{AP}\times\vec{v}=1\times(-2)-0\times3=-2$，$\lvert\vec{AP}\times\vec{v}\rvert=2$；$\lvert\vec{v}\rvert=\sqrt{3^2+(-2)^2}=\sqrt{13}$，故 $d=\dfrac{2}{\sqrt{13}}=\dfrac{2\sqrt{13}}{13}$。
			\par\textbf{理由：} 代入向量法距离公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boldsymbol{d=\dfrac{2\sqrt{13}}{13}}$。该例说明一般式直线也可通过构造方向向量与一点套用向量公式。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 的顶点 $A(1,2),\;B(3,5),\;C(6,1)$，求顶点 $A$ 到边 $BC$ 所在直线的距离 $d$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（方向向量）：}] 取 $BC$ 方向向量 $\vec{v}=C-B=(3,-4)$，并取点 $B(3,5)$ 作为直线上一点。
			\par\textbf{理由：} 由两点可确定方向向量。

		\item[\textbf{第二步（向量坐标）：}] 计算 $\vec{BA}=A-B=(1-3,\;2-5)=(-2,-3)$。
			\par\textbf{理由：} 坐标差。

		\item[\textbf{第三步（距离求解）：}] 叉积 $\vec{BA}\times\vec{v}=(-2)\times(-4)-(-3)\times3=8+9=17$；$\lvert\vec{v}\rvert=\sqrt{3^2+(-4)^2}=5$；$d=\dfrac{17}{5}=3.4$。
			\par\textbf{理由：} 代入向量法距离公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boldsymbol{d=3.4}$。此方法可快速计算三角形高线长度，体现向量法在几何中的应用。
\end{examplebox}
