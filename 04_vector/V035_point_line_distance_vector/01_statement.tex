	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（点与方向）：}}] 直线 $l$ 过点 $A(x_0,y_0)$，方向向量 $\vec{v}=(v_x,v_y)\neq\mathbf{0}$。对任意点 $P(x_1,y_1)$，记 $\vec{AP}=(x_1-x_0,\;y_1-y_0)$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（向量叉积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			距离等于以 $\vec{AP}$ 和 $\vec{v}$ 为邻边的平行四边形的高。
			\[
				d = \frac{\lvert\vec{AP}\times\vec{v}\rvert}{\lvert\vec{v}\rvert}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（坐标展开）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			使用坐标直接计算叉积的绝对值。
			\[
				d = \frac{\lvert (x_1-x_0)v_y - (y_1-y_0)v_x \rvert}{\sqrt{v_x^2+v_y^2}}
			\]
	\end{description}

\end{statementbox}
