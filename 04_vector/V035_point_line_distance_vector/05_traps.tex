\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（叉积分式混淆）}}] 错误记叉积公式为 $\vec{AP}\times\vec{v}=(x_1-x_0)v_x - (y_1-y_0)v_y$。
		\item[\textbf{\textcolor{CTrap}{正确理解（记忆要点）：}}] 二维叉积为 $\vec{AP}\times\vec{v}=(x_1-x_0)v_y - (y_1-y_0)v_x$，可视为行列式。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆点积）：}}] 易与点积公式 $(x_1-x_0)v_x + (y_1-y_0)v_y$ 混淆，造成符号和项错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗漏绝对值）}}] 直接将叉积值（可能为负数）代入分子，得到负距离。
		\item[\textbf{\textcolor{CTrap}{正确理解（取绝对值）：}}] 距离恒为非负数，必须对叉积取绝对值：$\lvert\vec{AP}\times\vec{v}\rvert$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视符号）：}}] 叉积可正可负，表示有向面积；距离标量无方向。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（方向向量为零）}}] 使用零向量作为方向向量，导致分母为零或方向不确定。
		\item[\textbf{\textcolor{CTrap}{正确理解（非零限制）：}}] 方向向量必须满足 $\vec{v}\neq\mathbf{0}$，否则直线无定义。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略前提）：}}] 零向量没有方向，不能代表直线，公式不适用。
	\end{description}
\end{trapbox}
