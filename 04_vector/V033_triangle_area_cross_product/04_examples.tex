\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$O(0,0), A(3,4), B(1,2)$，求$\triangle OAB$的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（向量表示）：}] 写出向量 $\vec{OA}=(3,4)$，$\vec{OB}=(1,2)$。
			\par\textbf{理由：} 由点坐标直接得到向量坐标。
		\item[\textbf{第二步（套用公式）：}] 计算面积
			\[
				S = \frac12|3\times2 - 4\times1| = \frac12|6-4| = 1.
			\]
			\par\textbf{理由：} 代入原点三角形面积公式 $S=\frac12|x_1y_2 - x_2y_1|$。
		\item[\textbf{第三步（得出结果）：}] 所以 $\triangle OAB$ 的面积 $S=1$。
			\par\textbf{理由：} 计算绝对值并取半。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{S=1}$.

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 顶点为 $A(1,2)$, $B(4,5)$, $C(2,7)$，求其面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平移处理）：}] 将点 $A$ 平移至原点：令 $O'(0,0)=A$，则 $B' = (4-1,5-2) = (3,3)$，$C' = (2-1,7-2) = (1,5)$。
			\par\textbf{理由：} 三角形平移不改变面积。
		\item[\textbf{第二步（套用公式）：}] 以 $O'$ 为原点，$B',C'$ 为顶点，面积
			\[
				S = \frac12|3\times5 - 3\times1| = \frac12|15-3| = 6.
			\]
			\par\textbf{理由：} 应用原点坐标面积公式。
		\item[\textbf{第三步（得出结论）：}] 故 $\triangle ABC$ 的面积为 $6$。
			\par\textbf{理由：} 平移后面积不变，直接得到原三角形面积。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{S=6}$.

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $O(0,0)$, $A(a,2)$, $B(3,4)$ 三点构成三角形，且面积为 $5$，求实数 $a$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（向量表示）：}] 写出 $\vec{OA}=(a,2)$，$\vec{OB}=(3,4)$。
			\par\textbf{理由：} 由坐标直接得到。
		\item[\textbf{第二步（列方程）：}] 由面积公式得
			\[
				\frac12|a\times4 - 2\times3| = 5 \quad\Longrightarrow\quad |4a - 6| = 10.
			\]
			\par\textbf{理由：} 代入 $S=\frac12|x_1y_2 - x_2y_1|$，等式两边同乘2。
		\item[\textbf{第三步（解方程）：}] 由 $|4a-6|=10$ 得 $4a-6=10$ 或 $4a-6=-10$。解得 $a=4$ 或 $a=-1$。
			\par\textbf{理由：} 绝对值方程拆分为两个一次方程求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{a=4}$ 或 $\boxed{a=-1}$.
\end{examplebox}
