\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{以原点为顶点的三角形面积等于两向量叉积绝对值的一半：$S=\frac12|x_1y_2 - x_2y_1|$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（原点顶点）：} 三角形必须有一个顶点在原点 $O$。
		\item \textbf{条件2（坐标已知）：} 其余两顶点的坐标 $(x_1,y_1)$ 和 $(x_2,y_2)$ 必须明确给出或可计算。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（必加绝对值）：} 计算时不要遗漏绝对值符号，因为 $x_1y_2 - x_2y_1$ 可能为负。
		\item \textbf{检查项（共线验证）：} 代入公式前应先验证三点是否共线；共线时面积为 $0$，公式仍然成立。
	\end{itemize}
\end{summarybox}
