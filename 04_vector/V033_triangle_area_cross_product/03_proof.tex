\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	\textbf{使用向量叉积的几何意义，将三角形面积转化为两个向量叉积模的一半，然后通过坐标运算得到具体的表达式。}

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（向量设点）：}]
			设 $\vec{OA} = (x_1, y_1)$，$\vec{OB} = (x_2, y_2)$，则三角形 $OAB$ 的面积 $S$ 满足
			\[
				S = \frac12 | \vec{OA} \times \vec{OB} |.
			\]
			\par\textbf{理由：}
			以 $\vec{OA},\vec{OB}$ 为邻边的平行四边形面积等于 $|\vec{OA} \times \vec{OB}|$，而三角形面积是平行四边形面积的一半。

		\item[\textbf{步骤二（叉积运算）：}]
			计算 $\vec{OA} \times \vec{OB}$ 的坐标：
			\[
				\vec{OA} \times \vec{OB} =
				\begin{vmatrix}
					x_1 & y_1 \\
					x_2 & y_2
			\end{vmatrix}
				= x_1 y_2 - x_2 y_1,
			\]
			因此
			\[
				|\vec{OA} \times \vec{OB}| = |x_1 y_2 - x_2 y_1|.
			\]
			\par\textbf{理由：}
			平面向量叉积（有向面积）等于对应二阶行列式的值；取绝对值即得平行四边形面积。

		\item[\textbf{步骤三（公式代入）：}]
			将叉积模代入面积公式：
			\[
				S = \frac12 |x_1 y_2 - x_2 y_1|.
			\]
			\par\textbf{理由：}
			由步骤一和步骤二直接推导得出。

		\item[\textbf{步骤四（共线情况）：}]
			若 $O, A, B$ 三点共线，则向量 $\vec{OA}$ 与 $\vec{OB}$ 平行，满足 $x_1 y_2 - x_2 y_1 = 0$，故 $S = 0$。
			\par\textbf{理由：}
			共线时存在常数 $\lambda$ 使 $(x_2,y_2)=\lambda(x_1,y_1)$，代入即得行列式为零。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{综上，以原点为顶点的三角形面积公式为 $S = \frac12 |x_1 y_2 - x_2 y_1|$，当三点共线时面积为 $0$。该公式是向量叉积在几何中的直接应用，是解析几何中求三角形面积的基本工具。}
\end{proofbox}
