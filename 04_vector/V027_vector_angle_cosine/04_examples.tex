\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $|\vec{a}|=2$，$|\vec{b}|=3$，$\vec{a}\cdot\vec{b}=3$，求 $\vec{a}$ 与 $\vec{b}$ 夹角 $\theta$ 的余弦值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（整理已知量）：}] 明确已知模长和数量积。
			\[
				|\vec{a}|=2,\quad |\vec{b}|=3,\quad \vec{a}\cdot\vec{b}=3
			\]
			\par\textbf{理由：}
			余弦公式需要 $|\vec{a}|,|\vec{b}|,\vec{a}\cdot\vec{b}$ 三个量。

		\item[\textbf{第二步（套用公式）：}] 代入夹角余弦公式。
			\[
				\begin{aligned}
					\cos\theta &= \frac{\vec{a}\cdot\vec{b}}{|\vec{a}||\vec{b}|} \\
					&= \frac{3}{2\times3}
			\end{aligned}
			\]
			\par\textbf{理由：}
			直接使用结论 $\cos\theta = \frac{\vec{a}\cdot\vec{b}}{|\vec{a}||\vec{b}|}$。

		\item[\textbf{第三步（数值计算）：}] 化简分数。
			\[
				\cos\theta = \frac{3}{6} = \frac12
			\]
			\par\textbf{理由：}
			算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\cos\theta = \dfrac12$，即 $\theta = 60^\circ$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\vec{a}=(1,0)$，$\vec{b}=(1,\sqrt{3})$，求 $\vec{a}$ 与 $\vec{b}$ 的夹角余弦值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（坐标运算）：}] 计算模长和数量积。
			\[
				|\vec{a}|=1,\quad |\vec{b}|=\sqrt{1^2+(\sqrt{3})^2}=2,\quad \vec{a}\cdot\vec{b}=1\times1+0\times\sqrt{3}=1
			\]
			\par\textbf{理由：}
			坐标形式下数量积等于 $x_1x_2+y_1y_2$，模长等于 $\sqrt{x^2+y^2}$。

		\item[\textbf{第二步（公式代入）：}] 代入夹角余弦公式。
			\[
				\begin{aligned}
					\cos\theta &= \frac{\vec{a}\cdot\vec{b}}{|\vec{a}||\vec{b}|} \\
					&= \frac{1}{1\times2}
			\end{aligned}
			\]
			\par\textbf{理由：}
			直接使用公式 $\cos\theta = \frac{\vec{a}\cdot\vec{b}}{|\vec{a}||\vec{b}|}$。

		\item[\textbf{第三步（化简结果）：}] 求得余弦值。
			\[
				\cos\theta = \frac12
			\]
			\par\textbf{理由：}
			分数化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\cos\theta = \dfrac12$，即 $\theta = 60^\circ$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\vec{a}=(2,1)$，$\vec{b}=(1,m)$，且 $\vec{a}$ 与 $\vec{b}$ 的夹角为锐角，求 $m$ 的取值范围。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（锐角条件）：}] 锐角等价于 $\cos\theta>0$，即 $\vec{a}\cdot\vec{b}>0$。
			\[
				\vec{a}\cdot\vec{b} = 2\times1+1\times m = 2+m
			\]
			由 $\vec{a}\cdot\vec{b}>0$ 得 $2+m>0$，解得 $m>-2$。
			\par\textbf{理由：}
			因为 $|\vec{a}|>0,|\vec{b}|>0$，$\cos\theta$ 的符号仅由 $\vec{a}\cdot\vec{b}$ 决定。

		\item[\textbf{第二步（排除同向）：}] 当 $\vec{a}$ 与 $\vec{b}$ 同向时，夹角为 $0^\circ$，不是锐角，需排除。
			\[
				\frac{2}{1} = \frac{1}{m} \Rightarrow m = \frac12
			\]
			\par\textbf{理由：}
			同向时对应坐标成比例，$m=\frac12$ 时 $\vec{a}=2\vec{b}$，方向相同。

		\item[\textbf{第三步（综合范围）：}] 合并条件，$m > -2$ 且 $m \neq \frac12$。
			\[
				m \in (-2,\frac12) \cup (\frac12,+\infty)
			\]
			\par\textbf{理由：}
			锐角要求数量积为正且两向量不共线同向。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m \in (-2,\frac12) \cup (\frac12,+\infty)$。}
\end{examplebox}
