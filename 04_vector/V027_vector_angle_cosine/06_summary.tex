\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{向量夹角余弦等于数量积除以模长之积，即 $\cos\theta = \frac{\vec{a}\cdot\vec{b}}{|\vec{a}||\vec{b}|}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（非零向量）：} 两向量 $\vec{a},\vec{b}$ 均不为零向量，否则夹角无定义。
		\item \textbf{条件2（范围定义）：} 夹角 $\theta$ 范围是 $[0,\pi]$，$\cos\theta$ 在此区间内单调递减。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略非零）：} 应用公式前务必检查向量是否非零，避免分母为零。
		\item \textbf{检查项（符号对应）：} $\cos\theta$ 的正负反映夹角类型，要结合具体范围判断。
	\end{itemize}
\end{summarybox}
