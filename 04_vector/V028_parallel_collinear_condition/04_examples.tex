\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\vec{a}=(3,2)$，$\vec{b}=(-6,-4)$，判断 $\vec{a}$ 与 $\vec{b}$ 是否平行。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列出坐标）：}] $\vec{a}=(3,2)$，$\vec{b}=(-6,-4)$。
			\par\textbf{理由：} 直接写出已知坐标。
		\item[\textbf{第二步（计算交叉积）：}] 计算 $x_1y_2 = 3\times(-4) = -12$，$x_2y_1 = (-6)\times2 = -12$。
			\par\textbf{理由：} 代入公式 $x_1y_2 = x_2y_1$。
		\item[\textbf{第三步（得出结论）：}] 因为 $x_1y_2 = x_2y_1$，所以 $\vec{a}\parallel\vec{b}$。
			\par\textbf{理由：} 向量平行坐标判定定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\vec{a}\parallel\vec{b}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	判断三点 $A(1,2)$，$B(3,4)$，$C(5,6)$ 是否共线。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造向量）：}] 取 $\overrightarrow{AB}=(3-1,4-2)=(2,2)$，$\overrightarrow{AC}=(5-1,6-2)=(4,4)$。
			\par\textbf{理由：} 向量坐标由终点减起点。
		\item[\textbf{第二步（计算交叉积）：}] $x_1y_2 = 2\times4 = 8$，$x_2y_1 = 2\times4 = 8$，相等。
			\par\textbf{理由：} 代入平行公式。
		\item[\textbf{第三步（得出结论）：}] 由 $\overrightarrow{AB}\parallel\overrightarrow{AC}$ 且起点相同，得 $A,B,C$ 三点共线。
			\par\textbf{理由：} 三点共线等价于同起点两向量平行。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$A,B,C$ 三点共线。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知四边形 $ABCD$ 顶点 $A(0,0)$，$B(2,3)$，$C(4,0)$，$D(-1,2)$。判断 $AB$ 是否平行于 $CD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求向量坐标）：}] $\overrightarrow{AB}=(2-0,3-0)=(2,3)$，$\overrightarrow{CD}=(-1-4,2-0)=(-5,2)$。
			\par\textbf{理由：} 终点减起点。
		\item[\textbf{第二步（计算交叉积）：}] 计算 $x_1y_2 = 2\times2 = 4$，$x_2y_1 = (-5)\times3 = -15$。两者不相等。
			\par\textbf{理由：} 直接代入 $x_1y_2$ 与 $x_2y_1$。
		\item[\textbf{第三步（得出结论）：}] 因为 $x_1y_2 \neq x_2y_1$，故 $\overrightarrow{AB}\not\parallel\overrightarrow{CD}$，即 $AB\not\parallel CD$。
			\par\textbf{理由：} 平行判定条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AB$ 与 $CD$ 不平行。}
\end{examplebox}
