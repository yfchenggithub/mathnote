\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{非零向量平行 $\iff$ 坐标交叉乘积相等 $x_1y_2=x_2y_1$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（非零向量）：} 两向量均为非零向量。
		\item \textbf{条件 2（坐标已知）：} 向量坐标在平面直角坐标系中给出。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（非零前提）：} 使用前确认向量非零。
		\item \textbf{检查项（交叉顺序）：} 计算时注意 $x_1y_2$ 和 $x_2y_1$，不要颠倒。
	\end{itemize}
\end{summarybox}
