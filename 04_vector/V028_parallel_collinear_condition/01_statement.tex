\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（非零向量）：}}] 向量 $\vec{a}=(x_1,y_1)$，$\vec{b}=(x_2,y_2)$ 均为非零向量。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（坐标判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两向量平行等价于对应坐标交叉乘积相等。
			\[
				\vec{a}\parallel\vec{b} \iff x_1y_2 = x_2y_1.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（三点共线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三点共线等价于同起点向量坐标交叉积相等。
			\par
			设 $\overrightarrow{AB}=(x_1,y_1)$，$\overrightarrow{AC}=(x_2,y_2)$ 均为非零向量，则 $A,B,C$ 三点共线当且仅当
			\[
				x_1y_2 = x_2y_1.
			\]
	\end{description}
\end{statementbox}
