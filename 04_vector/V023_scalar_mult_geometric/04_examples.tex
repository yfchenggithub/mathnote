\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $|\vec{a}|=5$，$\lambda=-2$。求 $|\lambda\vec{a}|$，并判断 $\lambda\vec{a}$ 与 $\vec{a}$ 的方向关系。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件代入）：}]
			写出模长公式，代入已知量。
			\[
				|\lambda\vec{a}| = |\lambda|\,|\vec{a}|,\quad |\vec{a}|=5,\ \lambda=-2
			\]
			\par\textbf{理由：} 该公式是向量数乘模长关系的直接结论。
		\item[\textbf{第二步（模长计算）：}]
			计算 $|\lambda\vec{a}|$ 的具体数值。
			\[
				\begin{aligned}
					|\lambda\vec{a}| &= |-2|\times 5 \\ &= 2\times 5 \\ &= 10
			\end{aligned}
			\]
			\par\textbf{理由：} 代入 $|\lambda|=2$。
		\item[\textbf{第三步（方向结论）：}]
			根据 $\lambda=-2<0$，得到方向关系并表述。
			\[
				\lambda\vec{a} = -2\vec{a},\quad |\lambda\vec{a}|=10
			\]
			\par\textbf{理由：} $\lambda<0$ 时 $\lambda\vec{a}$ 与 $\vec{a}$ 方向相反，且可用 $-|\lambda|\vec{a}$ 表示。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$|\lambda\vec{a}|=10$，$\lambda\vec{a}$ 与 $\vec{a}$ 反向。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $|\vec{a}|=4$，$\vec{b}$ 与 $\vec{a}$ 平行且方向相反，$|\vec{b}|=12$。若 $\vec{b}=\lambda\vec{a}$，求 $\lambda$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设定表示）：}]
			由共线关系，设 $\vec{b}=\lambda\vec{a}$。
			\[
				\vec{b} = \lambda\vec{a},\quad \lambda\in\mathbb{R}
			\]
			\par\textbf{理由：} 若两个非零向量共线，则其中一个可表示为另一个的数乘。
		\item[\textbf{第二步（模长关系）：}]
			利用模长公式建立方程。
			\[
				|\vec{b}| = |\lambda|\,|\vec{a}| \Longrightarrow 12 = |\lambda|\times 4
			\]
			\par\textbf{理由：} 数乘模长公式 $|\lambda\vec{a}| = |\lambda|\,|\vec{a}|$。
		\item[\textbf{第三步（解出 $\lambda$）：}]
			求解 $|\lambda|$ 并根据方向定符号。
			\[
				\begin{aligned}
					|\lambda| &= \frac{12}{4} \\ &= 3
			\end{aligned}
			\]
			\par\textbf{理由：} 因为 $\vec{b}$ 与 $\vec{a}$ 方向相反，所以 $\lambda<0$，故 $\lambda=-3$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\lambda=-3$，$\vec{b}$ 与 $\vec{a}$ 反向。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知向量 $\vec{a}=(3,-4)$，实数 $\lambda$ 满足 $|\lambda\vec{a}|=15$。求 $\lambda$ 的值，并判断 $\lambda\vec{a}$ 与 $\vec{a}$ 的方向关系。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求原模长）：}]
			计算 $\vec{a}$ 的模。
			\[
				\begin{aligned}
					|\vec{a}| &= \sqrt{3^2+(-4)^2} \\ &= \sqrt{9+16} \\ &= 5
			\end{aligned}
			\]
			\par\textbf{理由：} 坐标表示下模长公式。
		\item[\textbf{第二步（列模长方程）：}]
			利用 $|\lambda\vec{a}| = |\lambda||\vec{a}|$ 列方程并解 $|\lambda|$。
			\[
				15 = |\lambda|\times 5 \ \Longrightarrow\ |\lambda|=3
			\]
			\par\textbf{理由：} 代入已知 $|\lambda\vec{a}|=15$ 和 $|\vec{a}|=5$。
		\item[\textbf{第三步（求 $\lambda$ 和方向）：}]
			由 $|\lambda|=3$ 得 $\lambda=\pm3$，确定方向关系。
			\[
				\lambda\vec{a}=
				\begin{cases}
					(9,-12), & \lambda=3,\
					\[
						2pt] (-9,12), & \lambda=-3.
				\end{cases}
				\]
				\par\textbf{理由:} $\lambda=3$ 时 $\lambda\vec{a}=3(3,-4)=(9,-12)$,与 $\vec{a}$ 同向;$\lambda=-3$ 时 $\lambda\vec{a}=-3(3,-4)=(-9,12)$,与 $\vec{a}$ 反向.
		\end{description}
		\textbf{\textcolor{CExample}{关键结论:}}
		$\lambda=3$ 或 $\lambda=-3$;$\lambda>0$ 时同向,$\lambda<0$ 时反向.
	\end{examplebox}
