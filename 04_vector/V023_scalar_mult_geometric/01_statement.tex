\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（非零向量）：}}]
			$\vec{a}\neq\vec{0}$，$\lambda\in\mathbb{R}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（数乘平行）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			实数与向量相乘所得向量与原向量始终平行（共线）。
			\[
				\lambda\vec{a} \parallel \vec{a}
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（模长公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数乘后的模长等于原模长乘以 $|\lambda|$。
			\[
				|\lambda\vec{a}| = |\lambda|\,|\vec{a}|
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（方向规律）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$\lambda$ 的正负决定 $\lambda\vec{a}$ 与 $\vec{a}$ 的方向相同或相反；$\lambda=0$ 得零向量。
			\[
				\lambda\vec{a} =
				\begin{cases}
					|\lambda|\,\vec{a}, & \lambda>0,\
					\[
						2pt] -|\lambda|\,\vec{a}, & \lambda<0,\
						\[
							2pt] \vec{0}, & \lambda=0.
					\end{cases}
					\]
				\item[\textbf{\textcolor{CStatement}{推广结论(共线定理):}}]
					\textbf{\textcolor{CStatement}{直观描述:}}
					两个非零向量若共线,则其中一个可由另一个数乘唯一表示.
					\[
						\vec{b}\parallel\vec{a},\ \vec{a}\neq\vec{0}\ \Longrightarrow\ \exists!\,\lambda\in\mathbb{R},\ \vec{b}=\lambda\vec{a}
					\]
			\end{description}

		\end{statementbox}
