\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（非零向量）：}}]
		      $\vec{a}\neq\vec{0}$，$\lambda\in\mathbb{R}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（数乘共线）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      实数与向量相乘所得向量与原向量始终共线；按高中向量的常用约定，也可表述为平行。
		      \[
			      \lambda\vec{a} \parallel \vec{a}
		      \]
		\item[\textbf{\textcolor{CStatement}{结论二（模长公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      数乘后的模长等于原模长乘以 $|\lambda|$。
		      \[
			      |\lambda\vec{a}| = |\lambda|\,|\vec{a}|
		      \]
		\item[\textbf{\textcolor{CStatement}{结论三（方向规律）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      $\lambda$ 的正负决定 $\lambda\vec{a}$ 与 $\vec{a}$ 的方向相同或相反；$\lambda=0$ 时得到零向量，不再讨论方向。
		      \[
			      \lambda\vec{a} =
			      \begin{cases}
				      |\lambda|\,\vec{a},  & \lambda>0, \\[2pt]
				      -|\lambda|\,\vec{a}, & \lambda<0, \\[2pt]
				      \vec{0},             & \lambda=0.
			      \end{cases}
		      \]
		\item[\textbf{\textcolor{CStatement}{推广结论（共线定理）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      若 $\vec{a}\neq\vec{0}$，且 $\vec{b}$ 与 $\vec{a}$ 共线，则 $\vec{b}$ 可由 $\vec{a}$ 的数乘唯一表示。
		      \[
			      \vec{b}\parallel\vec{a},\ \vec{a}\neq\vec{0}
			      \Longleftrightarrow
			      \exists!\,\lambda\in\mathbb{R},\ \vec{b}=\lambda\vec{a}
		      \]
	\end{description}
\end{statementbox}
