\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（公共起点）：}}] 两个向量$\vec{a}=\overrightarrow{OA}$，$\vec{b}=\overrightarrow{OB}$均以点$O$为起点。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（三角形法则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将$\vec{b}$平移使其起点与$\vec{a}$的终点重合，则$\vec{a}+\vec{b}$对应从$\vec{a}$起点指向$\vec{b}$终点的有向线段。
			\[
				\overrightarrow{OA}+\overrightarrow{AB}=\overrightarrow{OB},
			\]
			其中 $\overrightarrow{AB}=\vec{b}$。

		\item[\textbf{\textcolor{CStatement}{结论二（平行四边形法则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$\vec{a}$与$\vec{b}$不共线时，以$\overrightarrow{OA}$，$\overrightarrow{OB}$为邻边作平行四边形$OACB$，则从$O$指向$C$的对角线即为和向量。
			\[
				\overrightarrow{OA}+\overrightarrow{OB}=\overrightarrow{OC}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（交换律）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			向量加法满足交换律，即$\vec{a}+\vec{b}=\vec{b}+\vec{a}$，可由平行四边形对角线相等直接得出。
			\[
				\vec{a}+\vec{b}=\vec{b}+\vec{a}.
			\]
	\end{description}
\end{statementbox}
