\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{向量加法可通过三角形法则或平行四边形法则实现，核心是平移合成。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（起点关系）：} 两向量可平移到同起点（平行四边形法则）或首尾相接（三角形法则）。
		\item \textbf{条件 2（共线限制）：} 平行四边形法则要求两向量不共线，共线时只能用三角形法则。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（法则选择）：} 根据向量起点关系选择合适法则，注意区分和向量与差向量的方向。
		\item \textbf{检查项（图形闭合）：} 多个向量首尾相加时，检查最终指向是否回到起点（是否封闭）。
	\end{itemize}
\end{summarybox}
