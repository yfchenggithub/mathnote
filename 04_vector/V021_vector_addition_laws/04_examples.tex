\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知向量$\vec{a}$与$\vec{b}$起点均为$O$，$|\vec{a}|=3$，$|\vec{b}|=4$，且$\vec{a}\perp\vec{b}$，求$|\vec{a}+\vec{b}|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别法则）：}] 因为$\vec{a}$与$\vec{b}$起点相同且不共线，用平行四边形法则：以$\vec{a}$、$\vec{b}$为邻边作平行四边形$OACB$，则$\vec{a}+\vec{b}=\overrightarrow{OC}$。
			\par\textbf{理由：}
			平行四边形法则：当两向量不共线时，和向量为从起点出发的对角线。
		\item[\textbf{第二步（利用几何性质）：}] 由$\vec{a}\perp\vec{b}$，平行四边形为矩形，对角线长度可由勾股定理计算。
			\par\textbf{理由：}
			矩形对角线相等且等于邻边平方和的平方根。
		\item[\textbf{第三步（计算模长）：}] $OC=\sqrt{OA^2+AC^2}=\sqrt{3^2+4^2}=5$，故$|\vec{a}+\vec{b}|=5$。
			\par\textbf{理由：}
			代入数值计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|\vec{a}+\vec{b}|=5$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=2$，$BC=3$，$\angle ABC=60^\circ$，求$|\overrightarrow{AB}+\overrightarrow{BC}|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别加法）：}] $\overrightarrow{AB}$与$\overrightarrow{BC}$首尾相接，由三角形法则得$\overrightarrow{AB}+\overrightarrow{BC}=\overrightarrow{AC}$。
			\par\textbf{理由：}
			三角形法则：将$\overrightarrow{BC}$平移至与$\overrightarrow{AB}$首尾相接，和向量为$\overrightarrow{AC}$。
		\item[\textbf{第二步（转化问题）：}] 因此$|\overrightarrow{AB}+\overrightarrow{BC}|=|\overrightarrow{AC}|$，即线段$AC$的长度。
			\par\textbf{理由：}
			模长定义。
		\item[\textbf{第三步（余弦定理）：}] 在$\triangle ABC$中，已知$AB=2$，$BC=3$，夹角$\angle ABC=60^\circ$，由余弦定理：
			\[
				AC^2 = AB^2 + BC^2 - 2\cdot AB\cdot BC\cos\angle ABC.
			\]
			\par\textbf{理由：}
			余弦定理适用于任意三角形。
		\item[\textbf{第四步（计算结果）：}] 代入数值：$AC^2=4+9-2\cdot2\cdot3\cdot\frac12=13-6=7$，所以$AC=\sqrt{7}$，即$|\overrightarrow{AB}+\overrightarrow{BC}|=\sqrt{7}$。
			\par\textbf{理由：}
			算术平方根取正值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|\overrightarrow{AB}+\overrightarrow{BC}|=\sqrt{7}$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$M$是$BC$的中点，设$\overrightarrow{AB}=\vec{a}$，$\overrightarrow{AC}=\vec{b}$，用$\vec{a}$、$\vec{b}$表示$\overrightarrow{AM}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造平行四边形）：}] 以$AB$、$AC$为邻边作平行四边形$ABDC$，连接$AD$。由平行四边形法则，$\overrightarrow{AD}=\vec{a}+\vec{b}$。
			\par\textbf{理由：}
			平行四边形法则：同起点两向量的和等于以它们为邻边的平行四边形从起点出发的对角线。
		\item[\textbf{第二步（确定中点关系）：}] 因为四边形$ABDC$是平行四边形，所以对角线$AD$与$BC$互相平分。已知$M$是$BC$中点，故$M$也是$AD$中点。
			\par\textbf{理由：}
			平行四边形对角线互相平分。
		\item[\textbf{第三步（中点向量关系）：}] 因为$M$是$AD$中点，所以$\overrightarrow{AM}=\frac12\overrightarrow{AD}$。
			\par\textbf{理由：}
			若$M$是线段$AD$中点，则$\overrightarrow{AM}=\frac12\overrightarrow{AD}$。
		\item[\textbf{第四步（代入得结果）：}] 将$\overrightarrow{AD}=\vec{a}+\vec{b}$代入，得$\overrightarrow{AM}=\frac12(\vec{a}+\vec{b})$。
			\par\textbf{理由：}
			等量代换。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\overrightarrow{AM}=\frac12(\vec{a}+\vec{b})$。}
\end{examplebox}
