\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（基底选择受限）}}]
			误认为只有正交且单位长度的向量才能作为基底。
		\item[\textbf{\textcolor{CTrap}{正确理解（基底条件）}}]
			任意两个不共线的向量均可作为基底，与垂直、长度无关。
		\item[\textbf{\textcolor{CTrap}{错因分析（直角坐标惯性）}}]
			受直角坐标系思维影响，误将正交与单位长度当作必要前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（系数顺序混淆）}}]
			在分解式中常把基底的先后顺序弄反，导致系数与基底不匹配。
		\item[\textbf{\textcolor{CTrap}{正确理解（顺序对应）}}]
			分解式 $\vec{a}=\lambda_1\vec{e}_1+\lambda_2\vec{e}_2$ 中 $\lambda_1$ 必须对应 $\vec{e}_1$，$\lambda_2$ 必须对应 $\vec{e}_2$，顺序不可颠倒。
		\item[\textbf{\textcolor{CTrap}{错因分析（未按固定顺序）}}]
			书写或思考时未给基底规定顺序，随意调换位置。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（模长计算误区）}}]
			以为 $\vec{a}=(\lambda_1,\lambda_2)$ 的模长等于 $\sqrt{\lambda_1^2+\lambda_2^2}$，忽略基底不一定是正交单位基。
		\item[\textbf{\textcolor{CTrap}{正确理解（模长依赖基底）}}]
			只有基底是单位正交向量时，模长公式才成立；一般基底需通过内积计算。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆一般基底与坐标轴）}}]
			将向量坐标表示下的模长公式推广到所有基底，忽略了正交性。
	\end{description}

\end{trapbox}
