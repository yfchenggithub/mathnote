\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\par\textbf{平面内任意向量均可由一组不共线基底唯一线性表示，这是向量运算与坐标化的基石。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（基底不共线）：} 基底 $\vec{e}_1,\vec{e}_2$ 必须不共线，否则无法张成整个平面。
		\item \textbf{条件2（同平面）：} 所研究向量 $\vec{a}$ 与基底必须在同一平面内。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（唯一分解条件）：} 给定基底后分解系数唯一，但基底选择不唯一。
		\item \textbf{检查项（确认不共线）：} 使用基底前必须验证两向量不共线（系数不成比例）。
	\end{itemize}

\end{summarybox}
