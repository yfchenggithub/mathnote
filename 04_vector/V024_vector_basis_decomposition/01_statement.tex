\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（基底条件）：}}]
			平面内两个向量 $\vec{e}_1$ 与 $\vec{e}_2$ 不共线。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（唯一分解）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平面任意向量都可沿两个不共线方向唯一拆分。
			\[
				\vec{a} = \lambda_1 \vec{e}_1 + \lambda_2 \vec{e}_2, \quad \lambda_1,\lambda_2 \in \mathbb{R}
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（坐标表示）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当基底取单位正交向量时，系数就是直角坐标。
			\[
				\vec{a} = (x,y) \iff \vec{a} = x\vec{i} + y\vec{j},
			\]
			其中 $\vec{i},\vec{j}$ 为单位正交基底。
	\end{description}

\end{statementbox}
