\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用平行四边形对角线向量表达式，将问题转化为向量模平方的计算，通过展开后相加消去内积项。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（向量表示）：}] 设 $\vec{a}=\overrightarrow{AB}$，$\vec{b}=\overrightarrow{AD}$，则 $\overrightarrow{AC}=\vec{a}+\vec{b}$，$\overrightarrow{BD}=\vec{b}-\vec{a}$（模相等）。
			\[
				\overrightarrow{AC}=\vec{a}+\vec{b},
				\qquad
				\overrightarrow{BD}=\vec{b}-\vec{a}
			\]
			\par\textbf{理由：} 平行四边形加法法则，对角线向量等于邻边向量的和或差。

		\item[\textbf{步骤二（展开和向量模）：}] 计算 $|\vec{a}+\vec{b}|^2$。
			\[
				|\vec{a}+\vec{b}|^2 = (\vec{a}+\vec{b})\cdot(\vec{a}+\vec{b}) = |\vec{a}|^2+|\vec{b}|^2+2\vec{a}\cdot\vec{b}
			\]
			\par\textbf{理由：} 向量模平方等于自身数量积，利用数量积分配律展开。

		\item[\textbf{步骤三（展开差向量模）：}] 计算 $|\vec{a}-\vec{b}|^2$。
			\[
				|\vec{a}-\vec{b}|^2 = |\vec{a}|^2+|\vec{b}|^2-2\vec{a}\cdot\vec{b}
			\]
			\par\textbf{理由：} 同理，展开后中间项变号。

		\item[\textbf{步骤四（相加消元）：}] 将两式相加。
			\[
				|\vec{a}+\vec{b}|^2+|\vec{a}-\vec{b}|^2 = 2(|\vec{a}|^2+|\vec{b}|^2)
			\]
			\par\textbf{理由：} 交叉项 $2\vec{a}\cdot\vec{b}$ 与 $-2\vec{a}\cdot\vec{b}$ 抵消。

		\item[\textbf{步骤五（边长转换）：}] 记 $a=|\vec{a}|$，$b=|\vec{b}|$，则 $AC=|\vec{a}+\vec{b}|$，$BD=|\vec{a}-\vec{b}|$，且 $AB=a,\ AD=b$，由对边相等得 $BC=a,\ CD=b$。
			\[
				\begin{aligned}
					AC^2+BD^2 &= 2(|\vec{a}|^2+|\vec{b}|^2) \\
					&= 2(AB^2+AD^2) \\
					&= AB^2+BC^2+CD^2+DA^2
			\end{aligned}
			\]
			\par\textbf{理由：} 对边长度相等，代入恒等式得证。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	因此，平行四边形对角线平方和恒等于四边平方和，也等于两邻边平方和的二倍。
\end{proofbox}
