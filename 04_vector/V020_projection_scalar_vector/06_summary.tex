\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	只要 $\vec{b}\ne\vec{0}$，$\vec{a}$ 在 $\vec{b}$ 方向上的投影数量为
	\[
		\frac{\vec{a}\cdot\vec{b}}{|\vec{b}|},
	\]
	投影向量为
	\[
		\frac{\vec{a}\cdot\vec{b}}{|\vec{b}|^2}\vec{b}.
	\]
	投影数量是带符号的标量，投影向量是向量，二者不能混淆。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（投影方向非零）：} 投影到 $\vec{b}$ 方向时，必须有 $\vec{b}\ne\vec{0}$，否则分母无意义。
		\item \textbf{条件2（夹角表述）：} 若使用 $\theta=\angle(\vec{a},\vec{b})$，则还需 $\vec{a}\ne\vec{0}$，并规定 $\theta\in[0,\pi]$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式区分）：} 投影数量分母是 $|\vec{b}|$，投影向量分母是 $|\vec{b}|^2$，因为投影数量还要再乘以单位向量 $\dfrac{\vec{b}}{|\vec{b}|}$。
		\item \textbf{检查项（符号判断）：} 计算后注意 $\vec{a}\cdot\vec{b}$ 的符号。点积为正时，投影向量与 $\vec{b}$ 同向；点积为负时，投影向量与 $\vec{b}$ 反向；点积为 $0$ 时，投影向量为 $\vec{0}$，没有确定方向。
		\item \textbf{概念项（数量与模长）：} 投影数量可正可负，而投影向量的模长一定非负：
		      \[
			      \bigl|\text{投影向量}\bigr|
			      =
			      \frac{|\vec{a}\cdot\vec{b}|}{|\vec{b}|}.
		      \]
	\end{itemize}
\end{summarybox}
