\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知向量 $\vec{a}=(1,2)$，$\vec{b}=(4,-3)$，求 $\vec{a}$ 在 $\vec{b}$ 方向上的投影数量。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算点积）：}]
		      \[
			      \vec{a}\cdot\vec{b}=1\times4+2\times(-3)=4-6=-2.
		      \]
		      \par\textbf{理由：} 向量点积的坐标运算法则。

		\item[\textbf{第二步（求模长）：}]
		      \[
			      |\vec{b}|=\sqrt{4^2+(-3)^2}=\sqrt{16+9}=5.
		      \]
		      \par\textbf{理由：} 向量模长的坐标公式。

		\item[\textbf{第三步（套用公式）：}]
		      \[
			      \text{投影数量}
			      =
			      \frac{\vec{a}\cdot\vec{b}}{|\vec{b}|}
			      =
			      \frac{-2}{5}
			      =
			      -\frac25.
		      \]
		      \par\textbf{理由：} 投影数量公式为 $\dfrac{\vec{a}\cdot\vec{b}}{|\vec{b}|}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 投影数量为 $-\dfrac25$。负号表示 $\vec{a}$ 在 $\vec{b}$ 方向上的有向长度为负，即对应的投影向量与 $\vec{b}$ 反向。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $|\vec{a}|=3$，$|\vec{b}|=4$，$\vec{a}\cdot\vec{b}=6$，求 $\vec{a}$ 在 $\vec{b}$ 方向上的投影向量。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求投影数量）：}]
		      \[
			      \text{投影数量}
			      =
			      \frac{\vec{a}\cdot\vec{b}}{|\vec{b}|}
			      =
			      \frac64
			      =
			      \frac32.
		      \]
		      \par\textbf{理由：} 投影数量公式为 $\dfrac{\vec{a}\cdot\vec{b}}{|\vec{b}|}$。

		\item[\textbf{第二步（求投影向量系数）：}]
		      \[
			      \lambda
			      =
			      \frac{\vec{a}\cdot\vec{b}}{|\vec{b}|^2}
			      =
			      \frac{6}{4^2}
			      =
			      \frac{6}{16}
			      =
			      \frac38.
		      \]
		      \par\textbf{理由：} 投影向量为 $\dfrac{\vec{a}\cdot\vec{b}}{|\vec{b}|^2}\vec{b}$。

		\item[\textbf{第三步（写投影向量）：}]
		      \[
			      \text{投影向量}
			      =
			      \lambda\vec{b}
			      =
			      \frac38\vec{b}.
		      \]
		      \par\textbf{理由：} 投影向量是 $\vec{b}$ 的实数倍。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 投影向量为 $\dfrac38\vec{b}$。因为 $\vec{a}\cdot\vec{b}=6>0$，所以投影向量与 $\vec{b}$ 同向。

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 在三角形 $ABC$ 中，$AB=AC=1$，$\angle A=120^\circ$，求 $\overrightarrow{AB}$ 在 $\overrightarrow{AC}$ 方向上的投影向量。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算点积）：}]
		      \[
			      \overrightarrow{AB}\cdot\overrightarrow{AC}
			      =
			      |AB||AC|\cos120^\circ
			      =
			      1\times1\times\left(-\frac12\right)
			      =
			      -\frac12.
		      \]
		      \par\textbf{理由：} 点积的几何定义。

		\item[\textbf{第二步（求投影向量系数）：}]
		      \[
			      \lambda
			      =
			      \frac{\overrightarrow{AB}\cdot\overrightarrow{AC}}{|\overrightarrow{AC}|^2}
			      =
			      \frac{-\frac12}{1^2}
			      =
			      -\frac12.
		      \]
		      \par\textbf{理由：} 投影向量公式为 $\dfrac{\vec{a}\cdot\vec{b}}{|\vec{b}|^2}\vec{b}$。

		\item[\textbf{第三步（写投影向量）：}]
		      \[
			      \text{投影向量}
			      =
			      \lambda\overrightarrow{AC}
			      =
			      -\frac12\overrightarrow{AC}.
		      \]
		      \par\textbf{理由：} 投影向量与 $\overrightarrow{AC}$ 共线，系数为 $-\dfrac12$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 投影向量为 $-\dfrac12\overrightarrow{AC}$。因为点积为负，所以方向与 $\overrightarrow{AC}$ 相反。
