\[
	\text{投影向量}
\]

\[
	\vec a \text{ 在 } \vec b \text{ 方向的投影，公式：}
\]

\[
	|\vec a|\cos\theta
	=
	|\vec a|\frac{\vec a\cdot \vec b}{|\vec a|\cdot|\vec b|}
	=
	\frac{\vec a\cdot \vec b}{|\vec b|},
	\quad
	\theta \text{ 为 } \vec a \text{ 与 } \vec b \text{ 的夹角}
\]

\[
	\text{（注意：投影是只有大小，没有方向）}
\]

\[
	\vec a \text{ 在 } \vec b \text{ 方向的投影向量，公式：}
\]

\[
	|\vec a|(\cos\theta)\frac{\vec b}{|\vec b|},
	\quad
	\frac{\vec b}{|\vec b|}
	\text{ 为向量 } \vec b \text{ 的单位向量}
\]

\[
	\text{（注意：投影向量有大小，也有方向）}
\]
