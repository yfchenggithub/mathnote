\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（概念混淆）：}}] 认为投影数量是向量，或投影向量是标量，将两者混为一谈。
		\item[\textbf{\textcolor{CTrap}{正确理解（概念区分）：}}] 投影数量是标量，可正、可负、也可为 $0$；投影向量是向量，与 $\vec{b}$ 共线，两者通过“投影数量乘以 $\vec{b}$ 方向单位向量”联系。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义不清）：}}] 未区分“有向长度”和“向量”，对投影的定义理解流于表面。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公式错误）：}}] 求投影向量时，误写为 $\dfrac{\vec{a}\cdot\vec{b}}{|\vec{b}|}\vec{b}$，忘记分母应为 $|\vec{b}|^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式记忆）：}}] 投影数量为
		      \[
			      \frac{\vec{a}\cdot\vec{b}}{|\vec{b}|},
		      \]
		      投影向量为
		      \[
			      \frac{\vec{a}\cdot\vec{b}}{|\vec{b}|^2}\vec{b}.
		      \]
		      前者分母是 $|\vec{b}|$，后者分母是 $|\vec{b}|^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导缺失）：}}] 只记住投影数量公式，未将投影数量再乘以单位向量 $\dfrac{\vec{b}}{|\vec{b}|}$，导致分母少乘一个 $|\vec{b}|$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号忘记）：}}] 计算投影数量时只取正值，忽略夹角为钝角时投影数量应为负。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号规律）：}}] 投影数量
		      \[
			      \frac{\vec{a}\cdot\vec{b}}{|\vec{b}|}
		      \]
		      是有向长度，$\vec{a}\cdot\vec{b}$ 的正负直接决定符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（几何直觉不足）：}}] 思维停留在“投影就是长度”的直观上，忽略了投影数量是带符号的标量。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（零向量忽略）：}}] 认为 $\vec{a}$ 和 $\vec{b}$ 都必须非零，或者在 $\vec{a}\cdot\vec{b}=0$ 时仍强行判断方向。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件边界）：}}] 投影到 $\vec{b}$ 方向时，核心条件是 $\vec{b}\ne\vec{0}$。若 $\vec{a}=\vec{0}$，投影数量为 $0$，投影向量为 $\vec{0}$；若 $\vec{a}\cdot\vec{b}=0$，投影向量也是 $\vec{0}$，没有确定方向。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件照搬）：}}] 把“夹角公式需要两个向量非零”和“投影公式只要求投影方向非零”混在一起。
	\end{description}
\end{trapbox}
