\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（方向易反）}}] 误认为$\vec{a}-\vec{b}=\overrightarrow{AB}$，即差向量方向画反。
		\item[\textbf{\textcolor{CTrap}{正确理解（方向判定）：}}] 差向量方向由减向量终点指向被减向量终点，使用“后指前”口诀（减向量为后）。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}] 受加法交换律影响，习惯认为被减数在前、结果方向与前一致。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（起点不同）}}] 对起点不同的两个向量直接相减，忽略平移步骤。
		\item[\textbf{\textcolor{CTrap}{正确理解（共起点前提）：}}] 应用减法三角形法则前，必须先将两向量平移至同一起点。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}] 没有注意向量减法定义建立在共起点基础上。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号运算错）}}] 在向量线性运算中，$-\vec{b}$ 的方向与 $\vec{b}$ 相反，但误写为模长或坐标变号错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（减法转化加法）：}}] $\vec{a}-\vec{b}=\vec{a}+(-\vec{b})$，减法可化为加法处理，注意相反向量的坐标变号。
		\item[\textbf{\textcolor{CTrap}{错因分析（运算混淆）：}}] 混淆向量减法与数乘的符号处理。
	\end{description}
\end{trapbox}
