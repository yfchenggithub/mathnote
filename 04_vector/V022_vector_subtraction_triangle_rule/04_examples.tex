\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知点O(0,0)，A(2,3)，B(5,7)。设$\vec{a}=\overrightarrow{OA}$，$\vec{b}=\overrightarrow{OB}$，求$\vec{a}-\vec{b}$，并指出其起点和终点。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写向量）：}] $\vec{a}=(2,3)$，$\vec{b}=(5,7)$。
			\par\textbf{理由：}
			坐标由点坐标直接写出。
		\item[\textbf{第二步（坐标相减）：}] $\vec{a}-\vec{b}=(2-5,3-7)$，即 $(-3,-4)$。
			\par\textbf{理由：}
			减法三角形法则的坐标形式：对应坐标相减。
		\item[\textbf{第三步（几何对应）：}] 因为$\vec{a}-\vec{b}=\overrightarrow{BA}$，所以起点为B(5,7)，终点为A(2,3)。
			\par\textbf{理由：}
			由法则，差向量等于从减向量终点B指向被减向量终点A的向量。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\vec{a}-\vec{b}=(-3,-4)$}，对应$\overrightarrow{BA}$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知共起点向量$\vec{a},\vec{b}$，且$|\vec{a}|=3$，$|\vec{b}|=4$，$\angle AOB=60^\circ$。求$|\vec{a}-\vec{b}|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定差向量）：}] 由减法法则，$\vec{a}-\vec{b}=\overrightarrow{BA}$，所以$|\vec{a}-\vec{b}|=|\overrightarrow{BA}|$。
			\par\textbf{理由：}
			通过法则将差向量模转化为B、A两点间距离。
		\item[\textbf{第二步（模长公式）：}] $|\vec{a}-\vec{b}|^2=|\vec{a}|^2+|\vec{b}|^2-2|\vec{a}||\vec{b}|\cos\angle AOB$。
			\par\textbf{理由：}
			向量模长平方公式，也称余弦定理向量形式。
		\item[\textbf{第三步（代入计算）：}]
			\[
				\begin{aligned}
					|\vec{a}-\vec{b}|^2 &= 3^2 + 4^2 - 2 \cdot 3 \cdot 4 \cdot \cos60^\circ \\
					&= 9 + 16 - 12 \\
					&= 13.
			\end{aligned}
			\]
			所以 $|\vec{a}-\vec{b}| = \sqrt{13}$。
			\par\textbf{理由：}
			代入已知数值，注意夹角为已知角。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|\vec{a}-\vec{b}|=\sqrt{13}$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在平行四边形ABCD中，设$\overrightarrow{AB}=\vec{a}$，$\overrightarrow{AD}=\vec{b}$。试用$\vec{a},\vec{b}$表示$\overrightarrow{DB}$和$\overrightarrow{BD}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平移共起点）：}] 将$\vec{a}$和$\vec{b}$平移到公共起点A，即$\overrightarrow{AB}=\vec{a}$，$\overrightarrow{AD}=\vec{b}$。
			\par\textbf{理由：}
			根据题目设定，A为共同起点。
		\item[\textbf{第二步（应用减法法则）：}] 差向量$\vec{a}-\vec{b}$等于从D指向B，即$\overrightarrow{DB}=\vec{a}-\vec{b}$。
			\par\textbf{理由：}
			以A为起点，$\vec{a}-\vec{b}$由减向量终点D指向被减向量终点B。
		\item[\textbf{第三步（交换顺序）：}] 同样，$\vec{b}-\vec{a}=\overrightarrow{BD}$。
			\par\textbf{理由：}
			交换被减数与减数，得相反向量，方向由B指向D。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\overrightarrow{DB}=\vec{a}-\vec{b}$，$\overrightarrow{BD}=\vec{b}-\vec{a}$。}
\end{examplebox}
