\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	利用向量加法三角形法则的逆运算，通过移项和相反向量推导。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（加法关系）：}] 对于任意三点 $O,A,B$，由向量加法三角形法则有：
			\[
				\overrightarrow{OA}+\overrightarrow{AB}=\overrightarrow{OB}.
			\]
			\par\textbf{理由：}
			从 $O$ 到 $A$ 再到 $B$，与直接 $O$ 到 $B$ 等价，这是向量加法的定义。

		\item[\textbf{步骤二（移项变形）：}] 将 $\overrightarrow{OA}$ 移至右边：
			\[
				\begin{aligned}
					\overrightarrow{AB} &= \overrightarrow{OB} - \overrightarrow{OA} \\
					&= \vec{b} - \vec{a}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			等式两边同时减去 $\overrightarrow{OA}$，向量减法定义为加法的逆运算。

		\item[\textbf{步骤三（取相反向量）：}] 因为 $\overrightarrow{BA}=-\overrightarrow{AB}$，代入得：
			\[
				\begin{aligned}
					\overrightarrow{BA} &= -(\vec{b} - \vec{a}) \\
					&= \vec{a} - \vec{b}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			相反向量长度相等、方向相反，负号改变方向。

		\item[\textbf{步骤四（公理化表述）：}] 综上，对任意共起点向量 $\vec{a}=\overrightarrow{OA},\vec{b}=\overrightarrow{OB}$，均有
			\[
				\boxed{\vec{a}-\vec{b}=\overrightarrow{BA}}.
			\]
			\par\textbf{理由：}
			该等式即为向量减法的三角形法则。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{所以，差向量 $\vec{a}-\vec{b}$ 恒等于从减向量终点 $B$ 指向被减向量终点 $A$ 的向量 $\overrightarrow{BA}$。}
\end{proofbox}
