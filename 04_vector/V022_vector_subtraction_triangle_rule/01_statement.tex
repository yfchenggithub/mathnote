	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（共起点）：}}] 向量 $\vec{a}$ 与 $\vec{b}$ 起点重合为 $O$，即 $\vec{a}=\overrightarrow{OA}$，$\vec{b}=\overrightarrow{OB}$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（差向量公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			差向量 $\vec{a}-\vec{b}$ 等于从 $B$ 指向 $A$ 的向量。
			\[
				\boxed{\vec{a}-\vec{b}=\overrightarrow{BA}}
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（交换顺序）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			交换被减数与减数得相反向量 $\overrightarrow{AB}$。
			\[
				\boxed{\vec{b}-\vec{a}=\overrightarrow{AB}}
			\]
	\end{description}

\end{statementbox}
