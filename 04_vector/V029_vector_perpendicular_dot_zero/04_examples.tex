\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知向量 $\vec{a}=(2,1)$，$\vec{b}=(-1,2)$，判断 $\vec{a}$ 与 $\vec{b}$ 是否垂直。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出坐标）：}]
			明确 $\vec{a}=(2,1)$，$\vec{b}=(-1,2)$。
			\par\textbf{理由：}垂直判据需要坐标。
		\item[\textbf{第二步（计算点积）：}]
			计算 $2\times(-1)+1\times2=-2+2=0$。
			\par\textbf{理由：}垂直条件为 $x_1x_2+y_1y_2=0$。
		\item[\textbf{第三步（判断垂直）：}]
			点积为0，故 $\vec{a}\perp\vec{b}$。
			\par\textbf{理由：}垂直的充要条件。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\vec{a}\perp\vec{b}$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知向量 $\vec{a}=(3,-1)$，$\vec{b}=(k,6)$，且 $\vec{a}\perp\vec{b}$，求实数 $k$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列方程）：}]
			由 $\vec{a}\perp\vec{b}$ 得 $3\cdot k + (-1)\cdot6 = 0$。
			\par\textbf{理由：}垂直 $\iff$ $x_1x_2+y_1y_2=0$。
		\item[\textbf{第二步（化简）：}]
			$3k-6=0$。
			\par\textbf{理由：}简化方程。
		\item[\textbf{第三步（解出参数）：}]
			解得 $k=2$。
			\par\textbf{理由：}方程求解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$k=2$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	求与向量 $\vec{a}=(1,-2)$ 垂直的一个单位向量 $\vec{e}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设向量列方程）：}]
			设 $\vec{e}=(x,y)$，由垂直得 $1\cdot x + (-2)\cdot y = 0$，即 $x-2y=0$。
			\par\textbf{理由：}垂直条件 $x_1x_2+y_1y_2=0$。
		\item[\textbf{第二步（单位条件）：}]
			由 $|\vec{e}|=1$ 得 $x^2+y^2=1$。
			\par\textbf{理由：}单位向量模为1。
		\item[\textbf{第三步（联立求解）：}]
			将 $x=2y$ 代入 $x^2+y^2=1$ 得 $(2y)^2+y^2=5y^2=1$， 故 $y=\pm\frac{1}{\sqrt5}$，$x=2y=\pm\frac{2}{\sqrt5}$。
			\par\textbf{理由：}方程组求解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\vec{e}=\left(\frac{2}{\sqrt5},\frac{1}{\sqrt5}\right)$ 或 $\left(-\frac{2}{\sqrt5},-\frac{1}{\sqrt5}\right)$}
\end{examplebox}
