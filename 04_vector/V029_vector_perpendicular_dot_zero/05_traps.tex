\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略非零）：}}]
			认为零向量与任何向量垂直，从而在应用 $x_1x_2+y_1y_2=0$ 时不检查向量是否非零。
		\item[\textbf{\textcolor{CTrap}{正确理解（非零前提）：}}]
			垂直定义要求两向量均非零，零向量与任何向量不讨论垂直。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视条件）：}}]
			没有注意结论中“非零向量”的前提限制。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（运算失误）：}}]
			计算 $x_1x_2+y_1y_2$ 时，误作 $x_1x_2$ 或 $y_1y_2$ 单独为零，或错误地使用减法。
		\item[\textbf{\textcolor{CTrap}{正确理解（乘加结构）：}}]
			数量积公式是 $x_1x_2+y_1y_2$，即对应坐标相乘后相加。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆模糊）：}}]
			对数量积坐标形式不熟练，容易与平行条件的差积形式混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（混用公式）：}}]
			用 $x_1y_2-x_2y_1=0$ 作为垂直判定。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分条件）：}}]
			垂直条件为 $x_1x_2+y_1y_2=0$，平行条件为 $x_1y_2-x_2y_1=0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			未区分垂直与平行的坐标判据，死记硬背时容易混淆。
	\end{description}

\end{trapbox}
