\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{当 $\vec b\neq\vec 0$ 时，$\vec a$ 在 $\vec b$ 方向上的投影数量是有向长度，由公式}
	\[
		\frac{\vec a\cdot\vec b}{|\vec b|}
	\]
	\textbf{计算。}

	\medskip
	\textbf{\textcolor{CSummary}{核心公式}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{投影数量：}
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|}.
		      \]

		\item \textbf{夹角形式：} 若 $\vec a\neq\vec 0$，且 $\theta$ 为 $\vec a$ 与 $\vec b$ 的夹角，则
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|}
			      =
			      |\vec a|\cos\theta .
		      \]

		\item \textbf{投影向量：}
		      \[
			      \left(\frac{\vec a\cdot\vec b}{|\vec b|}\right)\frac{\vec b}{|\vec b|}
			      =
			      \frac{\vec a\cdot\vec b}{|\vec b|^2}\vec b.
		      \]
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（分母非零）：} $\vec b$ 必须为非零向量，即
		      \[
			      |\vec b|\neq0.
		      \]
		      否则投影方向不确定，公式无意义。

		\item \textbf{条件2（夹角对应）：} 若使用 $|\vec a|\cos\theta$ 或通过符号判断夹角，则通常还需 $\vec a\neq\vec 0$，并设夹角范围为
		      \[
			      0\le\theta\le\pi.
		      \]

		\item \textbf{特殊情况：} 当 $\vec a=\vec 0$ 且 $\vec b\neq\vec 0$ 时，投影数量为 $0$，投影向量为 $\vec 0$，但不需要讨论 $\vec a$ 与 $\vec b$ 的夹角。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号正负）：} 投影数量符号由 $\cos\theta$ 决定：锐角为正，直角为零，钝角为负。

		\item \textbf{易错点（数量与向量）：} 投影数量是标量，投影向量才是向量。不要把
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|}
		      \]
		      和
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|^2}\vec b
		      \]
		      混为一谈。

		\item \textbf{检查项（分母非零）：} 使用
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|}
		      \]
		      前务必检查 $|\vec b|\neq0$，否则公式无效。
	\end{itemize}

\end{summarybox}
