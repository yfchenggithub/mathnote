\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（投影方向非零）：}}] 向量 $\vec b$ 必须为非零向量：
		      \[
			      \vec b \neq \vec 0.
		      \]
		      因为零向量没有确定方向，所以不能作为投影方向。

		\item[\textbf{\textcolor{CStatement}{条件2（夹角公式前提）：}}] 若使用 $\lvert\vec a\rvert\cos\theta$ 这一形式，则通常还需 $\vec a\neq \vec 0$，此时设 $\vec a$ 与 $\vec b$ 的夹角为 $\theta$，其中
		      \[
			      0 \le \theta \le \pi.
		      \]
		      特别地，当 $\vec a=\vec 0$ 且 $\vec b\neq\vec 0$ 时，$\vec a$ 在 $\vec b$ 方向上的投影数量为 $0$，投影向量为 $\vec 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（投影数量）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} $\vec a$ 在 $\vec b$ 方向上的投影数量是带有符号的标量，表示 $\vec a$ 沿 $\vec b$ 方向的有向长度。
		      \[
			      \frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}.
		      \]
		      若 $\vec a\neq\vec 0$，且 $\theta$ 是 $\vec a$ 与 $\vec b$ 的夹角，则
		      \[
			      \frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}
			      =
			      \lvert\vec a\rvert\cos\theta .
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（投影向量）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 投影向量是投影数量乘以 $\vec b$ 方向的单位向量。
		      \[
			      \left(\frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}\right)\frac{\vec b}{\lvert\vec b\rvert}
			      =
			      \frac{\vec a\cdot\vec b}{\lvert\vec b\rvert^2}\,\vec b.
		      \]
	\end{description}

\end{statementbox}
