\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略方向）}}] 认为投影数量永远非负，忽略投影数量是有符号标量。

		\item[\textbf{\textcolor{CTrap}{正确理解（符号判定）：}}] 当 $\vec a\neq\vec 0,\ \vec b\neq\vec 0$ 时，投影数量的正负由 $\cos\theta$ 决定：
		      \[
			      0\le\theta<90^\circ
			      \quad\Rightarrow\quad
			      \text{投影数量为正};
		      \]
		      \[
			      \theta=90^\circ
			      \quad\Rightarrow\quad
			      \text{投影数量为零};
		      \]
		      \[
			      90^\circ<\theta\le180^\circ
			      \quad\Rightarrow\quad
			      \text{投影数量为负}.
		      \]

		\item[\textbf{\textcolor{CTrap}{错因分析（向量投影本质）：}}] 混淆了线段长度与有向投影。线段长度一定非负，而投影数量可以为负。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分母非零）}}] 试图计算 $\vec b=\vec 0$ 时的投影数量，带入公式导致分母为零。

		\item[\textbf{\textcolor{CTrap}{正确理解（条件限制）：}}] 投影数量公式要求
		      \[
			      \vec b\neq\vec 0.
		      \]
		      零向量方向不确定，不能作为投影方向，因此投影无意义。

		\item[\textbf{\textcolor{CTrap}{错因分析（公式前提遗忘）：}}] 只记公式形式，忽略分母不能为零以及投影方向必须确定的前提。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（概念混淆）}}] 将投影数量与投影向量混为一谈，认为投影数量也是向量。

		\item[\textbf{\textcolor{CTrap}{正确理解（区分标量向量）：}}] 投影数量是标量：
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|}.
		      \]
		      投影向量是向量：
		      \[
			      \frac{\vec a\cdot\vec b}{|\vec b|^2}\vec b.
		      \]
		      两者相差一个 $\vec b$ 方向的单位向量。

		\item[\textbf{\textcolor{CTrap}{错因分析（几何意义不清）：}}] 没有理解投影数量只表示有向长度，投影向量才同时包含大小和方向。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（零向量夹角误判）}}] 看到投影数量为 $0$，就直接判断 $\theta=90^\circ$。

		\item[\textbf{\textcolor{CTrap}{正确理解（补充前提）：}}] 只有在 $\vec a\neq\vec 0,\ \vec b\neq\vec 0$ 且夹角 $\theta$ 有定义时，才能由投影数量为 $0$ 推出
		      \[
			      \cos\theta=0,
			      \quad
			      \theta=90^\circ.
		      \]
		      若 $\vec a=\vec 0$，投影数量也是 $0$，但不能说零向量与 $\vec b$ 的夹角是 $90^\circ$。

		\item[\textbf{\textcolor{CTrap}{错因分析（边界条件遗漏）：}}] 忽略了夹角公式 $\lvert\vec a\rvert\cos\theta$ 对非零向量的要求。
	\end{description}

\end{trapbox}
