\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	投影数量本质是 $\vec a$ 在 $\vec b$ 方向上的有向长度。证明时要先注意条件：$\vec b$ 必须非零；若使用夹角 $\theta$，通常还要考虑 $\vec a\neq\vec 0$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（先看零向量边界）：}] 当 $\vec a=\vec 0$ 且 $\vec b\neq\vec 0$ 时，
		      \[
			      \frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}
			      =
			      \frac{0}{\lvert\vec b\rvert}
			      =
			      0.
		      \]
		      \par\textbf{理由：}
		      零向量在任意非零方向上的投影数量为 $0$，投影向量也为 $\vec 0$。此时不需要讨论 $\vec a$ 与 $\vec b$ 的夹角。

		\item[\textbf{步骤二（设定非零情形）：}] 下面考虑 $\vec a\neq\vec 0,\ \vec b\neq\vec 0$ 的情形。设 $\vec a$ 与 $\vec b$ 的夹角为 $\theta$，其中
		      \[
			      0\le \theta\le \pi.
		      \]
		      \par\textbf{理由：}
		      两个非零向量的夹角才有通常意义下的定义。

		\item[\textbf{步骤三（几何投影）：}] 在夹角为 $\theta$ 的情形下，$\vec a$ 在 $\vec b$ 方向上的投影数量为
		      \[
			      \lvert\vec a\rvert\cos\theta .
		      \]
		      \par\textbf{理由：}
		      过 $\vec a$ 终点向 $\vec b$ 所在直线作垂线，垂足与原点的有向距离即为投影数量，由直角三角形关系可得。若夹角为钝角，则 $\cos\theta<0$，投影数量为负。

		\item[\textbf{步骤四（点积定义）：}] 根据点积定义，
		      \[
			      \vec a\cdot\vec b
			      =
			      \lvert\vec a\rvert\lvert\vec b\rvert\cos\theta .
		      \]
		      \par\textbf{理由：}
		      点积的几何意义是两个向量的模长乘积再乘以夹角余弦。

		\item[\textbf{步骤五（代数变形）：}] 因为 $\vec b\neq\vec 0$，所以 $\lvert\vec b\rvert\neq 0$。将上式两边同时除以 $\lvert\vec b\rvert$，得到
		      \[
			      \frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}
			      =
			      \lvert\vec a\rvert\cos\theta .
		      \]
		      \par\textbf{理由：}
		      分母非零，除法合法；右边正是 $\vec a$ 在 $\vec b$ 方向上的有向投影长度。

		\item[\textbf{步骤六（得到投影向量）：}] 投影向量等于投影数量乘以 $\vec b$ 方向的单位向量：
		      \[
			      \left(\frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}\right)\frac{\vec b}{\lvert\vec b\rvert}
			      =
			      \frac{\vec a\cdot\vec b}{\lvert\vec b\rvert^2}\vec b.
		      \]
		      \par\textbf{理由：}
		      投影数量只是标量，乘上单位方向 $\dfrac{\vec b}{\lvert\vec b\rvert}$ 后，才得到真正的投影向量。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{综上，当 $\vec b\neq\vec 0$ 时，$\vec a$ 在 $\vec b$ 方向上的投影数量为}
	\[
		\frac{\vec a\cdot\vec b}{\lvert\vec b\rvert}.
	\]
	\textbf{若 $\vec a\neq\vec 0$，它也等于}
	\[
		\lvert\vec a\rvert\cos\theta .
	\]
	\textbf{对应的投影向量为}
	\[
		\frac{\vec a\cdot\vec b}{\lvert\vec b\rvert^2}\vec b.
	\]

\end{proofbox}
