\begin{summarybox}
  \textbf{\textcolor{CSummary}{一句话核心}}
  \textbf{绕任意点旋转：先减中心，旋转相对坐标，再加中心；旋转保长度，正负角分方向。}

  \medskip
  \textbf{\textcolor{CSummary}{考场四步}}
  \begin{enumerate}[leftmargin=*,itemsep=0.28em]
    \item 读准旋转中心 $B(a,b)$ 和角度 $\theta$；顺时针取负。
    \item 求相对坐标 $u=x_1-a,\ v=y_1-b$。
    \item 求 $s=u\cos\theta-v\sin\theta,\ t=u\sin\theta+v\cos\theta$。
    \item 加回中心：$\boxed{Q=(a+s,b+t)}$；必要时验证 $BQ=BP$。
  \end{enumerate}

  \medskip
  \textbf{\textcolor{CSummary}{速用公式}}
  \[
  \boxed{\begin{aligned}
    Q&=(a+u\cos\theta-v\sin\theta,\ b+u\sin\theta+v\cos\theta),\\[-1pt]
    45^\circ:\ &(s,t)=\bigl((u-v)/\sqrt2,(u+v)/\sqrt2\bigr),\\
    90^\circ:\ &(s,t)=(-v,u),\quad
    -90^\circ:\ (s,t)=(v,-u).
  \end{aligned}}
  \]
  \textbf{反求角度（$BP=BQ>0$，结果相差整周）：} $\cos\theta=(us+vt)/(u^2+v^2)$，$\sin\theta=(ut-vs)/(u^2+v^2)$。

  \medskip
  \textbf{\textcolor{CSummary}{识别信号与关联}}
  \begin{itemize}[leftmargin=*,itemsep=0.3em]
    \item 题目明确说“绕某点旋转\textbf{某角度}”或“三角形旋转重合”：优先用\textbf{V040通式}。
    \item 题目只说“\textbf{垂直＋等长＋公共端点}”：优先识别 $\pm90^\circ$，调用\textbf{V039快法}。
    \item 只有角度、没有等长：不能直接认作纯旋转；先检查是否还存在长度变化。
    \item 遇到 $BP=BQ$ 且指定一个非直角角度：可利用 V040 建立两个一次坐标关系，优先于大量距离方程展开。
  \end{itemize}
  需判断唯一转向时，先限定 $-180^\circ<\theta\le180^\circ$，再讨论最短转动。
  \textbf{易错提醒：} 旋转不改变长度；未知顺逆要分两支； $P=B$ 时不能从两点反推角度。
\end{summarybox}
