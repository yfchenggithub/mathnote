\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（分母非零）}}]
			使用定比分点坐标公式时，直接代入 $x=\frac{x_1+\lambda x_2}{1+\lambda}$ 而忽略 $\lambda=-1$ 导致分母为零的情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母限制）：}}]
			公式成立的前提是 $\lambda\neq -1$。当 $\lambda=-1$ 时，点 $P$ 不存在（除非 $P_1=P_2$，但通常视为无意义）。
		\item[\textbf{\textcolor{CTrap}{错因分析（未验条件）：}}]
			学生对分式有意义的条件不够重视，解题时未检查分母是否为零。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（符号判定）}}]
			认为 $\lambda>0$ 表示外分点，$\lambda<0$ 表示内分点。
		\item[\textbf{\textcolor{CTrap}{正确理解（λ符号作用）：}}]
			$\lambda>0$ 时 $\overrightarrow{P_1P}$ 与 $\overrightarrow{PP_2}$ 同向，点 $P$ 在线段 $P_1P_2$ 内部（内分点）；$\lambda<0$ 且 $\lambda\neq -1$ 时反向，点 $P$ 在线段延长线上（外分点）。
		\item[\textbf{\textcolor{CTrap}{错因分析（方向混淆）：}}]
			对向量的方向与数量的符号关系理解不清，死记硬背。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式记忆）}}]
			记忆公式为 $x=\frac{\lambda x_1+x_2}{\lambda+1}$ 或 $x=\frac{x_1+x_2}{1+\lambda}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式）：}}]
			应记忆为 $x=\frac{x_1+\lambda x_2}{1+\lambda}$，分子是 $x_1$ 与 $\lambda x_2$ 的和，分母是 $1+\lambda$。
		\item[\textbf{\textcolor{CTrap}{错因分析（权重错位）：}}]
			未理解加权平均的对应关系，混淆了 $x_1$ 和 $x_2$ 的系数。
	\end{description}

\end{trapbox}
