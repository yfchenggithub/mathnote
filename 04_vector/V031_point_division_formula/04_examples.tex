\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $P_1(-1,2)$, $P_2(3,0)$，点 $P$ 满足 $\overrightarrow{P_1P}=3\overrightarrow{PP_2}$，求 $P$ 坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			$\lambda=3$。
			\par\textbf{理由：} 由已知向量等式可得分比 $\lambda=3$。
		\item[\textbf{第二步（代入横坐标）：}]
			由定比分点横坐标公式：
			\[
				\begin{aligned}
					x &= \frac{x_1+\lambda x_2}{1+\lambda} \\ &= \frac{-1+3\cdot 3}{1+3} \\ &= \frac{-1+9}{4} \\ &= \frac{8}{4} \\ &= 2.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用公式。
		\item[\textbf{第三步（代入纵坐标）：}]
			同理，由纵坐标公式：
			\[
				\begin{aligned}
					y &= \frac{y_1+\lambda y_2}{1+\lambda} \\ &= \frac{2+3\cdot 0}{1+3} \\ &= \frac{2}{4} \\ &= 0.5.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P(2,0.5)$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知点 $P(2,5)$ 分有向线段 $P_1P_2$ 的比为 $\lambda=-2$，且 $P_1(4,3)$，求 $P_2$ 坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设坐标）：}]
			设 $P_2(x_2,y_2)$。
			\par\textbf{理由：} 未知端点需设坐标表示。
		\item[\textbf{第二步（利用横坐标公式）：}]
			由横坐标公式：
			\[
				\begin{aligned}
					2 &= \frac{4 + (-2)x_2}{1+(-2)} \\ &= \frac{4-2x_2}{-1} \\ &= -4+2x_2.
			\end{aligned}
			\]
			解方程：
			\[
				\begin{aligned}
					2 &= -4+2x_2 \\ 2x_2 &= 6 \\ x_2 &= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入已知值解方程。
		\item[\textbf{第三步（利用纵坐标公式）：}]
			同理，由纵坐标公式：
			\[
				\begin{aligned}
					5 &= \frac{3 + (-2)y_2}{1+(-2)} \\ &= \frac{3-2y_2}{-1} \\ &= -3+2y_2.
			\end{aligned}
			\]
			解方程：
			\[
				\begin{aligned}
					5 &= -3+2y_2 \\ 2y_2 &= 8 \\ y_2 &= 4.
			\end{aligned}
			\]
			\par\textbf{理由：} 同理。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P_2(3,4)$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$A(2,1)$, $B(4,3)$, $C(6,1)$。（1）求 $BC$ 中点 $M$ 的坐标；（2）设重心 $G$ 满足 $AG:GM=2:1$（即 $\lambda=2$），利用定比分点公式求 $G$ 坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求中点M）：}]
			由中点公式：
			\[
				x_M = \frac{x_B+x_C}{2},
			\]
			\[
				x_M = \frac{4+6}{2},
			\]
			\[
				x_M = 5,
			\]
			\[
				y_M = \frac{y_B+y_C}{2},
			\]
			\[
				y_M = \frac{3+1}{2},
			\]
			\[
				y_M = 2.
			\]
			\par\textbf{理由：} 中点坐标公式（$\lambda=1$ 的定比分点）。
		\item[\textbf{第二步（分析分比）：}]
			$G$ 分线段 $AM$ 的比为 $\lambda=2$（$AG:GM=2:1$），即 $G$ 满足 $\overrightarrow{AG}=2\overrightarrow{GM}$。
			\par\textbf{理由：} 三角形重心性质：重心将中线分为 $2:1$ 两部分。
		\item[\textbf{第三步（代入公式求G）：}]
			由定比分点公式：
			\[
				x_G = \frac{x_A+\lambda x_M}{1+\lambda},
			\]
			\[
				x_G = \frac{2+2\cdot5}{1+2},
			\]
			\[
				x_G = \frac{12}{3},
			\]
			\[
				x_G = 4,
			\]
			\[
				y_G = \frac{y_A+\lambda y_M}{1+\lambda},
			\]
			\[
				y_G = \frac{1+2\cdot2}{3},
			\]
			\[
				y_G = \frac{5}{3}.
			\]
			\par\textbf{理由：} 直接套用定比分点坐标公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$G(4, \frac{5}{3})$}（可验证为三个顶点坐标的算术平均）。
\end{examplebox}
