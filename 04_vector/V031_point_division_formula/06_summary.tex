\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{定比分点坐标公式是线段分点坐标的线性加权表达式，沟通了几何分比与代数坐标。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（分点条件）：} 点 $P$ 满足 $\overrightarrow{P_1P}=\lambda\overrightarrow{PP_2}$（$\lambda\in\mathbb{R},\lambda\neq -1$）。
		\item \textbf{条件2（分母非零）：} $\lambda\neq -1$ 确保分母不为零。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母为零）：} 使用前必须确认 $\lambda\neq -1$，否则公式无意义。
		\item \textbf{检查项（符号判断）：} 根据 $\lambda$ 符号判断内分（$\lambda>0$）或外分（$\lambda<0$）。
	\end{itemize}

\end{summarybox}
