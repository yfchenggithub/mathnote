\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（分点条件）：}}]
			已知两点 $P_1(x_1,y_1)$, $P_2(x_2,y_2)$, 点 $P(x,y)$ 满足 $\overrightarrow{P_1P} = \lambda \overrightarrow{PP_2}$, 其中 $\lambda\in\mathbb{R}$ 且 $\lambda\neq -1$.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（坐标公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			分点 $P$ 的坐标由两端点坐标与分比 $\lambda$ 加权平均得到。
			\[
				x = \frac{x_1 + \lambda x_2}{1+\lambda},
				\qquad
				y = \frac{y_1 + \lambda y_2}{1+\lambda}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（中点特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $\lambda=1$ 时, $P$ 为线段 $P_1P_2$ 的中点, 坐标公式简化为中点坐标公式。
			\[
				x = \frac{x_1 + x_2}{2},
				\qquad
				y = \frac{y_1 + y_2}{2}.
			\]
	\end{description}

\end{statementbox}
