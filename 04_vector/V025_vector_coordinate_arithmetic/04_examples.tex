\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\vec{a}=(2,-1)$, $\vec{b}=(3,4)$，求： (1) $\vec{a}+\vec{b}$; (2) $\vec{a}-\vec{b}$; (3) $2\vec{a}$; (4) $3\vec{a}-\vec{b}$; (5) $\vec{a}\cdot\vec{b}$.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（加减运算）：}]
			\[
				\vec{a}+\vec{b}=(2+3,-1+4)=(5,3),\quad \vec{a}-\vec{b}=(2-3,-1-4)=(-1,-5)
			\]
			\par\textbf{理由：} 向量加减法坐标公式，对应坐标相加或相减.
		\item[\textbf{第二步（数乘运算）：}]
			\[
				2\vec{a}=(2\times 2,2\times(-1))=(4,-2)
			\]
			\par\textbf{理由：} 数乘公式，每个坐标乘以2.
		\item[\textbf{第三步（混合运算）：}]
			\[
				3\vec{a}-\vec{b}=3(2,-1)-(3,4)=(6-3,-3-4)=(3,-7)
			\]
			\par\textbf{理由：} 先数乘再减法.
		\item[\textbf{第四步（数量积）：}]
			\[
				\vec{a}\cdot\vec{b}=2\times3+(-1)\times4=6-4=2
			\]
			\par\textbf{理由：} 点乘的坐标公式 $x_1x_2+y_1y_2$.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{\vec{a}+\vec{b}=(5,3),\;\vec{a}-\vec{b}=(-1,-5),\;2\vec{a}=(4,-2),\;3\vec{a}-\vec{b}=(3,-7),\;\vec{a}\cdot\vec{b}=2}$.

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\vec{a}=(m,2)$, $\vec{b}=(1,-3)$，且 $2\vec{a}+3\vec{b}=(-1,-5)$，求 $m$ 的值.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线性组合坐标化）：}]
			利用坐标运算公式
			\[
				2\vec{a}+3\vec{b}=2(m,2)+3(1,-3)=(2m+3,4-9)=(2m+3,-5)
			\]
			\par\textbf{理由：} 数乘和加法坐标运算，分别计算横纵坐标.
		\item[\textbf{第二步（对比条件）：}]
			已知 $2\vec{a}+3\vec{b}=(-1,-5)$，故
			\[
				(2m+3,-5)=(-1,-5)
			\]
			\par\textbf{理由：} 向量相等对应坐标相等.
		\item[\textbf{第三步（解方程）：}]
			由横坐标相等得 $2m+3=-1$，解得 $m=-2$.
			\par\textbf{理由：} 解一元一次方程.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{m=-2}$.

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\vec{a}=(2,-1)$, $\vec{b}=(-3,4)$，求 $|2\vec{a}-3\vec{b}|$.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算向量坐标）：}]
			先求 $2\vec{a}-3\vec{b}$ 的坐标.
			\[
				2\vec{a}-3\vec{b}=2(2,-1)-3(-3,4)=(4+9,\,-2-12)=(13,-14)
			\]
			\par\textbf{理由：} 数乘与减法坐标公式.
		\item[\textbf{第二步（计算模的平方）：}]
			利用点乘公式，
			\[
				|2\vec{a}-3\vec{b}|^2 = (13,-14)\cdot(13,-14)=13\times13+(-14)\times(-14)=169+196=365
			\]
			\par\textbf{理由：} 模的平方等于向量与自身的点乘，点乘使用坐标公式.
		\item[\textbf{第三步（开方得模）：}]
			\[
				|2\vec{a}-3\vec{b}| = \sqrt{365}
			\]
			\par\textbf{理由：} 模长非负，直接开平方.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{|2\vec{a}-3\vec{b}|=\sqrt{365}}$.
\end{examplebox}
