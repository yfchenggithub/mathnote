\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（单位正交基底）：}}]
			平面向量在单位正交基底 $\{\mathbf{i},\mathbf{j}\}$ 下的坐标为 $\vec{a}=(x_1,y_1)$, $\vec{b}=(x_2,y_2)$，$\lambda\in\mathbb{R}$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（向量加减法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对应坐标分别相加或相减。
			\[
				\vec{a}\pm\vec{b}=(x_1\pm x_2,\; y_1\pm y_2)
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（数乘运算）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每个坐标乘以相同倍数。
			\[
				\lambda\vec{a}=(\lambda x_1,\; \lambda y_1)
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（数量积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对应坐标乘积之和。
			\[
				\vec{a}\cdot\vec{b}=x_1x_2+y_1y_2
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（线性组合）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任意数乘与加减组合可直接运算坐标。
			\[
				k_1\vec{a}+k_2\vec{b}=(k_1x_1+k_2x_2,\; k_1y_1+k_2y_2)
			\]
	\end{description}

\end{statementbox}
