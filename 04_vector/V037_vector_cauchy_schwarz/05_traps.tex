\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略绝对值符）}}]
		      把 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$ 误写成 $\vec{a}\cdot\vec{b} \le |\vec{a}||\vec{b}|$。后者虽然恒成立，但只给出上界，不能替代原不等式。

		\item[\textbf{\textcolor{CTrap}{正确理解（使用绝对不等式）：}}]
		      应牢记 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$，然后可拆分为 $-|\vec{a}||\vec{b}| \le \vec{a}\cdot\vec{b} \le |\vec{a}||\vec{b}|$。

		\item[\textbf{\textcolor{CTrap}{错因分析（符号性质混淆）：}}]
		      忽略数量积可正可负，导致只保留上界，丢失 $\vec{a}\cdot\vec{b} \ge -|\vec{a}||\vec{b}|$ 这一侧。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（等号条件不完整）：}}]
		      认为等号只在 $\vec{a}$ 与 $\vec{b}$ 同向时成立，忽略反向共线及零向量情形。

		\item[\textbf{\textcolor{CTrap}{正确理解（共线含零向量）：}}]
		      等号成立当且仅当 $\vec{a}\parallel\vec{b}$（包括同向、反向及零向量）。

		\item[\textbf{\textcolor{CTrap}{错因分析（共线定义不清）：}}]
		      对共线的概念理解片面，或忘记零向量与任何向量共线。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（平方开方失符号）：}}]
		      由 $(\vec{a}\cdot\vec{b})^2 \le |\vec{a}|^2|\vec{b}|^2$ 直接得 $\vec{a}\cdot\vec{b} \le |\vec{a}||\vec{b}|$，丢失绝对值。

		\item[\textbf{\textcolor{CTrap}{正确理解（开方需加绝对值）：}}]
		      开方结果必须加绝对值，即 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$。

		\item[\textbf{\textcolor{CTrap}{错因分析（算术根理解错）：}}]
		      算术平方根是非负的，必须有绝对值符号。
	\end{description}
\end{trapbox}
