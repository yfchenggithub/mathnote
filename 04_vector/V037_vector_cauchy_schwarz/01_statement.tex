\begin{statementbox}
	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件 1（向量条件）：}}]
			$\vec{a},\vec{b}$ 为任意向量（平面或空间）。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（绝对值不等式）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			数量积的绝对值不超过模长的乘积。\n
			\[
				\n |\vec{a}\cdot\vec{b}| \le |\vec{a}|\,|\vec{b}|\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（平方形式）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			平方后便于代数变形，无需处理绝对值。\n
			\[
				\n (\vec{a}\cdot\vec{b})^2 \le |\vec{a}|^2|\vec{b}|^2\n
			\]
			\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{等号/取等条件：}} 等号成立当且仅当 $\vec{a}$ 与 $\vec{b}$ 共线（含零向量；共线包括同向和反向）。\n
\end{statementbox}
