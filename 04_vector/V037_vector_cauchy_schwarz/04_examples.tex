\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知 $\vec{a}=(1,2),\ \vec{b}=(3,4)$，计算 $|\vec{a}\cdot\vec{b}|$ 与 $|\vec{a}||\vec{b}|$，验证柯西–施瓦茨不等式。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求点积）：}]
			\n $\vec{a}\cdot\vec{b}=1\times3+2\times4=11$。\n
			\par\textbf{理由：} 利用向量点积的坐标运算法则：$\vec{a}\cdot\vec{b}=x_1x_2+y_1y_2$。\n
		\item[\textbf{第二步（算模长积）：}]
			\n $|\vec{a}|=\sqrt{1^2+2^2}=\sqrt{5}$，$|\vec{b}|=\sqrt{3^2+4^2}=5$，故 $|\vec{a}||\vec{b}|=5\sqrt{5}$。\n
			\par\textbf{理由：} 向量模长公式 $|\vec{v}|=\sqrt{x^2+y^2}$。\n
		\item[\textbf{第三步（比较验证）：}]
			\n $11 < 5\sqrt{5}\approx 11.18$，因此 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$；由于 $11 \neq 5\sqrt{5}$，等号不成立，$\vec{a},\vec{b}$ 不共线。\n
			\par\textbf{理由：} 代入不等式直接比较，等号条件不满足。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{验证了柯西–施瓦茨不等式，且等号不成立。}\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知 $|\vec{a}|=2,\ |\vec{b}|=3$，求 $\vec{a}\cdot\vec{b}$ 的最大值和最小值，并指出取得最值时 $\vec{a},\vec{b}$ 的方向关系。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（应用不等式）：}]
			\n 由 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|=2\times3=6$，得到 $-6 \le \vec{a}\cdot\vec{b} \le 6$。\n
			\par\textbf{理由：} 绝对值不等式 $|x|\le A$ 等价于 $-A\le x\le A$。\n
		\item[\textbf{第二步（等号条件分析）：}]
			\n 等号成立当且仅当 $\vec{a}\parallel\vec{b}$。当 $\vec{a},\vec{b}$ 同向时 $\vec{a}\cdot\vec{b}=6$（最大）；反向时 $\vec{a}\cdot\vec{b}=-6$（最小）。\n
			\par\textbf{理由：} 共线时点积等于模长乘积乘以 $\pm1$，符号由方向决定。\n
		\item[\textbf{第三步（得出结论）：}]
			\n 最大值 $6$（同向），最小值 $-6$（反向）。\n
			\par\textbf{理由：} 端点即为最值，由不等式保证。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\vec{a}\cdot\vec{b}$ 的取值范围是 $[-6,6]$，端点对应共线情形。}\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知 $|\vec{a}|=1$，且 $|\vec{a}\cdot\vec{b}|=2$，求 $|\vec{b}|$ 的最小值，并判断此时 $\vec{a},\vec{b}$ 的关系。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（套用不等式）：}]
			\n 由 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$ 代入 $|\vec{a}|=1,|\vec{a}\cdot\vec{b}|=2$ 得 $2 \le 1\cdot|\vec{b}|$，即 $|\vec{b}|\ge 2$。\n
			\par\textbf{理由：} 不等式直接给出 $|\vec{b}|$ 的下界。\n
		\item[\textbf{第二步（确定最小值）：}]
			\n 因此 $|\vec{b}|$ 的最小值为 $2$。\n
			\par\textbf{理由：} 下界可取到，取等时 $|\vec{b}|=2$ 满足不等式等号。\n
		\item[\textbf{第三步（等号条件与方向）：}]
			\n 等号成立 $\Rightarrow \vec{a}\parallel\vec{b}$。又 $|\vec{a}\cdot\vec{b}|=2 = |\vec{a}||\vec{b}|$，故 $\vec{a}\cdot\vec{b} = \pm 2$；同向时 $\vec{a}\cdot\vec{b}=2$，反向时 $\vec{a}\cdot\vec{b}=-2$。但题目只给绝对值，故两种共线情况均可能。\n
			\par\textbf{理由：} 等号条件要求共线，点积符号取决于方向。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$|\vec{b}|_{\min}=2$，此时 $\vec{a}\parallel\vec{b}$（同向或反向）。}\n
\end{examplebox}
