\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n通过构造向量 $\vec{a}-t\vec{b}$ 并利用其模平方非负，得到关于 $t$ 的二次不等式，再由判别式非负导出结论。\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n\n当 $\vec{b}=\vec{0}$ 时，不等式两边为 $0$，结论显然成立（等号成立）；下面假设 $\vec{b}\neq \vec{0}$，此时 $|\vec{b}|>0$。\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（构造非负式）：}]
			\n
			\[
				\n |\vec{a} - t\vec{b}|^2 \ge 0\n
			\]
			\n
			\par\textbf{理由：} 任意向量模平方恒非负；$t$ 为任意实数。\n\n
		\item[\textbf{步骤二（展开平方）：}]
			\n
			\[
				\n |\vec{a}|^2 - 2t(\vec{a}\cdot\vec{b}) + t^2|\vec{b}|^2 \ge 0\n
			\]
			\n
			\par\textbf{理由：} 利用 $|\vec{v}|^2 = \vec{v}\cdot\vec{v}$ 展开完全平方。\n\n
		\item[\textbf{步骤三（判别式非正）：}]
			\n 由上述不等式对任意 $t$ 成立，故关于 $t$ 的二次函数 $f(t)=|\vec{b}|^2 t^2 - 2(\vec{a}\cdot\vec{b})t + |\vec{a}|^2$ 的判别式 $\Delta \le 0$：\n
			\[
				\n \Delta = 4(\vec{a}\cdot\vec{b})^2 - 4|\vec{a}|^2|\vec{b}|^2 \le 0\n
			\]
			\n
			\par\textbf{理由：} 二次项系数 $|\vec{b}|^2 > 0$（因 $\vec{b}\neq \vec{0}$），且 $f(t)\ge 0$ 恒成立，故判别式 $\Delta \le 0$。\n\n
		\item[\textbf{步骤四（化简得不等式）：}]
			\n 化简得 $(\vec{a}\cdot\vec{b})^2 \le |\vec{a}|^2|\vec{b}|^2$，两边开方得 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$：\n
			\[
				\n |\vec{a}\cdot\vec{b}| \le |\vec{a}|\,|\vec{b}|\n
			\]
			\n
			\par\textbf{理由：} 代数化简与非负实数开方。\n\n
		\item[\textbf{步骤五（等号条件分析）：}]
			\n 等号成立 $\Leftrightarrow \Delta = 0 \Leftrightarrow$ 存在 $t$ 使 $|\vec{a}-t\vec{b}|=0$，即 $\vec{a}=t\vec{b}$。因此向量共线：\n
			\[
				\n \vec{a} \parallel \vec{b} \quad (\text{含零向量})\n
			\]
			\n
			\par\textbf{理由：} 判别式为零对应二次函数有重根，此时 $\vec{a}-t\vec{b}=\vec{0}$。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n\textbf{综上，} 对任意向量 $\vec{a},\vec{b}$，恒有 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$，等号成立当且仅当 $\vec{a} \parallel \vec{b}$。该证明是 Cauchy-Schwarz 不等式向量形式的标准推导，核心是利用平方非负构造二次函数。\n
\end{proofbox}
