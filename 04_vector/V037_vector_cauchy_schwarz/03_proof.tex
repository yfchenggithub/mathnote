\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过构造向量 $\vec{a}-t\vec{b}$ 并利用其模平方非负，得到关于 $t$ 的二次不等式，再由判别式非正导出结论。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}

	当 $\vec{b}=\vec{0}$ 时，不等式两边为 $0$，结论显然成立（等号成立）；下面假设 $\vec{b}\neq \vec{0}$，此时 $|\vec{b}|>0$。

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（构造非负式）：}]
		      \[
			      |\vec{a} - t\vec{b}|^2 \ge 0
		      \]
		      \par\textbf{理由：} 任意向量模平方恒非负；$t$ 为任意实数。

		\item[\textbf{步骤二（展开平方）：}]
		      \[
			      |\vec{a}|^2 - 2t(\vec{a}\cdot\vec{b}) + t^2|\vec{b}|^2 \ge 0
		      \]
		      \par\textbf{理由：} 利用 $|\vec{v}|^2 = \vec{v}\cdot\vec{v}$ 展开完全平方。

		\item[\textbf{步骤三（判别式非正）：}]
		      由上述不等式对任意 $t$ 成立，故关于 $t$ 的二次函数
		      \[
			      f(t)=|\vec{b}|^2 t^2 - 2(\vec{a}\cdot\vec{b})t + |\vec{a}|^2
		      \]
		      恒有 $f(t)\ge 0$。因为二次项系数 $|\vec{b}|^2>0$，所以判别式 $\Delta\le 0$，即
		      \[
			      \Delta = 4(\vec{a}\cdot\vec{b})^2 - 4|\vec{a}|^2|\vec{b}|^2 \le 0
		      \]
		      \par\textbf{理由：} 开口向上的二次函数若对任意实数 $t$ 都非负，则其图像与 $t$ 轴至多有一个交点，故判别式不大于 $0$。

		\item[\textbf{步骤四（化简得不等式）：}]
		      化简得
		      \[
			      (\vec{a}\cdot\vec{b})^2 \le |\vec{a}|^2|\vec{b}|^2
		      \]
		      两边开方得
		      \[
			      |\vec{a}\cdot\vec{b}| \le |\vec{a}|\,|\vec{b}|
		      \]
		      \par\textbf{理由：} 两边均为非负量，开方时左边为 $|\vec{a}\cdot\vec{b}|$。

		\item[\textbf{步骤五（等号条件分析）：}]
		      在 $\vec{b}\neq \vec{0}$ 的前提下，等号成立
		      \[
			      \Longleftrightarrow \Delta = 0
			      \Longleftrightarrow \exists\,t\in\mathbb{R},\ |\vec{a}-t\vec{b}|=0
			      \Longleftrightarrow \vec{a}=t\vec{b}.
		      \]
		      因此 $\vec{a}$ 与 $\vec{b}$ 共线：
		      \[
			      \vec{a} \parallel \vec{b}.
		      \]
		      若 $\vec{b}=\vec{0}$，等号也成立，且零向量通常视为与任意向量共线。

		      \par\textbf{理由：} 判别式为零对应二次函数有重根，此时非负式能取到 $0$，即 $\vec{a}-t\vec{b}=\vec{0}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{综上，} 对任意向量 $\vec{a},\vec{b}$，恒有
	\[
		|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|,
	\]
	等号成立当且仅当 $\vec{a} \parallel \vec{b}$（含零向量）。该证明是 Cauchy-Schwarz 不等式向量形式的标准推导，核心是利用平方非负构造二次函数。
\end{proofbox}
