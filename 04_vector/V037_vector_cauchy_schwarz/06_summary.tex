\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{对任意向量 $\vec{a},\vec{b}$，恒有 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$，等号成立当且仅当 $\vec{a}\parallel\vec{b}$（含零向量）。}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（任意向量）：} $\vec{a},\vec{b}$ 为任何平面向量或空间向量。\n
		\item \textbf{条件 2（等号条件）：} 需要 $\vec{a}$ 与 $\vec{b}$ 共线（包括零向量）。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（忽略绝对值）：} 务必使用绝对值不等式 $|\vec{a}\cdot\vec{b}|$，而非直接 $\vec{a}\cdot\vec{b}$。\n
		\item \textbf{检查项（等号条件）：} 利用不等式求最值时，须验证等号能否成立，并考虑共线的正反方向。\n
	\end{itemize}
	\n
\end{summarybox}
