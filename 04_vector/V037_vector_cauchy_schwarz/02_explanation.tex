\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n向量数量积的绝对值不会超过两个向量模长的乘积，因为数量积相当于“模长相乘再乘以夹角的余弦”，而余弦的绝对值不超过 1。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（投影与夹角）：}]
			由数量积定义 $\vec{a}\cdot\vec{b}=|\vec{a}||\vec{b}|\cos\theta$，其中 $\theta$ 为两向量夹角。\n
		\item[\textbf{要点二（余弦有界性）：}]
			余弦函数满足 $|\cos\theta|\le 1$，代入即得 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（投影长度约束）}}\n\n$\vec{a}$ 在 $\vec{b}$ 方向上的投影长度为 $|\vec{a}|\cos\theta$，其绝对值不超过 $|\vec{a}|$，故乘积 $|\vec{a}||\vec{b}|\cos\theta$ 的绝对值不超过 $|\vec{a}||\vec{b}|$。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（基本不等式约束）}}\n\n该不等式是向量内积空间中的基本不等式，用于约束点积大小，可推出三角不等式等结论。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（最值·共线）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（最值问题）：} 已知 $|\vec{a}|,|\vec{b}|$ 求 $\vec{a}\cdot\vec{b}$ 的最大、最小值，或已知 $|\vec{a}|$ 与 $\vec{a}\cdot\vec{b}$ 求 $|\vec{b}|$ 的最小值。\n
		\item \textbf{考法二（共线判定）：} 利用等号条件判断向量共线（常出现在几何证明与坐标系中）。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n\textbf{问题根本是余弦绝对值 $\le 1$：} 不论向量维数如何，总有 $|\cos\theta|\le 1$，所以 $|\vec{a}\cdot\vec{b}| \le |\vec{a}||\vec{b}|$。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（向量运算）：} 直接应用公式判断数量积与模长乘积的大小关系。\n
		\item \textbf{场景二（坐标转化）：} 在直角坐标系中化为 $(x_1x_2+y_1y_2)^2 \le (x_1^2+y_1^2)(x_2^2+y_2^2)$，用于求解变量范围或证明不等式。\n
	\end{itemize}
	\n
\end{explanationbox}
