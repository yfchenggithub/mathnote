\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略非零条件）}}]
		      错误地认为只要 $\vec{a}+\vec{b}+\vec{c}=\mathbf{0}$ 就能构成非退化三角形，而不检查向量是否非零。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须非零）}}]
		      结论要求 $\vec{a},\vec{b},\vec{c}$ 均为非零向量；若出现零向量，则无法构成非退化三角形。
		\item[\textbf{\textcolor{CTrap}{错因分析（零向量致退化）}}]
		      零向量的方向不确定，且长度为 $0$，无法作为三角形的一边，导致图形退化为线段或点。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略不共线条件）}}]
		      仅凭三个向量和为 $\mathbf{0}$ 就断言构成非退化三角形，未检查是否两两不共线。
		\item[\textbf{\textcolor{CTrap}{正确理解（两两不共线）}}]
		      只有三个非零向量两两不共线时，和为零才等价于构成非退化三角形；若存在共线，则三点共线，不能形成非退化三角形。
		\item[\textbf{\textcolor{CTrap}{错因分析（共线致退化）}}]
		      在 $\vec{a}+\vec{b}+\vec{c}=\mathbf{0}$ 的前提下，若其中两个向量共线，则第三个向量也会与它们共线，首尾相接后只能形成退化图形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（方向顺序错误）}}]
		      误认为只要三个向量都来自三角形的边，和就一定为零。
		\item[\textbf{\textcolor{CTrap}{正确理解（首尾顺序要一致）}}]
		      必须按照 $\vec{a} = \overrightarrow{AB},\ \vec{b} = \overrightarrow{BC},\ \vec{c} = \overrightarrow{CA}$ 的顺序，即围绕三角形一周，方向连续；若顺序或方向混乱（如 $\vec{c} = \overrightarrow{BA}$），则和不一定为零。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆起点终点）}}]
		      向量方向影响加法的结果；若改变某个向量的方向或起点终点，会破坏三角形回路，导致和向量非零。
	\end{description}

\end{trapbox}
