\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	设非零向量 $\vec{a},\vec{b},\vec{c}$ 满足 $\vec{a}+\vec{b}+\vec{c}=\mathbf{0}$，且 $|\vec{a}|=2$，$|\vec{b}|=3$，$\vec{a}$ 与 $\vec{b}$ 的夹角为 $60^\circ$，求 $|\vec{c}|$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代换消元）：}]
		      由 $\vec{a}+\vec{b}+\vec{c}=\mathbf{0}$ 得 $\vec{c} = -(\vec{a}+\vec{b})$。
		      \par\textbf{理由：} 移项变形。
		\item[\textbf{第二步（模长平方）：}]
		      \[
			      \begin{aligned}
				      |\vec{c}|^2
				       & = |\vec{a}+\vec{b}|^2                                       \\
				       & = |\vec{a}|^2+|\vec{b}|^2+2|\vec{a}||\vec{b}|\cos 60^\circ.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 向量模长平方公式。
		\item[\textbf{第三步（代入求值）：}]
		      代入数据 $4+9+2\times2\times3\times\frac12 = 13+6 = 19$，故 $|\vec{c}| = \sqrt{19}$。
		      \par\textbf{理由：} 算术运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|\vec{c}| = \sqrt{19}$}.

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知非零向量 $\vec{a},\vec{b},\vec{c}$ 满足 $\vec{a}+\vec{b}+\vec{c}=\mathbf{0}$，且 $|\vec{a}|=|\vec{b}|=1$，$|\vec{c}|=\sqrt{2}$，求 $\vec{a}$ 与 $\vec{b}$ 的夹角 $\theta$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（和为零转化）：}]
		      由 $\vec{a}+\vec{b}+\vec{c}=\mathbf{0}$ 得 $\vec{a}+\vec{b} = -\vec{c}$，两端取模并平方得 $|\vec{a}+\vec{b}|^2 = |\vec{c}|^2$。
		      \par\textbf{理由：} 移项后利用相反向量模长相等。
		\item[\textbf{第二步（展开）：}]
		      $|\vec{a}|^2+|\vec{b}|^2+2|\vec{a}||\vec{b}|\cos\theta = |\vec{c}|^2$。
		      \par\textbf{理由：} 向量模长平方展开。
		\item[\textbf{第三步（求解）：}]
		      代入 $1+1+2\cos\theta = 2$，得 $2+2\cos\theta=2$，$\cos\theta=0$，所以 $\theta = 90^\circ$。
		      \par\textbf{理由：} 因为 $0^\circ\leq \theta\leq 180^\circ$，所以 $\cos\theta=0$ 对应 $\theta=90^\circ$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\theta = 90^\circ$}.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，设 $\overrightarrow{AB} = \vec{c}$，$\overrightarrow{BC} = \vec{a}$，$\overrightarrow{CA} = \vec{b}$，已知 $|\vec{a}|=3$，$|\vec{b}|=4$，$|\vec{c}|=5$，求 $\vec{a}\cdot\vec{b} + \vec{b}\cdot\vec{c} + \vec{c}\cdot\vec{a}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（三角形封闭）：}]
		      由向量回路性质，$\vec{a}+\vec{b}+\vec{c} = \mathbf{0}$。
		      \par\textbf{理由：} 三角形三边向量首尾相接，$\overrightarrow{BC}+\overrightarrow{CA}+\overrightarrow{AB}=\mathbf{0}$。
		\item[\textbf{第二步（平方展开）：}]
		      对 $\vec{a}+\vec{b}+\vec{c} = \mathbf{0}$ 两边平方：
		      \[
			      \begin{aligned}
				      0
				       & = |\vec{a}|^2+|\vec{b}|^2+|\vec{c}|^2                                       \\
				       & \quad + 2(\vec{a}\cdot\vec{b} + \vec{b}\cdot\vec{c} + \vec{c}\cdot\vec{a}).
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 完全平方展开。
		\item[\textbf{第三步（计算）：}]
		      设 $S=\vec{a}\cdot\vec{b} + \vec{b}\cdot\vec{c} + \vec{c}\cdot\vec{a}$。代入 $9+16+25 = 50$，得 $50 + 2S = 0$，故 $S = -25$。
		      \par\textbf{理由：} 解一元一次方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\vec{a}\cdot\vec{b} + \vec{b}\cdot\vec{c} + \vec{c}\cdot\vec{a} = -25$}.
