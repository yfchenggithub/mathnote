\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（非零且不共线）：}}]
		      三个平面向量 $\vec{a},\vec{b},\vec{c}$ 均为非零向量，且两两不共线。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（和为零与三角形）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      三个向量首尾相接可构成封闭三角形。
		      \[
			      \vec{a}+\vec{b}+\vec{c}=\mathbf{0}
			      \iff
			      \exists\;A,B,C,\ \vec{a}=\overrightarrow{AB},\ \vec{b}=\overrightarrow{BC},\ \vec{c}=\overrightarrow{CA}.
		      \]
		\item[\textbf{\textcolor{CStatement}{推广结论（三角形回路）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      三角形三边按顺序的向量和恒为零向量。
		      \[
			      \overrightarrow{AB}+\overrightarrow{BC}+\overrightarrow{CA}=\mathbf{0}.
		      \]
	\end{description}

\end{statementbox}
