\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（任意向量）}}]
			向量 $\mathbf{a},\mathbf{b}$ 为平面或空间中的任意向量。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（模三角不等式）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个向量模的和差运算满足三角不等式关系。
			\[
				\bigl| |\mathbf{a}| - |\mathbf{b}| \bigr| \leq |\mathbf{a} \pm \mathbf{b}| \leq |\mathbf{a}| + |\mathbf{b}|.
			\]
			等号成立条件：对于 $|\mathbf{a}+\mathbf{b}|$，右边等号成立当且仅当 $\mathbf{a}$ 与 $\mathbf{b}$ 同向共线，左边等号成立当且仅当反向共线；对于 $|\mathbf{a}-\mathbf{b}|$，右边等号成立当且仅当 $\mathbf{a}$ 与 $\mathbf{b}$ 反向共线，左边等号成立当且仅当同向共线。

			\medskip
		\item[\textbf{\textcolor{CStatement}{常见变形（多向量模和）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			多个向量和的模不超过各向量模的和。
			\[
				|\mathbf{a}_1 + \mathbf{a}_2 + \cdots + \mathbf{a}_n| \le |\mathbf{a}_1| + |\mathbf{a}_2| + \cdots + |\mathbf{a}_n|.
			\]
			等号成立当且仅当所有向量同向共线。
	\end{description}

\end{statementbox}
