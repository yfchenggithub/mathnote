\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（取等条件混淆）}}]
			错误地认为 $|\mathbf{a}+\mathbf{b}|$ 取最大值时向量反向，$|\mathbf{a}-\mathbf{b}|$ 取最大值时同向。
		\item[\textbf{\textcolor{CTrap}{正确理解（同向和反向对应）}}]
			$|\mathbf{a}+\mathbf{b}|$ 取等时 $\mathbf{a},\mathbf{b}$ 同向，$|\mathbf{a}-\mathbf{b}|$ 取等时 $\mathbf{a},\mathbf{b}$ 反向。
		\item[\textbf{\textcolor{CTrap}{错因分析（几何方向错误）}}]
			对向量减法法则不熟，误以为差向量指向被减向量终点，导致方向判断失误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗漏下限）}}]
			只记住 $|\mathbf{a}\pm\mathbf{b}|\le|\mathbf{a}|+|\mathbf{b}|$，而忽略 $|\mathbf{a}\pm\mathbf{b}|\ge||\mathbf{a}|-|\mathbf{b}||$。
		\item[\textbf{\textcolor{CTrap}{正确理解（下限同样重要）}}]
			三角不等式是双向的，左边给出模的下界，在求最小值时必用。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆片面）}}]
			加法不等式直观，减法不等式容易被忽视，需要对称记忆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（模的线性误用）}}]
			直接展开 $|\mathbf{a}\pm\mathbf{b}|$ 为 $|\mathbf{a}|\pm|\mathbf{b}|$，尤其在数值计算中偷懒。
		\item[\textbf{\textcolor{CTrap}{正确理解（模不满足线性）}}]
			除非已知向量共线方向，否则模的和差与向量和差的模不相等。
		\item[\textbf{\textcolor{CTrap}{错因分析（类比绝对值错误）}}]
			将实数的 $|a\pm b|\le |a|+|b|$ 与 $|a\pm b|\ge ||a|-|b||$ 错误理解为恒等，忽略向量方向。
	\end{description}

\end{trapbox}
