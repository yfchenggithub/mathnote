\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{向量三角不等式 $||\mathbf{a}|-|\mathbf{b}||\le|\mathbf{a}\pm\mathbf{b}|\le|\mathbf{a}|+|\mathbf{b}|$ 是刻画向量模取值范围的基本工具，可用于最值、证明与判断。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（任意向量）：} 对任意平面或空间向量 $\mathbf{a},\mathbf{b}$ 都成立，无需额外条件。
		\item \textbf{条件2（共线取等）：} 等号仅当 $\mathbf{a}$ 与 $\mathbf{b}$ 共线时才能达到，方向决定具体等号类型。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（取等方向）：} 和差取等方向相反，务必根据具体运算判断共线方向。
		\item \textbf{检查项（模分配误区）：} 不要将 $|\mathbf{a}\pm\mathbf{b}|$ 直接拆成 $|\mathbf{a}|\pm|\mathbf{b}|$，必须结合共线性。
	\end{itemize}

\end{summarybox}
