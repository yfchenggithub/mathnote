\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（非零向量前提）：}}] 向量 $\vec{a} \neq \vec{0}$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（单位向量定义）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将非零向量除以自身长度得到长度 1 的同向向量。
			\[
				\vec{e} = \frac{\vec{a}}{|\vec{a}|}, \quad |\vec{e}| = 1.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（向量分解）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任何一个非零向量可以表示为长度乘以单位向量。
			\[
				\vec{a} = |\vec{a}| \vec{e}.
			\]
	\end{description}
\end{statementbox}
