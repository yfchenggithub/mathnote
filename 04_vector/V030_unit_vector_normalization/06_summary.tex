\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{通过除以自身模长，将非零向量的长度标准化为1，得到同方向的单位向量。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（非零前提）：} 原向量必须是非零向量，即 $|\vec{a}| \neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（非零条件）：} 零向量无单位向量，使用前必须检查。
		\item \textbf{检查项（模长验证）：} 得到单位向量后应验证 $|\vec{e}|=1$。
	\end{itemize}
\end{summarybox}
