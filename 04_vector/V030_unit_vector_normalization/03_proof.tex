\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	要证明两点：一方面 $|\vec{e}|=1$，另一方面 $\vec{e}$ 与 $\vec{a}$ 方向相同。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（构造表达式）：}] 令 $\vec{e} = \dfrac{\vec{a}}{|\vec{a}|}$。
			\[
				\vec{e} = \frac{\vec{a}}{|\vec{a}|}
			\]
			\par\textbf{理由：}
			这是单位向量的定义。

		\item[\textbf{步骤二（计算模长）：}] 计算 $|\vec{e}|$。
			\[
				\begin{aligned}
					|\vec{e}| &= \left| \frac{\vec{a}}{|\vec{a}|} \right| \\
					&= \frac{|\vec{a}|}{|\vec{a}|} \\
					&= 1
			\end{aligned}
			\]
			\par\textbf{理由：}
			向量的模满足 $|\lambda\vec{a}| = |\lambda||\vec{a}|$，这里 $\lambda = \frac{1}{|\vec{a}|} > 0$，所以 $\left|\frac{\vec{a}}{|\vec{a}|}\right| = \frac{|\vec{a}|}{|\vec{a}|}=1$。

		\item[\textbf{步骤三（判断方向）：}] $\vec{e}$ 与 $\vec{a}$ 方向相同。
			\[
				\vec{e} = \frac{1}{|\vec{a}|}\vec{a}
			\]
			\par\textbf{理由：}
			$\frac{1}{|\vec{a}|} > 0$ 为正实数，正数乘以向量不改变方向，只会缩放长度。

		\item[\textbf{步骤四（归纳结论）：}] 由以上两步，$\vec{e}$ 是与 $\vec{a}$ 同方向的单位向量。
			\[
				\vec{e} \parallel \vec{a}, \quad |\vec{e}| = 1.
			\]
			\par\textbf{理由：}
			综合模长和方向结论。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	对任意非零向量 $\vec{a}$，按照公式 $\vec{e} = \dfrac{\vec{a}}{|\vec{a}|}$ 构造出的向量长度恒为 1，且方向与原向量一致，称为单位向量。
\end{proofbox}
