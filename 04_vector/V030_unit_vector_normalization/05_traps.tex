\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（零向量特例）}}] 错误地认为零向量也有单位向量。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须非零）：}}] 只有非零向量才能定义同向单位向量；零向量方向不定，无单位向量。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略前提）：}}] 没有注意定义中“非零向量”的前提条件。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（方向取反）}}] 求与 $\vec{a}$ 方向相反的单位向量时，仍用 $\dfrac{\vec{a}}{|\vec{a}|}$ 导致方向错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（添加负号）：}}] 方向相反的单位向量应为 $-\dfrac{\vec{a}}{|\vec{a}|}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号遗漏）：}}] 忽略系数正负对方向的影响；同向为正标量，反向为负标量。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（长度方向混淆）}}] 误以为单位向量类似常数“1”，忽略其方向属性。
		\item[\textbf{\textcolor{CTrap}{正确理解（模一方向定）：}}] 单位向量的长度恒为1，但方向随原向量变化；两个不同的非零向量对应不同的单位向量。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念切割）：}}] 将单位向量与单位长度混淆，未认识到向量是“大小+方向”的整体。
	\end{description}
\end{trapbox}
