\begin{trapbox}

	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略定义域）}}]
			使用等价关系时，直接取对数而忘记检验 $x+a>0$ 是否成立。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义域优先）：}}]
			必须先确认 $x+a>0$，否则对数无意义，等价关系失效。
		\item[\textbf{\textcolor{CTrap}{错因分析（基础不牢）：}}]
			对数的真数必须大于0，这是函数定义域要求，不可忽略。
	\end{description}

	\medskip
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（条件误扩）}}]
			错误认为 $e^{x}(x+a)=k$ 和 $\ln(x+a)=\ln k - x$ 等价时不需要 $k>0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（参数限制）：}}]
			只有 $k>0$ 时才能取对数，且需 $x+a>0$；若 $k\leq0$ 则无对数意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（推理不足）：}}]
			取对数默认要求等式右边大于0，不能自动省略。
	\end{description}

	\medskip
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（逆用混乱）}}]
			由 $\ln(x+a)=-x$ 推回 $e^{x}(x+a)=1$ 时，忘记 $e^{\ln(x+a)}=x+a$ 成立的条件仍是 $x+a>0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（逆用前提）：}}]
			从对数式反推指数式时，同样要求 $x+a>0$，否则指数后无法恢复原式。
		\item[\textbf{\textcolor{CTrap}{错因分析（对称注意）：}}]
			等价关系是双向的，条件必须同时满足两个方向。
	\end{description}

\end{trapbox}
