\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{指数乘积式 $e^{x}(x+a)=1$ 与对数负值式 $\ln(x+a)=-x$ 在 $x+a>0$ 时等价，可互相代换。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*,itemsep=0.5em,topsep=0.4em]
		\item \textbf{条件 1（定义域约束）：} $x+a>0$ 是前提，保证对数有意义。
		\item \textbf{条件 2（等式形式）：} 形如 $e^{x}(x+a)=1$ 或 $\ln(x+a)=-x$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*,itemsep=0.5em,topsep=0.4em]
		\item \textbf{易错点（定义域先行）：} 使用前必须验证 $x+a>0$。
		\item \textbf{检查项（常数匹配）：} 若等式右边不是1而是其他正数 $k$，应调整为 $\ln(x+a)=\ln k - x$。
	\end{itemize}

\end{summarybox}
