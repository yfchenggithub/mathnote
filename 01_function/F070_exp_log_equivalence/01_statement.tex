\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（参数条件）：}}]
			设常数 $a\in\mathbb{R}$，实数 $x$ 满足 $x+a>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（等价代换）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			指数乘积等式可转化为对数负值等式。
			\[
				e^{x}(x+a)=1 \quad\Longleftrightarrow\quad \ln(x+a)=-x.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（参数化形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			常数 $1$ 换成任意正数 $k$，等式形式保持不变。
			\[
				e^{x}(x+a)=k\;(k>0) \quad\Longleftrightarrow\quad \ln(x+a)=\ln k - x.
			\]
	\end{description}

\end{statementbox}
