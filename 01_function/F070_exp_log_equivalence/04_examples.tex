\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}已知 $a=2$，$x>-2$，且 $e^{x}(x+2)=1$。求 $\ln(x+2)+x$ 的值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}]
			条件满足 $e^{x}(x+a)=1$ 形式，且 $x+a>0$。
			\par\textbf{理由：}可直接应用等价关系。
			\[
				e^{x}(x+2)=1,\quad x+2>0.
			\]
		\item[\textbf{第二步（应用结论）：}]
			由等价关系得 $\ln(x+2)=-x$。
			\par\textbf{理由：}套用 $\ln(x+a)=-x$。
			\[
				\ln(x+2)=-x.
			\]
		\item[\textbf{第三步（计算目标）：}]
			计算 $\ln(x+2)+x = -x + x = 0$。
			\par\textbf{理由：}代入第二步结果。
			\[
				\ln(x+2)+x = 0.
			\]
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\ln(x+2)+x=0$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}设 $f(x)=e^{x}(x+1)-\ln(x+1)-x$，$x>-1$。已知 $f(x)$ 在 $x=x_0$ 处取极值，且满足 $e^{x_0}(x_0+1)=1$。求 $f(x_0)$ 的值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}]
			极值点满足 $e^{x_0}(x_0+1)=1$ 且 $x_0+1>0$。
			\par\textbf{理由：}条件符合等价关系。
			\[
				e^{x_0}(x_0+1)=1.
			\]
		\item[\textbf{第二步（应用结论）：}]
			由等价关系得 $\ln(x_0+1)=-x_0$。
			\par\textbf{理由：}代入结论。
			\[
				\ln(x_0+1) = -x_0.
			\]
		\item[\textbf{第三步（代入求值）：}]
			$f(x_0)=1 - \ln(x_0+1) - x_0 = 1 - (-x_0) - x_0 = 1$。
			\par\textbf{理由：}将 $e^{x_0}(x_0+1)=1$ 和 $\ln(x_0+1)=-x_0$ 代入 $f$ 表达式。
			\[
				f(x_0)=1.
			\]
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$f(x_0)=1$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}证明：当 $x>-1$ 时，$e^{x}(x+1) \ge \ln(x+1)+x+1$，并求等号成立条件。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（构造函数）：}]
			设 $g(x)=e^{x}(x+1)-\ln(x+1)-x-1$，则原不等式等价于 $g(x)\ge 0$。
			\par\textbf{理由：}构造函数，将不等式转化为极值问题。
			\[
				g(x)=e^{x}(x+1)-\ln(x+1)-x-1,\quad x>-1.
			\]
		\item[\textbf{第二步（求导变形）：}]
			计算 $g'(x)=e^{x}(x+2)-\frac{1}{x+1}-1$。令 $g'(x)=0$，化简得 $e^{x}(x+1)=1$。
			\par\textbf{理由：}移项通分，并约去正因子 $x+2$。
			\[
				g'(x)=0 \Rightarrow e^{x}(x+2) = \frac{1}{x+1}+1.
			\]
			\[
				\Rightarrow e^{x}(x+2) = \frac{x+2}{x+1}.
			\]
			\[
				\Rightarrow e^{x} = \frac{1}{x+1}.
			\]
			\[
				\Rightarrow e^{x}(x+1) = 1.
			\]
		\item[\textbf{第三步（隐元代换）：}]
			设 $x_0$ 满足 $g'(x_0)=0$，则 $e^{x_0}(x_0+1)=1$。由等价关系得 $\ln(x_0+1)=-x_0$。
			\par\textbf{理由：}直接应用等价关系。
			\[
				e^{x_0}(x_0+1)=1 \;\Longleftrightarrow\; \ln(x_0+1)=-x_0.
			\]
		\item[\textbf{第四步（极值计算）：}]
			计算 $g(x_0)=1 - \ln(x_0+1) - x_0 -1 = 1 - (-x_0) - x_0 -1 = 0$。
			\par\textbf{理由：}代入 $e^{x_0}(x_0+1)=1$ 和 $\ln(x_0+1)=-x_0$。
			\[
				g(x_0)=0.
			\]
		\item[\textbf{第五步（单调性分析）：}]
			通过 $g''(x)=e^{x}(x+3)+\frac{1}{(x+1)^2}>0$ 知 $g'(x)$ 在 $x>-1$ 上严格单调递增，故 $g'(x)$ 有唯一零点 $x_0$；且 $x<x_0$ 时 $g'(x)<0$，$x>x_0$ 时 $g'(x)>0$，所以 $x_0$ 为 $g(x)$ 的极小值点，也是最小值点。因此 $g(x)\ge g(x_0)=0$。
			\par\textbf{理由：}二阶导恒正，一阶导单调，极值唯一为最小值。
			\[
				g''(x)=e^{x}(x+3)+\frac{1}{(x+1)^2}>0.
			\]
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	原不等式成立，等号成立当且仅当 $e^{x}(x+1)=1$（$x>-1$）。
\end{examplebox}
