\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{连续严格单调函数在闭区间端点异号 $\Longleftrightarrow$ 方程有唯一实根。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（端点异号）：} $f(m)f(n)<0$。
		\item \textbf{条件 2（连续严格单调）：} $f$ 在 $[m,n]$ 上连续且严格单调。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（双条件缺一不可）：} 必须同时满足连续性和严格单调性，零点定理只保证存在，单调保证唯一。
		\item \textbf{检查项（确保乘积为负）：} 计算 $f(m)f(n)$ 时，必须严格小于零，等于零时结论不适用。
	\end{itemize}

\end{summarybox}
