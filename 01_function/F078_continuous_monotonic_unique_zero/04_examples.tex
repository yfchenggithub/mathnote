\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	判断函数 $f(x)=x^3-3x+1$ 在区间 $[0,1]$ 上是否有唯一零点？若有，请说明理由。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判断连续单调）：}]
			$f$ 是多项式，在 $[0,1]$ 上连续。求导 $f'(x)=3x^2-3=3(x^2-1)$，在 $[0,1]$ 上 $x^2-1\le 0$，且仅在 $x=1$ 时为零，故 $f$ 在 $[0,1]$ 上严格单调递减。
			\par\textbf{理由：} 导函数在区间内非正且不恒为零，函数严格递减。
		\item[\textbf{第二步（计算端点符号）：}]
			计算 $f(0)=1>0$，$f(1)=1-3+1=-1<0$。所以 $f(0)f(1)=-1<0$。
			\par\textbf{理由：} 代入端点计算，乘积为负，符合异号条件。
		\item[\textbf{第三步（套用结论得根）：}]
			由结论，因为 $f$ 在 $[0,1]$ 上连续且严格单调递减，且 $f(0)f(1)<0$，所以方程 $f(x)=0$ 在 $(0,1)$ 内存在唯一实根。
			\par\textbf{理由：} 直接应用充要条件。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{在 $(0,1)$ 内有唯一实根。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=\ln x - 2$，判断方程 $f(x)=0$ 在哪个整数区间 $(k,k+1)$ 内有唯一实根，并说明理由。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判断连续单调）：}]
			$f(x)=\ln x-2$ 在 $(0,+\infty)$ 上连续，且 $\ln x$ 严格单调递增，故 $f$ 在正区间上严格单调递增。
			\par\textbf{理由：} 对数函数 $\ln x$ 底 $e>1$，单调递增。
		\item[\textbf{第二步（寻找异号整数对）：}]
			计算 $f(7)=\ln7-2\approx -0.054<0$，$f(8)=\ln8-2\approx 0.079>0$。所以 $f(7)f(8)<0$。
			\par\textbf{理由：} 通过估值找到相邻整数使函数值异号。
		\item[\textbf{第三步（套用整数推论）：}]
			由推广结论，$f$ 在 $[7,8]$ 上连续且严格单调递增，且 $f(7)f(8)<0$，故方程在 $(7,8)$ 内有唯一实根，且该开区间内无整数根。
			\par\textbf{理由：} 整数区间推论直接应用。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{方程在 $(7,8)$ 内有唯一实根。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=x^3-3x+a$ 在区间 $[0,1]$ 上严格单调递减。若方程 $f(x)=0$ 在 $(0,1)$ 内有唯一实根，求实数 $a$ 的取值范围。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证单调条件）：}]
			$f'(x)=3x^2-3=3(x^2-1)$，在 $[0,1]$ 上 $x^2-1\le 0$，且仅在 $x=1$ 时为零，因此 $f$ 在 $[0,1]$ 上严格单调递减。题目条件已给定，实际可验证。
			\par\textbf{理由：} 导数非正且不恒为零确保严格单调递减。
		\item[\textbf{第二步（套用充要条件）：}]
			由结论，在连续严格单调的条件下，$(0,1)$ 内有唯一实根等价于 $f(0)f(1)<0$。
			\par\textbf{理由：} 充要条件直接用于参数确定。
		\item[\textbf{第三步（解不等式求范围）：}]
			$f(0)=a$，$f(1)=1-3+a=a-2$。因此 $a(a-2)<0$，解得 $0<a<2$。
			\par\textbf{理由：} 乘法不等式转化为一元二次不等式，解集为 $0<a<2$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a$ 的取值范围是 $(0,2)$。}
\end{examplebox}
