\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{极值点的充要条件是导数在驻点两侧变号，极值点个数等于变号根个数。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（函数可导）：} 函数在区间上可导。
		\item \textbf{条件 2（存在驻点）：} 找到导数为零的点。
		\item \textbf{条件 3（变号检验）：} 验证每个驻点两侧导数异号。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（驻点不一定）：} 不是所有驻点都是极值点。
		\item \textbf{检查项（变号验证）：} 对于每个驻点必须验证导数左右符号。
	\end{itemize}
\end{summarybox}
