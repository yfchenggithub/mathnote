\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（函数可导）：}}] 函数 $f(x)$ 在区间 $D$ 上可导，$x_0$ 为方程 $f'(x)=0$ 的实根。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（极值点判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			极值点等价于导数在两侧异号。
			\[
				\exists\delta>0,\ \forall x_1\in (x_0-\delta,x_0),\ \forall x_2\in (x_0,x_0+\delta),\ f'(x_1)f'(x_2)<0.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（极值点计数）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			极值点个数等于变号根的个数。
			\[
				\begin{aligned}
					&\bigl|\{x_0\in D\mid f'(x_0)=0,\exists\delta>0,\\
					&\qquad \forall x_1\in (x_0-\delta,x_0),\forall x_2\in (x_0,x_0+\delta), f'(x_1)f'(x_2)<0\}\bigr|.
			\end{aligned}
			\]
	\end{description}
\end{statementbox}
