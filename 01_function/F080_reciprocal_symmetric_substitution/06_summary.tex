\begin{summarybox}
    \textbf{\textcolor{CSummary}{一句话核心}}
    \textbf{倒数配对先整体换元，高次变低次；每次必须带上新变量范围并回代检验。}

    \medskip
    \textbf{\textcolor{CSummary}{核心公式（$t\ne0$ 为实数）}}
    \begin{itemize}[leftmargin=*, itemsep=0.52em, topsep=0.35em]
      \item \textbf{和型：} $u=t+t^{-1}$，$|u|\ge2$。
        \[
          t^2+t^{-2}=u^2-2,\quad
          t^3+t^{-3}=u^3-3u,\quad
          t^4+t^{-4}=u^4-4u^2+2.
        \]
      \item \textbf{差型：} $v=t-t^{-1}\in\mathbb R$，$t^2+t^{-2}=v^2+2$，$t^3-t^{-3}=v^3+3v$。
      \item \textbf{仅偶次：} 令 $z=t^2+t^{-2}\ge2$，则 $t^4+t^{-4}=z^2-2$。
      \item \textbf{高次递推：} $S_0=2$、$S_1=u$、$S_{n+1}=uS_n-S_{n-1}$（$n\ge1$）。
      \item \textbf{另一类整体（三角和、积）：} 在独立的三角题中令
        $t=\sin\theta+\cos\theta$，则
        \[
        t^2=1+2\sin\theta\cos\theta,\quad
        (1+\sin\theta)(1+\cos\theta)=\frac{(1+t)^2}{2}.
        \]
        当 $0\le\theta\le\pi/2$ 时 $1\le t\le\sqrt2$，
        $t=\sqrt2$ 当且仅当 $\theta=\pi/4$。此处 $t$ 与倒数型原变量是不同题目的记号。
    \end{itemize}

    \medskip
    \textbf{\textcolor{CSummary}{考场识别与操作}}
    \begin{itemize}[leftmargin=*, itemsep=0.46em, topsep=0.35em]
      \item \textbf{识别：} 同次正负幂成对、系数完整首尾对称、两项乘积为 $1$。
      \item \textbf{迁移识别：} 看见 $\sin\theta+\cos\theta$ 与
        $\sin\theta\cos\theta$ 同时出现，想到
        $\sin^2\theta+\cos^2\theta=1$；\textbf{先展开，后整体换元}。
      \item \textbf{换元：} 优先选重复最多、次数最低的整个式子；有时 $z=t^2+t^{-2}$ 比 $u=t+t^{-1}$ 更快。
      \item \textbf{范围：} $u$ 不能取 $(-2,2)$；$v$ 可以取所有实数；$z\ge2$。
      \item \textbf{收尾：} 和型回代 $t^2-ut+1=0$，差型回代 $t^2-vt-1=0$；最值必须验证等号能取到。
    \end{itemize}

    \medskip
    \textbf{\textcolor{CSummary}{最小反例与检查项}}
    \begin{itemize}[leftmargin=*, itemsep=0.43em, topsep=0.35em]
      \item \textbf{错把范围扩大：} $u=1$ 不可能等于任何非零实数的 $t+1/t$。
      \item \textbf{错把和当差：} $t=1$ 时 $t-1/t=0$，所以差型变量没有 $|v|\ge2$ 的限制。
      \item \textbf{错判符号：} $t=2$ 与 $t=1/2$ 的 $u$ 相同，但 $t^2-t^{-2}$ 符号相反。
      \item \textbf{联系抛物线题：} 距离平方化成 $16z(z-3)$、$z\ge4$，故最小距离为 $8$，不是 $0$。
      \item \textbf{2023 抛物线面积题：} 用三角整体
        $t=\cos\theta+\sin\theta\in[1,\sqrt2]$，
        得 $S\ge4/(1+t)^2\ge12-8\sqrt2$；必须先证明没有漏掉射线朝右的情况。
      \item \textbf{三角换元易错：} 不应凭空跳出平方形式；
        $\theta=5\pi/4$ 时 $\sin\theta+\cos\theta=-\sqrt2$，
        说明 $t\ge1$ 只适用于本节规定的第一象限角。
      \item \textbf{数值核对：} $t^4+t^{-4}-3(t^2+t^{-2})$ 的最小值是 $-4$（$t=\pm1$）。
    \end{itemize}
\end{summarybox}
