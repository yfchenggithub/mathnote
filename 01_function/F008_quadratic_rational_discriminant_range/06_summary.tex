\begin{summarybox}
  \textbf{\textcolor{CSummary}{一句话记住}}

  \medskip
  \textbf{求 $y$ 能取哪些值，就先问这些 $y$ 能不能找到对应的 $x$。}让分式等于 $y$，转成关于 $x$ 的方程；有实根，才说明这个 $y$ 能取到。

  \medskip
  \textbf{\textcolor{CSummary}{四步答题顺序}}
  \begin{enumerate}[leftmargin=*,itemsep=0.35em]
    \item \textbf{查分母：}优先处理 $dx^2+ex+f\ne0$；若 $e^2-4df<0$，任何实数 $x$ 都能用。
    \item \textbf{列方程：}
    \[
      (a-dy)x^2+(b-ey)x+(c-fy)=0.
    \]
    \item \textbf{算范围：}当 $a-dy\ne0$ 时，解
    \[
      \boxed{(b-ey)^2-4(a-dy)(c-fy)\ge0.}
    \]
    \item \textbf{查漏项：}把 $y=\dfrac ad$ 单独代入，判断方程有没有实数解。
  \end{enumerate}

  \medskip
  \textbf{\textcolor{CSummary}{最重要的提醒}}
  \begin{itemize}[leftmargin=*,itemsep=0.3em]
    \item \textbf{$\Delta\ge0$ 不等于什么都不用查：}只对二次项确实存在时直接有效。
    \item \textbf{极限靠近的数不一定取到：}尤其要查 $y=\dfrac ad$。
    \item \textbf{能拆分就先拆分：}如 $\dfrac{2x^2+3}{x^2+1}=2+\dfrac1{x^2+1}$，比套公式更快。
    \item \textbf{遇到最大、最小值：}还要确认等号何时成立，才能决定区间端点加不加括号。
  \end{itemize}

  \medskip
  \textbf{\textcolor{CSummary}{考场速记}}
  \[
    \boxed{\text{先查分母}\ \longrightarrow\ \text{去分母}\ \longrightarrow\ \Delta(y)\ge0\ \longrightarrow\ \text{单查 }y=\tfrac ad.}
  \]

\begin{summarybox}
\textbf{\textcolor{CSummary}{数形结合速记：先找到图形，再求范围}}
  \begin{itemize}[leftmargin=*,itemsep=0.4em]
    \item \textbf{抛物线法：}令 $X=dx^2+ex+f$、$Y=ax^2+bx+c$，
      原函数值就是 $Y/X$，即原点到动点 $(X,Y)$ 的连线斜率。
      对例 2，动点满足 $(X-Y-1)^2=4(Y-1)$，边界斜率来自两条切线。
    \item \textbf{单位圆法：}对例 2 令 $D=x^2+2x+2$，由
      $D^2-(D-2)^2=(2x+2)^2$ 得到
      \[
        u=\frac{D-2}{D},\quad v=\frac{2x+2}{D},\quad
        u^2+v^2=1,\quad y=\frac32-\frac12u-v.
      \]
      通过圆与直线相交、相切求范围；但 $(1,0)$ 并不能由有限实数 $x$ 到达。
    \item \textbf{端点检查永远不能省：}抛物线法要查 $a-dy=0$；
      圆法要查相切点是否真能由原来的 $x$ 得到。
  \end{itemize}

  \textbf{选择顺序：}能拆分就先拆分；需要通用快速计算时用判别式；
  要讲清楚\textbf{“判别式为何等于相切”}，优先讲抛物线斜率；
  想进一步体验勾股定理如何联系函数值域，再讲单位圆。
\end{summarybox}

\begin{summarybox}
\textbf{\textcolor{CSummary}{F008 七种主方法速览}}

\begin{enumerate}[leftmargin=*,itemsep=0.5em]
  \item \textbf{判别式法（通用计算）：}固定 $y$，看 $(a-dy)x^2+(b-ey)x+(c-fy)=0$ 有没有实根；单查 $y=a/d$。
  \item \textbf{两条抛物线比较法（看见相切）：}把分子 $P(x)$ 与 $y$ 倍的分母 $yQ(x)$ 画在同一坐标系里；相交则取到该 $y$，边界常在相切时出现。
  \item \textbf{距离平方比法（勾股定理）：}若分子分母可配成正的平方和，把它们看作动点到两个固定点的距离平方，再看对应的圆能否碰到动点所在的直线。
  \item \textbf{抛物线斜率法（原点连线）：}令 $X=Q(x),Y=P(x)$，求 $Y/X$，即从原点引线可取的斜率。
  \item \textbf{单位圆法（平方差）：}先从原式找出真正的平方和恒等式，再化成 $u^2+v^2=1$；必须查圆上有没有无法到达的点。
  \item \textbf{倒数对称法（特殊提速）：}本题 $F(-x-1)=1/F(x)$，所以求得一端就能求另一端，但它本身不能决定端点数值。
  \item \textbf{导数法（高中常规）：}看导数从正变负或从负变正的位置，结合定义域与远处函数的变化，确定全局最值。
\end{enumerate}

\medskip
\textbf{\textcolor{CSummary}{本题共同答案}}
\[
 F(x)=\frac{x^2+1}{x^2+2x+2},\qquad
 \boxed{F(x)\in\left[\frac{3-\sqrt5}{2},\frac{3+\sqrt5}{2}\right].}
\]
两端分别在 $x=(\sqrt5-1)/2$、$x=-(\sqrt5+1)/2$ 取到。

\medskip
\textbf{选择建议：}考试优先用1.；要解释“判别式与相切”优先用2.、4.；
想从勾股定理自然得到图形，可用3.；5.适合进一步深化；6.用于快速核对；7.用于已学导数后的对照。
原有\textbf{拆分法}与\textbf{平方验证法}也继续保留，作为不必硬套七法的简便技巧。

\textbf{最后都要检查：}分母是否为 $0$、二次项是否消失、图形端点能否真的达到。
\end{summarybox}
