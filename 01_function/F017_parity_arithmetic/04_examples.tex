\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1：奇函数相加}}

	设 $f(x)=x^3$，$g(x)=x$，求 $h(x)=f(x)+g(x)$ 的奇偶性。

	两个函数的定义域都是 $\mathbb{R}$，且
	\[
		f(-x)=-f(x),\qquad g(-x)=-g(x).
	\]
	因此 $f$、$g$ 都是奇函数，奇加奇仍为奇。
	直接检验也可得到
	\[
		h(-x)=(-x)^3+(-x)=-(x^3+x)=-h(x).
	\]
	故 $h(x)=x^3+x$ 是奇函数。

	\medskip
	\textbf{\textcolor{CExample}{例 2：偶函数相乘}}

	设 $f(x)=x^2$，$g(x)=\cos x$，求
	$\varphi(x)=f(x)g(x)$ 的奇偶性。

	两个函数的定义域都是 $\mathbb{R}$。
	由 $(-x)^2=x^2$ 和 $\cos(-x)=\cos x$，
	可知 $f$、$g$ 都是偶函数。于是
	\[
		\varphi(-x)=(-x)^2\cos(-x)
		=x^2\cos x
		=\varphi(x).
	\]
	故 $\varphi(x)=x^2\cos x$ 是偶函数。

	\medskip
	\textbf{\textcolor{CExample}{例 3：先乘后加}}

	设 $f(x)=x^3$、$g(x)=x^2$，
	$h(x)=f(x)g(x)$，$k(x)=h(x)+f(x)$。

	$f$ 为奇函数，$g$ 为偶函数，所以
	$h=fg$ 为奇函数。再由奇加奇为奇，
	得到 $k=h+f$ 为奇函数。也可直接写出
	\[
		k(x)=x^5+x^3,\qquad
		k(-x)=-x^5-x^3=-k(x).
	\]

	\medskip
	\textbf{\textcolor{CExample}{例 4：除法要检查定义域}}

	设 $q(x)=x/x$。原式仅在 $x\ne0$ 时有定义。
	在这个关于原点对称的定义域上，
	\[
		q(-x)=\frac{-x}{-x}=1=q(x).
	\]
	故 $q$ 是偶函数。约分得到 $1$ 后，
	仍须保留原式的限制 $x\ne0$。
\end{examplebox}
