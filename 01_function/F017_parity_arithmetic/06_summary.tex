\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{奇偶函数的四则运算规则：加减法仅同奇同偶时保持原奇偶；乘除法类似正负数符号运算（同奇得偶、奇偶得奇、同偶得偶），除法需分母非零。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（定义域对称）：} 两函数定义域均关于原点对称。
		\item \textbf{条件 2（运算可行）：} 运算在公共定义域上进行，除法时分母恒不为零。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（前提忽略）：} 容易忽略定义域对称性和分母为零，导致错误判定。
		\item \textbf{检查项（符号核对）：} 使用乘法规则时，注意核对符号是否记忆正确（如奇$\times$奇$=$偶）。
	\end{itemize}

\end{summarybox}
