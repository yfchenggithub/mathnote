\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（定义域忽略）}}]
			判断函数奇偶性前，未检查定义域是否关于原点对称，直接套用运算规则。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义域条件）：}}]
			运算前必须确保两个函数的定义域均关于原点对称，且运算在公共定义域上进行。
		\item[\textbf{\textcolor{CTrap}{错因分析（前提遗漏）：}}]
			学生只关注运算规则本身，忘记奇偶函数定义中的定义域对称性前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分母为零）}}]
			应用除法规则时，忽视分母函数恒不为零的条件，直接套用乘法规则。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母非零）：}}]
			除法运算要求分母函数在公共定义域内恒不为零，否则分式无意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（粗心忽略）：}}]
			受乘法规则影响，默认除法只需类比符号运算，未考虑分母为零的约束。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号记反）}}]
			记忆乘法规则时出现错误，例如认为“奇函数乘奇函数得奇函数”。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确规则）：}}]
			奇$\times$奇$=$偶，奇$\times$偶$=$奇，偶$\times$偶$=$偶；乘法规则类比正负数乘积。
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性思维）：}}]
			学生将正数乘法习惯带入：正正得正对应奇奇得奇？实际奇偶规则相反，需要专门记忆。
	\end{description}

\end{trapbox}
