\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	证明的基本方法是根据奇偶函数的定义，将目标函数中的自变量 $x$ 替换为 $-x$，并利用已知函数的奇偶性进行化简，最终判断结果与 $F(x)$ 的关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（加减规则证明）：}]
			设 $f,g$ 均为奇函数则 $(f\pm g)(-x)=-(f\pm g)(x)$，均为偶函数则 $(f\pm g)(-x)=(f\pm g)(x)$。公式如下：
			\[
				\begin{aligned}
					&f(-x)=-f(x),\; g(-x)=-g(x)\;\Rightarrow\;(f\pm g)(-x)=-(f\pm g)(x),\\&f(-x)=f(x),\; g(-x)=g(x)\;\Rightarrow\;(f\pm g)(-x)=(f\pm g)(x).
			\end{aligned}
			\]
			\par\textbf{理由：} 直接利用奇偶函数定义代入并提取公因式。
		\item[\textbf{步骤二（乘法规则证明）：}]
			设 $f,g$ 奇偶性如下，则乘积性质：
			\[
				\begin{aligned}
					&f(-x)=-f(x),\; g(-x)=-g(x)\;\Rightarrow\;(fg)(-x)\\
					&\qquad = f(x)g(x)\\
					&\qquad = (fg)(x),\
					\[
						2mm]&f(-x)=-f(x),\; g(-x)=g(x)\;\Rightarrow\;(fg)(-x)\\
						&\qquad = -f(x)g(x)\\
						&\qquad = -(fg)(x),\
						\[
							2mm]&f(-x)=f(x),\; g(-x)=g(x)\;\Rightarrow\;(fg)(-x)\\
							&\qquad = f(x)g(x)\\
							&\qquad = (fg)(x).
					\end{aligned}
					\]
					\par\textbf{理由:} 根据乘法交换性和奇偶定义逐项化简.
				\item[\textbf{步骤三(除法规则证明):}]
					在分母 $g(x)\neq0$ 的前提下,除法规则与乘法规则一致:
					\[
						\begin{aligned}
							&g(x)\neq0,\; f(-x)=-f(x),\; g(-x)=-g(x)\;\Rightarrow\;\left(\frac{f}{g}\right)(-x)\\
							&\qquad = \frac{f(x)}{g(x)}\\
							&\qquad = \left(\frac{f}{g}\right)(x),\
							\[
								2mm]&g(x)\neq0,\; f(-x)=-f(x),\; g(-x)=g(x)\;\Rightarrow\;\left(\frac{f}{g}\right)(-x)\\
								&\qquad = -\frac{f(x)}{g(x)}\\
								&\qquad = -\left(\frac{f}{g}\right)(x),\
								\[
									2mm]&g(x)\neq0,\; f(-x)=f(x),\; g(-x)=g(x)\;\Rightarrow\;\left(\frac{f}{g}\right)(-x)\\
									&\qquad = \frac{f(x)}{g(x)}\\
									&\qquad = \left(\frac{f}{g}\right)(x).
							\end{aligned}
							\]
							\par\textbf{理由:} 除法化为乘法,分母非零保证运算有意义.
					\end{description}

					\medskip
					\textbf{\textcolor{CProof}{结论回扣}}
					\textbf{以上推导验证了奇偶函数在四则运算下的结果,说明运算规则完全由奇偶定义通过代数变形得到,便于快速应用.}
				\end{proofbox}
