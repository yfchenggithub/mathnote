\begin{statementbox}
\textbf{\textcolor{CStatement}{核心结论一：交换横纵坐标}}
\input{assets/tikz/F095_point}

\textbf{一句话记住：}关于 $y=x$ 对称，横纵坐标互换。
\[
\boxed{P(a,b)\longleftrightarrow P'(b,a)}.
\]
任意曲线 $F(x,y)=0$ 的镜像方程为
\[
\boxed{F(y,x)=0}.
\]
原方程若含范围限制，也要一并交换。例如 $y=x^2,\ x\ge0$ 的镜像是 $x=y^2,\ y\ge0$，即 $y=\sqrt{x}$。

\textbf{函数图象：} $y=f(x)$ 的镜像为 $x=f(y)$。若 $f:D\to E$ 一一对应（$E=f(D)$），镜像才是反函数 $y=f^{-1}(x)$。
\end{statementbox}

\begin{statementbox}
\textbf{\textcolor{CStatement}{核心结论二：反函数与自身对称}}
\input{assets/tikz/F095_self}

若 $f:D\to E$ 一一对应，则
\[
\begin{aligned} f^{-1}(f(x))&=x\quad(x\in D),\\ f(f^{-1}(y))&=y\quad(y\in E).\end{aligned}
\]
原函数定义域与值域在反函数中互换。若 $f(D)\subseteq D$ 且对任意 $x\in D$ 有 $f(f(x))=x$，则 $f(D)=D$，且原函数自身就是反函数。
\[
\boxed{y=f(x)\text{ 自身关于 }y=x\text{ 对称}}
\quad\Longleftrightarrow\quad
\boxed{f(D)\subseteq D,\ f(f(x))=x\ (x\in D)}.
\]
典型模型：$y=x$、$y=b-x$、$y=k/x$（$k\ne0$）、$y=a+k/(x-a)$（$x\ne a$）。若严格递增且自身对称，则只能有 $f(x)=x$。
\end{statementbox}
\clearpage
\begin{statementbox}
\textbf{\textcolor{CStatement}{核心结论三：交点、斜率与单调性}}
\input{assets/tikz/F095_intersection}

\begin{itemize}[leftmargin=*,itemsep=.4em]
\item 若 $f$ 在 $D$ 上严格递增，原函数与反函数的公共点必在 $y=x$ 上，故交点只需求 $f(x)=x$（$x\in D$）。
\item 严格递减时不能直接使用上述交点结论；公共点满足 $x\in D,\ f(x)\in D,\ f(f(x))=x$。
\item 非水平、非竖直直线斜率 $k$ 的镜像斜率为 $1/k$。水平线和竖直线互换。
\item 严格递增（递减）的可逆函数，其反函数仍严格递增（递减）。
\item 若 $f$ 在 $a$ 附近的区间上可逆，且在 $a$ 可导、$f'(a)\ne0$，反函数在 $b=f(a)$ 处连续，则
$\displaystyle (f^{-1})'(b)=1/f'(a)$（导数按相应定义域理解）。
\end{itemize}
\end{statementbox}
