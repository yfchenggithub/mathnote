\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（由一个周期推出一组对称轴）}}

	\textbf{\textcolor{CExample}{题目：}}
	设定义在 $\mathbb{R}$ 上的函数 $f(x)$ 以 $4$ 为一个周期，且 $x=1$ 是它的一条对称轴。写出由这些条件必然能够推出的一组对称轴。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取条件）：}]
		      已知
		      \[
			      T=4,
			      \qquad
			      a=1.
		      \]

		\item[\textbf{第二步（应用连锁结论）：}]
		      由
		      \[
			      x=a+\frac{kT}{2},
			      \quad k\in\mathbb{Z},
		      \]
		      得
		      \[
			      x=1+\frac{4k}{2}=1+2k,
			      \quad k\in\mathbb{Z}.
		      \]

		\item[\textbf{第三步（辨析结论范围）：}]
		      因为题目只说 $4$ 是一个周期，没有说明它是最小正周期，所以只能说
		      \[
			      x=1+2k,
			      \quad k\in\mathbb{Z},
		      \]
		      是一组必然存在的对称轴，不能断言它们就是全部对称轴。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{x=1+2k,\quad k\in\mathbb{Z}}
	\]
	均为对称轴。

	\textbf{\textcolor{CExample}{例 2（对称中心连锁）}}

	\textbf{\textcolor{CExample}{题目：}}
	若函数 $f(x)$ 在 $\mathbb{R}$ 上以 $\pi$ 为一个周期，且关于点
	\[
		\left(\frac{\pi}{6},3\right)
	\]
	中心对称，写出由这些条件必然能够推出的一组对称中心。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取条件）：}]
		      \[
			      T=\pi,
			      \qquad
			      (b,c)=\left(\frac{\pi}{6},3\right).
		      \]

		\item[\textbf{第二步（应用连锁结论）：}]
		      由
		      \[
			      \left(b+\frac{kT}{2},\,c\right),
			      \quad k\in\mathbb{Z},
		      \]
		      代入可得
		      \[
			      \left(
			      \frac{\pi}{6}+\frac{k\pi}{2},
			      \,3
			      \right),
			      \quad k\in\mathbb{Z}.
		      \]

		\item[\textbf{第三步（辨析结论范围）：}]
		      题目只给出一个周期 $\pi$，所以这里得到的是一组必然存在的对称中心，不应直接称为全部对称中心。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{
			\left(
			\frac{\pi}{6}+\frac{k\pi}{2},
			\,3
			\right),
			\quad k\in\mathbb{Z}
		}
	\]
	均为对称中心。

	\textbf{\textcolor{CExample}{例 3（两条对称轴推出一个周期）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定义在 $\mathbb{R}$ 上的函数 $f(x)$ 有两条不同的对称轴
	\[
		x=-1
		\quad\text{和}\quad
		x=3.
	\]
	证明 $f$ 是周期函数，并求出它的一个正周期。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算轴间距）：}]
		      \[
			      |3-(-1)|=4.
		      \]

		\item[\textbf{第二步（两倍间距得到周期）：}]
		      \[
			      P=2\times4=8.
		      \]

		\item[\textbf{第三步（直接验证）：}]
		      由 $x=-1$ 为对称轴，
		      \[
			      f(t)=f(-2-t).
		      \]

		      再利用 $x=3$ 为对称轴，
		      \[
			      f(-2-t)
			      =f\bigl(6-(-2-t)\bigr)
			      =f(t+8).
		      \]

		      因此
		      \[
			      f(t+8)=f(t)
		      \]
		      对任意 $t\in\mathbb{R}$ 成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{8\text{ 是 }f\text{ 的一个正周期}}
	\]
	但不能仅凭这两条对称轴断定 $8$ 是最小正周期。

	\textbf{\textcolor{CExample}{例 4（为什么不能轻易写“全部”）}}

	\textbf{\textcolor{CExample}{题目：}}
	函数
	\[
		f(x)=\cos x
	\]
	以 $4\pi$ 为一个周期，且 $x=0$ 是一条对称轴。

	利用连锁结论可以推出哪些对称轴？这些是否为全部对称轴？

	\textbf{\textcolor{CExample}{解答：}}

	由 $T=4\pi$、$a=0$，可推出
	\[
		x=0+\frac{k\cdot4\pi}{2}
		=2k\pi,
		\quad k\in\mathbb{Z},
	\]
	均为对称轴。

	但 $\cos x$ 的最小正周期是 $2\pi$，它的全部对称轴为
	\[
		x=k\pi,
		\quad k\in\mathbb{Z}.
	\]

	因此，用一个非最小正周期得到的只是部分对称轴，不能写成全部对称轴。

	\textbf{\textcolor{CExample}{例 5（最小正周期下确定全部对称轴）}}

	\textbf{\textcolor{CExample}{题目：}}
	设 $T_0=6$ 是函数 $f$ 的最小正周期，且 $x=1$ 是函数图像的一条对称轴。求函数图像的全部对称轴。

	\textbf{\textcolor{CExample}{解答：}}

	由最小正周期下的完整结论，全部对称轴恰为
	\[
		x=1+\frac{kT_0}{2}
		=1+3k,
		\quad k\in\mathbb{Z}.
	\]

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{x=1+3k,\quad k\in\mathbb{Z}}
	\]
	是函数图像的全部对称轴。
