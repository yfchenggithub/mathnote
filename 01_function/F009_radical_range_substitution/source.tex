% F009 换元法求根式函数的值域：简要结论（供内容生产端读取）
% 换元时必须保留新变量的准确范围。
\[
 t=\sqrt{mx+n}\ge0\quad(m\ne0),\qquad x=\frac{t^2-n}{m}.
\]
\[
 x=a\sin\theta,\quad -\frac{\pi}{2}\le\theta\le\frac{\pi}{2},\quad
 \sqrt{a^2-x^2}=a\cos\theta\qquad(a>0).
\]
\[
 x=a\tan\theta,\quad -\frac{\pi}{2}<\theta<\frac{\pi}{2},\quad
 \sqrt{x^2+a^2}=\frac{a}{\cos\theta}\qquad(a>0).
\]
\[
 \sqrt{x^2+a^2}-x=\frac{a^2}{\sqrt{x^2+a^2}+x}\qquad(a>0).
\]
\[
 x-\sqrt{x^2-a^2}=\frac{a^2}{x+\sqrt{x^2-a^2}}
 \qquad(x\ge a>0).
\]
