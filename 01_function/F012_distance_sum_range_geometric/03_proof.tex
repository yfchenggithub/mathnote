\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	构造点 $A(m,-a)$、$B(n,b)$、$P(x,0)$，
	把 $f(x)$ 化为 $|PA|+|PB|$，再用两点之间线段最短。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（对应距离）：}]
		由两点间距离公式，
		\[
			|PA|=\sqrt{(x-m)^2+a^2},\qquad
			|PB|=\sqrt{(x-n)^2+b^2}.
		\]
		因此 $f(x)=|PA|+|PB|$。

		\item[\textbf{步骤二（找到最短路径）：}]
		由三角形不等式，
		\[
			f(x)=|PA|+|PB|\geq |AB|.
		\]
		因为 $a,b>0$，$A$、$B$ 分居 $x$ 轴两侧，
		线段 $AB$ 必与 $x$ 轴相交。
		取 $P$ 为这一交点，就有 $P$ 在线段 $AB$ 上，
		故等号确实能够成立。

		\item[\textbf{步骤三（计算最小值）：}]
		\[
			\begin{aligned}
				|AB|
				&=\sqrt{(m-n)^2+\bigl((-a)-b\bigr)^2}\\
				&=\sqrt{(m-n)^2+(a+b)^2}.
			\end{aligned}
		\]
		线段 $AB$ 与 $x$ 轴的交点横坐标为
		\[
			x=m+\frac{a}{a+b}(n-m)
			 =\frac{bm+an}{a+b}.
		\]

		\item[\textbf{步骤四（确定值域）：}]
		$f$ 在 $\mathbb{R}$ 上连续，最小值能够取得；
		且当 $x\to+\infty$ 时，$f(x)\to+\infty$。
		因此其值域为
		\[
			\left[\sqrt{(m-n)^2+(a+b)^2},+\infty\right).
		\]
	\end{description}
\end{proofbox}
