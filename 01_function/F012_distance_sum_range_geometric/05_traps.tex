\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（对称点设定）}}] 构造点时将 $A$ 设为 $(m,a)$ 而不是 $(m,-a)$．
		\item[\textbf{\textcolor{CTrap}{正确理解（须设异侧）：}}] 必须构造 $A(m,-a)$，使得 $A$ 与 $B(n,b)$ 分别在 $x$ 轴两侧，才能直接利用“两点间线段最短”求最小值．
		\item[\textbf{\textcolor{CTrap}{错因分析（同侧转化错）：}}] 若 $A,B$ 在 $x$ 轴同侧，$|PA|+|PB|$ 的最小值通常需要对称变换，不能直接套用 $|AB|$．
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（共线位置）}}] 误认为只要 $A,P,B$ 三点共线就能取最小值．
		\item[\textbf{\textcolor{CTrap}{正确理解（须线段上）：}}] 只有当 $P$ 落在线段 $AB$ 上时，$|PA|+|PB|=|AB|$；若 $P$ 在线段延长线上，$|PA|+|PB|>|AB|$．
		\item[\textbf{\textcolor{CTrap}{错因分析（延长线误解）：}}] 三角形不等式 $|PA|+|PB|\ge|AB|$ 的等号成立条件为 $P$ 在线段 $AB$ 上，不是任意共线都成立．
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（纵坐标差）}}] 计算 $|AB|$ 时误用 $(a-b)^2$ 代替 $(a+b)^2$．
		\item[\textbf{\textcolor{CTrap}{正确理解（纵差应求和）：}}] 因为 $A(m,-a)$、$B(n,b)$，纵坐标差为 $b-(-a)=a+b$，故 $|AB|=\sqrt{(m-n)^2+(a+b)^2}$．
		\item[\textbf{\textcolor{CTrap}{错因分析（符号遗漏）：}}] 误认为 $A$ 的纵坐标为 $a$ 导致差为 $b-a$；或者忽略负号，写成 $b-a$，造成最小值错误．
	\end{description}
\end{trapbox}
