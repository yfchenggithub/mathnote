\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视三次项系数不为零）：}}]
		      直接使用结论时，忘记检查三次项系数 $A$ 是否为 $0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（函数定义）：}}]
		      三次函数 $f(x)=Ax^3+Bx^2+Cx+D$ 要求 $A\neq0$。若 $A=0$，函数退化为二次函数，不能再使用三次函数极值判别式。
		\item[\textbf{\textcolor{CTrap}{错因分析（前提缺失）：}}]
		      只记住了公式 $\Delta=B^2-3AC$，却忘了它是三次函数的结论，前提是 $A\neq0$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（混淆判别式）：}}]
		      误把一元二次方程判别式 $b^2-4ac$ 当成三次函数极值判别式。
		\item[\textbf{\textcolor{CTrap}{正确理解（极值判别式来源）：}}]
		      三次函数极值判别式为
		      \[
			      \Delta=B^2-3AC,
		      \]
		      它来自导函数 $f'(x)=3Ax^2+2Bx+C$ 的判别式
		      \[
			      \Delta_{f'}=(2B)^2-4\cdot3A\cdot C=4(B^2-3AC).
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}]
		      $B^2-3AC$ 和二次方程判别式形式相近，但系数来源不同，不能混用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略 $A$ 的符号）：}}]
		      在 $\Delta>0$ 时，只记得 $A>0$ 的情况，默认函数总是先增后减再增。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号决定趋势）：}}]
		      当 $\Delta>0$ 时，函数一定有两个极值点，但单调趋势还要看 $A$ 的符号：
		      \[
			      \begin{cases}
				      A>0: & \text{增}\rightarrow\text{减}\rightarrow\text{增}, \\[2pt]
				      A<0: & \text{减}\rightarrow\text{增}\rightarrow\text{减}.
			      \end{cases}
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
		      常见题目中三次项系数多为正数，容易忽略 $A<0$ 时导函数抛物线开口向下。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（把 $\Delta=0$ 当成有极值）：}}]
		      看到 $f'(x)=0$ 有根，就误以为原函数一定有极值点。
		\item[\textbf{\textcolor{CTrap}{正确理解（重根不变号）：}}]
		      当 $\Delta=0$ 时，$f'(x)$ 只有一个二重根，导函数在该点两侧不变号，原函数没有极值点。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆驻点与极值点）：}}]
		      驻点只是 $f'(x)=0$ 的点，极值点还要求函数在该点附近发生由增到减或由减到增的变化。
	\end{description}
\end{trapbox}
