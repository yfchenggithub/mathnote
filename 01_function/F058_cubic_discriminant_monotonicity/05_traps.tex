	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视a≠0）：}}]
			直接使用结论时，忘记检查三次项系数 $a$ 是否为 $0$.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（函数定义）：}}]
			三次函数定义要求 $a\neq0$，若 $a=0$ 则函数退化为二次函数，结论不适用.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（前提缺失）：}}]
			生搬硬套结论，忽略适用范围.\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（混淆判别式）：}}]
			误将一元二次方程判别式 $b^2-4ac$ 当作三次函数极值判别式.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（极值判别式）：}}]
			三次函数极值判别式为 $\Delta = b^2-3ac$，源自导函数 $f'(x)=3ax^2+2bx+c$ 的判别式.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}]
			与二次方程判别式形式相近，系数不同容易记错.\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略a符号）：}}]
			在 $\Delta>0$ 时，只记得 $a>0$ 的情况，默认函数先增后减再增.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（符号决定）：}}]
			当 $a<0$ 时，函数先减后增再减；$a>0$ 时先增后减再增.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
			常见题目中 $a$ 多正数，导致学生习惯性忽略 $a<0$ 的情形.\n
	\end{description}
