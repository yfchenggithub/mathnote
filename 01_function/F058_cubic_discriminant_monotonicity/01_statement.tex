\begin{statementbox}
	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（函数形式）：}}]
			$f(x)=ax^3+bx^2+cx+d$，其中 $a\neq0$，$a,b,c,d\in\mathbb{R}$.\n
		\item[\textbf{\textcolor{CStatement}{条件2（判别式定义）：}}]
			定义 $\Delta = b^2 - 3ac$.\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（极值点存在）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $\Delta>0$ 时，函数 $f(x)$ 有两个极值点，图像出现两次转向.\n
			\[
				\n \Delta = b^2 - 3ac > 0.\n
			\]
			\n \n
		\item[\textbf{\textcolor{CStatement}{结论二（单调性判断）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $\Delta\le 0$ 时，函数 $f(x)$ 无极值点，若 $a>0$ 则单调递增，若 $a<0$ 则单调递减.\n
			\[
				\n \Delta = b^2 - 3ac \le 0.\n
			\]
			\n \n
		\item[\textbf{\textcolor{CStatement}{推广结论（常见变形）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			导函数 $f'(x)=3ax^2+2bx+c$ 的判别式 $\Delta_{f'} = 4(b^2-3ac) = 4\Delta$，故 $\Delta>0$ 等价于 $f'(x)=0$ 有两个不同实根.\n
			\[
				\n \Delta_{f'} = 4(b^2-3ac) = 4\Delta.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
