\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三次函数）：}}]
		      函数
		      \[
			      f(x)=Ax^3+Bx^2+Cx+D,
		      \]
		      其中 $A\neq0$，$A,B,C,D\in\mathbb{R}$。

		\item[\textbf{\textcolor{CStatement}{条件 2（判别式定义）：}}]
		      定义三次函数极值判别式
		      \[
			      \Delta=B^2-3AC.
		      \]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两个极值点）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      当 $\Delta>0$ 时，导函数有两个不同零点，原函数有两个极值点。
		      \[
			      \boxed{\Delta=B^2-3AC>0.}
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（没有极值点）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      当 $\Delta\le0$ 时，导函数没有两个不同零点，原函数没有极值点；若 $A>0$，则 $f(x)$ 在 $\mathbb{R}$ 上单调递增；若 $A<0$，则 $f(x)$ 在 $\mathbb{R}$ 上单调递减。
		      \[
			      \boxed{\Delta=B^2-3AC\le0.}
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（导函数判别式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      导函数
		      \[
			      f'(x)=3Ax^2+2Bx+C
		      \]
		      的判别式为
		      \[
			      \boxed{\Delta_{f'}=(2B)^2-4\cdot3A\cdot C=4(B^2-3AC)=4\Delta.}
		      \]
		      因此，$\Delta>0$ 等价于 $f'(x)=0$ 有两个不同实根。
	\end{description}
\end{statementbox}
