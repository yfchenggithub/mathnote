\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n判别式 $\Delta = b^2-3ac$ 直接决定导函数 $f'(x)=3ax^2+2bx+c$ 是否有零点，从而决定原函数是否有极值点.\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（极值点来源）：}]
			极值点出现在导数为零处；对于三次函数，导数是二次函数，其零点的个数决定极值点个数.\n
		\item[\textbf{要点二（判别式角色）：}]
			二次函数 $f'(x)$ 的判别式 $\Delta_{f'}=4(b^2-3ac)=4\Delta$，故 $\Delta$ 的符号完全决定 $f'(x)$ 的根的情况.\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（导函数图像）}}\n\n$f'(x)$ 是开口由 $a$ 决定的抛物线：其与 $x$ 轴的交点对应原函数 $f(x)$ 的驻点；$\Delta>0$ 时有两个交点，$\Delta\le0$ 时无交点或相切.\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（方程判别式）}}\n\n极值点个数问题转化为二次方程 $3ax^2+2bx+c=0$ 的判别式问题，无需解出具体根即可判断.\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（快速判趋势）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（选填提速）：} 在选择题中直接利用 $\Delta$ 符号判断三次函数单调性，避免求导列表.\n
		\item \textbf{考法二（含参分类）：} 参数影响二次项系数时，分类讨论 $\Delta$ 的正负确定极值点个数.\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n将“三次函数的极值”与“二次判别式”搭建桥梁：只需关注系数 $b^2-3ac$ 的符号，避开求导后的繁琐计算.\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（解析式已知）：} 直接给出 $a,b,c$ 的值或范围，判断极值点个数或单调区间.\n
		\item \textbf{场景二（含参讨论）：} 参数出现在系数中，利用 $\Delta$ 的符号讨论极值点个数及单调性.\n
	\end{itemize}
	\n
\end{explanationbox}
