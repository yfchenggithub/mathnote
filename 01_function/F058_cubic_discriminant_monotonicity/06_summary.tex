\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{三次函数 $f(x)=Ax^3+Bx^2+Cx+D$ 的极值点个数由 $\Delta=B^2-3AC$ 决定；单调趋势还要结合三次项系数 $A$ 的符号判断。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三次函数）：} 函数必须是 $f(x)=Ax^3+Bx^2+Cx+D$，且 $A\neq0$。
		\item \textbf{条件 2（判别式定义）：} 使用三次函数极值判别式 $\Delta=B^2-3AC$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{核心结论}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{$\Delta>0$：} 有两个极值点。若 $A>0$，趋势为增 $\rightarrow$ 减 $\rightarrow$ 增；若 $A<0$，趋势为减 $\rightarrow$ 增 $\rightarrow$ 减。
		\item \textbf{$\Delta\le0$：} 没有极值点。若 $A>0$，函数在 $\mathbb{R}$ 上单调递增；若 $A<0$，函数在 $\mathbb{R}$ 上单调递减。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式来源）：} $B^2-3AC$ 来自导函数 $f'(x)=3Ax^2+2Bx+C$ 的判别式，不是普通二次方程的 $b^2-4ac$。
		\item \textbf{检查项（符号双检）：} 先看 $\Delta$ 判断极值点个数，再看 $A$ 判断单调趋势。
		\item \textbf{特别注意（$\Delta=0$）：} 此时导函数有一个二重零点，但原函数没有极值点，只可能出现水平拐点。
	\end{itemize}
\end{summarybox}
