\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n三次函数的极值点个数和单调性趋势完全由判别式 $\Delta = b^2-3ac$ 的符号及三次项系数 $a$ 的符号决定.\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（函数形式）：} 函数为三次函数 $f(x)=ax^3+bx^2+cx+d$，且 $a\neq0$.\n
		\item \textbf{条件2（判别式定义）：} 定义 $\Delta = b^2-3ac$.\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（前提检验）：} 使用前确认 $a\neq0$，并选用正确判别式 $b^2-3ac$，而非 $b^2-4ac$.\n
		\item \textbf{检查项（符号双检）：} 先检查 $\Delta$ 符号判断极值点有无，再检查 $a$ 的符号确定单调趋势（$a>0$ 增 $\rightarrow$ 减 $\rightarrow$ 增，$a<0$ 减 $\rightarrow$ 增 $\rightarrow$ 减）.\n
	\end{itemize}
	\n
\end{summarybox}
