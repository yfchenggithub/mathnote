\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	\textbf{先求导，把三次函数极值问题转化为二次函数零点问题；再用判别式判断导函数是否有两个不同实根，从而判断原函数是否有极值点。}

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（求导）：}]
		      由
		      \[
			      f(x)=Ax^3+Bx^2+Cx+D
		      \]
		      可得
		      \[
			      f'(x)=3Ax^2+2Bx+C.
		      \]
		      \par\textbf{理由：} 按多项式求导法则逐项求导。

		\item[\textbf{步骤二（建立判别式关系）：}]
		      把 $f'(x)$ 看作关于 $x$ 的二次函数，其中二次项系数为 $3A$，一次项系数为 $2B$，常数项为 $C$。因此
		      \[
			      \begin{aligned}
				      \Delta_{f'}
				       & =(2B)^2-4\cdot3A\cdot C \\
				       & =4B^2-12AC              \\
				       & =4(B^2-3AC)             \\
				       & =4\Delta.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 二次方程 $ux^2+vx+w=0$ 的判别式为 $v^2-4uw$。

		\item[\textbf{步骤三（$\Delta>0$ 时）：}]
		      若 $\Delta>0$，则 $\Delta_{f'}=4\Delta>0$，所以方程 $f'(x)=0$ 有两个不同实根，记为 $x_1<x_2$。
		      \[
			      f'(x)=3A(x-x_1)(x-x_2).
		      \]
		      \par\textbf{理由：} 判别式大于零时，二次方程有两个不同实根。

		\item[\textbf{步骤四（$A>0$ 的单调趋势）：}]
		      若 $A>0$，则 $f'(x)$ 开口向上，故
		      \[
			      \begin{aligned}
				      x<x_1     & \Rightarrow f'(x)>0, \\
				      x_1<x<x_2 & \Rightarrow f'(x)<0, \\
				      x>x_2     & \Rightarrow f'(x)>0.
			      \end{aligned}
		      \]
		      因此 $f(x)$ 先增后减再增，在 $x_1,x_2$ 处分别取得一个极大值和一个极小值。
		      \par\textbf{理由：} 导函数符号由正变负产生极大值，由负变正产生极小值。

		\item[\textbf{步骤五（$A<0$ 的单调趋势）：}]
		      若 $A<0$，则 $f'(x)$ 开口向下，故
		      \[
			      \begin{aligned}
				      x<x_1     & \Rightarrow f'(x)<0, \\
				      x_1<x<x_2 & \Rightarrow f'(x)>0, \\
				      x>x_2     & \Rightarrow f'(x)<0.
			      \end{aligned}
		      \]
		      因此 $f(x)$ 先减后增再减，在 $x_1,x_2$ 处分别取得一个极小值和一个极大值。
		      \par\textbf{理由：} 导函数符号由负变正产生极小值，由正变负产生极大值。

		\item[\textbf{步骤六（$\Delta<0$ 时）：}]
		      若 $\Delta<0$，则 $\Delta_{f'}<0$，导函数 $f'(x)$ 没有实根，所以 $f'(x)$ 恒不变号。由于 $f'(x)$ 的二次项系数为 $3A$，故 $A>0$ 时 $f'(x)>0$ 恒成立，$A<0$ 时 $f'(x)<0$ 恒成立。
		      \par\textbf{理由：} 没有实根的二次函数整体在 $x$ 轴同一侧，其符号由开口方向决定。

		\item[\textbf{步骤七（$\Delta=0$ 时）：}]
		      若 $\Delta=0$，则 $f'(x)$ 有一个二重根
		      \[
			      x_0=-\frac{B}{3A},
		      \]
		      并且
		      \[
			      f'(x)=3A\left(x+\frac{B}{3A}\right)^2.
		      \]
		      所以 $A>0$ 时 $f'(x)\ge0$，$A<0$ 时 $f'(x)\le0$，导函数在 $x_0$ 两侧不变号，原函数没有极值点。
		      \par\textbf{理由：} 导函数只是与 $x$ 轴相切，没有穿过 $x$ 轴，因此原函数只出现水平拐点，不出现极大值或极小值。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	综上，$\Delta=B^2-3AC>0$ 时，$f(x)$ 有两个极值点；$\Delta\le0$ 时，$f(x)$ 没有极值点。若 $\Delta\le0$，则 $A>0$ 时 $f(x)$ 在 $\mathbb{R}$ 上单调递增，$A<0$ 时 $f(x)$ 在 $\mathbb{R}$ 上单调递减。
\end{proofbox}
