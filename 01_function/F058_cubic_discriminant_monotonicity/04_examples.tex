\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 判断函数 $f(x)=x^3-3x^2+2x$ 的极值点个数，并说明其单调性趋势.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（确定系数）：}]
			$a=1,\ b=-3,\ c=2$.\n
			\par\textbf{理由：} 与标准形式 $f(x)=ax^3+bx^2+cx+d$ 对照.\n\n
		\item[\textbf{第二步（计算判别式）：}]
			$\Delta = b^2-3ac = (-3)^2-3\cdot1\cdot2 = 9-6=3>0$.\n
			\par\textbf{理由：} 代入公式 $\Delta = b^2-3ac$.\n\n
		\item[\textbf{第三步（套用结论）：}]
			因为 $\Delta>0$ 且 $a=1>0$，故函数有两个极值点，且单调性趋势为：增 $\rightarrow$ 减 $\rightarrow$ 增（即 $f(x)$ 先增后减再增）.\n
			\par\textbf{理由：} 由结论：$\Delta>0$ 时有两个极值点；$a>0$ 时图像先增后减再增.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $\Delta=3>0$，$a=1>0$，故 $f(x)$ 有两个极值点，单调区间为 $(-\infty,x_1)$ 递增，$(x_1,x_2)$ 递减，$(x_2,+\infty)$ 递增，其中 $x_1,x_2$ 为 $f'(x)=0$ 的两根.\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 设函数 $f(x)=x^3-3x^2+ax$（$a\in\mathbb{R}$），讨论 $f(x)$ 极值点的个数.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（确定系数）：}]
			$a=1,\ b=-3,\ c=a$（此处 $a$ 为参数）.\n
			\par\textbf{理由：} 与标准形式 $f(x)=ax^3+bx^2+cx+d$ 对应.\n\n
		\item[\textbf{第二步（计算判别式）：}]
			$\Delta = b^2-3ac = (-3)^2-3\cdot1\cdot a = 9-3a$.\n
			\par\textbf{理由：} 代入判别式公式.\n\n
		\item[\textbf{第三步（分类讨论）：}]
			当 $\Delta>0$ 即 $a<3$ 时，有两个极值点；当 $\Delta\le0$ 即 $a\ge3$ 时，无极值点（且由于 $a=1>0$，此时函数单调递增）.\n
			\par\textbf{理由：} 由结论：$\Delta>0$ 有两个极值点；$\Delta\le0$ 无极值点，$a>0$ 时单调递增.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $a<3$ 时 $f(x)$ 有两个极值点；$a\ge3$ 时 $f(x)$ 无极值点.\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知函数 $f(x)=ax^3-3x^2+x+1$ 在 $\mathbb{R}$ 上单调递增，求实数 $a$ 的取值范围.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求导计算）：}]
			$f'(x)=3ax^2-6x+1$.\n
			\par\textbf{理由：} 多项式求导.\n\n
		\item[\textbf{第二步（讨论开口方向）：}]
			若 $a\le0$，则 $f'(x)$ 开口向下（或一次）且二次项系数非正，不能恒为非负，故不满足单调递增；若 $a>0$，则 $f'(x)$ 开口向上，有可能恒非负.\n
			\par\textbf{理由：} 单调递增要求 $f'(x)\ge0$ 恒成立，开口向下的二次函数不可能恒非负.\n\n
		\item[\textbf{第三步（计算条件）：}]
			当 $a>0$ 时，$f'(x)\ge0$ 恒成立等价于判别式 $\Delta_{f'}=(-6)^2-4\cdot3a\cdot1=36-12a\le0$，解得 $a\ge3$.\n
			\par\textbf{理由：} 开口向上的二次函数恒非负的充要条件是判别式不大于零.\n\n
		\item[\textbf{第四步（综合结论）：}]
			综合 $a>0$ 与 $a\ge3$ 得 $a\ge3$. 故 $a$ 的取值范围是 $[3,+\infty)$.\n
			\par\textbf{理由：} 当 $a\ge3$ 时，$\Delta_{f'}\le0$，满足条件.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $a\ge3$ 时，$f(x)$ 在 $\mathbb{R}$ 上单调递增.\n
\end{examplebox}
