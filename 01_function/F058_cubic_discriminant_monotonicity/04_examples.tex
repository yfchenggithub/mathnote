\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	判断函数 $f(x)=x^3-3x^2+2x$ 的极值点个数，并说明其单调性趋势。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定系数）：}]
		      与标准形式
		      \[
			      f(x)=Ax^3+Bx^2+Cx+D
		      \]
		      对照，得到 $A=1,
			      B=-3,
			      C=2$。
		      \par\textbf{理由：} 只需要三次项、二次项、一次项系数即可计算 $\Delta$。

		\item[\textbf{第二步（计算判别式）：}]
		      \[
			      \Delta=B^2-3AC=(-3)^2-3\cdot1\cdot2=9-6=3>0.
		      \]
		      \par\textbf{理由：} 代入三次函数极值判别式 $\Delta=B^2-3AC$。

		\item[\textbf{第三步（判断极值点个数）：}]
		      因为 $\Delta>0$，所以 $f'(x)=0$ 有两个不同实根，函数有两个极值点。
		      \par\textbf{理由：} $\Delta>0$ 等价于导函数有两个不同零点。

		\item[\textbf{第四步（判断单调趋势）：}]
		      由于 $A=1>0$，所以函数单调性趋势为
		      \[
			      \text{增}\ \longrightarrow\ \text{减}\ \longrightarrow\ \text{增}.
		      \]
		      \par\textbf{理由：} $A>0$ 时，导函数抛物线开口向上，两根之外 $f'(x)>0$，两根之间 $f'(x)<0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\Delta=3>0$，故 $f(x)$ 有两个极值点；又 $A>0$，所以 $f(x)$ 先增后减再增。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	设函数 $f(x)=x^3-3x^2+ax$，其中 $a\in\mathbb{R}$，讨论 $f(x)$ 极值点的个数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定系数）：}]
		      这里的 $a$ 是题目参数。为避免混淆，与标准形式 $f(x)=Ax^3+Bx^2+Cx+D$ 对照，得到
		      \[
			      A=1,\qquad B=-3,\qquad C=a.
		      \]
		      \par\textbf{理由：} 三次项系数是 $1$，一次项系数是参数 $a$。

		\item[\textbf{第二步（计算判别式）：}]
		      \[
			      \Delta=B^2-3AC=(-3)^2-3\cdot1\cdot a=9-3a.
		      \]
		      \par\textbf{理由：} 直接代入 $\Delta=B^2-3AC$。

		\item[\textbf{第三步（分类讨论）：}]
		      \[
			      \begin{cases}
				      \Delta>0 \Longleftrightarrow 9-3a>0 \Longleftrightarrow a<3, \\[2pt]
				      \Delta\le0 \Longleftrightarrow 9-3a\le0 \Longleftrightarrow a\ge3.
			      \end{cases}
		      \]
		      因此，$a<3$ 时有两个极值点；$a\ge3$ 时没有极值点。
		      \par\textbf{理由：} $\Delta>0$ 有两个极值点，$\Delta\le0$ 无极值点。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$a<3$ 时，$f(x)$ 有两个极值点；$a\ge3$ 时，$f(x)$ 没有极值点。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=ax^3-3x^2+x+1$ 在 $\mathbb{R}$ 上单调递增，求实数 $a$ 的取值范围。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导）：}]
		      \[
			      f'(x)=3ax^2-6x+1.
		      \]
		      \par\textbf{理由：} 要判断原函数在 $\mathbb{R}$ 上单调递增，通常转化为判断 $f'(x)\ge0$ 恒成立。

		\item[\textbf{第二步（排除 $a\le0$）：}]
		      若 $a<0$，则 $f'(x)$ 是开口向下的二次函数，不可能恒有 $f'(x)\ge0$；若 $a=0$，则 $f'(x)=-6x+1$，也不可能恒有 $f'(x)\ge0$。因此必须有 $a>0$。
		      \par\textbf{理由：} 单调递增要求导函数不能在某些区间为负。

		\item[\textbf{第三步（判别式条件）：}]
		      当 $a>0$ 时，$f'(x)=3ax^2-6x+1$ 开口向上。要使 $f'(x)\ge0$ 恒成立，需要且只需要其判别式不大于零：
		      \[
			      (-6)^2-4\cdot3a\cdot1\le0.
		      \]
		      即
		      \[
			      36-12a\le0,
		      \]
		      解得
		      \[
			      a\ge3.
		      \]
		      \par\textbf{理由：} 开口向上的二次函数恒不小于 $0$ 的充要条件是判别式 $\le0$。

		\item[\textbf{第四步（综合结论）：}]
		      由 $a>0$ 与 $a\ge3$ 综合，得到
		      \[
			      \boxed{a\ge3.}
		      \]
		      \par\textbf{理由：} 当 $a\ge3$ 时，$f'(x)\ge0$ 恒成立，故 $f(x)$ 在 $\mathbb{R}$ 上单调递增。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$a\ge3$ 时，$f(x)$ 在 $\mathbb{R}$ 上单调递增。
\end{examplebox}
