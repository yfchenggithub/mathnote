\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{分段函数 $f(x)=
	\begin{cases}
		g(x)&x\le x_0\\h(x)&x>x_0
	\end{cases}
	$ 在 $\mathbb{R}$ 上单调递增 $\iff$ $g$ 在 $(-\infty,x_0]$ 上递增，$h$ 在 $(x_0,+\infty)$ 上递增，且 $g(x_0)\le\lim_{x\to x_0^+}h(x)$；递减时将“递增”换为“递减”，不等号反向。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（分段定义）：} 函数由两个分段表达式构成，分界点明确。
		\item \textbf{条件2（单调一致）：} 每个分段必须在各自区间上单调递增（或递减）。
		\item \textbf{条件3（衔接比较）：} 必须比较左段终值与右段起点极限的大小。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（衔接遗忘）：} 易只做段内单调，漏掉分界点处不等式。
		\item \textbf{检查项（方向确认）：} 解题前先明确递增还是递减，再写不等号方向。
		\item \textbf{检查项（极限替代）：} 当右段不含 $x_0$ 时，切勿直接用 $h(x_0)$，应使用右极限。
	\end{itemize}

\end{summarybox}
