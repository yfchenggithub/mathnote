\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（遗漏特殊点）：}}] 误认为对任意恒成立问题都可用此方法。
		\item[\textbf{\textcolor{CTrap}{正确理解（需有取等点）：}}] 只有存在取等点（$f(x_0,a)=0$）才能使用，否则导数条件不成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}] 缺乏对取等条件重要性的认识，盲目套用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（方向混淆）：}}] 左端点取等且原不等式为 $\ge0$ 时，误以为 $f'(x_0)\le0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（左正右负）：}}] 左端点导数应 $\ge0$，右端点导数应 $\le0$（对 $\ge0$ 不等式）。
		\item[\textbf{\textcolor{CTrap}{错因分析（区间方向）：}}] 忽略区间延伸方向对导数符号的影响。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（跳过充分性）：}}] 只算出必要条件就下结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须验证充分性）：}}] 必须验证参数在该范围内原不等式恒成立，才能确定最终答案。
		\item[\textbf{\textcolor{CTrap}{错因分析（以偏概全）：}}] 认为必要条件即充要条件，忽略验证环节。
	\end{description}
\end{trapbox}
