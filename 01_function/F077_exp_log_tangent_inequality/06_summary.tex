\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{指对同构基本不等式：$e^x\ge e x$（实数）和 $x\ge e\ln x$（正数）及含参变形 $\frac{e^x}{a}\ge e(x-\ln a)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（指数域）：} 指数不等式对任意实数 $x$ 成立。
		\item \textbf{条件 2（对数域）：} 对数不等式要求 $x>0$。
		\item \textbf{条件 3（参数正数）：} 含参变形需要 $a>0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（定义域）：} 对数不等式和变形不能忽略 $x>0$ 和 $a>0$。
		\item \textbf{检查项（等号条件）：} 使用不等式时务必检查等号能否取得。
	\end{itemize}
\end{summarybox}
