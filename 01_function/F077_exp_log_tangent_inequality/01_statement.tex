\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（实数全体）：}}] 对于指数不等式，变量 $x$ 为任意实数。
		\item[\textbf{\textcolor{CStatement}{条件 2（正数区间）：}}] 对于对数不等式，变量 $x$ 取所有正实数。
		\item[\textbf{\textcolor{CStatement}{条件 3（参数范围）：}}] 对于含参变形，参数 $a>0$ 且 $x$ 为任意实数。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（指数不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			指数函数 $e^x$ 恒在直线 $y=ex$ 上方（或相切）。
			\[
				e^x \ge e x \quad (\forall x\in\mathbb{R})
			\]
			等号当且仅当 $x=1$。

		\item[\textbf{\textcolor{CStatement}{结论二（对数不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $x>0$ 时，$x$ 恒不小于 $e\ln x$。
			\[
				x \ge e\ln x \quad (\forall x>0)
			\]
			等号当且仅当 $x=e$。

		\item[\textbf{\textcolor{CStatement}{推广结论（含参变形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			引入参数 $a>0$，不等式变为分式形式。
			\[
				\frac{e^x}{a} \ge e(x-\ln a) \quad (\forall a>0,\; x\in\mathbb{R})
			\]
			等号当且仅当 $x=\ln a+1$。
	\end{description}

\end{statementbox}
