% F035 指数幂的运算法则｜03 用三种最容易懂的方法说明为什么成立
\begin{proofbox}
    \textbf{\textcolor{CProof}{证明从哪里开始？}}\par
    先只考虑\textbf{正整数指数}：因为 $a^3=a\cdot a\cdot a$，可以真正数清楚有几个 $a$。下面分别说明三条规律；这是初中知识就能理解的部分。

    \medskip
    \textbf{\textcolor{CProof}{方法一｜把两串接起来：同底相乘指数相加}}\par
    对正整数 $m,n$：
    \[
      \begin{aligned}
        a^ma^n
        &=\bigl(\underbrace{a\cdots a}_{m\text{ 个}}\bigr)
          \bigl(\underbrace{a\cdots a}_{n\text{ 个}}\bigr)\\
        &=\underbrace{a\cdots a}_{m+n\text{ 个}}=a^{m+n}.
      \end{aligned}
    \]
    \textbf{为什么对？} 乘法结合律允许我们去掉中间的分组；两串同样的 $a$ 接在一起，总数相加。

    \medskip
    \textbf{\textcolor{CProof}{方法二｜数“每组几个”：幂的乘方指数相乘}}\par
    对正整数 $m,n$，$(a^m)^n$ 就是\textbf{$n$ 组 $a^m$ 相乘}，每组又有 $m$ 个 $a$：
    \[
      (a^m)^n=\underbrace{a^m\cdot a^m\cdots a^m}_{n\text{ 组}}
      =a^{\underbrace{m+\cdots+m}_{n\text{ 个 }m}}=a^{mn}.
    \]
    \textbf{为什么对？} 组数 $\times$ 每组数量，就是相同因子的总数。

    \end{center}

    \medskip
    \textbf{\textcolor{CProof}{方法三｜把相同因子放一起：积的乘方}}\par
    对正整数 $m$，$(ab)^m$ 是 $m$ 组 $ab$ 相乘。运用乘法交换律、结合律：
    \[
      (ab)^m=\underbrace{(ab)(ab)\cdots(ab)}_{m\text{ 组}}
        =\underbrace{a\cdots a}_{m\text{ 个}}\,
         \underbrace{b\cdots b}_{m\text{ 个}}=a^mb^m.
    \]
    \textbf{为什么对？} 只是交换乘法因子的顺序，没有改变任何因子的个数。

    \end{center}

    \medskip
    \textbf{\textcolor{CProof}{负整数、分数、实数指数怎么办？}}\par
    \begin{itemize}[leftmargin=*,itemsep=0.35em,topsep=0.3em]
        \item \textbf{零次与负整数次：} 对 $a\ne0$ 规定 $a^0=1$、$a^{-k}=1/a^k$（$k$ 为正整数），例如 $a^3a^{-2}=a$，与 $a^{3+(-2)}=a$ 一致。
        \item \textbf{分数指数：} 对 $a>0$，$a^{p/q}=\sqrt[q]{a^p}$（$p$ 为整数，$q$ 为正整数）。例如 $(a^{1/2})^2=(\sqrt a)^2=a=a^{(1/2)\times2}$。
        \item \textbf{任意实数指数：} 高中教材把正底数的指数幂定义扩展到实数指数，三条运算法则继续成立。上面\textbf{数因子}的证明只针对正整数指数，不能直接当作所有实数指数的证明。
    \end{itemize}

    \input{assets/tikz/F035_exponent_rules_03_proof_04.tex}
    \end{center}

    \textbf{\textcolor{CProof}{最后记住：}}\textbf{加法}来自“把两组接起来”，\textbf{乘法}来自“每组数量乘组数”，\textbf{分别乘方}来自“重新排列相同因子”。
\end{proofbox}
