\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略定义域）}}] 求对数型函数的定义域或解对数不等式时，忘记检查真数大于$0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（先定义域）}}] 涉及对数函数的任何问题，必须先保证自变量满足真数$>0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（习惯性疏忽）}}] 习惯将所有函数定义域默认为$\mathbb{R}$，遗忘对数函数的限制条件。
	\end{description}

\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（底数分类遗漏）}}] 处理含有参数$a$的对数函数单调性或不等式时，直接假设底数$>1$，未分$0<a<1$讨论。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论）}}] 当底数含参数$a$时，必须分$a>1$和$0<a<1$两类分别处理。
		\item[\textbf{\textcolor{CTrap}{错因分析（默认递增）}}] 默认对数函数都是增函数，忽略底数在$(0,1)$时的递减情形。
	\end{description}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号记忆混淆）}}] 记错函数值符号，比如误以为$a>1$时$x>1$对应$y<0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号法则）}}] $a>1$时，$x>1\Rightarrow y>0$，$0<x<1\Rightarrow y<0$；$0<a<1$时反之。
		\item[\textbf{\textcolor{CTrap}{错因分析（机械记忆）}}] 不结合单调性与定点推导符号，死记硬背导致混淆。
\end{description}

\input{assets/tikz/F039_trap_negative_output}
\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
 \item[\textbf{\textcolor{CTrap}{易错点四（把真数为正误当函数值为正）}}] 错说“对数函数永远大于$0$”，甚至认为$\log_2\tfrac12$不能算。
 \item[\textbf{\textcolor{CTrap}{正确理解（两种正号分清）}}] \textbf{必须为正的是真数}：$x>0$；函数值可以小于零，例如$\log_2\tfrac12=-1$。底数同样要求$a>0$且$a\ne1$。
 \item[\textbf{\textcolor{CTrap}{错因分析}}] 把指数函数的“输出大于$0$”错搬到对数函数的“输出”上。两者交换横纵坐标后，限制应放在\textbf{对数的输入}上。
\end{description}
\end{trapbox}

\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
 \item[\textbf{\textcolor{CTrap}{易错点五（对称轴和定点混淆）}}] 错说$\log_2x$与$\log_{1/2}x$关于$y$轴对称，或说指数图像与对数图像都经过$(0,1)$。
 \item[\textbf{\textcolor{CTrap}{正确理解（用点检查）}}] 两个底数互倒：$(2,1)$对应$(2,-1)$，是\textbf{关于$x$轴}；指数$y=2^x$中的$(1,2)$对应对数$y=\log_2x$中的$(2,1)$，是\textbf{关于$y=x$}。对数图像不能有横坐标$0$。
 \item[\textbf{\textcolor{CTrap}{错因分析}}] 没有核查对称前后的点坐标，只凭曲线外观猜对称轴。
\end{description}

\begin{trapbox}
\textbf{\textcolor{CTrap}{补充易错（底数是否合法）}}
\input{assets/tikz/F039_base_legality}
不能把$\log_1x$或负底数直接当作本结论中的函数。$1^t=1$无法反求唯一指数；负底数的实数次幂也不覆盖全体实数。先核对$a>0,a\ne1$。

\end{trapbox}
\clearpage
\begin{trapbox}
\textbf{\textcolor{CTrap}{补充易错（别把定点写反）}}
\input{assets/tikz/F039_fixed_point_error}
$(0,1)$属于指数函数的共同点；对数函数应令真数为$1$，得到$(1,0)$。横坐标为$0$时对数没有意义。

\end{trapbox}
\clearpage
\begin{trapbox}
\textbf{\textcolor{CTrap}{补充易错（参数也是底数或真数时）}}
\input{assets/tikz/F039_parameter_inputs}
以$\log_{a-1}(a+1)$为例：必须同时有$a-1>0$、$a-1\ne1$、$a+1>0$，得到$a>1$且$a\ne2$，不能只检查真数。

\end{trapbox}
\clearpage
\begin{trapbox}
\textbf{\textcolor{CTrap}{补充易错（无限靠近不等于与轴相交）}}
\input{assets/tikz/F039_no_y_intercept}
即使图像越来越靠近$y$轴，每个允许的横坐标仍严格大于$0$。$x=0$不在定义域内，不能凭绘图尾部猜出一个$y$轴交点。
\end{trapbox}
