\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	看到 $x f'(x)+f(x)$（或 $x f'(x)+n f(x)$），立即构造 $F(x)=x f(x)$（或 $G(x)=x^n f(x)$），通过导数符号判断单调性。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（导数关系）：} 题干中出现 $x f'(x)+f(x)$ 或其变式。
		\item \textbf{条件2（可导性）：} 函数在区间上可导。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号决定）：} 必须有明确的正负号才能判断单调性。
		\item \textbf{检查项（系数匹配）：} 使用推广 $n$ 次幂时，确认 $n$ 与条件中系数一致。
	\end{itemize}

\end{summarybox}
