\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（函数可导性）：}}] 设函数 $f(x)$ 在区间 $I$ 上可导。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（导数公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量与函数乘积的导数等于 $x$ 乘导数加函数本身。
			\[
				F'(x)=x f'(x)+f(x), \quad F(x)=x f(x).
			\]
			若 $x f'(x)+f(x) > 0$（或 $<0$），则 $F(x)$ 在 $I$ 上严格单调递增（或递减）。

		\item[\textbf{\textcolor{CStatement}{推广结论（$n$ 次幂推广）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$x^n$ 乘函数的导数公式。
			\[
				G'(x)=x^{\,n-1}\bigl[x f'(x)+n f(x)\bigr], \quad G(x)=x^n f(x).
			\]
	\end{description}

\end{statementbox}
