\begin{examplebox}

	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定义在 $\mathbb{R}$ 上的可导函数 $f(x)$ 满足 $x f'(x)+f(x)>0$，判断函数 $F(x)=x f(x)$ 的单调性。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 观察到条件为 $x f'(x)+f(x)>0$，这正是主结论中的表达式，故构造函数 $F(x)=x f(x)$。
			\par\textbf{理由：}主结论指出 $F'(x)=x f'(x)+f(x)$。
		\item[\textbf{第二步（求导分析）：}] 由 $F'(x)=x f'(x)+f(x)>0$，知 $F'(x)$ 在 $\mathbb{R}$ 上恒正。
			\par\textbf{理由：}导数公式。
		\item[\textbf{第三步（单调结论）：}] 导函数恒大于 $0$，所以 $F(x)$ 在 $\mathbb{R}$ 上严格单调递增。
			\par\textbf{理由：}导数正号函数递增。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$F(x)$ 在 $\mathbb{R}$ 上单调递增。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定义在 $(0,+\infty)$ 上的可导函数 $f(x)$ 满足 $x f'(x)+2f(x)>0$，判断函数 $G(x)=x^2 f(x)$ 的单调性。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别系数）：}] 条件中 $x f'(x)+2f(x)$ 与推广形式 $x f'(x)+n f(x)$ 对比，知 $n=2$，故构造 $G(x)=x^2 f(x)$。
			\par\textbf{理由：}推广结论中 $G'(x)=x^{n-1}[x f'(x)+n f(x)]$。
		\item[\textbf{第二步（求导化简）：}] 对 $G(x)$ 求导，由推广公式得
			\[
				\begin{aligned}
					G'(x) &= x^{2-1}\bigl[ x f'(x) + 2 f(x) \bigr] \\
					&= x\bigl[ x f'(x) + 2 f(x) \bigr].
			\end{aligned}
			\]
			\par\textbf{理由：}直接套用 $n=2$。
		\item[\textbf{第三步（判断符号）：}] 由于 $x>0$，且 $x f'(x)+2f(x)>0$，相乘得 $G'(x)>0$。因此 $G(x)$ 在 $(0,+\infty)$ 上单调递增。
			\par\textbf{理由：}正数乘以正数仍为正。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$G(x)$ 在 $(0,+\infty)$ 上单调递增。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定义在 $\mathbb{R}$ 上的可导函数 $f(x)$ 满足 $x f'(x)+f(x)>0$，且 $f(2)=1$。解不等式 $(x-1) f(x-1) > 2$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造函数）：}] 注意到不等式左边为 $(x-1) f(x-1)$，可视为 $F(t)=t f(t)$ 在 $t=x-1$ 处的值。故构造 $F(t)=t f(t)$。
			\par\textbf{理由：}主结论的形式。
		\item[\textbf{第二步（单调性分析）：}] 由条件 $x f'(x)+f(x)>0$ 对所有实数 $x$ 成立，则 $F'(t)=t f'(t)+f(t)>0$，因此 $F(t)$ 在 $\mathbb{R}$ 上严格单调递增。
			\par\textbf{理由：}导函数恒正。
		\item[\textbf{第三步（转化求解）：}] 计算 $F(2)=2 f(2)$，由 $f(2)=1$ 得 $F(2)=2$。原不等式化为 $F(x-1) > F(2)$。由 $F$ 单调递增，得 $x-1 > 2$，即 $x>3$。
			\par\textbf{理由：}单调递增时，函数值大的自变量也大。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	不等式解集为 $(3,+\infty)$。
