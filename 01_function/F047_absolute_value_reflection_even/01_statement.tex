	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（定义域限制）：}}]
			函数 $f(x)$ 的定义域包含区间 $[0,+\infty)$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（翻折变换）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			变换 $y=f(|x|)$ 为偶函数，图像保留 $y=f(x)$ 在 $y$ 轴右侧部分，并关于 $y$ 轴翻折到左侧。
			\[
				\boxed{f(|x|) =
				\begin{cases}
					f(x), & x \ge 0,\
					\[
						2pt] f(-x), & x < 0.
				\end{cases}
					}
				\]
			\item[\textbf{\textcolor{CStatement}{推广结论(值域不变性):}}]
				\textbf{\textcolor{CStatement}{直观描述:}}
				变换 $y=f(|x|)$ 的值域由 $y=f(x)$ 在 $x\ge 0$ 时的值域完全决定.
				\[
					\{\, f(|x|) \mid x\in\mathbb{R} \,\} = \{\, f(x) \mid x\ge 0 \,\}.
				\]
		\end{description}

	\end{statementbox}
