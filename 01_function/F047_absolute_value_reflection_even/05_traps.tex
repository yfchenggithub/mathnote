\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（翻折方向混淆）}}]
			误认为 $y=f(|x|)$ 的图像是将 $y=f(x)$ 在 $y$ 轴左侧的部分翻折到右侧。
		\item[\textbf{\textcolor{CTrap}{正确理解（翻折规则）：}}]
			实际是保留 $y=f(x)$ 在 $y$ 轴右侧的部分，然后关于 $y$ 轴翻折到左侧；左侧部分并不保留。
		\item[\textbf{\textcolor{CTrap}{错因分析（对称对象）：}}]
			混淆了 $f(|x|)$ 与 $|f(x)|$ 的变换规则。$f(|x|)$ 作用于自变量，保留右侧翻折左侧；$|f(x)|$ 作用于函数值，保留上方翻折下方。对变换对象理解错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（定义域误判）}}]
			误认为 $f(|x|)$ 的定义域与 $f(x)$ 相同，忽略了绝对值的非负化效应。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义域扩大）：}}]
			$f(|x|)$ 要求 $|x|$ 在 $f$ 定义域内。若 $f$ 的定义域仅包含 $[0,+\infty)$，则 $f(|x|)$ 的定义域是全体实数 $\mathbb{R}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视绝对值）：}}]
			没有意识到 $|x|$ 总是非负，以为自变量的符号直接决定函数值；实际上先取绝对值才代入 $f$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（单调性误推）}}]
			误认为 $f(|x|)$ 在 $x<0$ 时的单调性与 $x>0$ 时相同。
		\item[\textbf{\textcolor{CTrap}{正确理解（单调性相反）：}}]
			由于 $f(|x|)$ 是偶函数，在对称区间上的单调性相反：若在 $[0,+\infty)$ 上递增，则在 $(-\infty,0]$ 上递减。
		\item[\textbf{\textcolor{CTrap}{错因分析（奇偶性影响）：}}]
			直接套用 $f(x)$ 的单调性，忽略了偶函数图像关于 $y$ 轴对称导致左右对称区间单调性相反。
	\end{description}

\end{trapbox}
