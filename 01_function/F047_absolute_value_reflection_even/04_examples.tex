\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	设 $f(x)=2x+1$，其定义域为 $[0,+\infty)$。求 $y=f(|x|)$ 的解析式，并判断其奇偶性。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入绝对值）：}]
			将 $|x|$ 代入 $f(x)$ 表达式中，得 $f(|x|)=2|x|+1$。
			\par\textbf{理由：} 因为 $f(x)=2x+1$，将 $x$ 替换为 $|x|$。
		\item[\textbf{第二步（分段化简）：}]
			根据绝对值定义，当 $x\ge0$ 时 $|x|=x$，$f(|x|)=2x+1$；当 $x<0$ 时 $|x|=-x$，$f(|x|)=-2x+1$。
			\[
				f(|x|) =
				\begin{cases}
					2x+1, & x\ge0,\
					\[
						2pt] -2x+1, & x<0.
				\end{cases}
				\]
				\par\textbf{理由:} 直接代入绝对值表达式.
			\item[\textbf{第三步(奇偶性判定):}]
				计算 $f(|-x|)$.
				\[
					f(|-x|)=2|-x|+1=2|x|+1=f(|x|).
				\]
				\par\textbf{理由:} 满足 $f(|-x|)=f(|x|)$,故 $y=f(|x|)$ 是偶函数.
		\end{description}
		\textbf{\textcolor{CExample}{关键结论:}}
		\textbf{$f(|x|)$ 是偶函数,图像关于 $y$ 轴对称;其分段表达式如上,值域为 $[1,+\infty)$.}

		\medskip
		\textbf{\textcolor{CExample}{例 2(稍变形)}}
		\textbf{\textcolor{CExample}{题目:}}
		设 $f(x)=\dfrac{1}{x+1}$,定义域为 $[0,+\infty)$.求 $y=f(|x|)$ 的单调区间.
		\textbf{\textcolor{CExample}{解题步骤:}}
		\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
			\item[\textbf{第一步(写出 $f(|x|)$ 解析式):}]
				由 $f(x)=\frac{1}{x+1}$,得 $f(|x|)=\frac{1}{|x|+1}$.
				\par\textbf{理由:} 替换自变量.
			\item[\textbf{第二步(分析 $x\ge0$ 单调性):}]
				$x\ge0$ 时,$f(|x|)=\frac{1}{x+1}$,其为减函数.
				\par\textbf{理由:} $x$ 增大,分母增大,分数值减小.
			\item[\textbf{第三步(利用偶函数对称性):}]
				由于 $y=f(|x|)$ 是偶函数,在对称区间上单调性相反.故在 $x\le0$ 时函数为增函数.
				\par\textbf{理由:} 偶函数在 $(-\infty,0]$ 与 $[0,+\infty)$ 上的单调性相反(前提是在 $[0,+\infty)$ 单调).
		\end{description}
		\textbf{\textcolor{CExample}{关键结论:}}
		\textbf{$f(|x|)$ 在 $(-\infty,0]$ 上单调递增,在 $[0,+\infty)$ 上单调递减.}

		\textbf{\textcolor{CExample}{例 3(综合应用)}}
		\textbf{\textcolor{CExample}{题目:}}
		设 $f(x)=x^2-2x$,定义域为 $[0,+\infty)$.(1)求 $y=f(|x|)$ 的解析式并画出草图;(2)讨论方程 $f(|x|)=m$ 的实根个数.
		\textbf{\textcolor{CExample}{解题步骤:}}
		\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
			\item[\textbf{第一步(求 $f(|x|)$ 解析式):}]
				替换得 $f(|x|)=|x|^2-2|x| = x^2 - 2|x|$(因为 $|x|^2=x^2$).
				\[
					f(|x|) =
					\begin{cases}
						x^2-2x, & x\ge0,\\ x^2+2x, & x<0.
				\end{cases}
				\]
				\par\textbf{理由:} 当 $x\ge0$ 时 $|x|=x$;当 $x<0$ 时 $|x|=-x$ 代入.
			\item[\textbf{第二步(绘制草图):}]
				在 $x\ge0$ 时,$f(x)=x^2-2x=(x-1)^2-1$,开口向上,顶点 $(1,-1)$,过 $(0,0)$ 和 $(2,0)$.利用偶函数翻折得到 $x<0$ 部分:点对称.
				\par\textbf{理由:} 图像关于 $y$ 轴对称,只需画出右侧,然后对称到左侧.
			\item[\textbf{第三步(根个数讨论):}]
				方程 $f(|x|)=m$ 的根个数即直线 $y=m$ 与 $y=f(|x|)$ 图像的交点个数.由图像:
				\begin{itemize}[leftmargin=*]
				\item $m<-1$ 时:与图像无交点,根个数 0;
				\item $m=-1$ 时:与图像有 $2$ 个交点(左右顶点);
				\item $-1<m<0$ 时:$4$ 个交点;
				\item $m=0$ 时:$3$ 个交点($x=-2,0,2$);
				\item $m>0$ 时:$2$ 个交点.
			\end{itemize}
				\par\textbf{理由:} 数形结合,通过水平线截取图像.
		\end{description}
		\textbf{\textcolor{CExample}{关键结论:}}
		\textbf{$f(|x|)$ 的分段解析式为 $f(|x|)=x^2-2|x|$,图像呈“W”形,关于 $y$ 轴对称;方程根的个数与 $m$ 的关系如上.}
	\end{examplebox}
