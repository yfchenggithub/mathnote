\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{变换 $y=f(|x|)$ 将 $y=f(x)$ 在 $x\ge 0$ 的部分保留并关于 $y$ 轴对称翻折，生成一个偶函数，其值域与原函数在 $x\ge0$ 的值域相同。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（定义域条件）：} 函数 $f(x)$ 的定义域必须包含区间 $[0,+\infty)$。
		\item \textbf{条件 2（变换前提）：} 适用于图像变换、偶函数构造、含绝对值函数的化简等场景。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（翻折方向）：} 保留右侧，翻折左侧；不要弄反。
		\item \textbf{检查项（值域确认）：} $f(|x|)$ 的值域等于 $f(x)$ 在 $x\ge0$ 时的值域，可用于验证。
	\end{itemize}

\end{summarybox}
