\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（符号判断）：}}] 误以为 $F'(x)$ 的符号同时受分子 $f'(x)-f(x)$ 和分母 $e^x$ 的影响，从而无法确定单调性。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母恒正）：}}] 分母 $e^x>0$ 恒成立，所以 $F'(x)$ 的符号完全由分子 $f'(x)-f(x)$ 决定。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略非零）：}}] 忽略了指数函数值域为正的基本性质，导致复杂化。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（单调方向）：}}] 混淆 $f'(x)-f(x)>0$ 对应 $F(x)$ 递增还是递减。
		\item[\textbf{\textcolor{CTrap}{正确理解（同号确定）：}}] 因为 $e^x>0$，所以 $F'(x)$ 与 $f'(x)-f(x)$ 同号：$>0$ 则递增，$<0$ 则递减。
		\item[\textbf{\textcolor{CTrap}{错因分析（求导记忆）：}}] 商法则求导结果处理不当，或对单调性与导数符号的关系不熟练。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（构造形式）：}}] 在解不等式 $f(x) > C e^x$ 时，直接尝试对 $f(x)$ 求导或移项，而未联想到构造 $F(x)$。
		\item[\textbf{\textcolor{CTrap}{正确理解（目标转化）：}}] 应将不等式化为 $\dfrac{f(x)}{e^x} > C$，即 $F(x) > C$，再借助 $F(x)$ 的单调性求解。
		\item[\textbf{\textcolor{CTrap}{错因分析（定势不足）：}}] 未形成“看到 $f'(x)-f(x)$ 就构造 $F(x)$”的条件反射，导致解题方向错误。
	\end{description}
\end{trapbox}
