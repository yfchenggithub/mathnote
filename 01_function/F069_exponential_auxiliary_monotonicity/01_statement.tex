\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（函数可导）：}}] 函数 $f(x)$ 在区间 $I$ 上可导。
		\item[\textbf{\textcolor{CStatement}{条件2（不等式可化）：}}] 已知关于 $f'(x)$ 与 $f(x)$ 的不等式可化为 $f'(x)-f(x) > 0$（或 $<0$, $\geq 0$, $\leq 0$）的形式。
	\end{description}
	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（构造单调性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			构造函数 $F(x)=\dfrac{f(x)}{e^x}$，其导数符号与 $f'(x)-f(x)$ 一致，从而判断单调性。
			\[
				F(x)=\frac{f(x)}{e^x},\quad F'(x)=\frac{f'(x)-f(x)}{e^x}.
			\]
			因此：
			\begin{itemize}
			\item 若 $f'(x)-f(x)>0$，则 $F(x)$ 单调递增；
			\item 若 $f'(x)-f(x)<0$，则 $F(x)$ 单调递减。
		\end{itemize}
			可用于比较函数值或解抽象不等式。

		\item[\textbf{\textcolor{CStatement}{常见变形（等价形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $F(x)$ 写为 $e^{-x}f(x)$，导数符号相同。
			\[
				G(x)=e^{-x}f(x),\quad G'(x)=e^{-x}\bigl(f'(x)-f(x)\bigr).
			\]
			$G'(x)$ 的符号由 $f'(x)-f(x)$ 决定，单调性与 $F(x)$ 一致。
	\end{description}
\end{statementbox}
