\begin{examplebox}
    \textbf{\textcolor{CExample}{例 1（最基础：3 步求值域）}}

    \textbf{题目：} 求 $y=\dfrac{2x+1}{x-1}$ 的值域，并写出图像对称中心。

    \textbf{解：}
    \begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
        \item[\textbf{第 1 步（先看限制）：}] $x\ne1$；且 $a=2,b=1,c=1,d=-1$，$ad=-2\ne1=bc$，可以用结论。
        \item[\textbf{第 2 步（分离常数）：}]
            \[
                2x+1=2(x-1)+3,\qquad
                y=2+\frac3{x-1}.
            \]
        \item[\textbf{第 3 步（读出答案）：}] $\dfrac3{x-1}$ 能取遍所有非零实数，所以 $y\ne2$；中心是 $(1,2)$。
    \end{description}
    \textbf{答案：} 值域 $(-\infty,2)\cup(2,+\infty)$；对称中心 $(1,2)$。

    \textbf{\textcolor{CExample}{例 2（参数题：别漏掉常数情况）}}

    \textbf{题目：} 已知 $y=\dfrac{3x+m}{x+2}$（$x\ne-2$）的值域是“除 $3$ 外的所有实数”，求 $m$ 的范围。

    \textbf{解：}
    \begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
        \item[\textbf{第 1 步（先拆开）：}]
            \[
                3x+m=3(x+2)+(m-6),\qquad
                y=3+\frac{m-6}{x+2}.
            \]
        \item[\textbf{第 2 步（为什么要分情况）：}] 如果 $m=6$，剩下的分式项就是 $0$，此时 $y$ 恒等于 $3$，值域只有 $\{3\}$，与题目相反。
        \item[\textbf{第 3 步（其余情况）：}] 如果 $m\ne6$，$\dfrac{m-6}{x+2}$ 能取遍所有非零数，值域恰好为 $y\ne3$。
    \end{description}
    \textbf{答案：} $\boxed{m\ne6}$。

    \textbf{题目：} 求 $y=\dfrac{2x+1}{x-1}$ 在 $x\ge2$ 时的值域。

    \textbf{解：}
    \begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
        \item[\textbf{第 1 步（仍然先拆）：}] $y=2+\dfrac3{x-1}$。
        \item[\textbf{第 2 步（利用 $x$ 的新限制）：}] $x\ge2$ 意味着 $x-1\ge1$，因此 $0<\dfrac3{x-1}\le3$。
        \item[\textbf{第 3 步（两边加 $2$）：}] $2<y\le5$。$x=2$ 时 $y=5$，而 $3/(x-1)$ 始终大于 $0$，所以 $y=2$ 仍取不到。
    \end{description}
    \textbf{答案：} $\boxed{(2,5]}$。注意这里\textbf{不是}“除了 $2$ 的所有实数”。

    \textbf{\textcolor{CExample}{例 4（反向验证：某个函数值能否出现？）}}

    \textbf{题目：} 对 $y=\dfrac{x+3}{2x-4}$，判断 $y=\dfrac12$ 和 $y=2$ 是否能取到，并求图像对称中心。

    \textbf{解：}
    \begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
        \item[\textbf{第 1 步（拆开）：}]
            \[
                y=\frac12+\frac5{2(x-2)}\qquad(x\ne2).
            \]
        \item[\textbf{第 2 步（判断 $y=1/2$）：}] 后面一项不等于 $0$，所以 $y=1/2$ 不可能。
        \item[\textbf{第 3 步（判断 $y=2$）：}] 令 $\dfrac{x+3}{2x-4}=2$，得 $x+3=4x-8$，所以 $x=11/3$；它不等于 $2$，因而确实能取到。
        \item[\textbf{第 4 步（读中心）：}] 不能取的 $x$ 是 $2$，不能取的 $y$ 是 $1/2$，中心为 $(2,1/2)$。
    \end{description}
    \textbf{答案：} $y=1/2$ 不能取到；$y=2$ 可以取到；中心 $\boxed{(2,\,1/2)}$。
