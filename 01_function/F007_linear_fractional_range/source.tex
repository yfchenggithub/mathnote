% F007  一次分式函数值域公式：分离常数法
% 简短、独立的结论记录（详细说明见 01--06）
\[
\begin{gathered}
 y=\frac{ax+b}{cx+d},\quad a,b,c,d\in\mathbb R,\quad c\ne0,\quad ad\ne bc,\\
 x\ne-\frac dc,\qquad
 y=\frac ac+\frac{bc-ad}{c(cx+d)},\\
 \boxed{\text{值域：}\;\left\{y\in\mathbb R\mid y\ne\frac ac\right\}},\qquad
 \text{图像对称中心：}\;\left(-\frac dc,\frac ac\right).
\end{gathered}
\]
