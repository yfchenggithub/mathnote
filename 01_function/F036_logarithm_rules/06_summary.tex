\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对数运算将乘除幂转化为加减倍，换底公式统一底数。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（底数真数满足）：} 所有底数 $a,c>0$ 且 $a,c\neq1$；真数 $M,N,b>0$。
		\item \textbf{条件 2（指数为任意实数）：} 幂的对数中指数 $m\in\mathbb{R}$ 均成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（勿缺定义域）：} 每次运算前确认真数、底数符合条件。
		\item \textbf{检查项（活用逆运算）：} 合并与展开要能互推，避免 $\log_a(M+N)$ 错误。
	\end{itemize}

\end{summarybox}
