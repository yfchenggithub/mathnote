\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CTrap}{易错点一（真数不检查）：}}]
			\textbf{错法：} $\log_2[(-2)\times(-4)]=\log_2(-2)+\log_2(-4)$。

			\textbf{为什么错？} 左边是 $\log_2 8=3$，有意义；右边出现负数的对数，根本不能计算。\textbf{记住：} 拆成几个对数后，每个真数也必须大于 $0$。

		\item[\textbf{\textcolor{CTrap}{易错点二（把加法也拆开）：}}]
			\textbf{错法：} $\log_2(1+1)=\log_2 1+\log_2 1$。

			\textbf{为什么错？} 左边 $\log_2 2=1$，右边 $0+0=0$，显然不相等。\textbf{记住：} 可以拆真数的\textbf{乘、除、乘方}，不能直接拆\textbf{加、减}。

		\item[\textbf{\textcolor{CTrap}{易错点三（换底颠倒）：}}]
			\textbf{错法：} $\log_4 8=\frac{\log_2 4}{\log_2 8}=\frac23$。

			\textbf{为什么错？} $4^{3/2}=8$，所以正确答案是 $\frac32$。\textbf{记住：} 换底后\textbf{上面是原真数，下边是原底数}：$\frac{\log_2 8}{\log_2 4}=\frac32$。

		\item[\textbf{\textcolor{CTrap}{易错点四（平方后忘记检查）：}}]
			\textbf{错法：} $\log_2((-2)^2)=2\log_2(-2)$。

			\textbf{为什么错？} 左边 $\log_2 4=2$，右边的 $\log_2(-2)$ 没有意义。\textbf{记住：} $\log_a M^m=m\log_a M$ 的常用条件是 $M>0$，不能见到指数就随便移出来。
	\end{description}
\end{trapbox}
