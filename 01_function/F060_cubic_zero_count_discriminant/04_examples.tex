\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	判断函数\(f(x)=x^3-3x+1\)的零点个数.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导函数）：}]
			\[
				f'(x)=3x^2-3.
			\]
			\par\textbf{理由：} 由导数公式可得.
		\item[\textbf{第二步（求极值点）：}]
			令\(f'(x)=0\)得\(x=\pm1\).
			\[
				f'(x)=0\ \Rightarrow\ x=-1,\ x=1.
			\]
			\par\textbf{理由：} 解一元二次方程.
		\item[\textbf{第三步（计算极值乘积）：}]
			计算\(f(-1)=(-1)^3-3(-1)+1=3\), \(f(1)=1-3+1=-1\).
			\[
				f(-1)=3,\quad f(1)=-1,\quad f(-1)f(1)=-3<0.
			\]
			\par\textbf{理由：} 代入函数解析式.
		\item[\textbf{第四步（下结论）：}]
			由零点判定定理，极值乘积为负，故函数有三个零点.
			\[
				\Rightarrow N=3.
			\]
			\par\textbf{理由：} 结论直接应用.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{\(f(x)=x^3-3x+1\)有三个零点.}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	判断函数\(f(x)=x^3-3x+2\)的零点个数，并指出重根位置.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导函数）：}]
			\[
				f'(x)=3x^2-3.
			\]
			\par\textbf{理由：} 同上.
		\item[\textbf{第二步（求极值点）：}]
			\[
				f'(x)=0\ \Rightarrow\ x=-1,\ x=1.
			\]
			\par\textbf{理由：} 同上.
		\item[\textbf{第三步（计算极值乘积）：}]
			\[
				f(-1)=(-1)^3-3(-1)+2=4,\quad f(1)=1-3+2=0.
			\]
			\[
				f(-1)f(1)=0.
			\]
			\par\textbf{理由：} 代入计算.
		\item[\textbf{第四步（判零点个数与重根）：}]
			乘积为零，故有两个零点，且极值点处为零的点\(x=1\)是重根.
			\[
				\Rightarrow N=2.
			\]
			\par\textbf{理由：} 结论中当乘积为零时有两个零点，并包含一个重根.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{\(f(x)=x^3-3x+2\)有两个零点，其中\(x=1\)为二重根.}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数\(f(x)=x^3-3x+k\)（\(k\in\mathbb{R}\)），讨论\(k\)取何值时，\(f(x)\)分别有一个、两个、三个零点.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导与极值点）：}]
			\[
				f'(x)=3x^2-3,\quad f'(x)=0\ \Rightarrow\ x=-1,\ x=1.
			\]
			\par\textbf{理由：} 导数计算.
		\item[\textbf{第二步（极值乘积）：}]
			\[
				f(-1)=k+2,\quad f(1)=k-2,\quad f(-1)f(1)=k^2-4.
			\]
			\par\textbf{理由：} 代入含参解析式.
		\item[\textbf{第三步（分类讨论零点个数）：}]
			根据乘积\(k^2-4\)的符号，由判定定理得：
			\[
				\begin{cases}
					k^2-4<0\ \Rightarrow\ -2<k<2\ \Rightarrow\ N=3,\\ k^2-4=0\ \Rightarrow\ k=\pm2\ \Rightarrow\ N=2,\\ k^2-4>0\ \Rightarrow\ k>2\ \vee\ k<-2\ \Rightarrow\ N=1.
			\end{cases}
			\]
			其中\(N\)为零点个数.
			\par\textbf{理由：} 零点判定定理直接应用，结合解不等式.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{当\(-2<k<2\)时，三个零点；当\(k=\pm2\)时，两个零点（一个重根，\(k=2\)时重根\(x=1\)，\(k=-2\)时重根\(x=-1\)）；当\(k>2\)或\(k<-2\)时，一个零点.}
\end{examplebox}
