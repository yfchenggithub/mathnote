\begin{statementbox}
	\begin{center}\input{assets/tikz/F022_period_translation}\end{center}
\textbf{以下各式分别作为独立条件使用，所得 $T$ 是一个正周期，不保证最小。}\par
\medskip
\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（平移封闭性）：}}]
			函数$f(x)$的定义域$D$满足：若$x\in D$，则$x\pm a\in D$，$x\pm b\in D$（其中$a,b$为非零实常数）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（加常数型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量加常数$|a|$后函数值重复出现。
			\[
				f(x+a)=f(x)
			\]
			\[
				T=|a|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（反号型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量加$|a|$后函数值变为相反数，再重复需两倍。
			\[
				f(x+a)=-f(x)
			\]
			\[
				T=2|a|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（倒数型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量加$|a|$后函数值变为倒数（或负倒数），再重复需两倍。
			\[
				f(x+a)=\pm\frac{1}{f(x)}\;(f(x)\neq0)
			\]
			\[
				T=2|a|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论四（差值型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两侧自变量差为定值$|a-b|$时函数值相等，该差即为周期。
			\[
				f(x+a)=f(x+b)\;(a\neq b)
			\]
			\[
				T=|a-b|.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{推广结论（整数倍周期）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{推广结论（整数倍周期）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若$T$是一个周期，则它的任意非零整数倍也是周期。
			\[
				f(x+kT)=f(x),
				\qquad
				k\in\mathbb{Z},\;k\neq0.
			\]
	\end{description}

\end{statementbox}

\textbf{\textcolor{CStatement}{新增：两类分式型，连续作用两次还原}}

设 $a\ne0$，非空定义域 $D$ 满足 $x\in D\Rightarrow x\pm a\in D$；下列等式对每个 $x\in D$ 成立。两种条件分别使用。

\textbf{分式型一（要求处处 $f(x)\ne1$）：}
\[
 f(x+a)=\frac{f(x)+1}{f(x)-1}
 \quad\Longrightarrow\quad
 \boxed{f(x+2a)=f(x),\quad T=2|a|.}
\]

\textbf{分式型二（要求处处 $f(x)\ne-1$）：}
\[
 f(x+a)=\frac{1-f(x)}{1+f(x)}
 \quad\Longrightarrow\quad
 \boxed{f(x+2a)=f(x),\quad T=2|a|.}
\]
