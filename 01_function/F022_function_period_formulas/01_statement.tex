\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（平移封闭性）：}}]
			函数$f(x)$的定义域$D$满足：若$x\in D$，则$x\pm a\in D$，$x\pm b\in D$（其中$a,b$为非零实常数）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（加常数型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量加常数$|a|$后函数值重复出现。
			\[
				f(x+a)=f(x)
			\]
			\[
				T=|a|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（反号型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量加$|a|$后函数值变为相反数，再重复需两倍。
			\[
				f(x+a)=-f(x)
			\]
			\[
				T=2|a|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（倒数型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			自变量加$|a|$后函数值变为倒数（或负倒数），再重复需两倍。
			\[
				f(x+a)=\pm\frac{1}{f(x)}\;(f(x)\neq0)
			\]
			\[
				T=2|a|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论四（差值型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两侧自变量差为定值$|a-b|$时函数值相等，该差即为周期。
			\[
				f(x+a)=f(x+b)\;(a\neq b)
			\]
			\[
				T=|a-b|.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{推广结论（整数倍周期）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{推广结论（整数倍周期）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若$T$是一个周期，则它的任意非零整数倍也是周期。
			\[
				f(x+kT)=f(x),
				\qquad
				k\in\mathbb{Z},\;k\neq0.
			\]
	\end{description}

\end{statementbox}
