\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	求函数 $f(x)=2x^3+5x^2-4x+1$ 的导函数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分项求导）：}] 根据和差法则，$f'(x)=(2x^3)'+(5x^2)'-(4x)'+(1)'$。
			\par\textbf{理由：}
			和差法则允许将和的导数拆成导数的和。
		\item[\textbf{第二步（幂函数求导）：}] 分别计算：$(2x^3)'=6x^2$，$(5x^2)'=10x$，$(4x)'=4$，$(1)'=0$。
			\par\textbf{理由：}
			幂函数求导公式 $(x^n)'=nx^{n-1}$，常数导数为0。
		\item[\textbf{第三步（合并结果）：}] 合并各项得 $f'(x)=6x^2+10x-4$。
			\par\textbf{理由：}
			和差法则的逆向。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$f'(x)=6x^2+10x-4$}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	求函数 $y=x^2\cos x$ 的导数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别乘积）：}] 令 $u=x^2$，$v=\cos x$，则 $y=uv$。
			\par\textbf{理由：}
			符合乘积结构，应使用乘积法则。
		\item[\textbf{第二步（分别求导）：}] 计算 $u'=2x$，$v'=-\sin x$。
			\par\textbf{理由：}
			幂函数和三角函数的导数公式。
		\item[\textbf{第三步（套用公式）：}] 代入乘积法则 $y'=u'v+uv'$，得 $y'=2x\cos x+x^2(-\sin x)=2x\cos x - x^2\sin x$。
			\par\textbf{理由：}
			乘积法则直接套用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y'=2x\cos x - x^2\sin x$}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	求函数 $y=\frac{\sin x}{x+1}$ 的导数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别分式）：}] 令 $u=\sin x$，$v=x+1$，分母 $v(x)\neq 0$。
			\par\textbf{理由：}
			商法则适用条件：分母不为零。
		\item[\textbf{第二步（分别求导）：}] 计算 $u'=\cos x$，$v'=1$。
			\par\textbf{理由：}
			三角和幂函数求导公式。
		\item[\textbf{第三步（套用商公式）：}] 代入商法则 $y'=\frac{u'v-uv'}{v^2}$，得 $y'=\frac{\cos x\cdot(x+1)-\sin x\cdot1}{(x+1)^2}$，化简得 $y'=\frac{(x+1)\cos x-\sin x}{(x+1)^2}$。
			\par\textbf{理由：}
			商法则直接应用，注意分母平方。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y'=\frac{(x+1)\cos x-\sin x}{(x+1)^2}$}
\end{examplebox}
