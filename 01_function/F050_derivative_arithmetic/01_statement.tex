\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（可导条件）：}}] 函数 $u(x), v(x)$ 在点 $x$ 处可导。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（和差求导）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			多个函数加减求导可分别求导再相加减。
			\[
				(u\pm v)' = u' \pm v'.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（乘积求导）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			前导后不导加前不导后导。
			\[
				(uv)' = u'v + uv'.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（商求导）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			子导母不导减子不导母导，分母平方。
			\[
				\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2} \quad (v(x) \neq 0).
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒（分母条件）：}}
	分母 $v(x)$ 必须不为零，且 $v(x)$ 可导。
\end{statementbox}
