\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{导数四则运算是“加减直接拆，乘积交叉乘，商要平方且分子有减号”。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（可导条件）：} 函数 $u$ 和 $v$ 在 $x$ 处可导。
		\item \textbf{条件2（分母非零）：} 使用商法则时，必须成立 $v(x)\neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（三项易错）：} 乘积漏项、商符号倒置、忘分母平方。
		\item \textbf{检查项（公式核对）：} 使用前确认可导和分母非零，使用后核对中间项是否齐全。
	\end{itemize}
\end{summarybox}
