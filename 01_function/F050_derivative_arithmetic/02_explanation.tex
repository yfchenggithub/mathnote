\begin{explanationbox}

	\textbf{\textcolor{CExplanation}{一句话直觉}}

	\textbf{加减求导可拆开，乘积求导交叉乘，商求导分母平方子导母不导减子不导母导。}

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（和差法则）：}] 对两个函数和（或差）求导，等于分别求导再相加（减）。步骤上与先后次序无关。
		\item[\textbf{要点二（乘积法则）：}] 乘积求导必须同时保留原函数和导数的交叉项，“前导后不导”加上“前不导后导”，缺一不可。
		\item[\textbf{要点三（商法则）：}] 商的导数使用专门公式：分母是原分母的平方，分子是“子的导数乘母减去子乘母的导数”。注意分母不能为零。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（变化率叠加）}}
	若 $u(x)$ 和 $v(x)$ 代表两个独立变化量，则其和的变化率等于各自变化率之和；积的变化率则与两个量的当前值与变化率均有关，不可分割。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（线性特性）}}
	求导算子对加法和数乘满足线性性，即 $(cu)'=cu'$ 和 $(u+v)'=u'+v'$。但乘法不满足线性，必须按照乘积法则计算，不能拆开。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（基础计算）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接求导）：} 给定函数解析式，利用四则法则求导函数。高考必考。
		\item \textbf{考法二（切线斜率）：} 求某点的导数值即切线斜率，需要先正确求导后再代入。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}

	\textbf{乘积法则可以记忆为“各求导一次依次乘对方原函数再相加”，类似于多项展开但不相同。商法则可通过把 $\frac{u}{v}$ 看作 $u \cdot v^{-1}$ 再结合乘积法则推导。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（多项式求导）：} 对多项式函数，逐项求导即可，由和差法则与幂函数导数完成。
		\item \textbf{场景二（分式求导）：} 对形如 $\frac{f}{g}$ 的函数，直接使用商法则，注意分母平方和分子符号。
	\end{itemize}
\end{explanationbox}
