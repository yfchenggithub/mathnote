\begin{examplebox}
	\textbf{\textcolor{CExample}{例1｜一眼求值（直接还原）}}
	计算 $2^{\log_2 3}$。

	\textbf{看底数：}外面是 $2$，$\log$ 的底数也是 $2$；里面的 $3>0$。因此
	\[
	\boxed{2^{\log_2 3}=3}.
	\]
	\textbf{想一想：}不必先算出 $\log_2 3$ 等于多少，知道它是“让 $2$ 的乘方等于 $3$ 的指数”就够了。

	\medskip
	\textbf{\textcolor{CExample}{例2｜底数不同，先统一（考试常考）}}
	计算 $4^{\log_2 3}$。

	\textbf{第一步：}这里的底数是 $4$ 和 $2$，\textbf{不能直接写成 $3$}。但 $4=2^2$。
	\textbf{第二步：}利用幂的乘方法则：
	\[
	\begin{aligned}
	4^{\log_2 3}
	&=(2^2)^{\log_2 3}\\
	&=(2^{\log_2 3})^2\\
	&=3^2=\boxed{9}.
	\end{aligned}
	\]
	\textbf{图中记住关键变化：}

	\end{center}
	\textbf{同类再练：}因为 $9=3^2$，所以 $9^{\log_3 5}=(3^{\log_3 5})^2=5^2=25$。

	\medskip
	\textbf{\textcolor{CExample}{例3｜解方程（先看有无意义）}}
	解 $5^{\log_5(2x-1)}=8$。

	\textbf{先写条件：}$2x-1>0$，即 $x>\frac12$。
	\textbf{再同底还原：}
	\[
	5^{\log_5(2x-1)}=2x-1=8.
	\]
	所以 $x=\frac92$。检验：$\frac92>\frac12$，\textbf{符合条件}。
	\[
	\boxed{x=\frac92}.
	\]

	\medskip
	\textbf{\textcolor{CExample}{例4｜原有进阶例题（含二次方程）}}
	解 $2^{\log_2(x^2-3x+2)}=4$。

	\textbf{第一步：}对数要求 $x^2-3x+2>0$。
	\textbf{第二步：}在这个条件下同底还原，得到
	\[
	x^2-3x+2=4\quad\Longleftrightarrow\quad x^2-3x-2=0.
	\]
	\textbf{第三步：}用求根公式得
	\[
	\boxed{x=\frac{3+\sqrt{17}}2\quad\text{或}\quad x=\frac{3-\sqrt{17}}2}.
	\]
	\textbf{最后检查：}这两个解都满足 $x^2-3x+2=4>0$，所以都保留。
\end{examplebox}
