\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话快记}}
	\textbf{同底才能消，真数必须正；先取对数再乘方，原数就回来。}

	\[
	\boxed{a^{\log_a N}=N}\quad(N>0),
	\qquad
	\boxed{\log_a(a^x)=x}\quad(x\in\mathbb{R}).
	\]
	两式都要求 $a>0$ 且 $a\neq 1$。

	\medskip
	\textbf{\textcolor{CSummary}{考场三步快用}}
	\begin{enumerate}[leftmargin=*, itemsep=0.3em, topsep=0.3em]
		\item \textbf{先看：}是不是“指数套对数”或“对数套指数”？
		\item \textbf{再查：}底数相同吗？对数里面的数大于 $0$ 吗？
		\item \textbf{最后算：}同底直接还原；不同底数先变形。
	\end{enumerate}

	\textbf{\textcolor{CSummary}{一分钟自测}}
	$3^{\log_3 7}=\boxed{7}$；
	$4^{\log_2 3}=\boxed{9}$。
	\textbf{为什么第二题不是 $3$？}因为 $4=2^2$，还原后还要平方。
\end{summarybox}
