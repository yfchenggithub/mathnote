\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（符号直接套用）}}] 错误认为由 $f'(x)+f(x)>0$ 可直接推出 $f(x)$ 严格递增。
		\item[\textbf{\textcolor{CTrap}{正确理解（对象区分）：}}] $f'(x)+f(x)>0$ 说明的是 $e^x f(x)$ 单调递增，而不是 $f(x)$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆单调性）：}}] 未构造 $e^x f(x)$，直接对 $f(x)$ 进行导数符号判断，忽略了乘积构造的转化作用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（构造混淆）}}] 遇到 $f'(x)-f(x)>0$ 仍套用 $e^x$ 构造。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号对称）：}}] 对于 $f'(x)-f(x)$，应构造 $e^{-x}f(x)$，其导数为 $e^{-x}[f'(x)-f(x)]$；同理 $f'(x)+f(x)$ 用 $e^x f(x)$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆错误）：}}] 误以为所有线性组合都用 $e^x$，未考虑指数函数的求导特点。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（常数条件误判）}}] 由 $f'(x)+f(x)=0$ 直接得 $f(x)$ 为常数。
		\item[\textbf{\textcolor{CTrap}{正确理解（构造常数）：}}] $f'(x)+f(x)=0$ 推出 $e^x f(x)$ 为常数，即 $e^x f(x)=C$，从而 $f(x)=Ce^{-x}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略乘积因子）：}}] 没有构造 $e^x f(x)$，而是试图直接积分 $f'(x)=-f(x)$，对高中生可能造成方法混乱。
	\end{description}
\end{trapbox}
