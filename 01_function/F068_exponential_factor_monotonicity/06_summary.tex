\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{条件中出现 $f'(x)+f(x)$ 时，构造 $e^x f(x)$，利用其导数将条件转化为单调性。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（函数可导）：} $f(x)$ 在相关区间上可导。
		\item \textbf{条件2（符号已知）：} 已知 $f'(x)+f(x) > 0$（或 $<0$、$=0$）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（对象区分）：} $f'(x)+f(x)$ 的符号影响的是 $e^x f(x)$ 的单调性，而非 $f(x)$。
		\item \textbf{检查项（构造匹配）：} 若条件为 $f'(x)-f(x)$，应构造 $e^{-x}f(x)$ 而非 $e^x f(x)$。
	\end{itemize}
\end{summarybox}
