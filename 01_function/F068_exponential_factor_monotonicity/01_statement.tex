\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（函数可导）：}}] $f(x)$ 在区间 $I$ 上可导。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（导数结构）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			构造 $e^x f(x)$ 将 $f'(x)+f(x)$ 与导数建立联系。
			\[
				F(x)=e^{x}f(x) \quad \Rightarrow \quad F'(x)=e^{x}[f'(x)+f(x)].
			\]
			\begin{itemize}
			\item 若 $f'(x)+f(x)>0$ 在 $I$ 上恒成立，则 $F(x)$ 严格递增；
			\item 若 $f'(x)+f(x)<0$ 在 $I$ 上恒成立，则 $F(x)$ 严格递减；
			\item 若 $f'(x)+f(x)=0$ 恒成立，则 $F(x)$ 为常数。
		\end{itemize}
		\item[\textbf{\textcolor{CStatement}{常见变形（常数因子）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $e^x$ 换为 $e^{kx}$，导数变为 $e^{kx}[f'(x)+kf(x)]$。
			\[
				G(x)=e^{kx}f(x) \quad \Rightarrow \quad G'(x)=e^{kx}[f'(x)+kf(x)].
			\]
	\end{description}
\end{statementbox}
