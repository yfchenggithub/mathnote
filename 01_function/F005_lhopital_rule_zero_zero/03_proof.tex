\begin{proofbox}
\textbf{\textcolor{CProof}{第一部分：$0/0$ 型的证明范围}}\par\medskip
\textbf{\textcolor{CProof}{证明一：最基础的一类——直接从导数定义出发}}
\begin{center}\input{assets/tikz/F005_derivative_secants.tex}\end{center}


\textbf{适用条件：} $f(a)=g(a)=0$，$f'(a)$、$g'(a)$ 都存在，且 $g'(a)\ne0$。

\begin{description}[style=nextline, leftmargin=2.6em, labelsep=0.4em, itemsep=0.55em]
    \item[\textbf{第1步：写出原式。}]
    因为 $f(a)=g(a)=0$，当 $x\ne a$ 时，
    \[
    \frac{f(x)}{g(x)}
    =\frac{f(x)-f(a)}{g(x)-g(a)}.
    \]
    \textbf{为什么这样写？} 为了凑出导数定义中的“函数值的变化量除以 $x$ 的变化量”。

    \item[\textbf{第2步：上下同时除以 $x-a$。}]
    \[
    \frac{f(x)-f(a)}{g(x)-g(a)}
    =\frac{\dfrac{f(x)-f(a)}{x-a}}
           {\dfrac{g(x)-g(a)}{x-a}}.
    \]
    这只是普通分式运算，式子并没有改变。

    \item[\textbf{第3步：让 $x$ 接近 $a$。}]
    依照导数定义，
    \[
    \lim_{x\to a}\frac{f(x)-f(a)}{x-a}=f'(a),\qquad
    \lim_{x\to a}\frac{g(x)-g(a)}{x-a}=g'(a).
    \]
    因为 $g'(a)\ne0$，可以把两个结果相除，得到
    \[
    \boxed{\displaystyle\lim_{x\to a}\frac{f(x)}{g(x)}
      =\frac{f'(a)}{g'(a)}.}
    \]
\end{description}

\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{证明二：两边都含相同的零因子——因式分解与求导对照}}
\begin{center}\input{assets/tikz/F005_common_zero_factor.tex}\end{center}


\textbf{适用情况：} 对于多项式，若分子、分母都能写成
\[
 f(x)=(x-a)^m A(x),\qquad g(x)=(x-a)^m B(x),
\]
其中 $m$ 是正整数，$A(x),B(x)$ 是多项式，且 $A(a),B(a)$ 都不为 $0$。

\begin{description}[style=nextline, leftmargin=2.6em, labelsep=0.4em, itemsep=0.55em]
    \item[\textbf{第1步：先按初中代数约分。}]
    在 $x\ne a$ 时，
    \[
       \frac{f(x)}{g(x)}=\frac{A(x)}{B(x)}.
    \]
    因而
    \[
       \lim_{x\to a}\frac{f(x)}{g(x)}=\frac{A(a)}{B(a)}.
    \]

    \item[\textbf{第2步：分别求导。}]
    按乘积求导法则，
    \[
    \begin{aligned}
    f'(x)&=(x-a)^{m-1}\bigl[mA(x)+(x-a)A'(x)\bigr],\\
    g'(x)&=(x-a)^{m-1}\bigl[mB(x)+(x-a)B'(x)\bigr].
    \end{aligned}
    \]

    \item[\textbf{第3步：同样约去公共部分。}]
    对足够接近 $a$ 且不等于 $a$ 的 $x$，
    \[
      \frac{f'(x)}{g'(x)}
      =\frac{mA(x)+(x-a)A'(x)}{mB(x)+(x-a)B'(x)}.
    \]
    因为 $A,B$ 是多项式，令 $x\to a$，带有 $(x-a)$ 的项都趋于 $0$：
    \[
    \lim_{x\to a}\frac{f'(x)}{g'(x)}
      =\frac{mA(a)}{mB(a)}=\frac{A(a)}{B(a)}.
    \]
\end{description}

这就亲手验证：\textbf{当分子、分母有同样次数的零因子时，约分法和分别求导法会得到同一个极限。}

\textbf{代入看一遍：}
\[
\begin{aligned}
\lim_{x\to1}\frac{(x-1)^2}{3(x-1)^2+2(x-1)^3}
&=\lim_{x\to1}\frac{1}{3+2(x-1)}=\frac13,\\
\lim_{x\to1}\frac{2(x-1)}{6(x-1)+6(x-1)^2}
&=\frac13.
\end{aligned}
\]
第二行正是分别求导后的式子。

\medskip
\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{证明三：一个三角函数例子——不用洛必达也能证明}}
\begin{center}\input{assets/tikz/F005_unit_circle_squeeze.tex}\end{center}


\textbf{要证明：} $\displaystyle\lim_{x\to0}\frac{\sin x}{x}=1$，其中角度按弧度计算。

\begin{description}[style=nextline, leftmargin=2.6em, labelsep=0.4em, itemsep=0.55em]
    \item[\textbf{第1步：先研究 $0<x<\frac\pi2$。}]
    画半径为 $1$ 的圆，设圆心 $O=(0,0)$，圆上的点 $A=(1,0)$、
    $B=(\cos x,\sin x)$。从 $A$ 作竖直方向的直线，与射线 $OB$ 相交于 $T=(1,\tan x)$。
    小三角形 $OAB$ 在扇形 $OAB$ 内，扇形又在大三角形 $OAT$ 内。
    用“底乘高除以 $2$”算出两个三角形的面积，再用单位圆扇形面积 $\frac{x}{2}$，便得到
    \[
      \frac{\sin x}{2}<\frac{x}{2}<\frac{\tan x}{2},
      \qquad\text{即}\qquad\sin x<x<\tan x.
    \]
    这里 $x$ 是弧度数，所用的只是圆和三角形的面积比较。

    \item[\textbf{第2步：整理成要研究的比值。}]
    因为此时 $x>0$ 且 $\cos x>0$，
    \[
      \sin x<x<\frac{\sin x}{\cos x}
      \quad\Longrightarrow\quad
      \cos x<\frac{\sin x}{x}<1.
    \]

    \item[\textbf{第3步：让角度越来越小。}]
    当 $x\to0^+$ 时，$\cos x\to1$，所以夹在 $\cos x$ 和 $1$ 之间的 $\frac{\sin x}{x}$ 也只能趋向 $1$。
    对于 $x<0$，利用 $\sin(-x)=-\sin x$，有
    \[
      \frac{\sin(-x)}{-x}=\frac{\sin x}{x},
    \]
    因此左右两边结果相同，
    \[
      \boxed{\displaystyle\lim_{x\to0}\frac{\sin x}{x}=1.}
    \]
\end{description}

\textbf{这一证明的价值：}它没有先使用 $\sin x$ 的求导公式，避免用一个依赖该极限建立的求导公式反过来证明同一个极限。
如果题目只要求计算，也可在已经掌握求导公式后用洛必达作为捷径，但不能把那种写法当作上面几何证明的替代。

\medskip
\textbf{\textcolor{CProof}{结论回扣}}

第一部分的三条路线分别说明：\textbf{导数定义能解释最简单的一类、因式分解能验证多项式的一大类、几何比较能独立解决一个经典三角式。}
它们都不等于对洛必达法则所有情况的完整证明；本页只讲高中范围内真正能够说明白的内容。

\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{第二部分：$\infty/\infty$ 型的证明范围}}

下面只证明几类\textbf{高中知识能够完整说清的具体情况}。我们不使用超出高中范围的一般证明。

\medskip
\textbf{\textcolor{CProof}{证明四：多项式除以多项式——为什么最高次项最重要？}}

\textbf{适用情况：} 设 $x\to+\infty$，分子 $f(x)$、分母 $g(x)$ 都是次数至少为 $1$ 的多项式，最高次项分别是 $Ax^m$、$Bx^n$，其中 $A,B\ne0$，$m,n$ 为正整数。

\begin{description}[style=nextline, leftmargin=2.5em, labelsep=0.4em, itemsep=0.48em]
\item[\textbf{第1步：把最高次幂提出。}]
将分子、分母写成
\[
 f(x)=x^m\bigl(A+u(x)\bigr),\qquad
 g(x)=x^n\bigl(B+v(x)\bigr),
\]
这里 $u(x)$、$v(x)$ 是低次项除以对应最高次幂后剩下的式子；因为每项都含 $\frac1x$ 或它的更高次幂，所以 $u(x)\to0$、$v(x)\to0$。
因此
\[
\frac{f(x)}{g(x)}
=x^{m-n}\frac{A+u(x)}{B+v(x)}.
\]
\textbf{第2步：分别求导，看看它会变成什么。}
求导后最高次项分别是 $mAx^{m-1}$、$nBx^{n-1}$，所以同样可以写成
\[
\frac{f'(x)}{g'(x)}
=x^{m-n}\frac{mA+u_1(x)}{nB+v_1(x)},
\]
其中 $u_1(x)\to0$、$v_1(x)\to0$；并且对足够大的 $x$，分母导数不会为 $0$。
\item[\textbf{第3步：按最高次数比较。}]
\begin{itemize}[leftmargin=*, itemsep=0.2em]
\item $m<n$：$x^{m-n}\to0$，两种比值的极限都为 $0$；
\item $m=n$：原比值 $\to\frac AB$，导数比值 $\to\frac{mA}{nB}=\frac AB$；
\item $m>n$：$x^{m-n}\to+\infty$。两种比值都无限增大，且正负号都由 $\frac AB$ 决定，因为 $\frac mn>0$。
\end{itemize}
\end{description}

\textbf{得到的结论：}对于这类多项式分式，只用高中代数与求导就能严格说明，两种方法算出的极限相同（包括确定趋向 $+\infty$ 或 $-\infty$ 的情况）。

\textbf{最快做法：}事实上第一步往往就可以直接算完；在高考常规解题中，这通常比使用洛必达法则更省步骤。
\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{证明五：根号与 $x$——先处理绝对值再比较导数}}

考虑
\[\lim_{x\to+\infty}\frac{\sqrt{x^2+4}}{x}.\]
\textbf{第1步：直接化简。} 当 $x>0$，$\sqrt{x^2}=x$，于是
\[
\frac{\sqrt{x^2+4}}{x}=\sqrt{1+\frac4{x^2}}\longrightarrow1.
\]
\textbf{第2步：再验证分别求导的结果。}
\[
\lim_{x\to+\infty}\frac{\bigl(\sqrt{x^2+4}\bigr)'}{(x)'}
=\lim_{x\to+\infty}\frac{x}{\sqrt{x^2+4}}=1.
\]
两种结果一致，且分母导数恒为 $1$。

\textbf{眼前一亮的延伸：} 如果改为 $x\to-\infty$，就不能把 $\sqrt{x^2}$ 直接写成 $x$，因为 $\sqrt{x^2}=|x|=-x$。这时
\[
\lim_{x\to-\infty}\frac{\sqrt{x^2+4}}{x}=-1,
\qquad
\lim_{x\to-\infty}\frac{x}{\sqrt{x^2+4}}=-1.
\]
即使分子趋向 $+\infty$、分母趋向 $-\infty$，也属于两边\textbf{绝对值都无限增大}的情况，且要保留负号。
\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{证明六：$\ln x$ 与 $x$——不用一般法则也能严谨比较快慢}}

\textbf{要证：} $\displaystyle\lim_{x\to+\infty}\frac{\ln x}{x}=0$。

\begin{description}[style=nextline, leftmargin=2.5em, labelsep=0.4em, itemsep=0.45em]
\item[\textbf{第1步：构造一个更容易比较的式子。}]
在 $x\ge1$ 时，设
\[h(x)=2\sqrt{x}-\ln x.\]
\item[\textbf{第2步：只用高中求导比较它的大小。}]
\[
 h'(x)=\frac1{\sqrt{x}}-\frac1x
=\frac{\sqrt{x}-1}{x}\ge0\qquad(x\ge1).
\]
所以 $h(x)$ 随 $x$ 增大不会减小；又因为 $h(1)=2$，故 $h(x)\ge2>0$，也就是
\[\ln x\le2\sqrt{x}\qquad(x\ge1).\]
\item[\textbf{第3步：把分子分母一起除以 $x$。}]
当 $x\ge1$ 时，$\ln x\ge0$，因此
\[
 0\le\frac{\ln x}{x}\le\frac2{\sqrt{x}}\longrightarrow0.
\]
由此严格得到 $\displaystyle\lim_{x\to+\infty}\frac{\ln x}{x}=0$。
\item[\textbf{第4步：对照求导后的比值。}]
\[
 \lim_{x\to+\infty}\frac{(\ln x)'}{(x)'}
 =\lim_{x\to+\infty}\frac{1/x}{1}=0.
\]
\end{description}

\textbf{这里到底证明了什么？}我们独立证明了这个\emph{具体例子}，并验证求导法给出相同结果；\textbf{没有因此证明任意两个无穷大函数都可以直接分别求导。}
\end{proofbox}
