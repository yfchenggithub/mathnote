\begin{proofbox}
\textbf{\textcolor{CProof}{证明一：最基础的一类——直接从导数定义出发}}
\begin{center}\input{assets/tikz/F005_derivative_secants.tex}\end{center}


\textbf{适用条件：} $f(a)=g(a)=0$，$f'(a)$、$g'(a)$ 都存在，且 $g'(a)\ne0$。

\begin{description}[style=nextline, leftmargin=2.6em, labelsep=0.4em, itemsep=0.55em]
    \item[\textbf{第1步：写出原式。}]
    因为 $f(a)=g(a)=0$，当 $x\ne a$ 时，
    \[
    \frac{f(x)}{g(x)}
    =\frac{f(x)-f(a)}{g(x)-g(a)}.
    \]
    \textbf{为什么这样写？} 为了凑出导数定义中的“函数值的变化量除以 $x$ 的变化量”。

    \item[\textbf{第2步：上下同时除以 $x-a$。}]
    \[
    \frac{f(x)-f(a)}{g(x)-g(a)}
    =\frac{\dfrac{f(x)-f(a)}{x-a}}
           {\dfrac{g(x)-g(a)}{x-a}}.
    \]
    这只是普通分式运算，式子并没有改变。

    \item[\textbf{第3步：让 $x$ 接近 $a$。}]
    依照导数定义，
    \[
    \lim_{x\to a}\frac{f(x)-f(a)}{x-a}=f'(a),\qquad
    \lim_{x\to a}\frac{g(x)-g(a)}{x-a}=g'(a).
    \]
    因为 $g'(a)\ne0$，可以把两个结果相除，得到
    \[
    \boxed{\displaystyle\lim_{x\to a}\frac{f(x)}{g(x)}
      =\frac{f'(a)}{g'(a)}.}
    \]
\end{description}

\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{证明二：两边都含相同的零因子——因式分解与求导对照}}
\begin{center}\input{assets/tikz/F005_common_zero_factor.tex}\end{center}


\textbf{适用情况：} 对于多项式，若分子、分母都能写成
\[
 f(x)=(x-a)^m A(x),\qquad g(x)=(x-a)^m B(x),
\]
其中 $m$ 是正整数，$A(x),B(x)$ 是多项式，且 $A(a),B(a)$ 都不为 $0$。

\begin{description}[style=nextline, leftmargin=2.6em, labelsep=0.4em, itemsep=0.55em]
    \item[\textbf{第1步：先按初中代数约分。}]
    在 $x\ne a$ 时，
    \[
       \frac{f(x)}{g(x)}=\frac{A(x)}{B(x)}.
    \]
    因而
    \[
       \lim_{x\to a}\frac{f(x)}{g(x)}=\frac{A(a)}{B(a)}.
    \]

    \item[\textbf{第2步：分别求导。}]
    按乘积求导法则，
    \[
    \begin{aligned}
    f'(x)&=(x-a)^{m-1}\bigl[mA(x)+(x-a)A'(x)\bigr],\\
    g'(x)&=(x-a)^{m-1}\bigl[mB(x)+(x-a)B'(x)\bigr].
    \end{aligned}
    \]

    \item[\textbf{第3步：同样约去公共部分。}]
    对足够接近 $a$ 且不等于 $a$ 的 $x$，
    \[
      \frac{f'(x)}{g'(x)}
      =\frac{mA(x)+(x-a)A'(x)}{mB(x)+(x-a)B'(x)}.
    \]
    因为 $A,B$ 是多项式，令 $x\to a$，带有 $(x-a)$ 的项都趋于 $0$：
    \[
    \lim_{x\to a}\frac{f'(x)}{g'(x)}
      =\frac{mA(a)}{mB(a)}=\frac{A(a)}{B(a)}.
    \]
\end{description}

这就亲手验证：\textbf{当分子、分母有同样次数的零因子时，约分法和分别求导法会得到同一个极限。}

\textbf{代入看一遍：}
\[
\begin{aligned}
\lim_{x\to1}\frac{(x-1)^2}{3(x-1)^2+2(x-1)^3}
&=\lim_{x\to1}\frac{1}{3+2(x-1)}=\frac13,\\
\lim_{x\to1}\frac{2(x-1)}{6(x-1)+6(x-1)^2}
&=\frac13.
\end{aligned}
\]
第二行正是分别求导后的式子。

\medskip
\end{proofbox}

\begin{proofbox}
\textbf{\textcolor{CProof}{证明三：一个三角函数例子——不用洛必达也能证明}}
\begin{center}\input{assets/tikz/F005_unit_circle_squeeze.tex}\end{center}


\textbf{要证明：} $\displaystyle\lim_{x\to0}\frac{\sin x}{x}=1$，其中角度按弧度计算。

\begin{description}[style=nextline, leftmargin=2.6em, labelsep=0.4em, itemsep=0.55em]
    \item[\textbf{第1步：先研究 $0<x<\frac\pi2$。}]
    画半径为 $1$ 的圆，设圆心 $O=(0,0)$，圆上的点 $A=(1,0)$、
    $B=(\cos x,\sin x)$。从 $A$ 作竖直方向的直线，与射线 $OB$ 相交于 $T=(1,\tan x)$。
    小三角形 $OAB$ 在扇形 $OAB$ 内，扇形又在大三角形 $OAT$ 内。
    用“底乘高除以 $2$”算出两个三角形的面积，再用单位圆扇形面积 $\frac{x}{2}$，便得到
    \[
      \frac{\sin x}{2}<\frac{x}{2}<\frac{\tan x}{2},
      \qquad\text{即}\qquad\sin x<x<\tan x.
    \]
    这里 $x$ 是弧度数，所用的只是圆和三角形的面积比较。

    \item[\textbf{第2步：整理成要研究的比值。}]
    因为此时 $x>0$ 且 $\cos x>0$，
    \[
      \sin x<x<\frac{\sin x}{\cos x}
      \quad\Longrightarrow\quad
      \cos x<\frac{\sin x}{x}<1.
    \]

    \item[\textbf{第3步：让角度越来越小。}]
    当 $x\to0^+$ 时，$\cos x\to1$，所以夹在 $\cos x$ 和 $1$ 之间的 $\frac{\sin x}{x}$ 也只能趋向 $1$。
    对于 $x<0$，利用 $\sin(-x)=-\sin x$，有
    \[
      \frac{\sin(-x)}{-x}=\frac{\sin x}{x},
    \]
    因此左右两边结果相同，
    \[
      \boxed{\displaystyle\lim_{x\to0}\frac{\sin x}{x}=1.}
    \]
\end{description}

\textbf{这一证明的价值：}它没有先使用 $\sin x$ 的求导公式，避免用一个依赖该极限建立的求导公式反过来证明同一个极限。
如果题目只要求计算，也可在已经掌握求导公式后用洛必达作为捷径，但不能把那种写法当作上面几何证明的替代。

\medskip
\textbf{\textcolor{CProof}{结论回扣}}

三条路线分别说明：\textbf{导数定义能解释最简单的一类、因式分解能验证多项式的一大类、几何比较能独立解决一个经典三角式。}
它们都不等于对洛必达法则所有情况的完整证明；本页只讲高中范围内真正能够说明白的内容。

\end{proofbox}
