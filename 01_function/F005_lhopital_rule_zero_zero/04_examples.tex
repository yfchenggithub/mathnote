\begin{examplebox}
\textbf{\textcolor{CExample}{第一部分：$0/0$ 型例题（例1--例5）}}



\textbf{\textcolor{CExample}{例1：最简单的 $0$ 比 $0$，先约分更快}}

\textbf{题目：} $\displaystyle\lim_{x\to2}\frac{x^2-4}{x-2}$。

\textbf{解法一（推荐，因式分解）：}
\[
\lim_{x\to2}\frac{(x-2)(x+2)}{x-2}
=\lim_{x\to2}(x+2)=4.
\]

\textbf{解法二（分别求导，作对照）：}
\[
\lim_{x\to2}\frac{x^2-4}{x-2}
=\lim_{x\to2}\frac{2x}{1}=4.
\]
\textbf{为什么能用？} 分子、分母都趋于 $0$，在 $2$ 附近能求导，分母导数为 $1$，求导后的比值有极限。

\medskip
\textbf{\textcolor{CExample}{例2：根号式，高中更推荐有理化}}

\textbf{题目：} $\displaystyle\lim_{x\to4}\frac{\sqrt{x}-2}{x-4}$。

\textbf{解法一（推荐）：} 分子、分母都趋于 $0$。分子分母同乘 $\sqrt{x}+2$：
\[
\begin{aligned}
\frac{\sqrt{x}-2}{x-4}
&=\frac{(\sqrt{x}-2)(\sqrt{x}+2)}{(x-4)(\sqrt{x}+2)}\\
&=\frac{x-4}{(x-4)(\sqrt{x}+2)}
=\frac1{\sqrt{x}+2}.
\end{aligned}
\]
所以极限为 $\frac14$。

\textbf{解法二（求导对照）：}
\[
\lim_{x\to4}\frac{\sqrt{x}-2}{x-4}
=\lim_{x\to4}\frac{\frac1{2\sqrt{x}}}{1}
=\frac14.
\]
\textbf{提醒：} 已经能用有理化一步算出，不必刻意使用新法则。

\medskip
\begin{minipage}{\linewidth}
\textbf{\textcolor{CExample}{例3：对数与平方，求导明显省力}}

\textbf{题目：} $\displaystyle\lim_{x\to0}\frac{\ln(1+x)-x}{x^2}$，其中 $x>-1$。
\end{minipage}

\textbf{第1步：} $x\to0$ 时，$\ln(1+x)-x\to0$，$x^2\to0$。
\textbf{第2步：} 分子导数为
\[
\bigl[\ln(1+x)-x\bigr]'=\frac1{1+x}-1=-\frac{x}{1+x},
\]
分母导数为 $2x$，当 $x$ 接近 $0$ 但不等于 $0$ 时 $2x\ne0$。
\textbf{第3步：} 求导后先约去 $x$：
\[
\begin{aligned}
\lim_{x\to0}\frac{\ln(1+x)-x}{x^2}
&=\lim_{x\to0}\frac{-\frac{x}{1+x}}{2x}\\
&=\lim_{x\to0}\left(-\frac1{2(1+x)}\right)
=\boxed{-\frac12}.
\end{aligned}
\]
\textbf{关键：} 一次求导后虽表面上还是 $0$ 比 $0$，但能直接约分，没有必要再求一次导。

\medskip
\textbf{\textcolor{CExample}{例4：求导一次不够，就继续检查}}

\textbf{题目：} $\displaystyle\lim_{x\to0}\frac{x-\sin x}{x^3}$。

\textbf{第一次：} 分子、分母都趋于 $0$，分别求导：
\[
\lim_{x\to0}\frac{x-\sin x}{x^3}
=\lim_{x\to0}\frac{1-\cos x}{3x^2}.
\]
\textbf{第二次：} 新分式仍是 $0$ 比 $0$，并满足求导条件：
\[
\lim_{x\to0}\frac{1-\cos x}{3x^2}
=\lim_{x\to0}\frac{\sin x}{6x}.
\]
\textbf{第三次：} 又是 $0$ 比 $0$，再分别求导：
\[
\lim_{x\to0}\frac{\sin x}{6x}
=\lim_{x\to0}\frac{\cos x}{6}
=\boxed{\frac16}.
\]
\textbf{更快的最后一步：} 已知 $\lim_{x\to0}\frac{\sin x}{x}=1$ 时，第二次后也可直接得 $\frac16$。
\textbf{提醒：} 每一次都需要重新检查，不能只写“连续求导三次”却不管条件。

\medskip
\textbf{\textcolor{CExample}{例5：有参数时，先看系数}}

\textbf{题目：} 已知
\[
\lim_{x\to1}\frac{x^2+(k-1)x-k}{x-1}=5,
\]
求 $k$。

\textbf{解：} 分子在 $x=1$ 时等于 $1+(k-1)-k=0$，
\[
 x^2+(k-1)x-k=(x-1)(x+k).
\]
因此
\[
\lim_{x\to1}\frac{(x-1)(x+k)}{x-1}=1+k=5,
\]
所以 $\boxed{k=4}$。

\textbf{另一种检查：} 分子导数为 $2x+k-1$，分母导数为 $1$，代入 $x=1$ 同样得到 $1+k$。
\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{第二部分：$\infty/\infty$ 型例题（例6--例10）}}

\textbf{\textcolor{CExample}{例6：两边都越来越大，先比最高次项}}

\textbf{题目：} $\displaystyle\lim_{x\to+\infty}\frac{2x^2+3x+1}{x^2+1}$。

\textbf{解法一（推荐）：} 分子分母同除以 $x^2$：
\[
\lim_{x\to+\infty}\frac{2+\frac3x+\frac1{x^2}}{1+\frac1{x^2}}=\boxed{2}.
\]
\textbf{解法二（求导对照）：}
\[
\lim_{x\to+\infty}\frac{2x^2+3x+1}{x^2+1}
=\lim_{x\to+\infty}\frac{4x+3}{2x}
=2.
\]
\textbf{思考：} 对多项式分式，比较最高次数通常比求导还快；方法是为解题服务的。

\medskip

\medskip
\textbf{\textcolor{CExample}{例7：最高次数不同，结果也可以是无穷大或零}}

\textbf{题目一：} $\displaystyle\lim_{x\to+\infty}\frac{x^3+x}{x^2+1}$。

\textbf{快速解法：} 把分子拆开：$x^3+x=x(x^2+1)$，所以原式就是 $x$，极限为 $+\infty$。

\textbf{求导核对：} 分子、分母都趋于 $+\infty$，
\[
\lim_{x\to+\infty}\frac{3x^2+1}{2x}=+\infty.
\]

\textbf{题目二：} $\displaystyle\lim_{x\to+\infty}\frac{x}{x^2+1}$。

\textbf{快速解法：} $\frac{x}{x^2+1}=\frac{1/x}{1+1/x^2}\to0$；求导后 $\frac1{2x}\to0$，结果相同。

\textbf{结论：}虽然两题上下都趋于无穷大，但答案一个是无穷大，一个是零，关键是\textbf{哪边增长得更快}。

\medskip
\textbf{\textcolor{CExample}{例8：对数与一次函数，导数比直接趋于零}}

\textbf{题目：} $\displaystyle\lim_{x\to+\infty}\frac{\ln x}{x}$。

当 $x\to+\infty$，$\ln x\to+\infty$，$x\to+\infty$；并且两边可导，分母导数为 $1$，因此
\[
\lim_{x\to+\infty}\frac{\ln x}{x}
=\lim_{x\to+\infty}\frac{1/x}{1}=\boxed{0}.
\]
\textbf{不用法则也能证明：}证明六已用高中求导得出 $0\le\frac{\ln x}{x}\le\frac2{\sqrt{x}}\to0$。
\textbf{直观理解：}对数虽然一直增大，但比 $x$ 增大得慢得多。

\medskip
\textbf{\textcolor{CExample}{例9：$x$ 接近有限位置，也能出现无穷比无穷}}

\textbf{题目：} $\displaystyle\lim_{x\to1^+}\frac{\frac2{x-1}}{\frac5{x-1}}$。

\textbf{快速解法：} 上下都趋向 $+\infty$，但可以把公共的 $\frac1{x-1}$ 直接约掉：
\[
\lim_{x\to1^+}\frac{\frac2{x-1}}{\frac5{x-1}}
=\lim_{x\to1^+}\frac25=\boxed{\frac25}.
\]
\textbf{求导核对：}
\[
\lim_{x\to1^+}\frac{-2/(x-1)^2}{-5/(x-1)^2}
=\frac25.
\]
\textbf{提醒：}不能因为 $x\to1$ 而误判为 $0/0$ 型，先看\textbf{分子、分母各自的函数值}。

\medskip
\textbf{\textcolor{CExample}{例10：$x\to-\infty$ 时，根号里的平方不能直接开成 $x$}}

\textbf{题目：} $\displaystyle\lim_{x\to-\infty}\frac{\sqrt{x^2+4}}{x}$。

因为 $x<0$，所以 $\sqrt{x^2}=|x|=-x$。先直接化简：
\[
\frac{\sqrt{x^2+4}}{x}
=-\sqrt{1+\frac4{x^2}}\longrightarrow\boxed{-1}.
\]
若分别求导，对照得
\[
\lim_{x\to-\infty}\frac{x/\sqrt{x^2+4}}{1}=-1.
\]
\textbf{关键：}分子趋于 $+\infty$、分母趋于 $-\infty$，也属于本型；千万不要把正负号丢掉。

\end{examplebox}
