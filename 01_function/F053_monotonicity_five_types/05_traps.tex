\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略定义域）}}]
			讨论含对数和分式的函数单调性时，未先求定义域，直接解 $f'(x)>0$，导致单调区间包含不在定义域内的点。
		\item[\textbf{\textcolor{CTrap}{正确理解（先定定义域）：}}]
			求导前先明确函数定义域，再在定义域内解 $f'(x)>0$ 和 $f'(x)<0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（习惯缺失）：}}]
			定义域是函数性质的基础，忽略定义域是常见系统性错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽视判别式）}}]
			对二次型导函数 $ax^2+bx+c$，不计算 $\Delta$ 就直接写解集，导致虚根情况未考虑。
		\item[\textbf{\textcolor{CTrap}{正确理解（依判别式分类）：}}]
			先计算 $\Delta$，若 $\Delta>0$ 有两实根，$\Delta=0$ 一根，$\Delta<0$ 无实根，再结合 $a$ 符号得解集。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略根存在）：}}]
			二次不等式解集依赖于判别式，不计算 $\Delta$ 必然无法正确分类。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（漏边界参数）}}]
			分类讨论时仅分 $k>0$ 和 $k<0$，忘记 $k=0$；或只分 $a>0$ 和 $a<0$，忽略 $a=0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（参数完全分类）：}}]
			对参数取值进行全域分类，包括边界值，每个子类都需讨论导函数符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维不严密）：}}]
			边界参数常使导函数结构变化，如退化或恒号，忽略会导致分类不完全。
	\end{description}

\end{trapbox}
