\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 讨论函数 $f(x)=\frac{1}{2}kx^2+2x+1$（$k\in\mathbb{R}$）的单调性。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导运算）：}]
			\[
				f'(x)=kx+2
			\]
			\par\textbf{理由：} 利用 $(x^2)'=2x$ 及线性求导法则。
		\item[\textbf{第二步（识别类型）：}]
			\[
				f'(x)=kx+2
			\]
			\par\textbf{理由：} 导函数为一次函数形式，参数 $k$ 为斜率。
		\item[\textbf{第三步（参数分类讨论）：}]
			\[
				kx+2>0
			\]
			\par\textbf{理由：} 当 $k>0$ 时，解为 $x>-\frac{2}{k}$，得增区间 $(-\frac{2}{k},\infty)$，减区间 $(-\infty,-\frac{2}{k})$。当 $k<0$ 时，解为 $x<-\frac{2}{k}$，得增区间 $(-\infty,-\frac{2}{k})$，减区间 $(-\frac{2}{k},\infty)$。当 $k=0$ 时，$f'(x)=2>0$ 恒成立，在 $\mathbb{R}$ 上单调增。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	单调区间由参数 $k$ 的正负和零点共同决定。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 讨论函数 $f(x)=x^3-3ax$（$a\in\mathbb{R}$）的单调性。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导运算）：}]
			\[
				f'(x)=3x^2-3a
			\]
			\par\textbf{理由：} 利用 $(x^3)'=3x^2$。
		\item[\textbf{第二步（识别类型）：}]
			\[
				f'(x)=3x^2-3a
			\]
			\par\textbf{理由：} 导函数为二次式，二次项系数 $3>0$，判别式 $\Delta=36a$，故按 $a$ 分类。
		\item[\textbf{第三步（参数分类讨论）：}]
			\[
				3x^2-3a>0
			\]
			\par\textbf{理由：} 当 $a<0$ 时 $\Delta<0$，$f'(x)>0$ 恒成立，$f(x)$ 在 $\mathbb{R}$ 上单调增。当 $a=0$ 时 $\Delta=0$，$f'(x)=3x^2\geq0$，$f(x)$ 在 $\mathbb{R}$ 上单调增。当 $a>0$ 时 $\Delta>0$，两根 $x=\pm\sqrt{a}$，$f'(x)>0$ 解为 $x<-\sqrt{a}$ 或 $x>\sqrt{a}$，增区间 $(-\infty,-\sqrt{a})\cup(\sqrt{a},\infty)$，减区间 $(-\sqrt{a},\sqrt{a})$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a\leq0$ 时 $f(x)$ 在 $\mathbb{R}$ 上单调增；$a>0$ 时单调增区间为 $(-\infty,-\sqrt{a})\cup(\sqrt{a},\infty)$，单调减区间为 $(-\sqrt{a},\sqrt{a})$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 讨论函数 $f(x)=\frac{1}{2}ax^2+\ln x$（$a\in\mathbb{R}$）的单调性。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导运算）：}]
			\[
				f'(x)=ax+\frac{1}{x}
			\]
			\par\textbf{理由：} $(\frac{1}{2}ax^2)'=ax$，$(\ln x)'=\frac{1}{x}$，定义域 $x>0$。
		\item[\textbf{第二步（识别类型）：}]
			\[
				f'(x)=\frac{ax^2+1}{x}
			\]
			\par\textbf{理由：} 通分后分子为二次式，分母 $x>0$，符号由分子决定，属于分式型。
		\item[\textbf{第三步（参数分类讨论）：}]
			\[
				\frac{ax^2+1}{x}>0\;(x>0)
			\]
			\par\textbf{理由：} 分母 $x>0$，故 $f'(x)>0$ 等价于 $ax^2+1>0$。若 $a\geq0$，则 $ax^2+1>0$ 恒成立，$f'(x)>0$，$f(x)$ 在 $(0,\infty)$ 上单调增。若 $a<0$，则 $ax^2+1>0$ 解为 $0<x<\sqrt{-\frac{1}{a}}$，故增区间 $(0,\sqrt{-\frac{1}{a}})$，减区间 $(\sqrt{-\frac{1}{a}},\infty)$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a\geq0$ 时 $f(x)$ 在 $(0,\infty)$ 上单调增；$a<0$ 时单调增区间为 $(0,\sqrt{-\frac{1}{a}})$，单调减区间为 $(\sqrt{-\frac{1}{a}},\infty)$。
\end{examplebox}
