\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（导函数可化标准）：}}]
			函数 $f(x)$ 可导，且导函数 $f'(x)$ 只含参数，并能化为以下五种标准结构之一。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（一次型讨论）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			按斜率正负和零点划分单调区间。
			\[
				f'(x)=kx+b
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（二次可分解）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			比较两根大小与二次项符号确定符号区间。
			\[
				f'(x)=a(x-m)(x-n)
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（二次不可分解）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			由判别式和二次项符号分三类讨论。
			\[
				f'(x)=ax^{2}+bx+c,\quad \Delta=b^{2}-4ac
			\]
		\item[\textbf{\textcolor{CStatement}{结论四（指数对数型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $f'(x)>0$ 化为 $x$ 与参数的比较。
			\[
				f'(x)=e^{x}-a,
				\qquad
				f'(x)=\ln x+a
			\]
		\item[\textbf{\textcolor{CStatement}{结论五（分式根式型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用穿根法结合定义域确定符号区间。
			\[
				f'(x)=\frac{x-a}{x}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（穿根法推广）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若导函数可分解为多个一次因式乘积，仍可用穿根法讨论。
			\[
				f'(x)=a\prod_{i=1}^{n}(x-r_i)
			\]
	\end{description}

\end{statementbox}
