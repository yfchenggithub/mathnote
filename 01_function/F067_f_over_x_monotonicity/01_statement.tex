\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定义域限制）：}}]
			函数$f(x)$在区间$I$上可导，且$0\notin I$（即$I$不包含$0$）。
	\end{description}
	\par
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（核心构造）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			构造函数后，分子正负决定单调性。
			\[
				F'(x)=\frac{x f'(x)-f(x)}{x^{2}}.
			\]
			因此$F'(x)$与$x f'(x)-f(x)$同号，从而可由后者的符号判断$F(x)$的单调性。
		\item[\textbf{\textcolor{CStatement}{常见变形（幂次推广）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若分母为$x^{n}$，分子相应变为$x f'(x)-n f(x)$。
			\[
				G(x)=\frac{f(x)}{x^{n}}\;\Rightarrow\; G'(x)=\frac{x f'(x)-n f(x)}{x^{n+1}}.
			\]
			若在区间 $I$ 上恒有 $x>0$，则 $G'(x)$ 与 $x f'(x)-n f(x)$ 同号（否则需考虑分母 $x^{n+1}$ 的符号）。
	\end{description}

\end{statementbox}
