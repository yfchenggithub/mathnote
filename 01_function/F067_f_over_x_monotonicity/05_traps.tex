\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（定义域忽略）}}]
			有些同学在使用结论时，直接对$F(x)$求导并判号，忘了原函数定义域要求$x\neq0$，导致错误地将单调区间扩展到包含$0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母不为零）：}}]
			使用该结论前必须确认区间$I$不包含$0$，否则$F(x)$在$x=0$处无定义，结论不适用。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视条件）：}}]
			只记住了“同号”关系，忽略了大前提$0\notin I$，导致对定义域把握不准。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（符号误判）}}]
			错误认为$x f'(x)-f(x)>0$能直接推出$F(x)$在包含$x=0$的区间上也单调递增，忽略了$x=0$处无定义。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母恒正）：}}]
			在$x\neq0$的区域，因为$x^{2}>0$，$F'(x)$与分子同号，所以单调性仅在$0$两侧分别讨论。
		\item[\textbf{\textcolor{CTrap}{错因分析（区间分割）：}}]
			不理解$x\neq0$导致定义域被分成两个区间，单调性需分区间说明，不能合并。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式混淆）}}]
			有些同学将结论记成$F'(x)$与$x f'(x)+f(x)$同号，或者用到$F(x)=x f(x)$时也用$x f'(x)-f(x)$判号，导致错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（形式对应）：}}]
			只有$F(x)=\frac{f(x)}{x}$对应的分子才是$x f'(x)-f(x)$；若$F(x)=x f(x)$，则$F'(x)=f(x)+x f'(x)$，与$f(x)+x f'(x)$同号。
		\item[\textbf{\textcolor{CTrap}{错因分析（死记硬背）：}}]
			没有理解公式的推导过程，仅凭记忆，容易混淆符号和形式。
	\end{description}

\end{trapbox}
