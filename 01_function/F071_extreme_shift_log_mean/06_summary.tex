\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{极值点偏移问题中，若条件可化为 $\frac{x_1-x_2}{\ln x_1-\ln x_2}=c$，则直接得 $x_1+x_2>2c$，$x_1x_2<c^2$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（正数且不等）：} $x_1,x_2>0$，且 $x_1\neq x_2$。
		\item \textbf{条件 2（标准结构）：} 能从 $f(x_1)=f(x_2)$ 准确变形得到对数平均形式，且 $c$ 为常数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（定义域与分母）：} 使用前必须检查 $x_1,x_2>0$ 且 $x_1\neq x_2$，否则结论无意义。
		\item \textbf{检查项（不等式方向）：} 和大于 $2c$，积小于 $c^2$，不可记反；可通过 $\sqrt{x_1x_2} < c$ 且 $c < \frac{x_1+x_2}{2}$ 验证方向。
	\end{itemize}

\end{summarybox}
