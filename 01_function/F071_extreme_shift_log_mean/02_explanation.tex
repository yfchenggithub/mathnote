\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}

	\textbf{该结论利用对数均值不等式：若两个正数的对数均值等于 $c$，那么 $c$ 位于几何平均与算术平均之间，从而直接推出和与积的不等式。}

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（条件识别）：}]
		      题目中出现 $x_1,x_2>0$、$x_1\neq x_2$，且能通过变形得到
		      \[
			      \frac{x_1-x_2}{\ln x_1-\ln x_2}=c
		      \]
		      的结构。该结构常来源于含有 $\ln x$ 与一次项 $kx$ 的函数等值条件。

		\item[\textbf{要点二（结论套用）：}]
		      一旦确定 $c$，就可直接写出 $x_1+x_2>2c$ 和 $x_1x_2<c^2$，不必再对原始函数反复构造差函数。

		\item[\textbf{要点三（方向来源）：}]
		      因为
		      \[
			      \sqrt{x_1x_2}<c<\frac{x_1+x_2}{2},
		      \]
		      所以由右边得到 $x_1+x_2>2c$，由左边得到 $x_1x_2<c^2$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（割线与平均）}}

	设 $y=\ln x$ 图像上两点为 $(x_1,\ln x_1)$、$(x_2,\ln x_2)$。割线斜率为
	\[
		\frac{\ln x_1-\ln x_2}{x_1-x_2},
	\]
	它的倒数
	\[
		\frac{x_1-x_2}{\ln x_1-\ln x_2}
	\]
	就是 $x_1,x_2$ 的对数均值。对数均值比几何平均大，比算术平均小。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（均值排序）}}

	对数平均是一种介于几何平均和算术平均之间的平均数。本结论利用这一排序关系，由对数平均等于 $c$ 反推出两数之和、两数之积的范围。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（极值点偏移）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接推导）：} 给出含参数函数 $f(x)$，由 $f(x_1)=f(x_2)$ 导出 $c$，再利用本结论证明 $x_1+x_2$ 与 $x_1x_2$ 的不等式。
		\item \textbf{考法二（不等式放缩）：} 在导数压轴题中，将对数均值不等式作为工具箱，快速得到变量范围，省去传统构造函数的繁琐计算。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}

	\textbf{本结论把两个变量的问题直接压缩成一个常数 $c$ 的问题，核心不是硬算 $x_1,x_2$，而是利用“对数均值夹在几何平均与算术平均之间”。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（极值点偏移）：} 若 $f(x_1)=f(x_2)$ 且 $x_1\neq x_2$，常可通过移项得到对数均值结构，从而直接判断 $x_1+x_2$ 与 $x_1x_2$ 的范围。
		\item \textbf{场景二（对数均值应用）：} 题目直接或间接给出形如 $\dfrac{a-b}{\ln a-\ln b}=k$ 的条件，问 $a+b$ 或 $ab$ 的范围，可直接套用。
	\end{itemize}
\end{explanationbox}
