\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正数且不等）：}}]
			$x_1,x_2 > 0$ 且 $x_1 \neq x_2$. 同时 $f(x_1)=f(x_2)$ 可变形为 $\frac{x_1-x_2}{\ln x_1 - \ln x_2} = c$（$c$为常数）.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（和不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两数之和大于参数 $c$ 的两倍.
			\[
				x_1 + x_2 > 2c.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（积不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两数之积小于参数 $c$ 的平方.
			\[
				x_1 x_2 < c^2.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{常见变形（对数均值不等式）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{常见变形（对数均值不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对数均值介于几何平均与算术平均之间.
			\[
				\begin{aligned}
					\sqrt{x_1 x_2} &< \frac{x_1-x_2}{\ln x_1 - \ln x_2} \\ &< \frac{x_1+x_2}{2}.
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
