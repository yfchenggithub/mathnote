\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正数且不等）：}}]
		      $x_1,x_2>0$ 且 $x_1\neq x_2$。

		\item[\textbf{\textcolor{CStatement}{条件 2（对数均值形式）：}}]
		      若由题设可变形得到
		      \[
			      \frac{x_1-x_2}{\ln x_1-\ln x_2}=c,
		      \]
		      其中 $c$ 为常数。由于左边是正数，所以必有 $c>0$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（和不等式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      两数之和大于参数 $c$ 的两倍。
		      \[
			      \boxed{x_1+x_2>2c.}
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（积不等式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      两数之积小于参数 $c$ 的平方。
		      \[
			      \boxed{x_1x_2<c^2.}
		      \]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{常见变形（对数均值不等式）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{对数均值夹逼：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      对数均值介于几何平均与算术平均之间。
		      \[
			      \boxed{
				      \sqrt{x_1x_2}
				      <
				      \frac{x_1-x_2}{\ln x_1-\ln x_2}
				      <
				      \frac{x_1+x_2}{2}
			      }
		      \]
	\end{description}
\end{statementbox}
