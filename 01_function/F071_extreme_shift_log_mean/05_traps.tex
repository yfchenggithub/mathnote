\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视定义域）}}]
			忽略 $x_1,x_2>0$ 的限制，将结论用于非正数。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义域要求）：}}]
			结论基于 $\ln x_1,\ln x_2$ 必须有意义，因此只能对正数使用。使用前必须检查 $x_1>0$ 且 $x_2>0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略对数定义）：}}]
			题目条件可能没有显式给出 $x_1,x_2>0$，但 $\ln$ 隐含该条件；学生若未注意，可能错误地应用到负数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分母非零条件）}}]
			忽略 $x_1 \neq x_2$，当 $x_1=x_2$ 时 $\ln x_1 - \ln x_2 = 0$ 导致分母为零，无定义。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母非零要求）：}}]
			结论的前提是 $x_1 \neq x_2$，否则 $c$ 无定义。在使用前务必确认 $x_1 \neq x_2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定式）：}}]
			在极值点偏移问题中 $x_1 \neq x_2$ 常隐式成立，但学生仍应主动检查，避免在等号情况下误用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（不等式方向记反）}}]
			混淆方向，写成 $x_1+x_2<2c$ 或 $x_1x_2>c^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（方向记忆）：}}]
			由对数均值不等式 $\sqrt{x_1x_2} < c$ 且 $c < \dfrac{x_1+x_2}{2}$ 可推出 $x_1+x_2>2c$ 且 $x_1x_2<c^2$。记忆：$c$ 比几何平均大，比算术平均小，故和比 $2c$ 大，积比 $c^2$ 小。
		\item[\textbf{\textcolor{CTrap}{错因分析（机械记忆）：}}]
			学生仅凭记忆，不理解不等式本质，颠倒方向。通过推导不等式理解就不会错。
	\end{description}

\end{trapbox}
