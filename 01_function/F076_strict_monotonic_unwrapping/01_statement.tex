\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（严格单调）：}}] 外层函数 $f(t)$ 在区间 $D$ 上严格单调（严格增或严格减）.
		\item[\textbf{\textcolor{CStatement}{条件 2（值域范围）：}}] 内层表达式 $g(x)$ 与 $h(x)$ 的值均落在 $D$ 内，即 $g(x), h(x) \in D$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（同构脱壳）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			方程两边同时去掉外层函数 $f$，转化为内层方程.
			\[
				f\bigl(g(x)\bigr) = f\bigl(h(x)\bigr) \iff g(x) = h(x).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（单射性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			严格单调函数在定义域内一一对应，相等的函数值对应相等的自变量.
			\[
				f(u) = f(v) \iff u = v
				\qquad
				(\forall u,v\in D).
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{适用范围：}}
	指数、对数、幂函数等复合方程的解方程问题，尤其当方程两侧结构相同（同构）时可直接脱壳.
\end{statementbox}
