\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{严格单调函数可“脱壳”：方程 $f(g(x))=f(h(x))$ 与 $g(x)=h(x)$ 等价，前提是 $f$ 严格单调且 $g,h$ 取值在 $f$ 的单调区间内.}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（严格单调）：} 外层函数 $f(t)$ 在区间 $D$ 上严格单调（严格增或严格减）.
		\item \textbf{条件 2（值域范围）：} 内层表达式 $g(x)$ 和 $h(x)$ 的值均落在 $D$ 内.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（单调性检查）：} 平方、绝对值等非严格单调函数不可直接脱壳；需先判断外层是否严格单调.
		\item \textbf{检查项（定义域验证）：} 脱壳后应检查内层表达式是否在外层函数定义域内，对数方程必须检验真数大于0.
	\end{itemize}
\end{summarybox}
