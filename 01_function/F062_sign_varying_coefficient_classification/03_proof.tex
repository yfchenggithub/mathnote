\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	利用不等式的基本性质：当两边乘除正数时不等号不变，乘除负数时不等号反转。因此必须按 $k(x)$ 的符号分段讨论，将原不等式等价转化为 $a$ 与 $\dfrac{h(x)}{k(x)}$ 的比较。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（符号划分）：}] 将 $D$ 按 $k(x)$ 的符号细分为三个子集：
			\[
				D_1=\{x\in D\mid k(x)>0\},\quad D_2=\{x\in D\mid k(x)<0\},\quad D_0=\{x\in D\mid k(x)=0\}.
			\]
			\par\textbf{理由：} 因为 $k(x)$ 变号，所以 $D_1,D_2,D_0$ 构成 $D$ 的一个划分，原不等式对 $D$ 恒成立等价于对每个子集恒成立。

		\item[\textbf{步骤二（正区间等价）：}] 对于任意 $x\in D_1$，由 $k(x)>0$，在 $a\cdot k(x)\ge h(x)$ 两边同除以正数 $k(x)$，不等号不变，得
			\[
				a\ge \frac{h(x)}{k(x)}.
			\]
			反之，若 $a\ge \dfrac{h(x)}{k(x)}$ 对一切 $x\in D_1$ 成立，则乘以 $k(x)>0$ 即得原不等式。
			\par\textbf{理由：} 正数除法保号性。

		\item[\textbf{步骤三（负区间等价）：}] 对于任意 $x\in D_2$，由 $k(x)<0$，同除以负数 $k(x)$，不等号反转，得
			\[
				a\le \frac{h(x)}{k(x)}.
			\]
			反之，若 $a\le \dfrac{h(x)}{k(x)}$ 对一切 $x\in D_2$ 成立，则乘以 $k(x)<0$ 时不等号再次反转，回到 $a\cdot k(x)\ge h(x)$。
			\par\textbf{理由：} 负数除法反转不等号。

		\item[\textbf{步骤四（零区间等价）：}] 对于任意 $x\in D_0$，$k(x)=0$，原不等式化为
			\[
				0\ge h(x).
			\]
			该条件不含 $a$，独立于 $a$ 的取值。
			\par\textbf{理由：} 代入 $k(x)=0$ 直接得到。

		\item[\textbf{步骤五（取交集）：}] 若 $a$ 同时满足步骤二、三、四中的条件（即 $a\ge \frac{h(x)}{k(x)}$ 在 $D_1$ 上恒成立，$a\le \frac{h(x)}{k(x)}$ 在 $D_2$ 上恒成立，且 $0\ge h(x)$ 在 $D_0$ 上恒成立），则原不等式对任意 $x\in D$ 成立。反之亦然。故 $a$ 的取值范围为三部分所得范围的交集。
			\par\textbf{理由：} 充要性来自每个步骤的等价变形。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	通过以上推导，我们将含参不等式 $a\cdot k(x)\ge h(x)$ 在变号系数情形下等价转化为三个子条件，然后分别求范围再取交集。这种分类讨论的思路是处理变号因子恒成立问题的标准方法。
\end{proofbox}
