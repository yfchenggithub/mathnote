\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{含参不等式恒成立，若参数系数变号，则按正、负、零分段求解，再取交集。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（变号系数）：} 参数系数 $k(x)$ 在区间 $D$ 上既取正值也取负值。
		\item \textbf{条件 2（不可直接分离）：} 参数不能直接通过不等式两边同除以 $k(x)$ 分离（因为 $k(x)$ 变号会导致方向不确定）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（零值检验）：} 必须单独验证 $k(x)=0$ 的点是否满足原不等式。
		\item \textbf{检查项（方向反转）：} 在 $k(x)<0$ 的区间，除以 $k(x)$ 后不等号必须反转。
	\end{itemize}
\end{summarybox}
