\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{将原函数 $y=f(x)$ 图像中位于 $x$ 轴下方的部分关于 $x$ 轴对称翻折到上方，即得 $y=|f(x)|$ 的图像。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（原函数已知）：} 需要已知 $y=f(x)$ 的图像或解析式，可画出其草图。
		\item \textbf{条件2（定义域不变）：} 变换前后定义域相同，保证逐点操作可行。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分情况处理）：} 必须根据 $f(x)$ 的符号分开处理，确保 $f(x)\ge0$ 部分不变，$f(x)<0$ 部分翻折。
		\item \textbf{检查项（翻折完整）：} 画图后检查 $x$ 轴下方是否还有残留部分，确保所有负值都已翻折。
	\end{itemize}
\end{summarybox}
