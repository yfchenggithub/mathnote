	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定义域前提）：}}] 设原函数 $y=f(x)$ 的定义域为 $D$，则变换后 $y=|f(x)|$ 的定义域仍为 $D$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（翻折规则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			保留 $x$ 轴上方的部分，将 $x$ 轴下方的部分关于 $x$ 轴对称翻折到上方。
			\[
				|f(x)| = \begin{cases}
				f(x), & f(x)\ge0,\\n    -f(x), & f(x)<0.
		\end{cases}
		\]
	\item[\textbf{\textcolor{CStatement}{推广结论（特殊情形）：}}]
		\textbf{\textcolor{CStatement}{直观描述：}}
		当 $f(x)\ge0$ 恒成立时，$|f(x)|=f(x)$，图像与原图重合；当 $f(x)\le0$ 恒成立时，$|f(x)|=-f(x)$，图像与原图关于 $x$ 轴对称。
\end{description}
\end{statementbox}
