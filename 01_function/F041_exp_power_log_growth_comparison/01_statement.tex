\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（底数与指数）：}}]
			实数$a>1$，实数$\alpha>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（自变量趋势）：}}]
			$x\to +\infty$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（指数与幂函数）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			指数函数$a^x$的增长速度远快于任意幂函数$x^\alpha$。
			\[
				\lim_{x\to+\infty}\frac{a^x}{x^\alpha}=+\infty.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（幂函数与对数函数）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			幂函数$x^\alpha$的增长速度远快于对数函数$\log_a x$。
			\[
				\lim_{x\to+\infty}\frac{x^\alpha}{\log_a x}=+\infty.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（对数正幂次）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对数函数的任意正数次幂仍然慢于幂函数。
			\[
				\lim_{x\to+\infty}\frac{x^\alpha}{(\log_a x)^\beta}=+\infty,
				\qquad
				\forall\beta>0.
			\]
	\end{description}

\end{statementbox}
