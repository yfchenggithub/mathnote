\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{当$a>1,\alpha>0$且$x\to+\infty$时，指数函数$a^x$的增长速度远大于幂函数$x^\alpha$，而幂函数$x^\alpha$的速度又远大于对数函数$\log_a x$，即$a^x \gg x^\alpha \gg \log_a x$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（底数大于1）：} $a>1$。
		\item \textbf{条件2（幂指数为正）：} $\alpha>0$。
		\item \textbf{条件3（自变量趋势）：} $x\to+\infty$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽视条件）：} 只在 $a>1,\alpha>0$ 时成立，不可用于 $a\leq1$ 或 $\alpha\leq0$。
		\item \textbf{检查项（参数确认）：} 使用前必须检查底数 $a>1$ 且指数 $\alpha>0$，否则不适用。
	\end{itemize}

\end{summarybox}
