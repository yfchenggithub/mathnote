\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话记住}}
	当$a>1$、$\alpha>0$且$x\to+\infty$时：
	\[
		\boxed{\underbrace{a^x}_{\text{指数}}\ \gg\
		\underbrace{x^\alpha}_{\text{幂}}\ \gg\
		\underbrace{\log_ax}_{\text{对数}}}.
	\]
	\textbf{指数最终超过幂，幂最终超过对数；超过的倍数越来越大。}

	\textbf{\textcolor{CSummary}{考试三步走}}
	\begin{enumerate}[leftmargin=*,itemsep=0.25em]
		\item \textbf{查条件：}$a>1$、$\alpha>0$、$x\to+\infty$。
		\item \textbf{认函数：}看清谁是$a^x$、谁是$x^\alpha$、谁是$\log_ax$。
		\item \textbf{看位置：}增长快的在分子，比值趋向$+\infty$；增长快的在分母，比值趋向$0$。
	\end{enumerate}
	\textbf{最后提醒：}$x=3$时$3^2>2^3$，所以这是一条\textbf{最终增长规律}，不是每个$x$下的大小排序。
\end{summarybox}
