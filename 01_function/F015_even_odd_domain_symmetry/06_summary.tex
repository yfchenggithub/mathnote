\begin{summarybox}
\textbf{一句话记住}

\textbf{先查输入能否左右配对，再查输出是相等还是相反。}

\medskip
\textbf{三步答题法}
\begin{enumerate}[leftmargin=*,itemsep=0.25em]
\item \textbf{写定义域：}任意允许的$x$，$-x$也得允许；若不满足，\textbf{直接判非奇非偶}。
\item \textbf{算$f(-x)$：}先把所有$x$换成$-x$，再化简。
\item \textbf{比函数值：}$f(-x)=f(x)$为\textbf{偶函数}；$f(-x)=-f(x)$为\textbf{奇函数}；都不满足就\textbf{非奇非偶}。等式要对所有允许的$x$成立。
\end{enumerate}

\medskip
\textbf{图像怎么记？}
\[
\begin{array}{c|c|c}
&\text{对应的点}&\text{图像对称}\\
\hline
\text{偶函数}&(x,y)\leftrightarrow(-x,y)&y\text{轴}\\
\text{奇函数}&(x,y)\leftrightarrow(-x,-y)&\text{原点}
\end{array}
\]

\medskip
\textbf{四个必须分清的细节}
\begin{itemize}[leftmargin=*,itemsep=0.25em]
\item 定义域对称\textbf{并不保证}函数一定奇或偶。
\item 奇函数\textbf{若在$0$处有定义}，才有$f(0)=0$；反过来不成立。
\item 既奇又偶的函数在\textbf{对称定义域}上只能处处为$0$。
\item 一组点符合要求\textbf{不足以证明}；找到一组不符合要求\textbf{足以否定}。
\end{itemize}

\medskip
\textbf{进阶一眼看懂}
\[
\boxed{f(x)=\frac{f(x)+f(-x)}2+\frac{f(x)-f(-x)}2}
\]
在定义域对称时，前一部分是偶函数，后一部分是奇函数。

\medskip
\textbf{这条结论的价值：}少画一半图，少算一半函数值，还能快速判断式子、解含参数题。
\end{summarybox}
