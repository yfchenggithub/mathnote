\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（可导条件）：}}]
			函数 $y = f(x)$ 在 $x_0$ 处可导，即极限 $\lim_{\Delta x \to 0} \frac{f(x_0 + \Delta x) - f(x_0)}{\Delta x}$ 存在且有限。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（切线点斜式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			以导数值为斜率，过切点写直线方程。
			\[
				\boxed{y - f(x_0) = f'(x_0)(x - x_0)}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（斜截形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将点斜式化为斜截式，便于读取截距。
			\[
				\boxed{y = f'(x_0)\,x + \bigl(f(x_0) - f'(x_0)x_0\bigr)}.
			\]
	\end{description}

\end{statementbox}
