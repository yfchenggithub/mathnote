\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{可导函数在切点处的切线方程可由导数值和切点坐标直接写为点斜式 $y - f(x_0) = f'(x_0)(x - x_0)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（可导条件）：} 函数 $y = f(x)$ 在 $x_0$ 处可导，即 $\displaystyle \lim_{\Delta x\to 0}\frac{f(x_0+\Delta x) - f(x_0)}{\Delta x}$ 存在。
		\item \textbf{条件2（切点已知）：} 已知切点横坐标 $x_0$，且对应点在曲线上。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（切点验证）：} 使用公式前必须确认题目要求的是“在点”还是“过点”，必要时设切点。
		\item \textbf{检查项（符号核对）：} 写出切线方程后，代入切点坐标检查是否满足。
	\end{itemize}

\end{summarybox}
