\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x) = x^2$，求曲线在点 $(1, f(1))$ 处的切线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求值求导）：}]
			计算 $f(1) = 1^2 = 1$，$f'(x) = 2x$，故 $f'(1) = 2$。
			\par\textbf{理由：} 幂函数求导公式 $f'(x) = nx^{n-1}$。
		\item[\textbf{第二步（套用公式）：}]
			代入切线方程公式：
			\[
				y - f(1) = f'(1)(x - 1) \quad\Longrightarrow\quad y - 1 = 2(x - 1).
			\]
			\par\textbf{理由：} 点斜式切线方程 $y - f(x_0) = f'(x_0)(x - x_0)$。
		\item[\textbf{第三步（化简结果）：}]
			整理得 $y = 2x - 1$。
			\par\textbf{理由：} 移项合并。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线方程为 $y = 2x - 1$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知曲线 $y = x^3 - 2x$，求其在点 $(1, -1)$ 处的切线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证切点）：}]
			代入 $x=1$ 得 $y = 1^3 - 2 \times 1 = -1$，恰为给定点，故该点在曲线上。
			\par\textbf{理由：} 切点必须在曲线上。
		\item[\textbf{第二步（求导取值）：}]
			$f'(x) = 3x^2 - 2$，所以 $f'(1) = 3 \times 1^2 - 2 = 1$。
			\par\textbf{理由：} 导数计算。
		\item[\textbf{第三步（点斜方程）：}]
			由点斜式得
			\[
				y - (-1) = 1 \cdot (x - 1) \quad\Longrightarrow\quad y + 1 = x - 1.
			\]
			\par\textbf{理由：} 切线方程公式。
		\item[\textbf{第四步（化简）：}]
			整理得 $y = x - 2$。
			\par\textbf{理由：} 移项合并。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线方程为 $y = x - 2$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知曲线 $y = x^2 - 2x + 3$，求过点 $(0, 1)$ 的切线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设切点）：}]
			设切点横坐标为 $t$，则切点为 $(t, t^2 - 2t + 3)$。
			\par\textbf{理由：} 切点未知，需设为参数。
		\item[\textbf{第二步（含参切线）：}]
			$f'(x) = 2x - 2$，故切线斜率 $k = 2t - 2$。切线方程为
			\[
				y - (t^2 - 2t + 3) = (2t - 2)(x - t).
			\]
			\par\textbf{理由：} 点斜式方程。
		\item[\textbf{第三步（代入定点）：}]
			切线过点 $(0, 1)$，代入得
			\[
				1 - (t^2 - 2t + 3) = (2t - 2)(0 - t).
			\]
			\par\textbf{理由：} 点在线上，坐标满足方程。
		\item[\textbf{第四步（解参数）：}]
			整理并解方程：
			\[
				1 - (t^2 - 2t + 3) = (2t - 2)(-t) \implies -t^2 + 2t - 2 = -2t^2 + 2t.
			\]
			移项得 $t^2 - 2 = 0$，所以 $t = \pm \sqrt{2}$。
			\par\textbf{理由：} 代数化简。
		\item[\textbf{第五步（代回得切线）：}]
			\begin{itemize}
			\item 当 $t = \sqrt{2}$，切点 $( \sqrt{2}, 5 - 2\sqrt{2})$，斜率 $2\sqrt{2} - 2$，切线方程为
				\[
					y - (5 - 2\sqrt{2}) = (2\sqrt{2} - 2)(x - \sqrt{2}).
				\]
			\item 当 $t = -\sqrt{2}$，切点 $( -\sqrt{2}, 5 + 2\sqrt{2})$，斜率 $-2\sqrt{2} - 2$，切线方程为
				\[
					y - (5 + 2\sqrt{2}) = (-2\sqrt{2} - 2)(x + \sqrt{2}).
				\]
		\end{itemize}
			\par\textbf{理由：} 将 $t$ 值代入含参切线。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{曲线过点 $(0,1)$ 的切线有两条，方程如上所示。}
\end{examplebox}
