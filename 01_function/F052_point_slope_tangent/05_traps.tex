\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（切点误判）}}]
			见到“过点 $P$ 的切线”就认为 $P$ 是切点，直接用点 $P$ 的坐标代入斜率公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（验证切点）：}}]
			先判断点是否在曲线上：若在，则切点即该点；若不在，则必须设切点 $(t, f(t))$ 重新计算。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			混淆“点 $P$ 处的切线”（切点确定）与“过点 $P$ 的切线”（切点可能变化）两种表述。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（斜率代错）}}]
			求切线时错误使用 $\frac{f(x_0+\Delta x)-f(x_0)}{\Delta x}$ 或其极限未化简就代替导数。
		\item[\textbf{\textcolor{CTrap}{正确理解（导数即斜率）：}}]
			切线斜率必须是导数值 $f'(x_0)$，只有极限存在时才能使用。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式误解）：}}]
			未理解导数定义中极限的作用，用割线斜率直接作为切线斜率。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号混淆）}}]
			点斜式公式记错，如写成 $y - x_0 = f'(x_0)(x - f(x_0))$ 或漏掉括号。
		\item[\textbf{\textcolor{CTrap}{正确理解（点斜结构）：}}]
			正确形式为 $y - f(x_0) = f'(x_0)(x - x_0)$，其中 $f(x_0)$ 和 $x_0$ 地位对称。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号记忆）：}}]
			记忆公式时不理解每个字母的意义，盲目套用。
	\end{description}

\end{trapbox}
