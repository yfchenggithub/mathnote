\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{极值第一充分条件：导数左正右负得极大，左负右正得极小，两侧同号则非极值。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（连续条件）：} 函数 $f(x)$ 在点 $x_0$ 处连续。
		\item \textbf{条件2（邻域可导）：} $f(x)$ 在 $x_0$ 的某去心邻域内可导。
		\item \textbf{条件3（候选点）：} $x_0$ 是 $f$ 的驻点（$f'(x_0)=0$）或不可导点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽视变号）：} 导数为零不一定是极值，必须验证左右导数符号是否改变。
		\item \textbf{检查项（符号判定）：} 在 $x_0$ 两侧各选一个邻近点，计算导数值并比较符号。
	\end{itemize}

\end{summarybox}
