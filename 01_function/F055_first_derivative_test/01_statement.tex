\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（连续可导）：}}]
			函数 $f(x)$ 在点 $x_0$ 处连续，在 $x_0$ 的某去心邻域内可导。
		\item[\textbf{\textcolor{CStatement}{条件2（候选点）：}}]
			$x_0$ 是 $f$ 的驻点（$f'(x_0)=0$）或不可导点。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（左正右负极大）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			左侧导数为正，右侧导数为负，函数先增后减，故 $f(x_0)$ 为极大值。
			\[
				\exists\delta>0,\; x\in(x_0-\delta,x_0)\Rightarrow f'(x)>0,\; x\in(x_0,x_0+\delta)\Rightarrow f'(x)<0.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（左负右正极小）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			左侧导数为负，右侧导数为正，函数先减后增，故 $f(x_0)$ 为极小值。
			\[
				\exists\delta>0,\; x\in(x_0-\delta,x_0)\Rightarrow f'(x)<0,\; x\in(x_0,x_0+\delta)\Rightarrow f'(x)>0.
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（两侧同号非极值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在 $x_0$ 左右两侧的导数符号相同，函数在 $x_0$ 两侧单调性一致，因此 $f(x_0)$ 不是极值。
			\[
				\exists\delta>0,\ \forall x_1\in(x_0-\delta,x_0),\ \forall x_2\in(x_0,x_0+\delta),\ f'(x_1)f'(x_2)>0.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{常见变形（二阶导数判定）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{变形结论（二阶可导判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $f$ 在 $x_0$ 处二阶可导且 $f'(x_0)=0$ 时，可由 $f''(x_0)$ 的符号判定：若 $f''(x_0)>0$，则 $f(x_0)$ 为极小值；若 $f''(x_0)<0$，则 $f(x_0)$ 为极大值。
			\[
				f'(x_0)=0,\; f''(x_0)>0 \implies \text{极小值};\qquad f'(x_0)=0,\; f''(x_0)<0 \implies \text{极大值}.
			\]
	\end{description}

\end{statementbox}
