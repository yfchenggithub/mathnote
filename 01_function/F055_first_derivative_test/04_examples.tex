\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $f(x)=x^2-4x+3$ 的极值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导找驻点）：}]
			$f'(x)=2x-4$。令 $f'(x)=0$，解得 $x=2$。
			\par\textbf{理由：} 驻点是可能的极值点。
		\item[\textbf{第二步（左右符号判定）：}]
			取 $x=1$（左侧），$f'(1)=2\times1-4=-2<0$；取 $x=3$（右侧），$f'(3)=2\times3-4=2>0$。左负右正。
			\par\textbf{理由：} 导数符号代表单调性方向。
		\item[\textbf{第三步（极值判定）：}]
			根据第一充分条件，左负右正表明 $x=2$ 为极小值点，极小值为 $f(2)=2^2-4\times2+3=-1$。
			\par\textbf{理由：} 导数变号判据（左负右正极小）。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x=2$ 是极小值点，极小值为 $f(2)=-1$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $f(x)=x^3$ 的极值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导找驻点）：}]
			$f'(x)=3x^2$。令 $f'(x)=0$，解得 $x=0$。
			\par\textbf{理由：} 驻点候选。
		\item[\textbf{第二步（左右符号分析）：}]
			取 $x=-1$（左侧），$f'(-1)=3>0$；取 $x=1$（右侧），$f'(1)=3>0$。两侧同为正号。
			\par\textbf{理由：} 导数在两侧同号。
		\item[\textbf{第三步（判定结果）：}]
			根据第一充分条件，左右同号非极值，$x=0$ 不是极值点。
			\par\textbf{理由：} 导数变号判据第三条。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x=0$ 不是极值点。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $f(x)=|x|$ 的极值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定候选点）：}]
			$f(x)=|x|$ 在 $x=0$ 处不可导（尖点），且在该点连续。因此 $x=0$ 是极值候选点。
			\par\textbf{理由：} 不可导点也可能是极值点。
		\item[\textbf{第二步（左右导数符号）：}]
			当 $x<0$ 时，$f(x)=-x$，$f'(x)=-1<0$；当 $x>0$ 时，$f(x)=x$，$f'(x)=1>0$。左负右正。
			\par\textbf{理由：} 直接计算左右导数可得。
		\item[\textbf{第三步（极值判定）：}]
			左负右正表明 $x=0$ 为极小值点，极小值为 $f(0)=0$。
			\par\textbf{理由：} 导数变号判据（左负右正极小）。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x=0$ 是极小值点，极小值为 $0$。}
\end{examplebox}
