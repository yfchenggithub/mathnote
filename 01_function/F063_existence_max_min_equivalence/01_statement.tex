\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（有最值区间）：}}]
			函数 $f(x)$ 在区间 $D$ 上有最大值和最小值，$a$ 为常数。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最大值转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			函数存在值不小于 $a$ 等价于最大值不小于 $a$。
			\[
				\boxed{\exists\, x\in D,\ f(x)\ge a \;\iff\; \max_{x\in D}f(x)\ge a}
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（最小值转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			函数存在值不大于 $a$ 等价于最小值不大于 $a$。
			\[
				\boxed{\exists\, x\in D,\ f(x)\le a \;\iff\; \min_{x\in D}f(x)\le a}
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{推广结论（严格不等形式）}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	严格不等也满足类似等价关系。
	\[
		\boxed{\exists\, x\in D,\ f(x)>a \;\iff\; \max_{x\in D}f(x)>a}
	\]
	\[
		\boxed{\exists\, x\in D,\ f(x)<a \iff\; \min_{x\in D}f(x)<a}
	\]
\end{statementbox}
