\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（最值存在前提）}}]
			使用结论时未确认函数在区间 $D$ 上是否有最大值和最小值。
		\item[\textbf{\textcolor{CTrap}{正确理解（检查最值条件）：}}]
			必须首先验证函数在 $D$ 上最值存在，例如闭区间连续或单调有界函数；若函数无最值（如 $f(x)=1/x$ 在 $(0,1)$ 上），则结论不适用。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}]
			学生习惯性套用结论，没有检查最值存在性的意识。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（量词混淆）}}]
			将存在性与恒成立混淆，误认为“存在 $x$ 使 $f(x)\ge a$”等价于“$\min_{x\in D}f(x)\ge a$”。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分存在恒立）：}}]
			存在性使用最大值转化：$\exists x,\ f(x)\ge a \iff \max f\ge a$；恒成立使用最小值转化：$\forall x,\ f(x)\ge a \iff \min f\ge a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（逻辑颠倒）：}}]
			对量词与最值的关系理解不清晰，未注意到存在性的本质是找一个点，只需最大值满足即可。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（等号遗漏）}}]
			认为存在 $x$ 使 $f(x)\ge a$ 必须满足 $\max f > a$，当 $\max f = a$ 时认为不存在。
		\item[\textbf{\textcolor{CTrap}{正确理解（等号可取）：}}]
			条件 $\max f \ge a$ 包含等号，当 $\max f = a$ 时，存在最大值点 $x_0$ 使得 $f(x_0)=a$，所以存在性成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号理解）：}}]
			对 $\ge$ 的含义理解不完整，忽略了等号部分。
	\end{description}

\end{trapbox}
