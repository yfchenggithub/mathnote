\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	函数 $f(x)=x^2$ 在区间 $[-1,2]$ 上有最大值和最小值。判断是否存在 $x\in[-1,2]$ 使得 $f(x)\ge 5$？若存在，求出 $x$；若不存在，说明理由。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求函数最值）：}]
			计算 $f(x)=x^2$ 在 $[-1,2]$ 上的最值。由函数性质，最大值在 $x=2$ 处取得，$f(2)=4$；最小值在 $x=0$ 处，$f(0)=0$。所以 $\max f=4$，$\min f=0$。
			\par\textbf{理由：} 二次函数在闭区间上的最值取决于端点与对称轴位置，此处对称轴 $x=0$ 在区间内。
		\item[\textbf{第二步（应用结论转化）：}]
			存在 $x$ 使 $f(x)\ge 5$ 等价于 $\max f\ge 5$。而 $4\ge5$ 不成立。
			\par\textbf{理由：} 由结论 $\exists x\in D,\ f(x)\ge a \iff \max_{x\in D}f(x)\ge a$，带入 $a=5$。
		\item[\textbf{第三步（给出答案）：}]
			因此不存在这样的 $x$。
			\par\textbf{理由：} 转化后条件不满足，故不存在。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{存在性条件等价于最值比较，当最值不满足时，不存在满足条件的点。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=x^3-3x$ 在 $[-2,2]$ 上有最大值和最小值。若存在 $x\in[-2,2]$ 使得 $f(x)\ge a$ 成立，求实数 $a$ 的取值范围。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求最大值）：}]
			对 $f(x)=x^3-3x$ 求导 $f'(x)=3x^2-3=3(x-1)(x+1)$，列表确定单调性：$[-2,-1]$ 递增，$[-1,1]$ 递减，$[1,2]$ 递增。计算 $f(-2)=-2,\ f(-1)=2,\ f(1)=-2,\ f(2)=2$。所以最大值 $\max f = 2$。
			\par\textbf{理由：} 导数法求三次函数在闭区间上的最值，需比较极值和端点值。
		\item[\textbf{第二步（转化存在性条件）：}]
			存在 $x$ 使 $f(x)\ge a$ 等价于 $\max f\ge a$，即 $2\ge a$。
			\par\textbf{理由：} 直接套用结论 $\exists x\in D,\ f(x)\ge a \iff \max_{x\in D}f(x)\ge a$。
		\item[\textbf{第三步（写出范围）：}]
			所以 $a$ 的取值范围是 $(-\infty,2]$。
			\par\textbf{理由：} 由不等式 $2\ge a$ 解出 $a\le 2$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{存在性求参问题转化为最值与参数的大小比较。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=2^x$ 在 $[-1,2]$ 上有最大值和最小值。若存在 $x_1,x_2\in[-1,2]$ 使得 $f(x_1)\ge a$ 且 $f(x_2)\le a$ 同时成立，求实数 $a$ 的取值范围。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求最值）：}]
			$f(x)=2^x$ 在 $[-1,2]$ 上单调递增，所以 $\min f = f(-1)=2^{-1}=1/2$，$\max f = f(2)=2^{2}=4$。
			\par\textbf{理由：} 指数函数 $2^x$ 为增函数，最值在端点取得。
		\item[\textbf{第二步（转化两条件）：}]
			存在 $x_1$ 使 $f(x_1)\ge a$ 等价于 $\max f\ge a$，即 $4\ge a$。存在 $x_2$ 使 $f(x_2)\le a$ 等价于 $\min f\le a$，即 $1/2\le a$。
			\par\textbf{理由：} 分别应用结论的两个方向。
		\item[\textbf{第三步（求交集）：}]
			两个条件同时成立需 $a\le 4$ 且 $a\ge 1/2$，所以 $a\in[1/2,\ 4]$。
			\par\textbf{理由：} 取两个不等式的交集。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{同时涉及 $\ge$ 和 $\le$ 的两个存在性条件时，分别转化后取交集。}
\end{examplebox}
