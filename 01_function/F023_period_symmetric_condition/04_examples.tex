\begin{examplebox}
\textbf{\textcolor{CExample}{例1｜最基础：已知加减 $6$，求周期}}
\input{assets/tikz/F023_example_twelve}


\textbf{题目：} 对任意实数 $x$ 有 $f(x+6)=f(x-6)$。求一个正周期。

\textbf{解：} 两个输入分别是 $x+6,x-6$，相隔 $6-(-6)=12$。所以
\[
\boxed{12\text{ 是一个正周期。}}
\]
\textbf{特别注意：} 题目如果问“最小正周期”，仅凭这句话还不能确定答案。

\medskip

\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例2｜连续跳格：$f(9)$ 怎么求？}}
\input{assets/tikz/F023_example_three_values}


\textbf{题目：} $f(x+2)=f(x-2)$ 对任意实数成立，已知 $f(1)=3$，求 $f(9)$。

\textbf{解：} 相隔 $4$ 的位置取值相同；从 $1$ 开始，$1\to5\to9$，每次加 $4$：
\[
f(9)=f(5)=f(1)=\boxed{3}.
\]
\textbf{一句话：} 先找重复的间隔，再把较大的自变量一步步减回已知位置。

\medskip

\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例3｜高考常见：超大的自变量，缩回已知区间}}
\input{assets/tikz/F023_periodic_square}


\textbf{题目：} $f$ 定义在 $\mathbb R$ 上，满足 $f(x+3)=f(x-3)$；当 $-3\le x<3$ 时 $f(x)=x^2$。求 $f(2024)$。

\textbf{第1步：} 一个正周期是 $6$。

\textbf{第2步：} $2024=337\times6+2$，因此 $f(2024)=f(2)$。

\textbf{第3步：} $2$ 在 $[-3,3)$ 内，可以代入 $x^2$：
\[
f(2024)=f(2)=2^2=\boxed{4}.
\]
\textbf{为什么可以放心减去 $337$ 个 $6$？} 因为每减一个周期，函数值都不变，减多少次都不变。

\medskip

\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例4｜变形：不是 $+a$ 与 $-a$ 也会有周期}}
\input{assets/tikz/F023_general_gap}


\textbf{题目：} $f(x+1)=f(x-3)$ 对任意 $x$ 成立，已知 $f(0)=7$，求 $f(100)$。

\textbf{解：} 两个位置相隔 $1-(-3)=4$；因此 $4$ 是一个正周期。
\[
100=25\times4,\qquad f(100)=f(0)=\boxed{7}.
\]
\textbf{点睛：} 不必硬凑成 $f(x+a)=f(x-a)$，直接算两个括号里的差更快。

\medskip

\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例5｜反过来写：当 $a$ 是负数}}
\input{assets/tikz/F023_negative_a}


\textbf{题目：} $f(x-3)=f(x+3)$ 对任意 $x$ 成立，求一个正周期。

\textbf{解：} 可以取 $a=-3$，则 $2a=-6$ 是一个带符号的重复位移。周期按正数写：
\[
T=2|-3|=\boxed{6}.
\]
\textbf{不要把正周期写成 $-6$。}

\medskip

\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例6｜辨析：能推出 $4$，却推不出“最小就是 $4$”}}
\input{assets/tikz/F023_not_minimal}


\textbf{题目：} $f(x)=\cos(\pi x)$。它满足 $f(x+2)=f(x-2)$ 吗？最小正周期是多少？

\textbf{解：} 由余弦函数的性质，$\cos(\pi(x+2))=\cos(\pi x)$，$\cos(\pi(x-2))=\cos(\pi x)$，所以条件成立，按本结论可知 $4$ 是一个正周期。

但是 $\cos(\pi(x+2))=\cos(\pi x)$，故更短的 $2$ 也是周期；因为 $\cos(\pi x)$ 的最小正周期为 $2$，所以\textbf{最小正周期是 $2$，不是 $4$}。

\textbf{点睛：} 一个函数可能有 $2,4,6,\ldots$ 等许多正周期；题设只保你找到一个，并不保证是最短的。

\medskip

\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例7｜两条对称轴，直接读出一个周期}}
\input{assets/tikz/F023_two_axes}


\textbf{题目：} 一个定义在 $\mathbb R$ 上的函数，其图像同时关于直线 $x=1$ 和 $x=4$ 对称。求一个正周期。

\textbf{解法一（看图想象）：} 两条竖直对称轴相隔 $4-1=3$，沿两条轴依次翻折，相当于水平方向移动 $2\times3=6$，故 $6$ 是一个正周期。

\textbf{解法二（写得严谨）：} 对称性给出 $f(2-x)=f(x)$ 与 $f(8-x)=f(x)$，从而
\[
f(x+6)=f\big(8-(x+6)\big)=f(2-x)=f(x).
\]
所以 $\boxed{6\text{ 是一个正周期}}$。\textbf{答题时只写后一行即可。}
\end{examplebox}
