\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{若函数$f(x)$满足$f(x+a)=f(x-a)$（$a\neq0$）对定义域内任意$x$成立，则$f(x)$是以$2a$为周期的周期函数。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（等式恒成立）：} 等式$f(x+a)=f(x-a)$必须对定义域内所有$x$恒成立。
		\item \textbf{条件2（非零常数）：} $a$为非零常数，以保证周期$2a\neq0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（周期值为2a）：} 容易错误认为周期是$a$，实际通过代换$x\rightarrow x+a$得到$f(x+2a)=f(x)$，故$T=2a$。
		\item \textbf{检查项（验证恒等性）：} 必须检查等式是否对任意$x$成立，若只对某些$x$成立则不能得出周期结论。
	\end{itemize}
\end{summarybox}
