\begin{summarybox}
\textbf{\textcolor{CSummary}{30 秒记住这道结论}}

\[
\boxed{f(x+a)=f(x-a)\quad\Longrightarrow\quad f(x+2a)=f(x).}
\]
\[
\boxed{T=2|a|\text{ 是一个正周期（不一定最小）。}}
\]
\textbf{一句话：两个自变量相差多少，就可以找到多长的重复间隔。}

\medskip
\textbf{\textcolor{CSummary}{为什么是两倍？}}

因为两边比较的不是 $x$ 和 $x+a$，而是 $x-a$ 和 $x+a$：
\[
(x+a)-(x-a)=2a.
\]

\medskip
\textbf{\textcolor{CSummary}{三种证明怎么选？}}
\begin{itemize}[leftmargin=*,itemsep=.28em]
\item \textbf{最快：} 直接算两个括号内输入的差。
\item \textbf{最稳：} 在原式中把 $x$ 换成 $x+a$，写出 $f(x+2a)=f(x)$。
\item \textbf{最直观：} 从任意位置开始，每走 $2|a|$ 一格，函数值就重复。
\end{itemize}

\medskip
\textbf{\textcolor{CSummary}{遇到这三种题怎么办？}}
\begin{itemize}[leftmargin=*,itemsep=.28em]
\item $f(x+6)=f(x-6)$：一个正周期为 $12$。
\item $f(x+1)=f(x-3)$：一个正周期为 $|1-(-3)|=4$。
\item $f(x+3)=f(x-3)$，求 $f(2024)$：先用周期 $6$ 把 $2024$ 减回已知区间。
\end{itemize}

\medskip
\textbf{\textcolor{CSummary}{最后检查三件事}}
\begin{enumerate}[leftmargin=*,itemsep=.25em]
\item 题目是否对\textbf{所有实数 $x$} 成立，函数是否在整个 $\mathbb R$ 上定义？
\item 题目要\textbf{一个正周期}，还是要\textbf{最小正周期}？
\item $a$ 是否为负数？写正周期时务必用 $2|a|$。
\end{enumerate}

\textbf{相关联的惊喜：} 一条图像若关于 $x=p,x=q$ 两条不同直线都对称，也有一个正周期 $2|p-q|$。
\end{summarybox}
