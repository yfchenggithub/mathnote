\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（可导条件）}}]
			设函数 $f(x)$ 在点 $x_0$ 处可导，且 $x_0$ 为 $f$ 定义域内的内点.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（必要关系）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			极值点处导数必为零.
			\[
				x_0\in\operatorname{Ext}(f)\;\Rightarrow\;f'(x_0)=0.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（不充分性）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			导数为零未必是极值点.
			\[
				f'(x_0)=0\;\not\Rightarrow\;x_0\in\operatorname{Ext}(f).
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（驻点极值关系）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			可导函数极值点一定是驻点，但驻点不一定极值.
			\[
				\operatorname{Ext}(f)\subseteq\{x_0\mid f'(x_0)=0\}.
			\]
	\end{description}

\end{statementbox}
