\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{对于可导函数，$f'(x_0)=0$ 是 $x_0$ 为极值点的必要不充分条件。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（可导前提）：} 函数 $f$ 在 $x_0$ 处可导.
		\item \textbf{条件 2（内点条件）：} $x_0$ 是定义域内点，不是端点.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（充要混淆）：} $f'(x_0)=0$ 不一定推出极值点，需用充分条件验证.
		\item \textbf{检查项（参数验证）：} 由必要条件解出参数后，必须回代检查是否为极值点.
	\end{itemize}

\end{summarybox}
