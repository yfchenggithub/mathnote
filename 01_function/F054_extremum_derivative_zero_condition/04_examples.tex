\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 已知函数 $f(x)=x^3+ax^2+1$ 在 $x=1$ 处有极值，求 $a$ 的值.\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导）：}]
			计算 $f'(x)=3x^2+2ax$.
			\par\textbf{理由：} 幂函数求导法则.
		\item[\textbf{第二步（列方程）：}]
			由极值必要条件，$f'(1)=0$，即 $3+2a=0$.
			\par\textbf{理由：} 若 $x_0$ 是极值点，则导数必为零（必要条件）.
		\item[\textbf{第三步（解参数）：}]
			解方程 $3+2a=0$，得 $a=-\dfrac{3}{2}$.
			\par\textbf{理由：} 移项得 $2a=-3$，故 $a=-\frac{3}{2}$.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a=-\dfrac{3}{2}$。经验证，$f''(1)=6+2a=3>0$，故 $x=1$ 为极小值点，符合题设.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 已知函数 $f(x)=x^3+ax^2+bx+1$ 在 $x=2$ 处有极值，且 $f(2)=1$，求 $a,b$ 的值，并判断 $x=2$ 处是极大值还是极小值.\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导）：}]
			计算 $f'(x)=3x^2+2ax+b$.
			\par\textbf{理由：} 幂函数求导法则.
		\item[\textbf{第二步（列方程求解）：}]
			由极值必要条件 $f'(2)=0$ 得 $12+4a+b=0$；由 $f(2)=1$ 得 $8+4a+2b+1=1$ 即 $4a+2b=-8$. 联立
			\[
				\begin{cases}
					12+4a+b=0,\\4a+2b=-8.
			\end{cases}
			\]
			解得 $a=-4,\ b=4$.
			\par\textbf{理由：} 极值点导数零，函数值已知代入.
		\item[\textbf{第三步（判定类型）：}]
			求二阶导数 $f''(x)=6x+2a$，代入 $x=2,a=-4$ 得 $f''(2)=12-8=4>0$，故 $x=2$ 为极小值点.
			\par\textbf{理由：} 若二阶导大于零，则为极小值（极值的充分条件）.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a=-4,\ b=4$；在 $x=2$ 处取得极小值.

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 已知函数 $f(x)=2x^3+ax^2+bx+1$ 在 $x=1$ 和 $x=-1$ 处都有极值，求 $a,b$ 的值，并求 $f(x)$ 的极值.\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求导）：}]
			计算 $f'(x)=6x^2+2ax+b$.
			\par\textbf{理由：} 幂函数求导法则.
		\item[\textbf{第二步（列方程求解）：}]
			由极值必要条件 $f'(1)=0$ 及 $f'(-1)=0$ 得
			\[
				\begin{cases}
					6+2a+b=0,\\6-2a+b=0.
			\end{cases}
			\]
			解得 $a=0,\ b=-6$.
			\par\textbf{理由：} 极值点处导数必为零.
		\item[\textbf{第三步（求极值）：}]
			由 $f'(x)=6x^2-6=6(x-1)(x+1)$ 知极值点为 $x=-1,1$。求二阶导数 $f''(x)=12x$，代入得 $f''(1)=12>0$ 极小，$f''(-1)=-12<0$ 极大。计算极值：$f(1)=2+0-6+1=-3$，$f(-1)=-2+0+6+1=5$.
			\par\textbf{理由：} 二阶导符号判定极值类型.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a=0,\ b=-6$；极大值 $5$ 在 $x=-1$，极小值 $-3$ 在 $x=1$.
\end{examplebox}
