\begin{summarybox}
  \textbf{\textcolor{CSummary}{一分钟记住：看开口、看交点、查等号}}

  \textbf{先确认：} $f(x)=ax^2+bx+c$ 为实系数\textbf{二次函数}（$a\ne0$），题目要求对\textbf{所有实数 $x$} 成立。
  \[
    \boxed{f(x)>0\ \text{恒成立}\quad\Longleftrightarrow\quad a>0\quad\text{且}\quad\Delta=b^2-4ac<0}
  \]

  \textbf{记成一句话：} 开口向上，而且\textbf{一点也不能碰到} $x$ 轴。

  \medskip
  \textbf{\textcolor{CSummary}{顺手类推（仍须 $a\ne0$、对所有实数）}}
  \[
    \begin{array}{c|c|c}
      \text{要求} & \text{开口} & \text{判别式}\\ \hline
      f(x)>0\ \text{恒成立} & a>0 & \Delta<0\\
      f(x)\ge0\ \text{恒成立} & a>0 & \Delta\le0\\
      f(x)<0\ \text{恒成立} & a<0 & \Delta<0\\
      f(x)\le0\ \text{恒成立} & a<0 & \Delta\le0
    \end{array}
  \]

  \textbf{\textcolor{CSummary}{考试三步走}}
  \begin{enumerate}[leftmargin=*,itemsep=.25em]
    \item \textbf{看题目范围：} 是不是对全部实数？二次项会不会变成 $0$？
    \item \textbf{看开口：} 恒正先写 $a>0$。
    \item \textbf{看交点：} 严格大于 $0$ 写 $\Delta<0$，再解参数范围并取公共部分。
  \end{enumerate}

  \textbf{最后别漏：} $\Delta=0$ 只保证\textbf{非负}；若 $a=0$，应回到一次式或常数式另行判断。
\end{summarybox}
