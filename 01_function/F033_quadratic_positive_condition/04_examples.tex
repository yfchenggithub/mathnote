\begin{examplebox}
  \textbf{\textcolor{CExample}{例 1｜最基础：已知开口向上，求参数}}

  \textbf{题目：} $f(x)=x^2-2kx+1$ 对一切实数 $x$ 都 $>0$，求 $k$ 的范围。

  \textbf{最快解法：} $a=1>0$ 已经满足，只需检查有无横轴交点：
  \[
    \Delta=(-2k)^2-4=4k^2-4<0
    \ \Longleftrightarrow\ k^2<1
    \ \Longleftrightarrow\ \boxed{-1<k<1}.
  \]
  \textbf{为什么边界不能取？} $k=1$ 时 $f(x)=(x-1)^2$，取 $x=1$ 就等于 $0$。

  \input{assets/tikz/F033_quadratic_positive_condition_04_examples_01.tex}
  \end{center}
  \textbf{考试记法：} 已经 $a>0$，就把力气用在 $\Delta<0$ 上。

  \medskip
  \textbf{\textcolor{CExample}{例 2｜两个条件都要检查}}

  \textbf{题目：} $f(x)=kx^2-2x+k$（$k\ne0$）对所有实数恒正，求 $k$。

  \textbf{最快解法：} 先看开口 $k>0$；再看横轴交点：
  \[
    \Delta=(-2)^2-4\cdot k\cdot k=4-4k^2<0
    \ \Longleftrightarrow\ k>1\ \text{或}\ k<-1.
  \]
  与 $k>0$ 取公共部分，得到 $\boxed{k>1}$。
  \textbf{为什么不能写 $|k|>1$ 就结束？} $k=-2$ 时抛物线开口向下，不可能恒正。

  \medskip
  \textbf{\textcolor{CExample}{例 3｜含参数的二次项：注意分界}}

  \textbf{题目：} $f(x)=(m-1)x^2+(m-1)x+1$（$m\ne1$）对所有实数恒正，求 $m$。

  \textbf{最快解法：} 因为题目说明是二次函数，先写 $m-1>0$，即 $m>1$，再求
  \[
    \Delta=(m-1)^2-4(m-1)=(m-1)(m-5)<0.
  \]
  所以 $1<m<5$，与 $m>1$ 合并，得到 $\boxed{1<m<5}$。
  \textbf{点醒一句：} $m=5$ 时恰好碰轴；$m>5$ 时会穿轴，不能恒正。

  \input{assets/tikz/F033_quadratic_positive_condition_04_examples_03.tex}
  \end{center}
  \textbf{再进一小步（容易漏分）：} 若\textbf{删去题目中的 $m\ne1$}，则 $m=1$ 时 $f(x)=1>0$ 也符合要求；新的答案应是 $\boxed{1\le m<5}$。\textbf{别把“系数为零”直接判成不合格。}
\end{examplebox}
