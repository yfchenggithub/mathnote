\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（函数形式）：}}] $f(x)=ax^2+bx+c$，其中 $a\neq0$。
		\item[\textbf{\textcolor{CStatement}{条件二（区间与对称轴）：}}] 区间 $[m,n]$ 固定，$m<n$；对称轴 $x_0=-\dfrac{b}{2a}$ 含参数。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（a>0最小值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			抛物线开口向上，最小值由对称轴位置决定，分三种情况：在区间左侧取左端点，在区间内取顶点，在区间右侧取右端点。
			\[
				f_{\min}=
				\begin{cases}
					f(m), & x_0<m,\\
					f(x_0), & m\le x_0\le n,\\
					f(n), & x_0>n.
			\end{cases}
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（a>0最大值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			最大值由对称轴与区间中点比较决定：对称轴在中点左侧时右端点较大，对称轴在中点右侧时左端点较大。
			\[
				f_{\max}=
				\begin{cases}
					f(n), & x_0\le\dfrac{m+n}{2},\\[6pt]
					f(m), & x_0>\dfrac{m+n}{2}.
			\end{cases}
			\]

		\item[\textbf{\textcolor{CStatement}{结论三(a<0最值对调):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			开口向下时,最小值与最大值表达式恰好互换.
			\[
				f_{\min}=
				\begin{cases}
					f(n), & x_0\le\dfrac{m+n}{2},\\[6pt]
					f(m), & x_0>\dfrac{m+n}{2},
			\end{cases}
				\qquad
				f_{\max}=
				\begin{cases}
					f(m), & x_0<m,\\
					f(x_0), & m\le x_0\le n,\\
					f(n), & x_0>n.
			\end{cases}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{常见变形(已知最值求参)}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{方法:}}] 已知最值求参数时,需按对称轴与区间的三种位置关系分类,令对应表达式等于已知最值,解出参数并检验是否满足分类条件.
	\end{description}
\end{statementbox}
