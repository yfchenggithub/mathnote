\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{定区间动轴求最值：分类讨论对称轴与区间位置，开口向上最小分三段、最大分两段；开口向下最值对调。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（二次函数且区间固定）：} $f(x)=ax^2+bx+c$（$a\neq0$），给定闭区间 $[m,n]$。
		\item \textbf{条件 2（对称轴含参数）：} 对称轴 $x_0=-\dfrac{b}{2a}$ 随参数变化，区间固定。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略分类前提）：} 务必先判定 $a$ 的正负，再选用对应公式；先判断对称轴是否在区间内。
		\item \textbf{检查项（验证取等条件）：} 分类边界点（如 $x_0=m$ 或 $x_0=\dfrac{m+n}{2}$）属于哪一类，需统一约定，避免重复或遗漏。
	\end{itemize}
\end{summarybox}
