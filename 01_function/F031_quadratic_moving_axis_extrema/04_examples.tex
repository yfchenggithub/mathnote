\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=x^2-2x+3$，区间 $[0,3]$，求 $f(x)$ 的最大值与最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定开口与对称轴）：}]
			$a=1>0$，对称轴 $x_0=-\dfrac{-2}{2\times1}=1$。
			\par\textbf{理由：} 二次项系数为正，开口向上，可套用 $a>0$ 的公式。
		\item[\textbf{第二步（求最小值）：}]
			因为 $x_0=1\in[0,3]$，故最小值在顶点取：
			\[
				f_{\min}=f(1)=1^2-2\times1+3=2.
			\]
			\par\textbf{理由：} 对称轴在区间内，属于 $m\le x_0\le n$ 情形。
		\item[\textbf{第三步（求最大值）：}]
			比较 $x_0$ 与区间中点：$\dfrac{m+n}{2}=1.5$，$x_0=1\le1.5$，故最大值在右端点：
			\[
				f_{\max}=f(3)=3^2-2\times3+3=6.
			\]
			\par\textbf{理由：} 对称轴在中点左侧，右端点离对称轴更远。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$f_{\min}=2$，$f_{\max}=6$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $f(x)=x^2-2ax+1$，区间 $[0,2]$，$a$ 为参数，求 $f(x)$ 的最小值 $g(a)$ 的表达式。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定对称轴）：}]
			二次项系数 $1>0$，对称轴 $x_0=a$。
			\par\textbf{理由：} 开口向上，最小值需按对称轴位置分类。
		\item[\textbf{第二步（分类讨论最小值）：}]
			分三种情况：
			\par\textbf{理由：}
			\begin{itemize}
			\item 若 $a<0$，则 $x_0$ 在区间左侧，函数在 $[0,2]$ 递增，最小值 $f(0)=1$。
			\item 若 $0\le a\le2$，则 $x_0$ 在区间内，最小值 $f(a)=a^2-2a^2+1=1-a^2$。
			\item 若 $a>2$，则 $x_0$ 在区间右侧，函数在 $[0,2]$ 递减，最小值 $f(2)=4-4a+1=5-4a$。
		\end{itemize}
		\item[\textbf{第三步（整合表达式）：}]
			得 $g(a)=
			\begin{cases}
				1, & a<0,\\ 1-a^2, & 0\le a\le2,\\ 5-4a, & a>2.
		\end{cases}
			$
			\par\textbf{理由：} 对应最小值的三段公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最小值 $g(a)$ 为关于 $a$ 的分段函数，需根据对称轴 $x_0=a$ 与区间 $[0,2]$ 的关系分三类。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $f(x)=x^2-2ax+1$ 在区间 $[0,2]$ 上的最小值为 $-3$，求 $a$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设定分类情形）：}]
			由例 2，最小值 $g(a)$ 分三段，分别令其等于 $-3$。
			\par\textbf{理由：} 已知最值反求参数，需分类解方程。
		\item[\textbf{第二步（解方程与检验）：}]
			\par\textbf{理由：}
			\begin{itemize}
			\item 情形 1：$a<0$，$g(a)=1=-3$，无解。
			\item 情形 2：$0\le a\le2$，$g(a)=1-a^2=-3$ 得 $a^2=4$，$a=2$ 或 $a=-2$（$a=-2$ 不满足 $0\le a\le2$，舍去；$a=2$ 满足）。检验：$a=2$ 时 $x_0=2$ 属于区间端点，仍符合分类 $0\le a\le2$，成立。
			\item 情形 3：$a>2$，$g(a)=5-4a=-3$ 得 $a=2$，不满足 $a>2$，舍去。
		\end{itemize}
		\item[\textbf{第三步（确定答案）：}]
			只有 $a=2$ 满足条件。
			\par\textbf{理由：} 代入 $f(x)=x^2-4x+1$，$[0,2]$ 上的最小值确实为 $f(2)=4-8+1=-3$，符合。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	已知最值求参数时，必须对对称轴位置分类，令对应表达式等于已知最值，解后检验是否属于该分类区间。
\end{examplebox}
