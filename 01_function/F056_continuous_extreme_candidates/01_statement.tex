\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（函数连续性）：}}]
			$f(x)$ 在闭区间 $[a,b]$ 上连续。
		\item[\textbf{\textcolor{CStatement}{条件2（可导条件）：}}]
			$f(x)$ 在开区间 $(a,b)$ 内可导（允许有限个不可导点）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最值点确定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			最大值与最小值只能在端点、导数零点或不可导点处取得。
			\[
				\begin{aligned}
					M_{\max} &= \max\{f(a), f(b), f(x_1),\dots,f(x_k)\},\\ m_{\min} &= \min\{f(a), f(b), f(x_1),\dots,f(x_k)\}.
			\end{aligned}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（单调性简化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若函数单调，则最值只取决于两端点。
			\[
				\begin{aligned}
					f'(x)\ge0\ (\forall x\in(a,b)) &\Longrightarrow \max f = f(b),\ \min f = f(a);\\ f'(x)\le0\ (\forall x\in(a,b)) &\Longrightarrow \max f = f(a),\ \min f = f(b).
			\end{aligned}
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{使用提醒：}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item 务必验证函数的连续性与可导性，特别是不忽略不可导点。
		\item 比较所有可疑点（端点、驻点、不可导点）的函数值，不能遗漏。
	\end{itemize}

\end{statementbox}
