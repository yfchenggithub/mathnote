\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{任意与存在混合型不等式可转化为最值比较：$\min$ 配 $\ge$，$\max$ 配 $\le$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（函数有最值）：} 函数 $g$ 在 $I_1$ 上、$h$ 在 $I_2$ 上分别存在所需最值（最小值或最大值）。
		\item \textbf{条件2（区间明确）：} 区间 $I_1,I_2$ 给定，函数表达式已知。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（方向匹配）：} $\ge$ 方向用 $\min$，$\le$ 方向用 $\max$，不可混淆。
		\item \textbf{检查项（最值存在）：} 确保函数在区间上有最值，否则需用确界替代。
	\end{itemize}
\end{summarybox}
