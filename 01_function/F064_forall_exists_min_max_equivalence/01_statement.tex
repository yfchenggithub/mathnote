\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（存在最值）：}}] 函数 $g$ 在区间 $I_1$ 上存在最小值（或最大值），函数 $h$ 在区间 $I_2$ 上存在最小值（或最大值），具体类型由不等号方向确定。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最小值转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任意 $g$ 不小于某个 $h$ 值，等价于 $g$ 的最小值不小于 $h$ 的最小值。
			\[
				\forall x_1\in I_1,\ \exists x_2\in I_2:\ g(x_1)\ge h(x_2) \Longleftrightarrow \min_{x\in I_1} g(x) \ge \min_{x\in I_2} h(x).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（最大值转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任意 $g$ 不大于某个 $h$ 值，等价于 $g$ 的最大值不大于 $h$ 的最大值。
			\[
				\forall x_1\in I_1,\ \exists x_2\in I_2:\ g(x_1)\le h(x_2) \Longleftrightarrow \max_{x\in I_1} g(x) \le \max_{x\in I_2} h(x).
			\]
	\end{description}
\end{statementbox}
