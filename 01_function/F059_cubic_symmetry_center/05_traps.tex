\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆拐点极值）}}]
			误认为对称中心是极大值或极小值点。
		\item[\textbf{\textcolor{CTrap}{正确理解（拐点非极值点）：}}]
			拐点是凹凸性改变点，不是极值点。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			将二阶导为零与一阶导为零的条件混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（坐标符号错误）}}]
			计算时漏掉负号，得 $x_0=\frac{b}{3a}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式符号）：}}]
			正确公式为 $x_0=-\frac{b}{3a}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（移项漏负号）：}}]
			解 $6ax+2b=0$ 移项后忘记变号。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略三次条件）}}]
			误认为对任意 $a$ 成立，包括 $a=0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须为三次）：}}]
			仅当 $a\neq0$ 时函数为三次函数才适用。
		\item[\textbf{\textcolor{CTrap}{错因分析（盲目套公式）：}}]
			未确认函数类型直接套用公式。
	\end{description}

\end{trapbox}
