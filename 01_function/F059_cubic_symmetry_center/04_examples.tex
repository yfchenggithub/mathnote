\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	求函数 $f(x)=x^3+3x^2-1$ 的对称中心。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求二阶导数）：}]
			$f'(x)=3x^2+6x$，$f''(x)=6x+6$。
			\par\textbf{理由：} 对称中心横坐标由二阶导为零确定。
		\item[\textbf{第二步（令二阶导为零）：}]
			解 $6x+6=0$，得 $x=-1$。
			\par\textbf{理由：} 拐点横坐标。
		\item[\textbf{第三步（代入原函数）：}]
			$f(-1)=(-1)^3+3(-1)^2-1$，得 $f(-1)=1$。
			\par\textbf{理由：} 得到纵坐标。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{对称中心为 $(-1,1)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知三次函数 $f(x)=ax^3+bx^2+cx+d$ 的图像关于点 $(1,2)$ 中心对称，且 $a=1$，求 $b$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（公式横坐标）：}]
			对称中心横坐标公式 $x_0=-\dfrac{b}{3a}$。由条件 $x_0=1$，$a=1$，得 $1=-\dfrac{b}{3}$。
			\par\textbf{理由：} 直接代入公式。
		\item[\textbf{第二步（解一次方程）：}]
			两边乘以 $3$ 得 $3=-b$，故 $b=-3$。
			\par\textbf{理由：} 求解 $b$。
		\item[\textbf{第三步（验证纵坐标）：}]
			$f(1)=1+b+c+d$，由 $f(1)=2$ 得 $1+b+c+d=2$。代入 $b=-3$ 得 $c+d=4$（可供后续使用，本题不需）。
			\par\textbf{理由：} 纵坐标条件用于一致性检查。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$b=-3$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x)=x^3-3x^2+4x+2$，求 $f(0)+f(2)$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求对称中心）：}]
			$x_0=-\dfrac{b}{3a}$，代入 $a=1,\ b=-3$ 得 $x_0=1$。$f(1)=1-3+4+2$，得 $4$，对称中心为 $(1,4)$。
			\par\textbf{理由：} 套用对称中心公式。
		\item[\textbf{第二步（写对称关系）：}]
			由中心对称性质，$f(1+t)+f(1-t)=2f(1)$。由 $f(1)=4$ 得 $f(1+t)+f(1-t)=8$。
			\par\textbf{理由：} 对称中心定义。
		\item[\textbf{第三步（代入求值）：}]
			取 $t=1$，则 $f(0)+f(2)=8$。
			\par\textbf{理由：} $0=1-1$，$2=1+1$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$f(0)+f(2)=8$。}
\end{examplebox}
