	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三次函数）：}}]
			函数 $f(x)=ax^3+bx^2+cx+d$，其中 $a\neq 0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对称中心）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三次函数图像的唯一对称中心是其凹凸性改变点（拐点）。
			\[
				\boxed{\left(-\frac{b}{3a},\; f\!\left(-\frac{b}{3a}\right)\right)}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（b为零特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当二次项系数为零时对称中心位于 $y$ 轴上。
			\[
				\boxed{(0,\; d)}
			\]
	\end{description}

\end{statementbox}
