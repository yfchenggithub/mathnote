\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{三次函数 $f(x)=ax^3+bx^2+cx+d\ (a\neq0)$ 的对称中心即拐点，坐标为 $\left(-\dfrac{b}{3a},\ f\!\left(-\dfrac{b}{3a}\right)\right)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三次函数）：} 函数必须为三次函数，即 $a\neq0$。
		\item \textbf{条件 2（定义域）：} 全体实数 $\mathbb{R}$，无特殊限制。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（横坐标符号）：} 注意 $x_0=-\dfrac{b}{3a}$ 中的负号不可遗漏。
		\item \textbf{检查项（二阶导验证）：} 可通过 $f''(x)=0$ 快速获得横坐标，再进行对称性验证。
	\end{itemize}

\end{summarybox}
