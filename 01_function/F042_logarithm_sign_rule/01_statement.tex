\begin{statementbox}
  \textbf{\textcolor{CStatement}{先回答一个问题}}\quad
  不知道 $\log_2 7$ 的准确值，也能判断它是正数还是负数吗？\textbf{能！只要把底数、真数分别和 $1$ 比一比。}

  \medskip
  \textbf{\textcolor{CStatement}{使用条件}}\quad $a>0,\ a\neq1,\ b>0$（底数是正数但不能等于 $1$；真数必须为正）。

  \medskip
  \textbf{\textcolor{CStatement}{核心结论：底真同侧为正，异侧为负}}
  \[
    \boxed{\ \log_a b>0\iff(a-1)(b-1)>0\ },\qquad
    \boxed{\ \log_a b<0\iff(a-1)(b-1)<0\ }.
  \]
  \begin{center}
  \input{assets/tikz/F042_logarithm_sign_rule_01_statement_01.tex}
  \end{center}

  \textbf{\textcolor{CStatement}{四种情况，一眼看懂}}
  \[
    \begin{array}{c|cc}
      &b>1&0<b<1\\ \hline
      a>1 & + & -\\
      0<a<1 & - & +
    \end{array}
  \]
  \textbf{例：}\,$\log_2 4=2>0$，$\log_2\frac14=-2<0$，
  $\log_{\frac12}\frac14=2>0$，$\log_{\frac12}4=-2<0$。

  \textbf{\textcolor{CStatement}{边界提醒}}\quad $\log_a1=0$；$a=1$ 或 $b\leq0$ 时，对数不存在。先检查条件，再判正负。
\end{statementbox}
