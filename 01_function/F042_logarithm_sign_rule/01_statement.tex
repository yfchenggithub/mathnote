\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定义域范围）：}}] $a>0$, $a\neq 1$, $b>0$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正负判号）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对数的正负由底数与真数是否同时位于1的同侧决定。
			\[
				\log_a b > 0 \iff (a-1)(b-1) > 0,
				\qquad
				\log_a b < 0 \iff (a-1)(b-1) < 0.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（区间判号）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			底真同区间（都大于1或都小于1）对数为正，底真跨区间对数为负。
			\[
				\begin{aligned}
					&\log_a b>0 \iff (a>1\land b>1)\lor(0<a<1\land 0<b<1);\\
					&\log_a b<0 \iff (a>1\land 0<b<1)\lor(0<a<1\land b>1).
			\end{aligned}
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{使用提醒：}}
	注意检验真数 $b>0$；当 $b=1$ 时 $\log_a b = 0$，不适用该法则。
\end{statementbox}
