\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 比较 $e^{0.05}$ 与 $1.05$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
		      观察发现 $e^{0.05}$ 对应 $e^x$ 中 $x=0.05$，$1.05$ 对应 $x+1$ 中 $x=0.05$。
		      \par\textbf{理由：} 根据指数切线不等式 $e^x\ge x+1$，等号仅当 $x=0$ 成立。
		\item[\textbf{第二步（套用结论）：}]
		      取 $x=0.05$，由 $e^x\ge x+1$ 得
		      \[
			      e^{0.05}\ge1.05.
		      \]
		      又因为 $0.05\neq0$，所以等号不成立。
		      \par\textbf{理由：} 代入不等式，注意取等条件。
		\item[\textbf{第三步（得到大小关系）：}]
		      因此
		      \[
			      e^{0.05}>1.05.
		      \]
		      \par\textbf{理由：} 非取等点处严格大于。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$e^{0.05}>1.05$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 比较 $\ln 1.08$ 与 $0.08$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
		      $1.08=1+0.08$，对应 $\ln(1+t)$ 中 $t=0.08$；$0.08$ 对应 $t$。
		      \par\textbf{理由：} 对数变形不等式 $\ln(1+t)\le t$ 要求 $t>-1$，这里 $0.08>-1$。
		\item[\textbf{第二步（套用结论）：}]
		      取 $t=0.08$，由 $\ln(1+t)\le t$ 得
		      \[
			      \ln1.08\le0.08.
		      \]
		      又因为 $0.08\neq0$，所以等号不成立。
		      \par\textbf{理由：} 代入不等式，注意取等条件。
		\item[\textbf{第三步（得到大小关系）：}]
		      因此
		      \[
			      \ln1.08<0.08.
		      \]
		      \par\textbf{理由：} 非取等点处严格小于。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\ln 1.08<0.08$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 证明：对任意实数 $x$，有 $e^x\ge e\,x$，并指出等号成立条件。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变量代换）：}]
		      在指数不等式 $e^u\ge u+1$ 中，令 $u=x-1$。
		      \par\textbf{理由：} 因为 $x\in\mathbb{R}$，所以 $u=x-1\in\mathbb{R}$，不等式仍成立。
		\item[\textbf{第二步（得到新不等式）：}]
		      代入得
		      \[
			      e^{x-1}\ge (x-1)+1=x.
		      \]
		      \par\textbf{理由：} 左边变成 $e^{x-1}$，右边化简为 $x$。
		\item[\textbf{第三步（变形为目标形式）：}]
		      两边同时乘以正数 $e$，得
		      \[
			      e^x\ge e\,x.
		      \]
		      \par\textbf{理由：} 因为 $e>0$，乘以 $e$ 不改变不等号方向。
		\item[\textbf{第四步（确定取等条件）：}]
		      原不等式 $e^u\ge u+1$ 的等号当且仅当 $u=0$，即 $x-1=0$，所以 $x=1$。
		      \par\textbf{理由：} 取等条件随代换同步变化。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\forall x\in\mathbb{R}$，$e^x\ge e\,x$，等号当且仅当 $x=1$。}
