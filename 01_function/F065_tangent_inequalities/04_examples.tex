\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 比较 $e^{0.05}$ 与 $1.05$ 的大小。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（识别结构）：}]
			观察发现 $e^{0.05}$ 对应 $e^x$ 中 $x=0.05$，$1.05$ 对应 $x+1$ 中 $x=0.05$。\n
			\par\textbf{理由：}\n 根据指数切线不等式 $e^x\ge x+1$，等号仅当 $x=0$ 成立。\n
		\item[\textbf{第二步（套用结论）：}]
			取 $x=0.05$，由 $e^x\ge x+1$ 得 $e^{0.05}\ge 1.05$，且 $x=0.05\neq0$，故等号不成立，为严格大于。\n
			\par\textbf{理由：}\n 代入不等式，注意取等条件。\n
		\item[\textbf{第三步（得到大小关系）：}]
			因此 $e^{0.05}>1.05$。\n
			\par\textbf{理由：}\n 严格大于。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$e^{0.05}>1.05$}\n\n

	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 比较 $\ln 1.08$ 与 $0.08$ 的大小。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（识别结构）：}]
			$1.08=1+0.08$，对应 $\ln(1+t)$ 中 $t=0.08$；$0.08$ 对应 $t$。\n
			\par\textbf{理由：}\n 对数变形不等式 $\ln(1+t)\le t$（$t>-1$）。\n
		\item[\textbf{第二步（套用结论）：}]
			取 $t=0.08$，由 $\ln(1+t)\le t$ 得 $\ln 1.08\le 0.08$，且 $t=0.08\neq0$，等号不成立。\n
			\par\textbf{理由：}\n 代入不等式，注意取等条件。\n
		\item[\textbf{第三步（得到大小关系）：}]
			因此 $\ln 1.08<0.08$。\n
			\par\textbf{理由：}\n 严格小于。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\ln 1.08<0.08$}\n\n

	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 证明：对任意实数 $x$，有 $e^x\ge ex$，并指出等号成立条件。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（变量代换）：}]
			在指数不等式 $e^x\ge x+1$ 中，将 $x$ 替换为 $x-1$。\n
			\par\textbf{理由：}\n $x-1$ 仍是实数，不等式仍成立。\n
		\item[\textbf{第二步（得到新不等式）：}]
			$e^{x-1}\ge (x-1)+1$，即 $e^{x-1}\ge x$。\n
			\par\textbf{理由：}\n 代入后左边 $e^{x-1}$，右边 $x$。\n
		\item[\textbf{第三步（变形为目标形式）：}]
			两边同时乘以 $e$（$e>0$），得 $e^x\ge ex$。\n
			\par\textbf{理由：}\n $e^{x-1}\ge x$ 等价于 $e^x\ge ex$。\n
		\item[\textbf{第四步（确定取等条件）：}]
			原不等式等号在 $x=0$ 时成立；替换后 $x-1=0$ 即 $x=1$ 时等号成立。故 $e^x\ge ex$ 等号当且仅当 $x=1$。\n
			\par\textbf{理由：}\n 替换不改变等号对应变量关系。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\forall x\in\mathbb{R},\ e^x\ge ex$，等号当 $x=1$。}\n
\end{examplebox}
