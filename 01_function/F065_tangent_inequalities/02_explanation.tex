\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n指数函数 $e^x$ 和对数函数 $\ln x$ 在特殊点处的切线是它们图象的“支撑线”，一个恒在下方，一个恒在上方。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（对数构造）：}]
			构造函数 $f(x)=\ln x-(x-1)$，利用导数研究最值，证明 $f(x)\le 0$。\n
		\item[\textbf{要点二（指数构造）：}]
			构造函数 $g(x)=e^x-(x+1)$，同样利用导数证明 $g(x)\ge 0$。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（切线位置）}}\n\n$y=\ln x$ 在 $(1,0)$ 处的切线为 $y=x-1$，除切点外曲线全部位于切线下方；$y=e^x$ 在 $(0,1)$ 处的切线为 $y=x+1$，除切点外曲线全部位于切线上方。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（不等式放缩）}}\n\n将指数或对数函数转化为一次函数，从而简化代数式比较或证明，常用于导数压轴题中的放缩步骤。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（导数放缩）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（比较大小）：} 直接代入具体数值，快速判断大小关系。\n
		\item \textbf{考法二（证明不等式）：} 将复杂函数放缩为一次函数，辅助证明。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n\textbf{两个不等式本质是同一个性质：$e^x$ 和 $\ln x$ 互为反函数，它们的切线关于直线 $y=x$ 对称。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（估值比较）：} 当自变量接近 $1$ 或 $0$ 时，切线近似效果很好，如估计 $\ln 1.1$、$e^{0.1}$。\n
		\item \textbf{场景二（放缩证明）：} 在导数综合题中，用 $e^x\ge x+1$ 或 $\ln x\le x-1$ 对目标进行放缩，简化求值或证明。\n
	\end{itemize}
	\n
\end{explanationbox}
