\begin{statementbox}
	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件一（定义域限制）：}}]
			对数不等式要求 $x>0$；指数不等式对任意实数 $x$ 成立。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（对数切线）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			对数函数 $\ln x$ 的图象在点 $(1,0)$ 处的切线为 $y=x-1$，曲线位于切线下方。\n
			\[
				\n \ln x \le x-1,
				\qquad
				x>0.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{结论二（指数切线）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			指数函数 $e^x$ 的图象在点 $(0,1)$ 处的切线为 $y=x+1$，曲线位于切线上方。\n
			\[
				\n e^x \ge x+1,
				\qquad
				x\in\mathbb{R}.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（常见变形）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过变量代换得到其他常用放缩形式。\n
			\[
				\n \ln(1+t)\le t\;(t>-1),
				\qquad
				e^x\ge ex\;(x\in\mathbb{R}).\n
			\]
			\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{等号/取等条件：}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em]
		\n
		\item 对于 $\ln x\le x-1$，等号成立当且仅当 $x=1$；\n
		\item 对于 $e^x\ge x+1$，等号成立当且仅当 $x=0$；\n
		\item 对于 $\ln(1+t)\le t$，等号成立当且仅当 $t=0$（即 $x=1$）；\n
		\item 对于 $e^x\ge ex$，等号成立当且仅当 $x=1$（即 $e^x\ge e\cdot x$）。\n
	\end{itemize}
	\n
\end{statementbox}
