\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略对数定义域）：}}]
		      使用 $\ln x\le x-1$ 时，容易默认任何 $x$ 都可以代入，从而错误地用于 $x\le0$ 的情形。
		\item[\textbf{\textcolor{CTrap}{正确理解（范围限制）：}}]
		      仅当 $x>0$ 时不等式 $\ln x\le x-1$ 才有意义；若 $x\le0$，$\ln x$ 无定义。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义域忽略）：}}]
		      未注意对数函数定义域，随意套用导致错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（取等条件遗忘）：}}]
		      在比较大小或证明时，只写不等式方向而不说明等号何时成立，导致结论不严谨。
		\item[\textbf{\textcolor{CTrap}{正确理解（等号条件）：}}]
		      使用 $e^x\ge x+1$ 时，相等只在 $x=0$ 成立；使用 $\ln x\le x-1$ 时，相等只在 $x=1$ 成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（取值验证忽略）：}}]
		      未代入特殊点验证等号，导致范围判断不精确。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（变形范围遗忘）：}}]
		      使用 $\ln(1+t)\le t$ 时，忘记 $t>-1$ 的条件，错误地用于 $t\le -1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（变形范围）：}}]
		      变形来源于 $\ln x\le x-1$，令 $x=1+t$，所以 $1+t>0$，即 $t>-1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（变量代换条件丢失）：}}]
		      变量代换后未回推新变量的取值范围。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（把 $e^x\ge e\,x$ 误读成 $e^x\ge e^x$）：}}]
		      看到 $e^x\ge ex$ 时，容易把右边的 $ex$ 误读为指数 $e^x$。
		\item[\textbf{\textcolor{CTrap}{正确理解（乘法写清楚）：}}]
		      这里的 $ex$ 表示 $e\cdot x$，为了避免误读，建议写成 $e^x\ge e\,x$ 或 $e^x\ge e\cdot x$。
		\item[\textbf{\textcolor{CTrap}{错因分析（排版歧义）：}}]
		      数学排版中相邻字母常表示乘法，但在指数函数附近容易造成视觉混淆。
	\end{description}
\end{trapbox}
