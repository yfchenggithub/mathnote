\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{两个切线不等式及其变形是导数放缩中最常用的工具，可将指数、对数函数放缩为一次函数。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（对数定义域）：} $\ln x\le x-1$ 要求 $x>0$；变形 $\ln(1+t)\le t$ 要求 $t>-1$。
		\item \textbf{条件 2（指数无条件）：} $e^x\ge x+1$ 对任意实数成立；$e^x\ge e\,x$ 也对任意实数成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{核心结论}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{对数切线：} $\ln x\le x-1$，$x>0$，等号当且仅当 $x=1$。
		\item \textbf{指数切线：} $e^x\ge x+1$，$x\in\mathbb{R}$，等号当且仅当 $x=0$。
		\item \textbf{常见变形：} $\ln(1+t)\le t$，$t>-1$；$e^x\ge e\,x$，$x\in\mathbb{R}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略取等）：} 使用不等式时务必考虑等号成立条件，否则可能丢失精确值。
		\item \textbf{检查项（取值范围）：} 涉及对数时先确认真数大于零，涉及指数时无需额外限制。
		\item \textbf{排版提醒：} $e^x\ge e\,x$ 中右边表示 $e\cdot x$，不是 $e^x$。
	\end{itemize}
\end{summarybox}
