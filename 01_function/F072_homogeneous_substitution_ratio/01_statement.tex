\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（零次齐次）：}}]
			不等式 $F(x_1,x_2)\ge0$ 为零次齐次（即对任意 $\lambda>0$，$F(\lambda x_1,\lambda x_2)=F(x_1,x_2)$），且 $x_1,x_2>0$，$x_1\neq x_2$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（比值换元）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过比值 $t=x_1/x_2$ 将二元不等式化为一元不等式。
			\[
				F(x_1,x_2)\ge0\;\Longleftrightarrow\;\varphi(t)\ge0,\quad \varphi(t)=F(t,1),\;t>0,\;t\neq1.
			\]
			应用时常设 $x_1>x_2$，此时 $t>1$。
		\item[\textbf{\textcolor{CStatement}{常见变形（对数均值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对数均值不等式 $\dfrac{x_1-x_2}{\ln x_1-\ln x_2}<\dfrac{x_1+x_2}{2}$（$x_1\neq x_2$）经换元化为 $\dfrac{t-1}{\ln t}<\dfrac{t+1}{2}$（$t>1$）。
			\[
				\frac{x_1-x_2}{\ln x_1-\ln x_2}<\frac{x_1+x_2}{2}\;\Longleftrightarrow\;\frac{t-1}{\ln t}<\frac{t+1}{2},\quad t>1.
			\]
	\end{description}

\end{statementbox}
