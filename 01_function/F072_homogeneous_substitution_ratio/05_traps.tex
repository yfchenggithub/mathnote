\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（齐次性验证）}}]
			对非零次齐次的不等式使用比值换元。
		\item[\textbf{\textcolor{CTrap}{正确理解（必需验证齐次）：}}]
			换元前提是 $F(\lambda x_1,\lambda x_2)=F(x_1,x_2)$ 对所有 $\lambda>0$ 成立。必须先用定义验证。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件未检查）：}}]
			错误认为所有双变量不等式都可进行比值换元，忽略了齐次性要求。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（定义域遗漏）}}]
			换元后忽略 $t\neq1$ 或 $t>0$ 的限制。
		\item[\textbf{\textcolor{CTrap}{正确理解（明确定义域条件）：}}]
			由 $x_1,x_2>0$ 得 $t>0$，由 $x_1\neq x_2$ 得 $t\neq1$。作答时明确标出。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提约束）：}}]
			忘记原题 $x_1\neq x_2$ 或 $x_1,x_2>0$，导致定义域扩大或缩小。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（对称性误判）}}]
			默认 $t>1$ 就代表全部情况，未考虑 $0<t<1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（按需分类讨论）：}}]
			若不等式关于 $x_1,x_2$ 不对称，必须分别讨论 $t>1$ 和 $0<t<1$，或通过 $t\leftrightarrow 1/t$ 转换。
		\item[\textbf{\textcolor{CTrap}{错因分析（滥用对称性）：}}]
			错误认为换元后函数总是关于 $t$ 与 $1/t$ 对称，未作验证即只取一侧。
	\end{description}

\end{trapbox}
