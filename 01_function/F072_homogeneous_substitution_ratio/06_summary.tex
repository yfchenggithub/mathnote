\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{对于零次齐次双变量不等式 $F(x_1,x_2)\ge0$（$x_1,x_2>0$，$x_1\neq x_2$），令 $t=x_1/x_2$，则问题等价于单变量不等式 $\varphi(t)=F(t,1)\ge0$（$t>0,t\neq1$）。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（零次齐次）：} 不等式必须满足 $F(\lambda x_1,\lambda x_2)=F(x_1,x_2)$ 对所有 $\lambda>0$ 成立。
		\item \textbf{条件2（正数且不等）：} $x_1,x_2>0$ 且 $x_1\neq x_2$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（验证齐次性）：} 换元前必须确认函数为零次齐次，不可滥用。
		\item \textbf{检查项（定义域与对称）：} 换元后记得 $t>0$ 且 $t\neq1$；若不等式不对称，需同时考虑 $t>1$ 和 $0<t<1$，或通过 $t\leftrightarrow 1/t$ 转化。
	\end{itemize}

\end{summarybox}
