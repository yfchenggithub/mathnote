\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（漏乘内层导数）}}] 求 $y=\sin 2x$ 的导数时，误得 $y'=\cos 2x$，遗漏因子 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（内外分别求）：}}] 正确运算：$y' = \cos(2x)\cdot (2x)'$，故 $y' = 2\cos 2x$。
		\item[\textbf{\textcolor{CTrap}{错因分析（未分内外）：}}] 没有将函数拆解为内外层，忽略了内层函数的导数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（外层代入遗漏）}}] 求 $y=\ln(\sin x)$ 的导数时，误得 $y'=\frac{1}{\sin x}$，缺少 $\cos x$ 项。
		\item[\textbf{\textcolor{CTrap}{正确理解（每层都乘）：}}] 正确运算：$y' = \frac{1}{\sin x}\cdot(\sin x)'$，即 $y' = \frac{\cos x}{\sin x}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（求导不全）：}}] 外层求导后忘记乘以内层导数，只算了外层。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（层次混淆少乘）}}] 求 $y=\ln(\cos 2x)$ 导数时，误得 $y'=-\tan 2x$，遗漏因子 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（逐层全乘）：}}] 正确运算：$y' = \frac{1}{\cos 2x}\cdot(-\sin 2x)\cdot 2$，得 $y' = -2\tan 2x$。
		\item[\textbf{\textcolor{CTrap}{错因分析（结构不清）：}}] 未正确识别函数嵌套的层数，导致遗漏最内层导数。
	\end{description}
\end{trapbox}
