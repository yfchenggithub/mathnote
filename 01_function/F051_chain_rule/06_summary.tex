\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{复合函数的导数等于外层导数乘内层导数，代入再乘。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（内层可导）：} 内层函数 $u=g(x)$ 在 $x$ 处可导。
		\item \textbf{条件2（外层可导）：} 外层函数 $y=f(u)$ 在对应 $u=g(x)$ 处可导。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（逐层求导）：} 每层导数都要乘，不能遗漏。
		\item \textbf{检查项（内层代入）：} 外层导数求完后必须将内层函数代入自变量位置。
	\end{itemize}
\end{summarybox}
