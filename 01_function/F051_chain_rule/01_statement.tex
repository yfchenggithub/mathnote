\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（内层可导）：}}] 函数 $u=g(x)$ 在点 $x$ 处可导。
		\item[\textbf{\textcolor{CStatement}{条件2（外层可导）：}}] 函数 $y=f(u)$ 在对应点 $u=g(x)$ 处可导。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（复合可导）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			复合函数的导数等于外层导数乘以内层导数。
			\[
				\frac{dy}{dx}=f'(g(x))\cdot g'(x)
			\]
			\[
				\frac{dy}{dx}=\frac{dy}{du}\cdot\frac{du}{dx}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（多层嵌套）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每次嵌套都乘一次内层导数，逐层求导相乘。
			\[
				\frac{dy}{dx}=f'(g(h(x)))\cdot g'(h(x))\cdot h'(x).
			\]
	\end{description}
\end{statementbox}
