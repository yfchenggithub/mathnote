  \textbf{\textcolor{CStatement}{这条结论解决什么问题？}}\par\nopagebreak[4]

  题目只说函数或导函数\textbf{关于哪条直线、哪个点对称}，没有给出解析式，怎样迅速推断另一个图像的对称性？关键是比较对称位置的\textbf{高度}和\textbf{斜率}。

  \textbf{\textcolor{CStatement}{一眼看懂：同一个中点，图像如何改变}}\par\nopagebreak[4]

  \medskip
  \textbf{\textcolor{CStatement}{使用条件}}
  \begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
    \item[\textbf{条件一（可导）：}] $f(x)$ 在 $\mathbb R$ 上可导；$f'(x)$ 表示它的导函数。
    \item[\textbf{条件二（对称位置）：}] $a,b$ 为固定实数，$t$ 为任意实数；$a+t$ 和 $a-t$ 距离 $a$ 一样远。
  \end{description}

  \medskip
  \textbf{\textcolor{CStatement}{结论一：原函数轴对称 $\Longleftrightarrow$ 导函数中心对称}}
  \[
    \boxed{f(a+t)=f(a-t)\ \Longleftrightarrow\ f'(a+t)=-f'(a-t)}
  \]
  $f$ 关于\textbf{直线 $x=a$} 对称，当且仅当 $f'$ 关于\textbf{点 $(a,0)$} 对称。

  \medskip
  \textbf{\textcolor{CStatement}{结论二：原函数中心对称 $\Longleftrightarrow$ 导函数轴对称}}
  \[
    \boxed{f(a+t)+f(a-t)=2f(a)\ \Longleftrightarrow\ f'(a+t)=f'(a-t)}
  \]
  $f$ 关于\textbf{点 $(a,f(a))$} 对称，当且仅当 $f'$ 关于\textbf{直线 $x=a$} 对称。

  \medskip
  \textbf{\textcolor{CStatement}{结论三：导函数关于 $(a,b)$ 中心对称}}
  \[
    f'(a+t)+f'(a-t)=2b
    \ \Longleftrightarrow\ 
    \boxed{f(a+t)-f(a-t)=2bt}.
  \]
  \begin{itemize}[leftmargin=*,itemsep=0.25em,topsep=0.2em]
    \item $b=0$：原函数关于 $x=a$ 对称。
    \item $b\ne0$：原函数\textbf{不关于 $x=a$ 对称}；但可能关于别的直线对称。
  \end{itemize}

  \medskip
  \textbf{\textcolor{CStatement}{速记口诀：}}\textbf{原轴导中，原中导轴；对称位置的横坐标中点始终是 $a$。}
\end{statementbox}
