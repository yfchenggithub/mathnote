
\[
	\text{圆锥曲线硬解定理}
\]
\[
	\text{3. 直线与抛物线}
\]

\[
	\text{（1）抛物线开口向右}
\]

联立
\[
	\begin{cases}
		Ax+By+C=0,\quad (A\ne 0), \\[4pt]
		y^2=2px,\quad (p>0),
	\end{cases}
\]
得：
\[
	A^2x^2+(2AC-2B^2p)x+C^2=0
\]
或
\[
	Ay^2+2Bpy+2Cp=0.
\]

\[
	x_1+x_2=-\frac{2AC-2B^2p}{A^2},
	\qquad
	x_1x_2=\frac{C^2}{A^2}
\]
或
\[
	y_1+y_2=-\frac{2Bp}{A},
	\qquad
	y_1y_2=\frac{2Cp}{A}.
\]

\[
	\text{（2）抛物线开口向左}
\]

联立
\[
	\begin{cases}
		Ax+By+C=0,\quad (A\ne 0), \\[4pt]
		y^2=-2px,\quad (p>0),
	\end{cases}
\]
得：
\[
	A^2x^2+(2AC+2B^2p)x+C^2=0
\]
或
\[
	Ay^2-2Bpy-2Cp=0.
\]

\[
	x_1+x_2=-\frac{2AC+2B^2p}{A^2},
	\qquad
	x_1x_2=\frac{C^2}{A^2}
\]
或
\[
	y_1+y_2=\frac{2Bp}{A},
	\qquad
	y_1y_2=-\frac{2Cp}{A}.
\]

\[
	\text{（3）抛物线开口向上}
\]

联立
\[
	\begin{cases}
		Ax+By+C=0,\quad (A\ne 0), \\[4pt]
		x^2=2py,\quad (p>0),
	\end{cases}
\]
得：
\[
	Bx^2+2Apx+2Cp=0
\]
或
\[
	B^2y^2+(2BC-2A^2p)y+C^2=0.
\]

\[
	x_1+x_2=-\frac{2Ap}{B},
	\qquad
	x_1x_2=\frac{2Cp}{B}
\]
或
\[
	y_1+y_2=-\frac{2BC-2A^2p}{B^2},
	\qquad
	y_1y_2=\frac{C^2}{B^2}.
\]

\[
	\text{（4）抛物线开口向下}
\]

联立
\[
	\begin{cases}
		Ax+By+C=0,\quad (A\ne 0), \\[4pt]
		x^2=-2py,\quad (p>0),
	\end{cases}
\]
得：
\[
	Bx^2-2Apx-2Cp=0
\]
或
\[
	B^2y^2+(2BC+2A^2p)y+C^2=0.
\]

\[
	x_1+x_2=\frac{2Ap}{B},
	\qquad
	x_1x_2=-\frac{2Cp}{B}
\]
或
\[
	y_1+y_2=-\frac{2BC+2A^2p}{B^2},
	\qquad
	y_1y_2=\frac{C^2}{B^2}.
\]
