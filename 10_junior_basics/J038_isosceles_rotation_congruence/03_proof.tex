\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	通过 SAS 证明 $\triangle ABD\cong\triangle ACE$，关键证明夹角 $\angle BAD = \angle CAE$；然后利用旋转全等得出对应边 $BD$ 与 $CE$ 的夹角关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（等腰条件）：}]
			由题设，$\triangle ABC$ 和 $\triangle ADE$ 都是等腰三角形，有
			\[
				AB = AC,
				\qquad
				AD = AE,
				\qquad
				\angle BAC = \angle DAE = \alpha .
			\]
			\par\textbf{理由：}
			直接来自已知条件；这是全等所需的边和角准备。

		\item[\textbf{步骤二（夹角相等）：}]
			因为 $\angle BAC = \angle DAE$，两边同时加上 $\angle CAD$，得到
			\[
				\angle BAC + \angle CAD = \angle DAE + \angle CAD,
			\]
			即 $\angle BAD = \angle CAE$。（若图形位置导致相减，同样可得相等关系，此处取典型情况。）
			\par\textbf{理由：}
			利用等式的可加性（或有向角的加减）。

		\item[\textbf{步骤三（全等判定）：}]
			在 $\triangle ABD$ 与 $\triangle ACE$ 中，
			\[
				\begin{aligned}
					AB &= AC \quad (\text{given}),\\
					AD &= AE \quad (\text{given}),\\
					\angle BAD &= \angle CAE \quad (\text{proved}),
			\end{aligned}
			\]
			所以 $\triangle ABD \cong \triangle ACE$（SAS）。
			因此 $BD = CE$。
			\par\textbf{理由：}
			两边及其夹角对应相等，判定三角形全等；全等三角形对应边相等。

		\item[\textbf{步骤四（夹角关系）：}]
			由 $\triangle ABD \cong \triangle ACE$，且因为 $AB = AC$，$\angle BAC = \alpha$，所以 $AB$ 绕点 $A$ 旋转 $\alpha$ 后与 $AC$ 重合；同理 $AD$ 绕 $A$ 旋转 $\alpha$ 后与 $AE$ 重合。因此 $\triangle ABD$ 整体绕点 $A$ 旋转 $\alpha$ 后与 $\triangle ACE$ 重合。于是 $BD$ 也绕点 $A$ 旋转 $\alpha$ 后与 $CE$ 重合，故 $BD$ 与 $CE$ 的夹角等于 $\alpha$（或 $180^\circ - \alpha$，取决于旋转方向）。
			\[
				\angle(BD, CE) = \alpha \ \text{or} \ 180^\circ - \alpha .
			\]
			\par\textbf{理由：}
			旋转重合性；对应线段旋转相同角度，夹角等于旋转角或其补角。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{综上，}$\triangle ABD \cong \triangle ACE$，$BD = CE$，且 $BD$ 与 $CE$ 的夹角为 $\alpha$ 或 $180^\circ-\alpha$，即手拉手模型的核心结论。

\end{proofbox}
