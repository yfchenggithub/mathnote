\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	设 $\triangle ABC$ 和 $\triangle ADE$ 均为等腰直角三角形，$\angle BAC = \angle DAE = 90^\circ$，$AB = AC$，$AD = AE$。将 $\triangle ADE$ 绕点 $A$ 旋转任意角度，连接 $BD$ 与 $CE$。求证：$BD = CE$，且 $BD \perp CE$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件识别）：}]
			由题设，$\triangle ABC$ 和 $\triangle ADE$ 都是等腰直角三角形，因此 $AB = AC$，$AD = AE$，且顶角 $\angle BAC = \angle DAE = 90^\circ$，满足手拉手模型的前提。
			\par\textbf{理由：}
			直接使用已知条件。

		\item[\textbf{第二步（全等证明）：}]
			因为 $\angle BAC = \angle DAE$，所以 $\angle BAD = \angle CAE$（两边同时加上或减去 $\angle CAD$）。在 $\triangle ABD$ 和 $\triangle ACE$ 中，
			\[
				\begin{aligned}
					AB &= AC \quad (\text{given}),\\
					AD &= AE \quad (\text{given}),\\
					\angle BAD &= \angle CAE \quad (\text{proved}),
			\end{aligned}
			\]
			所以 $\triangle ABD \cong \triangle ACE$（SAS）。从而 $BD = CE$。
			\par\textbf{理由：}
			利用手拉手模型的 SAS 全等。

		\item[\textbf{第三步（垂直推得）：}]
			由手拉手模型的推广结论，$BD$ 与 $CE$ 所在直线的夹角等于顶角 $90^\circ$ 或 $180^\circ-90^\circ=90^\circ$，故 $BD \perp CE$。
			\par\textbf{理由：}
			夹角性质：$\angle(BD, CE) = \alpha$ 或 $180^\circ-\alpha$，本节 $\alpha=90^\circ$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BD = CE$ 且 $BD \perp CE$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 和 $\triangle ADE$ 都是等边三角形，共顶点 $A$。将 $\triangle ADE$ 绕点 $A$ 旋转至某一位置，连接 $BD$ 与 $CE$，交于点 $P$。求 $\angle BPC$ 的度数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件识别）：}]
			等边三角形三边相等，三角均为 $60^\circ$。故 $AB = AC$，$AD = AE$，$\angle BAC = \angle DAE = 60^\circ$，满足手拉手条件。
			\par\textbf{理由：}
			等边三角形的性质。

		\item[\textbf{第二步（全等证明）：}]
			由 $\angle BAC = \angle DAE$ 得 $\angle BAD = \angle CAE$。因此
			\[
				\triangle ABD \cong \triangle ACE \ (\text{SAS}),
			\]
			从而 $BD = CE$，且 $\angle ABD = \angle ACE$。
			\par\textbf{理由：}
			手拉手全等。

		\item[\textbf{第三步（夹角计算）：}]
			由手拉手模型的推广结论，$BD$ 与 $CE$ 的夹角等于顶角 $60^\circ$ 或 $180^\circ-60^\circ=120^\circ$。因此 $\angle BPC = 60^\circ$ 或 $120^\circ$（具体取决于旋转方向，题目中若未指定，则两个都可能）。
			\par\textbf{理由：}
			直接套用夹角性质。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle BPC = 60^\circ$ 或 $120^\circ$，且 $BD = CE$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AB = AC$，$\angle BAC = 90^\circ$，$D$ 为 $BC$ 边上一点。以 $AD$ 为腰作等腰直角三角形 $ADE$，$\angle DAE = 90^\circ$，$AD = AE$，连接 $CE$。求证：$CE = BD$，且 $CE \perp BD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件识别）：}]
			$\triangle ABC$ 是等腰直角三角形，$AB = AC$，$\angle BAC = 90^\circ$；$\triangle ADE$ 是等腰直角三角形，$AD = AE$，$\angle DAE = 90^\circ$。两三角形共顶点 $A$，顶角相等，满足手拉手条件。
			\par\textbf{理由：}
			已知条件。

		\item[\textbf{第二步（全等证明）：}]
			因为 $\angle BAC = \angle DAE = 90^\circ$，所以 $\angle BAC - \angle DAC = \angle DAE - \angle DAC$，即 $\angle BAD = \angle CAE$。在 $\triangle ABD$ 和 $\triangle ACE$ 中，
			\[
				\begin{aligned}
					AB &= AC \quad (\text{given}),\\
					AD &= AE \quad (\text{given}),\\
					\angle BAD &= \angle CAE \quad (\text{proved}),
			\end{aligned}
			\]
			所以 $\triangle ABD \cong \triangle ACE$（SAS）。故 $BD = CE$，$\angle ABD = \angle ACE$。
			\par\textbf{理由：}
			SAS 全等。

		\item[\textbf{第三步（垂直证明）：}]
			在等腰直角三角形 $ABC$ 中，$\angle ABC = \angle ACB = 45^\circ$。由 $\angle ABD = \angle ACE$，得 $\angle ACE = 45^\circ$。因此 $\angle BCE = \angle ACB + \angle ACE = 45^\circ + 45^\circ = 90^\circ$，即 $CE \perp BC$。因为 $D$ 在 $BC$ 上，所以 $BC$ 即直线 $BD$，故 $CE \perp BD$。
			\par\textbf{理由：}
			全等三角形对应角相等；等腰直角三角形底角性质。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$CE = BD$，且 $CE \perp BD$。}
