	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（等腰共顶）：}}]
			设 $\triangle ABC$ 与 $\triangle ADE$ 有公共顶点 $A$，且满足 $AB = AC$，$AD = AE$，顶角 $\angle BAC = \angle DAE = \alpha$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（旋转全等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $\triangle ADE$ 绕 $A$ 旋转任意角度后，连接 $BD$ 与 $CE$，所得 $\triangle ABD$ 与 $\triangle ACE$ 全等。
			\[
				\triangle ABD \cong \triangle ACE \quad (\text{SAS}).
			\]
			从而 $BD = CE$。

		\item[\textbf{\textcolor{CStatement}{推广结论（夹角关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$BD$ 与 $CE$ 所在直线的夹角等于顶角 $\alpha$ 或 $180^\circ - \alpha$。
			\[
				\angle(BD, CE) = \alpha \ \text{or} \ 180^\circ - \alpha .
			\]
	\end{description}

\end{statementbox}
