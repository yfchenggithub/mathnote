\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件遗漏）}}]
			误认为只要两个等腰三角形共顶点，连接对应边后的三角形就全等。
		\item[\textbf{\textcolor{CTrap}{正确理解（顶角相等）：}}]
			两个等腰三角形不仅需共顶点，还必须顶角相等，才能保证夹角 $\angle BAD = \angle CAE$，从而满足 SAS 全等。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略条件）：}}]
			只记住了模型的外形（等腰 + 共顶点），忽略了核心的角相等条件。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（夹角补角）}}]
			认为 $BD$ 与 $CE$ 的夹角总是等于顶角 $\alpha$。
		\item[\textbf{\textcolor{CTrap}{正确理解（补角可能）：}}]
			夹角可能等于 $\alpha$，也可能等于 $180^\circ - \alpha$，取决于旋转方向和相对位置。
		\item[\textbf{\textcolor{CTrap}{错因分析（方向忽略）：}}]
			没有考虑有向角，默认旋转同向导致夹角相等。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（旋转限制）}}]
			认为只有某个特定旋转角度下全等才成立。
		\item[\textbf{\textcolor{CTrap}{正确理解（任意旋转）：}}]
			任意旋转角度下全等均成立，因为全等条件与旋转角无关。
		\item[\textbf{\textcolor{CTrap}{错因分析（动态误解）：}}]
			对旋转不变性理解不足，误以为旋转会破坏全等关系。
	\end{description}
\end{trapbox}
