\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(1,2)$，$B(3,5)$，$C(4,1)$。求以 $A,B,C$ 为顶点的平行四边形的第四个顶点 $D$ 的坐标（所有可能）。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列出坐标）：}]记 $A(1,2)$，$B(3,5)$，$C(4,1)$。
			\par\textbf{理由：}准备代入公式。
		\item[\textbf{第二步（分情况套公式）：}]利用结论，分别计算三种情况：

			以 $A,C$ 为对角：$D_1 = (1+4-3,\;2+1-5) = (2,-2)$;

			以 $A,B$ 为对角：$D_2 = (1+3-4,\;2+5-1) = (0,6)$;

			以 $B,C$ 为对角：$D_3 = (3+4-1,\;5+1-2) = (6,4)$。
			\par\textbf{理由：}利用结论公式 $D = (x_1+x_3-x_2,\;y_1+y_3-y_2)$ 等，代入坐标计算。
		\item[\textbf{第三步（写出答案）：}]所以 $D$ 的可能坐标为 $(2,-2)$，$(0,6)$，$(6,4)$。
			\par\textbf{理由：}三种对角配对均满足平行四边形条件。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{利用对点坐标和相等可快速求出所有可能的第四个顶点。}
	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知平行四边形 $ABCD$ 中，$A(1,2)$，$B(3,5)$，$D(2,-2)$，且 $A$ 与 $C$ 为对角顶点，求顶点 $C$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定对角关系）：}]由 $A$ 与 $C$ 为对角顶点，则 $B$ 与 $D$ 为另一组对角顶点。
			\par\textbf{理由：}平行四边形中，若一对对角顶点确定，另一对也随之确定。
		\item[\textbf{第二步（列等式）：}]根据对角顶点坐标和相等，得 $A+C = B+D$，即 $C = B+D-A$。
			\par\textbf{理由：}对角线互相平分，中点坐标相同。
		\item[\textbf{第三步（计算）：}]$C = (3+2-1,\;5+(-2)-2) = (4,1)$。
			\par\textbf{理由：}代入坐标进行加减运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$C(4,1)$。}
	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(1,2)$，$B(3,5)$，点 $C$ 在直线 $y=2x$ 上，且以 $A,B,C$ 为顶点的平行四边形第四个顶点 $D(0,1)$，求点 $C$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设出C坐标）：}]设 $C(t,2t)$，因为其在直线 $y=2x$ 上。
			\par\textbf{理由：}减少未知数个数。
		\item[\textbf{第二步（分类讨论）：}]根据平行四边形对角顶点关系，分三种情况。

			若 $A,C$ 为对角，则 $C = B+D-A = (2,4)$，即 $t=2$;

			若 $A,B$ 为对角，则 $C = A+B-D = (4,6)$，不符合 $y=2x$;

			若 $B,C$ 为对角，则 $C = A+D-B = (-2,-2)$，不符合 $y=2x$。
			\par\textbf{理由：}由对角顶点坐标和相等求出 $C$ 坐标，并结合 $C$ 在 $y=2x$ 上筛选。
		\item[\textbf{第三步（验证共线并确定答案）：}]以上只有 $C(2,4)$ 符合条件。验证 $A,B,C$ 不共线：$k_{AB}=3/2$，$k_{AC}=2\neq 3/2$。故 $C(2,4)$ 即为所求。
			\par\textbf{理由：}三点共线无法构成平行四边形。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$C(2,4)$。}
\end{examplebox}
