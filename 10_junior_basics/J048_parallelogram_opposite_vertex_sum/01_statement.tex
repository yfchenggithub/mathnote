\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（三点不共线）：}}]在平面直角坐标系中，已知不共线的三点 $A(x_1,y_1)$，$B(x_2,y_2)$，$C(x_3,y_3)$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（A、C为对角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			以 $A,C$ 为对角顶点，$D$ 坐标为 $A,C$ 坐标和减去 $B$。
			\[
				D(x_1+x_3-x_2,\; y_1+y_3-y_2)
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（A、B为对角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			以 $A,B$ 为对角顶点，$D$ 坐标为 $A,B$ 坐标和减去 $C$。
			\[
				D(x_1+x_2-x_3,\; y_1+y_2-y_3)
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（B、C为对角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			以 $B,C$ 为对角顶点，$D$ 坐标为 $B,C$ 坐标和减去 $A$。
			\[
				D(x_2+x_3-x_1,\; y_2+y_3-y_1)
			\]
	\end{description}
	\medskip
	\textbf{\textcolor{CStatement}{推广结论（坐标和相等）：}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	平行四边形两对角顶点坐标和相等。
	\[
		x_A+x_C = x_B+x_D,\quad y_A+y_C = y_B+y_D
	\]
\end{statementbox}
