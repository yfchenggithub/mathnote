\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{平行四边形第四个顶点坐标等于对角顶点坐标和减去第三点坐标。}
	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（三点不共线）：}只有三点不共线才能构成平行四边形。
		\item \textbf{条件2（对角配对）：}必须明确哪两点是对角顶点。
	\end{itemize}
	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（漏解）：}容易忽略其他可能的对角配对，导致漏解。应分三种情况分别计算。
		\item \textbf{检查项（共线判定）：}使用公式前应验证三点不共线。
	\end{itemize}
\end{summarybox}
