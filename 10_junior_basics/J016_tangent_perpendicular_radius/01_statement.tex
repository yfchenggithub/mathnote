\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（直线过圆上点）：}}]直线 $l$ 经过圆 $\odot O$ 上一点 $A$，且 $l \perp OA$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（判定定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			垂直于半径且过圆上对应点的直线是圆的切线。
			\[
				l \perp OA,\ A \in \odot O \implies l \cap \odot O = \{A\}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（性质定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆的切线垂直于过切点的半径。
			\[
				l \cap \odot O = \{A\} \implies l \perp OA.
			\]
	\end{description}
\end{statementbox}
