	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，已知 $\odot O$，$OA$ 为半径，点 $A$ 在圆上，直线 $l$ 经过点 $A$ 且 $l \perp OA$。求证：$l$ 是 $\odot O$ 的切线。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件识别）：}]已知 $A \in \odot O$，$l \perp OA$。
			\par\textbf{理由：}题目直接给出，无需推导。
		\item[\textbf{第二步（定理套用）：}]根据切线判定定理：若直线过圆上一点且垂直于该点半径，则该直线是圆的切线。
			\par\textbf{理由：}这是定理本身内容。
		\item[\textbf{第三步（得出结论）：}]因此 $l$ 是 $\odot O$ 的切线。
			\par\textbf{理由：}判定定理成立，符合条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$l$ 是 $\odot O$ 的切线}。

	\textbf{\textcolor{CExample}{例 2（解析应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $C: (x-1)^2+(y-2)^2=5$，点 $A(3,1)$，直线 $l$ 过点 $A$ 且与直线 $CA$ 垂直。判断直线 $l$ 与圆 $C$ 的位置关系，并说明理由。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（点在圆上验证）：}]代入 $A$ 坐标：$(3-1)^2+(1-2)^2=4+1=5$，满足圆方程，故 $A\in$ 圆 $C$。
			\par\textbf{理由：}圆上点坐标满足方程。
		\item[\textbf{第二步（求斜率垂直）：}]圆心 $C(1,2)$，$k_{CA}=\frac{1-2}{3-1}=-\frac{1}{2}$。因 $l\perp CA$，故 $k_l=2$。
			\par\textbf{理由：}垂直直线斜率乘积为 $-1$。
		\item[\textbf{第三步（判定切线）：}]$l$ 过圆上 $A$ 且 $l\perp CA$，由切线判定定理知 $l$ 是圆 $C$ 的切线，故相切。
			\par\textbf{理由：}判定定理直接使用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$l$ 与圆 $C$ 相切，直线 $l$ 是圆 $C$ 的切线}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $AB$ 是 $\odot O$ 的直径，$C$ 是圆上一点，$AD$ 平分 $\angle BAC$ 交 $\odot O$ 于 $D$，$DE\perp AC$ 于 $E$。求证：$DE$ 是 $\odot O$ 的切线。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（角平分线条件）：}]因为 $AD$ 平分 $\angle BAC$，所以 $\angle BAD = \angle DAC$。
			\par\textbf{理由：}角平分线定义。
		\item[\textbf{第二步（等腰与平行）：}]连结 $OD$，由 $OA=OD$ 得 $\angle ODA = \angle OAD$。又 $\angle OAD = \angle BAD$，所以 $\angle ODA = \angle DAC$，故 $OD\parallel AC$。
			\par\textbf{理由：}等边对等角；内错角相等则平行。
		\item[\textbf{第三步（垂直与切线）：}]因为 $DE\perp AC$ 且 $OD\parallel AC$，所以 $OD\perp DE$。又 $D$ 在 $\odot O$ 上，根据切线判定定理，$DE$ 是 $\odot O$ 的切线。
			\par\textbf{理由：}垂直传递结论；切线判定定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$DE$ 是 $\odot O$ 的切线}。
