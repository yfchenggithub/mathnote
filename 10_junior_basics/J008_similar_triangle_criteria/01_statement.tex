\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}

	设 $\triangle ABC$ 与 $\triangle A'B'C'$ 的对应边长分别为 $a,b,c$ 与 $a',b',c'$，对应角分别为 $\angle A,\angle B,\angle C$ 与 $\angle A',\angle B',\angle C'$。若 $\angle C=\angle C'=90^\circ$，则简记为直角情形。

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两角相等）：}}] $\angle A=\angle A'$, $\angle B=\angle B'$。
		\item[\textbf{\textcolor{CStatement}{条件2（边角边）：}}] $\dfrac{a}{a'}=\dfrac{b}{b'}$ 且 $\angle C=\angle C'$。
		\item[\textbf{\textcolor{CStatement}{条件3（三边比例）：}}]
			\[
				\begin{aligned}
					\frac{a}{a'} &= \frac{b}{b'} \\
					&= \frac{c}{c'}
			\end{aligned}
			\]
		\item[\textbf{\textcolor{CStatement}{条件4（斜边直角边）：}}] 在直角三角形中，$\angle C=\angle C'=90^\circ$ 且 $\dfrac{c}{c'}=\dfrac{a}{a'}$（或 $\dfrac{c}{c'}=\dfrac{b}{b'}$）。
		\item[\textbf{\textcolor{CStatement}{条件5（平行截线）：}}] 在$\triangle ABC$中，点$D\in AB$, $E\in AC$, $DE\parallel BC$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（三角形相似）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当两三角形满足上述任一条件时，它们的形状完全相同，对应角相等、对应边成比例。
			\[
				\triangle ABC \sim \triangle A'B'C'
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（面积比性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			相似三角形的面积比等于对应边长比的平方（相似比的平方）。
			\[
				\begin{aligned}
					\frac{S_{\triangle ABC}}{S_{\triangle A'B'C'}} &= \left(\frac{a}{a'}\right)^2 \\
					&= k^2
			\end{aligned}
			\]
			（其中 $k$ 为相似比，$k = \dfrac{a}{a'} = \dfrac{b}{b'} = \dfrac{c}{c'}$）
	\end{description}
\end{statementbox}
