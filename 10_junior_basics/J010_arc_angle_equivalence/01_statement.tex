\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（同圆或等圆）：}}] 必须在同圆或等圆中，即圆心相同或半径相等。
		\item[\textbf{\textcolor{CStatement}{条件2（顶点同侧）：}}] 圆周角的顶点必须位于所对弧的同一侧，以保证所对的是同一段弧。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（等弧对等角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			如果两条弧相等，那么它们所对的圆周角相等。
			\[
				\overset{\frown}{AB} = \overset{\frown}{CD} \Rightarrow \angle APB = \angle CQD.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（等角对等弧）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			如果两个圆周角相等，那么它们所对的弧相等。
			\[
				\angle APB = \angle CQD \Rightarrow \overset{\frown}{AB} = \overset{\frown}{CD}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（同弧对等角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同一弧所对的任意圆周角都相等。
			\[
				\angle ACB = \angle ADB.
			\]
	\end{description}
\end{statementbox}
