	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图所示，在圆$O$中，弧$AB$ = 弧$CD$，点$P$、$Q$在圆上，且$\angle APB$和$\angle CQD$所对的弧分别是$\overset{\frown}{AB}$和$\overset{\frown}{CD}$。若$\angle APB = 35^\circ$，求$\angle CQD$的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别等弧）：}] 已知条件中给出弧$AB$与弧$CD$相等。
			\par\textbf{理由：} 题目直接给出$\overset{\frown}{AB} = \overset{\frown}{CD}$。
		\item[\textbf{第二步（套用结论）：}] 根据“等弧所对的圆周角相等”，得$\angle APB = \angle CQD$。
			\par\textbf{理由：} 同圆或等圆中，等弧所对的圆周角相等（推论）。
		\item[\textbf{第三步（计算结果）：}] 因为$\angle APB = 35^\circ$，所以$\angle CQD = 35^\circ$。
			\par\textbf{理由：} 代入数值。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle CQD = 35^\circ$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$AB$是圆$O$的直径，$C$、$D$是圆上位于$AB$同侧的两点，且$\angle CAB = \angle DBA$。求证：弧$AC$ = 弧$BD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定角所对弧）：}] $\angle CAB$所对的弧是$\overset{\frown}{CB}$（不含$A$），$\angle DBA$所对的弧是$\overset{\frown}{DA}$（不含$B$）。
			\par\textbf{理由：} 圆周角顶点与圆上两点的连线所夹的弧（劣弧）。
		\item[\textbf{第二步（等角推等弧）：}] 由$\angle CAB = \angle DBA$，及等圆周角所对的弧相等，得$\overset{\frown}{CB} = \overset{\frown}{DA}$。
			\par\textbf{理由：} 等角对等弧的推论。
		\item[\textbf{第三步（利用直径等分圆）：}] 因为$AB$是直径，所以半圆$\overset{\frown}{ACB}$与半圆$\overset{\frown}{BDA}$相等，即$\overset{\frown}{AC} + \overset{\frown}{CB} = \overset{\frown}{BD} + \overset{\frown}{DA}$。
			\par\textbf{理由：} 直径将圆周分为两个相等的半圆。
		\item[\textbf{第四步（等量减等量）：}] 由$\overset{\frown}{CB} = \overset{\frown}{DA}$，两边同时减去，得$\overset{\frown}{AC} = \overset{\frown}{BD}$。
			\par\textbf{理由：} 等量减等量差相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\overset{\frown}{AC} = \overset{\frown}{BD}$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，在圆$O$中，点$A,B,C,D$在圆上，且$\angle ABC = \angle DCB$。求证：$\angle BAD = \angle CDA$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等角推等弧）：}] $\angle ABC$所对的弧是$\overset{\frown}{AC}$，$\angle DCB$所对的弧是$\overset{\frown}{DB}$。由$\angle ABC = \angle DCB$及等圆周角对等弧，得$\overset{\frown}{AC} = \overset{\frown}{DB}$。
			\par\textbf{理由：} 等圆周角所对的弧相等。
		\item[\textbf{第二步（等弧推等角）：}] $\angle BAD$所对的弧是$\overset{\frown}{BD}$，$\angle CDA$所对的弧是$\overset{\frown}{CA}$。因为$\overset{\frown}{AC} = \overset{\frown}{DB}$即$\overset{\frown}{CA} = \overset{\frown}{BD}$，由等弧所对的圆周角相等，得$\angle BAD = \angle CDA$。
			\par\textbf{理由：} 等弧对等角。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle BAD = \angle CDA$。
