\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{在同圆或等圆中，等弧与等圆周角可以互相推出。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（同圆或等圆）：} 两个圆必须半径相同（等圆）或为同一个圆。
		\item \textbf{条件 2（同侧对应）：} 圆周角的顶点应位于所对弧的同一侧，以确保所对的弧是同一段弧。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（同圆前提）：} 不可忽略“同圆或等圆”前提，否则等角不一定推出等弧。
		\item \textbf{检查项（弧角对应）：} 每一步推理前，确认所找的弧与圆周角是否匹配（角两边所夹弧）。
	\end{itemize}
\end{summarybox}
