\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆切线判定）}}]
			误认为只要直线与半径垂直，该直线就是圆的切线.
		\item[\textbf{\textcolor{CTrap}{正确理解（垂直且过端点）：}}]
			直线必须垂直于半径，且垂足为该半径的外端点（即点在圆上），才是切线.
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义）：}}]
			垂直是切线性质定理的逆定理的条件之一，但缺少“点在圆上”不构成充要条件.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗漏竖直切线）}}]
			在求切线方程时，设点斜式 $y-y_0=k(x-x_0)$，容易忽略斜率不存在的情况.
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论）：}}]
			应先考虑切线竖直（$x=x_0$）是否满足条件，再设点斜式.
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定式）：}}]
			大多数直线方程均设为点斜式，未检查特殊情形，导致漏解.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（误用结论）}}]
			从圆外一点求切线时，仍然用“垂直半径”的条件，但此时切点未知，无法直接使用.
		\item[\textbf{\textcolor{CTrap}{正确理解（距离公式法）：}}]
			圆外一点求切线应设直线方程，利用圆心到直线的距离等于半径求解，而非依赖垂直关系.
		\item[\textbf{\textcolor{CTrap}{错因分析（滥用结论）：}}]
			切线性质定理适用于已知切点的情况，对于切点未知的列方程问题，应使用代数判别式或距离公式.
	\end{description}
\end{trapbox}
