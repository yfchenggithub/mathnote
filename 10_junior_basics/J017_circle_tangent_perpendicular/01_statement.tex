	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（切线与切点）：}}]
			直线 $l$ 与圆 $\odot O$ 相切于一点 $P$，$O$ 为圆心，$r$ 为半径.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（半径垂直切线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆的切线与过切点的半径互相垂直.
			\[
				OP \perp l.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（切线判定定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过圆上一点且垂直于该点半径的直线是圆的切线.
			\[
				P \in \odot O,\ l \perp OP \Longrightarrow l \cap \odot O = \{P\}.
			\]
	\end{description}
\end{statementbox}
