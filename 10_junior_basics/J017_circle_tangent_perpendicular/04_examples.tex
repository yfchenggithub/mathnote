\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（已知切点）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $x^2+y^2=5$，点 $P(1,2)$ 在圆上，求过点 $P$ 的切线方程.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求圆心半径）：}]
			圆心 $O(0,0)$，半径 $r=\sqrt5$.
			\par\textbf{理由：} 标准圆方程直接读出.
		\item[\textbf{第二步（求斜率）：}]
			$k_{OP}=\frac{2-0}{1-0}$，故 $k_{OP}=2$，由 $OP\perp l$ 得切线斜率 $k_l=-\frac12$.
			\par\textbf{理由：} 垂直直线斜率乘积为 $-1$.
		\item[\textbf{第三步（写方程）：}]
			点斜式 $y-2=-\frac12(x-1)$，整理得 $x+2y-5=0$.
			\par\textbf{理由：} 代入切点和斜率.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线方程为 $x+2y-5=0$}.

	\medskip
	\textbf{\textcolor{CExample}{例2（点斜式求切线）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $(x+1)^2+(y-3)^2=13$，直线 $l$ 经过点 $A(1,0)$ 且与圆相切，求 $l$ 的方程.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（位置判断）：}]
			将 $A(1,0)$ 代入圆方程得 $(1+1)^2+(0-3)^2=13$，故 $A$ 在圆上，是切点.
			\par\textbf{理由：} 点坐标满足方程即点在圆上.
		\item[\textbf{第二步（求斜率）：}]
			圆心 $C(-1,3)$，$k_{CA}=\frac{0-3}{1-(-1)}$，得 $k_{CA}=-\frac32$，故 $k_l=\frac23$.
			\par\textbf{理由：} $k_{CA}\cdot k_l=-1$.
		\item[\textbf{第三步（写方程）：}]
			点斜式 $y-0=\frac23(x-1)$，整理得 $2x-3y-2=0$.
			\par\textbf{理由：} 过 $A$ 且斜率已知.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线方程为 $2x-3y-2=0$}.

	\medskip
	\textbf{\textcolor{CExample}{例3（垂线求切点）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $x^2+y^2=2$ 与直线 $l: x+y-2=0$ 相切于点 $P$，求切点 $P$ 的坐标.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求圆心半径）：}]
			圆心 $O(0,0)$，半径 $r=\sqrt2$.
			\par\textbf{理由：} 标准方程.
		\item[\textbf{第二步（求垂线方程）：}]
			$l$ 斜率 $-1$，由 $OP\perp l$ 得 $OP$ 斜率 $1$，故 $OP$ 方程为 $y=x$.
			\par\textbf{理由：} 垂直关系.
		\item[\textbf{第三步（联立求切点）：}]
			解 $
			\begin{cases}
				y=x,\\x^2+y^2=2
		\end{cases}
			$ 得 $x=\pm1$，相应 $y=\pm1$，代入 $l$ 知 $(1,1)$ 满足 $x+y-2=0$，故 $P(1,1)$.
			\par\textbf{理由：} 切点同时在圆上和直线上.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切点 $P(1,1)$}.
\end{examplebox}
