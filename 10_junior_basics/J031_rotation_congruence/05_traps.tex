\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（对应点混淆）}}]
			在旋转全等中，误认为点 $A$ 对应点 $B'$，导致对应边、对应角找错。
		\item[\textbf{\textcolor{CTrap}{正确理解（顺序对应）：}}]
			旋转后，原图形中的每个点按照旋转顺序映射到唯一对应点，应保持顺序一致（如 $A\to A'$, $B\to B'$, $C\to C'$）。
		\item[\textbf{\textcolor{CTrap}{错因分析（对应意识弱）：}}]
			认为旋转只是整体移动，未注意点与点的一一对应关系。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（旋转角误判）}}]
			把 $\angle AOB$ 当作旋转角，而实际上旋转角应是 $\angle AOA'$。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义明确）：}}]
			旋转角是同一对应点旋转前后与旋转中心连线所夹的角。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）：}}]
			未区分旋转角与三角形内角。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略旋转中心）}}]
			在证明全等时，直接认为任意变换（如平移、旋转）都全等，忽略旋转中心必须固定。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件必备）：}}]
			本结论的等距等角性质依赖于固定的旋转中心；平移、反射虽也全等，但不涉及旋转中心，不能直接套用此结论。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件忽略）：}}]
			未注意旋转定义中包含旋转中心和旋转角两个要素。
	\end{description}

\end{trapbox}
