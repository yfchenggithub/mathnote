\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 已知 $\triangle ABC$ 绕点 $O$ 旋转 $60^\circ$ 得到 $\triangle A'B'C'$，且 $AB=5$，$BC=6$，$CA=7$。求 $A'B'$，$B'C'$，$C'A'$ 的长度。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别全等）：}]
			由旋转保距保角知 $\triangle ABC \cong \triangle A'B'C'$。
			\par\textbf{理由：} 旋转 $\theta$ 角得到的图形与原图形全等。
		\item[\textbf{第二步（对应边相等）：}]
			全等三角形对应边相等，所以 $A'B' = AB$，$B'C' = BC$，$C'A' = CA$。
			\par\textbf{理由：} 三角形全等性质。
		\item[\textbf{第三步（代入数值）：}]
			因为 $AB = 5$，$BC = 6$，$CA = 7$，所以 $A'B' = 5$，$B'C' = 6$，$C'A' = 7$。
			\par\textbf{理由：} 等量替换。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{旋转全等直接推出对应边相等。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 如图，正方形 $ABCD$ 中，$E$ 是 $CD$ 上一点，将 $\triangle ADE$ 绕点 $A$ 顺时针旋转 $90^\circ$ 得到 $\triangle ABF$。求证：$DE=BF$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别旋转全等）：}]
			绕点 $A$ 旋转 $90^\circ$，$\triangle ADE \cong \triangle ABF$。
			\par\textbf{理由：} 旋转保距保角，故两三角形全等。
		\item[\textbf{第二步（确定对应边）：}]
			在全等中，$A$ 对应 $A$，$D$ 对应 $B$，$E$ 对应 $F$，故对应边 $DE$ 与 $BF$ 相等。
			\par\textbf{理由：} 全等三角形对应边相等。
		\item[\textbf{第三步（得证）：}]
			所以 $DE = BF$。
			\par\textbf{理由：} 由第二步直接得到。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{通过旋转构造全等，可直接证明线段相等。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 如图，$P$ 是等边 $\triangle ABC$ 内一点，将 $\triangle ABP$ 绕点 $B$ 逆时针旋转 $60^\circ$ 得到 $\triangle CBP'$。若 $BP=4$，求 $\triangle BPP'$ 的周长。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别全等与旋转角）：}]
			旋转得 $\triangle ABP \cong \triangle CBP'$，且旋转角 $\angle PBP' = 60^\circ$（$BP$ 旋转到 $BP'$）。
			\par\textbf{理由：} 旋转保距保角，对应点与旋转中心连线夹角等于旋转角。
		\item[\textbf{第二步（得等边三角形）：}]
			由全等得 $BP = BP'$；又 $BP = 4$，所以 $BP' = 4$。在 $\triangle BPP'$ 中，$BP = BP'$ 且 $\angle PBP' = 60^\circ$，故 $\triangle BPP'$ 是等边三角形，从而 $PP' = 4$。
			\par\textbf{理由：} 等边三角形判定。
		\item[\textbf{第三步（求周长）：}]
			计算得 $C_{\triangle BPP'} = BP + BP' + PP'$；代入 $BP = 4$，$BP' = 4$，$PP' = 4$，因此 $C_{\triangle BPP'} = 12$。
			\par\textbf{理由：} 等边三角形三边相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{旋转全等结合旋转角性质，可推出等边三角形，进而求周长。}
