\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（旋转中心与角度）：}}]
			平面内任一图形绕定点 $O$ 旋转角 $\theta$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（全等性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			旋转前后图形形状大小完全一样。
			\[
				\triangle ABC \cong \triangle A'B'C'.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（距等性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每个点到旋转中心的距离保持不变。
			\[
				\begin{aligned}
					OA &= OA', \\ OB &= OB', \\ OC &= OC'.
			\end{aligned}
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（角等性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每个点绕 $O$ 转过的角度相同。
			\[
				\begin{aligned}
					\angle AOA' &= \theta, \\ \angle BOB' &= \theta, \\ \angle COC' &= \theta.
			\end{aligned}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（对应边相等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			原三角形的三边与旋转后的三边分别相等。
			\[
				\begin{aligned}
					AB &= A'B', \\ BC &= B'C', \\ CA &= C'A'.
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
