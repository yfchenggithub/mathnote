\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略定义域）}}]
			只计算顶点最值，不考虑动点横坐标的实际范围。
		\item[\textbf{\textcolor{CTrap}{正确理解（区间优先）}}]
			必须先检查顶点横坐标是否在给定区间 $[m,n]$ 内，若不在则最值只能在端点取得。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）}}]
			学生习惯直接使用顶点公式，忘记了二次函数是定义在区间上的．
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（符号错误）}}]
			计算对称轴 $x_0=-\frac{B}{2A}$ 时丢掉负号，或 $A$、$B$ 代入符号出错。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式记忆）}}]
			牢记对称轴公式并仔细计算，尤其当 $B$ 为负值时注意符号处理。
		\item[\textbf{\textcolor{CTrap}{错因分析（粗心运算）}}]
			对公式记忆不牢固，代入数值时不注意符号变化．
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（单调误判）}}]
			当对称轴不在区间内时，不能正确判断最值在哪个端点。
		\item[\textbf{\textcolor{CTrap}{正确理解（画图辅助）}}]
			画二次函数示意图，结合抛物线开口方向和对称轴位置确定端点处的函数值大小关系。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏图像）}}]
			不画图像仅凭推理，容易弄反单调方向．
	\end{description}

\end{trapbox}
