\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知二次函数 $S(x)=x^{2}-4x+5$，$x\in[0,3]$，求 $S(x)$ 的最大值和最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			确定二次项系数 $A=1>0$，区间 $[m,n]=[0,3]$。
			\par\textbf{理由：} $A>0$ 开口向上，最小值在顶点，最大值在端点。
		\item[\textbf{第二步（顶点判定）：}]
			计算对称轴：$x_{0}=-\frac{B}{2A}=2$，因为 $2\in[0,3]$，所以最小值在顶点。
			\[
				S_{\min}=S(2)=2^{2}-4\cdot2+5=1.
			\]
			\par\textbf{理由：} 顶点公式给出最小值。
		\item[\textbf{第三步（端点比较）：}]
			计算端点值：$S(0)=5$，$S(3)=3^{2}-12+5=2$，比较得最大值 $\max\{S(0),S(3)\}=5$。
			\[
				S_{\max}=5.
			\]
			\par\textbf{理由：} 开口向上时最大值出现在离对称轴较远的端点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最小值 $1$，最大值 $5$。

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知二次函数 $S(x)=-2x^{2}+12x-10$，$x\in[1,4]$，求 $S(x)$ 的最大值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			确定 $A=-2<0$，区间 $[1,4]$。
			\par\textbf{理由：} $A<0$ 开口向下，最大值在顶点。
		\item[\textbf{第二步（顶点判定）：}]
			对称轴：$x_{0}=-\frac{B}{2A}=3$，$3\in[1,4]$，故最大值在 $x_{0}$ 处。
			\[
				S_{\max}=S(3)=-2\cdot9+36-10=8.
			\]
			\par\textbf{理由：} 顶点公式 $S(x_{0})=\frac{4AC-B^{2}}{4A}$ 直接计算得 $8$。
		\item[\textbf{第三步（验证端点）：}]
			计算端点值：$S(1)=0$，$S(4)=6$，均小于 $8$，验证最大值正确。
			\[
				S(1)=0,\quad S(4)=6.
			\]
			\par\textbf{理由：} 端点值作为候选，但顶点在区间内时最值在顶点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最大值 $8$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知二次函数 $S(x)=x^{2}+2x+3$，$x\in[0,2]$，求 $S(x)$ 的最小值和最大值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			确定 $A=1>0$，区间 $[0,2]$。
			\par\textbf{理由：} $A>0$ 开口向上，最值可能受端点影响。
		\item[\textbf{第二步（对称轴判定）：}]
			对称轴：$x_{0}=-\frac{B}{2A}=-1$，$-1\notin[0,2]$，顶点不在区间内。
			\par\textbf{理由：} 对称轴在区间左侧，函数在 $[0,2]$ 上单调递增。
		\item[\textbf{第三步（端点最值）：}]
			最小值在左端点 $x=0$，最大值在右端点 $x=2$。
			\[
				S(0)=3,
				\qquad
				S(2)=4+4+3=11.
			\]
			\par\textbf{理由：} 开口向上且对称轴在左侧，函数单调递增，故最小值在左端点，最大值在右端点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最小值 $3$，最大值 $11$。
\end{examplebox}
