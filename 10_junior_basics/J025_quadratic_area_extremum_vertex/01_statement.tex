\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（二次函数形式）：}}]
			设动点横坐标为 $x$，目标面积 $S(x)=Ax^{2}+Bx+C$，其中 $A\neq 0$，且 $x$ 的取值范围为闭区间 $[m,n]$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（顶点公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积最值可由顶点坐标公式直接求得。
			\[
				x_{0}=-\frac{B}{2A},
				\qquad
				y_{0}=\frac{4AC-B^{2}}{4A}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（区间端点判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			最值可能在顶点取，也可能在区间端点取，取决于对称轴是否在区间内。
			\begin{itemize}
			\item 若 $A>0$ 且 $x_{0}\in[m,n]$，则 $S(x)$ 在 $x_{0}$ 处取最小值；
			\item 若 $A<0$ 且 $x_{0}\in[m,n]$，则 $S(x)$ 在 $x_{0}$ 处取最大值；
			\item 若 $x_{0}\notin[m,n]$，则最值在端点 $x=m$ 或 $x=n$ 处取得。
		\end{itemize}

	\end{description}
	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	当 $x=x_{0}$（且 $x_{0}\in[m,n]$）或 $x=m$、$x=n$ 时取得最值。
\end{statementbox}
