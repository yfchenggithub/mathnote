\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用非负性 $a_i\ge0$ 和和为零的条件，通过不等式放缩得到 $0\le a_j\le0$，从而推出 $a_j=0$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（转化条件）：}]
		      已知 $a_i\ge0$（$i=1,2,\dots,n$）且
		      \[
			      \sum_{i=1}^n a_i=0.
		      \]
		      \par\textbf{理由：} 这是题目条件的代数表达。

		\item[\textbf{步骤二（建立下界）：}]
		      对任意 $j\in\{1,2,\dots,n\}$，由非负性有 $a_j\ge0$。
		      \par\textbf{理由：} 每一项都是非负数。

		\item[\textbf{步骤三（建立上界）：}]
		      由于其余各项也都非负，所以
		      \[
			      a_j\le a_1+a_2+\dots+a_n=\sum_{i=1}^n a_i=0.
		      \]
		      \par\textbf{理由：} 非负数的总和不小于其中任意一项。

		\item[\textbf{步骤四（夹逼为零）：}]
		      结合 $a_j\ge0$ 和 $a_j\le0$，得 $a_j=0$。
		      \par\textbf{理由：} 一个实数同时不小于 $0$ 且不大于 $0$，只能等于 $0$。

		\item[\textbf{步骤五（逐项推广）：}]
		      因为 $j$ 可以取 $1,2,\dots,n$ 中的任意值，所以
		      \[
			      a_1=a_2=\dots=a_n=0.
		      \]
		      \par\textbf{理由：} 任意一项都能按同样方法推出为零。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，有限个非负数之和为零时，每一项都必须为零。}
\end{proofbox}
