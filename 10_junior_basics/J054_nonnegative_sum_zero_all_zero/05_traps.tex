\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视非负前提）}}]
		      学生盲目认为只要和为零，每个加数必为零，而忽略这些加数必须都是非负数。
		\item[\textbf{\textcolor{CTrap}{正确理解（检查每项非负）：}}]
		      使用结论前必须确认每一项都是非负数。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件误用）：}}]
		      将结论的条件“非负数”丢失，导致在负数或任意实数上错误推广。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗漏部分项）}}]
		      在多个非负项和为零的方程中，只解出部分项为零，遗漏其他项。
		\item[\textbf{\textcolor{CTrap}{正确理解（逐项为零）：}}]
		      必须令所有非负项同时为零，得到完整方程组。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维疏漏）：}}]
		      只关注了明显项，忽略其他项，造成漏解。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（定义域忽略）}}]
		      在根式等非负项中，只考虑非负性和和为零，忽略表达式自身的定义域，导致无效解。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义域先行）：}}]
		      求解前先确定各项的定义域，然后令每项为零，解得的参数必须在定义域内。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件不全）：}}]
		      非负性只是其中一重限制，表达式的定义域也必须满足。
	\end{description}
\end{trapbox}
