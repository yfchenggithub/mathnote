\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（非负项有限）：}}]
		      设 $a_1,a_2,\dots,a_n$ 是有限个非负数，即
		      \[
			      a_i\ge 0\quad (i=1,2,\dots,n).
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（每项必为零）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 若有限个非负数的总和为零，则每个加数都必须为零。
		      \[
			      a_1+a_2+\dots+a_n=0
			      \quad\Rightarrow\quad
			      a_1=a_2=\dots=a_n=0.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（常见非负形式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 绝对值、平方、算术平方根等非负项之和为零时，也可以逐项令零。
		      \[
			      |x_1|+|x_2|+\dots+|x_n|=0
			      \quad\Rightarrow\quad
			      x_1=x_2=\dots=x_n=0.
		      \]
		      类似地，若 $\sum_{i=1}^n x_i^2=0$，则 $x_1=x_2=\dots=x_n=0$；若 $y_i\ge0$ 且 $\sum_{i=1}^n \sqrt{y_i}=0$，则 $y_1=y_2=\dots=y_n=0$。
	\end{description}
\end{statementbox}
