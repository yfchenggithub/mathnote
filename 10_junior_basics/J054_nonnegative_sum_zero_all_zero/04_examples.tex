\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $|x-2|+|y+1|=0$，求实数 $x,y$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别非负项）：}]
		      $|x-2|$ 和 $|y+1|$ 均为非负数。
		      \par\textbf{理由：} 绝对值非负。
		\item[\textbf{第二步（套用结论）：}]
		      由非负数之和为零，得 $|x-2|=0$ 且 $|y+1|=0$。
		      \par\textbf{理由：} 若有限个非负数之和为零，则每个为零。
		\item[\textbf{第三步（求解）：}]
		      由 $|x-2|=0$ 得 $x=2$；由 $|y+1|=0$ 得 $y=-1$。
		      \par\textbf{理由：} 绝对值等于零，则内部表达式为零。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$(x,y)=(2,-1)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $(2x-1)^2+(y+3)^2=0$，求实数 $x,y$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别非负项）：}]
		      $(2x-1)^2$ 和 $(y+3)^2$ 均为非负数。
		      \par\textbf{理由：} 平方非负。
		\item[\textbf{第二步（套用结论）：}]
		      由平方和为零，得 $(2x-1)^2=0$ 且 $(y+3)^2=0$。
		      \par\textbf{理由：} 非负数之和为零则每个为零。
		\item[\textbf{第三步（求解）：}]
		      由 $(2x-1)^2=0$ 得 $2x-1=0$，即 $x=\dfrac12$；由 $(y+3)^2=0$ 得 $y=-3$。
		      \par\textbf{理由：} 平方为零则底数为零。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$x=\dfrac12,\,y=-3$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $\sqrt{x-1}+\sqrt{y+2}=0$，求实数 $x,y$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别非负项）：}]
		      $\sqrt{x-1}$ 和 $\sqrt{y+2}$ 均为非负数，且需满足 $x\ge1$，$y\ge-2$。
		      \par\textbf{理由：} 算术平方根非负，且被开方数非负。
		\item[\textbf{第二步（套用结论）：}]
		      由非负数之和为零，得 $\sqrt{x-1}=0$ 且 $\sqrt{y+2}=0$。
		      \par\textbf{理由：} 有限个非负数之和为零则每个为零。
		\item[\textbf{第三步（求解验证）：}]
		      由 $\sqrt{x-1}=0$ 得 $x=1$；由 $\sqrt{y+2}=0$ 得 $y=-2$。经验证，$x=1\ge1$，$y=-2\ge-2$，满足定义域。
		      \par\textbf{理由：} 根号为零则被开方数为零。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$x=1,\,y=-2$。}
\end{examplebox}
