\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{有限个非负数之和为零，则每个非负数必为零。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（各项非负）：} 每一项都是非负数，即 $a_i\ge0$。
		\item \textbf{条件 2（和为零）：} 所有项的和等于零，即 $\sum_{i=1}^n a_i=0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略非负性）：} 应用结论前必须确认每一项确实是非负数。
		\item \textbf{检查项（定义域相容）：} 对于根式等非负表达式，还需检查定义域及各项同时为零是否可能。
	\end{itemize}
\end{summarybox}
