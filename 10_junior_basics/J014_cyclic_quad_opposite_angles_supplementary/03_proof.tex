\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用圆周角定理，将对角之和转化为所对弧的度数之和，再由整个圆周为 $360^\circ$ 得出互补.

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（确定对弧）：}]
			在圆 $O$ 中，四边形 $ABCD$ 内接，顶点按顺序 $A,B,C,D$.
			$\angle A$ 所对的弧是 $\widehat{BCD}$，$\angle C$ 所对的弧是 $\widehat{BAD}$.
			\[
				\angle A \longleftrightarrow \widehat{BCD},
				\qquad
				\angle C \longleftrightarrow \widehat{BAD}.
			\]
			\par\textbf{理由：}
			圆周角由其两边与圆的交点确定所对弧.

		\item[\textbf{步骤二（弧长和）：}]
			$\widehat{BCD}$ 和 $\widehat{BAD}$ 恰好构成整个圆周.
			\[
				\widehat{BCD} + \widehat{BAD} = 360^\circ.
			\]
			\par\textbf{理由：}
			整个圆周的度数为 $360^\circ$.

		\item[\textbf{步骤三（圆周角公式）：}]
			圆周角等于所对弧的度数的一半.
			\[
				\angle A = \frac{1}{2}\widehat{BCD},
				\qquad
				\angle C = \frac{1}{2}\widehat{BAD}.
			\]
			\par\textbf{理由：}
			圆周角定理.

		\item[\textbf{步骤四（求和得互补）：}]
			相加两式：
			\[
				\begin{aligned}
					\angle A + \angle C &= \frac{1}{2}(\widehat{BCD}+\widehat{BAD}) \\
					&= \frac{1}{2}\times 360^\circ \\
					&= 180^\circ.
			\end{aligned}
			\]
			\par\textbf{理由：}
			代入运算；同理可得 $\angle B + \angle D = 180^\circ$.
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	因此，圆内接四边形对角互补：$\angle A+\angle C=180^\circ$，$\angle B+\angle D=180^\circ$.
\end{proofbox}
