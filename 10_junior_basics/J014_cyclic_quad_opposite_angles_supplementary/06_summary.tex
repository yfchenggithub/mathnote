\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆内接四边形对角互补，即 $\angle A+\angle C=180^\circ$，$\angle B+\angle D=180^\circ$.}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（共圆前提）：} 四边形 $ABCD$ 的四个顶点在同一圆周上.
		\item \textbf{条件2（对角识别）：} 互补角必须是 $\angle A$ 与 $\angle C$、$\angle B$ 与 $\angle D$ 这两组对角.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（邻角误区）：} 不要误认为邻角也互补.
		\item \textbf{检查项（逆用条件）：} 若已知一组对角互补，则可判定四点共圆，但需确认是对角.
	\end{itemize}
\end{summarybox}
