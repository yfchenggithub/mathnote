\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{一组平行线（至少三条）截两条直线，所截得的对应线段成比例。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（三线平行）：} 至少有三条不重合的平行线。
		\item \textbf{条件2（两截线）：} 两条直线与这组平行线都相交。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（对应顺序）：} 严格按照平行线顺序确定对应线段，不可错位。
		\item \textbf{检查项（平行线数）：} 检查平行线是否至少三条，不足时结论不直接成立。
	\end{itemize}
\end{summarybox}
