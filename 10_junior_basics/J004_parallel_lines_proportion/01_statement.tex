\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（平行线截两线）：}}] 至少三条平行线（如 $l_1\parallel l_2\parallel l_3$）分别与两条直线 $a$、$b$ 相交，交点按顺序为 $A,B,C$ 和 $D,E,F$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对应成比例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三条平行线截两条直线，所截线段对应成比例。
			\[
				\frac{AB}{BC}=\frac{DE}{EF},
				\qquad
				\frac{AB}{AC}=\frac{DE}{DF}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（三角形截线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形中，平行于一边的直线截其他两边，所得线段对应成比例。
			\[
				\frac{AD}{DB}=\frac{AE}{EC},
				\qquad
				\frac{AD}{AB}=\frac{AE}{AC}.
			\]
	\end{description}
\end{statementbox}
