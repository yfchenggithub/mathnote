\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{点关于直线对称，对称轴是该两点连线的垂直平分线。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件一（正向应用）} 已知两点关于某直线对称，可推出对称轴垂直平分连线。
		\item \textbf{条件二（逆向应用）} 若一条直线垂直平分一条线段，则两端点关于该直线对称。
		\item \textbf{条件三（坐标形式）} 在解析几何中，可用中点公式和斜率关系列方程。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（缺一不可）} 垂直和平分两个条件必须同时满足，不能只用一个。
		\item \textbf{检查项（斜率讨论）} 使用斜率关系时，注意斜率不存在或为零的特殊情况。
	\end{itemize}
\end{summarybox}
