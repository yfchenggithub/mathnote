	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（点线对称）}}] 平面内点 $P$ 与点 $P'$ 关于直线 $l$ 对称（沿 $l$ 折叠重合）。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中垂线性质）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对称轴垂直平分对称点的连线。记 $M$ 为 $PP'$ 的中点。
			\[
				l \perp PP' \ \land\ M \in l.
			\]

		\item[\textbf{\textcolor{CStatement}{逆命题（等价条件）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若一条直线同时垂直且平分一条线段，则该直线是对称轴。
			\[
				l \perp PP' \ \land\ M \in l.
			\]
	\end{description}
\end{statementbox}
