\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{通过作对称点将同侧折线最短转化为两点间线段最短。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（定点同侧）：} 两定点在直线同侧，动点在直线上。
		\item \textbf{条件2（对称可行）：} 可准确作出一个点的对称点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（同侧对称）：} 必须作对称，不能直接连。
		\item \textbf{检查项（交点验证）：} 确认$P$是$A'B$与$l$的交点。
	\end{itemize}
\end{summarybox}
