\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定点同侧）：}}] 点$A,B$在直线$l$同侧，$P$是$l$上动点。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对称转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			作$A$关于$l$的对称点$A'$，连接$A'B$交$l$于$P$，则$PA+PB$最小。
			\[
				(PA+PB)_{\min}=A'B.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（异侧情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若$A,B$在$l$异侧，直接连接$AB$交$l$于$P$，则$PA+PB$最小。
			\[
				(PA+PB)_{\min}=AB.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	$P$是$A'B$与$l$的交点。
\end{statementbox}
