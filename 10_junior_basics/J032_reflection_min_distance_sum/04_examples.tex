\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线$l:y=0$（$x$轴），点$A(1,2)$，$B(3,1)$，在$l$上求一点$P$，使得$PA+PB$最小，并求最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（对称变换）：}] 作$A$关于$x$轴的对称点$A'(1,-2)$。
			\par\textbf{理由：} 关于$x$轴对称，横坐标不变，纵坐标相反。
		\item[\textbf{第二步（连线求交）：}] 直线$A'B$的斜率$k=\frac{1-(-2)}{3-1}=\frac{3}{2}$，方程为$y+2=\frac{3}{2}(x-1)$。令$y=0$，解得$x=\frac{7}{3}$，故$P\left(\frac{7}{3},0\right)$。
			\par\textbf{理由：} $P$是$A'B$与$l$的交点。
		\item[\textbf{第三步（计算最值）：}] $A'B=\sqrt{(3-1)^2+(1+2)^2}=\sqrt{4+9}=\sqrt{13}$，所以$(PA+PB)_{\min}=\sqrt{13}$。
			\par\textbf{理由：} 两点间距离公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P\left(\frac{7}{3},0\right)$，最小值$\sqrt{13}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线$l:y=x$，点$A(1,2)$，$B(3,4)$都在$l$同侧（$y>x$），求$l$上点$P$使$PA+PB$最小，并求最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（对称变换）：}] 作$A$关于$y=x$的对称点$A'(2,1)$。
			\par\textbf{理由：} 关于$y=x$对称，交换横纵坐标。
		\item[\textbf{第二步（连线求交）：}] 直线$A'B$的斜率$k=\frac{4-1}{3-2}=3$，方程为$y-1=3(x-2)$，即$y=3x-5$。与$y=x$联立解得$x=2.5$，$y=2.5$，故$P(2.5,2.5)$。
			\par\textbf{理由：} $P$是直线交点。
		\item[\textbf{第三步（计算最值）：}] $A'B=\sqrt{(3-2)^2+(4-1)^2}=\sqrt{1+9}=\sqrt{10}$，所以$(PA+PB)_{\min}=\sqrt{10}$。
			\par\textbf{理由：} 两点间距离公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P(2.5,2.5)$，最小值$\sqrt{10}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线$l:y=x+1$，点$A(0,0)$，$B(2,0)$在$l$同侧（下方），求$l$上点$P$使$PA+PB$最小，并求最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求对称点）：}] 设$A$关于$l$的对称点$A'(x,y)$。由$AA'\perp l$得$\frac{y-0}{x-0}=-1$，即$y=-x$；由中点在$l$上得$\frac{y}{2}=\frac{x}{2}+1$，即$y=x+2$。联立解得$x=-1$，$y=1$，故$A'(-1,1)$。
			\par\textbf{理由：} 对称点满足垂直且中点在轴上。
		\item[\textbf{第二步（连线求交）：}] 直线$A'B$的斜率$k=\frac{0-1}{2-(-1)}=-\frac{1}{3}$，方程为$y-1=-\frac{1}{3}(x+1)$，即$y=-\frac{1}{3}x+\frac{2}{3}$。与$l$联立得$-\frac{1}{3}x+\frac{2}{3}=x+1$，解得$x=-\frac{1}{4}$，$y=\frac{3}{4}$，故$P\left(-\frac{1}{4},\frac{3}{4}\right)$。
			\par\textbf{理由：} $P$是两直线交点。
		\item[\textbf{第三步（计算最值）：}] $A'B=\sqrt{(2+1)^2+(0-1)^2}=\sqrt{9+1}=\sqrt{10}$，所以$(PA+PB)_{\min}=\sqrt{10}$。
			\par\textbf{理由：} 两点间距离公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P\left(-\frac{1}{4},\frac{3}{4}\right)$，最小值$\sqrt{10}$。}
\end{examplebox}
