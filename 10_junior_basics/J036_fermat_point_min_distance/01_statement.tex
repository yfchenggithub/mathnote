\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1(全锐角三角形):}}]
			三角形三个内角均小于 $120^\circ$.
		\item[\textbf{\textcolor{CStatement}{条件2(含钝角三角形):}}]
			三角形有一个内角(设为 $\angle A$)大于或等于 $120^\circ$.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一(费马点存在性):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			费马点位于三角形内部,与三顶点连线两两成 $120^\circ$ 角.
			\[
				\angle APB = \angle BPC = \angle CPA = 120^\circ,
			\]
			且 $PA+PB+PC$ 取得最小值.
		\item[\textbf{\textcolor{CStatement}{结论二(费马点退化情形):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			若三角形某内角不小于 $120^\circ$,该角顶点即为费马点,距离和等于该角两边长度之和.
			\[
				PA+PB+PC = AB+AC
			\]
			(此时 $\angle A \ge 120^\circ$).
		\item[\textbf{\textcolor{CStatement}{常见变形(旋转构造性质):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			向外作等边三角形,其外顶点与对顶点的连线长度即为最小距离和.
			\[
				CD = PA+PB+PC
			\]
			(其中 $\triangle ABD$ 为等边三角形).
	\end{description}

\end{statementbox}
