\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{费马点是到三角形三顶点距离之和最小的点:当所有内角小于$120^\circ$时,在形内且三连线夹角均为$120^\circ$;当有一角不小于$120^\circ$时,该角顶点即为费马点,最小值为该角两边长度之和.}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1(全锐角三角形):} 三角形三个内角均小于 $120^\circ$.
		\item \textbf{条件2(含钝角三角形):} 三角形有一个内角大于或等于 $120^\circ$.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点(忽略角度条件):} 使用费马点前必须确认三角形内角是否都小于 $120^\circ$,不能盲目套用内部点结论.
		\item \textbf{检查项(公式选取):} 全锐角时利用旋转构造求 $BC'$(最小值);退化时直接使用 $AB+AC$,无需旋转.
	\end{itemize}

\end{summarybox}
