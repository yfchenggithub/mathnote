\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{直径所对的圆周角是直角，直角圆周角所对的弦是直径。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（直径前提）：} 圆中有一条弦是直径。
		\item \textbf{条件2（顶点位置）：} 圆周角的顶点在圆上（不与直径端点重合）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（误用条件）：} 任意弦不保证直角，只有直径才行。
		\item \textbf{检查项（先证后算）：} 使用直角三角形性质前务必先确认 $\angle ACB=90^\circ$。
	\end{itemize}
\end{summarybox}
