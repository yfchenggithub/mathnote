\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$AB$ 是 $\odot O$ 的直径，点 $C$ 在 $\odot O$ 上，$\angle CAB = 40^\circ$，求 $\angle CBA$ 的度数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] $AB$ 是 $\odot O$ 的直径，$C$ 在 $\odot O$ 上。
			\par\textbf{理由：} 已知条件。
		\item[\textbf{第二步（应用定理）：}] 由直径所对圆周角是直角，得
			\[
				\angle ACB = 90^\circ.
			\]
			\par\textbf{理由：} 定理直接应用。
		\item[\textbf{第三步（角度运算）：}] 在 $\triangle ACB$ 中，$\angle CAB+\angle CBA = 90^\circ$，
			\[
				\angle CBA = 90^\circ - 40^\circ = 50^\circ.
			\]
			\par\textbf{理由：} 直角三角形两锐角互余。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle CBA = 50^\circ$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$\triangle ABC$ 的三个顶点都在 $\odot O$ 上，且 $\angle A = 90^\circ$，求证：$BC$ 是 $\odot O$ 的直径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] $A,B,C$ 在 $\odot O$ 上，$\angle A = 90^\circ$。
			\par\textbf{理由：} 已知。
		\item[\textbf{第二步（逆定理推理）：}] 由于 $\angle A$ 是 $90^\circ$ 的圆周角，它所对的弦 $BC$ 必为直径。
			\[
				\angle A = 90^\circ \implies O\in BC.
			\]
			\par\textbf{理由：} 直径所对圆周角是直角的逆定理。
		\item[\textbf{第三步（结论）：}] 所以 $BC$ 是 $\odot O$ 的直径。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$BC$ 是 $\odot O$ 的直径。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$AB$ 是 $\odot O$ 的直径，$C$ 是 $\odot O$ 上一点，$CD \perp AB$ 于 $D$。求证：$CD^2 = AD \cdot DB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用直径定理）：}] 由 $AB$ 是直径，得
			\[
				\angle ACB = 90^\circ.
			\]
			\par\textbf{理由：} 直径所对的圆周角是直角。
		\item[\textbf{第二步（直角三角形高）：}] 在 $\triangle ACB$ 中，$\angle ACB = 90^\circ$，$CD \perp AB$，所以 $CD$ 是斜边上的高。
			\par\textbf{理由：} 直角三角形斜边高定义。
		\item[\textbf{第三步（射影定理）：}] 由射影定理，直角三角形斜边上的高满足
			\[
				CD^2 = AD \cdot DB.
			\]
			\par\textbf{理由：} 射影定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$CD^2 = AD \cdot DB$。
\end{examplebox}
