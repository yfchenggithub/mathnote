\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（过重心都平分）：}}]
			认为三角形中过重心的直线一定平分面积。
		\item[\textbf{\textcolor{CTrap}{正确理解（过重心不一定平分）：}}]
			过重心的直线不一定平分面积（例如过重心平行于底边的直线）；中线虽过重心且平分，但不能因此认为所有过重心的直线都平分。
		\item[\textbf{\textcolor{CTrap}{错因分析（重心非中心）：}}]
			将重心误认为三角形的对称中心，误以为过中心直线都平分。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（误认仅对角线）：}}]
			在平行四边形或圆中，只认为两条对角线平分面积，忽略其他过中心的直线。
		\item[\textbf{\textcolor{CTrap}{正确理解（任意过心线均可）：}}]
			对于中心对称图形，任何过对称中心的直线都平分其面积。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略旋转对称）：}}]
			习惯性认为只有对称轴或对角线才具有平分功能，未从旋转对称角度思考。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（盲目取PG直线）：}}]
			过定点 $P$ 作三角形面积平分线时，直接取 $PG$ 而不验证 $PG$ 是否为中线。
		\item[\textbf{\textcolor{CTrap}{正确理解（验证PG过顶点）：}}]
			只有当 $PG$ 经过顶点（即 $PG$ 为中线）时，$PG$ 才平分面积；否则不一定。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略中线前提）：}}]
			记忆了“重心+定点”的方法，但忘记检查是否满足中线条件，导致错误。
	\end{description}

\end{trapbox}
