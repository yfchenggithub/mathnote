\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	平行四边形 $ABCD$ 的顶点坐标为 $A(0,0), B(4,0), C(6,2), D(2,2)$。求其对称中心 $O$ 的坐标，并写出两条过 $O$ 且平分该平行四边形面积的直线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求中心坐标）：}]
			由 $A(0,0),C(6,2)$，得 $O$ 为 $AC$ 的中点：
			\[
				O=\left(\frac{0+6}{2},\frac{0+2}{2}\right)=(3,1).
			\]
			\par\textbf{理由：} 平行四边形对角线互相平分。
		\item[\textbf{第二步（应用性质）：}]
			由结论，过对称中心 $O$ 的任意直线均平分面积。故从过 $O$ 的直线中任取两条：取 $l_1:y=1$（水平线），$l_2$ 过 $O$ 且斜率为 $\frac13$（即 $y=\frac13 x$）。
			\[
				l_1:y=1,
				\qquad
				l_2:y=\frac13 x.
			\]
			\par\textbf{理由：} 过 $O$ 的直线都平分面积，所以可任意选取。
		\item[\textbf{第三步（写出方程）：}]
			因此，两条直线方程即为 $l_1:y=1$ 和 $l_2:y=\frac13 x$（答案不唯一）。
			\par\textbf{理由：} 只需直线过点 $O$ 即可。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	过平行四边形对称中心的任意直线平分其面积。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 的顶点 $A(0,0), B(4,0), C(0,3)$。求 $BC$ 边上的中线方程，并验证该中线平分 $\triangle ABC$ 的面积。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求中点坐标）：}]
			$BC$ 的中点 $D$ 的坐标：
			\[
				D=\left(\frac{4+0}{2},\frac{0+3}{2}\right)=\left(2,\frac32\right).
			\]
			\par\textbf{理由：} 中点坐标公式。
		\item[\textbf{第二步（求中线方程）：}]
			中线 $AD$ 的斜率 $k=\frac{3/2-0}{2-0}=\frac34$，方程为
			\[
				AD: y=\frac34 x.
			\]
			\par\textbf{理由：} 两点确定一条直线。
		\item[\textbf{第三步（验证面积平分）：}]
			因为 $D$ 为 $BC$ 中点，所以 $BD=DC$。$\triangle ABD$ 与 $\triangle ADC$ 等底同高，故面积相等。具体计算：$S_{\triangle ABC}=\frac12\cdot4\cdot3=6$，则
			\[
				S_{\triangle ABD}=S_{\triangle ADC}=\frac12 S_{\triangle ABC}=3.
			\]
			\par\textbf{理由：} 等底同高或面积公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	三角形的中线平分面积。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 的顶点 $A(0,0), B(4,0), C(0,3)$，重心为 $G$。点 $P$ 在 $AG$ 上，且 $AP:PG=1:2$。是否存在过 $P$ 的直线平分 $\triangle ABC$ 的面积？若存在，写出该直线方程并说明理由。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求重心及点P）：}]
			重心 $G=\left(\frac{0+4+0}{3},\frac{0+0+3}{3}\right)=\left(\frac43,1\right)$。由 $AP:PG=1:2$，得 $P=\frac{2A+G}{3}=\left(\frac23,\frac13\right)$。
			\[
				G=\left(\frac43,1\right),\quad P=\left(\frac23,\frac13\right).
			\]
			\par\textbf{理由：} 重心坐标公式和定比分点。
		\item[\textbf{第二步（判断PG是否过顶点）：}]
			直线 $PG$ 即直线 $AG$（因为 $P,A,G$ 共线），而 $AG$ 是中线（$A$ 与 $BC$ 中点 $D$ 的连线）。故 $PG$ 经过顶点 $A$，由推广结论，$PG$ 平分 $\triangle ABC$ 的面积。
			\par\textbf{理由：} 推广结论：过重心 $G$ 与定点 $P$ 的直线若过顶点，则为中线，从而平分面积。
		\item[\textbf{第三步（写出直线方程）：}]
			直线 $AG$ 的方程：过 $A(0,0)$，斜率 $k=\frac{1}{4/3}=\frac34$，所以
			\[
				l: y=\frac34 x.
			\]
			\par\textbf{理由：} 点斜式求直线方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	当定点 $P$ 与重心 $G$ 的连线经过三角形顶点时，该直线平分三角形面积。
\end{examplebox}
