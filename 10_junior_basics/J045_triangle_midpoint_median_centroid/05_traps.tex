	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式符号混淆）}}] 将中线长公式$m_a^2=\frac12(b^2+c^2)-\frac14a^2$中的减号误写为加号.
		\item[\textbf{\textcolor{CTrap}{正确理解（公式结构）：}}] 中线平方等于相邻两边平方和之半减去底边平方的四分之一，中间是减号.
		\item[\textbf{\textcolor{CTrap}{错因分析（符号记忆）：}}] 推导时互补角余弦抵消后产生减号，记忆不牢导致符号错误.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分中线比例颠倒）}}] 混淆$AG:GD=2:1$与$AG:AD=1:2$或$GD:AG=2:1$.
		\item[\textbf{\textcolor{CTrap}{正确理解（分点位置）：}}] 重心$G$更靠近顶点$A$，故$AG = \frac23 AD$, $GD = \frac13 AD$，所以$AG:GD=2:1$.
		\item[\textbf{\textcolor{CTrap}{错因分析（比例习惯）：}}] 许多学生习惯按从近到远的比例，但实际重心到顶点距离较长.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（中位线定理不全）}}] 仅记住“$EF=\frac12 BC$”而忽略平行关系，或在证明平行时忘记长度条件.
		\item[\textbf{\textcolor{CTrap}{正确理解（双重结论）：}}] 中位线$EF\parallel BC$且$EF=\frac12 BC$，两者同时成立，缺一不可.
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆片面）：}}] 中位线定理经常用于求长度，但平行性在几何证明中同样关键.
	\end{description}
\end{trapbox}
