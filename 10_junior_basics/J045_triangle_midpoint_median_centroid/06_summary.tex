\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{中点、中线、重心是三角形中的三大对称要素，提供长度关系（中线长公式）、平行关系（中位线）、向量关系（重心向量和为零）。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三角形条件）：} 题目明确给出三角形的边、中点或中线.
		\item \textbf{条件 2（向量背景）：} 涉及顶点向量或重心时，考虑重心向量和为零.
		\item \textbf{条件 3（构造辅助线）：} 遇中点可倍长中线构造平行四边形.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号比例）：} 中线长公式注意减号；重心分中线$2:1$勿颠倒；中位线定理平行与长度缺一不可.
		\item \textbf{检查项（适用前提）：} 使用中线长公式先确认三边对应；直角三角形斜边中线定理需直角在顶点$A$.
	\end{itemize}
\end{summarybox}
