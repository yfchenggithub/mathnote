\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=5$, $AC=6$, $BC=7$，求中线$AD$的长（结果可保留根号）.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定对应边）：}] 因$BC$对应边$a$，所以$a=7$，$b=AC=6$, $c=AB=5$.
			\par\textbf{理由：} 中线$AD$所对边是$BC$.
		\item[\textbf{第二步（代入中线长公式）：}]
			\[
				\begin{aligned}
					m_a^2 &= \frac12(b^2+c^2) - \frac14 a^2 \\
					&= \frac12(36+25) - \frac14\cdot 49 \\
					&= \frac12\cdot 61 - \frac{49}{4} \\
					&= \frac{61}{2} - \frac{49}{4} \\
					&= \frac{122-49}{4} \\
					&= \frac{73}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用公式计算.
		\item[\textbf{第三步（开平方得中线）：}]
			\[
				\begin{aligned}
					m_a &= \sqrt{\frac{73}{4}} \\
					&= \frac{\sqrt{73}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 长度取正.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AD = \dfrac{\sqrt{73}}{2}$（约4.27）}.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=5$, $AC=7$, $\angle A=60^\circ$，$AD$是中线，$G$是重心，求$AG$的长.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求第三边）：}] 由余弦定理：
			\[
				\begin{aligned}
					BC^2 &= AB^2+AC^2-2\cdot AB\cdot AC\cos60^\circ \\
					&= 25+49-2\cdot5\cdot7\cdot\frac12 \\
					&= 74-35 \\
					&= 39,
			\end{aligned}
			\]
			所以$a = BC = \sqrt{39}$.
			\par\textbf{理由：} 已知两边夹角可求第三边.
		\item[\textbf{第二步（求中线长）：}] 由中线长公式：
			\[
				\begin{aligned}
					m_a^2 &= \frac12(49+25) - \frac14\cdot 39 \\
					&= 37 - \frac{39}{4} \\
					&= \frac{148-39}{4} \\
					&= \frac{109}{4},
			\end{aligned}
			\]
			故$m_a = \dfrac{\sqrt{109}}{2}$.
			\par\textbf{理由：} 用中线长公式计算出中线长度.
		\item[\textbf{第三步（重心分中线）：}] 重心$G$将中线$AD$按$AG:GD=2:1$分割，所以
			\[
				\begin{aligned}
					AG &= \frac23 AD \\
					&= \frac23 \cdot \frac{\sqrt{109}}{2} \\
					&= \frac{\sqrt{109}}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 重心分中线比例$2:1$（从顶点到重心占$2/3$）.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AG = \dfrac{\sqrt{109}}{3}$}.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AD$是中线，延长$AD$至点$E$使$DE=AD$，连接$CE$. 求证：$AB=CE$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（标注中点）：}] 由$AD$是中线的条件得$BD=DC$；由作图得$AD=DE$，且$A,D,E$共线.
			\par\textbf{理由：} 中线定义和倍长中线构造.
		\item[\textbf{第二步（判定平行四边形）：}] 考虑四边形$ABEC$，其对角线$AE$与$BC$互相平分（因为$AD=DE$，$BD=DC$），故$ABEC$是平行四边形.
			\par\textbf{理由：} 对角线互相平分的四边形是平行四边形.
		\item[\textbf{第三步（平行四边形性质）：}] 由平行四边形对边相等得$AB = CE$.
			\par\textbf{理由：} 平行四边形对边相等.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AB = CE$成立}. 倍长中线构造平行四边形是证明线段相等的经典辅助线.
\end{examplebox}
