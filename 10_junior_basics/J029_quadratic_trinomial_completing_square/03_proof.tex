\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	将二次三项式写成$a(x^2+\frac{b}{a}x)+c$，然后在括号内配成完全平方，再合并常数。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（提取系数）：}]
			\[
				ax^2+bx+c = a\left(x^2+\frac{b}{a}x\right)+c.
			\]
			\par\textbf{理由：} 因 $a\neq 0$，提出 $a$，不影响恒等性。

		\item[\textbf{步骤二（括号配方）：}]
			\[
				\begin{aligned}
					x^2+\frac{b}{a}x
					&= \left(x^2+\frac{b}{a}x + \left(\frac{b}{2a}\right)^2\right) - \left(\frac{b}{2a}\right)^2 \\
					&= \left(x+\frac{b}{2a}\right)^2 - \frac{b^2}{4a^2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 加一次项系数一半的平方再减去它，保持恒等；利用完全平方公式合并。

		\item[\textbf{步骤三（乘以系数）：}]
			\[
				\begin{aligned}
					a\left[\left(x+\frac{b}{2a}\right)^2-\frac{b^2}{4a^2}\right]+c
					&= a\left(x+\frac{b}{2a}\right)^2 - \frac{b^2}{4a} \\
					&\quad +c.
			\end{aligned}
			\]
			\par\textbf{理由：} 分配律 $a\cdot\left(\cdot\right)$ 分别乘开。

		\item[\textbf{步骤四（合并常数）：}]
			\[
				\begin{aligned}
					-\frac{b^2}{4a}+c
					&= \frac{-b^2}{4a} + \frac{4ac}{4a} \\
					&= \frac{-b^2+4ac}{4a} \\
					&= \frac{4ac-b^2}{4a}.
			\end{aligned}
			\]
			\par\textbf{理由：} 将 $c$ 写成 $\frac{4ac}{4a}$ 后合并分子。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{综上，有}
	\[
		ax^2+bx+c = a\left(x+\frac{b}{2a}\right)^2+\frac{4ac-b^2}{4a}.
	\]
	\textbf{恒等式得证。}
\end{proofbox}
