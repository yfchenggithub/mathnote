\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	求二次函数 $y=2x^2-4x+1$ 的顶点坐标和最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取系数）：}]
			\[
				y = 2(x^2-2x)+1.
			\]
			\par\textbf{理由：} 提出二次项系数 $2$。

		\item[\textbf{第二步（括号配方）：}]
			\[
				\begin{aligned}
					y &= 2(x^2-2x)+1 \\
					&= 2[(x-1)^2-1]+1 \\
					&= 2(x-1)^2-1.
			\end{aligned}
			\]
			\par\textbf{理由：} 括号内配完全平方，再乘以系数并合并常数。

		\item[\textbf{第三步（读出结论）：}]
			\[
				(1,-1),\quad y_{\min} = -1.
			\]
			\par\textbf{理由：} 因为 $a=2>0$，开口向上，顶点纵坐标即最小值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值} $-1$，顶点 $(1,-1)$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	求函数 $y=-x^2+2x+3$ 的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取负号）：}]
			\[
				y = -(x^2-2x)+3.
			\]
			\par\textbf{理由：} 提出负号，便于配方。

		\item[\textbf{第二步（括号配方）：}]
			\[
				\begin{aligned}
					y &= -(x^2-2x)+3 \\
					&= -[(x-1)^2-1]+3 \\
					&= -(x-1)^2+4.
			\end{aligned}
			\]
			\par\textbf{理由：} 括号内配完全平方，再乘以负号并合并常数。

		\item[\textbf{第三步（读出最值）：}]
			\[
				y_{\max}=4,\quad (x=1).
			\]
			\par\textbf{理由：} 因为 $a=-1<0$，开口向下，顶点纵坐标即最大值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最大值} $4$，顶点 $(1,4)$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	证明：对任意实数 $x$，不等式 $2x^2-4x+3>0$ 恒成立。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（配方化简）：}]
			\[
				\begin{aligned}
					2x^2-4x+3 &= 2(x^2-2x)+3 \\
					&= 2[(x-1)^2-1]+3 \\
					&= 2(x-1)^2+1.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用配方法将二次式写成顶点式。

		\item[\textbf{第二步（平方非负）：}]
			\[
				\begin{aligned}
					2(x-1)^2+1 &\ge 1 \\
					&> 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 因为 $(x-1)^2\ge 0$，所以 $2(x-1)^2\ge 0$，加1得 $2(x-1)^2+1\ge 1>0$。

		\item[\textbf{第三步（得出结论）：}]
			\[
				2x^2-4x+3 > 0 \quad (\forall x\in\mathbb{R}).
			\]
			\par\textbf{理由：} 左边最小值为 $1$，始终大于 $0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{不等式恒成立}，配方得 $2(x-1)^2+1\ge1>0$。
