\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（遗漏系数a）}}] 错误认为 $ax^2+bx+c = \left(x+\frac{b}{2a}\right)^2 + \frac{4ac-b^2}{4a}$，缺少系数 $a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数不能漏）：}}] 正确恒等式为 $ax^2+bx+c = a\left(x+\frac{b}{2a}\right)^2 + \frac{4ac-b^2}{4a}$，必须保留公因子 $a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忘记提取）：}}] 提取二次项系数后，处理括号内配方时未将 $a$ 乘回，导致变形不等价。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（常数符号错误）}}] 错误写出 $\frac{4ac+b^2}{4a}$ 或 $\frac{b^2-4ac}{4a}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（分子顺序）：}}] 常数项是 $\frac{4ac-b^2}{4a}$，即分子中 $4ac$ 减 $b^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（合并出错）：}}] 合并 $-\frac{b^2}{4a}+c$ 时，通分后分子符号误算，或与求根公式 $\Delta = b^2-4ac$ 混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（最值类型混淆）}}] 无论 $a$ 正负，一律将 $\frac{4ac-b^2}{4a}$ 当作最小值。
		\item[\textbf{\textcolor{CTrap}{正确理解（开口定最大最小）：}}] 当 $a>0$ 时 $\frac{4ac-b^2}{4a}$ 是最小值；当 $a<0$ 时是最大值。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略开口方向）：}}] 只记住了顶点纵坐标是最值，但未联系二次项系数 $a$ 的符号确定最大还是最小。
	\end{description}
\end{trapbox}
