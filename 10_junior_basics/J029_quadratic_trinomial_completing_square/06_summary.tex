\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{配方恒等式将二次三项式化为顶点式，直接确定对称轴和最值。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（二次项非零）：} $a \neq 0$。
		\item \textbf{条件2（实数系数）：} $a,b,c \in \mathbb{R}$，$x \in \mathbb{R}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（遗漏系数$a$）：} 恒等式中不可缺失因子 $a$。
		\item \textbf{检查项（符号核对）：} 常数项 $\frac{4ac-b^2}{4a}$ 分子为 $4ac-b^2$，不可颠倒。
	\end{itemize}
\end{summarybox}
