\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正三角形内点）：}}]
			在正三角形 $ABC$ 内部任取一点 $P$，记 $PA=a$, $PB=b$, $PC=c$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（旋转构造关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $\triangle ABP$ 绕 $B$ 逆时针旋转 $60^\circ$ 后，可构造出等边三角形与已知边长的关系。
			\[
				PB = PQ = BQ = b,\quad CQ = a.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（余弦定理公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在 $\triangle PQC$ 中，由余弦定理可得 $PC$ 与 $PA, PB$ 及 $\angle APB$ 的关系。
			\[
				c^2 = a^2 + b^2 - 2ab\cos(\angle APB - 60^\circ).
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（勾股特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $a^2+b^2=c^2$ 时，$\angle APB = 150^\circ$。
			\[
				a^2+b^2=c^2 \quad\Rightarrow\quad \angle APB = 150^\circ.
			\]
	\end{description}

\end{statementbox}
