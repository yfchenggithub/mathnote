\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 在正三角形 $ABC$ 内部一点 $P$，已知 $PA=2$, $PB=3$, $\angle APB=120^\circ$，求 $PC$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}]
			点 $P$ 在正三角形内部，满足鸡爪模型使用条件。
			\par\textbf{理由：} 正三角形内部一点可旋转构造。
		\item[\textbf{第二步（代入数据）：}]
			由 $c^2 = a^2+b^2-2ab\cos(\angle APB-60^\circ)$，代入 $a=2$, $b=3$, $\angle APB=120^\circ$。
			\[
				c^2 = 2^2+3^2-2\cdot2\cdot3\cos(120^\circ-60^\circ).
			\]
			\par\textbf{理由：} 直接套用结论公式。
		\item[\textbf{第三步（计算）：}]
			计算得
			\[
				\begin{aligned}
					c^2 &= 4+9-12\cos60^\circ \\ &= 13-12\times\frac12 \\ &= 13-6 \\ &= 7,
			\end{aligned}
			\]
			故 $c=\sqrt7$。
			\par\textbf{理由：} 算术运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$PC=\sqrt7$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 在正三角形 $ABC$ 内部一点 $P$，已知 $PA=2$, $PB=3$, $PC=\sqrt7$，求 $\angle APB$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入公式）：}]
			由 $c^2 = a^2+b^2-2ab\cos(\angle APB-60^\circ)$，代入 $a=2$, $b=3$, $c=\sqrt7$。
			\[
				(\sqrt7)^2 = 2^2+3^2-2\cdot2\cdot3\cos(\angle APB-60^\circ).
			\]
			\par\textbf{理由：} 套用公式。
		\item[\textbf{第二步（化简方程）：}]
			化简得 $7=13-12\cos\theta$，其中 $\theta=\angle APB-60^\circ$。
			\[
				\begin{aligned}
					12\cos\theta &= 13-7 \\ &= 6,
			\end{aligned}
			\]
			所以 $\cos\theta = \frac12$。
			\par\textbf{理由：} 代数运算。
		\item[\textbf{第三步（求角度）：}]
			因为 $0^\circ<\angle APB<180^\circ$，所以 $-60^\circ<\theta<120^\circ$，在此范围内 $\cos\theta=\frac12$ 得 $\theta=60^\circ$。再由 $\angle APB = \theta + 60^\circ$，故 $\angle APB = 120^\circ$。
			\par\textbf{理由：} 反三角函数及范围判断。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle APB = 120^\circ$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 在正三角形 $ABC$ 内部一点 $P$，已知 $PA=3$, $PB=4$, $PC=5$，求 $\angle APB$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入公式）：}]
			由 $c^2 = a^2+b^2-2ab\cos(\angle APB-60^\circ)$，代入 $a=3$, $b=4$, $c=5$。
			\[
				25 = 9+16-2\cdot3\cdot4\cos(\angle APB-60^\circ).
			\]
			\par\textbf{理由：} 直接套用结论。
		\item[\textbf{第二步（化简）：}]
			化简得 $25=25-24\cos\theta$，其中 $\theta=\angle APB-60^\circ$。
			\[
				24\cos\theta=0\quad\Rightarrow\quad \cos\theta=0.
			\]
			\par\textbf{理由：} 代数运算。
		\item[\textbf{第三步（求角与讨论）：}]
			由 $\cos\theta=0$ 且 $0^\circ<\angle APB<180^\circ$ 得 $\theta=90^\circ$。再由 $\angle APB = \theta + 60^\circ$，故 $\angle APB = 150^\circ$。
			\par\textbf{理由：} 余弦值为零时角度为 $90^\circ$ 或 $270^\circ$，在范围内取 $90^\circ$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle APB = 150^\circ$。
