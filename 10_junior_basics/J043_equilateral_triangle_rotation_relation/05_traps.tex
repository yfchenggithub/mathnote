\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（角度差方向）}}]
			在公式 $c^2 = a^2+b^2-2ab\cos(\angle APB-60^\circ)$ 中，将角度写成 $\angle APB+60^\circ$。
		\item[\textbf{\textcolor{CTrap}{正确理解（几何推导）}}]
			由旋转构造得 $\angle PQC = \angle APB-60^\circ$，因此余弦定理中应使用减 $60^\circ$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混乱）}}]
			旋转 $60^\circ$ 可能误以为角度增加 $60^\circ$，实际是旋转后 $PB$ 转到 $QB$ 造成的角差。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（特例角度）}}]
			当 $a^2+b^2=c^2$ 时，误以为 $\angle APB=90^\circ$。
		\item[\textbf{\textcolor{CTrap}{正确理解（代入计算）}}]
			代入公式得 $\cos(\angle APB-60^\circ)=0$，结合 $\angle APB$ 范围得 $\angle APB=150^\circ$。
		\item[\textbf{\textcolor{CTrap}{错因分析（定势思维）}}]
			平面几何中常见的直角关系导致误判，此题因 $60^\circ$ 角差平移，直角出现在 $90^\circ$ 处。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（条件缺失）}}]
			忽略点 $P$ 必须在正三角形内部，将外部点直接套用公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（构造前提）}}]
			旋转构造依赖 $P$ 在内部，才能保证旋转后 $Q$ 位置使得 $\angle PQC=\angle APB-60^\circ$。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件感知不足）}}]
			不区分点 $P$ 位置，片面记忆公式。
	\end{description}

\end{trapbox}
