\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（定点与参数）：}}] 平面内一定点$O$，给定缩放比$k>0$及旋转角$\theta$（逆时针为正）。
		\item[\textbf{\textcolor{CStatement}{条件 2（主从关系）：}}] 从动点$B$满足$\overrightarrow{OB}$由$\overrightarrow{OA}$绕$O$旋转$\theta$并缩放$k$倍，即
			\[
				z_B - z_O = k e^{i\theta}(z_A - z_O).
			\]
		\item[\textbf{\textcolor{CStatement}{条件 3（主动轨迹）：}}] 主动点$A$在某已知直线或圆上运动。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（直线得直线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点$A$在直线上运动时，点$B$的轨迹也是直线，由原直线绕$O$旋转$\theta$并缩放$k$得到。
			\[
				z_A = z_O + t\,\mathbf{u} \;\Longrightarrow\; z_B = z_O + k e^{i\theta} t\,\mathbf{u},
			\]
			其中$\mathbf{u}$为原直线的方向向量。

		\item[\textbf{\textcolor{CStatement}{推广结论（圆得圆）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点$A$在圆上运动时，点$B$的轨迹也是圆，其圆心由原圆心经同一变换得到，半径为原半径的$k$倍。
			\[
				z_{C_B} - z_O = k e^{i\theta}(z_{C_A} - z_O),\quad r_B = k\, r_A.
			\]
	\end{description}
