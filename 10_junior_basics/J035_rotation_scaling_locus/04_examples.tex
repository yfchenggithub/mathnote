\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定点$O(0,0)$，$k=2$，$\theta=90^\circ$（逆时针）。主动点$A$在直线$y=1$上运动，从动点$B$满足$z_B = 2 e^{i\pi/2} z_A$，求$B$的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}] 条件符合瓜豆原理，$A$在直线上运动，故$B$轨迹为直线。
			\par\textbf{理由：} 由“直线得直线”结论。
		\item[\textbf{第二步（代数推导）：}] 设$A$坐标为$(x,1)$，对应复数$z_A=x+i$。代入得
			\[
				\begin{aligned}
					z_B &= 2e^{i\pi/2}z_A \\
					&= 2i(x+i) \\
					&= 2ix - 2,
			\end{aligned}
			\]
			即 $z_B = -2 + 2xi$。因此$B$的坐标为$(-2,2x)$，横坐标恒为$-2$，纵坐标随$x$变化。
			\par\textbf{理由：} 复数乘法直接计算，得出$B$的横坐标为常数。
		\item[\textbf{第三步（写出方程）：}] 故$B$的轨迹方程为$x=-2$。
			\par\textbf{理由：} 轨迹是竖直直线。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直线$x=-2$由原直线$y=1$绕$O$逆时针旋转$90^\circ$并缩放$2$倍得到。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定点$O(1,0)$，$k=\frac13$，$\theta=90^\circ$。主动点$A$在圆$(x-2)^2+y^2=9$上运动，从动点$B$满足$z_B - z_O = \frac13 e^{i\pi/2}(z_A - z_O)$，求$B$的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}] $A$在圆上运动，符合“圆得圆”条件。
			\par\textbf{理由：} 瓜豆原理圆结论。
		\item[\textbf{第二步（计算圆心与半径）：}] 原圆心$C_A(2,0)$，半径$R_A=3$。变换中心$O(1,0)$。新圆心$C_B$满足
			\[
				\begin{aligned}
					z_{C_B} - 1 &= \frac13 e^{i\pi/2}(2-1) \\
					&= \frac13 i,
			\end{aligned}
			\]
			故 $z_{C_B}=1+\frac13 i$，即 $C_B\left(1,\frac13\right)$。新半径 $R_B = \frac13 \times 3$，即 $R_B = 1$。
			\par\textbf{理由：} 圆心变换公式与半径直接乘$k$。
		\item[\textbf{第三步（写出方程）：}] $B$轨迹方程为$(x-1)^2+\left(y-\frac13\right)^2=1$。
			\par\textbf{理由：} 圆的标准方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{圆$(x-2)^2+y^2=9$经以$O(1,0)$为中心旋转$90^\circ$缩放$\frac13$得圆$(x-1)^2+(y-\frac13)^2=1$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定点$A(1,0)$，$k=2$，$\theta=60^\circ$。动点$B$在圆$x^2+y^2=4$上运动，动点$C$满足$z_C - z_A = 2 e^{i\pi/3}(z_B - z_A)$，求$C$的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}] 变换中心为$A$，$B$在圆上运动，$C$为从动点，符合瓜豆原理。
			\par\textbf{理由：} 应用圆得圆结论。
		\item[\textbf{第二步（计算圆心与半径）：}] $B$的圆心$O_B(0,0)$，半径$R_B=2$。新圆心$C_C$满足
			\[
				\begin{aligned}
					z_{C_C} - 1 &= 2 e^{i\pi/3}(0-1) \\
					&= -2e^{i\pi/3}.
			\end{aligned}
			\]
			计算 $-2(\cos60^\circ + i\sin60^\circ) = -1 - i\sqrt3$。故
			\[
				\begin{aligned}
					z_{C_C} &= 1 + (-1 - i\sqrt3) \\
					&= -i\sqrt3,
			\end{aligned}
			\]
			即 $C_C(0,-\sqrt3)$。新半径 $R_C = 2 \times 2$，即 $R_C = 4$。
			\par\textbf{理由：} 圆心变换公式，半径乘$k$。
		\item[\textbf{第三步（写出方程）：}] $C$轨迹方程为$x^2 + (y+\sqrt3)^2 = 16$。
			\par\textbf{理由：} 圆的标准方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{圆$x^2+y^2=4$经以$A(1,0)$为中心旋转$60^\circ$缩放$2$倍得圆$x^2+(y+\sqrt3)^2=16$。}
\end{examplebox}
