\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略桥长）：}}]
			求折线最小值时，直接计算 $|AB|$，认为最小值就是 $|AB|$。
		\item[\textbf{\textcolor{CTrap}{正确理解（包含桥长）：}}]
			折线包含长度固定的桥长 $d$，最小值应为 $|A'B|+d$，其中 $A'$ 是 $A$ 平移 $d$ 后的点。
		\item[\textbf{\textcolor{CTrap}{错因分析（未单独考虑）：}}]
			误将折线等同于直线，未意识到桥长是独立的固定部分，平移法可将其分离。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（平移方向）：}}]
			平移 $A$ 时，不沿垂直于平行线的方向，而是沿任意方向。
		\item[\textbf{\textcolor{CTrap}{正确理解（垂直平移）：}}]
			必须沿垂直于 $l_1,l_2$ 的方向平移，才能保证 $AA' \parallel MN$ 且 $AA'=MN$，从而利用平行四边形转化。
		\item[\textbf{\textcolor{CTrap}{错因分析（破坏构造）：}}]
			若方向不垂直，则 $AA'$ 与 $MN$ 不平行，无法构造平行四边形，等量关系 $AM=A'N$ 不成立。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（同异混淆）：}}]
			在 $A,B$ 位于两平行线同侧时，直接平移 $A$ 并使用 $|A'B|+d$ 计算最小值。
		\item[\textbf{\textcolor{CTrap}{正确理解（异侧条件）：}}]
			模型要求 $A,B$ 位于异侧；若同侧，先作对称点再平移，或改用其他方法。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件误用）：}}]
			同侧时，平移后 $A'$ 与 $B$ 仍位于同侧，$A'B$ 不穿过两平行线，取等点 $N$ 不存在于 $l_2$ 上，导致结论错误。
	\end{description}

\end{trapbox}
