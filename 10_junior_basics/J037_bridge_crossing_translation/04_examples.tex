\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两平行线 $l_1:y=0$，$l_2:y=4$，间距 $d=4$。点 $A(-1,-2)$ 在 $l_1$ 外侧下方，点 $B(5,6)$ 在 $l_2$ 外侧上方，且 $A,B$ 位于两平行线异侧。求折线 $A\to M\to N\to B$ 长度的最小值，其中 $M\in l_1$，$N\in l_2$，$MN\perp l_1,l_2$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平移构造）：}]
			将点 $A$ 沿垂直向上方向平移 $4$ 个单位，得 $A'(-1,2)$。
			\par\textbf{理由：} 平移方向垂直于平行线，长度等于桥长，使 $AA'=d$ 且 $AA'\parallel MN$，从而可构造平行四边形。
		\item[\textbf{第二步（距离计算）：}]
			由距离公式，
			\[
				|A'B| = \sqrt{(5-(-1))^2+(6-2)^2},
			\]
			\[
				|A'B| = \sqrt{36+16},
			\]
			\[
				|A'B| = \sqrt{52},
			\]
			\[
				|A'B| = 2\sqrt{13}.
			\]
			\par\textbf{理由：} 两点间距离公式逐步计算 $A'$ 到 $B$ 的距离。
		\item[\textbf{第三步（套用结论）：}]
			折线最小值 $L_{\min}=|A'B|+d$，代入数值得 $L_{\min}=2\sqrt{13}+4$。
			\par\textbf{理由：} 根据造桥选址模型，最小值等于 $|A'B|+d$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$L_{\min}=2\sqrt{13}+4$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两平行线 $l_1:x=0$，$l_2:x=3$，间距 $d=3$。点 $A(-1,2)$ 在 $l_1$ 左侧（外侧），点 $B(5,4)$ 在 $l_2$ 右侧（外侧），且 $A,B$ 位于异侧。求折线 $A\to M\to N\to B$ 的最小值，并求此时 $M,N$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平移构造）：}]
			将点 $A$ 沿水平向右平移 $3$ 个单位，得 $A'(2,2)$。
			\par\textbf{理由：} 平移方向垂直于平行线（水平），长度等于桥长 $d$。
		\item[\textbf{第二步（距离计算）：}]
			\[
				|A'B| = \sqrt{(5-2)^2+(4-2)^2},
			\]
			\[
				|A'B| = \sqrt{9+4},
			\]
			\[
				|A'B| = \sqrt{13}.
			\]
			故 $L_{\min}= \sqrt{13}+3$。
			\par\textbf{理由：} 距离公式与模型结论。
		\item[\textbf{第三步（求取等坐标）：}]
			直线 $A'B$ 方程：斜率 $k=\frac{4-2}{5-2}=\frac{2}{3}$，过 $(2,2)$，得 $y-2=\frac{2}{3}(x-2)$。令 $x=3$，得 $y=2+\frac{2}{3}$，即 $y=\frac{8}{3}$，故 $N(3,\frac{8}{3})$。由 $MN\perp l_1,l_2$ 且 $M\in l_1$，得 $M$ 与 $N$ 纵坐标相同，故 $M(0,\frac{8}{3})$。
			\par\textbf{理由：} 取等条件要求 $A',N,B$ 共线，所以 $N$ 为 $A'B$ 与 $l_2$ 的交点；$MN$ 为垂线段，故 $M$ 在过 $N$ 的垂线上。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最小值为 $\sqrt{13}+3$，取等时 $M(0,\frac{8}{3}), N(3,\frac{8}{3})$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两平行线 $l_1:y=0$，$l_2:y=4$，间距 $d=4$。点 $A(-2,-1)$ 在 $l_1$ 外侧下方，点 $B(5,7)$ 在 $l_2$ 外侧上方，且 $A,B$ 位于异侧。求折线 $A\to M\to N\to B$ 的最小值，并求此时 $M,N$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平移构造）：}]
			将点 $A$ 沿垂直向上平移 $4$ 个单位，得 $A'(-2,3)$。
			\par\textbf{理由：} 平移方向垂直于平行线，长度等于桥长，可构造平行四边形。
		\item[\textbf{第二步（距离计算）：}]
			\[
				|A'B| = \sqrt{(5-(-2))^2+(7-3)^2},
			\]
			\[
				|A'B| = \sqrt{49+16},
			\]
			\[
				|A'B| = \sqrt{65}.
			\]
			故 $L_{\min}= \sqrt{65}+4$。
			\par\textbf{理由：} 距离公式与模型结论。
		\item[\textbf{第三步（求取等坐标）：}]
			直线 $A'B$ 斜率 $k=\frac{7-3}{5-(-2)}=\frac{4}{7}$，过 $A'(-2,3)$，方程为 $y-3=\frac{4}{7}(x+2)$。令 $y=4$，得 $4-3=1$，所以 $\frac{4}{7}(x+2)=1$，解得 $x+2=\frac{7}{4}$，$x=-\frac{1}{4}$。故 $N(-\frac{1}{4},4)$。由 $MN\perp l_1,l_2$，$M$ 与 $N$ 横坐标相同，所以 $M(-\frac{1}{4},0)$。
			\par\textbf{理由：} 取等条件要求 $A',N,B$ 共线，且 $MN$ 为垂线段。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最小值为 $\sqrt{65}+4$，取等时 $M(-\frac{1}{4},0), N(-\frac{1}{4},4)$。
