\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（两平行线）：}}]
			设 $l_1 \parallel l_2$，间距为 $d$。
		\item[\textbf{\textcolor{CStatement}{条件 2（定点异侧）：}}]
			定点 $A$ 在 $l_1$ 外侧，定点 $B$ 在 $l_2$ 外侧，且 $A,B$ 位于两平行线异侧。
		\item[\textbf{\textcolor{CStatement}{条件 3（垂桥限制）：}}]
			线段 $MN$ 满足 $M \in l_1$，$N \in l_2$，$MN \perp l_1,l_2$，$|MN| = d$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最短折线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			折线长度最小值等于平移点到 $B$ 的距离与桥长之和。
			\[
				L_{\min} = |A'B| + d,
			\]
			其中 $A'$ 是点 $A$ 沿与 $l_1,l_2$ 垂直的方向平移 $d$ 至对岸一侧所得的点。
		\item[\textbf{\textcolor{CStatement}{推广结论（多线平移）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对三条平行线 $l_1,l_2,l_3$，间距 $d_1,d_2$，$A$ 在 $l_1$ 外侧，$B$ 在 $l_3$ 外侧，且 $A,B$ 异侧，则折线最小值可通过逐次平移得到。
			\[
				L_{\min} = |A''B| + d_1 + d_2,
			\]
			其中 $A''$ 是点 $A$ 先平移 $d_1$ 再平移 $d_2$ 所得的点。
	\end{description}
	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	当且仅当 $A',N,B$ 三点共线，且 $N$ 为直线 $A'B$ 与 $l_2$ 的交点，$M$ 为过 $N$ 且垂直于 $l_1,l_2$ 的直线与 $l_1$ 的交点时取等。
\end{statementbox}
