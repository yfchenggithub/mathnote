\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对角互补 $\Leftrightarrow$ 四点共圆；外角等于内对角、同侧张角相等均为等价表述。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（四点构四边形）：} 四点 $A,B,C,D$ 不共线，且按序连接形成四边形。
		\item \textbf{条件2（对角求和）：} 计算一组对角（如 $\angle A+\angle C$）是否等于 $180^\circ$；或验证外角与内对角相等。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（对角判断）：} 必须识别正确的对角，避免用邻角求和。
		\item \textbf{检查项（四点位置）：} 确认四点不共线，且四边形封闭。
	\end{itemize}
\end{summarybox}
