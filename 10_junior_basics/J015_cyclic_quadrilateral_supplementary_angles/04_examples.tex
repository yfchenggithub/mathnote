\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	四边形 $ABCD$ 中，$\angle A = 110^\circ$，$\angle C = 70^\circ$。判断 $A,B,C,D$ 四点是否共圆。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算对角和）：}]
			$\angle A + \angle C = 110^\circ + 70^\circ = 180^\circ$。
			\par\textbf{理由：} 直接求和。
		\item[\textbf{第二步（判定共圆）：}]
			由对角互补判定定理，若一组对角和为 $180^\circ$，则四点共圆。故 $A,B,C,D$ 共圆。
			\par\textbf{理由：} 结论的直接应用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	四边形 $ABCD$ 的一组对角互补（$\angle A+\angle C=180^\circ$），因此 $A,B,C,D$ 四点共圆。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在四边形 $ABCD$ 中，$\angle ABC = 90^\circ$，$\angle ADC = 90^\circ$。求证：$A,B,C,D$ 四点共圆。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别对角）：}]
			指出 $\angle B$ 和 $\angle D$ 是四边形的一组对角。
			\par\textbf{理由：} 四边形 $ABCD$ 的顶点顺序。
		\item[\textbf{第二步（求和）：}]
			$\angle B + \angle D = 90^\circ + 90^\circ = 180^\circ$。
			\par\textbf{理由：} 已知条件代入。
		\item[\textbf{第三步（判定）：}]
			根据对角互补判定定理，$A,B,C,D$ 四点共圆。
			\par\textbf{理由：} 结论直接应用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	由于 $\angle B+\angle D=180^\circ$，故 $A,B,C,D$ 共圆。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在锐角三角形 $\triangle ABC$ 中，$BD \perp AC$ 于点 $D$，$CE \perp AB$ 于点 $E$。求证：$B,C,D,E$ 四点共圆。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定同底）：}]
			取 $BC$ 为底，$D,E$ 在 $BC$ 的同侧。
			\par\textbf{理由：} $D,E$ 分别为 $AC,AB$ 上的垂足，都在 $\triangle ABC$ 内部。
		\item[\textbf{第二步（张角相等）：}]
			由 $CE\perp AB$ 得 $\angle BEC = 90^\circ$，由 $BD\perp AC$ 得 $\angle BDC = 90^\circ$，故 $\angle BEC = \angle BDC$。
			\par\textbf{理由：} 垂直定义。
		\item[\textbf{第三步（判定共圆）：}]
			对同底 $BC$，$D,E$ 两点的张角相等，所以 $B,C,D,E$ 四点共圆。
			\par\textbf{理由：} 同侧张角相等判定定理（等价于对角互补）。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle BEC=\angle BDC=90^\circ$，故 $B,C,D,E$ 共圆。
\end{examplebox}
