\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（四点不共线）：}}] 平面内不共线的四点$A,B,C,D$按序连成四边形$ABCD$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对角互补）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			四边形一组对角之和等于$180^\circ$。
			\[
				\angle A+\angle C=180^\circ \iff \angle B+\angle D=180^\circ .
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（外角内角等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外角等于内对角，或同侧张角相等。
			延长 $BC$ 至 $E$，则
			\[
				\angle DCE=\angle A .
			\]
			对同底 $AB$，有
			\[
				\angle ADB=\angle ACB .
			\]
	\end{description}
\end{statementbox}
