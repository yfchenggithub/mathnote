\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 求二次函数 $y=2x^2-4x+1$ 的判别式 $\Delta$，并判断其图像与 $x$ 轴的交点个数，再求函数的最小值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求判别式）：}]
			由 $a=2,b=-4,c=1$，计算判别式：\n
			\[
				\n
				\begin{aligned}
					\n\Delta &= b^2-4ac \\\n &= (-4)^2-4\times2\times1 \\\n &= 16-8=8.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 直接代入判别式定义。\n
		\item[\textbf{第二步（判交点个数）：}]
			因为 $\Delta=8>0$，所以图像与 $x$ 轴有两个交点。\n
			\par\textbf{理由：} $\Delta>0$ 对应两个不等实根，即两个交点。\n
		\item[\textbf{第三步（求最小值）：}]
			由于 $a=2>0$，函数有最小值。由最值公式：\n
			\[
				\n
				\begin{aligned}
					\ny_{\min} &= -\frac{\Delta}{4a} \\\n &= -\frac{8}{4\times2} \\\n &= -1.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 最值公式 $-\dfrac{\Delta}{4a}$，且 $a>0$ 时取最小值。\n
	\end{description}
	\n\textbf{\textcolor{CExample}{关键结论：}} $\Delta=8>0$，故有两个交点；最小值 $y_{\min}=-1$。\n\n

	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知二次函数 $y=x^2+mx+1$ 的图像与 $x$ 轴有交点，求实数 $m$ 的取值范围。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（写判别式）：}]
			$a=1,b=m,c=1$，则 $\Delta = m^2-4$。\n
			\par\textbf{理由：} 判别式公式。\n
		\item[\textbf{第二步（列条件）：}]
			图像与 $x$ 轴有交点 $\iff$ 方程有实根 $\iff \Delta\ge0$，故 $m^2-4\ge0$。\n
			\par\textbf{理由：} $\Delta\ge0$ 保证实根存在。\n
		\item[\textbf{第三步（解不等式）：}]
			由 $m^2\ge4$ 得 $|m|\ge2$，即 $m\le -2$ 或 $m\ge2$。\n
			\par\textbf{理由：} 解二次不等式得到范围。\n
	\end{description}
	\n\textbf{\textcolor{CExample}{关键结论：}} 当 $m\le -2$ 或 $m\ge2$ 时，图像与 $x$ 轴有交点。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知二次函数 $f(x)=ax^2-2ax+1$ （$a>0$）的最小值为 $-1$。（1）求 $a$ 的值；（2）判断该函数图像与 $x$ 轴的交点情况。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（配方法）：}]
			\n
			\[
				\n
				\begin{aligned}
					\nf(x) &= a(x^2-2x)+1 \\\n &= a[(x-1)^2-1]+1 \\\n &= a(x-1)^2+1-a,\n
			\end{aligned}
				\n
			\]
			\n顶点纵坐标为 $1-a$。\n
			\par\textbf{理由：} 配方得到顶点式。\n
		\item[\textbf{第二步（利用最值求 $a$）：}]
			已知最小值 $-1$，且 $a>0$，故最小值即为顶点纵坐标，所以 $1-a=-1$，解得 $a=2$。\n
			\par\textbf{理由：} 开口向上时，最小值在顶点处取得。\n
		\item[\textbf{第三步（判交点）：}]
			当 $a=2$ 时，$f(x)=2x^2-4x+1$，计算判别式：\n
			\[
				\n\Delta = (-4)^2-4\times2\times1 = 8,\n
			\]
			\n由于 $8>0$，所以图像与 $x$ 轴有两个交点。\n
			\par\textbf{理由：} $\Delta>0$ 直接得出两个交点。\n
	\end{description}
	\n\textbf{\textcolor{CExample}{关键结论：}} $a=2$；$\Delta=8>0$，故图像与 $x$ 轴有两个交点。\n
\end{examplebox}
