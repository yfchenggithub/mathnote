	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略$a\neq0$）：}}]
			解题时直接使用判别式或最值公式，但没有确认 $a\neq0$。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（先检二次项）：}}]
			使用判别式 $\Delta$ 和公式 $y_{\text{极}}=-\dfrac{\Delta}{4a}$ 前，必须验证二次项系数 $a\neq0$，否则函数可能是一次函数，公式失效。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义）：}}]
			误以为所有形如 $ax^2+bx+c$ 的式子都是二次函数，忽略 $a=0$ 的可能性，导致条件不满足时仍套用公式。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（交点误判与开口）：}}]
			认为 $\Delta>0$ 且开口向上时才有两个交点，或者认为 $\Delta<0$ 时图像一定在 $x$ 轴上方。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（交点只靠$\Delta$）：}}]
			交点个数完全由 $\Delta$ 决定，与开口方向无关：$\Delta>0$ 恒有两个交点；$\Delta<0$ 恒无交点。开口只决定图像在交点之间是在 $x$ 轴上方还是下方。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆位置）：}}]
			将交点个数与图像恒正恒负的条件混淆，恒正需要 $\Delta<0$ 且 $a>0$，但交点个数只由 $\Delta$ 决定。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（极值类型忽略$a$）：}}]
			直接使用 $y_{\text{极}}=-\dfrac{\Delta}{4a}$ 求最值，但不判断 $a$ 的正负，导致将最小值写成最大值。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（先定开口再定极）：}}]
			最值类型由 $a$ 的符号决定：$a>0$ 时 $-\dfrac{\Delta}{4a}$ 是最小值，$a<0$ 时是最大值。必须单独注明。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（机械套用）：}}]
			只记住公式，忘记公式中 $a$ 的符号影响极值类型，或误以为公式直接给出最小值。\n
	\end{description}
