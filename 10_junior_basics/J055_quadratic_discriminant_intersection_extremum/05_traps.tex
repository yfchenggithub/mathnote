\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略 $a\neq0$）：}}]
		      解题时直接使用判别式或最值公式，但没有确认 $a\neq0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（先检二次项）：}}]
		      使用判别式 $\Delta$ 和公式 $y_{\text{顶}}=-\dfrac{\Delta}{4a}$ 前，必须验证二次项系数 $a\neq0$，否则函数可能退化为一次函数，公式失效。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义）：}}]
		      误以为所有形如 $ax^2+bx+c$ 的式子都是二次函数，忽略了 $a=0$ 的可能性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（交点误判与开口）：}}]
		      认为 $\Delta>0$ 且开口向上时才有两个交点，或者认为 $\Delta<0$ 时图像一定在 $x$ 轴上方。
		\item[\textbf{\textcolor{CTrap}{正确理解（交点只看 $\Delta$）：}}]
		      交点个数只由 $\Delta$ 决定，与开口方向无关：$\Delta>0$ 有两个交点，$\Delta=0$ 有一个交点，$\Delta<0$ 无交点。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆位置）：}}]
		      交点个数与图像恒正、恒负不是同一个问题。恒正需要 $a>0$ 且 $\Delta<0$；恒负需要 $a<0$ 且 $\Delta<0$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（极值类型忽略 $a$）：}}]
		      直接使用 $-\dfrac{\Delta}{4a}$ 求最值，但不判断 $a$ 的正负，导致把最小值写成最大值，或把最大值写成最小值。
		\item[\textbf{\textcolor{CTrap}{正确理解（先定开口再定值）：}}]
		      $a>0$ 时 $-\dfrac{\Delta}{4a}$ 是最小值；$a<0$ 时 $-\dfrac{\Delta}{4a}$ 是最大值。必须单独说明极值类型。
		\item[\textbf{\textcolor{CTrap}{错因分析（机械套用）：}}]
		      只记住公式，忘记公式中 $a$ 的符号决定开口方向，也决定极值类型。
	\end{description}
\end{trapbox}
