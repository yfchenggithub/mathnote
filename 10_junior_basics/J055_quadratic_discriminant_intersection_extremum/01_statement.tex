	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（交点个数）：}}]
			\n\textbf{\textcolor{CStatement}{直观描述：}} 判别式 $\Delta$ 的符号决定图像与 $x$ 轴的交点个数。\n
			\[
				\n\Delta>0 \iff \text{两个交点},\quad\n\Delta=0 \iff \text{一个交点(相切)},\quad\n\Delta<0 \iff \text{无交点}.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{结论二（最值公式）：}}]
			\n\textbf{\textcolor{CStatement}{直观描述：}} 函数最值由 $\Delta$ 和 $a$ 的符号共同决定。\n
			\[
				\ny_{\text{极}} = -\frac{\Delta}{4a}.\n
			\]
			\n当 $a>0$ 时，函数有最小值；当 $a<0$ 时，函数有最大值。\n
	\end{description}

	\[
		\n\left(-\frac{b}{2a},\ -\frac{\Delta}{4a}\right).\n
