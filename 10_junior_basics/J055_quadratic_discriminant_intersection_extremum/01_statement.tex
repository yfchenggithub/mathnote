\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（函数形式）：}}]
		      一元二次函数
		      \[
			      y=ax^2+bx+c,\qquad a\neq0,
		      \]
		      定义域为全体实数，判别式记为
		      \[
			      \Delta=b^2-4ac.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（交点个数）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 判别式 $\Delta$ 的符号决定图像与 $x$ 轴的交点个数。
		      \[
			      \begin{cases}
				      \Delta>0 \iff \text{两个交点},     \\
				      \Delta=0 \iff \text{一个交点（相切）}, \\
				      \Delta<0 \iff \text{无交点}.
			      \end{cases}
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（顶点纵坐标与最值）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 顶点纵坐标为 $-\dfrac{\Delta}{4a}$；当 $a>0$ 时它是最小值，当 $a<0$ 时它是最大值。
		      \[
			      y_{\text{顶}}=-\frac{\Delta}{4a}.
		      \]
		      当 $a>0$ 时，函数有最小值；当 $a<0$ 时，函数有最大值。
	\end{description}

	\textbf{\textcolor{CStatement}{推广结论（顶点坐标）：}}
	\textbf{\textcolor{CStatement}{直观描述：}} 二次函数的顶点坐标与判别式 $\Delta$ 直接相关：
	\[
		\left(-\frac{b}{2a},\ -\frac{\Delta}{4a}\right).
	\]
\end{statementbox}
