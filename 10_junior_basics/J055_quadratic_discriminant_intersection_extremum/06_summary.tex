\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{判别式 $\Delta$ 既决定二次函数图像与 $x$ 轴的交点个数，又能通过 $-\dfrac{\Delta}{4a}$ 得到顶点纵坐标，从而确定函数的最值。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（函数形式）：} 二次函数 $y=ax^2+bx+c$，且 $a\neq0$。
		\item \textbf{条件 2（定义域）：} 自变量 $x$ 取全体实数；若只在某个区间上求最值，则还需比较区间端点与顶点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略系数）：} 使用 $\Delta$ 和顶点纵坐标公式前，必须确认 $a\neq0$。
		\item \textbf{检查项（极值类型）：} 求最值时务必根据 $a>0$ 或 $a<0$ 判断是最大值还是最小值。
	\end{itemize}
\end{summarybox}
