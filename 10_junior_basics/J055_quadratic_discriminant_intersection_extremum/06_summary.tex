\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{判别式 $\Delta$ 既决定了二次函数图像与 $x$ 轴的交点个数，又与 $a$ 共同确定了函数的最值。}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（函数形式）：} 二次函数 $y=ax^2+bx+c$ 且 $a\neq0$。\n
		\item \textbf{条件 2（定义域）：} 自变量 $x$ 取全体实数，否则最值可能发生变化。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（忽略系数）：} 使用 $\Delta$ 和 $y_{\text{极}}$ 公式前必须确认 $a\neq0$。\n
		\item \textbf{检查项（极值类型）：} 求最值时务必根据 $a>0$ 或 $a<0$ 确定是最大值还是最小值。\n
	\end{itemize}
	\n
\end{summarybox}
