\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n判别式 $\Delta$ 如同二次函数的"眼睛"，既能判定图像与 $x$ 轴相交的个数，又能直接导出函数的最值。\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（判定根个数）：}]
			二次方程 $ax^2+bx+c=0$ 的判别式 $\Delta$ 的正负对应方程实根的个数，从而决定函数图像与 $x$ 轴交点的个数。\n
		\item[\textbf{要点二（导出最值）：}]
			通过配方法 $y=a\left(x+\dfrac{b}{2a}\right)^2-\dfrac{\Delta}{4a}$，可知顶点纵坐标为 $-\dfrac{\Delta}{4a}$，当 $a>0$ 时是最小值，$a<0$ 时是最大值。\n
	\end{description}
	\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（判距定位）}}\n\n$\Delta$ 的符号实际上刻画了抛物线顶点 $\left(-\dfrac{b}{2a},-\dfrac{\Delta}{4a}\right)$ 与 $x$ 轴的距离关系：$\Delta>0$ 时顶点在 $x$ 轴下方（若 $a>0$）或上方（若 $a<0$），与 $x$ 轴相交；$\Delta=0$ 时顶点恰在 $x$ 轴上；$\Delta<0$ 时顶点远离 $x$ 轴，无交点。\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（根系结合）}}\n\n判别式 $\Delta$ 来自二次方程的求根公式 $x=\dfrac{-b\pm\sqrt{\Delta}}{2a}$，它统一了根的存在性、个数与函数图像的关系，是沟通方程与函数的核心桥梁。\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（高频应用）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（交点判定）：} 直接利用 $\Delta$ 判定二次函数图像与 $x$ 轴的交点情况，尤其是参数含在 $b$ 或 $c$ 中时，转化为解不等式。\n
		\item \textbf{考法二（最值计算）：} 已知函数解析式，直接代入 $\Delta$ 和 $a$ 求最值，或根据最值反求参数。\n
	\end{itemize}
	\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n把 $\Delta$ 既看成根的判别式，又视为顶点纵坐标的分子，一次记忆，双重应用。\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（判交点）：} 已知二次函数，求其与 $x$ 轴交点情况。\n
		\item \textbf{场景二（求最值）：} 已知二次函数，求最大值或最小值。\n
		\item \textbf{场景三（恒成立）：} 利用 $\Delta<0$ 且 $a>0$ 或 $a<0$ 判断二次函数恒正或恒负。\n
	\end{itemize}
	\n
\end{explanationbox}
