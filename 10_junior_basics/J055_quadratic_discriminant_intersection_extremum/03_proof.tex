\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	先通过配方法把二次函数化为顶点式，得到顶点纵坐标与 $\Delta$ 的关系；再通过求根公式，利用 $\Delta$ 判断实根个数，从而得到图像与 $x$ 轴的交点个数。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（配方变形）：}]
		      将 $y=ax^2+bx+c$ 配方：
		      \[
			      \begin{aligned}
				      y
				       & =a\left(x^2+\frac{b}{a}x\right)+c                               \\
				       & =a\left[\left(x+\frac{b}{2a}\right)^2-\frac{b^2}{4a^2}\right]+c \\
				       & =a\left(x+\frac{b}{2a}\right)^2-\frac{b^2}{4a}+c                \\
				       & =a\left(x+\frac{b}{2a}\right)^2+\frac{4ac-b^2}{4a}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 配方法是将二次函数化为顶点式的标准手段，便于看出对称轴和顶点坐标。

		\item[\textbf{步骤二（引入判别式）：}]
		      因为 $\Delta=b^2-4ac$，所以 $4ac-b^2=-\Delta$。于是
		      \[
			      y=a\left(x+\frac{b}{2a}\right)^2-\frac{\Delta}{4a},
		      \]
		      顶点纵坐标为
		      \[
			      y_{\text{顶}}=-\frac{\Delta}{4a}.
		      \]
		      \par\textbf{理由：} 将 $4ac-b^2$ 用 $-\Delta$ 代入即可得到顶点纵坐标。

		\item[\textbf{步骤三（判断最值）：}]
		      由于 $a\neq0$，当 $a>0$ 时抛物线开口向上，顶点处取得最小值；当 $a<0$ 时抛物线开口向下，顶点处取得最大值。因此
		      \[
			      \begin{cases}
				      y_{\min}=-\dfrac{\Delta}{4a}, & a>0, \\[4pt]
				      y_{\max}=-\dfrac{\Delta}{4a}, & a<0.
			      \end{cases}
		      \]
		      \par\textbf{理由：} 二次函数在顶点处取得唯一极值，极值类型由开口方向决定。

		\item[\textbf{步骤四（解对应方程）：}]
		      令 $y=0$，得到二次方程 $ax^2+bx+c=0$。由求根公式可得
		      \[
			      x=\frac{-b\pm\sqrt{\Delta}}{2a},
		      \]
		      其中 $\Delta=b^2-4ac$。
		      \par\textbf{理由：} 图像与 $x$ 轴的交点横坐标，就是对应二次方程的实根。

		\item[\textbf{步骤五（判别式与根）：}]
		      根号内的 $\Delta$ 决定实根个数：
		      \[
			      \begin{cases}
				      \Delta>0 \Longrightarrow \text{两个不等实根},       \\
				      \Delta=0 \Longrightarrow \text{两个相等实根（一个重根）}, \\
				      \Delta<0 \Longrightarrow \text{无实根}.
			      \end{cases}
		      \]
		      \par\textbf{理由：} 在实数范围内，$\sqrt{\Delta}$ 是否存在以及是否为零，决定实根的个数与形式。

		\item[\textbf{步骤六（对应交点）：}]
		      二次函数图像与 $x$ 轴交点的横坐标正是 $ax^2+bx+c=0$ 的实根，故
		      \[
			      \begin{cases}
				      \Delta>0 \iff \text{两个交点},     \\
				      \Delta=0 \iff \text{一个交点（相切）}, \\
				      \Delta<0 \iff \text{无交点}.
			      \end{cases}
		      \]
		      \par\textbf{理由：} 方程实根的个数等于函数图像与 $x$ 轴交点的个数。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	综上，判别式 $\Delta$ 不仅决定二次函数图像与 $x$ 轴的交点个数，而且与二次项系数 $a$ 共同确定函数的最值。因此 $\Delta$ 是研究二次函数性质的重要工具。
\end{proofbox}
