\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n先通过配方法将二次函数化为顶点式，得到顶点纵坐标与 $\Delta$ 的关系；再通过求解二次方程，利用 $\Delta$ 判断实根个数，从而得到交点个数。\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（配方变形）：}]
			将 $y=ax^2+bx+c$ 配方：\n
			\[
				\n
				\begin{aligned}
					\ny &= a\left(x^2+\frac{b}{a}x\right)+c \\\n &= a\left[\left(x+\frac{b}{2a}\right)^2-\frac{b^2}{4a^2}\right]+c \\\n &= a\left(x+\frac{b}{2a}\right)^2-\frac{b^2}{4a}+c \\\n &= a\left(x+\frac{b}{2a}\right)^2+\frac{4ac-b^2}{4a}.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 配方法是将二次函数化为顶点式的标准手段，便于看出对称轴和顶点坐标。\n\n
		\item[\textbf{步骤二（引入判别式）：}]
			注意到 $\Delta=b^2-4ac$，则 $4ac-b^2=-\Delta$，于是顶点纵坐标为 $-\dfrac{\Delta}{4a}$：\n
			\[
				\ny_{\text{顶}} = -\frac{\Delta}{4a}.\n
			\]
			\n
			\par\textbf{理由：} 将 $4ac-b^2$ 用 $-\Delta$ 代入，化简得顶点纵坐标的简洁形式。\n\n
		\item[\textbf{步骤三（判断最值）：}]
			由于 $a\neq0$，当 $a>0$ 时抛物线开口向上，顶点处取得最小值 $-\dfrac{\Delta}{4a}$；当 $a<0$ 时开口向下，顶点处取得最大值 $-\dfrac{\Delta}{4a}$。\n
			\[
				\ny_{\text{极}} =
				\begin{cases}
					\n\text{最小值} & (a>0),\\\n\text{最大值} & (a<0).\n
			\end{cases}
				\n
			\]
			\n
			\par\textbf{理由：} 二次函数在顶点处取得唯一极值，极值类型由开口方向决定。\n\n
		\item[\textbf{步骤四（解对应方程）：}]
			令 $y=0$ 得 $ax^2+bx+c=0$，其解为\n
			\[
				\nx = \frac{-b\pm\sqrt{\Delta}}{2a},\n
			\]
			\n其中 $\Delta=b^2-4ac$。\n
			\par\textbf{理由：} 通过求根公式得到二次方程的解的表达形式。\n\n
		\item[\textbf{步骤五（判别式与根）：}]
			根号内的 $\Delta$ 决定了实根个数：\n
			\[
				\n
				\begin{cases}
					\n\Delta>0 & \Longrightarrow\; \text{两个不等实根},\\\n\Delta=0 & \Longrightarrow\; \text{两个相等实根(一个重根)},\\\n\Delta<0 & \Longrightarrow\; \text{无实根}.\n
			\end{cases}
				\n
			\]
			\n
			\par\textbf{理由：} 在实数范围内，当且仅当 $\sqrt{\Delta}$ 有意义且不为零时，根的个数及形式随之确定。\n\n
		\item[\textbf{步骤六（对应交点）：}]
			二次函数图像与 $x$ 轴交点的横坐标即为 $ax^2+bx+c=0$ 的实根，故 $\Delta$ 的符号直接给出了交点个数：\n
			\[
				\n
				\begin{cases}
					\n\Delta>0 & \iff \text{两个交点},\\\n\Delta=0 & \iff \text{一个交点(相切)},\\\n\Delta<0 & \iff \text{无交点}.\n
			\end{cases}
				\n
			\]
			\n
			\par\textbf{理由：} 函数零点（即方程实根）的个数等于图像穿过 $x$ 轴的点数。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n综上，判别式 $\Delta$ 不仅决定了二次函数图像与 $x$ 轴的交点个数，而且其与二次项系数 $a$ 共同确定了函数的最值（$y_{\text{极}}=-\dfrac{\Delta}{4a}$）。因此 $\Delta$ 是研究二次函数性质的核心工具。\n
\end{proofbox}
