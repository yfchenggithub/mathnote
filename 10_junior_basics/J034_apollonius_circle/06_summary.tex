\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{已知两定点 $A,B$ 及定比 $k>0,\ k\neq 1$，则满足 $PA=kPB$ 的点 $P$ 的轨迹是一个圆，圆心在直线 $AB$ 上，半径为 $R=\dfrac{k}{|1-k^2|}\cdot |AB|$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（两定点）：} 存在两个固定点 $A,B$。
		\item \textbf{条件 2（比值条件）：} 常数 $k>0$ 且 $k\neq 1$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（半径绝对值）：} 半径公式必须加绝对值：$R=\dfrac{k}{|1-k^2|}\cdot |AB|$。
		\item \textbf{检查项（$k=1$ 退化）：} 当 $k=1$ 时轨迹退化为中垂线，不能误用圆结论。
	\end{itemize}
\end{summarybox}
