\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n已知两定点$A,B$及定比$k>0,k\neq1$，则满足$PA=kPB$的点$P$的轨迹是一个圆，圆心在$AB$上，半径为$R=\frac{k}{|1-k^2|}\cdot AB$.\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（两定点）：} 存在两个固定点$A,B$.\n
		\item \textbf{条件2（比值条件）：} 常数$k>0$且$k\neq1$.\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（半径绝对值）：} 半径公式必须加绝对值：$R=\frac{k}{|1-k^2|}\cdot AB$.\n
		\item \textbf{检查项（k=1退化）：} 当$k=1$时轨迹退化为中垂线，不能误用圆结论.\n
	\end{itemize}
	\n\n
\end{summarybox}
