	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（k=1特例）}}]
			误以为所有满足$PA=kPB$的轨迹都是圆，忽略$k=1$时的退化情形.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（中垂线情形）：}}]
			当$k=1$时，动点到$A,B$距离相等，轨迹是线段$AB$的中垂线.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（特例遗漏）：}}]
			只记忆了圆的一般结论，未考虑比值等于1时条件的变化.\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（半径缺绝对值）}}]
			直接写$R=\frac{k}{1-k^2}\cdot AB$，当$k>1$时$1-k^2<0$，会得到负半径，导致错误.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（加绝对值）：}}]
			半径公式应为$R=\frac{k}{|1-k^2|}\cdot AB$，保证半径非负.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（符号忽略）：}}]
			未考虑$1-k^2$可能为负，忘记绝对值.\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（圆心位置错误）}}]
			错误认为圆心在$AB$中点，或者使用内分点作为圆心.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（有向距离）：}}]
			圆心在直线$AB$上，到$A$的有向距离为$\frac{k^2}{k^2-1}\cdot AB$，方向由$A$指向$B$.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（公式混淆）：}}]
			混淆了内分点$C$、外分点$D$与圆心的关系，未正确使用有向距离公式.\n
	\end{description}
	\n\n
