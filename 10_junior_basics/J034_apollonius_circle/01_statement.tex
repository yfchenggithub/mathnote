\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（定点定比）：}}]
		      给定两个定点 $A,B$ 以及常数 $k>0,\ k\neq 1$。动点 $P$ 满足
		      \[
			      PA=kPB.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（轨迹为圆）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      动点 $P$ 到 $A$ 的距离等于 $k$ 倍到 $B$ 的距离，轨迹是一个圆。
		      \[
			      R=\frac{k}{|1-k^{2}|}\,|AB|.
		      \]
		      圆心在直线 $AB$ 上，圆心相对 $A$ 的有向距离为
		      \[
			      \frac{k^{2}}{k^{2}-1}\,|AB|,
		      \]
		      其中正方向为 $A\to B$。

		\item[\textbf{\textcolor{CStatement}{推广结论（$k=1$ 退化）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      当 $k=1$ 时，条件变为 $PA=PB$，轨迹退化为线段 $AB$ 的中垂线。
	\end{description}
\end{statementbox}
