	\n\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（定点定比）：}}]
			给定两个定点 $A,B$ 以及常数 $k>0,k\neq1$.\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（轨迹为圆）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			动点 $P$ 到 $A$ 的距离等于 $k$ 倍到 $B$ 的距离，轨迹是一个圆.\n
			\[
				\n R = \frac{k}{|1-k^{2}|}\,|AB|.\n
			\]
			\n 圆心在直线$AB$上，到$A$的有向距离为$\frac{k^{2}}{k^{2}-1}\,|AB|$（正方向为$A\to B$）.\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（$k=1$退化）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $k=1$ 时轨迹退化为线段 $AB$ 的中垂线.\n
			\[
				\n PA = PB.\n
			\]
			\n\n
	\end{description}
	\n\n
\end{statementbox}
