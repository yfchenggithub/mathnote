\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n通过坐标法，将几何条件转化为代数方程，经配方得到圆的标准形式，进而确定圆心半径，并验证内、外分点构成直径.\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（建系设点）：}]
			设 $A(0,0)$, $B(d,0)$ ($d>0$)，动点 $P(x,y)$.\n
			\[
				\n A(0,0),\quad B(d,0),\quad P(x,y).\n
			\]
			\n
			\par\textbf{理由：} 将几何元素用坐标表示，便于列方程.\n\n
		\item[\textbf{步骤二（列距离比）：}]
			由 $PA = k PB$ 平方得 $PA^{2}=k^{2} PB^{2}$，即\n
			\[
				\n x^{2}+y^{2}=k^{2}\bigl[(x-d)^{2}+y^{2}\bigr].\n
			\]
			\n
			\par\textbf{理由：} 平方消除根号，转化为代数等式.\n\n
		\item[\textbf{步骤三（展开整理）：}]
			展开括号并移项：\n
			\[
				\n (1-k^{2})x^{2}+(1-k^{2})y^{2}+2k^{2} d\,x - k^{2} d^{2}=0.\n
			\]
			\n
			\par\textbf{理由：} 将所有项移到左边，合并同类项.\n\n
		\item[\textbf{步骤四（配方变形）：}]
			当 $k\neq1$ 时，两边除以 $1-k^{2}$ 并配方：\n
			\[
				\n
				\begin{aligned}
					\n x^{2}+y^{2}+\frac{2k^{2} d}{1-k^{2}}x-\frac{k^{2} d^{2}}{1-k^{2}} &= 0 \\ \Bigl(x+\frac{k^{2} d}{1-k^{2}}\Bigr)^{2} + y^{2}\n &= \Bigl(\frac{k d}{|1-k^{2}|}\Bigr)^{2}.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 配方得到标准圆方程，注意 $1-k^{2}$ 可能为负，半径取绝对值.\n\n
		\item[\textbf{步骤五（圆心半径）：}]
			对比标准式得圆心 $O$ 和半径 $R$：\n
			\[
				\n O\Bigl(-\frac{k^{2} d}{1-k^{2}},\,0\Bigr),
				\qquad\n R = \frac{k d}{|1-k^{2}|}.\n
			\]
			\n 当 $k>1$ 时 $1-k^{2}<0$，可改写为 $O\Bigl(\dfrac{k^{2} d}{k^{2}-1},\,0\Bigr)$.\n
			\par\textbf{理由：} 由圆的标准方程直接读出圆心和半径.\n\n
		\item[\textbf{步骤六（验证直径）：}]
			内分点 $C$ 满足 $\dfrac{CA}{CB}=k$，得 $C\Bigl(\dfrac{k d}{1+k},0\Bigr)$；\n 外分点 $D$ 满足 $\dfrac{DA}{DB}=k$（$D$ 在直线 $AB$ 上），得 $D\Bigl(\dfrac{k d}{k-1},0\Bigr)$.\n 验证 $CD$ 为直径：\n
			\[
				\n
				\begin{aligned}
					\n \frac{1}{2}\!\left(\frac{k d}{1+k}+\frac{k d}{k-1}\right) &= \frac{k^{2} d}{k^{2}-1},\\n \Bigl|\frac{k d}{k-1}-\frac{k d}{1+k}\Bigr| &= 2\cdot\frac{k d}{|1-k^{2}|},\\n \Bigl|\frac{k d}{k-1}-\frac{k d}{1+k}\Bigr| &= 2R.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 中点横坐标等于圆心横坐标，且 $CD$ 长度等于 $2R$，故 $CD$ 为直径.\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n通过坐标法严格证明了：满足 $PA=kPB$ 的点的轨迹是以 $AB$ 的内、外分点 $C,D$ 为直径端点的圆，圆心在直线 $AB$ 上，半径 $R=\dfrac{k}{|1-k^{2}|}\,|AB|$. 当 $k=1$ 时配方过程分母为零，轨迹退化为中垂线.\n\n
\end{proofbox}
