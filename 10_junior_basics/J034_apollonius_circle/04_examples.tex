\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知点$A(0,0),B(4,0)$，动点$P$满足$PA=2PB$，求点$P$的轨迹方程.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（计算参数）：}]
			$d=AB=4$，$k=2$.\n
			\par\textbf{理由：} 确定基本量.\n
		\item[\textbf{第二步（求圆心半径）：}]
			半径$R=\frac{k}{|1-k^2|}d=\frac{2}{3}\cdot4=\frac{8}{3}$；圆心在有向距离$\frac{k^2}{k^2-1}d=\frac{4}{3}\cdot4=\frac{16}{3}$处，故圆心$O(\frac{16}{3},0)$.\n
			\par\textbf{理由：} 直接套用阿波罗尼斯圆结论.\n
		\item[\textbf{第三步（写方程）：}]
			轨迹方程为$\bigl(x-\frac{16}{3}\bigr)^2+y^2=\frac{64}{9}$.\n
			\par\textbf{理由：} 圆的标准方程.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{轨迹是以$(\frac{16}{3},0)$为圆心、$\frac{8}{3}$为半径的圆.}\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知点$A(-1,0),B(1,0)$，动点$Q$满足$QA=3QB$，求点$Q$的轨迹方程，并判断原点$O(0,0)$是否在该圆上.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（计算参数）：}]
			$d=AB=2$，$k=3$.\n
			\par\textbf{理由：} 代入公式前先求距离和比值.\n
		\item[\textbf{第二步（求圆心半径）：}]
			半径$R=\frac{3}{|1-9|}\cdot2=\frac{3}{8}\cdot2=\frac34$；圆心在有向距离$\frac{9}{9-1}\cdot2=\frac{9}{8}\cdot2=\frac94$处，从$A$向$B$方向，$A(-1,0)$，故圆心$O_1(\frac54,0)$.\n
			\par\textbf{理由：} 利用圆心有向距离公式.\n
		\item[\textbf{第三步（检验原点）：}]
			圆方程为$(x-\frac54)^2+y^2=\frac{9}{16}$，代入$(0,0)$得$\frac{25}{16}\neq\frac{9}{16}$，故原点不在圆上.\n
			\par\textbf{理由：} 点满足方程才在圆上.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{轨迹方程为$(x-\frac54)^2+y^2=\frac{9}{16}$，原点不在该圆上.}\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知点$A(0,0),B(6,0)$，动点$P$满足$PA=\frac12 PB$，求$\triangle ABP$面积的最大值.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（得轨迹）：}]
			$d=6$，$k=\frac12$，半径$R=\frac{0.5}{|1-0.25|}\cdot6=\frac{0.5}{0.75}\cdot6=4$；圆心有向距离$\frac{0.25}{0.25-1}\cdot6=\frac{0.25}{-0.75}\cdot6=-2$，故圆心$O_2(-2,0)$. 轨迹为以$(-2,0)$为圆心、$4$为半径的圆.\n
			\par\textbf{理由：} 套用阿波罗尼斯圆结论.\n
		\item[\textbf{第二步（面积表示）：}]
			底边$AB=6$在$x$轴上，高为$P$的纵坐标绝对值$|y|$，面积$S=\frac12\cdot6\cdot|y|=3|y|$.\n
			\par\textbf{理由：} 三角形面积公式.\n
		\item[\textbf{第三步（求最值）：}]
			圆上点的纵坐标绝对值最大等于半径$4$，故$S_{\max}=3\times4=12$.\n
			\par\textbf{理由：} 圆上点到$x$轴的最大距离为半径.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\triangle ABP$面积的最大值为$12$.}\n\n
\end{examplebox}
