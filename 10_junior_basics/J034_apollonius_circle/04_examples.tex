\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知点 $A(0,0),B(4,0)$，动点 $P$ 满足 $PA=2PB$，求点 $P$ 的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算参数）：}]
		      $d=AB=4$，$k=2$。
		      \par\textbf{理由：} 确定基本量。
		\item[\textbf{第二步（求圆心半径）：}]
		      半径
		      \[
			      R=\frac{k}{|1-k^{2}|}d=\frac{2}{3}\cdot 4=\frac{8}{3};
		      \]
		      圆心相对 $A$ 的有向距离
		      \[
			      \frac{k^{2}}{k^{2}-1}d=\frac{4}{3}\cdot 4=\frac{16}{3},
		      \]
		      故圆心为 $\left(\frac{16}{3},0\right)$。
		      \par\textbf{理由：} 直接套用阿波罗尼斯圆结论。
		\item[\textbf{第三步（写方程）：}]
		      轨迹方程为
		      \[
			      \left(x-\frac{16}{3}\right)^2+y^2=\frac{64}{9}.
		      \]
		      \par\textbf{理由：} 圆的标准方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}} \textbf{轨迹是以 $\left(\frac{16}{3},0\right)$ 为圆心、$\frac{8}{3}$ 为半径的圆。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知点 $A(-1,0),B(1,0)$，动点 $Q$ 满足 $QA=3QB$，求点 $Q$ 的轨迹方程，并判断原点 $O(0,0)$ 是否在该圆上。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算参数）：}]
		      $d=AB=2$，$k=3$。
		      \par\textbf{理由：} 代入公式前先求距离和比值。
		\item[\textbf{第二步（求圆心半径）：}]
		      半径
		      \[
			      R=\frac{3}{|1-9|}\cdot 2=\frac{3}{4};
		      \]
		      圆心相对 $A$ 的有向距离
		      \[
			      \frac{9}{9-1}\cdot 2=\frac{9}{4}.
		      \]
		      因为 $A(-1,0)$，所以圆心为 $\left(\frac{5}{4},0\right)$。
		      \par\textbf{理由：} 利用圆心有向距离公式。
		\item[\textbf{第三步（检验原点）：}]
		      圆方程为
		      \[
			      \left(x-\frac{5}{4}\right)^2+y^2=\frac{9}{16}.
		      \]
		      代入 $(0,0)$ 得 $\dfrac{25}{16}\neq \dfrac{9}{16}$，故原点不在圆上。
		      \par\textbf{理由：} 点满足方程才在圆上。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}} \textbf{轨迹方程为 $\left(x-\frac{5}{4}\right)^2+y^2=\frac{9}{16}$，原点不在该圆上。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知点 $A(0,0),B(6,0)$，动点 $P$ 满足 $PA=\dfrac{1}{2}PB$，求 $\triangle ABP$ 面积的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（得轨迹）：}]
		      $d=6$，$k=\dfrac{1}{2}$。半径
		      \[
			      R=\frac{\frac{1}{2}}{\left|1-\frac{1}{4}\right|}\cdot 6=4;
		      \]
		      圆心相对 $A$ 的有向距离
		      \[
			      \frac{\frac{1}{4}}{\frac{1}{4}-1}\cdot 6=-2,
		      \]
		      故轨迹为以 $(-2,0)$ 为圆心、$4$ 为半径的圆。
		      \par\textbf{理由：} 套用阿波罗尼斯圆结论。
		\item[\textbf{第二步（面积表示）：}]
		      底边 $AB=6$ 在 $x$ 轴上，高为 $P$ 的纵坐标绝对值 $|y|$，面积
		      \[
			      S=\frac{1}{2}\cdot 6\cdot |y|=3|y|.
		      \]
		      \par\textbf{理由：} 三角形面积公式。
		\item[\textbf{第三步（求最值）：}]
		      圆上点的纵坐标绝对值最大等于半径 $4$，故
		      \[
			      S_{\max}=3\times 4=12.
		      \]
		      \par\textbf{理由：} 圆上点到 $x$ 轴的最大距离为半径。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\triangle ABP$ 面积的最大值为 $12$。}
