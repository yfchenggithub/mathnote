\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 在 Rt$\triangle ABC$ 中，$\angle C=90^\circ$，$AC=3$，$BC=4$，求 $AB$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定三角形）：}]
			已知 $\angle C=90^\circ$，所以 $\triangle ABC$ 为直角三角形，$AB$ 为斜边。
			\par\textbf{理由：} 直角三角形的斜边是直角所对的边。
		\item[\textbf{第二步（代入公式）：}]
			由勾股定理 $AB^2 = AC^2+BC^2$。代入 $AC=3, BC=4$ 得 $3^2+4^2=25$，所以 $AB^2=25$。
			\par\textbf{理由：} 勾股定理公式。
		\item[\textbf{第三步（求算结果）：}]
			由 $AB^2=25$ 开方得 $AB=5$（边长取正）。
			\par\textbf{理由：} 边长非负。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AB=5$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 在弦图构造中，大正方形的面积为 $169$，中间小正方形的面积为 $49$，求直角三角形两条直角边长 $a,b$（$a\ge b>0$）。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（面积开方）：}]
			由大正方形面积 $c^2=169$，得 $c=13$；由小正方形面积 $(a-b)^2=49$，得 $a-b=7$（取正）。
			\par\textbf{理由：} 正方形边长等于面积的正平方根。
		\item[\textbf{第二步（建立方程组）：}]
			由勾股定理及大正方形面积得 $a^2+b^2=169$，由小正方形面积得 $(a-b)^2=49$，展开得 $a^2-2ab+b^2=49$。两式相减得 $2ab=120$，即 $ab=60$。
			\par\textbf{理由：} 完全平方公式与已知等式的关系。
		\item[\textbf{第三步（解方程）：}]
			联立 $a-b=7$ 与 $ab=60$，解得 $a=12,b=5$（或 $a=5,b=12$，但 $a\ge b$，故取 $a=12,b=5$）。
			\par\textbf{理由：} 解一元二次方程组或试根法。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直角边长为 $12$ 和 $5$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 在弦图模型中，大正方形的边长为 $13$，小正方形的面积比大正方形的面积少 $48$，求直角边 $a$ 与 $b$ 的乘积 $ab$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求小正方形面积）：}]
			大正方形边长 $13$，面积 $c^2=169$；小正方形面积比大正方形少 $48$，故为 $121$，得边长 $d=11$，即 $a-b=11$（$a\ge b$）。
			\par\textbf{理由：} 面积差条件；正方形面积公式。
		\item[\textbf{第二步（应用弦图关系）：}]
			弦图面积恒等式 $c^2 = 2ab + (a-b)^2$，代入 $c^2=169$，$(a-b)^2=121$，得 $169 = 2ab + 121$。
			\par\textbf{理由：} 弦图模型的面积关系式。
		\item[\textbf{第三步（解出乘积）：}]
			移项得 $2ab = 48$，故 $ab = 24$。
			\par\textbf{理由：} 一元一次方程求解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$ab=24$。}
