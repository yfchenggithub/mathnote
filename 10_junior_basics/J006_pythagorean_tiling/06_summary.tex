\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{弦图模型揭示了直角三角形三边的数量关系：$a^2+b^2=c^2$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（直角存在）：} 存在直角三角形，两直角边长 $a,b$（$a\ge b>0$），斜边长 $c$。
		\item \textbf{条件 2（图形构造）：} 能够将此直角三角形全等复制四个，并按弦图的旋转位置摆放，形成边长为 $c$ 的大正方形和边长为 $a-b$ 的小正方形。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（边长归属）：} 弦图中的大正方形是斜边构成的，边长为 $c$，不是 $a+b$。
		\item \textbf{检查项（面积系数）：} 四个直角三角形总面积是 $2ab$，不是 $4ab$。
	\end{itemize}

\end{summarybox}
