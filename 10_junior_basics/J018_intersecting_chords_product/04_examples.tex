\begin{examplebox}

	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在圆 $O$ 中，弦 $AB$ 与 $CD$ 交于点 $P$. 已知 $PA=2$, $PB=6$, $PC=3$, 求 $PD$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写定理关系）：}] 由相交弦定理得 $PA\cdot PB = PC\cdot PD$.
			\par\textbf{理由：} 定理直接给出乘积等式.
		\item[\textbf{第二步（代入数据）：}] $2\times 6 = 3\times PD$ 即 $12 = 3PD$.
			\par\textbf{理由：} 代入已知数值.
		\item[\textbf{第三步（求解未知量）：}] 两边除以 $3$ 得 $PD = 4$.
			\par\textbf{理由：} 等式性质.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PD=4$}.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	圆内两弦 $AB$, $CD$ 交于 $P$, 且 $AB=13$, $PA=5$, $PC=4$, 求 $PD$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求另一段）：}] 因为 $P$ 在线段 $AB$ 上，$PB = AB - PA = 13 - 5 = 8$.
			\par\textbf{理由：} $AB$ 被 $P$ 分成两段之和.
		\item[\textbf{第二步（列乘积等式）：}] 由相交弦定理 $PA\cdot PB = PC\cdot PD$ 得 $5\times 8 = 4\times PD$.
			\par\textbf{理由：} 代入已知量.
		\item[\textbf{第三步（解出 $PD$）：}] $40 = 4PD$, 所以 $PD = 10$.
			\par\textbf{理由：} 等式性质.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PD=10$}.

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	圆内两弦 $AB$, $CD$ 交于 $P$. 已知 $PA = x+2$, $PB = x-2$, $PC = x$, $PD = 8$（其中 $x>2$），求 $x$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列方程）：}] 由相交弦定理得 $(x+2)(x-2) = x \cdot 8$.
			\par\textbf{理由：} $PA\cdot PB = PC\cdot PD$.
		\item[\textbf{第二步（展开化简）：}] 左边用平方差公式 $x^2 - 4 = 8x$, 移项得 $x^2 - 8x - 4 = 0$.
			\par\textbf{理由：} 代数展开与移项.
		\item[\textbf{第三步（解二次方程）：}] 由求根公式
			\[
				\begin{aligned}
					x &= \frac{8 \pm \sqrt{64 + 16}}{2} \\
					&= \frac{8 \pm \sqrt{80}}{2} \\
					&= 4 \pm 2\sqrt{5}.
			\end{aligned}
			\]
			因 $x>2$，取 $x = 4 + 2\sqrt{5}$.
			\par\textbf{理由：} 二次方程求解，并考虑条件舍去负根 $4-2\sqrt{5} \approx 4 - 4.47 = -0.47$（不满足 $x>2$）.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x = 4 + 2\sqrt{5}$}.
