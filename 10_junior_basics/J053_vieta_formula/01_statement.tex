\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（使用前提）：}}]
			实系数一元二次方程 $ax^2+bx+c=0$（$a\neq0$），且判别式 $\Delta=b^2-4ac\geq0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（根与系数）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两根之和与积可直接由系数得到，无需解方程。
			\[
				x_1+x_2=-\frac{b}{a}
			\]
			\[
				x_1x_2=\frac{c}{a}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（逆用构造）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知两根之和与两根之积可构造原二次方程。
			\[
				x^2-(x_1+x_2)x+x_1x_2=0
			\]
	\end{description}

\end{statementbox}
