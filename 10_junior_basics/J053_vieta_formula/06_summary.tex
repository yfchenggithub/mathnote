\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{韦达定理建立了二次方程两根之和、两根之积与系数之间的直接代数关系，广泛用于求对称式的值、构造方程等。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准形式）：} 方程必须为一元二次方程 $ax^2+bx+c=0$，且二次项系数 $a\neq0$。
		\item \textbf{条件 2（实根存在）：} 判别式 $\Delta=b^2-4ac\geq0$，保证两根为实数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（判别式验证）：} 务必在 $\Delta\geq0$ 前提下使用韦达定理，否则会得到无意义的结果（复数根高中不要求）。
		\item \textbf{检查项（符号与分母）：} 求和时不要漏掉负号，求积时不要忘记除以 $a$（当 $a\neq1$ 时）。
	\end{itemize}

\end{summarybox}
