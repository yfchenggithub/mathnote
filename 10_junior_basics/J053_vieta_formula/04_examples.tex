\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知二次方程 $2x^2-4x+1=0$，求它的两根之和与两根之积。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判别确认）：}]
			确认 $a=2$, $b=-4$, $c=1$，计算判别式 $\Delta=(-4)^2-4\cdot2\cdot1=8$，可知 $\Delta>0$，所以有实根。
			\par\textbf{理由：} 韦达定理只适用于实根情形，必须先验证 $\Delta\geq0$。
		\item[\textbf{第二步（套用公式）：}]
			直接由韦达定理：
			\[
				\begin{aligned}
					x_1+x_2 &=-\frac{b}{a} \\ &=-\frac{-4}{2} \\ &=2,
			\end{aligned}
			\]
			\[
				\begin{aligned}
					x_1x_2 &=\frac{c}{a} \\ &=\frac{1}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入系数公式即可。
		\item[\textbf{第三步（结果答写）：}]
			因此两根之和为 $2$，两根之积为 $\dfrac{1}{2}$。
			\par\textbf{理由：} 计算无误。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{满足条件时，直接套用公式可迅速得到和与积，避免解方程。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	关于 $x$ 的方程 $x^2+mx+6=0$ 的一个根为 $2$，求另一个根和 $m$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设根条件）：}]
			设另一根为 $x_2$。由韦达定理，根与系数满足 $x_1x_2=6$，$x_1+x_2=-m$，其中 $x_1=2$。
			\par\textbf{理由：} 已知一根，可直接使用韦达定理中的积关系。
		\item[\textbf{第二步（利用积）：}]
			由 $x_1x_2=6$ 得 $2\cdot x_2=6$，所以 $x_2=3$。
			\par\textbf{理由：} 积公式直接解出另一根。
		\item[\textbf{第三步（利用和）：}]
			由 $x_1+x_2 = -m$ 得 $2+3 = -m$，所以 $m = -5$。
			\par\textbf{理由：} 和公式建立联系，解出参数。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{已知一根时，先利用积求另一根，再利用和求参数。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	若 $x_1,x_2$ 是方程 $x^2-3x+1=0$ 的两根，求 $x_1^2+x_2^2$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写和积）：}]
			由韦达定理得 $x_1+x_2=3$，$x_1x_2=1$。
			\par\textbf{理由：} 直接套用公式。
		\item[\textbf{第二步（恒等变形）：}]
			将目标式变形：
			\[
				x_1^2+x_2^2 = (x_1+x_2)^2-2x_1x_2.
			\]
			\par\textbf{理由：} 利用完全平方公式将平方和转化为和与积。
		\item[\textbf{第三步（代入计算）：}]
			代入 $x_1+x_2=3$, $x_1x_2=1$ 得 $3^2-2\cdot1=7$。
			\par\textbf{理由：} 直接数值运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{利用韦达定理将对称式转化为和与积的代数式，无需解出具体根。}
\end{examplebox}
