\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（平行截取点）：}}] 点 $D$ 在 $AB$ 上，点 $E$ 在 $AC$ 上，且 $DE\parallel BC$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（相似三角形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平行线截三角形两边，所得小三角形与原三角形形状相同，对应边成比例。
			\[
				\frac{AD}{AB} = \frac{AE}{AC} = \frac{DE}{BC} = k,\quad \triangle ADE\sim\triangle ABC
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（面积与线段比）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有对应线段（高、中线、角平分线）长度比均为 $k$；面积比是 $k^2$。
			\[
				\frac{S_{\triangle ADE}}{S_{\triangle ABC}} = k^2 = \left(\frac{AD}{AB}\right)^2
			\]
	\end{description}

\end{statementbox}
