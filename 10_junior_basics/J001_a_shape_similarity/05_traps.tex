\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（缺少平行条件）}}] 误以为只要点 $D$、$E$ 分别在 $AB$、$AC$ 上，$\triangle ADE$ 就与 $\triangle ABC$ 相似。
		\item[\textbf{\textcolor{CTrap}{正确理解（需平行前提）：}}] 必须满足 $DE\parallel BC$ 才能保证 $\triangle ADE\sim\triangle ABC$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略关键条件）：}}] 忽略平行线是判断相似的关键条件。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（面积比误用一次方）}}] 认为面积比等于边长比（$k$），即 $S_{\triangle ADE}:S_{\triangle ABC}=AD:AB$。
		\item[\textbf{\textcolor{CTrap}{正确理解（面积比平方关系）：}}] $S_{\triangle ADE}:S_{\triangle ABC}=k^2$，其中 $k$ 为相似比。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆长度与面积缩放）：}}] 面积是二维量，长度是一维量，放大倍数为平方关系。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（对应边对应错误）}}] 写比例时对应边弄错，如 $\frac{AD}{DB}=\frac{DE}{BC}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（严格对应顶点）：}}] 比例式的分子是小三角形边，分母是大三角形对应边，如 $\frac{AD}{AB}=\frac{DE}{BC}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（未按对应顶点写比例）：}}] 没有明确对应关系，盲目套比例。
	\end{description}
\end{trapbox}
