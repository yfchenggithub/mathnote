\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	由平行线得出两组同位角相等，进而证三角形相似；再利用相似性质得到对应边比例和面积比。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（证角等）：}] 因为 $DE\parallel BC$，所以 $\angle ADE = \angle ABC$，$\angle AED = \angle ACB$，又 $\angle A$ 为公共角，因此 $\triangle ADE \sim \triangle ABC$（AA相似）。
			\[
				\angle A = \angle A,\quad \angle ADE = \angle ABC,\quad \angle AED = \angle ACB
			\]
			\par\textbf{理由：}平行线同位角相等；两角对应相等的三角形相似。

		\item[\textbf{步骤二（定比例）：}] 由 $\triangle ADE \sim \triangle ABC$，得对应边成比例：
			\[
				\frac{AD}{AB} = \frac{AE}{AC} = \frac{DE}{BC} = k
			\]
			\par\textbf{理由：}相似三角形定义。

		\item[\textbf{步骤三（面积比）：}] 根据相似三角形性质，面积比等于相似比的平方：
			\[
				\frac{S_{\triangle ADE}}{S_{\triangle ABC}} = k^2 = \left(\frac{AD}{AB}\right)^2
			\]
			\par\textbf{理由：}相似三角形的面积比等于相似比的平方（可由对应高比例推导：$S_{\triangle ADE}=\frac12 DE\cdot h_1$，$S_{\triangle ABC}=\frac12 BC\cdot h_2$，且$\frac{h_1}{h_2}=k$，故面积比$=k\cdot k=k^2$）。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{核心要点：}$DE\parallel BC \Rightarrow$ 小三角形$\sim$大三角形 $\Rightarrow$ 线段成比例、面积平方比。
\end{proofbox}
