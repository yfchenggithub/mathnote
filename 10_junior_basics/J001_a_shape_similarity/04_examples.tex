	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$DE\parallel BC$，$AD=2$，$DB=3$，求 $DE:BC$ 及 $S_{\triangle ADE}:S_{\triangle ABC}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}] 由 $DE\parallel BC$ 得 $\triangle ADE\sim\triangle ABC$。
			\par\textbf{理由：}平行截相似。
		\item[\textbf{第二步（求相似比）：}] 相似比 $k=\frac{AD}{AB}=\frac{2}{2+3}=\frac{2}{5}$。
			\par\textbf{理由：}$AB=AD+DB$。
		\item[\textbf{第三步（得比例）：}] $DE:BC=k=2:5$，面积比 $S_{\triangle ADE}:S_{\triangle ABC}=k^2=4:25$。
			\par\textbf{理由：}相似三角形对应边成比例，面积比等于相似比的平方。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$DE:BC=2:5$，$S_{\triangle ADE}:S_{\triangle ABC}=4:25$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$DE\parallel BC$，若 $S_{\triangle ADE}:S_{BCED}=4:5$，求 $AD:AB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（转化面积比）：}] $S_{BCED}=S_{\triangle ABC}-S_{\triangle ADE}$，所以 $S_{\triangle ADE}:S_{\triangle ABC}=4:(4+5)=4:9$。
			\par\textbf{理由：}整体减部分。
		\item[\textbf{第二步（面积比求相似比）：}] 由面积比等于相似比的平方，得 $k^2=\frac{4}{9}$，故 $k=\frac{2}{3}$（取正）。
			\par\textbf{理由：}相似三角形性质。
		\item[\textbf{第三步（得比例）：}] $AD:AB=k=2:3$。
			\par\textbf{理由：}对应边比等于相似比。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AD:AB=2:3$。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，在 $\triangle ABC$ 中，$DE\parallel BC$，$AD:DB=2:3$，$\triangle ADE$ 的面积为 $8$，求四边形 $BCED$ 的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求相似比）：}] $AD:DB=2:3$，故 $AD:AB=2:5$，即 $k=\frac{2}{5}$。
			\par\textbf{理由：}比例加法。
		\item[\textbf{第二步（面积比求大三角形面积）：}] 由 $S_{\triangle ADE}:S_{\triangle ABC}=k^2=\frac{4}{25}$，设 $S_{\triangle ADE}=8$，则 $S_{\triangle ABC}=8\times\frac{25}{4}=50$。
			\par\textbf{理由：}解比例。
		\item[\textbf{第三步（求四边形面积）：}] $S_{BCED}=S_{\triangle ABC}-S_{\triangle ADE}=50-8=42$。
			\par\textbf{理由：}面积相减。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{四边形 $BCED$ 面积为 $42$。}
