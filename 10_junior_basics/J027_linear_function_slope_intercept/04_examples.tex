\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一次函数图像经过点 $A(1,3)$ 和点 $B(3,7)$，求该一次函数的解析式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求斜率）：}] 设 $A(x_1,y_1)=(1,3)$, $B(x_2,y_2)=(3,7)$，代入斜率公式：
			\[
				k = \frac{y_2 - y_1}{x_2 - x_1} = \frac{7-3}{3-1} = \frac{4}{2} = 2.
			\]
			\par\textbf{理由：} 直线上两点确定唯一的斜率。

		\item[\textbf{第二步（求截距）：}] 设解析式为 $y = kx + b$，将 $k=2$ 及点 $A(1,3)$ 代入：
			\[
				3 = 2\cdot 1 + b \;\Longrightarrow\; b = 1.
			\]
			\par\textbf{理由：} 点的坐标满足方程。

		\item[\textbf{第三步（写解析式）：}] 所以一次函数解析式为 $\boxed{y = 2x + 1}$。
			\par\textbf{理由：} 代入 $k=2,b=1$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y=2x+1$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l$ 的斜率为 $k=-3$，且与 $y$ 轴交于点 $(0,4)$，求 $l$ 的方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定截距）：}] 直线与 $y$ 轴交点的纵坐标即为纵截距 $b$，故 $b = 4$。
			\par\textbf{理由：} $b$ 的定义。

		\item[\textbf{第二步（代入解析式）：}] 将 $k=-3$，$b=4$ 代入 $y = kx + b$ 得
			\[
				y = -3x + 4.
			\]
			\par\textbf{理由：} 一次函数的标准形式。

		\item[\textbf{第三步（写出答案）：}] 所求直线方程为 $\boxed{y = -3x + 4}$。
			\par\textbf{理由：} 已确定 $k,b$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y=-3x+4$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l$ 平行于直线 $y = 2x + 1$，且过点 $P(2,1)$，求 $l$ 的方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定斜率）：}] 平行直线的斜率相等，所以 $l$ 的斜率 $k = 2$。
			\par\textbf{理由：} 两直线平行 $\Leftrightarrow k_1 = k_2$。

		\item[\textbf{第二步（点斜式求方程）：}] 利用点斜式 $y - y_0 = k(x - x_0)$，代入 $k=2$, $P(2,1)$：
			\[
				y - 1 = 2(x - 2).
			\]
			\par\textbf{理由：} 已知一点和斜率可直接写方程。

		\item[\textbf{第三步（化为斜截式）：}] 整理得
			\[
				y = 2x - 3.
			\]
			\par\textbf{理由：} 展开括号并移项。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y = 2x - 3$}。
\end{examplebox}
