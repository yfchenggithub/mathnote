\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（线段和差待证）：}}]
			已知三条线段 $L,a,b$，需要证明 $a+b=L$ 或类似的线段和差关系。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（截长法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在长线段 $L$ 上截取一段等于 $a$，再通过全等证明剩余部分等于 $b$。
			\[
				L_1 = a,\quad L_2 = b \;\Longrightarrow\; a+b = L.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（补短法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将短线段 $a$ 延长至 $AC=a+b$，再通过全等证明 $AC$ 等于长线段 $L$。
			\[
				AC = a+b \;\Longrightarrow\; AC = L.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（角平分线模型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在角平分线模型中常使用截长补短法构造全等，完成线段和差证明。
	\end{description}

\end{statementbox}
