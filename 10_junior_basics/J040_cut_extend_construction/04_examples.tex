\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$\angle B = 80^\circ$，$\angle C = 40^\circ$，$AD$ 平分 $\angle BAC$ 交 $BC$ 于 $D$。求证：$AB + BD = AC$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（截长构造）：}]
			在 $AC$ 上截取点 $E$ 使 $AE = AB$，连接 $DE$。
			\par\textbf{理由：} 截长法作辅助线，将 $AB$ 转移到 $AC$ 上，剩余 $EC$ 需等于 $BD$。
		\item[\textbf{第二步（证全等）：}]
			证明 $\triangle ABD \cong \triangle AED$。因为 $AE=AB$，$\angle BAD = \angle EAD$（$AD$ 平分），$AD$ 公共，所以全等（SAS）。
			\par\textbf{理由：} 由全等得到 $BD=DE$，$\angle AED = \angle B = 80^\circ$。
		\item[\textbf{第三步（角度推算）：}]
			$\angle C = 40^\circ$，则 $\angle DEC = 180^\circ - 80^\circ = 100^\circ$，$\angle EDC = 180^\circ - 100^\circ - 40^\circ = 40^\circ$，所以 $DE = EC$，即 $EC = BD$。因此 $AC = AE + EC = AB + BD$。
			\par\textbf{理由：} 三角形内角和与等角对等边。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AB + BD = AC$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AD$ 平分 $\angle BAC$，$\angle B = 2\angle C$。求证：$AB + BD = AC$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（补短构造）：}]
			延长 $AB$ 至点 $E$，使 $BE = BD$，连接 $DE$。
			\par\textbf{理由：} 补短法，构造 $AE = AB+BD$，只需证 $AE = AC$。
		\item[\textbf{第二步（角度导出）：}]
			由 $BE=BD$ 得 $\angle E = \angle BDE = \frac12\angle ABC = \angle C$。
			\par\textbf{理由：} 等腰三角形性质和 $\angle B =2\angle C$。
		\item[\textbf{第三步（证全等）：}]
			在 $\triangle ADE$ 和 $\triangle ADC$ 中，$\angle E = \angle C$，$\angle BAD = \angle CAD$（$AD$ 平分），$AD$ 公共，所以 $\triangle ADE \cong \triangle ADC$（AAS），得 $AE = AC$。
			\par\textbf{理由：} AAS全等。
		\item[\textbf{第四步（等量代换）：}]
			由 $AE = AB+BE = AB+BD = AC$，即原式成立。
			\par\textbf{理由：} 等量代换。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AB + BD = AC$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在正方形 $ABCD$ 中，$E$、$F$ 分别在边 $BC$、$CD$ 上，$\angle EAF = 45^\circ$。求证：$EF = BE + DF$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（补短构造）：}]
			延长 $CB$ 至点 $G$，使 $BG = DF$，连接 $AG$。
			\par\textbf{理由：} 补短法，将 $DF$ 转移到 $BG$，需证 $EG = EF$。
		\item[\textbf{第二步（证全等一）：}]
			证明 $\triangle ABG \cong \triangle ADF$。由 $AB=AD$，$BG=DF$，$\angle ABG = \angle ADF = 90^\circ$，得全等（SAS）。所以 $AG=AF$，$\angle GAB = \angle FAD$。
			\par\textbf{理由：} SAS全等。
		\item[\textbf{第三步（证全等二）：}]
			证明 $\triangle AEG \cong \triangle AEF$。因为 $\angle GAE = \angle GAB + \angle BAE = \angle FAD + \angle BAE = 45^\circ = \angle EAF$，又 $AE=AE$，$AG=AF$，所以全等（SAS），得 $EG=EF$。
			\par\textbf{理由：} SAS全等。
		\item[\textbf{第四步（等量代换）：}]
			$EF = EG = BE + BG = BE + DF$，原式得证。
			\par\textbf{理由：} 等量代换。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$EF = BE + DF$}。
\end{examplebox}
