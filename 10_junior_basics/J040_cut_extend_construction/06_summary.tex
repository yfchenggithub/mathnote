\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{截长补短法是通过构造全等三角形，将线段和差关系拆分为几段相等关系，再等量代换证明目标等式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件一（线段和差待证）：} 需证明 $a+b=L$ 或类似和差关系。
		\item \textbf{条件二（全等可构造）：} 图形中存在或可以添加辅助线构造全等三角形。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略全等步骤）：} 不能跳过全等直接得出相等。
		\item \textbf{检查项（全等对应正确）：} 写全等时注意对应顶点顺序，避免对应错误。
	\end{itemize}

\end{summarybox}
