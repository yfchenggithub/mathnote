\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（截长法误用）}}]
			误认为截取后剩余部分自动等于另一条短线段，不证全等直接断言。
		\item[\textbf{\textcolor{CTrap}{正确理解（需证全等）：}}]
			截取后只能保证 $L_1=a$，$L_2$ 必须通过全等证明才可与 $b$ 相等。
		\item[\textbf{\textcolor{CTrap}{错因分析（略证跳步）：}}]
			学生跳步，默认 $L_2=b$，导致逻辑链条残缺，可能失分。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（补短方向错）}}]
			补短时延长方向任意，造成构造的三角形无法与原图形建立全等关系。
		\item[\textbf{\textcolor{CTrap}{正确理解（延时应关联）：}}]
			延长方向应使新线段与已知边、角形成可证全等的结构，通常沿与角平分线或等边相关的方向。
		\item[\textbf{\textcolor{CTrap}{错因分析（盲目延长）：}}]
			未分析图形，随意延长导致全等条件缺失，辅助线失效。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（全等对应乱）}}]
			证明全等时未注意对应顶点书写顺序，错误使用SSA或对应边角位置不对。
		\item[\textbf{\textcolor{CTrap}{正确理解（对应要精确）：}}]
			严格按照SAS、ASA、AAS、SSS，写好对应顶点和边角顺序。
		\item[\textbf{\textcolor{CTrap}{错因分析（对应混淆）：}}]
			学生习惯写“三角形ABC全等于三角形DEF”，但对应关系不清导致等式无效。
	\end{description}

\end{trapbox}
