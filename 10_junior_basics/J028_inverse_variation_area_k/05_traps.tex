\begin{trapbox}
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（面积未取绝对值）}}]
			已知反比例函数 $y = \dfrac{k}{x}$，计算矩形面积时直接写 $S = k$，没有加绝对值。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（面积恒正）：}}]
			矩形面积 $S = |k|$，无论 $k>0$ 还是 $k<0$，面积都是正数。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（符号混淆）：}}]
			学生误将 $k$ 的代数符号当作面积符号，忽略了面积的非负性。\n
	\end{description}
	\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（三角形面积漏除 $2$）}}]
			计算直角三角形面积时，直接得到 $|k|$，未乘以 $\frac{1}{2}$。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（面积减半）：}}]
			直角三角形面积是矩形面积的一半，即 $S_{\triangle} = \frac{1}{2}|k|$。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（公式记忆错误）：}}]
			混淆三角形面积与矩形面积，或不清楚三角形面积需乘以 $\frac{1}{2}$。\n
	\end{description}
	\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（由面积求 $k$ 忽略符号）}}]
			已知矩形面积 $S=6$，直接得 $k=6$，不考虑 $k$ 的正负。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（符号由象限定）：}}]
			由 $S=|k|$ 得 $|k|=6$，需根据双曲线所在象限确定 $k=6$ 或 $k=-6$。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视双曲线分布）：}}]
			学生认为面积 $S$ 等于 $k$，忘记 $k$ 可以是负数，且面积不能直接决定 $k$ 的符号。\n
	\end{description}
	\n
\end{trapbox}
