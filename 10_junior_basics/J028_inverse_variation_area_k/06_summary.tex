\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{过反比例函数 $y=\dfrac{k}{x}$（$k\neq 0$）图象上任一点 $P$ 作两坐标轴的垂线，所得矩形面积为 $|k|$，对应直角三角形面积为 $\dfrac{1}{2}|k|$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（点在图象）：} 点 $P$ 必须在 $y=\dfrac{k}{x}$ 的图象上，即 $x_0y_0=k$。
		\item \textbf{条件 2（作垂线）：} 过点 $P$ 分别作 $x$ 轴、$y$ 轴的垂线，得到矩形或直角三角形。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（面积取绝对值）：} 面积恒为正，计算时务必使用 $|k|$，不要直接用 $k$。
		\item \textbf{检查项（三角形面积减半）：} 三角形面积是矩形面积的一半，不要忘记乘以 $\dfrac{1}{2}$。
	\end{itemize}
\end{summarybox}
