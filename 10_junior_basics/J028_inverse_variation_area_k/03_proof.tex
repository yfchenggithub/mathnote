\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n要证明矩形面积等于 $|k|$，关键利用点 $P$ 的坐标满足函数解析式，从而 $x_0 y_0 = k$。再通过直角坐标系中长度公式计算面积，最后取绝对值。\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（坐标关系）：}]
			由点 $P(x_0, y_0)$ 在反比例函数 $y = \dfrac{k}{x}$ 的图象上，有\n
			\[
				\n y_0 = \frac{k}{x_0} \quad \Rightarrow \quad x_0 y_0 = k.\n
			\]
			\n
			\par\textbf{理由：} 代入解析式得 $x_0 \cdot \dfrac{k}{x_0} = k$，所以 $x_0 y_0 = k$（$k$ 可为正或负）。\n\n
		\item[\textbf{步骤二（矩形面积表达）：}]
			矩形 $OAPB$ 的顶点为 $O(0,0)$，$A(x_0,0)$，$B(0,y_0)$，$P(x_0,y_0)$，其面积\n
			\[
				\n S = OA \cdot OB = |x_0| \cdot |y_0|.\n
			\]
			\n
			\par\textbf{理由：} $OA$ 是 $P$ 的横坐标的绝对值，$OB$ 是纵坐标的绝对值。\n\n
		\item[\textbf{步骤三（矩形面积化简）：}]
			由绝对值性质 $|x_0||y_0| = |x_0 y_0|$，代入步骤一的结果得\n
			\[
				\n S = |x_0 y_0| = |k|.\n
			\]
			\n
			\par\textbf{理由：} $x_0 y_0 = k$，故 $|x_0 y_0| = |k|$。矩形面积证毕。\n\n
		\item[\textbf{步骤四（三角形面积）：}]
			直角三角形 $OAP$（或 $OBP$）的面积是矩形面积的一半：\n
			\[
				\n S_{\triangle} = \frac{1}{2} \cdot OA \cdot AP = \frac{1}{2} |x_0| \cdot |y_0| = \frac{1}{2} |k|.\n
			\]
			\n
			\par\textbf{理由：} 矩形的对角线将矩形分为两个全等的直角三角形，每个面积为矩形的一半。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n综上，反比例函数 $y = \dfrac{k}{x}$（$k \neq 0$）的图象上任一点 $P$，过 $P$ 作坐标轴的垂线围成的矩形面积为 $|k|$，对应的直角三角形面积为 $\dfrac{1}{2}|k|$。\n
\end{proofbox}
