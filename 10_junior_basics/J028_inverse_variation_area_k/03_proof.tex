\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	要证明矩形面积等于 $|k|$，关键是利用点 $P$ 的坐标满足函数解析式，从而得到 $x_0y_0=k$。再用直角坐标系中的长度关系计算面积，最后取绝对值。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（坐标关系）：}]
		      由点 $P(x_0,y_0)$ 在反比例函数 $y=\dfrac{k}{x}$ 的图象上，有
		      \[
			      y_0=\frac{k}{x_0}\quad\Rightarrow\quad x_0y_0=k.
		      \]
		      \par\textbf{理由：} 代入解析式即可得到 $x_0\cdot \dfrac{k}{x_0}=k$，所以 $x_0y_0=k$（$k$ 可为正或负）。

		\item[\textbf{步骤二（矩形面积表达）：}]
		      设 $A(x_0,0)$，$B(0,y_0)$，则矩形 $OAPB$ 的顶点为 $O(0,0)$、$A(x_0,0)$、$P(x_0,y_0)$、$B(0,y_0)$，其面积为
		      \[
			      S=OA\cdot OB=|x_0|\cdot |y_0|.
		      \]
		      \par\textbf{理由：} $OA$ 是 $P$ 的横坐标的绝对值，$OB$ 是 $P$ 的纵坐标的绝对值。

		\item[\textbf{步骤三（矩形面积化简）：}]
		      由绝对值性质 $|x_0||y_0|=|x_0y_0|$，代入步骤一的结果得
		      \[
			      S=|x_0y_0|=|k|.
		      \]
		      \par\textbf{理由：} 因为 $x_0y_0=k$，所以 $|x_0y_0|=|k|$。矩形面积证毕。

		\item[\textbf{步骤四（三角形面积）：}]
		      直角三角形 $OAP$（或 $OBP$）的面积是矩形面积的一半：
		      \[
			      S_{\triangle}=\frac{1}{2}\cdot OA\cdot AP=\frac{1}{2}|x_0|\cdot |y_0|=\frac{1}{2}|k|.
		      \]
		      \par\textbf{理由：} 矩形的对角线将矩形分成两个全等的直角三角形，每个面积为矩形面积的一半。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	综上，反比例函数 $y=\dfrac{k}{x}$（$k\neq 0$）的图象上任一点 $P$，过 $P$ 作坐标轴的垂线围成的矩形面积为 $|k|$，对应的直角三角形面积为 $\dfrac{1}{2}|k|$。
\end{proofbox}
