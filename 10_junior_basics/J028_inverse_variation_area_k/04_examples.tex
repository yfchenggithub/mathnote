\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知反比例函数 $y = \dfrac{k}{x}$ 的图象经过点 $A(3,2)$。过点 $A$ 分别作 $x$ 轴、$y$ 轴的垂线，垂足为 $B$、$C$。求矩形 $OBAC$ 的面积和直角三角形 $OAB$ 的面积。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求比例系数）：}]
			将 $A(3,2)$ 代入 $y = \dfrac{k}{x}$，得 $2 = \dfrac{k}{3}$，所以 $k = 6$。\n
			\par\textbf{理由：} 点在图象上，坐标满足解析式。\n
		\item[\textbf{第二步（矩形面积）：}]
			利用系数 $k$ 的几何意义，矩形 $OBAC$ 的面积 $S = |k| = 6$。\n
			\par\textbf{理由：} 矩形面积等于 $|k|$。\n
		\item[\textbf{第三步（三角形面积）：}]
			直角三角形 $OAB$ 的面积 $S_{\triangle} = \dfrac{1}{2}|k| = 3$。\n
			\par\textbf{理由：} 直角三角形面积是矩形面积的一半。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{矩形面积 $S = 6$，三角形面积 $S_{\triangle} = 3$。}\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知反比例函数 $y = \dfrac{k}{x}$ 的图象上有一点 $P$，过点 $P$ 作 $x$ 轴的垂线，垂足为 $A$。若 $\triangle OAP$ 的面积为 $5$，求 $k$ 的值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（设出坐标）：}]
			设点 $P(x_0, y_0)$，则 $A(x_0,0)$，$OA = |x_0|$，$AP = |y_0|$。\n
			\par\textbf{理由：} 垂足 $A$ 的横坐标与 $P$ 相同，纵坐标为 $0$。\n
		\item[\textbf{第二步（面积表示）：}]
			$\triangle OAP$ 的面积 $S_{\triangle} = \dfrac{1}{2} |x_0||y_0| = \dfrac{1}{2} |x_0 y_0| = \dfrac{1}{2} |k|$。\n
			\par\textbf{理由：} $P$ 在 $y = \dfrac{k}{x}$ 上，故 $x_0 y_0 = k$。\n
		\item[\textbf{第三步（解出 $k$）：}]
			由 $S_{\triangle}=5$ 得 $\dfrac{1}{2}|k| = 5$，所以 $|k| = 10$，因此 $k = \pm 10$。\n
			\par\textbf{理由：} 面积公式给出 $|k|$，$k$ 的正负由双曲线所在象限决定，题目未明确时两个都有可能。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$k = \pm 10$。}\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知反比例函数 $y = \dfrac{k}{x}$ 与一次函数 $y = x-1$ 的图象交于点 $A(2,1)$ 和点 $B$。过点 $A$、$B$ 分别向两坐标轴作垂线，围成的两个矩形面积分别为 $S_A$、$S_B$，求 $S_A + S_B$。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求 $k$）：}]
			将 $A(2,1)$ 代入 $y = \dfrac{k}{x}$，得 $1 = \dfrac{k}{2}$，所以 $k = 2$。故反比例函数为 $y = \dfrac{2}{x}$。\n
			\par\textbf{理由：} 点 $A$ 在图象上。\n
		\item[\textbf{第二步（求 $B$ 坐标）：}]
			联立 $y = x-1$ 与 $y = \dfrac{2}{x}$，得 $x-1 = \dfrac{2}{x}$，即 $x^2 - x - 2 = 0$。解得 $x = 2$ 或 $x = -1$。代回得 $y = 1$ 或 $y = -2$，故 $B(-1,-2)$。\n
			\par\textbf{理由：} 两函数图象的交点同时满足两个解析式，解方程组得到另一交点。\n
		\item[\textbf{第三步（面积和）：}]
			由 $k$ 的几何意义，每个矩形面积 $S = |k|$，故 $S_A = |k| = 2$，$S_B = |k| = 2$，$S_A + S_B = 4$。\n
			\par\textbf{理由：} 过反比例函数图象上任一点作坐标轴垂线，矩形面积恒等于 $|k|$，与该点坐标无关。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$S_A + S_B = 4$。}\n
\end{examplebox}
