	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（切线条件）：}}]
			点$P$在$\odot O$外，$PA$、$PB$分别切$\odot O$于$A$、$B$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（切线长相等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从圆外一点引出的两条切线长度相等。
			\[
				PA=PB
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（角平分关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆心与点P的连线平分两条切线所夹的角。
			\[
				\angle APO = \angle BPO
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（垂直平分弦）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			设$C=OP\cap AB$，则$OP$垂直$AB$且$AC=BC$。
			\[
				OP\perp AB,\quad AC=BC.
			\]
	\end{description}

\end{statementbox}
