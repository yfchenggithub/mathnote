\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{切线长相等，角平分关系。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（外点切线）：} 点$P$在$\odot O$外，且$PA$、$PB$是$\odot O$的切线。
		\item \textbf{条件2（垂直前提）：} 切线与半径垂直$OA\perp PA$，$OB\perp PB$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（概念区分）：} 切线长是数值，切线是直线；注意条件：点必须在外、线必须相切。
		\item \textbf{检查项（推论使用）：} 垂直平分弦、角平分等推论需先证明，不能直接当作定理。
	\end{itemize}

\end{summarybox}
