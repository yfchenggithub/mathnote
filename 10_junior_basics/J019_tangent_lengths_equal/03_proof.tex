\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	通过连接圆心与切点，构造直角三角形，利用“斜边直角边（HL）”证明两三角形全等，进而得到边相等与角相等。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（证垂直）：}]
			由切线性质，$OA\perp PA$，$OB\perp PB$，因此$\triangle PAO$和$\triangle PBO$均为直角三角形。
			\[
				\angle OAP = \angle OBP = 90^\circ.
			\]
			\par\textbf{理由：} 圆的切线垂直于过切点的半径。
		\item[\textbf{步骤二（半径相等）：}]
			在$\triangle PAO$和$\triangle PBO$中，$OA$、$OB$是$\odot O$的半径，故相等。
			\[
				OA = OB.
			\]
			\par\textbf{理由：} 同圆半径相等。
		\item[\textbf{步骤三（公共边）：}]
			两个三角形共享边$OP$。
			\[
				OP = OP.
			\]
			\par\textbf{理由：} 公共边。
		\item[\textbf{步骤四（HL全等）：}]
			两个直角三角形满足斜边和一条直角边对应相等，故全等。
			\[
				\triangle PAO \cong \triangle PBO.
			\]
			\par\textbf{理由：} $OA=OB$（直角边），$OP=OP$（斜边），$\angle OAP=\angle OBP=90^\circ$，符合HL判定。
		\item[\textbf{步骤五（切线长相等）：}]
			由全等得对应边相等。
			\[
				PA = PB.
			\]
			\par\textbf{理由：} 全等三角形对应边相等。
		\item[\textbf{步骤六（角相等）：}]
			由全等得对应角相等。
			\[
				\angle APO = \angle BPO.
			\]
			\par\textbf{理由：} 全等三角形对应角相等。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{由以上推导可知，从圆外一点引两条切线，切线长相等，且该点与圆心的连线平分两切线的夹角。}
\end{proofbox}
