\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{一线三等角模型：共线、三组等角且 $AD\parallel BE$，则 $\triangle ABD\sim\triangle BCE$，对应边成比例。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三点共线）：} 点$A,B,C$共线。
		\item \textbf{条件 2（三角相等）：} $\angle ABD=\angle BCE=\angle DBE=\theta$。
		\item \textbf{条件 3（平行）：} $AD\parallel BE$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（对应边顺序）：} 比例必须按对应顶点：$\frac{AB}{BC}=\frac{BD}{CE}=\frac{AD}{BE}$。
		\item \textbf{检查项（平行检查）：} 应用模型前务必确认 $AD\parallel BE$ 成立。
	\end{itemize}

\end{summarybox}
