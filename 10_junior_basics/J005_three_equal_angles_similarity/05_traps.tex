\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（对应边匹配错误）}}]
			常见错误比例：$\dfrac{AB}{CE}=\dfrac{BD}{BE}$，未按对应顶点。
		\item[\textbf{\textcolor{CTrap}{正确理解（顶点顺序对应）：}}]
			严格按顶点顺序：$\dfrac{AB}{BC}=\dfrac{BD}{CE}=\dfrac{AD}{BE}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（相似对应不清）：}}]
			学生未掌握相似三角形中对应顶点的判定方法，随意匹配。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（全等条件误用）}}]
			误以为只要有一组边相等就全等，如认为$AB=BC$可推出全等，忽略相似比是否为$1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（相似比需为1）：}}]
			全等要求相似比$k=1$，即所有对应边相等。仅一组对应边相等不能直接得全等，需先确认相似且该组边为对应边。
		\item[\textbf{\textcolor{CTrap}{错因分析（全等判定混淆）：}}]
			学生混淆全等判定定理，忘记全等是相似的特例。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略平行条件）}}]
			在未确认 $AD\parallel BE$ 的情况下直接套用结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（平行是关键）：}}]
			本模型要求 $AD\parallel BE$，否则 $\angle ADB\neq\angle DBE$，相似不成立。解题时务必先验证或构造平行。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件疏漏）：}}]
			学生只关注到三个等角，忽略模型对平行关系的依赖。
	\end{description}

\end{trapbox}
