\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，点$A,B,C$共线，$\angle ABD=\angle BCE=\angle DBE=60^{\circ}$，$AD\parallel BE$，已知$AB=6$，$BC=4$，$AD=6$，$BD=6$。求$CE$和$BE$的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}]
			由共线、等角及$AD\parallel BE$得$\triangle ABD\sim\triangle BCE$。
			\par\textbf{理由：}满足一线三等角模型条件。
		\item[\textbf{第二步（求相似比）：}]
			相似比 $k=\dfrac{AB}{BC}=\dfrac{6}{4}=1.5$。
			\par\textbf{理由：}相似三角形对应边比例。
		\item[\textbf{第三步（计算所求边）：}]
			$CE=\dfrac{BD}{k}=\dfrac{6}{1.5}=4$，$BE=\dfrac{AD}{k}=\dfrac{6}{1.5}=4$。
			\par\textbf{理由：}代入比例式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$CE=4$，$BE=4$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（全等）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，点$A,B,C$共线，$\angle ABD=\angle BCE=\angle DBE=60^{\circ}$，$AD\parallel BE$，且$AB=BC=4$，$AD=4$，$BD=4$。求$CE$和$BE$的长，并判断$\triangle ABD$与$\triangle BCE$是否全等。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（相似）：}]
			由模型得$\triangle ABD\sim\triangle BCE$。
			\par\textbf{理由：}满足条件。
		\item[\textbf{第二步（全等判定）：}]
			相似比 $k=\dfrac{AB}{BC}=1$，故$\triangle ABD\cong\triangle BCE$。
			\par\textbf{理由：}相似比为$1$即全等。
		\item[\textbf{第三步（求边）：}]
			全等三角形对应边相等，$CE=BD=4$，$BE=AD=4$。
			\par\textbf{理由：}全等性质。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$CE=4$，$BE=4$，且$\triangle ABD\cong\triangle BCE$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，点$A,B,C$共线，$\angle ABD=\angle BCE=\angle DBE=60^{\circ}$，$AD\parallel BE$，$AB=4$，$BC=6$，$\triangle ABD$的面积为$8\sqrt{3}$，求$\triangle BCE$的面积。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求相似比）：}]
			由模型得$\triangle ABD\sim\triangle BCE$，相似比 $k=\dfrac{AB}{BC}=\dfrac{4}{6}=\dfrac{2}{3}$。
			\par\textbf{理由：}相似三角形对应边比例。
		\item[\textbf{第二步（面积比）：}]
			面积比等于相似比的平方，即$\dfrac{S_{\triangle ABD}}{S_{\triangle BCE}}=k^2=\dfrac{4}{9}$。
			\par\textbf{理由：}相似三角形面积比性质。
		\item[\textbf{第三步（计算面积）：}]
			$S_{\triangle BCE}=S_{\triangle ABD}\cdot\dfrac{9}{4}=8\sqrt{3}\times\dfrac{9}{4}=18\sqrt{3}$。
			\par\textbf{理由：}代入已知面积。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\triangle BCE$的面积为$18\sqrt{3}$。}
