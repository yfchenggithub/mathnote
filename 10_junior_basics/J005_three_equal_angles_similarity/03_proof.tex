\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	利用 $AD\parallel BE$ 得 $\angle ADB=\theta$，从而在 $\triangle ABD$ 中求出 $\angle BAD$；再在 $\triangle BCE$ 中通过平角与内角和求出 $\angle CBE$，发现 $\angle BAD=\angle CBE$，结合已知 $\angle ABD=\angle BCE$，得证相似。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（由平行得等角）：}]
			$\because AD\parallel BE$，$\therefore \angle ADB=\angle DBE=\theta$。
			\par\textbf{理由：}两直线平行，内错角相等。
		\item[\textbf{步骤二（求 $\angle BAD$）：}]
			在 $\triangle ABD$ 中，$\angle ABD=\theta$，$\angle ADB=\theta$，由三角形内角和
			\[
				\angle BAD = 180^\circ - \theta - \theta = 180^\circ - 2\theta.
			\]
			\par\textbf{理由：}三角形内角和定理。
		\item[\textbf{步骤三（求 $\angle EBC$）：}]
			点 $A,B,C$ 共线，$\angle ABD+\angle DBE+\angle EBC=180^\circ$。代入 $\angle ABD=\angle DBE=\theta$，得
			\[
				\angle EBC = 180^\circ - 2\theta.
			\]
			\par\textbf{理由：}平角定义。
		\item[\textbf{步骤四（求 $\angle CBE$ 与 $\angle BEC$）：}]
			在 $\triangle BCE$ 中，$\angle BCE=\theta$，$\angle EBC=180^\circ-2\theta$（即 $\angle CBE$），由内角和得
			\[
				\angle BEC = 180^\circ - \theta - (180^\circ-2\theta) = \theta.
			\]
			故 $\angle CBE = 180^\circ - 2\theta$.
			\par\textbf{理由：}三角形内角和定理。
		\item[\textbf{步骤五（相似结论）：}]
			在 $\triangle ABD$ 与 $\triangle BCE$ 中，
			\[
				\angle ABD = \angle BCE = \theta,
				\qquad
				\angle BAD = \angle CBE = 180^\circ - 2\theta,
			\]
			\[
				\therefore \triangle ABD \sim \triangle BCE.
			\]
			\par\textbf{理由：}两角对应相等的两个三角形相似。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{因此对应边成比例：$\dfrac{AB}{BC}=\dfrac{BD}{CE}=\dfrac{AD}{BE}$。全等条件：当一组对应边相等时，$\triangle ABD\cong\triangle BCE$。}
\end{proofbox}
