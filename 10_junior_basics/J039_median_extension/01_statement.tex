	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（中点与中线）：}}]
			在$\triangle ABC$中，$D$为$BC$的中点，$AD$为中线。
		\item[\textbf{\textcolor{CStatement}{条件2（倍长构造）：}}]
			延长$AD$至点$E$，使得$DE = AD$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（连接BE全等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			连接$BE$，可得$\triangle ADC$与$\triangle EDB$全等，实现边角转移。
			\[
				\triangle ADC \cong \triangle EDB \quad (\mathrm{SAS}).
			\]
			\[
				AC = BE,
				\qquad
				AC \parallel BE.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（连接CE全等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			连接$CE$，可得$\triangle ADB$与$\triangle EDC$全等，得到另一组相等和平行。
			\[
				\triangle ADB \cong \triangle EDC \quad (\mathrm{SAS}).
			\]
			\[
				AB = CE,
				\qquad
				AB \parallel CE.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（平行四边形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同时连接$BE$和$CE$，可得四边形$ABEC$为平行四边形。
			\[
				AC = BE,\quad AC \parallel BE,\quad AB = CE,\quad AB \parallel CE.
			\]
	\end{description}

\end{statementbox}
