	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，在$\triangle ABC$中，$D$为$BC$的中点，延长$AD$至$E$使$DE=AD$，连接$BE$。求证：$AC=BE$且$AC\parallel BE$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（准备条件）：}]
			由中点得$BD=CD$，由作图得$AD=ED$，对顶角$\angle ADC=\angle EDB$。
			\[
				BD=CD,\quad AD=ED,\quad \angle ADC=\angle EDB.
			\]
			\par\textbf{理由：}中点定义、作图规定、对顶角相等。
		\item[\textbf{第二步（证全等）：}]
			在$\triangle ADC$和$\triangle EDB$中，$AD=ED$，$\angle ADC=\angle EDB$，$DC=DB$，故$\triangle ADC\cong\triangle EDB$（SAS）。
			\[
				\triangle ADC\cong\triangle EDB\ (\mathrm{SAS}).
			\]
			\par\textbf{理由：}两边及其夹角对应相等。
		\item[\textbf{第三步（得出结论）：}]
			由全等得$AC=EB$，且$\angle ACD=\angle EBD$，所以$AC\parallel BE$。
			\[
				AC=EB,
				\qquad
				\angle ACD=\angle EBD\ \Rightarrow\ AC\parallel BE.
			\]
			\par\textbf{理由：}全等三角形对应边相等，对应角相等；内错角相等等价于两直线平行。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AC=BE$ 且 $AC\parallel BE$}.

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=5$，$AC=3$，$AD$为$BC$边上的中线，求$AD$的取值范围。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（倍长构造）：}]
			延长$AD$至$E$，使$DE=AD$，连接$BE$。
			\[
				DE=AD.
			\]
			\par\textbf{理由：}利用倍长中线构造全等，转移边AC。
		\item[\textbf{第二步（等量代换）：}]
			由倍长中线法知$\triangle ADC\cong\triangle EDB$，故$AC=BE=3$，且$AE=2AD$。
			\[
				AC=BE=3,
				\qquad
				AE=2AD.
			\]
			\par\textbf{理由：}全等三角形对应边相等；倍长定义。
		\item[\textbf{第三步（三边关系）：}]
			在$\triangle ABE$中，由三边关系得
			\[
				\begin{aligned}
					AB-BE &< AE < AB+BE,\\ 5-3 &< 2AD < 5+3,\\ 2 &< 2AD < 8,\\ 1 &< AD < 4.
			\end{aligned}
			\]
			\par\textbf{理由：}三角形两边之和大于第三边，两边之差小于第三边。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$AD$的取值范围是$\mathbf{1<AD<4}$.

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$\angle BAC=90^\circ$，$AD$是$BC$边上的中线。求证：$AD=\dfrac{1}{2}BC$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（倍长构造）：}]
			延长$AD$至$E$，使$DE=AD$，连接$BE$、$CE$。
			\[
				DE=AD.
			\]
			\par\textbf{理由：}构造平行四边形ABEC对角线互相平分。
		\item[\textbf{第二步（证平行四边形）：}]
			由$BD=CD$，$AD=ED$，得四边形$ABEC$是平行四边形（对角线互相平分）。
			\[
				BD=CD,\ AD=ED.
			\]
			\par\textbf{理由：}对角线互相平分的四边形是平行四边形。
		\item[\textbf{第三步（证矩形）：}]
			因为$\angle BAC=90^\circ$，平行四边形$ABEC$中有一个直角，故$ABEC$是矩形。
			\[
				\angle BAC=90^\circ.
			\]
			\par\textbf{理由：}有一个角是直角的平行四边形是矩形。
		\item[\textbf{第四步（得结论）：}]
			矩形对角线相等，所以$BC=AE$，而$AE=2AD$，故$AD=\dfrac{1}{2}BC$。
			\[
				BC=AE,\ AE=2AD\ \Rightarrow\ AD=\frac12 BC.
			\]
			\par\textbf{理由：}矩形对角线相等；倍长定义。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\mathbf{AD=\dfrac{1}{2}BC}$.
\end{examplebox}
