\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{倍长中线法通过构造全等三角形（或平行四边形），实现边角条件转移，常用于中点问题。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（中点条件）：}题中存在线段中点或中线。
		\item \textbf{条件2（转移边角）：}需要证明线段相等、平行或转移分散条件。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（全等判定）：}避免SSA误用，坚持SAS。
		\item \textbf{检查项（平行依据）：}证明平行必须基于角相等，不能仅靠等长。
	\end{itemize}

\end{summarybox}
