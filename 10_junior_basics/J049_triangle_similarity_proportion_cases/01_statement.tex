	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（点在线段上）：}}]
			点 $D$ 在边 $AB$ 上，点 $E$ 在边 $AC$ 上。
		\item[\textbf{\textcolor{CStatement}{条件 2（相似前提）：}}]
			$\triangle ADE$ 与 $\triangle ABC$ 相似（对应顶点未指定）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（顶点相同）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点 $A$ 对应点 $A$，点 $D$ 对应点 $B$，点 $E$ 对应点 $C$。
			\[
				\frac{AD}{AB}=\frac{AE}{AC}
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（顶点交换）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点 $A$ 对应点 $A$，点 $D$ 对应点 $C$，点 $E$ 对应点 $B$。
			\[
				\frac{AD}{AC}=\frac{AE}{AB}
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（平行条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若 $DE\parallel BC$，则只出现第一种相似关系。
			\[
				\frac{AD}{AB}=\frac{AE}{AC}=\frac{DE}{BC}
			\]
	\end{description}

\end{statementbox}
