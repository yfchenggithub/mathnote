\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AB=6$，$AC=4$，点 $D$ 在 $AB$ 上，点 $E$ 在 $AC$ 上，且 $\triangle ADE$ 与 $\triangle ABC$ 相似（对应顶点未指定）。已知 $AD=2$，求 $AE$ 的长度。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别对应）：}]
			由结论知，满足 $\triangle ADE\sim\triangle ABC$ 有两种对应，均需考虑。
			\par\textbf{理由：}
			相似对应未指定，必须分类。

		\item[\textbf{第二步（第一种对应）：}]
			若 $\triangle ADE\sim\triangle ABC$（即 $A\leftrightarrow A,\;D\leftrightarrow B,\;E\leftrightarrow C$），则
			\[\frac{AD}{AB}=\frac{AE}{AC}
				\;\Longrightarrow\; \frac{2}{6}=\frac{AE}{4}
				\;\Longrightarrow\; AE=\frac{4}{3}.\]
				\par\textbf{理由：}
				直接代入比例式。

			\item[\textbf{第三步（第二种对应）：}]
				若 $\triangle ADE\sim\triangle ACB$（即 $A\leftrightarrow A,\;D\leftrightarrow C,\;E\leftrightarrow B$），则
				\[\frac{AD}{AC}=\frac{AE}{AB}
					\;\Longrightarrow\; \frac{2}{4}=\frac{AE}{6}
					\;\Longrightarrow\; AE=3.\]
					\par\textbf{理由：}
					代入另一比例式。
			\end{description}

			\textbf{\textcolor{CExample}{关键结论：}}
			\textbf{$AE=\dfrac{4}{3}$ 或 $AE=3$，两种都满足 $E$ 在线段 $AC$ 上（因为 $AE\leqslant AC=4$）。}

			\medskip
			\textbf{\textcolor{CExample}{例 2（稍变形）}}

			\textbf{\textcolor{CExample}{题目：}}
			在 $\triangle ABC$ 中，$AB=6$，$AC=4$，点 $D$ 在 $AB$ 上，$AD=5$，点 $E$ 在 $AC$ 上，且 $\triangle ADE$ 与 $\triangle ABC$ 相似（对应顶点未指定）。求 $AE$ 的长，并判断点 $E$ 是否在线段 $AC$ 上。

			\textbf{\textcolor{CExample}{解题步骤：}}
			\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
				\item[\textbf{第一步（分类计算比例）：}]
					由结论得两种可能：
					\[\frac{5}{6}=\frac{AE}{4}\;\Rightarrow\;AE=\frac{10}{3}\approx3.33,
						\qquad
						\frac{5}{4}=\frac{AE}{6}\;\Rightarrow\;AE=\frac{15}{2}=7.5.\]
						\par\textbf{理由：}
						分别代入 $\frac{AD}{AB}=\frac{AE}{AC}$ 和 $\frac{AD}{AC}=\frac{AE}{AB}$。

					\item[\textbf{第二步（检验点 $E$ 位置）：}]
						由于 $E$ 在线段 $AC$ 上，必须有 $0<AE\leqslant AC=4$。$AE=\frac{10}{3}\approx3.33\leqslant4$ 满足；$AE=7.5>4$ 不满足，故舍去。
						\par\textbf{理由：}
						几何条件限制以实际线段为界。

					\item[\textbf{第三步（确认结果）：}]
						因此只有第一种对应成立，$AE=\dfrac{10}{3}$。
						\par\textbf{理由：}
						第二种对应导致 $E$ 落在线段 $AC$ 之外，不符合题意。
				\end{description}

				\textbf{\textcolor{CExample}{关键结论：}}
				\textbf{$AE=\dfrac{10}{3}$，且只有 $\triangle ADE\sim\triangle ABC$（$A\leftrightarrow A,\;D\leftrightarrow B,\;E\leftrightarrow C$）满足点 $E$ 在线段 $AC$ 上。}

				\medskip
				\textbf{\textcolor{CExample}{例 3（综合应用）}}

				\textbf{\textcolor{CExample}{题目：}}
				在 $\triangle ABC$ 中，$AB=8$，$AC=6$，$\angle A=60^\circ$。点 $D$ 在 $AB$ 上，$AD=4$，点 $E$ 在 $AC$ 上，且 $\triangle ADE$ 与 $\triangle ABC$ 相似（对应顶点未指定）。求 $\triangle ADE$ 的面积。

				\textbf{\textcolor{CExample}{解题步骤：}}
				\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
					\item[\textbf{第一步（求两种AE）：}]
						由结论得：
						\[\frac{4}{8}=\frac{AE}{6}\;\Rightarrow\;AE=3,
							\qquad
							\frac{4}{6}=\frac{AE}{8}\;\Rightarrow\;AE=\frac{16}{3}\approx5.33.\]
							\par\textbf{理由：}
							分别代入两个比例式。

						\item[\textbf{第二步（验证线段条件）：}]
							因为 $E$ 在线段 $AC$ 上，$0<AE\leqslant6$。$AE=3$ 和 $AE=\frac{16}{3}\approx5.33$ 均 $\leqslant6$，所以两种都成立。
							\par\textbf{理由：}
							位置约束$AE\leqslant AC$二者都满足。

						\item[\textbf{第三步（计算两种面积）：}]
							$\triangle ADE$ 的面积 $S=\frac12\cdot AD\cdot AE\cdot\sin\angle A$。

							情形一（$AE=3$）：$S_1=\frac12\cdot4\cdot3\cdot\sin60^\circ=\frac12\cdot12\cdot\frac{\sqrt3}{2}=3\sqrt3$。

							情形二（$AE=\frac{16}{3}$）：$S_2=\frac12\cdot4\cdot\frac{16}{3}\cdot\frac{\sqrt3}{2}=\frac{16\sqrt3}{3}$。
							\par\textbf{理由：}
							三角形面积公式 $S=\frac12ab\sin C$，代入数据。

						\item[\textbf{第四步（给出结论）：}]
							因此 $\triangle ADE$ 的面积可能为 $3\sqrt3$ 或 $\dfrac{16\sqrt3}{3}$。
							\par\textbf{理由：}
							两种对应均符合条件，面积不同皆需考虑。
					\end{description}

					\textbf{\textcolor{CExample}{关键结论：}}
					\textbf{当 $\triangle ADE\sim\triangle ABC$ 时，$AE=3$，$S=3\sqrt3$；当 $\triangle ADE\sim\triangle ACB$ 时，$AE=\dfrac{16}{3}$，$S=\dfrac{16\sqrt3}{3}$。}
				\end{examplebox}
