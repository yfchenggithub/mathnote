\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{当 $D$ 在 $AB$ 上、$E$ 在 $AC$ 上且 $\triangle ADE\sim\triangle ABC$ 时，对应关系有两种可能，比例式相应有两种。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（点位置）：} 点 $D$ 在边 $AB$ 上，点 $E$ 在边 $AC$ 上。
		\item \textbf{条件 2（相似条件）：} $\triangle ADE$ 与 $\triangle ABC$ 相似，未指明顶点对应。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分类讨论）：} 必须列出两种比例式，分别求解。
		\item \textbf{检查项（点在线段）：} 计算所得边长应满足 $E$ 在线段 $AC$ 上的条件 $0<AE\leq AC$，舍去不合理解。
	\end{itemize}
\end{summarybox}
