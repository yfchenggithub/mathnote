\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对于相似图形，面积比等于相似比的平方。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（图形相似）：} 两个图形必须相似（对应角相等，对应边长成比例）。
		\item \textbf{条件 2（相似比已知）：} 相似比 $k$ 已知或可通过线段比求得。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略平方）：} 面积比 $\neq$ 相似比，必须 $\frac{S_1}{S_2}=k^2$。
		\item \textbf{检查项（前提确认）：} 使用结论前先验证图形是否相似。
	\end{itemize}
\end{summarybox}
