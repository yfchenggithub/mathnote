\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础直接应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC\sim\triangle A'B'C'$，相似比为 $\frac{3}{4}$，且 $S_{A'B'C'}=32$，求 $S_{ABC}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用面积比定理）：}]
			由面积比定理得：
			\[
				\frac{S_{ABC}}{S_{A'B'C'}} = \frac{9}{16}.
			\]
			\par\textbf{理由：} 面积比等于相似比平方，代入 $k=\frac{3}{4}$，$k^2=\frac{9}{16}$。
		\item[\textbf{第二步（建立比例方程）：}]
			代入 $S_{A'B'C'}=32$ 得：
			\[
				\frac{S_{ABC}}{32}=\frac{9}{16}.
			\]
			\par\textbf{理由：} 将已知代入比例式。
		\item[\textbf{第三步（求解面积结果）：}]
			解方程：
			\[
				S_{ABC}=32\times\frac{9}{16}.
			\]
			\par\textbf{理由：} 两边乘 $32$，得 $S_{ABC}=18$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_{ABC}=18$}。

	\textbf{\textcolor{CExample}{例 2（变形——已知高比）}}

	\textbf{\textcolor{CExample}{题目：}}
	两个相似三角形对应高之比为 $2:3$，且它们的面积之和为 $130$，求每个三角形的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由高比得相似比）：}]
			对应高之比等于相似比，故：
			\[
				k=\frac{2}{3}.
			\]
			\par\textbf{理由：} 相似三角形对应高成比例且等于相似比。
		\item[\textbf{第二步（计算面积比）：}]
			由面积比定理得：
			\[
				\frac{S_1}{S_2} = \frac{4}{9}.
			\]
			\par\textbf{理由：} 面积比等于相似比平方，$k^2=\frac{4}{9}$。
		\item[\textbf{第三步（列方程求解）：}]
			设两三角形面积分别为 $4x$ 和 $9x$，则：
			\[
				4x+9x=130.
			\]
			\par\textbf{理由：} 由面积和列方程。
			解得 $x=10$，故面积分别为 $40$ 和 $90$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{两三角形面积分别为 $40$ 和 $90$}。

	\textbf{\textcolor{CExample}{例 3（综合——已知周长和与面积和）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC\sim\triangle DEF$，周长之比为 $2:5$，且面积之和为 $87$，求 $\triangle ABC$ 与 $\triangle DEF$ 的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（周长比得相似比）：}]
			周长比等于相似比，故：
			\[
				k=\frac{2}{5}.
			\]
			\par\textbf{理由：} 相似三角形周长之比等于相似比。
		\item[\textbf{第二步（计算面积比）：}]
			由面积比定理得：
			\[
				\frac{S_{ABC}}{S_{DEF}} = \frac{4}{25}.
			\]
			\par\textbf{理由：} 面积比等于相似比平方，$k^2=\frac{4}{25}$。
		\item[\textbf{第三步（列方程求解）：}]
			设 $S_{ABC}=4x$，$S_{DEF}=25x$，则：
			\[
				4x+25x=87.
			\]
			\par\textbf{理由：} 由面积和列方程。
			解得 $x=3$，故 $S_{ABC}=12$，$S_{DEF}=75$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_{ABC}=12$，$S_{DEF}=75$}。
