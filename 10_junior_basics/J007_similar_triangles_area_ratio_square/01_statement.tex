\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（两个相似图形）：}}] 两个图形相似，相似比为 $k$，面积分别为 $S_1$ 和 $S_2$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面积比公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积之比等于相似比的平方。
			\[
				\frac{S_1}{S_2}=k^2.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（对应线段平方比）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			相似三角形面积比等于任一对对应线段比的平方。
			\[
				\frac{S_1}{S_2} = \left(\frac{a_1}{a_2}\right)^2.
			\]
			\[
				\frac{S_1}{S_2} = \left(\frac{h_1}{h_2}\right)^2.
			\]
			\[
				\frac{S_1}{S_2} = \left(\frac{C_1}{C_2}\right)^2.
			\]
	\end{description}
\end{statementbox}
