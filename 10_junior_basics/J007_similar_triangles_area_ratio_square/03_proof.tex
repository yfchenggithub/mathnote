\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	考虑两个相似三角形，将面积用底和高表示，利用对应边和高成比例，推出面积比等于$k^2$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设对应元素）：}]
			设 $\triangle ABC\sim\triangle A'B'C'$，相似比 $k=\frac{a}{a'}=\frac{h}{h'}$（$a,a'$ 为对应边，$h,h'$ 分别为 $a,a'$ 边上的高）。
			\[
				\frac{a}{a'}=k,
				\qquad
				\frac{h}{h'}=k.
			\]
			\par\textbf{理由：} 相似三角形对应边成比例，对应高也成比例且等于相似比。

		\item[\textbf{步骤二（写出面积公式）：}]
			使用三角形面积公式：
			\[
				S=\frac12 a h,
				\qquad
				S'=\frac12 a' h'.
			\]
			\par\textbf{理由：} 任意三角形的面积等于底乘高的一半。

		\item[\textbf{步骤三（求面积比并化简）：}]
			先写面积比：
			\[
				\frac{S}{S'} = \frac{\frac12 a h}{\frac12 a' h'}.
			\]
			化简得：
			\[
				\frac{S}{S'} = \frac{a}{a'}\cdot\frac{h}{h'}.
			\]
			\par\textbf{理由：} 分子分母约去$\frac12$，分数相乘时分子分母分别相除。

		\item[\textbf{步骤四（代入相似比得结论）：}]
			将比值代入，得到面积比：
			\[
				\frac{S}{S'} = k^2.
			\]
			\par\textbf{理由：} 由 $\frac{S}{S'}=\frac{a}{a'}\cdot\frac{h}{h'}$，再代入 $\frac{a}{a'}=k$ 和 $\frac{h}{h'}=k$，得 $k\cdot k=k^2$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，对于任意两个相似三角形，面积比等于相似比的平方。该结论也可推广到一般相似图形。}
\end{proofbox}
