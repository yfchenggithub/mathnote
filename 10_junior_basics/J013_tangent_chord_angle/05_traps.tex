\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（切点角识别）}}] 误将顶点不在切点的角当作弦切角，或混淆切线与弦。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义核对）}}] 弦切角必须同时满足：顶点在切点，一边是切线，另一边是过切点的弦。
		\item[\textbf{\textcolor{CTrap}{错因分析（基础概念）}}] 对角定义的各要素把握不严。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（所夹弧判断）}}] 选错弦切角所夹的弧，导致圆周角选取错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（角内弧）}}] 所夹弧是位于角内部的那段圆弧，需根据图形判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（弧的两端）}}] 没有区分角内部和外部弧。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（圆周角对应）}}] 使用了不是同弧所对的圆周角。
		\item[\textbf{\textcolor{CTrap}{正确理解（同弧原则）}}] 所选圆周角必须与弦切角对应同一段弧。
		\item[\textbf{\textcolor{CTrap}{错因分析（混用弧）}}] 对弧与角的关系认识不清。
	\end{description}
\end{trapbox}
