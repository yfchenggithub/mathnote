\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过作辅助线，利用圆周角定理、切线性质和等腰三角形性质，建立角度关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}

	（下面证明针对弦切角为锐角（圆心在弦切角外部）的情形。直角或钝角情形可类似证明。）

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（作辅助线）：}]
			连接圆心$O$与$P,A,B$得半径$OP,OA,OB$。
			\[
				OP,\; OA,\; OB.
			\]
			\par\textbf{理由：}
			准备使用圆心角和圆周角。

		\item[\textbf{步骤二（圆周角定理）：}]
			因为$\angle PBA$和$\angle POA$对应同一弧$\wideparen{PA}$，所以
			\[
				\angle PBA = \frac{1}{2}\angle POA.
			\]
			\par\textbf{理由：}
			同弧所对的圆周角等于圆心角的一半。

		\item[\textbf{步骤三（切线性质）：}]
			$PT$是切线，$OP$为半径，所以$OP\perp PT$，即
			\[
				\angle OPT = 90^\circ.
			\]
			\par\textbf{理由：}
			切线垂直于过切点的半径。

		\item[\textbf{步骤四（等腰三角形）：}]
			在$\triangle OAP$中，$OA=OP$，所以
			\[
				\angle OAP = \angle OPA.
			\]
			\par\textbf{理由：}
			等腰三角形底角相等。

		\item[\textbf{步骤五（三角形内角和）：}]
			$\triangle OAP$中，
			\[
				\angle POA + \angle OAP + \angle OPA = 180^\circ.
			\]
			代入$\angle OAP = \angle OPA$得
			\[
				\angle POA + 2\angle OPA = 180^\circ,
			\]
			故
			\[
				\angle OPA = 90^\circ - \frac{1}{2}\angle POA.
			\]
			\par\textbf{理由：}
			三角形内角和为$180^\circ$，代数变形。

		\item[\textbf{步骤六（计算$\angle APT$）：}]
			因为此时圆心$O$在$\angle APT$外部，所以$\angle APT = \angle OPT - \angle OPA = 90^\circ - \angle OPA$。将$\angle OPA$代入：
			\[
				\begin{aligned}
					\angle APT &= 90^\circ - \left(90^\circ - \frac{1}{2}\angle POA\right) \\
					&= \frac{1}{2}\angle POA.
			\end{aligned}
			\]
			\par\textbf{理由：}
			角度差运算。

		\item[\textbf{步骤七（等量代换）：}]
			由步骤二知$\displaystyle\angle PBA = \frac{1}{2}\angle POA$，所以
			\[
				\angle APT = \angle PBA.
			\]
			\par\textbf{理由：}
			等量传递。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	至此，弦切角等于它所夹弧所对的圆周角（锐角情形）得证。
\end{proofbox}
