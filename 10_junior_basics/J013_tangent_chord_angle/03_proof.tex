\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	把弦切角分成两种：内部夹劣弧时是锐角，内部夹优弧时是钝角。两种情况都可以用“切线垂直半径”和“圆心角、圆周角关系”证明。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}

	设$\theta=\angle POA$，其中$\theta$表示劣弧$\overset{\frown}{PA}$所对的圆心角。

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（连接半径）：}]
			连接$OP,OA$。
			\[
				OP=OA.
			\]
			\par\textbf{理由：}
			同圆半径相等。

		\item[\textbf{步骤二（等腰三角形求底角）：}]
			在$\triangle OPA$中，$OP=OA$，所以
			\[
				\angle OPA=\angle OAP.
			\]
			又因为
			\[
				\angle POA+\angle OPA+\angle OAP=180^\circ,
			\]
			所以
			\[
				\angle OPA=90^\circ-\frac{1}{2}\theta.
			\]
			\par\textbf{理由：}
			等腰三角形底角相等，三角形内角和为$180^\circ$。

		\item[\textbf{步骤三（切线垂直半径）：}]
			因为$PT$是$\odot O$在$P$处的切线，$OP$为过切点的半径，所以
			\[
				OP\perp PT,\qquad \angle OPT=90^\circ.
			\]
			\par\textbf{理由：}
			圆的切线垂直于过切点的半径。

		\item[\textbf{步骤四（劣弧对应的弦切角）：}]
			若$\angle APT$内部夹的是劣弧$\overset{\frown}{PA}$，则
			\[
				\begin{aligned}
					\angle APT
					&=\angle OPT-\angle OPA \\
					&=90^\circ-\left(90^\circ-\frac{1}{2}\theta\right) \\
					&=\frac{1}{2}\theta.
				\end{aligned}
			\]
			\par\textbf{理由：}
			角度差运算。

		\item[\textbf{步骤五（优弧对应的弦切角）：}]
			若$\angle APT$内部夹的是优弧$\overset{\frown}{PA}$，则该弦切角是上一步锐角的补角，所以
			\[
				\angle APT=180^\circ-\frac{1}{2}\theta
				=\frac{1}{2}(360^\circ-\theta).
			\]
			\par\textbf{理由：}
			同一直线上的两个邻补角和为$180^\circ$，优弧所对圆心角为$360^\circ-\theta$。

		\item[\textbf{步骤六（与圆周角对应）：}]
			圆周角等于它所对弧的圆心角的一半。因此，无论$\angle APT$内部夹的是劣弧还是优弧，都有
			\[
				\angle APT=\angle PBA.
			\]
			\par\textbf{理由：}
			二者对应同一段弧。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	只要$\angle APT$是切线$PT$与弦$PA$形成的弦切角，且$\angle PBA$是同一段弧$\overset{\frown}{PA}$所对的圆周角，就有
	\[
		\angle APT=\angle PBA.
	\]
\end{proofbox}
