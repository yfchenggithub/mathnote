	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（切线与切点）：}}] 圆$O$中，$PT$是切线，$P$为切点。
		\item[\textbf{\textcolor{CStatement}{条件2（弦与弦切角）：}}] $PA$是过$P$的弦，$\angle APT$是弦切角。
		\item[\textbf{\textcolor{CStatement}{条件3（所夹弧）：}}] $\overset{\frown}{PA}$（位于$\angle APT$内部）是弦切角所夹的弧。
		\item[\textbf{\textcolor{CStatement}{条件4（圆周角点）：}}] $B$是$\odot O$上异于$P,A$的一点，$\angle PBA$是$\overset{\frown}{PA}$所对的圆周角。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（弦切角定理）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      弦切角等于它所夹弧所对的圆周角。
		      \[
			      \angle APT = \angle PBA.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（直径特例）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      当弦$PA$为直径时，弦切角为直角。
		      \[
			      \angle APT = 90^\circ.
		      \]
	\end{description}
\end{statementbox}
