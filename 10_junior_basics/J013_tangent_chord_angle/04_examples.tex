	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，在$\odot O$中，$PT$是切线，$P$为切点，$PA$是弦，$B$是$\odot O$上一点，$\angle PBA = 40^\circ$，求$\angle APT$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别弦切角）：}]
		      $\angle APT$的顶点为$P$（切点），两边$PT$（切线）和$PA$（弦），因此是弦切角。
		      \[
			      \angle APT.
		      \]
		      \par\textbf{理由：}
		      弦切角定义：顶点在切点，一边切线，一边过切点弦。

		\item[\textbf{第二步（确定所夹弧）：}]
		      $\angle APT$所夹的弧为$\overset{\frown}{PA}$（角内部的部分）。
		      \[
			      \overset{\frown}{PA}.
		      \]
		      \par\textbf{理由：}
		      弦切角所夹弧的定义。

		\item[\textbf{第三步（找圆周角）：}]
		      $\angle PBA$是$\overset{\frown}{PA}$所对的圆周角（$B$在圆上）。
		      \[
			      \angle PBA.
		      \]
		      \par\textbf{理由：}
		      顶点$B$在圆上，且$\overset{\frown}{PA}$在$\angle PBA$内部。

		\item[\textbf{第四步（套定理）：}]
		      由弦切角定理，$\angle APT = \angle PBA$。已知 $\angle PBA = 40^\circ$，故
		      \[
			      \angle APT = 40^\circ.
		      \]
		      \par\textbf{理由：}
		      定理直接应用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle APT = 40^\circ$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$AB$是$\odot O$的直径，$C$是$\odot O$上一点，过$C$的切线$CT$交$AB$延长线于$T$，连接$AC$。若$\angle CAB = 35^\circ$，求$\angle TCA$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别弦切角）：}]
		      $\angle TCA$的顶点$C$（切点），一边$CT$（切线），一边$CA$（弦），故为弦切角。
		      \[
			      \angle TCA.
		      \]
		      \par\textbf{理由：}
		      弦切角定义。

		\item[\textbf{第二步（确定所夹弧）：}]
		      $\angle TCA$所夹的弧为$\overset{\frown}{CA}$（位于角内部）。
		      \[
			      \overset{\frown}{CA}.
		      \]
		      \par\textbf{理由：}
		      所夹弧定义。

		\item[\textbf{第三步（找圆周角）：}]
		      $\angle CBA$是$\overset{\frown}{CA}$所对的圆周角（$B$在圆上）。
		      \[
			      \angle CBA.
		      \]
		      \par\textbf{理由：}
		      圆周角定义，且$B$在圆上。

		\item[\textbf{第四步（计算$\angle CBA$）：}]
		      因为$AB$是直径，所以$\angle ACB = 90^\circ$（直径所对圆周角为直角）。在$\triangle ABC$中，
		      \[
			      \begin{aligned}
				      \angle CBA & = 90^\circ - \angle CAB \\
				                 & = 90^\circ - 35^\circ   \\
				                 & = 55^\circ.
			      \end{aligned}
		      \]
		      \par\textbf{理由：}
		      直角三角形两锐角互余。

		\item[\textbf{第五步（套定理）：}]
		      由弦切角定理，$\angle TCA = \angle CBA$。故
		      \[
			      \angle TCA = 55^\circ.
		      \]
		      \par\textbf{理由：}
		      弦切角等于所夹弧所对的圆周角。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle TCA = 55^\circ$}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$\triangle ABC$内接于$\odot O$，过点$A$作$\odot O$的切线交$BC$延长线于$D$．求证：$\angle DAB = \angle ACB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别弦切角）：}]
		      $\angle DAB$的顶点$A$（切点），一边$AD$（切线），一边$AB$（弦），是弦切角。
		      \[
			      \angle DAB.
		      \]
		      \par\textbf{理由：}
		      弦切角定义。

		\item[\textbf{第二步（确定所夹弧）：}]
		      $\angle DAB$所夹的弧为$\overset{\frown}{AB}$（角内部弧）。
		      \[
			      \overset{\frown}{AB}.
		      \]
		      \par\textbf{理由：}
		      所夹弧定义。

		\item[\textbf{第三步（找圆周角）：}]
		      $\angle ACB$是$\overset{\frown}{AB}$所对的圆周角（$C$在圆上）。
		      \[
			      \angle ACB.
		      \]
		      \par\textbf{理由：}
		      圆周角定义。

		\item[\textbf{第四步（套定理）：}]
		      由弦切角定理，$\angle DAB = \angle ACB$。
		      \[
			      \angle DAB = \angle ACB.
		      \]
		      \par\textbf{理由：}
		      弦切角等于所夹弧所对的圆周角。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle DAB = \angle ACB$（证毕）}
