\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆外一点 $P$，$PT$ 是切线（$T$ 为切点），$PAB$ 是割线（$A$ 在 $P$、$B$ 之间）。若 $PT=6$，$PA=4$，求 $PB$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别定理）：}]
			根据题意，$P$ 在圆外，$PT$ 为切线，$PAB$ 为割线，满足切割线定理条件。
			\par\textbf{理由：} 切割线定理只要求圆外一点、一条切线、一条割线，符合即可使用。
		\item[\textbf{第二步（代入公式）：}]
			由切割线定理得 $PT^{2}=PA\cdot PB$。代入 $PT=6$, $PA=4$:
			\[
				36 = 4\cdot PB.
			\]
			\par\textbf{理由：} 等量关系直接代入。
		\item[\textbf{第三步（求解答案）：}]
			解方程，$PB=36/4=9$。
			\[
				PB = 9.
			\]
			\par\textbf{理由：} 一元一次方程求解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PB=9$，且弦 $AB=PB-PA=5$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	圆外一点 $P$，切线 $PT$ 长 $8$，割线 $PAB$ 交圆于 $A,B$，且 $PA:PB=1:2$（$A$ 在 $P$ 与 $B$ 之间）。求 $PA$ 和 $PB$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}]
			设 $PA=x$，则由比例 $PA:PB=1:2$ 得 $PB=2x$。
			\par\textbf{理由：} 利用比值设未知元，便于列方程。
		\item[\textbf{第二步（列等积方程）：}]
			由切割线定理 $PT^{2}=PA\cdot PB$，代入 $PT=8$, $PA=x$, $PB=2x$:
			\[
				64 = x \cdot 2x = 2x^{2}.
			\]
			\par\textbf{理由：} 等量代换。
		\item[\textbf{第三步（解方程得结果）：}]
			解 $2x^{2}=64$，得 $x^{2}=32$，因为 $x>0$，所以 $x=\sqrt{32}=4\sqrt{2}$。
			\[
				x=4\sqrt{2},\quad 2x=8\sqrt{2}.
			\]
			\par\textbf{理由：} 二次方程算术平方根取正。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA=4\sqrt{2}$，$PB=8\sqrt{2}$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，圆外一点 $P$，$PT$ 切圆于 $T$，$PAB$ 和 $PCD$ 是两条割线，分别交圆于 $A,B$ 和 $C,D$。已知 $PA=3$，$AB=5$，$PC=4$，点 $A$ 在 $P$ 与 $B$ 之间，$C$ 在 $P$ 与 $D$ 之间。求 $CD$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求切线长）：}]
			对割线 $PAB$，先计算 $PB$：$PB=PA+AB=3+5=8$。由切割线定理 $PT^{2}=PA\cdot PB$，得
			\[
				PT^{2}=3\cdot 8=24,\quad PT=2\sqrt{6}.
			\]
			\par\textbf{理由：} 在第一条割线中利用定理求得切线长，作为桥梁。
		\item[\textbf{第二步（对另一割线列式）：}]
			对割线 $PCD$，同样有 $PT^{2}=PC\cdot PD$。代入 $PT^{2}=24$, $PC=4$:
			\[
				24 = 4\cdot PD \quad\Rightarrow\quad PD=6.
			\]
			\par\textbf{理由：} 同一圆外点 $P$ 的切线长不变，故切割线定理对任意割线成立。
		\item[\textbf{第三步（求 $CD$）：}]
			由 $CD=PD-PC$，得
			\[
				CD = 6 - 4 = 2.
			\]
			\par\textbf{理由：} 线段差的关系。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$CD=2$。}
\end{examplebox}
