	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（圆外一点与切线、割线）：}}]
			圆外一点 $P$ 引圆的切线 $PT$（$T$ 为切点）和一条割线交圆于 $A,B$ 两点（$P,A,B$ 共线且 $A$ 在 $P$ 与 $B$ 之间）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（切割线等积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			切线长的平方等于割线上两段距离的乘积。
			\[
				PT^{2}=PA\cdot PB .
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（积为定值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对于给定圆外点，任意割线所得乘积恒等于切线长的平方。
			\[
				PA\cdot PB = PT^{2} \quad (\text{$PAB$ 为任意过 $P$ 的割线}).
			\]
	\end{description}

\end{statementbox}
