\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{切割线定理：圆外一点 $P$ 的切线长 $PT$ 的平方等于该点到割线与圆两交点 $A,B$ 的线段长乘积，即 $PT^{2}=PA\cdot PB$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（圆外一点）：} 点 $P$ 必须在圆的外部。
		\item \textbf{条件 2（存在切线）：} 从 $P$ 引出的直线中有一条与圆相切，切点为 $T$。
		\item \textbf{条件 3（存在割线）：} 从 $P$ 引出的另一条直线与圆相交于两个点 $A,B$（$A$ 靠近 $P$，$B$ 远离 $P$）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件检查）：} 使用前必须确认 $P$ 在圆外，且 $PT$ 是切线，$PAB$ 是割线（与圆有两个交点）。
		\item \textbf{检查项（公式精度）：} $PT^{2}$ 是平方，$PA\cdot PB$ 是乘积，两者相等，单位一致。
	\end{itemize}

\end{summarybox}
