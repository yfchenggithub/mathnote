\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件遗漏）}}]
			应用时忽略“点 $P$ 在圆外”的前提，或误将圆上点当作圆外点使用定理。
		\item[\textbf{\textcolor{CTrap}{正确理解（前提必备）：}}]
			切割线定理只适用于从圆外一点引切线和割线的情形，点 $P$ 在圆外是根本条件。若 $P$ 在圆内或圆上，结论不成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件不清）：}}]
			学生通常只记忆公式，忘记公式推导所依赖的几何设定，导致在题目未明确给出“圆外”时想当然套用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公式混淆）}}]
			将 $PT^{2}=PA\cdot PB$ 误记为 $PT=PA\cdot PB$ 或 $PT^{2}=PA+PB$ 等形式。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确记忆）：}}]
			公式是“切线长的平方等于割线上两段长的乘积”，即 $PT^{2}=PA\cdot PB$。可通过量纲帮助记忆：切线长单位一次，平方得面积；乘积也是面积，两边量纲一致。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆不牢）：}}]
			没有抓住等积式的特征，容易与比例关系混淆。建议反复默写并做简单数字验证。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号遗漏）}}]
			解方程 $x^{2}=k$ 时直接写 $x=\pm\sqrt{k}$，未结合线段长取正，导致多解或错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（长度非负）：}}]
			线段长度只能取非负值，所以 $x=\sqrt{k}$（$k>0$）。若 $k$ 由平方根得到，也需检查合理性。
		\item[\textbf{\textcolor{CTrap}{错因分析（负数误解）：}}]
			学生习惯解二次方程取两个根，未考虑几何实际意义的约束。应养成“先判断正负再取值”的习惯。
	\end{description}

\end{trapbox}
