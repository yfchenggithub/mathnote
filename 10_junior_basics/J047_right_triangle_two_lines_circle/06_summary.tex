\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{判断直角三角形动点轨迹的充要条件是“两线一圆”。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（定点A,B已知）：} 已知两定点坐标，可写出轨迹方程。
		\item \textbf{条件 2（动点C有限制）：} 动点满足直线或曲线方程，可联立求解。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分类与排除）：} 分类直角顶点，排除重合点。
		\item \textbf{检查项（代入验证）：} 求出点后验证点积为零并确认不与顶点重合。
	\end{itemize}

\end{summarybox}
