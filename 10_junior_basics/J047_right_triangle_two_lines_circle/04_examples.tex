\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知点 $A(1,2), B(-3,4)$，点 $C$ 在直线 $y=2$ 上，且 $\angle C=90^\circ$，求点 $C$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定轨迹类型）：}]
			由结论一，$\angle C=90^\circ$ 时 $C$ 在以 $AB$ 为直径的圆上（除 $A,B$）。
			\par\textbf{理由：} 两线一圆法。
		\item[\textbf{第二步（求圆方程）：}]
			$AB$ 中点 $(-1,3)$，半径平方 $r^2=\frac{(1+3)^2+(2-4)^2}{4}=5$，圆方程为 $(x+1)^2+(y-3)^2=5$。
			\par\textbf{理由：} 直径圆公式。
		\item[\textbf{第三步（联立直线）：}]
			与 $y=2$ 联立：$(x+1)^2+(2-3)^2=5 \Rightarrow (x+1)^2=4 \Rightarrow x=1$ 或 $x=-3$。
			\par\textbf{理由：} 解代数方程。
		\item[\textbf{第四步（排除重合点）：}]
			$C(1,2)$ 与 $A$ 重合，舍去；$C(-3,2)$ 与 $B$ 不重合，保留。
			\par\textbf{理由：} 三角形顶点必须互异。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	点 $C(-3,2)$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知点 $A(0,0), B(4,0)$，点 $C$ 在直线 $y=2x-4$ 上，且 $\angle A=90^\circ$，求点 $C$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别直角顶点）：}]
			由于 $\angle A=90^\circ$，$C$ 在过 $A$ 且垂直于 $AB$ 的直线上（除 $A$）。
			\par\textbf{理由：} 结论二。
		\item[\textbf{第二步（求垂线方程）：}]
			$AB$ 方向 $(4,0)$，垂线方向 $(0,1)$，过 $A(0,0)$ 的垂线方程为 $x=0$。
			\par\textbf{理由：} 垂直向量关系。
		\item[\textbf{第三步（联立直线）：}]
			由 $x=0$ 与 $y=2x-4$ 联立，得 $C(0,-4)$。
			\par\textbf{理由：} 解方程组。
		\item[\textbf{第四步（检验）：}]
			$C(0,-4) \neq A(0,0)$，有效。
			\par\textbf{理由：} 排除重合。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	点 $C(0,-4)$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知点 $A(0,0), B(6,0)$，点 $C$ 在直线 $x=3$ 上移动。是否存在 $C$ 使得 $\triangle ABC$ 为直角三角形？若存在，求所有 $C$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分类直角顶点）：}]
			可能直角顶点为 $C$，$A$ 或 $B$，需分别讨论。
			\par\textbf{理由：} 三角形三个角均可能是直角。
		\item[\textbf{第二步（$\angle C=90^\circ$）：}]
			以 $AB$ 为直径的圆方程为 $(x-3)^2+y^2=9$，与 $x=3$ 联立得 $y^2=9$，$y=\pm3$，得到 $C_1(3,3),C_2(3,-3)$，均不与 $A,B$ 重合，有效。
			\par\textbf{理由：} 结论一。
		\item[\textbf{第三步（$\angle A=90^\circ$）：}]
			过 $A$ 的垂线为 $x=0$，与 $x=3$ 无交点，故无解。
			\par\textbf{理由：} 结论二。
		\item[\textbf{第四步（$\angle B=90^\circ$）：}]
			过 $B$ 的垂线为 $x=6$，与 $x=3$ 无交点，故无解。
			\par\textbf{理由：} 结论三。
		\item[\textbf{第五步（总结）：}]
			存在两个点 $C$ 满足条件：$C(3,3)$ 或 $C(3,-3)$。
			\par\textbf{理由：} 检验后均为有效点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$C(3,3)$ 或 $C(3,-3)$。
\end{examplebox}
