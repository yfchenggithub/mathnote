\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（重合点误判）}}]
			错误：当 $C$ 与 $A$ 或 $B$ 重合时，点积也为 $0$，但三角形不存在。
		\item[\textbf{\textcolor{CTrap}{正确理解（排除A,B）}}]
			使用点积为零时必须同时要求 $C\neq A,B$。
		\item[\textbf{\textcolor{CTrap}{错因分析（顶点互异性）}}]
			忽略了三角形三顶点必须互异的基本要求。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（漏分类顶点）}}]
			错误：只考虑 $\angle C=90^\circ$ 的情况，忽略 $\angle A$ 或 $\angle B$ 也可能是直角。
		\item[\textbf{\textcolor{CTrap}{正确理解（三种可能）}}]
			当题目未指定直角顶点时，必须分三种情况讨论。
		\item[\textbf{\textcolor{CTrap}{错因分析（未分类讨论）}}]
			思维定势认为直角一定在动点 $C$ 处，缺乏分类意识。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（轨迹形状混淆）}}]
			错误：认为所有直角条件均对应圆的方程。
		\item[\textbf{\textcolor{CTrap}{正确理解（圆或直线区别）}}]
			直角在 $C$ 时对应圆，直角在 $A$ 或 $B$ 时对应直线。
		\item[\textbf{\textcolor{CTrap}{错因分析（垂直向量混淆）}}]
			混淆了不同顶点对应的向量点积组合，导致轨迹误判。
	\end{description}

\end{trapbox}
