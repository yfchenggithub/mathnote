\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（同圆等弧）：}}]
			在同圆或等圆中，设$\odot O$中弧$AB$所对的圆心角为$\angle AOB$，点$C$为圆上不与$A,B$重合的点，则$\angle ACB$为弧$AB$所对的圆周角。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（劣弧圆周角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当点$C$在优弧$AB$上（即$AB$为劣弧）时，圆周角等于圆心角的一半。
			\[
				\angle ACB = \frac{1}{2}\angle AOB.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（优弧圆周角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当点$C$在劣弧$AB$上（即$AB$为优弧）时，圆周角等于互补圆心角的一半。
			\[
				\angle ACB = \frac{1}{2}(360^\circ - \angle AOB).
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（等弧对等角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同弧或等弧所对的圆周角相等，即$\angle ACB = \angle ADB$（$D$为圆上另一点）。
			\[
				\angle ACB = \angle ADB.
			\]
	\end{description}

\end{statementbox}
