\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\odot O$中，$\angle AOB=80^\circ$，点$C$在优弧$AB$上（不与$A,B$重合），求$\angle ACB$的度数。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（弧位判断）：}]
			因为$C$在优弧$AB$上，所以$AB$为劣弧，$\angle ACB$是劣弧所对的圆周角。
			\par\textbf{理由：} 点在优弧上，所对弧为劣弧。
		\item[\textbf{第二步（直接套用）：}]
			由圆周角定理，
			\[
				\begin{aligned}
					\angle ACB &= \frac12\angle AOB \\ &= \frac12 \times 80^\circ \\ &= 40^\circ.
			\end{aligned}
			\]
			\par\textbf{理由：} 圆周角等于圆心角的一半。
		\item[\textbf{第三步（结论作答）：}]
			因此$\angle ACB = 40^\circ$。
			\par\textbf{理由：} 计算完成。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle ACB = 40^\circ$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\odot O$中，点$C$在劣弧$AB$上，$\angle ACB=120^\circ$，求$\angle AOB$的度数。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（弧位判断）：}]
			因为$C$在劣弧$AB$上，所以$AB$为优弧，$\angle ACB$是优弧所对的圆周角。
			\par\textbf{理由：} 点在劣弧上，所对弧为优弧。
		\item[\textbf{第二步（公式变形）：}]
			由圆周角定理（优弧），$\angle ACB = \frac12(360^\circ - \angle AOB)$，故$\angle AOB = 360^\circ - 2\angle ACB$．计算得
			\[
				\begin{aligned}
					\angle AOB &= 360^\circ - 2\angle ACB \\ &= 360^\circ - 2\times 120^\circ \\ &= 120^\circ.
			\end{aligned}
			\]
			\par\textbf{理由：} 移项求解与代入计算。
		\item[\textbf{第三步（结论作答）：}]
			因此$\angle AOB = 120^\circ$。
			\par\textbf{理由：} 计算完成。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle AOB = 120^\circ$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\odot O$中，$\angle AOB=120^\circ$，点$D$是弧$AB$的中点（在劣弧$AB$上），点$C$在优弧$AB$上，求$\angle ACB$和$\angle ADB$的度数。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（劣弧求角）：}]
			因为$C$在优弧$AB$上，所以
			\[
				\begin{aligned}
					\angle ACB &= \frac12\angle AOB \\ &= \frac12\times 120^\circ \\ &= 60^\circ.
			\end{aligned}
			\]
			\par\textbf{理由：} 同弧所对圆周角等于圆心角一半。
		\item[\textbf{第二步（优弧求角）：}]
			$D$在劣弧$AB$上，所以
			\[
				\begin{aligned}
					\angle ADB &= \frac12(360^\circ - \angle AOB) \\ &= \frac12(360^\circ - 120^\circ) \\ &= 120^\circ.
			\end{aligned}
			\]
			\par\textbf{理由：} 优弧所对圆周角等于互补圆心角的一半。
		\item[\textbf{第三步（结果验证）：}]
			$\angle ACB + \angle ADB = 60^\circ + 120^\circ = 180^\circ$，符合圆内接四边形对角互补。
			\par\textbf{理由：} 验证结果合理。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle ACB = 60^\circ$，$\angle ADB = 120^\circ$}
\end{examplebox}
