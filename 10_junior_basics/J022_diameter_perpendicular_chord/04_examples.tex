\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在圆 $O$ 中，半径 $R=5$，弦心距 $d=3$，弦 $CD$ 垂直于直径 $AB$。求弦 $CD$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入公式）：}] 由垂径定理得 $\left(\frac{l}{2}\right)^2 + d^2 = R^2$，代入 $R=5$，$d=3$：
			\[
				\left(\frac{l}{2}\right)^2 + 9 = 25.
			\]
			\par\textbf{理由：} 垂径定理公式。

		\item[\textbf{第二步（解半弦长）：}] 解得 $\left(\frac{l}{2}\right)^2 = 16$，故 $\frac{l}{2}=4$（取正）。
			\par\textbf{理由：} 平方法。

		\item[\textbf{第三步（得弦长）：}] 所以 $l = 2 \times 4 = 8$。
			\par\textbf{理由：} 半弦长乘以2。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{弦 $CD$ 的长为 $8$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在圆 $O$ 中，直径 $AB$ 垂直于弦 $CD$ 于点 $E$，且 $CD=8$，$OE=3$。求圆 $O$ 的半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（得半弦长）：}] $CD=8$，所以 $\frac{CD}{2}=4$。
			\par\textbf{理由：} 弦长的一半。

		\item[\textbf{第二步（代公式）：}] 由垂径定理，$R^2 = \left(\frac{CD}{2}\right)^2 + OE^2$，代入 $CD=8$，$OE=3$得：
			\[
				\begin{aligned}
					R^2 &= 4^2 + 3^2 \\
					&= 25.
			\end{aligned}
			\]
			\par\textbf{理由：} 勾股关系。

		\item[\textbf{第三步（求半径）：}] 故 $R = 5$（取正）。
			\par\textbf{理由：} 开平方。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{半径 $R=5$。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	圆 $O$ 的直径 $AB=10$，弦 $CD=8$，且 $CD\perp AB$ 于 $E$。求 $AE$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（得半径）：}] $AB=10$ 得半径 $R=5$。
			\par\textbf{理由：} 直径的一半。

		\item[\textbf{第二步（求弦心距）：}] 在 $\mathrm{Rt}\triangle OCE$ 中，$CE=\frac{CD}{2}=4$，$OC=R=5$，由垂径定理：
			\[
				\begin{aligned}
					OE &= \sqrt{R^2 - CE^2} \\
					&= \sqrt{5^2-4^2} \\
					&= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 勾股定理。

		\item[\textbf{第三步（求 $AE$）：}] 点 $E$ 可能在圆心 $O$ 与 $A$ 之间或与 $B$ 之间，故 $AE = AO \pm OE$，即 $AE=5-3=2$ 或 $AE=5+3=8$。
			\par\textbf{理由：} 点 $E$ 的位置不确定。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AE$ 的长为 $2$ 或 $8$。}
\end{examplebox}
