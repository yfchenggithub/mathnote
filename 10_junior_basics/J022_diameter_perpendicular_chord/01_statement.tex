\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（直径垂直弦）：}}] 在圆 $O$ 中，$AB$ 为直径，$CD$ 为弦，且 $AB \perp CD$ 于点 $E$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（垂直平分弦与弧）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直径 $AB$ 平分弦 $CD$，并平分 $CD$ 所对的两条弧。
			\[
				CE = ED,\quad \widehat{AC} = \widehat{AD},\quad \widehat{BC} = \widehat{BD}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（弦长公式链）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆的半径 $R$、弦长 $l$、弦心距 $d$ 构成直角三角形关系。
			\[
				\left(\frac{l}{2}\right)^2 + d^2 = R^2.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（逆定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若一条直径平分一条非直径的弦，则该直径垂直于该弦。
			\[
				(O\in AB) \land (CE=ED) \land (O\notin CD) \Longrightarrow (AB \perp CD).
			\]
	\end{description}

\end{statementbox}
