\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	先利用半径相等和垂直条件证明弦被平分，再证明直径两端与弦端点连线段相等，最后利用等弦对等弧得到弧相等。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（半径证平分）：}] 连接圆心 $O$ 与弦端点 $C$、$D$，得 $OC$、$OD$。
			\[
				\begin{aligned}
					& OC=OD,\ OE\perp CD \\
					&\quad \Longrightarrow \triangle OCE\cong\triangle ODE\ (\text{HL}) \\
					&\quad \Longrightarrow CE=ED.
			\end{aligned}
			\]
			\par\textbf{理由：} 等腰三角形底边上的高也是中线。

		\item[\textbf{步骤二（端点证等弦）：}] 连接 $AC$、$AD$。
			\[
				\begin{aligned}
					& AE=AE,\ CE=ED,\ \angle AEC=\angle AED=90^\circ \\
					&\quad \Longrightarrow \triangle AEC\cong\triangle AED\ (\text{SAS}) \\
					&\quad \Longrightarrow AC=AD.
			\end{aligned}
			\]
			\par\textbf{理由：} 两边及其夹角相等则全等。

		\item[\textbf{步骤三（等弦推等弧）：}] 在圆 $O$ 中，$AC=AD$ 得 $\widehat{AC}=\widehat{AD}$。同理连接 $BC$、$BD$ 可证 $BC=BD$，得 $\widehat{BC}=\widehat{BD}$。
			\[
				AC=AD \Longrightarrow \widehat{AC}=\widehat{AD},
				\qquad
				BC=BD \Longrightarrow \widehat{BC}=\widehat{BD}.
			\]
			\par\textbf{理由：} 同圆中，等弦所对的弧相等（通常指劣弧）。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{综上，直径 $AB\perp CD$ 于点 $E$ 时，$AB$ 平分弦 $CD$，且平分弦 $CD$ 所对的两条弧（$\widehat{AC}=\widehat{AD}$，$\widehat{BC}=\widehat{BD}$）。}
\end{proofbox}
