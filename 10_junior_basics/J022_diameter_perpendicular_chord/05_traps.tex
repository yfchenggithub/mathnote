\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（直径特殊情形）}}] 误认为只要直径平分弦，就自动垂直于弦。
		\item[\textbf{\textcolor{CTrap}{正确理解（非直径弦条件）：}}] 只有当被平分的弦不是直径时，平分该弦的直径才垂直于弦。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视特殊性）：}}] 没有考虑直径本身也是弦，且直径被任意直径平分，但未必垂直。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（漏乘二因子）}}] 在代入公式 $(\frac{l}{2})^2+d^2=R^2$ 后，直接得出 $l=\sqrt{R^2-d^2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（半弦再乘二）：}}] 正确的弦长公式为 $l=2\sqrt{R^2-d^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式记错）：}}] 忽略了 $\frac{l}{2}$ 表示半弦长，需要再乘以 $2$ 才得到弦长。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（弦心距定义）}}] 错误认为圆心到弦上任意一点的距离都是弦心距。
		\item[\textbf{\textcolor{CTrap}{正确理解（垂直距离）：}}] 弦心距特指圆心到弦的垂线段长度，且垂足就是弦的中点。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 混淆了“点到直线的距离”与“点到点的距离”，弦心距是前者，不是后者。
	\end{description}
\end{trapbox}
