\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	直接利用多项式乘法展开左边，通过合并同类项得到右边。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（平方差验证）：}]
			\[
				\begin{aligned}
					(a+b)(a-b) &= a^2 - ab + ab - b^2 \\
					&= a^2 - b^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 分配律相乘，交叉项抵消。

		\item[\textbf{步骤二（和平方验证）：}]
			\[
				\begin{aligned}
					(a+b)^2 &= (a+b)(a+b) \\
					&= a^2 + ab + ab + b^2 \\
					&= a^2 + 2ab + b^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方即自乘，分配律，合并同类项。

		\item[\textbf{步骤三（差平方验证）：}]
			\[
				\begin{aligned}
					(a-b)^2 &= (a-b)(a-b) \\
					&= a^2 - ab - ab + b^2 \\
					&= a^2 - 2ab + b^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 与和平方类似，但交叉项为负。

		\item[\textbf{步骤四（立方和验证）：}]
			\[
				\begin{aligned}
					(a+b)(a^2-ab+b^2) &= a^3 - a^2b + ab^2 \\
					&\quad + a^2b - ab^2 + b^3 \\
					&= a^3 + b^3.
			\end{aligned}
			\]
			\par\textbf{理由：} 逐项展开后，中间四项全部抵消。

		\item[\textbf{步骤五（立方差验证）：}]
			\[
				\begin{aligned}
					(a-b)(a^2+ab+b^2) &= a^3 + a^2b + ab^2 \\
					&\quad - a^2b - ab^2 - b^3 \\
					&= a^3 - b^3.
			\end{aligned}
			\]
			\par\textbf{理由：} 展开后中间四项抵消。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	以上五个步骤分别验证了三个乘法公式，左右恒等，故公式成立。
\end{proofbox}
