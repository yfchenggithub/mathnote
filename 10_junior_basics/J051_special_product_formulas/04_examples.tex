\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	计算 $(3x+2y)(3x-2y)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别公式）：}] 观察表达式结构为 $(a+b)(a-b)$，符合平方差公式。
			\par\textbf{理由：} 平方差公式要求两数和乘以两数差。
		\item[\textbf{第二步（代入公式）：}] 将 $a=3x$, $b=2y$ 代入 $a^2-b^2$，得到
			\[
				\begin{aligned}
					(3x+2y)(3x-2y) &= (3x)^2 - (2y)^2 \\
					&= 9x^2 - 4y^2.
			\end{aligned}
			\]
			\par\textbf{理由：} $(a+b)(a-b)=a^2-b^2$。
		\item[\textbf{第三步（得出结论）：}] 最终结果为 $9x^2-4y^2$，无需合并同类项。
			\par\textbf{理由：} 公式直接给出结果。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$9x^2-4y^2$}。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	展开 $(2a-3b)^2$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别公式）：}] 结构为 $(a-b)^2$，符合完全平方公式。
			\par\textbf{理由：} 完全平方公式形式为 $(a\pm b)^2=a^2\pm2ab+b^2$。
		\item[\textbf{第二步（代入展开）：}] 取 $a=2a$, $b=3b$，代入 $(a-b)^2=a^2-2ab+b^2$，得
			\[
				\begin{aligned}
					(2a-3b)^2 &= (2a)^2 - 2\cdot2a\cdot3b + (3b)^2 \\
					&= 4a^2 -12ab + 9b^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用完全平方公式。
		\item[\textbf{第三步（化简结果）：}] 整理后得到 $4a^2-12ab+9b^2$，这是最简形式。
			\par\textbf{理由：} 各项已合并，无同类项。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$4a^2-12ab+9b^2$}。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	化简 $(x+1)(x-1)(x^2+1)(x^4+1)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（首次平方差）：}] 先计算前两个因式 $(x+1)(x-1)$，根据平方差公式：
			\[
				(x+1)(x-1) = x^2 - 1.
			\]
			\par\textbf{理由：} $a=x$, $b=1$，代入 $a^2-b^2$。
		\item[\textbf{第二步（二次平方差）：}] 将第一步结果与第三个因式相乘：$(x^2-1)(x^2+1)$，再次使用平方差公式：
			\[
				(x^2-1)(x^2+1) = x^4 - 1.
			\]
			\par\textbf{理由：} 此处 $a=x^2$, $b=1$。
		\item[\textbf{第三步（三次平方差）：}] 将第二步结果与最后一个因式相乘：$(x^4-1)(x^4+1)$，第三次使用平方差公式：
			\[
				(x^4-1)(x^4+1) = x^8 - 1.
			\]
			\par\textbf{理由：} $a=x^4$, $b=1$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x^8-1$}。
\end{examplebox}
