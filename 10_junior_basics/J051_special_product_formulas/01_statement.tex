\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（变量类型）：}}] 实数或代数式。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{结论一（平方差公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个数的平方差等于两数和与两数差的乘积。
			\[
				a^2-b^2=(a+b)(a-b)
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（完全平方公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个数的和（或差）的平方等于它们的平方和加上（或减去）它们积的2倍。
			\[
				(a\pm b)^2=a^2\pm 2ab+b^2
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（立方和差公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个数的立方和（或差）等于两数和（或差）与它们的平方和减去（或加上）乘积的积。
			\[
				a^3\pm b^3=(a\pm b)(a^2\mp ab+b^2)
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（三项平方公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三个数和的平方等于各数平方和加上两两积的2倍。
			\[
				(a+b+c)^2=a^2+b^2+c^2+2ab+2bc+2ca
			\]
	\end{description}
\end{statementbox}
