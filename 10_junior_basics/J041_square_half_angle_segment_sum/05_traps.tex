\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件限制忽略）}}]
			错误地认为在任意四边形中都有 $EF = BE + DF$。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件严格）：}}]
			结论仅适用于正方形且 $\angle EAF = 45^\circ$ 的情形。
		\item[\textbf{\textcolor{CTrap}{错因分析（特殊推广）：}}]
			受直觉误导，忽略了正方形直角和邻边相等的特殊性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（符号记忆错误）}}]
			将乘积关系记为 $xy = a^2 + a(x+y)$ 或 $xy = a^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确定义）：}}]
			明确推导来源，$xy = a^2 - a(x+y)$，可通过代入具体数值验证。
		\item[\textbf{\textcolor{CTrap}{错因分析（粗心推导）：}}]
			展开时符号处理错误，或未正确移项。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（解的范围忽略）}}]
			解方程求出 $x$ 或 $y$ 后，未检查是否在 $[0,a]$ 内。
		\item[\textbf{\textcolor{CTrap}{正确理解（范围约束）：}}]
			$BE$ 和 $DF$ 均为边长上的线段，其值应满足 $x\in[0,a]$，$y\in[0,a]$。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏几何检验）：}}]
			只关注代数解，忽视几何位置约束。
	\end{description}

\end{trapbox}
