\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{正方形内 $45^\circ$ 角模型：$EF = BE + DF$ 且 $xy = a^2 - a(x+y)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（正方形45度角）：} 正方形 $ABCD$ 且 $\angle EAF=45^\circ$。
		\item \textbf{条件 2（变量定义）：} $BE = x$，$DF = y$，$EF = z$，边长 $AB = a$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件与符号）：} 注意结论只适用于正方形内含 $45^\circ$ 角；乘积关系符号切勿记反。
		\item \textbf{检查项（范围验证）：} 解出的 $x$ 和 $y$ 必须在 $[0,a]$ 内。
	\end{itemize}

\end{summarybox}
