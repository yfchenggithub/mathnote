\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}
	将 $\triangle ADF$ 绕点 $A$ 顺时针旋转 $90^\circ$ 至 $\triangle ABG$，则 $G, B, E$ 共线，可得 $EF = EG$，又 $EG = BE + DF$，故 $EF = BE + DF$。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（旋转构造）：}]
			将 $\triangle ADF$ 绕点 $A$ 顺时针旋转 $90^\circ$ 到 $\triangle ABG$，使 $AD$ 与 $AB$ 重合。
		\item[\textbf{要点二（全等共线）：}]
			由旋转得 $\angle GAE = \angle FAE$（均为 $45^\circ$），结合 $AG = AF$，得 $\triangle AEF \cong \triangle AEG$，从而 $EF = EG$；又 $EG = BE + DF$，故 $EF = BE + DF$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（旋转与全等）}}
	通过旋转将不在同一直线的线段集中，利用全等三角形实现等量代换。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（二次方程）}}
	在勾股定理的基础上导出 $x$ 与 $y$ 的二次关系式。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（两个结论）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接求和）：} 已知正方形边长及 $BE$，直接套用 $EF = BE+DF$ 及 $xy$ 关系求 $DF$ 和 $EF$。
		\item \textbf{考法二（列方程）：} 通过 $xy$ 关系列方程求未知量，常与一元二次方程结合。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}
	\textbf{看到正方形内 $45^\circ$ 角，立即想到旋转构造。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（求线段长）：} 已知正方形边长和一边上的分点，求另一分点对应的线段长。
		\item \textbf{场景二（综合计算）：} 结合勾股定理、面积、方程求解复杂问题。
	\end{itemize}

\end{explanationbox}
