\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略开口方向）：}}]
		      求最值时，不先判断 $a$ 的正负，直接把顶点值当作最大值或最小值。
		\item[\textbf{\textcolor{CTrap}{正确理解（先定开口方向）：}}]
		      $a>0$ 时顶点是最低点；$a<0$ 时顶点是最高点。必须先判断开口方向，再讨论最值。
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性思维）：}}]
		      只记住“二次函数看顶点”，却忘记顶点到底对应最大值还是最小值由开口方向决定。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略区间限制）：}}]
		      对称轴不在区间内时，仍然把 $f(h)$ 当作区间最值。
		\item[\textbf{\textcolor{CTrap}{正确理解（顶点必须在区间内）：}}]
		      只有 $h\in[m,n]$ 时，顶点值才可能成为区间最值；若 $h\notin[m,n]$，区间最值只能在端点 $m,n$ 中产生。
		\item[\textbf{\textcolor{CTrap}{错因分析（区间意识缺失）：}}]
		      把函数在全体实数上的最值，误当成函数在指定闭区间上的最值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（对称轴公式错误）：}}]
		      将对称轴算错，如写成 $h=\frac{b}{2a}$ 或 $h=-\frac{b}{a}$，导致分类位置全部错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（牢记公式）：}}]
		      二次函数 $f(x)=ax^2+bx+c$ 的对称轴为
		      \[
			      h=-\frac{b}{2a}.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（符号粗心）：}}]
		      对负号和分母 $2a$ 不敏感，特别是在含参数题中更容易出错。
	\end{description}
\end{trapbox}
