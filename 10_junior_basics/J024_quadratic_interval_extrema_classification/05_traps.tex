	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略开口方向）：}}]
			求最值时，不加判断直接使用 $h$ 代入得最值，不考虑 $a$ 的正负和区间限制。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（先定开口方向）：}}]
			必须先确定 $a>0$ 或 $a<0$，再选用相应的最值规律。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性思维）：}}]
			习惯所有二次函数看顶点，忘记区间和开口影响。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（顶点误当最值）：}}]
			区间内只要有对称轴，就把顶点值当最值，没有比较端点。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（顶点需在区间内）：}}]
			顶点值只是候选最值，还需比较端点值；若对称轴不在区间内，顶点不参与最值。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（区间意识缺失）：}}]
			只关注函数整体最值，忽略区间限制。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（对称轴公式错误）：}}]
			将 $h$ 算错，如 $h=\frac{b}{2a}$ 或 $h=-\frac{b}{a}$，导致分类错误。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（牢记公式 $h=-\frac{b}{2a}$）：}}]
			对称轴公式必须准确记忆，注意负号和分母 $2a$。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆不牢）：}}]
			公式结构混淆，或者符号处理粗心。\n
	\end{description}
