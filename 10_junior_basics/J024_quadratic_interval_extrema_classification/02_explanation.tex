\begin{explanationbox}
	\n\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n二次函数在闭区间上的最值，由开口方向和对称轴到区间的相对位置共同决定：离对称轴越近，函数值越接近极值。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（开口决定单调趋势）：}]
			$a>0$ 时先减后增，$a<0$ 时先增后减。\n
		\item[\textbf{要点二（对称轴区分最值点）：}]
			对称轴在区间内时，顶点是极值点；对称轴在区间外时，最值只在端点。\n
		\item[\textbf{要点三（端点比较定最值）：}]
			最大值/最小值需在可能点中比较得出。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（抛物线轴对称性）}}\n\n抛物线关于对称轴对称，区间离对称轴越远，函数值偏离顶点越大。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（配方法直接求最值）}}\n\n通过 $f(x)=a(x-h)^2+k$ 可立即得到顶点纵坐标 $k$，再结合 $a$ 的符号及区间判断最值。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（高频应用场景）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（直接求值域）：} 给定二次函数与闭区间，直接套用分类求最值。\n
		\item \textbf{考法二（含参讨论）：} 已知最值或值域关系反求参数，需要分类讨论开口与对称轴。\n
		\item \textbf{考法三（恒成立转化）：} 将 $f(x)\ge 0$ 在区间上恒成立转化为最小值 $\ge 0$。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n\textbf{“二次函数区间最值，开口定向，对称轴定位，端点兜底”。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（单一函数最值）：} 求 $f(x)=x^2-4x+5$ 在 $[2,5]$ 上的最值。\n
		\item \textbf{场景二（含参函数最值讨论）：} 求 $f(x)=x^2-2ax+1$ 在 $[0,1]$ 上的最小值。\n
		\item \textbf{场景三（最值求参数）：} 已知函数在区间上的最大值或最小值，求参数值。\n
	\end{itemize}
	\n
\end{explanationbox}
