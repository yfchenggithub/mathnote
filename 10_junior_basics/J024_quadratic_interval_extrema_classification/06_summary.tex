\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n二次函数在闭区间上的最值由开口方向与对称轴到区间的位置共同决定。\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（二次函数）：} $f(x)=ax^2+bx+c$（$a\neq 0$）。\n
		\item \textbf{条件 2（闭区间）：} $[m,n]$（$m<n$）。\n
		\item \textbf{条件 3（对称轴公式）：} $h=-\frac{b}{2a}$。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（开口判断）：} 先看 $a$ 的符号，再决定用 $a>0$ 还是 $a<0$ 的结论。\n
		\item \textbf{检查项（顶点验证）：} 只有对称轴在区间内时才考虑顶点值。\n
		\item \textbf{检查项（端点比较）：} 最大值、最小值均需比较端点和可能的顶点值。\n
	\end{itemize}
	\n
\end{summarybox}
