\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（函数定义）：}}]
		      二次函数 $f(x)=ax^2+bx+c$，其中 $a\neq 0$，对称轴为
		      \[
			      h=-\frac{b}{2a}.
		      \]
		\item[\textbf{\textcolor{CStatement}{条件 2（给定区间）：}}]
		      闭区间 $[m,n]$，且 $m<n$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（开口向上最值）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      开口向上时，对称轴在区间内则顶点取最小值；最大值一定在两个端点中产生。

		      当 $a>0$ 时，
		      \[
			      y_{\min}=\begin{cases}
				      f(m), & h\le m, \\[2pt]
				      f(h), & m<h<n,  \\[2pt]
				      f(n), & h\ge n,
			      \end{cases}
			      \qquad
			      y_{\max}=\max\{f(m),f(n)\}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（开口向下最值）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      开口向下时，对称轴在区间内则顶点取最大值；最小值一定在两个端点中产生。

		      当 $a<0$ 时，
		      \[
			      y_{\max}=\begin{cases}
				      f(m), & h\le m, \\[2pt]
				      f(h), & m<h<n,  \\[2pt]
				      f(n), & h\ge n,
			      \end{cases}
			      \qquad
			      y_{\min}=\min\{f(m),f(n)\}.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（含参分类讨论）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      含参二次函数求闭区间最值时，应先确定开口方向，再比较对称轴与区间端点的位置，最后代入端点或顶点求值；若二次项系数也含参数，还必须先讨论 $a=0$ 与 $a\neq0$ 的情形。
	\end{description}
\end{statementbox}
