\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 求 $f(x)=x^2-4x+3$ 在 $[0,3]$ 上的最大值与最小值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求对称轴）：}]
			$h=-\frac{-4}{2}=2$。\n
			\par\textbf{理由：} 对称轴公式 $h=-\frac{b}{2a}$，代入 $a=1,b=-4$。\n
		\item[\textbf{第二步（判断开口）：}]
			$a=1>0$，开口向上。\n
			\par\textbf{理由：} 二次项系数为正。\n
		\item[\textbf{第三步（分类求最值）：}]
			因为 $h=2\in[0,3]$，所以最小值 $f(2)=4-8+3=-1$。最大值在端点比较：$f(0)=3$，$f(3)=9-12+3=0$，故最大值为 $3$。\n
			\par\textbf{理由：} 开口向上时，对称轴在区间内，最小值在顶点，最大值在离对称轴较远的端点 $0$ 处（因为 $|0-2|=2 > |3-2|=1$）。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 最小值 $-1$，最大值 $3$。\n\n

	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知 $f(x)=x^2+2ax+2$ 在 $[-1,1]$ 上的最小值为 $1$，求实数 $a$ 的值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求对称轴）：}]
			$h=-a$。\n
			\par\textbf{理由：} $h=-\frac{b}{2a}=-\frac{2a}{2}=-a$。\n
		\item[\textbf{第二步（分类讨论最小值）：}]
			\n 当 $a\ge1$ 时，函数在 $[-1,1]$ 递增，最小值 $f(-1)=3-2a=1$，解得 $a=1$；\\\n 当 $-1<a<1$ 时，对称轴在区间内，最小值 $f(-a)=2-a^2=1$，解得 $a=\pm1$，无符合条件的解；\\\n 当 $a\le-1$ 时，函数递减，最小值 $f(1)=3+2a=1$，解得 $a=-1$。\n
			\par\textbf{理由：} 开口向上时，最小值按对称轴与区间位置分三类讨论，分别求解并验证范围。\n
		\item[\textbf{第三步（写出答案）：}]
			$a=1$ 或 $a=-1$。\n
			\par\textbf{理由：} 两步合并不冲突。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $a=1$ 或 $a=-1$。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 求函数 $f(x)=x^2-2x+3$ 在区间 $[t,t+1]$ 上的最小值 $g(t)$。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求对称轴）：}]
			$h=1$。\n
			\par\textbf{理由：} $h=-\frac{b}{2a}=1$。\n
		\item[\textbf{第二步（分情况求最小值）：}]
			\n 若 $t+1\le1$ 即 $t\le0$，则区间在对称轴左侧，递减，最小值 $g(t)=f(t+1)=t^2+2$；\\\n 若 $t<1<t+1$ 即 $0<t<1$，则区间含对称轴，最小值 $g(t)=f(1)=2$；\\\n 若 $t\ge1$，则区间在对称轴右侧，递增，最小值 $g(t)=f(t)=t^2-2t+3$。\n
			\par\textbf{理由：} 开口向上时，最小值出现在顶点或端点，取决于对称轴是否在区间内。\n
		\item[\textbf{第三步（得出分段表达式）：}]
			\n
			\[
				\n g(t)=\n
				\begin{cases}
					\n t^2+2, & t\le0,\\\n 2, & 0<t<1,\\\n t^2-2t+3, & t\ge1.\n
			\end{cases}
				\n
			\]
			\n
			\par\textbf{理由：} 由第二步三种情况合成。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $g(t)$ 如分段所示。\n
\end{examplebox}
