\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}求 $f(x)=x^2-4x+3$ 在 $[0,3]$ 上的最大值与最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求对称轴）：}]
		      \[
			      h=-\frac{-4}{2\cdot1}=2.
		      \]
		      \par\textbf{理由：}对称轴公式为 $h=-\frac{b}{2a}$，这里 $a=1,b=-4$。
		\item[\textbf{第二步（判断开口）：}]
		      因为 $a=1>0$，所以抛物线开口向上。
		      \par\textbf{理由：}二次项系数为正，开口向上。
		\item[\textbf{第三步（求最小值）：}]
		      因为 $h=2\in[0,3]$，所以最小值在顶点处取得：
		      \[
			      f(2)=2^2-4\cdot2+3=-1.
		      \]
		      \par\textbf{理由：}开口向上且对称轴在区间内，顶点取最小值。
		\item[\textbf{第四步（求最大值）：}]
		      比较端点值：
		      \[
			      f(0)=3,\qquad f(3)=9-12+3=0.
		      \]
		      所以最大值为 $3$。
		      \par\textbf{理由：}开口向上时，区间最大值一定在两个端点中产生。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}最小值为 $-1$，最大值为 $3$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}已知 $f(x)=x^2+2ax+2$ 在 $[-1,1]$ 上的最小值为 $1$，求实数 $a$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求对称轴）：}]
		      \[
			      h=-\frac{2a}{2}=-a.
		      \]
		      \par\textbf{理由：}对称轴公式为 $h=-\frac{b}{2a}$，此处二次项系数为 $1$，一次项系数为 $2a$。
		\item[\textbf{第二步（分类讨论）：}]
		      当 $a\ge1$ 时，$h=-a\le-1$，函数在 $[-1,1]$ 上递增，最小值为
		      \[
			      f(-1)=1-2a+2=3-2a.
		      \]
		      由 $3-2a=1$ 得 $a=1$，符合 $a\ge1$。

		      当 $-1<a<1$ 时，$h=-a\in(-1,1)$，最小值为
		      \[
			      f(-a)=a^2-2a^2+2=2-a^2.
		      \]
		      由 $2-a^2=1$ 得 $a=\pm1$，均不符合 $-1<a<1$。

		      当 $a\le-1$ 时，$h=-a\ge1$，函数在 $[-1,1]$ 上递减，最小值为
		      \[
			      f(1)=1+2a+2=3+2a.
		      \]
		      由 $3+2a=1$ 得 $a=-1$，符合 $a\le-1$。
		      \par\textbf{理由：}开口向上，最小值位置由对称轴与区间的位置决定。
		\item[\textbf{第三步（写出答案）：}]
		      \[
			      a=1\quad\text{或}\quad a=-1.
		      \]
		      \par\textbf{理由：}分类求出的有效解合并即可。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}$a=1$ 或 $a=-1$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}求函数 $f(x)=x^2-2x+3$ 在区间 $[t,t+1]$ 上的最小值 $g(t)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求对称轴）：}]
		      \[
			      h=-\frac{-2}{2\cdot1}=1.
		      \]
		      \par\textbf{理由：}对称轴公式为 $h=-\frac{b}{2a}$。
		\item[\textbf{第二步（分情况求最小值）：}]
		      若 $t+1\le1$，即 $t\le0$，则区间在对称轴左侧，函数递减，最小值为
		      \[
			      g(t)=f(t+1)=(t+1)^2-2(t+1)+3=t^2+2.
		      \]

		      若 $t<1<t+1$，即 $0<t<1$，则区间包含对称轴，最小值为
		      \[
			      g(t)=f(1)=2.
		      \]

		      若 $t\ge1$，则区间在对称轴右侧，函数递增，最小值为
		      \[
			      g(t)=f(t)=t^2-2t+3.
		      \]
		      \par\textbf{理由：}开口向上时，最小值出现在顶点或离顶点最近的端点。
		\item[\textbf{第三步（得出分段表达式）：}]
		      \[
			      g(t)=\begin{cases}
				      t^2+2,    & t\le0, \\[2pt]
				      2,        & 0<t<1, \\[2pt]
				      t^2-2t+3, & t\ge1.
			      \end{cases}
		      \]
		      \par\textbf{理由：}由第二步三种位置关系合并得到。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}$g(t)$ 如上式所示。
\end{examplebox}
