\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（角平分线条件）：}}]
			射线 $OC$ 平分 $\angle AOB$，点 $P$ 在 $OC$ 上。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（垂两边模型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过角平分线上一点向角两边作垂线，垂线段长度相等。
			\[
				PE\perp OA,\ PF\perp OB \Rightarrow PE=PF.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（截等长模型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在角的两边截取相等线段，可得全等三角形。
			\[
				OD=OE \Rightarrow \triangle ODP\cong\triangle OEP.
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（作平行模型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过角平分线上一点作一边的平行线，构成等腰三角形。
			\[
				PA\parallel OB \Rightarrow OA=AP.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（等腰三线合一）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			等腰三角形顶角平分线也是底边中线和高。
			\[
				AB=AC,\ \angle BAD=\angle CAD \Rightarrow AD\perp BC,\ BD=CD.
			\]
	\end{description}

\end{statementbox}
