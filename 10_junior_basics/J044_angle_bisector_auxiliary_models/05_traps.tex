\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（垂两边误用）}}]
			认为角平分线上一点到角两边任意连线都相等。
		\item[\textbf{\textcolor{CTrap}{正确理解（距离相等）：}}]
			只有作垂线得到的垂线段才相等，即点到角两边的距离相等。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略垂直）：}}]
			未注意到“距离”要求垂直，错误推广到一般线段。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（截等长条件）}}]
			在角两边截取等长线段后，直接认为两个三角形全等，忽略点不在角平分线上的情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（角分线必备）：}}]
			必须确保所连点 $P$ 在角平分线上，且 $OP$ 是公共边，才能用 SAS 证全等。
		\item[\textbf{\textcolor{CTrap}{错因分析（漏条件）：}}]
			缺少“$P$ 在角平分线上”这一关键条件。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（三线合一误用）}}]
			在普通三角形中，如果一条线既是角平分线又是中线，就以为它是等腰三角形。
		\item[\textbf{\textcolor{CTrap}{正确理解（判定与性质）：}}]
			“三线合一”是等腰三角形的性质，如果已知等腰，可得三者合一；反之，如果已知两条合一，可推出等腰，但不能直接认为三线合一自动成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆性质）：}}]
			把性质当判定使用，或忘记需要先证明等腰。
	\end{description}

\end{trapbox}
