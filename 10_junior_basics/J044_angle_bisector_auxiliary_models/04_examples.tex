\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，$AD$ 平分 $\angle BAC$，$DE\perp AB$ 于点 $E$，$DF\perp AC$ 于点 $F$。求证：$DE=DF$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（标定条件）：}]
			由 $AD$ 平分 $\angle BAC$ 得 $\angle EAD=\angle FAD$；由 $DE\perp AB$，$DF\perp AC$ 得 $\angle AED=\angle AFD=90^\circ$。
			\par\textbf{理由：} 角平分线定义；垂直定义。
		\item[\textbf{第二步（证全等）：}]
			在 $\triangle ADE$ 和 $\triangle ADF$ 中，
			\[
				\angle EAD = \angle FAD,\quad \angle AED = \angle AFD = 90^\circ,\quad AD = AD.
			\]
			\par\textbf{理由：} 第一步所得条件；公共边。所以 $\triangle ADE \cong \triangle ADF$（AAS）。
		\item[\textbf{第三步（得结论）：}]
			由全等得 $DE=DF$。
			\par\textbf{理由：} 全等三角形对应边相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{角平分线上一点到角两边的距离相等。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，$AD$ 平分 $\angle BAC$，点 $E$ 在 $AB$ 上，且 $AE=AC$，连接 $DE$。求证：$\triangle ADE \cong \triangle ADC$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（找条件）：}]
			已知 $AD$ 平分 $\angle BAC$，所以 $\angle EAD=\angle CAD$；又 $AE=AC$，$AD$ 是公共边。
			\par\textbf{理由：} 角平分线定义；题目所给条件。
		\item[\textbf{第二步（证全等）：}]
			在 $\triangle ADE$ 和 $\triangle ADC$ 中，
			\[
				AE = AC,\quad \angle EAD = \angle CAD,\quad AD = AD.
			\]
			\par\textbf{理由：} 第一步条件；SAS判定定理。所以 $\triangle ADE \cong \triangle ADC$（SAS）。
		\item[\textbf{第三步（得结论）：}]
			全等直接得 $\angle AED=\angle ACD$，$DE=DC$ 等。
			\par\textbf{理由：} 全等三角形对应边、角相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{在角平分线两侧截取等长线段，可构造全等三角形。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在等腰 $\triangle ABC$ 中，$AB=AC$，$AD$ 平分 $\angle BAC$，交 $BC$ 于点 $D$。过点 $D$ 作 $DE\perp AB$ 于 $E$，$DF\perp AC$ 于 $F$。求证：$BE=CF$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（三线合一）：}]
			在等腰 $\triangle ABC$ 中，$AB=AC$，$AD$ 平分 $\angle BAC$，由三线合一得 $AD\perp BC$，$BD=CD$。
			\par\textbf{理由：} 等腰三角形顶角平分线也是底边上的中线和高。
		\item[\textbf{第二步（垂两边得等距）：}]
			由 $AD$ 平分 $\angle BAC$，$DE\perp AB$，$DF\perp AC$，根据角平分线性质得 $DE=DF$。
			\par\textbf{理由：} 角平分线上一点到角两边距离相等。
		\item[\textbf{第三步（证全等得边等）：}]
			在 $\triangle BDE$ 和 $\triangle CDF$ 中，
			\[
				BD = CD,\quad DE = DF,\quad \angle BED = \angle CFD = 90^\circ.
			\]
			\par\textbf{理由：} 第一步得 $BD=CD$，第二步得 $DE=DF$，$DE\perp AB$，$DF\perp AC$ 得直角。所以 $\triangle BDE \cong \triangle CDF$（HL），从而 $BE=CF$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{综合运用三线合一和角平分线性质，可推导线段相等。}
\end{examplebox}
