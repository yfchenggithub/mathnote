\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（二次项非零）：}}] $y=ax^2+bx+c$ 中 $a\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（顶点与对称轴）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直接计算顶点坐标与对称轴。
			\[
				\left(-\frac{b}{2a},\frac{4ac-b^2}{4a}\right)
			\]
			\[
				x = -\frac{b}{2a}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（顶点式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			一般式化为顶点式，便于平移分析。
			\[
				y = a(x-h)^2+k,\quad h=-\frac{b}{2a},\; k=\frac{4ac-b^2}{4a}
			\]
	\end{description}
\end{statementbox}
