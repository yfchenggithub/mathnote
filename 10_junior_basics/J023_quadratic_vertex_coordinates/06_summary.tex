\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{二次函数顶点公式 $(-\frac{b}{2a},\frac{4ac-b^2}{4a})$ 直接给出抛物线的顶点坐标和对称轴。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（二次项系数非零）：} 函数必须为二次函数，即 $a\neq 0$。
		\item \textbf{条件 2（一般式形式）：} 函数必须写成 $y=ax^2+bx+c$ 的形式，否则需先转化。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（纵坐标符号）：} 纵坐标 $k=\frac{4ac-b^2}{4a}$，切勿错写成 $\frac{4ac+b^2}{4a}$ 或 $\frac{b^2-4ac}{4a}$。
		\item \textbf{检查项（配方验证）：} 对结果有疑问时，可用配方法验证顶点公式。
	\end{itemize}
\end{summarybox}
