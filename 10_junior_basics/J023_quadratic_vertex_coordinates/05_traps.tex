\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（顶点横坐标符号）}}] 容易将顶点横坐标误写为 $\dfrac{b}{2a}$，漏写负号。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号规则）：}}] 顶点横坐标应为 $-\dfrac{b}{2a}$，负号不能省略。
		\item[\textbf{\textcolor{CTrap}{错因分析（负号遗漏）：}}] 记忆公式不熟练，或混淆对称轴与 $x$ 轴截距。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（纵坐标公式符号）}}] 常见错误：$k = \dfrac{4ac+b^2}{4a}$ 或 $k = \dfrac{b^2-4ac}{4a}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（分子顺序）：}}] 纵坐标公式为 $k = \dfrac{4ac-b^2}{4a}$，分子是 $4ac$ 减 $b^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（与判别式混淆）：}}] 判别式 $\Delta = b^2-4ac$ 形式相似但符号相反，容易混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略$a\neq0$）}}] 在应用顶点公式时，未检查 $a$ 是否为零，直接当作二次函数处理。
		\item[\textbf{\textcolor{CTrap}{正确理解（二次函数定义）：}}] 二次函数要求 $a\neq 0$，否则公式不适用。
		\item[\textbf{\textcolor{CTrap}{错因分析（默认二次）：}}] 看到 $ax^2+bx+c$ 形式就默认是二次，未验证 $a\neq 0$。
	\end{description}
\end{trapbox}
