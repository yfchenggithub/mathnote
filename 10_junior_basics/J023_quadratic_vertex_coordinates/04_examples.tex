\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	求二次函数 $y=2x^2-4x+1$ 的顶点坐标和对称轴。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别系数）：}] $a=2$，$b=-4$，$c=1$。
			\par\textbf{理由：} 顶点公式直接依赖于一般式的系数。
		\item[\textbf{第二步（求横坐标）：}] 计算：
			\[
				\begin{aligned}
					h &= -\frac{b}{2a} \\
					&= -\frac{-4}{2\times 2} \\
					&= 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 对称轴坐标公式。
		\item[\textbf{第三步（求纵坐标）：}] 计算：
			\[
				\begin{aligned}
					k &= \frac{4ac-b^2}{4a} \\
					&= \frac{4\times 2\times 1 - (-4)^2}{4\times 2} \\
					&= \frac{8-16}{8} \\
					&= -1.
			\end{aligned}
			\]
			\par\textbf{理由：} 顶点纵坐标公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	顶点坐标为 $(1,-1)$，对称轴为直线 $x=1$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知二次函数 $y = x^2 + bx + c$ 的顶点在 $x$ 轴上，且对称轴为 $x=2$，求 $b$ 和 $c$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由对称轴求 $b$）：}] 对称轴公式 $x=-\dfrac{b}{2a}$，$a=1$，代入 $x=2$：$-\dfrac{b}{2}=2$，解得 $b=-4$。
			\par\textbf{理由：} 对称轴 $x= -\frac{b}{2a}$ 直接给出关系。
		\item[\textbf{第二步（由顶点纵坐标求 $c$）：}] 顶点纵坐标 $k = \dfrac{4ac-b^2}{4a}$，由顶点在 $x$ 轴得 $k=0$。代入 $a=1,b=-4$：$\dfrac{4c-16}{4}=0$，得 $c=4$。
			\par\textbf{理由：} 顶点在 $x$ 轴上意味着纵坐标为 $0$。
		\item[\textbf{第三步（验证）：}] $y=x^2-4x+4$，配方得 $y=(x-2)^2$，顶点 $(2,0)$，对称轴 $x=2$，符合题意。
			\par\textbf{理由：} 可通过配方验证结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$b=-4$，$c=4$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知二次函数 $y = x^2 - 2mx + m^2 + 3$，求顶点坐标（用 $m$ 表示），并说明顶点始终在哪条直线上。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别系数）：}] $a=1$，$b=-2m$，$c=m^2+3$。
			\par\textbf{理由：} 套用顶点公式前需提取系数。
		\item[\textbf{第二步（求横坐标）：}] 计算：
			\[
				\begin{aligned}
					h &= -\frac{b}{2a} \\
					&= -\frac{-2m}{2} \\
					&= m.
			\end{aligned}
			\]
			\par\textbf{理由：} 顶点横坐标公式。
		\item[\textbf{第三步（求纵坐标）：}] 计算：
			\[
				\begin{aligned}
					k &= \frac{4ac-b^2}{4a} \\
					&= \frac{4(m^2+3) - (2m)^2}{4} \\
					&= \frac{12}{4} \\
					&= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 顶点纵坐标公式，化简后与 $m$ 无关。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	顶点坐标为 $(m,3)$，始终在直线 $y=3$ 上。

\end{examplebox}
