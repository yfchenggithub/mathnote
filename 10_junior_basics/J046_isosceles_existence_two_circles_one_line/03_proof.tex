\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	等腰三角形两边相等转化为距离等式，进而得到直线或圆方程，与轨迹联立并剔除共线。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（距离公式）：}]
			设 $A(x_1,y_1), B(x_2,y_2), P(x,y)$，则 $|PA|=\sqrt{(x-x_1)^2+(y-y_1)^2}$，$|PB|=\sqrt{(x-x_2)^2+(y-y_2)^2}$，$|AB|=d=\sqrt{(x_1-x_2)^2+(y_1-y_2)^2}$。
			\par\textbf{理由：} 两点间距离公式。
		\item[\textbf{步骤二（等腰条件分类）：}]
			$\triangle ABP$ 为等腰三角形等价于 $|PA|=|PB|$ 或 $|PA|=|AB|$ 或 $|PB|=|AB|$。
			\par\textbf{理由：} 等腰三角形定义，两边相等。
		\item[\textbf{步骤三（平方得代数方程）：}]
			两边平方得到三个方程：
			\[
				\begin{aligned}
					&(x-x_1)^2+(y-y_1)^2 = (x-x_2)^2+(y-y_2)^2,\\ &(x-x_1)^2+(y-y_1)^2 = d^2,\\ &(x-x_2)^2+(y-y_2)^2 = d^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方消去根号，(i)可化简为直线（中垂线），(ii)(iii)为圆。
		\item[\textbf{步骤四（与轨迹联立）：}]
			将上述每个方程分别与轨迹方程 $f(x,y)=0$ 联立，得到候选点集。
			\par\textbf{理由：} $P$ 必须同时满足轨迹和等腰条件。
		\item[\textbf{步骤五（剔除共线解）：}]
			对每个候选点 $P$，若 $(x-x_1)(y_2-y_1) = (y-y_1)(x_2-x_1)$，则 $A,B,P$ 共线，舍去该解。
			\par\textbf{理由：} 共线时三角形退化。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{因此，采用两圆一线法联立求解并剔除共线解即得所有满足条件的点 $P$。}
\end{proofbox}
