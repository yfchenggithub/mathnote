\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(0,0)$，$B(4,0)$，点 $P$ 在直线 $x=0$ 上，求使 $\triangle ABP$ 为等腰三角形的点 $P$ 坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写基本量）：}]
			$d=|AB|=4$，轨迹方程为 $x=0$。
			\par\textbf{理由：} 两点间距离。
		\item[\textbf{第二步（中垂线情形）：}]
			$PA=PB$ 得中垂线 $x=2$，与 $x=0$ 无交点，无解。
			\par\textbf{理由：} 中垂线与轨迹平行。
		\item[\textbf{第三步（圆 $A$ 情形）：}]
			$PA=AB$ 得 $x^2+y^2=16$，代入 $x=0$ 得 $y=\pm4$，故 $P(0,4)$ 或 $(0,-4)$。
			\par\textbf{理由：} 联立求解。
		\item[\textbf{第四步（圆 $B$ 情形）：}]
			$PB=AB$ 得 $(x-4)^2+y^2=16$，代入 $x=0$ 得 $16+y^2=16$，$y=0$，得 $P(0,0)$。
			\par\textbf{理由：} 联立求解。
		\item[\textbf{第五步（剔除共线）：}]
			$P(0,0)$ 与 $A$ 重合且三点共线，剔除。有效点为 $P(0,4)$ 和 $P(0,-4)$。
			\par\textbf{理由：} 共线退化。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\triangle ABP$ 为等腰时 $P$ 坐标为 $(0,4)$ 或 $(0,-4)$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(0,0)$，$B(6,0)$，点 $P$ 在圆 $C:(x-3)^2+y^2=9$ 上，求使 $\triangle ABP$ 为等腰三角形的点 $P$ 坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写基本量）：}]
			$d=6$，轨迹圆 $C$。
			\par\textbf{理由：} 距离公式。
		\item[\textbf{第二步（中垂线情形）：}]
			$PA=PB$ 得中垂线 $x=3$，与 $C$ 联立得 $y^2=9$，$y=\pm3$，得 $P(3,3)$ 和 $(3,-3)$，不共线。
			\par\textbf{理由：} 代入圆方程求解。
		\item[\textbf{第三步（圆 $A$ 情形）：}]
			$PA=AB$ 得圆 $A:x^2+y^2=36$。与 $C$ 相减得 $6x-9=27$，$x=6$，代入得 $y=0$，得 $P(6,0)$ (与 $B$ 重合，共线)。
			\par\textbf{理由：} 两圆方程相减得直线，再代入。
		\item[\textbf{第四步（圆 $B$ 情形）：}]
			$PB=AB$ 得圆 $B:(x-6)^2+y^2=36$。与 $C$ 相减得 $-6x+27=27$，$x=0$，代入得 $y=0$，得 $P(0,0)$ (与 $A$ 重合，共线)。
			\par\textbf{理由：} 同上。
		\item[\textbf{第五步（剔除共线）：}]
			$P(6,0)$ 和 $P(0,0)$ 共线且与定点重合，剔除。有效点为 $P(3,3)$ 和 $(3,-3)$。
			\par\textbf{理由：} 共线退化。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	满足条件的点 $P$ 坐标为 $(3,3)$ 或 $(3,-3)$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(0,0)$，$B(2,0)$，点 $P$ 在抛物线 $y^2=2x$ 上，求使 $\triangle ABP$ 为等腰三角形的点 $P$ 坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写基本量）：}]
			$d=2$，抛物线 $y^2=2x$。
			\par\textbf{理由：} 距离和轨迹。
		\item[\textbf{第二步（中垂线情形）：}]
			$PA=PB$ 得中垂线 $x=1$，代入抛物线得 $y^2=2$，$y=\pm\sqrt{2}$，得 $P_1(1,\sqrt{2}), P_2(1,-\sqrt{2})$，不共线。
			\par\textbf{理由：} 联立求解。
		\item[\textbf{第三步（圆 $A$ 情形）：}]
			$PA=AB$ 得圆 $A:x^2+y^2=4$，与 $y^2=2x$ 联立消 $y$ 得 $x^2+2x-4=0$，解得 $x=-1+\sqrt{5}$（正根），进而 $y=\pm\sqrt{2\sqrt{5}-2}$，得 $P_{3,4}$，$y\neq0$，不共线。
			\par\textbf{理由：} 代入消元，解二次方程。
		\item[\textbf{第四步（圆 $B$ 情形）：}]
			$PB=AB$ 得圆 $B:(x-2)^2+y^2=4$，与 $y^2=2x$ 联立得 $x^2-2x=0$，$x=0$ 或 $x=2$。 \par $x=0$ 得 $P(0,0)$（与 $A$ 重合，共线，舍）；$x=2$ 得 $y=\pm2$，即 $P_5(2,2), P_6(2,-2)$，不共线。
			\par\textbf{理由：} 代入消元，解二次方程。
		\item[\textbf{第五步（汇总）：}]
			有效点 $P_1,P_2,P_{3,4},P_5,P_6$ 共六个。
			\par\textbf{理由：} 所有解均不共线（$P_{3,4}$ 纵坐标非零，其余不在 $x$ 轴）。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	共有六个点，坐标如上。
