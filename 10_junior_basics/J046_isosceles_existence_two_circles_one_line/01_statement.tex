\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两定点与轨迹）：}}]
			设两定点 $A(x_1,y_1)$, $B(x_2,y_2)$，距离 $d = |AB| = \sqrt{(x_1-x_2)^2+(y_1-y_2)^2}$。动点 $P(x,y)$ 满足轨迹方程 $f(x,y)=0$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{情形一（中垂线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$P$ 到 $A,B$ 距离相等，即 $P$ 在线段 $AB$ 的中垂线上。
			\[
				(x-x_1)^2+(y-y_1)^2 = (x-x_2)^2+(y-y_2)^2
			\]
		\item[\textbf{\textcolor{CStatement}{情形二（圆 $A$）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$PA = AB$，即 $P$ 在以 $A$ 为圆心、$|AB|$ 为半径的圆上。
			\[
				(x-x_1)^2+(y-y_1)^2 = d^2
			\]
		\item[\textbf{\textcolor{CStatement}{情形三（圆 $B$）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$PB = AB$，即 $P$ 在以 $B$ 为圆心、$|AB|$ 为半径的圆上。
			\[
				(x-x_2)^2+(y-y_2)^2 = d^2
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（解的存在性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点 $P$ 的候选位置为轨迹 $f(x,y)=0$ 与三种距离等式所确定曲线的交点，需剔除共线情形。
			\[
				\left\{ (x,y) \;\middle|\; f(x,y)=0 \land
				\begin{aligned}
					&\bigl( (x-x_1)^2+(y-y_1)^2 = (x-x_2)^2+(y-y_2)^2 \bigr) \\ &\quad \lor \bigl( (x-x_1)^2+(y-y_1)^2 = d^2 \bigr) \\ &\quad \lor \bigl( (x-x_2)^2+(y-y_2)^2 = d^2 \bigr)
			\end{aligned}
				\,\right\}.
			\]
	\end{description}

\end{statementbox}
