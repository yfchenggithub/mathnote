\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{等腰三角形存在性问题，转化为“两圆一线”与轨迹联立，并剔除共线解。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（两定点坐标）：} 已知两定点 $A,B$ 坐标，可求距离 $d=|AB|$。
		\item \textbf{条件2（动点轨迹方程）：} 动点 $P$ 满足轨迹方程 $f(x,y)=0$。
		\item \textbf{条件3（非退化要求）：} 最后需检验 $A,B,P$ 不共线。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分类完整）：} 必须考虑 $PA=PB$、$PA=AB$、$PB=AB$ 三种情况，不可遗漏。
		\item \textbf{检查项（共线检验）：} 求出坐标后务必验证三点是否共线，防止退化三角形。
	\end{itemize}

\end{summarybox}
