\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在四边形 $ABCD$ 中，$\angle A=80^\circ$，$\angle C=100^\circ$，$\angle B=70^\circ$。判断 $A,B,C,D$ 是否四点共圆，并求 $\angle D$ 的度数。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证互补）：}]
			计算 $\angle A+\angle C=80^\circ+100^\circ=180^\circ$。
			\par\textbf{理由：} 对角互补条件为 $180^\circ$。
		\item[\textbf{第二步（判定共圆）：}]
			因为 $\angle A+\angle C=180^\circ$，所以 $A,B,C,D$ 四点共圆。
			\par\textbf{理由：} 对角互补判定定理。
		\item[\textbf{第三步（求角度）：}]
			由圆内接四边形性质，$\angle D=180^\circ-\angle B=180^\circ-70^\circ=110^\circ$。
			\par\textbf{理由：} 对角互补。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle D=110^\circ$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在四边形 $ABCD$ 中，$\angle ABC+\angle ADC=180^\circ$，$AB=AD$，$\angle BAC=30^\circ$，$\angle BCA=20^\circ$。将 $\triangle ABC$ 绕点 $A$ 旋转，使 $AB$ 与 $AD$ 重合，点 $C$ 落在射线 $CD$ 上的点 $E$。求 $\angle ADE$ 的大小。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定互补）：}]
			已知 $\angle ABC+\angle ADC=180^\circ$，满足一组对角互补。
			\par\textbf{理由：} 条件直接给出。
		\item[\textbf{第二步（旋转全等）：}]
			由 $AB=AD$ 且对角互补，旋转得 $\triangle ABC\cong\triangle ADE$，$E$ 在 $CD$ 上。
			\par\textbf{理由：} 旋转全等结论。
		\item[\textbf{第三步（求角度）：}]
			在 $\triangle ABC$ 中，$\angle ABC=180^\circ-\angle BAC-\angle BCA=180^\circ-30^\circ-20^\circ=130^\circ$。由全等，$\angle ADE=\angle ABC=130^\circ$。
			\par\textbf{理由：} 全等对应角相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle ADE=130^\circ$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在四边形 $ABCD$ 中，$\angle ABC+\angle ADC=180^\circ$，$AB=AD$，$\angle BAD=60^\circ$，$BC=4$，$CD=2$。将 $\triangle ABC$ 绕点 $A$ 旋转，使 $AB$ 与 $AD$ 重合，点 $C$ 落在射线 $CD$ 上的点 $E$。求 $AC$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（旋转全等）：}]
			由 $AB=AD$ 且 $\angle ABC+\angle ADC=180^\circ$，旋转得 $\triangle ABC\cong\triangle ADE$，故 $DE=BC=4$，$AE=AC$。
			\par\textbf{理由：} 旋转全等结论。
		\item[\textbf{第二步（求线段）：}]
			因为 $C,D,E$ 共线，所以 $CE=CD+DE=2+4=6$。
			\par\textbf{理由：} 旋转后三点共线。
		\item[\textbf{第三步（等边推AC）：}]
			旋转角 $\angle CAE=\angle BAD=60^\circ$，且 $AC=AE$，故 $\triangle ACE$ 是等边三角形，所以 $AC=CE=6$。
			\par\textbf{理由：} 等腰三角形顶角 $60^\circ$ 推出等边。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AC=6$}
\end{examplebox}
