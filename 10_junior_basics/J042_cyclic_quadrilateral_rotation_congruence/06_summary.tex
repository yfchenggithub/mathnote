\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{对角互补判四点共圆；邻边相等加对角互补可旋转构造全等。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（对角互补）：} 四边形一组对角之和为 $180^\circ$。
		\item \textbf{条件 2（邻边相等）：} 四边形有一组邻边相等（如 $AB=AD$）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略共线）：} 旋转全等时需保证 $\angle B+\angle D=180^\circ$，否则 $E$ 不在 $CD$ 上。
		\item \textbf{检查项（对角判定）：} 判定四点共圆时确认是对角互补，而不是邻角。
	\end{itemize}

\end{summarybox}
