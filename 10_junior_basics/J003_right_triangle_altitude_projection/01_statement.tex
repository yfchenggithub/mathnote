\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（直角斜高）：}}]
			在 $\triangle ABC$ 中，$\angle C = 90^{\circ}$，$CD \perp AB$ 于 $D$。
		\item[\textbf{\textcolor{CStatement}{条件2（线段记号）：}}]
			记 $AD = m$，$DB = n$，$CD = h$，$AC = b$，$BC = a$，$AB = c$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（高平方）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			斜边上的高是两段斜边线段的比例中项。
			\[
				CD^{2} = AD \cdot DB.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（直角边平方一）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			一条直角边是其在斜边上的射影与斜边的比例中项。
			\[
				AC^{2} = AD \cdot AB.
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（直角边平方二）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			另一条直角边是其在斜边上的射影与斜边的比例中项。
			\[
				BC^{2} = BD \cdot AB.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（相似传递）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两小直角三角形分别与原直角三角形相似，且彼此相似。
			\[
				\triangle ACD \sim \triangle ABC \sim \triangle CBD.
			\]
			\textbf{\textcolor{CStatement}{比例关系：}}
			对应边成比例，如
			\[
				\frac{AC}{AD} = \frac{AB}{AC},\quad \frac{BC}{BD} = \frac{AB}{BC},\quad \frac{CD}{AD} = \frac{BD}{CD}.
			\]
	\end{description}

\end{statementbox}
