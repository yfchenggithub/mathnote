\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记错）：}}]
			误写为 $CD = AD \cdot DB$（忘记平方）或 $CD^2 = AD \cdot AB$（混淆对应边）。
		\item[\textbf{\textcolor{CTrap}{正确理解（平方关系）：}}]
			射影定理中等号左边是线段的平方，右侧是两条线段的乘积，不能丢掉平方。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆不牢）：}}]
			没有记住平方特征，或相似比例式对应边写错。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（条件忽视）：}}]
			在非直角三角形中应用射影定理，或垂足不在斜边上。
		\item[\textbf{\textcolor{CTrap}{正确理解（严格前提）：}}]
			定理只适用于直角三角形且直角顶点到斜边的垂线。
		\item[\textbf{\textcolor{CTrap}{错因分析（审图不细）：}}]
			解题前未检查角度和高线位置，盲目套用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开方出错）：}}]
			解方程 $CD^2 = 36$ 时误写 $CD = \pm 6$，线段长度取负。
		\item[\textbf{\textcolor{CTrap}{正确理解（非负性质）：}}]
			线段长度不能为负，开方后只取算术平方根。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略几何意义）：}}]
			忘记 $CD$ 是线段，长度必须非负。
	\end{description}

\end{trapbox}
