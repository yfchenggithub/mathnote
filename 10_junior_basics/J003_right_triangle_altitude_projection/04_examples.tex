\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在直角三角形 $ABC$ 中，$\angle C = 90^\circ$，$CD \perp AB$ 于 $D$。已知 $AD = 4$，$DB = 9$，求 $CD$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}]
			确认图形符合射影定理模型：$\triangle ABC$ 是直角三角形，$CD$ 是斜边上的高，已知两段 $AD$，$DB$。
			\par\textbf{理由：} 射影定理适用条件是直角三角形且斜边上有高。
		\item[\textbf{第二步（套用公式）：}]
			由射影定理 $CD^2 = AD \cdot DB = 4 \times 9 = 36$。
			\par\textbf{理由：} 射影定理第一式：斜边上的高的平方等于两段乘积。
		\item[\textbf{第三步（求结果）：}]
			开方得 $CD = \sqrt{36} = 6$（线段长度取正值）。
			\par\textbf{理由：} $CD$ 是线段长，非负，故取正根。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$CD = 6$}

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在直角三角形 $ABC$ 中，$\angle C = 90^\circ$，$CD \perp AB$ 于 $D$。已知 $AC = 5$，$AB = 13$，求 $AD$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分析已知量）：}]
			题目给出斜边 $AB$ 和一条直角边 $AC$，要求 $AD$（斜边上一段）。
			\par\textbf{理由：} $AD$ 是 $AC$ 在斜边上的射影，由 $AC^2 = AD \cdot AB$ 可解。
		\item[\textbf{第二步（列方程）：}]
			设 $AD = x$，代入射影定理第二式：$5^2 = x \cdot 13$，得 $25 = 13x$。
			\par\textbf{理由：} 应用 $AC^2 = AD \cdot AB$。
		\item[\textbf{第三步（求解）：}]
			解得 $x = \dfrac{25}{13}$。
			\par\textbf{理由：} 等式两边除以 $13$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AD = \dfrac{25}{13}$}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在直角三角形 $ABC$ 中，$\angle C = 90^\circ$，$CD \perp AB$ 于 $D$。已知 $CD = 6$，$AD = 4$，求 $AC, BC$ 的长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求另一段）：}]
			由射影定理第一式：$CD^2 = AD \cdot BD$，即 $36 = 4 \cdot BD$，解得 $BD = 9$。
			\par\textbf{理由：} 高平方等于两段乘积。
		\item[\textbf{第二步（求斜边长）：}]
			$AB = AD + BD = 4 + 9 = 13$。
			\par\textbf{理由：} $AB$ 被 $D$ 分成两段。
		\item[\textbf{第三步（求直角边）：}]
			由 $AC^2 = AD \cdot AB = 4 \times 13 = 52$，得 $AC = \sqrt{52} = 2\sqrt{13}$；由 $BC^2 = BD \cdot AB = 9 \times 13 = 117$，得 $BC = \sqrt{117} = 3\sqrt{13}$。
			\par\textbf{理由：} 分别应用射影定理第二、三式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AC = 2\sqrt{13}$，$BC = 3\sqrt{13}$}
\end{examplebox}
