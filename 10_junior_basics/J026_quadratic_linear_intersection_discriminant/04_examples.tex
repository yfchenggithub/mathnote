\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 判断二次函数 $y=x^{2}+1$ 与直线 $y=2x$ 的图象交点个数。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（代入联立）：}]
			将 $y=2x$ 代入 $y=x^{2}+1$，得\n
			\[
				\n x^{2}+1 = 2x.\n
			\]
			\n
			\par\textbf{理由：} 联立方程组，消去 $y$，得到关于 $x$ 的方程。\n\n
		\item[\textbf{第二步（移项化简）：}]
			移项并整理，得\n
			\[
				\n x^{2}-2x+1 = 0.\n
			\]
			\n
			\par\textbf{理由：} 化为标准一元二次方程形式。\n\n
		\item[\textbf{第三步（计算判别式）：}]
			计算判别式\n
			\[
				\n
				\begin{aligned}
					\Delta &= (-2)^{2} - 4 \times 1 \times 1 \\ &= 4 - 4 \\ &= 0.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} $\Delta=0$ 说明方程有一个重根，因此图象有一个交点（相切）。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $\Delta=0$，故图象有一个交点，即直线与抛物线相切。\n\n

	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知二次函数 $y=x^{2}-2x+1$ 与直线 $y=2x+m$ 的图象有两个交点，求实数 $m$ 的取值范围。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（联立整理）：}]
			代入 $y=2x+m$，移项得\n
			\[
				\n x^{2}-4x+(1-m) = 0.\n
			\]
			\n
			\par\textbf{理由：} 由 $y=x^{2}-2x+1$ 与 $y=2x+m$ 联立，消去 $y$ 并整理得到该方程。\n\n
		\item[\textbf{第二步（计算判别式）：}]
			\n
			\[
				\n
				\begin{aligned}
					\Delta &= (-4)^{2} - 4 \times 1 \times (1-m) \\ &= 16 - 4 + 4m \\ &= 12 + 4m.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 判别式公式 $\Delta = B^{2}-4AC$，其中 $A=1,\,B=-4,\,C=1-m$。\n\n
		\item[\textbf{第三步（列不等式求解）：}]
			由题意需 $\Delta>0$，即\n
			\[
				\n 12+4m > 0.\n
			\]
			\n
			\par\textbf{理由：} 两个交点对应方程有两个不等实根，所以 $\Delta>0$。解得 $m > -3$。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 当 $m > -3$ 时，二次函数与直线有两个交点。\n\n

	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知二次函数 $y=x^{2}-2kx+1$ 的图象与直线 $y=2x-1$ 有两个交点，且与 $x$ 轴无交点，求 $k$ 的取值范围。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（直线联立方程）：}]
			将 $y=2x-1$ 代入，移项得\n
			\[
				\n x^{2}-2(k+1)x+2 = 0.\n
			\]
			\n
			\par\textbf{理由：} 联立消去 $y$，得到关于 $x$ 的一元二次方程。\n\n
		\item[\textbf{第二步（直线判别式）：}]
			该方程的判别式为\n
			\[
				\n
				\begin{aligned}
					\Delta_{1} &= [-2(k+1)]^{2} - 4 \times 1 \times 2 \\ &= 4(k^{2}+2k-1).
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 两个交点要求该方程有两个不等实根，即 $\Delta_{1}>0$。\n\n
		\item[\textbf{第三步（x轴判别式）：}]
			与 $x$ 轴（$y=0$）联立，得 $x^{2}-2kx+1=0$，判别式\n
			\[
				\n
				\begin{aligned}
					\Delta_{2} &= (-2k)^{2} - 4 \\ &= 4(k^{2}-1).
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 与 $x$ 轴无交点要求该方程无实根，即 $\Delta_{2}<0$。\n\n
		\item[\textbf{第四步（解不等式组）：}]
			同时满足 $\Delta_{1}>0$ 和 $\Delta_{2}<0$，即\n
			\[
				\n k^{2}+2k-1 > 0,\quad k^{2}-1 < 0.\n
			\]
			\n
			\par\textbf{理由：} 解第一个不等式得 $k<-1-\sqrt{2}$ 或 $k>-1+\sqrt{2}$；解第二个得 $-1<k<1$。取交集得 $-1+\sqrt{2} < k < 1$。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $k$ 的取值范围是 $(-1+\sqrt{2},\,1)$。\n
\end{examplebox}
