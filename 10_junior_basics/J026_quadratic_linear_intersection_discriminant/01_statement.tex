\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（函数形式）：}}]
		      二次函数 $y=ax^{2}+bx+c$（$a\neq0$）与直线 $y=kx+m$（包括 $x$ 轴 $y=0$），联立消去 $y$ 后得关于 $x$ 的一元二次方程
		      \[
			      ax^{2}+(b-k)x+(c-m)=0.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（交点判别）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      判别式
		      \[
			      \Delta=(b-k)^{2}-4a(c-m)
		      \]
		      的符号决定交点个数：当 $\Delta>0$ 时有两个交点；当 $\Delta=0$ 时有一个交点（相切）；当 $\Delta<0$ 时无交点。

		\item[\textbf{\textcolor{CStatement}{常见变形（横轴特例）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      直线取 $x$ 轴，即 $k=0,\ m=0$，则判别式化为
		      \[
			      \Delta=b^{2}-4ac.
		      \]
		      此时当 $\Delta>0$ 时图象与 $x$ 轴有两个交点；当 $\Delta=0$ 时有一个交点（顶点在 $x$ 轴上）；当 $\Delta<0$ 时无交点。
	\end{description}
\end{statementbox}
