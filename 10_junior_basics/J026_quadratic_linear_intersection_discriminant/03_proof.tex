\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n将交点问题转化为一元二次方程根的问题，利用判别式判断根的个数，进而得出交点个数。\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（代入消元）：}]
			\n
			\[
				\n ax^{2}+bx+c = kx+m.\n
			\]
			\n
			\par\textbf{理由：} 联立 $y=kx+m$ 与 $y=ax^{2}+bx+c$，消去 $y$ 得到关于 $x$ 的等式。\n\n
		\item[\textbf{步骤二（移项整理）：}]
			\n
			\[
				\n ax^{2}+(b-k)x+(c-m)=0.\n
			\]
			\n
			\par\textbf{理由：} 将所有项移至左边，化为标准二次方程形式 $Ax^{2}+Bx+C=0$，其中 $A=a\neq0$, $B=b-k$, $C=c-m$。\n\n
		\item[\textbf{步骤三（计算判别式）：}]
			\n
			\[
				\n \Delta = (b-k)^{2}-4a(c-m).\n
			\]
			\n
			\par\textbf{理由：} 二次方程 $Ax^{2}+Bx+C=0$ 的判别式定义为 $\Delta = B^{2}-4AC$，代入即得。\n\n
		\item[\textbf{步骤四（判别式分类）：}]
			\n
			\[
				\n \Delta>0,\quad \Delta=0,\quad \Delta<0.\n
			\]
			\n
			\par\textbf{理由：} $\Delta$ 的符号对应方程根的不同情况：$\Delta>0$ 时有两个不等实根，从而有两个交点；$\Delta=0$ 时有一个重根，从而有一个交点（相切）；$\Delta<0$ 时无实根，从而无交点。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n综上所述，二次函数与一次函数图象交点个数由联立后一元二次方程的判别式 $\Delta$ 唯一决定：$\Delta>0$ 二交点；$\Delta=0$ 一切点；$\Delta<0$ 无交点。\n
\end{proofbox}
