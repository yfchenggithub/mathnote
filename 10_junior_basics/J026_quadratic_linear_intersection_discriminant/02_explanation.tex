\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n将交点问题转化为一元二次方程根的问题，判别式符号直接决定交点个数。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（代数转化）：}]
			将 $y=kx+m$ 代入 $y=ax^{2}+bx+c$，整理得 $ax^{2}+(b-k)x+(c-m)=0$。\n
		\item[\textbf{要点二（判别式判根）：}]
			方程根的个数由判别式 $\Delta=(b-k)^{2}-4a(c-m)$ 决定，根的个数等于图象交点个数。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（几何直观）}}\n\n当 $\Delta>0$ 时直线穿过抛物线，有两个交点；$\Delta=0$ 时直线与抛物线相切于一点；$\Delta<0$ 时直线与抛物线无公共点。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（方程思想）}}\n\n联立后的二次方程根的判别式是判断交点存在性的代数工具，实现几何问题代数化。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（高频考点）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（直接判交点）：} 给出具体二次函数和直线，计算 $\Delta$ 符号即可判断交点个数。\n
		\item \textbf{考法二（参数求范围）：} 已知交点个数条件（如两个交点、相切、无交点），列出 $\Delta>0$、$\Delta=0$ 或 $\Delta<0$ 的不等式或方程求解参数范围。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n遇到直线与抛物线问题，先联立、再算 $\Delta$、后由符号下结论，无需画图。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（线抛交点问题）：} 二次函数与一次函数图象相交、相切、无公共点的判别及参数求解。\n
		\item \textbf{场景二（x轴零点问题）：} 二次函数与 $x$ 轴交点个数问题，即 $k=0,\ m=0$ 时的特例。\n
	\end{itemize}
	\n
\end{explanationbox}
