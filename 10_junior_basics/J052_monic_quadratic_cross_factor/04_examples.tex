\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础直接应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	分解因式 $x^2+5x+6$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分解常数）：}] 将常数项6分解为 $2\times3$。
			\par\textbf{理由：} 寻找乘积等于6的两个数。
		\item[\textbf{第二步（检验和）：}] 计算 $2+3=5$，等于一次项系数5。
			\par\textbf{理由：} 条件要求 $a+b=p$。
		\item[\textbf{第三步（写出分解）：}] 由十字相乘法得 $x^2+5x+6=(x+2)(x+3)$。
			\par\textbf{理由：} 代入公式 $x^2+px+q=(x+a)(x+b)$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$(x+2)(x+3)$}.

	\medskip
	\textbf{\textcolor{CExample}{例2（符号变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	分解因式 $x^2-4x-21$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分解常数）：}] 将常数项 $-21$ 分解为 $3\times(-7)$ 或 $(-3)\times7$ 等，使乘积为 $-21$。
			\par\textbf{理由：} 常数项为负，两因数必异号。
		\item[\textbf{第二步（检验和）：}] 检验各对和：$3+(-7)=-4$，等于一次项系数 $-4$；而 $(-3)+7=4\neq-4$，舍去。
			\par\textbf{理由：} 条件 $a+b=p$，$p=-4$。
		\item[\textbf{第三步（写出分解）：}] 取 $a=3,\;b=-7$，得 $x^2-4x-21=(x+3)(x-7)$。
			\par\textbf{理由：} 代入公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$(x+3)(x-7)$}.

	\medskip
	\textbf{\textcolor{CExample}{例3（解方程）}}

	\textbf{\textcolor{CExample}{题目：}}
	解方程 $x^2-9x+20=0$（用十字相乘法因式分解求解）。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分解常数）：}] 将常数项20分解为 $(-4)\times(-5)$（乘积为正，两数同号且和须负）。
			\par\textbf{理由：} 一次项系数 $-9<0$，故两因数同负。
		\item[\textbf{第二步（检验和）：}] 计算 $(-4)+(-5)=-9$，等于一次项系数。
			\par\textbf{理由：} 条件 $a+b=p$，$p=-9$。
		\item[\textbf{第三步（因式分解并求解）：}] 由十字相乘法，$x^2-9x+20=(x-4)(x-5)=0$，所以 $x=4$ 或 $x=5$。
			\par\textbf{理由：} 若 $(x-4)(x-5)=0$，则 $x-4=0$ 或 $x-5=0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	方程的解为 \textbf{$x=4$ 或 $x=5$}.
\end{examplebox}
