\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知点 $P$ 在 $\odot O$ 外，割线 $PAB$ 交圆于 $A,B$，$PA=2$，$AB=4$；另一割线 $PCD$ 交圆于 $C,D$，$PC=3$。求 $PD$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}] 点 $P$ 在圆外，有两条割线 $PAB$ 和 $PCD$，符合割线定理。
			\par\textbf{理由：}
			确认定理使用条件：$P$ 在圆外，两条割线。
		\item[\textbf{第二步（套用定理）：}] 由割线定理得 $PA \cdot PB = PC \cdot PD$。
			\par\textbf{理由：}
			直接应用结论。
		\item[\textbf{第三步（代入求解）：}] $PA=2$，$AB=4$，则 $PB = PA+AB = 6$；代入得 $2 \times 6 = 3 \times PD$，解得 $PD = 4$。
			\par\textbf{理由：}
			代数运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PD = 4$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$P$ 为圆外一点，割线 $PAB$ 交圆于 $A,B$，$PA=4$，$PB=9$；另一割线 $PCD$ 交圆于 $C,D$，$PC=3$。求 $PD$ 和 $CD$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}] 点 $P$ 在圆外，两条割线，满足割线定理条件。
			\par\textbf{理由：}
			确认条件。
		\item[\textbf{第二步（套用定理）：}] 由割线定理得 $PA \cdot PB = PC \cdot PD$。
			\par\textbf{理由：}
			定理应用。
		\item[\textbf{第三步（计算求解）：}] 代入：$4 \times 9 = 3 \times PD$，解得 $PD = 12$；则 $CD = PD - PC = 12 - 3 = 9$。
			\par\textbf{理由：}
			先求 $PD$，再求弦长 $CD$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PD = 12$，$CD = 9$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	从圆外一点 $P$ 引一条割线 $PAB$ 交圆于 $A,B$，并引一条切线 $PT$ 切圆于 $T$。已知 $PA=4$，$PB=9$，求切线长 $PT$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 点 $P$ 在圆外，有一条割线和一条切线。
			\par\textbf{理由：}
			切线可视为割线的特殊情形（交点重合）。
		\item[\textbf{第二步（套用推广）：}] 由割线定理的推广（切线情形）得 $PA \cdot PB = PT^2$。
			\par\textbf{理由：}
			当割线 $PCD$ 与圆相切时，$C$ 与 $D$ 重合于 $T$，故 $PC \cdot PD = PT^2$。
		\item[\textbf{第三步（计算求解）：}] 代入 $4 \times 9 = PT^2$，得 $PT^2 = 36$，所以 $PT = 6$（取正值）。
			\par\textbf{理由：}
			切线长取正。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PT = 6$}。
