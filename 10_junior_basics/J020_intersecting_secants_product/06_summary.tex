\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{从圆外一点出发的两条割线，该点到两交点的线段之积相等，即 $PA \cdot PB = PC \cdot PD$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（点在圆外）：} 点 $P$ 在圆 $O$ 外部。
		\item \textbf{条件 2（两条割线）：} 有两条割线 $PAB$ 和 $PCD$ 分别交圆于 $A,B$ 和 $C,D$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件混淆）：} 注意定理只适用于点 $P$ 在圆外，圆内时应使用相交弦定理。
		\item \textbf{检查项（对应乘积）：} 乘积是 $PA \cdot PB$（近段 $\times$ 远段），不要混淆成其他线段组合。
	\end{itemize}
\end{summarybox}
