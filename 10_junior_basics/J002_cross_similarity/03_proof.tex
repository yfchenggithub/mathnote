\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	\par 通过平行线性质证明两对对应角相等，利用AA判定相似，再根据相似三角形性质得到比例式。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（平行得角相等）：}]
		      由$AB\parallel CD$，得：
		      \[
			      \angle OAB = \angle OCD,\quad \angle OBA = \angle ODC.
		      \]
		      \par\textbf{理由：}两直线平行，内错角（或同位角）相等。
		\item[\textbf{步骤二（对顶角相等）：}]
		      直线$AC$与$BD$相交于点$O$，所以：
		      \[
			      \angle AOB = \angle COD.
		      \]
		      \par\textbf{理由：}对顶角相等。
		\item[\textbf{步骤三（AA证相似）：}]
		      在$\triangle OAB$与$\triangle OCD$中，有两组角对应相等：
		      \[
			      \begin{aligned}
				      \angle OAB & = \angle OCD, \\
				      \angle AOB & = \angle COD.
			      \end{aligned}
		      \]
		      所以$\triangle OAB\sim\triangle OCD$。
		      \par\textbf{理由：}两角对应相等的两个三角形相似。
		\item[\textbf{步骤四（对应边成比例）：}]
		      由相似三角形对应边成比例，得：
		      \[
			      \frac{OA}{OC}=\frac{OB}{OD}=\frac{AB}{CD}.
		      \]
		      \par\textbf{理由：}相似三角形对应边比例相等。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\par\textbf{只要满足$AB\parallel CD$且$AC$、$BD$交于$O$，即可直接使用比例式$\frac{OA}{OC}=\frac{OB}{OD}=\frac{AB}{CD}$，无需重复证明。}
\end{proofbox}
