	\textbf{\textcolor{CExample}{例1（基础）}}
	\par\textbf{\textcolor{CExample}{题目：}} 如图，直线$AC$与$BD$交于点$O$，$AB\parallel CD$。已知$OA=3$，$OC=6$，$AB=4$。求$CD$的长。
	\par\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}]
			由$AB\parallel CD$且$AC$、$BD$交于$O$，得：
			\[
				\triangle OAB\sim\triangle OCD.
			\]
			\par\textbf{理由：}符合8字型相似模型。
		\item[\textbf{第二步（列比例式）：}]
			由相似得：
			\[
				\frac{OA}{OC}=\frac{AB}{CD}.
			\]
			\par\textbf{理由：}对应边成比例。
		\item[\textbf{第三步（求解未知）：}]
			代入已知：
			\[
				\frac{3}{6}=\frac{4}{CD}\;\Rightarrow\; CD=8.
			\]
			\par\textbf{理由：}解比例方程。
	\end{description}
	\par\textbf{\textcolor{CExample}{关键结论：}} \textbf{$CD=8$}。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\par\textbf{\textcolor{CExample}{题目：}} 梯形$ABCD$中，$AD\parallel BC$，对角线$AC$、$BD$交于点$O$。已知$AD=4$，$BC=6$，$OA=3$。求$OC$的长。
	\par\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}]
			由$AD\parallel BC$得：
			\[
				\triangle AOD\sim\triangle COB.
			\]
			\par\textbf{理由：}梯形中8字型相似。
		\item[\textbf{第二步（列比例式）：}]
			由相似得：
			\[
				\frac{AO}{CO}=\frac{AD}{BC}.
			\]
			\par\textbf{理由：}对应边成比例（注意A对应C，D对应B）。
		\item[\textbf{第三步（求解未知）：}]
			代入：
			\[
				\frac{3}{OC}=\frac{4}{6}\;\Rightarrow\; OC=4.5.
			\]
			\par\textbf{理由：}解比例。
	\end{description}
	\par\textbf{\textcolor{CExample}{关键结论：}} \textbf{$OC=4.5$}。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\par\textbf{\textcolor{CExample}{题目：}} 在8字型相似中，$AB\parallel CD$，$AB:CD=2:3$，$\triangle OAB$的面积为$12$，求$\triangle OCD$的面积。
	\par\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求相似比）：}]
			由$AB\parallel CD$得：
			\[
				\triangle OAB\sim\triangle OCD,\quad k=\frac{AB}{CD}=\frac{2}{3}.
			\]
			\par\textbf{理由：}对应边比就是相似比。
		\item[\textbf{第二步（面积比转化）：}]
			面积比等于相似比的平方：
			\[
				\frac{S_{\triangle OAB}}{S_{\triangle OCD}}=k^2=\left(\frac{2}{3}\right)^2=\frac{4}{9}.
			\]
			\par\textbf{理由：}相似三角形面积比性质。
		\item[\textbf{第三步（代值求解）：}]
			设$S_{\triangle OCD}=x$，则：
			\[
				\frac{12}{x}=\frac{4}{9}\;\Rightarrow\; x=27.
			\]
			\par\textbf{理由：}解比例。
	\end{description}
	\par\textbf{\textcolor{CExample}{关键结论：}} \textbf{$S_{\triangle OCD}=27$}。
