\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}
	\par两条平行线被两条交叉直线所截，形成的两个三角形形状一样。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（平行得等角）：}]
			因为$AB\parallel CD$，所以$\angle OAB = \angle OCD$，$\angle OBA = \angle ODC$。
		\item[\textbf{要点二（对顶角相等）：}]
			直线$AC$与$BD$相交，$\angle AOB = \angle COD$（对顶角）。
		\item[\textbf{要点三（两角证相似）：}]
			由两组角相等得$\triangle OAB\sim\triangle OCD$，继而推出对应边比例。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（平行相似）}}
	\par平行这一位置关系通过角相等转化为三角形相似，从而把长度比例联系起来。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（比例关系）}}
	\par已知平行可写出多组成比例线段，用于列方程求未知长度。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（模型应用）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接求边长）：}给出部分线段长，利用比例求未知边。
		\item \textbf{考法二（混合模型）：}结合其他几何模型，综合应用。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}
	\par把图形中的平行线看做“角转换的桥梁”，相交直线提供对顶角，自动完成两对等角的拼图。

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（梯形对角线）：}梯形中，上、下底平行，对角线交点构成8字型。
		\item \textbf{场景二（平行线分线段）：}任意两条直线被一组平行线所截，截得的对应线段成比例，8字型是特例。
	\end{itemize}

\end{explanationbox}
