\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\par\textbf{两条平行线截两条相交直线，所得两个三角形相似，对应边成比例。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（相交直线）：}直线$AC$与$BD$相交于$O$。
		\item \textbf{条件2（平行条件）：}$AB\parallel CD$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（对应边确定）：}务必根据角相等找对应，避免配错比例。
		\item \textbf{检查项（平行前提）：}使用前必须确认$AB\parallel CD$，否则模型不成立。
	\end{itemize}

\end{summarybox}
