	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（相交直线）：}}]
			直线$AC$与$BD$相交于点$O$。
		\item[\textbf{\textcolor{CStatement}{条件2（平行条件）：}}]
			$AB\parallel CD$。
	\end{description}
	\par\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（三角形相似）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个三角形形状相同，对应边成比例。
			\[
				\triangle OAB\sim\triangle OCD.
			\]
			且
			\[
				\frac{OA}{OC}=\frac{OB}{OD}=\frac{AB}{CD}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（面积关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积比等于对应边比的平方。
			\[
				\frac{S_{\triangle OAB}}{S_{\triangle OCD}}=\left(\frac{AB}{CD}\right)^2.
			\]
	\end{description}

\end{statementbox}
