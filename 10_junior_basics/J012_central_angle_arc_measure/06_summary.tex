\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆心角的度数等于它所对弧的度数。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（同圆等圆）：} 圆心角和所对弧必须在同一个圆或半径相等的圆中。
		\item \textbf{条件2（对应弧）：} 只能对圆心角所对的弧使用，不能混淆其他弧。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（等圆条件）：} 忽略“同圆或等圆”前提，直接比较弧长会导致错误。
		\item \textbf{检查项（单位一致）：} 所有角度单位必须统一使用“度”，避免与弧度混淆。
	\end{itemize}
\end{summarybox}
