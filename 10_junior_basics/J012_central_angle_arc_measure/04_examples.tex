\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在圆$O$中，圆心角$\angle AOB = 50^{\circ}$，求所对弧$\widehat{AB}$的度数（用$\deg$表示）。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别对应）：}] 确定$\angle AOB$是圆心角，$\widehat{AB}$是其所对的弧。
			\par\textbf{理由：}根据圆心角定义，顶点在圆心，两边交圆于$A,B$。
		\item[\textbf{第二步（套用定理）：}] 由圆心角定理，$\deg(\widehat{AB}) = \deg(\angle AOB) = 50^{\circ}$。
			\par\textbf{理由：}圆心角的度数等于所对弧的度数。
		\item[\textbf{第三步（写出答案）：}] $\deg(\widehat{AB}) = 50^{\circ}$。
			\par\textbf{理由：}代入已知角度。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\deg(\widehat{AB}) = 50^{\circ}$}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在圆$O$中，$\deg(\widehat{AB}) = 80^{\circ}$，点$C$是$\widehat{AB}$上一点，且$\deg(\widehat{AC}) = 30^{\circ}$，求$\deg(\widehat{BC})$和$\deg(\angle AOB)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（弧段相减）：}] $\deg(\widehat{BC}) = \deg(\widehat{AB}) - \deg(\widehat{AC}) = 80^{\circ} - 30^{\circ} = 50^{\circ}$。
			\par\textbf{理由：}弧的度数可以直接相加（减），因为同圆上连续弧。
		\item[\textbf{第二步（求圆心角）：}] 由圆心角定理，$\deg(\angle AOB) = \deg(\widehat{AB}) = 80^{\circ}$。
			\par\textbf{理由：}圆心角等于所对弧的度数。
		\item[\textbf{第三步（整合答案）：}] $\deg(\widehat{BC}) = 50^{\circ}$，$\deg(\angle AOB) = 80^{\circ}$。
			\par\textbf{理由：}计算结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\deg(\widehat{BC}) = 50^{\circ}, \deg(\angle AOB) = 80^{\circ}$}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在圆$O$中，$\angle AOB = 2\angle BOC$，且$\angle AOC = 120^{\circ}$，其中射线$OB$在$\angle AOC$内部。求$\deg(\widehat{AB})$和$\deg(\widehat{BC})$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列方程）：}] 设$\angle BOC = x$，则$\angle AOB = 2x$。由$\angle AOC = \angle AOB + \angle BOC = 3x = 120^{\circ}$，解得$x = 40^{\circ}$。
			\par\textbf{理由：}角的和等于已知角$\angle AOC$。
		\item[\textbf{第二步（计算各角）：}] 因此$\angle AOB = 80^{\circ}$，$\angle BOC = 40^{\circ}$。
			\par\textbf{理由：}代入$x$值。
		\item[\textbf{第三步（求弧的度数）：}] 由圆心角定理，$\deg(\widehat{AB}) = \deg(\angle AOB) = 80^{\circ}$，$\deg(\widehat{BC}) = \deg(\angle BOC) = 40^{\circ}$。
			\par\textbf{理由：}圆心角的度数等于所对弧的度数。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\deg(\widehat{AB}) = 80^{\circ}, \deg(\widehat{BC}) = 40^{\circ}$}
