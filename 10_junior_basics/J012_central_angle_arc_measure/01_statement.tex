\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（同圆等圆）：}}] 在同一个圆或半径相等的两个圆中（即等圆）。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（圆心角定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆心角的度数等于它所对弧的度数。
			\[
				\deg(\angle AOB) = \deg(\overset{\frown}{AB}).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（圆心角弧双向）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在同圆或等圆中，圆心角相等等价于所对弧相等（度数相等）。
			\[
				\angle AOB = \angle COD \iff \deg(\overset{\frown}{AB}) = \deg(\overset{\frown}{CD}).
			\]
	\end{description}
\end{statementbox}
