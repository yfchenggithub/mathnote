	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（椭圆与焦准）：}}] 椭圆标准方程 $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$（$a>b>0$），焦点 $F(c,0)$，对应准线 $l: x=\dfrac{a^2}{c}$，离心率 $e=\dfrac{c}{a}\in(0,1)$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦准距之比）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆上任意点到焦点距离与到准线距离的比值恒等于离心率。
			\[
				\frac{|PF|}{d(P,l)}=e.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（坐标轴推广）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点所在坐标轴决定准线的具体形式。
			若焦点在 $x$ 轴上，准线方程为 $x = \pm \dfrac{a^2}{c}$；
			若焦点在 $y$ 轴上，准线方程为 $y = \pm \dfrac{a^2}{c}$。
	\end{description}

\end{statementbox}
