\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{椭圆上的点到焦点与对应准线的距离之比恒为离心率 $e \in (0,1)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准方程或基本量）：} 已知椭圆标准方程或 $a, c, e$ 等量。
		\item \textbf{条件 2（焦点与准线对应）：} 左焦点对应左准线，右焦点对应右准线，不可混淆。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（准线公式记忆）：} 焦点在 $x$ 轴，准线为 $x = \pm a^2/c$；焦点在 $y$ 轴，准线为 $y = \pm a^2/c$。
		\item \textbf{检查项（$e$ 范围）：} 椭圆 $0<e<1$，双曲线 $e>1$，抛物线 $e=1$，使用前先判断曲线。
	\end{itemize}
\end{summarybox}
