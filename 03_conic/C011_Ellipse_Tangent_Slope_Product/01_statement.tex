\begin{statementbox}
\textbf{C011: 椭圆切线与径向斜率积定值}

设椭圆
\[
C:\ \frac{x^2}{a^2}+\frac{y^2}{b^2}=1,\qquad a>b>0,
\]
中心为 $O(0,0)$, 焦距参数 $c=\sqrt{a^2-b^2}$, 离心率 $e=\frac ca$.
设 $P(x_0,y_0)$ 是椭圆上一点, $l$ 是椭圆在 $P$ 处的切线.

\textbf{适用条件}: $x_0y_0\ne 0$, 即 $P$ 不在四个轴顶点上.
此时切线 $l$ 与中心连线 $OP$ 的斜率均存在, 分别记为 $k_l$ 和 $k_{OP}$, 则
\[
\boxed{k_l\,k_{OP}=-\frac{b^2}{a^2}=e^2-1.}
\]

\textbf{配套公式}: 椭圆在 $P$ 处的切线方程与两条斜率分别为
\[
\frac{x_0x}{a^2}+\frac{y_0y}{b^2}=1,\qquad
k_l=-\frac{b^2x_0}{a^2y_0},\qquad
k_{OP}=\frac{y_0}{x_0}.
\]

\textbf{等价形式}: 当 $k_{OP}\ne0$ 时,
\[
k_l=-\frac{b^2}{a^2k_{OP}}=\frac{e^2-1}{k_{OP}}.
\]

\textbf{顶点例外}: 若 $P=(\pm a,0)$, 则 $OP$ 水平、切线竖直;
若 $P=(0,\pm b)$, 则 $OP$ 竖直、切线水平.
两种情形下 $OP\perp l$, 但斜率乘积\textbf{没有定义}, 不可代入上面的乘积公式.
\end{statementbox}
