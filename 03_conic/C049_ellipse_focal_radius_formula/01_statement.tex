\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（椭圆标准式）：}}] 椭圆方程为 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$，其中 $a>b>0$，$c^2=a^2-b^2$，离心率 $e=\frac{c}{a}\in(0,1)$。
		\item[\textbf{\textcolor{CStatement}{条件2（焦点位置）：}}] 左、右焦点分别为 $F_1(-c,0)$，$F_2(c,0)$。
		\item[\textbf{\textcolor{CStatement}{条件3（点P坐标）：}}] 点 $P(x_0,y_0)$ 在椭圆上。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦半径长度随 $x_0$ 增大而左增右减。
			\[
				|PF_1| = a + e x_0,
				\qquad
				|PF_2| = a - e x_0.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（焦点在y轴）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			类似结构，纵坐标代替横坐标（$F_1$ 为下焦点，$F_2$ 为上焦点）。
			\[
				|PF_1| = a + e y_0,
				\qquad
				|PF_2| = a - e y_0.
			\]
	\end{description}


\medskip
\textbf{\textcolor{CStatement}{关键等价式（用于证明和考场检查）}}
设 $r_1=|PF_1|$、$r_2=|PF_2|$，则
\[
 r_1+r_2=2a,\qquad r_1-r_2=2ex_0.
\]
因此 $x_0=0$ 时两焦半径均为 $a$；$x_0>0$ 时左焦半径更长，$x_0<0$ 时右焦半径更长。
注意 $-a\le x_0\le a$，每条焦半径均在 $[a-c,a+c]$ 内。

\medskip
\textbf{\textcolor{CStatement}{推广的完整条件}}
焦点在 $y$ 轴时，标准方程改为
$\frac{x^2}{b^2}+\frac{y^2}{a^2}=1\ (a>b>0)$，
下、上焦点依次为 $F_1(0,-c)$、$F_2(0,c)$；
此时 $|PF_1|=a+ey_0$，$|PF_2|=a-ey_0$。
\end{statementbox}
