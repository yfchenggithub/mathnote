\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{焦点弦长由分母 $1-e^2\cos^2\theta$ 控制，随着 $|\cos\theta|$ 增大而增大；在所有过同一焦点的弦中，其取值范围为 $\frac{2b^2}{a}\le |PQ|\le 2a$。}

	\medskip

	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准椭圆）：} 椭圆方程为 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\;(a>b>0)$，焦点在 $x$ 轴。
		\item \textbf{条件 2（过焦点弦）：} 所给弦必须经过椭圆的一个焦点。
	\end{itemize}

	\medskip

	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（余弦平方）：} 公式分母是 $a^2-c^2\cos^2\theta$，$\cos\theta$ 必须平方，且 $\theta$ 为弦与长轴夹角。
		\item \textbf{检查项（通径验证）：} 令 $\theta=90^\circ$ 得 $|PQ|=\frac{2b^2}{a}$，可检验计算是否正确。
	\end{itemize}

\end{summarybox}
