\begin{explanationbox}

	\textbf{\textcolor{CExplanation}{一句话直觉}}

	焦点弦的长度由椭圆大小和直线方向共同决定：直线越偏向长轴方向，弦越长；越偏向短轴方向，弦越短。

	\medskip

	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（参数依赖）：}] 公式中只出现 $\cos^2\theta$，因此弦长与 $\theta$ 的正负无关，只与夹角的余弦绝对值有关。
		\item[\textbf{要点二（特殊取值）：}] 当 $\theta=0^\circ$ 时，$|PQ|=2a$（长轴长度）；当 $\theta=90^\circ$ 时，$|PQ|=\frac{2b^2}{a}$（通径）。
	\end{description}

	\medskip

	\textbf{\textcolor{CExplanation}{几何本质（极坐标视角）}}
	以焦点为极点时，设半通径
	\[
		\ell=\frac{b^2}{a}=a(1-e^2),
	\]
	椭圆的极坐标方程可写成
	\[
		\rho=\frac{\ell}{1\pm e\cos\varphi},
	\]
	其中正负号取决于所选焦点和极轴正方向。过焦点的弦对应方向 $\varphi$ 和 $\varphi+\pi$ 的两段极径之和，因此可得
	\[
		|PQ|=\frac{2\ell}{1-e^2\cos^2\varphi}
		=\frac{2a(1-e^2)}{1-e^2\cos^2\varphi}.
	\]

	\medskip

	\textbf{\textcolor{CExplanation}{代数意义（分式结构）}}
	公式是关于 $\cos^2\theta$ 的一次分式：分母 $a^2-c^2\cos^2\theta$，分子常数 $2ab^2$，因此 $|PQ|$ 随 $|\cos\theta|$ 增大而单调递增。

	\medskip

	\textbf{\textcolor{CExplanation}{考点价值（常见考法）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接求值）：} 给出椭圆方程和直线夹角，直接代入公式计算弦长，避免联立方程组。
		\item \textbf{考法二（反求参数）：} 已知焦点弦长和其他条件，列方程求解离心率、焦半距或椭圆方程。
	\end{itemize}

	\medskip

	\textbf{\textcolor{CExplanation}{顿悟点}}
	\textbf{公式的核心在于 $\cos^2\theta$ 的出现，它把角度信息转化为代数项，使得弦长计算从联立方程转化为单一分式。}

	\medskip

	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（焦点弦问题）：} 任何直接求过焦点的弦长问题，优先考虑使用该公式。
		\item \textbf{场景二（范围最值）：} 当问题涉及焦点弦长度变化时，利用 $\theta$ 的取值范围求最值。
	\end{itemize}

\end{explanationbox}
