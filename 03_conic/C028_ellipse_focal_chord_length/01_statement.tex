	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（标准设定）：}}] 椭圆$\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\;(a>b>0)$，焦点$F_1(-c,0),F_2(c,0)$，$c^2=a^2-b^2$，离心率$e=\frac{c}{a}$；弦$PQ$过其中一个焦点$F$，且与长轴（$x$轴）的夹角为$\theta$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦点弦长）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 焦点弦长由半长轴$a$、半短轴$b$、焦半距$c$和夹角$\theta$直接给出。
		      \[
			      |PQ|=\frac{2ab^2}{a^2-c^2\cos^2\theta}=\frac{2a(1-e^2)}{1-e^2\cos^2\theta}.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（通径特例）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 在所有过同一焦点的弦中，当弦垂直于长轴（$\theta=90^\circ$）时，得到最短焦点弦，即通径。
		      \[
			      |PQ|=\frac{2b^2}{a}.
		      \]
	\end{description}

\end{statementbox}
