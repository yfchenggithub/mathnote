	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{25}+\frac{y^2}{9}=1$，过右焦点 $F_2$ 的弦 $PQ$ 与长轴（$x$轴）的夹角为 $60^\circ$，求弦 $PQ$ 的长度。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 由椭圆方程得 $a=5,\; b=3,\; c=4,\; e=\frac{4}{5}$。
		      \par\textbf{理由：} 标准形式的系数直接读出 $a,b$，然后 $c^2=a^2-b^2$，$e=c/a$。

		\item[\textbf{第二步（代入公式）：}] 使用焦点弦长公式 $|PQ|=\frac{2ab^2}{a^2-c^2\cos^2\theta}$，代入 $\theta=60^\circ$，$\cos60^\circ=\frac12$：
		      \[
			      |PQ| = \frac{2\cdot5\cdot9}{25-16\cdot\left(\frac12\right)^2} = \frac{90}{25-4} = \frac{90}{21}.
		      \]
		      \par\textbf{理由：} 将 $a,b,c,\cos\theta$ 具体值代入并计算分母。

		\item[\textbf{第三步（化简结果）：}] 化简分数得 $|PQ| = \frac{30}{7}$。
		      \par\textbf{理由：} 分子分母同除以 $3$ 即可。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$|PQ| = \dfrac{30}{7}$}.

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知椭圆中心在原点，焦点在 $x$ 轴上，离心率 $e=\frac12$，过左焦点 $F_1$ 的弦 $AB$ 与长轴的夹角为 $60^\circ$，且 $|AB|=8$，求椭圆的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设方程列参）：}] 设椭圆方程为 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\;(a>b>0)$，由 $e=\frac12$ 得 $c=\frac a2$，$b^2=a^2-c^2=\frac34a^2$。
		      \par\textbf{理由：} 离心率与基本量关系。

		\item[\textbf{第二步（表达弦长）：}] 使用公式 $|AB|=\frac{2a(1-e^2)}{1-e^2\cos^2\theta}$，代入 $e=\frac12$，$\cos60^\circ=\frac12$：
		      \[
			      |AB| = \frac{2a(1-\frac14)}{1-\frac14\cdot\frac14} = \frac{2a\cdot\frac34}{1-\frac1{16}} = \frac{3a/2}{15/16} = \frac{8a}{5}.
		      \]
		      \par\textbf{理由：} 将参数代入并化简分式。

		\item[\textbf{第三步（求解参数）：}] 由 $|AB| = 8$ 得 $\frac{8a}{5}=8$，解得 $a=5$；进而 $c=\frac52$，$b^2=\frac34\cdot25=\frac{75}{4}$。故椭圆方程为 $\frac{x^2}{25}+\frac{4y^2}{75}=1$。
		      \par\textbf{理由：} 代回求 $b$，写出标准方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\dfrac{x^2}{25}+\dfrac{4y^2}{75}=1$}.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\;(a>b>0)$ 的焦距为 $2\sqrt3$，且过焦点 $F$ 的弦 $PQ$ 的最大长度是最小长度的 $4$ 倍，求椭圆的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（最值关系）：}] 由焦点弦长公式，在所有过同一焦点的弦中，当 $\theta=0^\circ$ 时 $|PQ|_{\max}=2a$，当 $\theta=90^\circ$ 时 $|PQ|_{\min}=\frac{2b^2}{a}$。根据题意 $2a=4\times\frac{2b^2}{a}$，化简得 $a^2=4b^2$，即 $a=2b$。
		      \par\textbf{理由：} 代入 $\theta$ 的端点值得最值，按题意列方程。

		\item[\textbf{第二步（利用焦距）：}] 焦距 $2c=2\sqrt3$，故 $c=\sqrt3$。椭圆中 $c^2=a^2-b^2$，将 $a=2b$ 代入得 $3=4b^2-b^2=3b^2$，解得 $b^2=1$，$a^2=4$。
		      \par\textbf{理由：} 椭圆基本关系 $c^2=a^2-b^2$。

		\item[\textbf{第三步（写出方程）：}] 所以标准方程为 $\frac{x^2}{4}+y^2=1$。
		      \par\textbf{理由：} 代入 $a,b$ 值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\dfrac{x^2}{4}+y^2=1$}.
