\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（余弦平方错误）：}}] 误记公式分母为 $a^2-c^2\cos\theta$ 或漏掉平方。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母平方形式）：}}] 正确分母应为 $a^2-c^2\cos^2\theta$，$\cos\theta$ 必须平方。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导忽略）：}}] 推导过程中二次项系数含 $\cos^2\theta$，直接记结论时易丢失平方。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（斜率转化错误）：}}] 已知直线斜率 $k$ 时，误将 $k$ 当作 $\cos\theta$，或只记 $\cos\theta=\frac{1}{\sqrt{1+k^2}}$ 而忘记公式中需要的是 $\cos^2\theta$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确定义）：}}] 若直线斜率为 $k$，则
		      \[
			      \tan\theta=k,\qquad \cos^2\theta=\frac{1}{1+k^2}.
		      \]
		      代入焦点弦长公式时，应使用 $\cos^2\theta$，而不是把 $k$ 当作 $\cos\theta$。
		\item[\textbf{\textcolor{CTrap}{补充提醒（竖直直线）：}}] 当直线竖直时，斜率不存在，此时 $\theta=90^\circ$，直接取 $\cos^2\theta=0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 混淆了正切与余弦的关系，或者忽略 $\cos^2\theta$ 与 $k$ 的直接联系。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（条件误用）：}}] 将焦点弦长公式直接用于椭圆中不过焦点的弦。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式前提）：}}] 公式推导源于直线过焦点，仅对过焦点的弦成立；一般弦要根据直线方程重新联立计算。若所在直线为非竖直直线 $y=kx+m$，则
		      \[
			      |AB|=\sqrt{1+k^2}\,|x_1-x_2|.
		      \]
		      若直线竖直，则应直接使用
		      \[
			      |AB|=|y_1-y_2|.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（条件迁移）：}}] 只记住公式形式，忽略其特殊的几何条件（过焦点）。
	\end{description}

\end{trapbox}
