\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\dfrac{x^{2}}{4}+\dfrac{y^{2}}{3}=1$ 的左右顶点为 $A_{1}(-2,0),A_{2}(2,0)$，点 $P(1,\frac{3}{2})$ 在椭圆上，求 $k_{PA_{1}}\cdot k_{PA_{2}}$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证点P在椭圆上）：}]
			代入 $P$ 到椭圆方程：$\dfrac{1^{2}}{4}+\dfrac{(3/2)^{2}}{3}=1$，成立。
			\par\textbf{理由：} 确保点 $P$ 满足椭圆方程，符合使用条件。
		\item[\textbf{第二步（确定椭圆参数）：}]
			由椭圆方程得 $a^{2}=4,b^{2}=3$。
			\par\textbf{理由：} 准备代入结论公式。
		\item[\textbf{第三步（套用第三定义）：}]
			根据椭圆第三定义，$k_{PA_{1}}\cdot k_{PA_{2}} = -\dfrac{b^{2}}{a^{2}}$。代入 $a^{2}=4,b^{2}=3$，即得 $k_{PA_{1}}\cdot k_{PA_{2}} = -\dfrac{3}{4}$。
			\par\textbf{理由：} 直接使用已知结论。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$-\dfrac{3}{4}$}

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $C:\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1\;(a>b>0)$ 的左右顶点为 $A_{1},A_{2}$，点 $P(2,1)$ 在 $C$ 上，且 $k_{PA_{1}}\cdot k_{PA_{2}} = -\dfrac{1}{2}$。求椭圆 $C$ 的标准方程和离心率 $e$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（参数关系推导）：}]
			由椭圆第三定义，$k_{PA_{1}}\cdot k_{PA_{2}} = -\dfrac{b^{2}}{a^{2}}$。已知 $k_{PA_{1}}\cdot k_{PA_{2}} = -\dfrac{1}{2}$，故 $-\dfrac{b^{2}}{a^{2}} = -\dfrac{1}{2}$，得 $b^{2} = \dfrac{a^{2}}{2}$。
			\par\textbf{理由：} 将已知乘积与第三定义结论对比。
		\item[\textbf{第二步（坐标代入方程）：}]
			将 $P(2,1)$ 代入椭圆方程得 $\dfrac{4}{a^{2}}+\dfrac{1}{b^{2}}=1$，再把 $b^{2}=\dfrac{a^{2}}{2}$ 代入得 $\dfrac{4}{a^{2}}+\dfrac{2}{a^{2}}=1$，解得 $a^{2}=6$，从而 $b^{2}=3$。
			\par\textbf{理由：} 利用椭圆方程建立等式求解。
		\item[\textbf{第三步（求离心率）：}]
			$e=\dfrac{c}{a}=\sqrt{1-\dfrac{b^{2}}{a^{2}}}=\sqrt{1-\dfrac{3}{6}}=\dfrac{\sqrt{2}}{2}$。
			\par\textbf{理由：} 离心率定义 $e=\sqrt{1-\dfrac{b^{2}}{a^{2}}}$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{椭圆 $\dfrac{x^{2}}{6}+\dfrac{y^{2}}{3}=1$，离心率 $e=\dfrac{\sqrt{2}}{2}$}

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\dfrac{x^{2}}{4}+\dfrac{y^{2}}{3}=1$ 的左右顶点为 $A_{1}(-2,0),A_{2}(2,0)$。设 $P$ 为椭圆上异于 $A_{1},A_{2}$ 的一点，直线 $PA_{1}$ 与 $x=4$ 交于点 $M$，直线 $PA_{2}$ 与 $x=4$ 交于点 $N$。证明：以线段 $MN$ 为直径的圆经过定点，并求出定点坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（纵坐标与斜率）：}]
			设 $k_{PA_{1}}=k_{1}$，$k_{PA_{2}}=k_{2}$。由点斜式，$PA_{1}:y=k_{1}(x+2)$，令 $x=4$ 得 $y_{M}=6k_{1}$；$PA_{2}:y=k_{2}(x-2)$，令 $x=4$ 得 $y_{N}=2k_{2}$。
			\par\textbf{理由：} 通过直线方程求交点纵坐标，将问题转化为斜率关系。
		\item[\textbf{第二步（第三定义求积）：}]
			由椭圆第三定义，$k_{1}k_{2}= -\dfrac{b^{2}}{a^{2}}$。代入 $a^{2}=4,b^{2}=3$ 得 $k_{1}k_{2}= -\dfrac{3}{4}$。故
			\[
				\begin{aligned}
					y_{M}y_{N} &= (6k_{1})(2k_{2}) \\ &= 12k_{1}k_{2} \\ &= 12\times\left(-\dfrac{3}{4}\right) \\ &= -9.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接代入第三定义，得到 $y_{M}y_{N}$ 为定值。
		\item[\textbf{第三步（直径式求定点）：}]
			以 $MN$ 为直径的圆方程为 $(x-4)^{2}+(y-y_{M})(y-y_{N})=0$，即 $(x-4)^{2}+y^{2}-(y_{M}+y_{N})y+y_{M}y_{N}=0$。代入 $y_{M}y_{N}=-9$ 得 $(x-4)^{2}+y^{2}-(y_{M}+y_{N})y-9=0$。令 $y=0$，得 $(x-4)^{2}-9=0$，解得 $x=1$ 或 $x=7$。所以圆恒过定点 $(1,0)$ 和 $(7,0)$。
			\par\textbf{理由：} 直径式方程中令 $y=0$ 消去含参项 $y_{M}+y_{N}$，得到仅与 $x$ 有关的方程，解得的点与 $P$ 无关。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{以 $MN$ 为直径的圆恒过定点 $(1,0)$ 和 $(7,0)$}
\end{examplebox}
