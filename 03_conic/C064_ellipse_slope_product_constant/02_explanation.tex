\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}
	椭圆上任意一点（除长轴端点外）与椭圆长轴两端点连线的斜率乘积为固定值，这个定值仅与椭圆的半长轴和半短轴有关，与点 $P$ 的位置无关。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（坐标表示斜率）：}]
			设 $P(x_{0},y_{0})$，则 $k_{PA_{1}}=\dfrac{y_{0}}{x_{0}+a}$，$k_{PA_{2}}=\dfrac{y_{0}}{x_{0}-a}$，乘积为 $\dfrac{y_{0}^{2}}{x_{0}^{2}-a^{2}}$。
		\item[\textbf{要点二（椭圆方程代入）：}]
			由椭圆方程 $\dfrac{x_{0}^{2}}{a^{2}}+\dfrac{y_{0}^{2}}{b^{2}}=1$，可得
			\[
				\begin{aligned}
					y_{0}^{2} &= b^{2}\left(1-\dfrac{x_{0}^{2}}{a^{2}}\right) \\ &= \dfrac{b^{2}}{a^{2}}(a^{2}-x_{0}^{2}).
			\end{aligned}
			\]
		\item[\textbf{要点三（化简化定）：}]
			将 $y_{0}^{2}$ 代入乘积表达式，约去 $a^{2}-x_{0}^{2}$（注意 $P$ 非端点，故 $x_{0}\neq\pm a$），立即得到 $-\dfrac{b^{2}}{a^{2}}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（类圆压缩）}}
	将圆 $x^{2}+y^{2}=a^{2}$ 沿 $y$ 轴压缩为椭圆 $\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1$。圆中直径所对圆周角为 $90^\circ$，对应斜率乘积 $-1$；压缩后 $y$ 坐标成比例变为 $y'=\dfrac{b}{a}y$，故斜率乘积变为 $-\dfrac{b^{2}}{a^{2}}$。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（定值结构）}}
	该定值仅依赖于椭圆形状（$a,b$），与 $P$ 的具体选取无关，体现了椭圆方程 $\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1$ 的代换对称性：分子 $x_{0}^{2}-a^{2}$ 与分子 $y_{0}^{2}$ 通过方程产生比例常数 $-\dfrac{b^{2}}{a^{2}}$。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（快速解题）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接求积）：} 已知椭圆方程和 $P$ 点位置，直接代入公式得斜率乘积，无需重复推导。
		\item \textbf{考法二（逆向求参数）：} 已知斜率乘积及椭圆方程一半，可反求 $a,b$ 或离心率 $e$。
		\item \textbf{考法三（串联韦达定理）：} 在综合题中，利用该结论简化涉及 $PA_{1},PA_{2}$ 斜率的运算，配合韦达定理建立关系。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}
	\textbf{关键在“消元”：} 将 $P$ 坐标表示为单变量（利用椭圆方程消去 $y_{0}^{2}$），使得乘积不再依赖 $x_{0}$，从而得到与 $P$ 无关的常数。学生应深刻体会“坐标代入、消元化简”的坐标法核心思路。

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（直接应用）：} 已知椭圆方程，求椭圆上某点与长轴端点连线的斜率乘积。
		\item \textbf{场景二（逆向求解）：} 已知椭圆上一点与长轴端点的斜率乘积，反求椭圆的 $a,b$ 或离心率。
		\item \textbf{场景三（定值证明）：} 在综合题中直接引用该结论，简化定值问题的计算。
	\end{itemize}

\end{explanationbox}
