\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略端点约束）}}]
			直接在长轴端点 $A_{1},A_{2}$ 上使用斜率公式，误认为结论也成立。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件必备）：}}]
			斜率定义要求分母不为零，故点 $P$ 不能与 $A_{1},A_{2}$ 重合；此外，结论推导中也约去了 $a^{2}-x_{0}^{2}$，若 $x_{0}=\pm a$ 则会除以零。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义域）：}}]
			将公式记忆为“任意点都成立”，忽略了斜率的自然定义域。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（焦点位置混淆）}}]
			将焦点在 $x$ 轴的结论 $-\dfrac{b^{2}}{a^{2}}$ 机械套用在焦点在 $y$ 轴的椭圆上。
		\item[\textbf{\textcolor{CTrap}{正确理解（a,b角色互换）：}}]
			当椭圆长轴在 $y$ 轴上时，方程变为 $\dfrac{y^{2}}{a^{2}}+\dfrac{x^{2}}{b^{2}}=1\;(a>b>0)$，结论相应为 $-\dfrac{a^{2}}{b^{2}}$（$a$ 为半长轴，$b$ 为半短轴）。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号记忆混乱）：}}]
			死记结论时未联系 $a,b$ 的几何意义，导致系数位置颠倒。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（正负号错误）}}]
			将乘积结果记为正数 $\dfrac{b^{2}}{a^{2}}$，漏掉负号。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号确定）：}}]
			两斜率一正一负（因 $A_{1},A_{2}$ 位于 $P$ 两侧），乘积必然为负，且最终化简得 $-\dfrac{b^{2}}{a^{2}}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导跳步）：}}]
			化简时忽略分子 $a^{2}-x_{0}^{2}$ 与分母 $x_{0}^{2}-a^{2}$ 互为相反数，直接约分得到 $\dfrac{b^{2}}{a^{2}}$ 而未考虑符号。
	\end{description}

\end{trapbox}
