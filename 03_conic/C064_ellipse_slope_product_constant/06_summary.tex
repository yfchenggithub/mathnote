\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{椭圆上任意一点 $P$（非长轴端点）与长轴两端点 $A_{1},A_{2}$ 连线的斜率乘积为定值 $-\dfrac{b^{2}}{a^{2}}$，仅由椭圆形状决定。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（椭圆标准形）：} 椭圆方程为 $\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1\;(a>b>0)$，焦点在 $x$ 轴上。
		\item \textbf{条件2（点P约束）：} 点 $P$ 在椭圆上，且 $P$ 不与 $A_{1}(-a,0),A_{2}(a,0)$ 重合。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（端点排除）：} 切勿在长轴端点处使用该结论（分母为零）。
		\item \textbf{检查项（焦点辨别）：} 若椭圆焦点在 $y$ 轴（方程 $\dfrac{y^{2}}{a^{2}}+\dfrac{x^{2}}{b^{2}}=1$），则结论变为 $-\dfrac{a^{2}}{b^{2}}$，注意 $a,b$ 角色互换。
	\end{itemize}

\end{summarybox}
