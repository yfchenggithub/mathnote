\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（椭圆标准形）：}}]
			椭圆 $\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1\;(a>b>0)$，长轴端点为 $A_{1}(-a,0),A_{2}(a,0)$。点 $P$ 在椭圆上，且异于 $A_{1},A_{2}$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（第三定义）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆上任何一点（非长轴端点）与长轴两端点连线的斜率乘积是恒定的负值。
			\[
				k_{PA_{1}}\cdot k_{PA_{2}} = -\frac{b^{2}}{a^{2}}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（焦点在$y$轴）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若椭圆长轴在 $y$ 轴上（方程 $\dfrac{y^{2}}{a^{2}}+\dfrac{x^{2}}{b^{2}}=1$，$a>b>0$），长轴端点为 $B_{1}(0,-a),B_{2}(0,a)$，则类似结论为
			\[
				k_{PB_{1}}\cdot k_{PB_{2}} = -\frac{a^{2}}{b^{2}}.
			\]
	\end{description}

\end{statementbox}
