\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	通过“设坐标 $\rightarrow$ 写斜率 $\rightarrow$ 代方程 $\rightarrow$ 消元化简”的标准解析几何流程，逐步推导定值。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设点表示斜率）：}]
			设 $P(x_{0},y_{0})$ 为椭圆上一点（$x_{0}\neq\pm a$）。长轴端点为 $A_{1}(-a,0)$，$A_{2}(a,0)$。则
			\[
				k_{PA_{1}}=\frac{y_{0}}{x_{0}+a},
				\qquad
				k_{PA_{2}}=\frac{y_{0}}{x_{0}-a}.
			\]
			\par\textbf{理由：} 直接利用两点间斜率公式 $k=\dfrac{y_{2}-y_{1}}{x_{2}-x_{1}}$。
		\item[\textbf{步骤二（写出乘积）：}]
			\[
				\begin{aligned}
					k_{PA_{1}}\cdot k_{PA_{2}} &= \frac{y_{0}}{x_{0}+a}\cdot\frac{y_{0}}{x_{0}-a} \\ &= \frac{y_{0}^{2}}{(x_{0}+a)(x_{0}-a)} \\ &= \frac{y_{0}^{2}}{x_{0}^{2}-a^{2}}.
			\end{aligned}
			\]
			\par\textbf{理由：} 分母利用平方差公式 $(x_{0}+a)(x_{0}-a)=x_{0}^{2}-a^{2}$。
		\item[\textbf{步骤三（椭圆方程变形）：}]
			因为 $P$ 在椭圆上，所以 $\dfrac{x_{0}^{2}}{a^{2}}+\dfrac{y_{0}^{2}}{b^{2}}=1$。移项得
			\[
				\begin{aligned}
					\frac{y_{0}^{2}}{b^{2}} &= 1-\frac{x_{0}^{2}}{a^{2}} \\ &= \frac{a^{2}-x_{0}^{2}}{a^{2}}.
			\end{aligned}
			\]
			两边乘以 $b^{2}$ 得到
			\[
				\begin{aligned}
					y_{0}^{2} &= b^{2}\cdot\frac{a^{2}-x_{0}^{2}}{a^{2}} \\ &= \frac{b^{2}}{a^{2}}(a^{2}-x_{0}^{2}).
			\end{aligned}
			\]
			\par\textbf{理由：} 从椭圆方程中解出 $y_{0}^{2}$，为代入消元做准备。
		\item[\textbf{步骤四（代入消元）：}]
			\[
				\begin{aligned}
					k_{PA_{1}}\cdot k_{PA_{2}} &= \frac{\dfrac{b^{2}}{a^{2}}(a^{2}-x_{0}^{2})}{x_{0}^{2}-a^{2}} \\ &= \frac{b^{2}}{a^{2}}\cdot\frac{a^{2}-x_{0}^{2}}{x_{0}^{2}-a^{2}}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入并整理因子。
		\item[\textbf{步骤五（化简得定值）：}]
			注意到 $a^{2}-x_{0}^{2}=-(x_{0}^{2}-a^{2})$，因此
			\[
				\frac{a^{2}-x_{0}^{2}}{x_{0}^{2}-a^{2}} = -1.
			\]
			故
			\[
				\begin{aligned}
					k_{PA_{1}}\cdot k_{PA_{2}} &= \frac{b^{2}}{a^{2}}\cdot(-1) \\ &= -\frac{b^{2}}{a^{2}}.
			\end{aligned}
			\]
			\par\textbf{理由：} 分子分母互为相反数，约分即得与 $P$ 无关的常数。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{至此，我们严格推导了椭圆第三定义：椭圆上任意一点（非长轴端点）与长轴两端点连线的斜率乘积恒等于 $-\dfrac{b^{2}}{a^{2}}$。该结论完全由椭圆方程决定，与点 $P$ 的具体位置无关。}
\end{proofbox}
