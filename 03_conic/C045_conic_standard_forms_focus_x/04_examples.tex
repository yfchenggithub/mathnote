	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆的标准方程为 $\frac{x^2}{25}+\frac{y^2}{16}=1$，求椭圆的焦点坐标。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			\[
				\begin{aligned}
					a^2 &= 25,\\
					b^2 &= 16.
			\end{aligned}
			\]
			\par\textbf{理由：} 椭圆标准方程 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ 中，分母即为 $a^2,b^2$。

		\item[\textbf{第二步（计算半焦距）：}]
			\[
				c = 3.
			\]
			\par\textbf{理由：} 由椭圆参数关系 $c^2=a^2-b^2=25-16=9$，所以 $c=3$。

		\item[\textbf{第三步（写出焦点坐标）：}]
			\[
				F_1(-3,0),\ F_2(3,0).
			\]
			\par\textbf{理由：} 焦点在 $x$ 轴上，横坐标为 $\pm c$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{椭圆的焦点坐标为 $(-3,0)$ 和 $(3,0)$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线方程为 $\frac{x^2}{9}-\frac{y^2}{16}=1$，求实轴长、虚轴长、焦点坐标。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			\[
				\begin{aligned}
					a^2 &= 9,\\
					b^2 &= 16.
			\end{aligned}
			\]
			\par\textbf{理由：} 双曲线标准方程 $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$ 中分母对应 $a^2,b^2$。

		\item[\textbf{第二步（计算焦距与轴长）：}]
			\[
				\begin{aligned}
					c &= 5,\\
					2a &= 6,\\
					2b &= 8.
			\end{aligned}
			\]
			\par\textbf{理由：} $c^2=a^2+b^2=9+16=25$，所以 $c=5$；实轴长 $2a=6$，虚轴长 $2b=8$。

		\item[\textbf{第三步（写出焦点坐标）：}]
			\[
				F_1(-5,0),\ F_2(5,0).
			\]
			\par\textbf{理由：} 焦点在 $x$ 轴上，坐标为 $(\pm c,0)$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{实轴长 $6$，虚轴长 $8$，焦点 $(5,0)$ 和 $(-5,0)$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆的焦点在 $x$ 轴上，焦距为 $8$，长轴长为 $10$，求椭圆的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定 $a$ 和 $c$）：}]
			\[
				\begin{aligned}
					a &= 5,\\
					c &= 4.
			\end{aligned}
			\]
			\par\textbf{理由：} 长轴长 $2a=10$，得 $a=5$；焦距 $2c=8$，得 $c=4$。

		\item[\textbf{第二步（计算 $b$）：}]
			\[
				b = 3.
			\]
			\par\textbf{理由：} $b^2=a^2-c^2=25-16=9$，所以 $b=3$。

		\item[\textbf{第三步（写出标准方程）：}]
			\[
				\frac{x^2}{25}+\frac{y^2}{9}=1.
			\]
			\par\textbf{理由：} 焦点在 $x$ 轴时椭圆标准方程形式为 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$，代入 $a^2=25,b^2=9$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{椭圆的标准方程为 $\frac{x^2}{25}+\frac{y^2}{9}=1$。}
