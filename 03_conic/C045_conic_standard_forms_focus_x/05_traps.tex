\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（参数混淆）}}] 在椭圆中，误认为 $c^2=a^2+b^2$，导致焦距计算错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（参数定义）}}] 椭圆中 $a^2=b^2+c^2$，即 $c^2=a^2-b^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆干扰）}}] 与双曲线中 $c^2=a^2+b^2$ 相混淆，未注意椭圆中 $a$ 最大。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（焦点定位）}}] 对于双曲线 $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$ 与 $\frac{y^2}{a^2}-\frac{x^2}{b^2}=1$，误将焦点判断到 $y$ 轴。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数定性）}}] 哪个二次项系数为正，焦点就在哪个轴上；标准形式中 $x^2$ 项系数为正时焦点在 $x$ 轴。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号误读）}}] 只关注分母大小，忽略符号，导致定位错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（抛物线条乱）}}] 将 $y^2=2px$ 的焦点记成 $(0,\frac{p}{2})$，准线记成 $y=-\frac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（焦点准线）}}] $y^2=2px$ 的焦点是 $(\frac{p}{2},0)$，准线是 $x=-\frac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（坐标混淆）}}] 与 $x^2=2py$ 的形式记混，未确认一次项对应的变量。
	\end{description}
\end{trapbox}
