\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（对称轴与中心）：}}] 圆锥曲线以坐标轴为对称轴；椭圆和双曲线的中心在原点，抛物线的顶点在原点。
		\item[\textbf{\textcolor{CStatement}{条件二（焦点位置）：}}] 焦点在 $x$ 轴上。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（椭圆标准方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			动点到两定点距离和为常数 $2a$，轨迹为椭圆。
			\[
				\frac{x^2}{a^2}+\frac{y^2}{b^2}=1,
				\qquad
				b^2=a^2-c^2,\ a>c>0.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（双曲线标准方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			动点到两定点距离差绝对值为常数 $2a$，轨迹为双曲线。
			\[
				\frac{x^2}{a^2}-\frac{y^2}{b^2}=1,
				\qquad
				b^2=c^2-a^2,\ 0<a<c.
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（抛物线标准方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			动点到定点与定直线距离相等，轨迹为抛物线。
			\[
				y^2=2px,
				\qquad
				p>0.
			\]
			焦点 $(\frac{p}{2},0)$，准线 $x=-\frac{p}{2}$。

		\item[\textbf{\textcolor{CStatement}{推广结论（焦点在纵轴）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点在 $y$ 轴时，椭圆、双曲线交换 $x$ 与 $y$，抛物线方程为 $x^2=2py$。
			\par
			椭圆：
			\[
				\frac{y^2}{a^2}+\frac{x^2}{b^2}=1,
				\qquad
				b^2=a^2-c^2.
			\]
			双曲线：
			\[
				\frac{y^2}{a^2}-\frac{x^2}{b^2}=1,
				\qquad
				b^2=c^2-a^2.
			\]
			抛物线：
			\[
				x^2=2py,
				\qquad
				p>0.
			\]
			焦点 $(0,\frac{p}{2})$，准线 $y=-\frac{p}{2}$。
	\end{description}

	\textbf{\textcolor{CStatement}{参数限制：}}
	\begin{itemize}[leftmargin=*, itemsep=0.3em]
		\item 椭圆：$a>c>0$，$b^2=a^2-c^2$，长轴在 $x$ 轴。
		\item 双曲线：$c>a>0$，$b^2=c^2-a^2$，实轴在 $x$ 轴。
		\item 抛物线：$p>0$，焦点在 $x$ 轴正半轴，开口向右。
	\end{itemize}

\end{statementbox}
