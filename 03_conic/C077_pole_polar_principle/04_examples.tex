\begin{examplebox}
\textbf{\textcolor{CExample}{例 1：一个圆，一次看清圆内、圆上、圆外}}
已知圆 $x^2+y^2=4$，求 $P_1(4,0)$、$P_2(2,0)$、$P_3(1,0)$、$P_4(0,0)$ 的极线，并解释其意义。

\textbf{第 1 步：}半径 $R=2$，故 $R^2=4$；圆的极线统一为 $x_0x+y_0y=4$。

\textbf{第 2 步：}代 $P_1(4,0)$ 得 $4x=4$，即 $x=1$。与圆联立：$1+y^2=4$，得 $y=\pm\sqrt3$，即两切点 $T_1(1,\sqrt3),T_2(1,-\sqrt3)$。检查切线是否经过 $P_1$：在 $T_1$ 的切线是 $x+\sqrt3y=4$，在 $T_2$ 的切线是 $x-\sqrt3y=4$；把 $(4,0)$ 代入都成立。因此 $x=1$ 是真正的切点弦所在直线。

\textbf{第 3 步：}代 $P_2(2,0)$ 得 $2x=4$，即 $x=2$。它与圆只有交点 $(2,0)$，是该点处的切线。

\textbf{第 4 步：}代 $P_3(1,0)$ 得 $x=4$。代回圆：$16+y^2=4$，即 $y^2=-12$，没有实数解。所以\textbf{极线存在，但没有实切点}。

\textbf{第 5 步：}代 $P_4(0,0)$ 得 $0=4$，不表示平面里的一条直线，因此圆心没有有限极线。

\textbf{结果：} $P_1\mapsto x=1$（切点弦）；$P_2\mapsto x=2$（切线）；$P_3\mapsto x=4$（没有实交点的极线）；$P_4$ 无有限极线。

\medskip
\textbf{\textcolor{CExample}{例 2：点在椭圆上，求切线（直接应用）}}
已知椭圆 $\dfrac{x^2}{4}+y^2=1$，点 $P\bigl(1,\dfrac{\sqrt3}{2}\bigr)$，求该点处切线。

\textbf{第 1 步：}把点代入曲线：$\dfrac{1^2}{4}+(\dfrac{\sqrt3}{2})^2=\dfrac14+\dfrac34=1$，说明 $P$ 在椭圆上。

\textbf{第 2 步：}由 $a^2=4,b^2=1,x_0=1,y_0=\sqrt3/2$，极线为
\[
\frac{1\cdot x}{4}+\frac{(\sqrt3/2)y}{1}=1.
\]

\textbf{第 3 步：}两边乘 $4$，得到 $x+2\sqrt3y=4$。因为 $P$ 在曲线上，此极线就是切线。

\textbf{结果：} $\boxed{x+2\sqrt3y=4}$。

\medskip
\textbf{\textcolor{CExample}{例 3：原题保留，但明确“有极线，不一定有切点弦”}}
已知双曲线 $x^2-\dfrac{y^2}{3}=1$，$P=(2,1)$。求极线，并判断是否存在两实切点。

\textbf{第 1 步：}代入极线公式（$A^2=1,B^2=3$）：
\[
2x-\frac{y}{3}=1\quad\Longleftrightarrow\quad 6x-y=3.
\]

\textbf{第 2 步：}由极线得 $y=6x-3$，代入双曲线：
\[
\begin{aligned}
x^2-\frac{(6x-3)^2}{3}&=1,\\
3x^2-(36x^2-36x+9)&=3,\\
-33x^2+36x-12&=0,\\
11x^2-12x+4&=0.
\end{aligned}
\]
判别式 $\Delta=(-12)^2-4\cdot11\cdot4=144-176=-32<0$，无实交点。

\textbf{结果：} 极线 $\boxed{6x-y=3}$ \textbf{存在}，但这条线上没有实切点，因此\textbf{实切点弦不存在}。不能因为能写出一个方程，就认定题目里的几何对象存在。

\medskip
\textbf{\textcolor{CExample}{例 4：双曲线也可以真的有两个切点}}
还是双曲线 $x^2-y^2/3=1$，把已知点换成 $P(0,2)$。

\textbf{第 1 步：}极线：$0\cdot x-\dfrac{2y}{3}=1$，解得 $y=-\dfrac32$。

\textbf{第 2 步：}把 $y=-3/2$ 代入双曲线：
\[
x^2-\frac{(-3/2)^2}{3}=1
\quad\Longrightarrow\quad x^2-\frac34=1
\quad\Longrightarrow\quad x^2=\frac74.
\]
解得两切点
\[
T_1\left(\frac{\sqrt7}{2},-\frac32\right),\qquad
T_2\left(-\frac{\sqrt7}{2},-\frac32\right).
\]

\textbf{第 3 步：}检查两点处的切线：
\[
\frac{\sqrt7}{2}x+\frac12y=1,
\qquad
-\frac{\sqrt7}{2}x+\frac12y=1,
\]
也即 $\sqrt7x+y=2$ 和 $-\sqrt7x+y=2$。将 $P(0,2)$ 代入，两式均成立。

\textbf{结果：}此时有两条实切线；两切点连线就是极线 $\boxed{y=-\dfrac32}$。

\textbf{\textcolor{CExample}{例 5：抛物线切点弦——不需要先猜两个切点}}
抛物线 $y^2=4x$，外点 $P(0,1)$。求从 $P$ 作两切线的切点连线，顺便验证切点。


\begin{center}\input{assets/tikz/C077_parabola_contact_chord.tex}\end{center}
\textbf{第 1 步：}比较 $y^2=4x$ 与 $y^2=2px$，得 $2p=4$，所以 $p=2$。

\textbf{第 2 步：}由 $y_0y=p(x+x_0)$，将 $x_0=0,y_0=1$ 代入：$y=2x$。这就是\textbf{候选切点连线}。

\textbf{第 3 步：}联立 $y=2x$ 和 $y^2=4x$，得 $(2x)^2=4x$，即 $4x(x-1)=0$。解出 $x=0$ 或 $1$，对应 $y=0$ 或 $2$。所以 $T_1(0,0),T_2(1,2)$ 两点不同。

\textbf{第 4 步：}切线检验：在 $T_1(0,0)$ 的切线是 $x=0$；在 $T_2(1,2)$ 的切线由 $2y=2(x+1)$ 得 $y=x+1$。两条线都经过 $P(0,1)$。

\textbf{结果：}切点弦所在直线为 $\boxed{y=2x}$，不是两条切线 $x=0$ 或 $y=x+1$。

\textbf{\textcolor{CExample}{例 6：配极原理——把点线关系反向使用}}
椭圆 $x^2/4+y^2=1$，已知 $P(4,0)$，取 $Q(1,1)$。验证“$Q$ 在 $P$ 的极线上”与“$P$ 在 $Q$ 的极线上”。

\textbf{第 1 步：} $P$ 的极线为 $4x/4+0\cdot y=1$，即 $x=1$；$Q(1,1)$ 满足它。

\textbf{第 2 步：} $Q$ 的极线为 $x/4+y=1$。代入 $P(4,0)$：$4/4+0=1$，所以 $P$ 也在 $Q$ 的极线上。

\textbf{第 3 步：}为什么两边恰好相等？因为 $4\cdot1/4+0\cdot1$ 交换乘数位置，数值没有改变。

\textbf{结果：}配极原理不需要另找对称轴，也不要求 $P,Q$ 在椭圆上；它只是在\textbf{两条对应直线之间交换点线条件}。

\textbf{\textcolor{CExample}{例 7：双曲线一个有限交点，不等于极线是切线}}
双曲线 $x^2-y^2/3=1$，点 $P(1,\sqrt3)$。

\textbf{第 1 步：}极线为 $x-\dfrac{\sqrt3y}{3}=1$，即 $x-\dfrac{y}{\sqrt3}=1$，所以 $y=\sqrt3(x-1)$。

\textbf{第 2 步：}代入双曲线：
\[
x^2-\frac{3(x-1)^2}{3}=1
\quad\Longrightarrow\quad x^2-(x-1)^2=1
\quad\Longrightarrow\quad 2x-1=1.
\]
得到唯一有限交点 $T(1,0)$。

\textbf{第 3 步：}双曲线在 $T(1,0)$ 的切线是 $x=1$，与极线 $x-y/\sqrt3=1$ \textbf{不是同一条线}。

\textbf{结果：}与曲线“只有一个有限交点”不必然意味着相切。出现一次方程时，要检查是否另一交点没有落在有限平面中；高考解答中只需知道\textbf{别把它误认为切线}。
