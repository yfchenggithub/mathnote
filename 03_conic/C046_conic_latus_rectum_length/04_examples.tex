\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^2}{16}+\frac{y^2}{9}=1$，求该椭圆的通径长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（参数识别）：}]
			由标准方程得 $a^2=16$，$b^2=9$，所以 $a=4$，$b=3$。
			\par\textbf{理由：} 椭圆标准方程中分母对应 $a^2$，$b^2$。
		\item[\textbf{第二步（公式套用）：}]
			通径长 $L=\frac{2b^2}{a}=\frac{2\times9}{4}=\frac{9}{2}$。
			\par\textbf{理由：} 直接应用通径公式 $L=2b^2/a$。
		\item[\textbf{第三步（结论确认）：}]
			故通径长为 $\frac{9}{2}$。
			\par\textbf{理由：} 得到最终结果。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{通径长 $L=9/2$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ ($a>b>0$) 的通径长为 $\sqrt{3}$，且 $b=1$，求 $a$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列公式）：}]
			由通径公式 $L=2b^2/a$，将 $L=\sqrt{3}$，$b=1$ 代入得 $\sqrt{3}=2\cdot1^2/a$。
			\par\textbf{理由：} 通径公式。
		\item[\textbf{第二步（解方程）：}]
			由 $\sqrt{3}=2/a$ 得 $a=2/\sqrt{3}$，即 $a=2\sqrt{3}/3$。
			\par\textbf{理由：} 解简单分式方程。
		\item[\textbf{第三步（条件检验）：}]
			$a=2\sqrt{3}/3\approx1.155>1=b$，满足 $a>b>0$，故 $a=2\sqrt{3}/3$。
			\par\textbf{理由：} 确保椭圆参数条件成立。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=2\sqrt{3}/3$。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $C:\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ ($a>b>0$) 的离心率 $e=\frac{\sqrt{2}}{2}$，且椭圆通径长与抛物线 $y^2=2px$ ($p>0$) 的通径长相等，求 $p$ 与 $a$ 的比值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（椭圆通径表达）：}]
			由 $e=c/a$，且 $e=\sqrt{2}/2$，得 $c=a/\sqrt{2}$。由 $b^2=a^2-c^2$，代入 $c^2=a^2/2$，得 $b^2=a^2-\frac{a^2}{2}$，即 $b^2=\frac{a^2}{2}$。故 $L_e=\frac{2b^2}{a}$，计算得 $L_e=a$。
			\par\textbf{理由：} 离心率定义及通径公式。
		\item[\textbf{第二步（抛物线通径）：}]
			抛物线通径长 $L_p=2p$。
			\par\textbf{理由：} 抛物线通径公式。
		\item[\textbf{第三步（相等条件）：}]
			由 $L_e=L_p$ 得 $a=2p$，所以 $\frac{p}{a}=\frac{1}{2}$。
			\par\textbf{理由：} 列等式并化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\displaystyle\frac{p}{a}=\frac{1}{2}$。}
\end{examplebox}
