\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（标准方程式）：}}]
			圆锥曲线为标准形式：椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ ($a>b>0$)，双曲线 $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$，抛物线 $y^2=2px$ 或 $x^2=2py$ ($p>0$)。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（有心曲线通径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆和双曲线中，过焦点且垂直于长轴（实轴）的弦称为通径，其长度为 $\frac{2b^2}{a}$。
			\[
				L = \frac{2b^2}{a}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（抛物线通径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			抛物线中，过焦点且垂直于对称轴的弦称为通径，其长度为 $2p$。
			\[
				L = 2p.
			\]
	\end{description}

\end{statementbox}
