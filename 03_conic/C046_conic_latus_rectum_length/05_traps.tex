\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略垂直）}}]
			认为过焦点的任意弦都是通径，直接使用通径公式计算长度。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义约束）：}}]
			通径必须同时满足“过焦点”和“垂直于对称轴”两个条件，其他焦点弦需使用一般弦长公式。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆片面）：}}]
			只记得通径过焦点，忽略了垂直要求，导致对通径的定义理解不完整。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公式颠倒）}}]
			将椭圆或双曲线的通径公式错记为 $2a^2/b$ 或 $b^2/a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式）：}}]
			正确通径公式为 $\frac{2b^2}{a}$，其中 $a$ 是长半轴（或实半轴），$b$ 是短半轴（或虚半轴）。
		\item[\textbf{\textcolor{CTrap}{错因分析（变量混淆）：}}]
			对 $a$、$b$ 在椭圆和双曲线中的作用不清晰，导致公式中 $a$、$b$ 位置颠倒。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（漏乘2）}}]
			将抛物线的通径长误记为 $p$，忽略 $2p$。
		\item[\textbf{\textcolor{CTrap}{正确理解（倍焦准距）：}}]
			抛物线通径长等于 $2p$（$p$ 为焦点到准线的距离）。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）：}}]
			不清楚通径与焦准距的关系，或与椭圆双曲线通径混淆。
	\end{description}

\end{trapbox}
