\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{抛物线$y^2=2px$的弦中点$M(x_0,y_0)$与弦斜率$k$满足$k=\dfrac{p}{y_0}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（弦非竖直）：}弦$AB$不与$y$轴平行，即斜率存在。
		\item \textbf{条件2（纵标非零）：}中点的纵坐标$y_0\neq0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件验证）：}使用公式前必须检查弦是否与$y$轴平行及$y_0$是否为零。
		\item \textbf{检查项（$p$值确认）：}确定$p$时应将方程化为$y^2=2px$形式，$p$为$x$系数的一半。
	\end{itemize}
\end{summarybox}
