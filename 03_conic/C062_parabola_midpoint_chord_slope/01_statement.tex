	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中点弦斜率）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			弦的斜率等于$p$除以中点纵坐标。
			\[
				k = \frac{p}{y_0}
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（反解形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			中点纵坐标等于$p$除以弦斜率（$k\neq0$）。
			\[
				y_0 = \frac{p}{k}
			\]
	\end{description}
