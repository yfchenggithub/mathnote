	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（标准方程与点在曲线上）：}}] 抛物线方程为 $y^2=2px$（$p>0$），且点 $P(x_0,y_0)$ 在该抛物线上。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦半径长度等于点横坐标与半通径之和。
			\[
				PF = x_0 + \frac{p}{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（开口向上情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对于开口向上的抛物线，焦半径等于纵坐标与半通径之和。
			\[
				PF = y_0 + \frac{p}{2}.
			\]
	\end{description}

\end{statementbox}
