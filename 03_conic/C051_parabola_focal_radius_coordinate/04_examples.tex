\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=4x$ 上一点 $P$ 的横坐标 $x_0=3$，求点 $P$ 到焦点 $F$ 的距离。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 由 $y^2=4x$ 得 $2p=4$，故 $p=2$。
			\par\textbf{理由：} 标准方程 $y^2=2px$ 对比。
		\item[\textbf{第二步（套用公式）：}] 用焦半径公式 $|PF| = x_0 + \frac{p}{2}$。
			\par\textbf{理由：} 本题结论直接应用。
		\item[\textbf{第三步（计算答案）：}] 代入 $x_0=3$，$p=2$，得 $|PF| = 3 + \frac{2}{2} = 4$。
			\par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|PF|=4$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在抛物线 $y^2=8x$ 上，点 $P$ 到焦点的距离为 $6$，求点 $P$ 的横坐标 $x_0$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 由 $y^2=8x$ 得 $2p=8$，$p=4$。
			\par\textbf{理由：} 标准方程对比。
		\item[\textbf{第二步（列方程）：}] 由焦半径公式 $|PF| = x_0 + \frac{p}{2}=6$。
			\par\textbf{理由：} 已知距离代入公式。
		\item[\textbf{第三步（解方程）：}] $x_0 + \frac{4}{2}=6$，解得 $x_0=4$。
			\par\textbf{理由：} 移项计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{横坐标 $x_0=4$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	抛物线 $y^2=4x$ 的焦点为 $F$，点 $P$ 在抛物线上且 $|PF|=5$，求 $\triangle OFP$ 的面积（$O$ 为坐标原点）。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（参数与横坐标）：}] 由 $y^2=4x$ 得 $p=2$，焦点 $F(1,0)$。由 $|PF|=5$ 得 $x_0+1=5$，所以 $x_0=4$。
			\par\textbf{理由：} 焦半径公式代入。
		\item[\textbf{第二步（求纵坐标）：}] 将 $x_0=4$ 代入 $y^2=4x$，得 $y_0^2=16$，$y_0=\pm 4$。
			\par\textbf{理由：} 点在抛物线上。
		\item[\textbf{第三步（求面积）：}] 底 $|OF|=1$，高 $|y_0|=4$，面积 $S=\frac12\times1\times4=2$。
			\par\textbf{理由：} 三角形面积公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\triangle OFP$ 的面积为 $2$。}
