\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（符号记反）}}] 误将焦半径公式记成 $|PF|=x_0-\dfrac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（坐标加半距）}}] 正确公式为 $|PF|=x_0+\dfrac{p}{2}$，因准线方程是 $x=-\dfrac{p}{2}$，所以距离为 $x_0-\left(-\dfrac{p}{2}\right)=x_0+\dfrac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（焦点混淆）}}] 可能记错准线方程或误将焦点坐标与准线位置混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略非负性）}}] 在去掉绝对值时未考虑 $x_0\ge 0$，导致公式适用范围出错。
		\item[\textbf{\textcolor{CTrap}{正确理解（点横非负）}}] 抛物线 $y^2=2px$ 上点的横坐标 $x_0\ge 0$，故 $|x_0+\dfrac{p}{2}|=x_0+\dfrac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（坐标范围）}}] 忽略了抛物线开口方向对坐标符号的限制。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开口误用）}}] 对于抛物线 $x^2=2py$（开口向上），仍用 $x_0$ 计算焦半径。
		\item[\textbf{\textcolor{CTrap}{正确理解（依开口定式）}}] 对于 $x^2=2py$，准线为 $y=-\dfrac{p}{2}$，焦半径为 $|PF|=y_0+\dfrac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（未灵活变换）}}] 死记公式而未理解公式来源于准线方程。
	\end{description}
\end{trapbox}
