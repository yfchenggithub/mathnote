\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $\frac{x^2}{4} - y^2 = 1$，点 $P(3,1)$ 在双曲线外部，求过 $P$ 的两条切线的切点弦所在直线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			从双曲线方程 $\frac{x^2}{4} - y^2 = 1$ 可得 $a^2 = 4$，$b^2 = 1$，且焦点在 $x$ 轴。
			\par\textbf{理由：} 标准形式 $\frac{x^2}{a^2} - \frac{y^2}{b^2}=1$ 对应 $a^2=4, b^2=1$。
		\item[\textbf{第二步（套用公式）：}]
			使用焦点在 $x$ 轴时的切点弦方程 $\frac{x_0 x}{a^2} - \frac{y_0 y}{b^2} = 1$。
			\par\textbf{理由：} 该定理给出切点弦方程与焦点位置有关。
		\item[\textbf{第三步（代入化简）：}]
			代入 $x_0 = 3$，$y_0 = 1$ 得 $\frac{3x}{4} - \frac{1\cdot y}{1} = 1$，即 $\frac{3x}{4} - y = 1$，整理得 $3x - 4y = 4$。
			\par\textbf{理由：} 直接代数运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	切点弦所在直线方程为 $\boxed{3x-4y=4}$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $\frac{y^2}{9} - \frac{x^2}{4} = 1$，点 $P(2,6)$ 在双曲线外部，求过 $P$ 的两条切线的切点弦方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数与焦点）：}]
			由 $\frac{y^2}{9} - \frac{x^2}{4} = 1$ 得 $a^2 = 9$，$b^2 = 4$，焦点在 $y$ 轴。
			\par\textbf{理由：} 标准形式 $\frac{y^2}{a^2} - \frac{x^2}{b^2}=1$ 对应 $a^2=9, b^2=4$。
		\item[\textbf{第二步（选择公式）：}]
			使用焦点在 $y$ 轴时的切点弦方程 $\frac{y_0 y}{a^2} - \frac{x_0 x}{b^2} = 1$。
			\par\textbf{理由：} 根据焦点位置选择对应公式。
		\item[\textbf{第三步（代入化简）：}]
			代入 $x_0=2, y_0=6$ 得 $\frac{6y}{9} - \frac{2x}{4} = 1$，化简为 $\frac{2y}{3} - \frac{x}{2} = 1$，两边乘以6得 $4y - 3x = 6$。
			\par\textbf{理由：} 代数化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	切点弦方程为 $\boxed{4y-3x=6}$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $\frac{x^2}{2} - \frac{y^2}{3} = 1$，一条切点弦所在直线方程为 $2x-3y=6$，求该切点弦对应的外部点 $P$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化标准形式）：}]
			将直线 $2x-3y=6$ 化为右端为 $1$ 的形式：$\frac{x}{3} - \frac{y}{2} = 1$。
			\par\textbf{理由：} 切点弦方程公式右侧为 $1$，因此需将直线方程改写为 $\frac{x_0 x}{a^2} - \frac{y_0 y}{b^2}=1$ 的结构。
		\item[\textbf{第二步（比较系数）：}]
			对比公式 $\frac{x_0 x}{a^2} - \frac{y_0 y}{b^2}=1$，得 $\frac{x_0}{a^2} = \frac{1}{3}$，$\frac{y_0}{b^2} = \frac{1}{2}$。由双曲线方程知 $a^2=2, b^2=3$。
			\par\textbf{理由：} 相同变量系数对应相等。
		\item[\textbf{第三步（解出 $P$）：}]
			$x_0 = \frac{a^2}{3} = \frac{2}{3}$，$y_0 = \frac{b^2}{2} = \frac{3}{2}$。所以 $P\left( \frac{2}{3}, \frac{3}{2} \right)$。
			\par\textbf{理由：} 代数计算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	外部点 $P$ 的坐标为 $\boxed{\left( \frac{2}{3}, \frac{3}{2} \right)}$。
\end{examplebox}
