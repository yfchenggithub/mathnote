\begin{summarybox}
\textbf{\textcolor{CSummary}{一、快记：两式 + 一个存在条件}}
\[
\boxed{\begin{aligned}
\frac{x^2}{a^2}-\frac{y^2}{b^2}=1
&\ \Longrightarrow\ AB:\ \frac{x_0x}{a^2}-\frac{y_0y}{b^2}=1;\\[3pt]
\frac{y^2}{a^2}-\frac{x^2}{b^2}=1
&\ \Longrightarrow\ AB:\ \frac{y_0y}{a^2}-\frac{x_0x}{b^2}=1.
\end{aligned}}
\]
先令 $q$ 等于把 $P(x_0,y_0)$ 代入标准方程左边所得的值，\textbf{两不同有限实切点存在 $\Longleftrightarrow q<1$ 且 $q\ne0$}。

\medskip
\textbf{\textcolor{CSummary}{二、快懂：为什么“平方换乘积”}}
\[
P\in\text{切点 $A$ 处切线}
\ \Longrightarrow\
\frac{x_1x_0}{a^2}-\frac{y_1y_0}{b^2}=1
\ \Longrightarrow\ A\in AB.
\]
对 $B$ 也成立，两个切点确定这条直线。核心是\textbf{切线式 + 代入 + 交换乘法顺序}。

\medskip
\begin{minipage}{\linewidth}
\textbf{\textcolor{CSummary}{三、三种证明的选择}}
\begin{enumerate}[leftmargin=*,itemsep=0.3em]
\item \textbf{最低门槛：}已知切线式，分别将 $P$ 代入 $A,B$ 处切线，说明 $A,B$ 共线。
\item \textbf{最快速证：}先写候选直线；任何候选交点的切线都通过 $P$，故两交点就是两切点。
\item \textbf{常规通用：}设 $ux+vy=1$，与双曲线联立，用 $\Delta=0$ 求出切点 $(a^2u,-b^2v)$，再利用切线过 $P$。
\end{enumerate}\end{minipage}


\medskip
\textbf{\textcolor{CSummary}{四、考场三步}}
\textbf{第1步} 认准焦点在 $x$ 轴还是 $y$ 轴，并算 $q$。\quad
\textbf{第2步} 检查 $q<1$ 且 $q\ne0$。\quad
\textbf{第3步} 再按原式保留分母与减号，把平方换成乘积。

\medskip
\textbf{\textcolor{CSummary}{五、三个最容易错的地方}}
\begin{itemize}[leftmargin=*,itemsep=0.25em]
\item $q=0$ 时不可能有两个不同的有限切点；$q>1$ 时没有实切点。
\item $P$ 在两条切线上，\textbf{通常不在切点弦 $AB$ 上}。
\item 竖直切线可能存在；用 $y=kx+d$ 推导时不得遗漏。
\end{itemize}

\textbf{一个检验例子：} $x^2/4-y^2=1,\ P=(0,2)$，算得 $q=-4$，切点弦是 $y=-1/2$。
\end{summarybox}
