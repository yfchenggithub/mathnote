	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（曲线与焦点）：}}]
			椭圆、双曲线或抛物线，焦点分别为$F_1,F_2$或$F$，对称轴方向为$d$（抛物线）。
		\item[\textbf{\textcolor{CStatement}{条件2（点线关系）：}}]
			$P$为曲线上任意一点，$l$为$P$处的切线，$n$为$P$处的法线。抛物线中过$P$作平行于对称轴且指向开口方向的射线$PH$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（椭圆法线平分）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆在$P$处的法线平分两焦半径夹角。
			\[
				\angle F_1 P n = \angle n P F_2 .
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（双曲线切线平分）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			双曲线在$P$处的切线平分两焦半径夹角。
			\[
				\angle F_1 P l = \angle l P F_2 .
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（抛物线切线平分）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			抛物线在$P$处的切线平分焦半径与平行轴射线的夹角。
			\[
				\angle F P l = \angle l P H .
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{推广结论（角平分表述）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{推广结论（角平分表述）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆法线平分$\angle F_1PF_2$，双曲线切线平分$\angle F_1PF_2$，抛物线切线平分$\angle FPH$。
	\end{description}

\end{statementbox}
