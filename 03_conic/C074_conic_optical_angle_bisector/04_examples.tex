\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆$C:\frac{x^2}{4}+\frac{y^2}{3}=1$，左右焦点$F_1(-1,0),F_2(1,0)$，点$P(1,\frac{3}{2})$在椭圆上。证明椭圆在$P$处的法线平分$\angle F_1PF_2$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求切线法线斜率）：}]
			切线斜率$k_t=-\frac{b^2x_0}{a^2y_0}=-\frac{3\cdot1}{4\cdot\frac{3}{2}}=-\frac{1}{2}$，由$k_t\cdot k_n=-1$得法线斜率$k_n=2$。
			\par\textbf{理由：} 利用椭圆切线斜率公式（隐函数求导）。
		\item[\textbf{第二步（求焦半径斜率）：}]
			$k_1=\frac{y_0}{x_0+1}=\frac{3/2}{2}=\frac{3}{4}$。$PF_2$垂直$x$轴（$x_0=1$等于$F_2$横坐标），故$k_2$不存在，倾斜角$\frac{\pi}{2}$。
			\par\textbf{理由：} 两点斜率公式。
		\item[\textbf{第三步（验证角平分）：}]
			设法线$n$的倾斜角为$\alpha$，$\tan\alpha=2$；$PF_1$的倾斜角为$\beta$，$\tan\beta=\frac{3}{4}$。需要证明$\alpha-\beta=\frac{\pi}{2}-\alpha$，即$2\alpha=\beta+\frac{\pi}{2}$。两边取正切：$\tan(2\alpha)=\frac{2\tan\alpha}{1-\tan^2\alpha}=\frac{4}{1-4}=-\frac{4}{3}$，$\tan(\beta+\frac{\pi}{2})=-\cot\beta=-\frac{1}{\tan\beta}=-\frac{4}{3}$。故等式成立，从而$\angle F_1Pn=\angle nPF_2$。
			\par\textbf{理由：} 利用正切的两倍角公式及诱导公式验证角度关系。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle F_1Pn=\angle nPF_2$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线$C:x^2-\frac{y^2}{3}=1$，焦点$F_1(-2,0),F_2(2,0)$，点$P(2,3)$在$C$上。（1）求$P$处的切线方程；（2）证明双曲线在$P$处的切线平分$\angle F_1PF_2$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求切线方程）：}]
			双曲线切线方程$x_0x-\frac{y_0y}{3}=1$，代入$x_0=2,y_0=3$得$2x-y=1$，即$y=2x-1$，切线斜率$k_t=2$。
			\par\textbf{理由：} 双曲线切线公式（或由导数）。
		\item[\textbf{第二步（求焦半径斜率）：}]
			$k_1=\frac{y_0}{x_0+2}=\frac{3}{4}$。$x_0=2=F_2$横坐标，故$PF_2$垂直$x$轴，$k_2$不存在，倾斜角$\frac{\pi}{2}$。
			\par\textbf{理由：} 两点斜率公式。
		\item[\textbf{第三步（验证角平分）：}]
			设切线$l$的倾斜角为$\phi$，$\tan\phi=2$；$PF_1$的倾斜角为$\beta$，$\tan\beta=\frac{3}{4}$。需证$\phi-\beta=\frac{\pi}{2}-\phi$，即$2\phi=\beta+\frac{\pi}{2}$。取正切：$\tan(2\phi)=\frac{2\tan\phi}{1-\tan^2\phi}=\frac{4}{1-4}=-\frac{4}{3}$，$\tan(\beta+\frac{\pi}{2})=-\cot\beta=-\frac{1}{\tan\beta}=-\frac{4}{3}$。故$2\phi=\beta+\frac{\pi}{2}$，因此$\angle F_1Pl=\angle lPF_2$。
			\par\textbf{理由：} 利用正切的两倍角及诱导公式，验证角相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle F_1Pl=\angle lPF_2$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线$C:y^2=4x$的焦点为$F$，点$A(5,2)$在$C$外。从焦点$F$发出的光线射到抛物线上的点$P$，经反射后平行于$x$轴且通过点$A$。根据抛物线光学性质（切线平分$\angle FPH$，$PH$平行于$x$轴正向），求点$P$的坐标及光线从$F$到$P$再到$A$的总路程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定反射点纵坐标）：}]
			反射光线平行于$x$轴，故其纵坐标等于$A$的纵坐标$2$，所以$P$的纵坐标$y_0=2$。
			\par\textbf{理由：} 反射光线过$P$和$A$且水平，纵坐标相同。
		\item[\textbf{第二步（求反射点横坐标）：}]
			将$y_0=2$代入抛物线方程$2^2=4x_0$，得$x_0=1$，所以$P(1,2)$。
			\par\textbf{理由：} 点在抛物线上满足方程。
		\item[\textbf{第三步（计算总路程）：}]
			$FP$的距离：$F(1,0),P(1,2)$，故$FP=2$；$PA$的距离：$|PA|=\sqrt{(5-1)^2+(2-2)^2}=4$，总路程$=2+4=6$。
			\par\textbf{理由：} 两点距离公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{反射光线平行于对称轴，总路程为$6$。}
\end{examplebox}
