\begin{statementbox}
\textbf{C013\quad 椭圆长轴端点切线相交性质}

设椭圆
\[
C:\quad \frac{x^2}{a^2}+\frac{y^2}{b^2}=1,\qquad a>b>0,
\]
长轴端点为 $A_1(-a,0)$、$A_2(a,0)$。取椭圆上动点
$P(x_0,y_0)\ne A_1,A_2$（等价于 $y_0\ne0$）。
过 $A_1,A_2$ 的切线分别为 $x=-a$、$x=a$；
它们与 $P$ 处切线分别相交于 $P_1,P_2$。
记 $d_1=|A_1P_1|$、$d_2=|A_2P_2|$，
$S$ 为四边形 $A_1P_1P_2A_2$ 的面积。

\begin{enumerate}
  \item \textbf{乘积定值（核心）}：
  \[
  \boxed{d_1d_2=b^2.}
  \]
  \item \textbf{面积最值}：四边形是以 $A_1P_1$、$A_2P_2$
  为平行边、间距为 $2a$ 的直角梯形（取等时为矩形），
  \[
  \boxed{S=a(d_1+d_2)\ge 2ab.}
  \]
  等号当且仅当 $P=(0,b)$ 或 $P=(0,-b)$；此时
  $d_1=d_2=b$，$S_{\min}=2ab$。
  \item \textbf{同侧性与无最大值}：$P_1,P_2$ 始终位于
  $x$ 轴同侧，且与 $P$ 同侧；随着 $P$ 趋近任一长轴端点，
  一个长度趋于 $0$，另一个趋于 $+\infty$，所以 $S$ 没有最大值。
\end{enumerate}

\textbf{等价速用形式}：若 $P$ 处切线为 $y=kx+m$，则
\[
\boxed{m^2-a^2k^2=b^2},\qquad
\boxed{S=2a|m|=2a\sqrt{a^2k^2+b^2}}.
\]
这里的乘积定值只针对 $x=\pm a$ 这两条\textbf{端点切线}，
不可无条件替换成椭圆的两条准线。
\end{statementbox}
