\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（直线方程）}}] $Ax+By+C=0$，其中$A,B$不同时为零.
		\item[\textbf{\textcolor{CStatement}{条件 2（椭圆形式）}}] 椭圆写成统一形式
		      \[
			      \frac{x^2}{\alpha}+\frac{y^2}{\beta}=1
			      \qquad(\alpha>0,\ \beta>0).
		      \]
		      焦点在$x$轴时，$\alpha=a^2,\ \beta=b^2$，且$a>b>0$；
		      焦点在$y$轴时，$\alpha=b^2,\ \beta=a^2$，且$a>b>0$.
		\item[\textbf{\textcolor{CStatement}{条件 3（消元前提）}}] 消去$y$需$B\neq0$，消去$x$需$A\neq0$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{统一结论}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 联立直线与标准椭圆，直接写出消元后二次方程系数及韦达定理.

		      若$B\neq0$，消去$y$，得到关于$x$的二次方程
		      \[
			      (A^2\alpha+B^2\beta)x^2+2AC\alpha x+(C^2\alpha-B^2\alpha\beta)=0.
		      \]
		      因此
		      \[
			      x_1+x_2=-\frac{2AC\alpha}{A^2\alpha+B^2\beta},
			      \qquad
			      x_1x_2=\frac{C^2\alpha-B^2\alpha\beta}{A^2\alpha+B^2\beta}.
		      \]

		      若$A\neq0$，消去$x$，得到关于$y$的二次方程
		      \[
			      (A^2\alpha+B^2\beta)y^2+2BC\beta y+(C^2\beta-A^2\alpha\beta)=0.
		      \]
		      因此
		      \[
			      y_1+y_2=-\frac{2BC\beta}{A^2\alpha+B^2\beta},
			      \qquad
			      y_1y_2=\frac{C^2\beta-A^2\alpha\beta}{A^2\alpha+B^2\beta}.
		      \]

		\item[\textbf{\textcolor{CStatement}{对应到两类标准椭圆}}]
		      焦点在$x$轴时，
		      \[
			      \frac{x^2}{a^2}+\frac{y^2}{b^2}=1
			      \qquad(a>b>0),
		      \]
		      只需令
		      \[
			      \alpha=a^2,\qquad \beta=b^2.
		      \]

		      焦点在$y$轴时，
		      \[
			      \frac{x^2}{b^2}+\frac{y^2}{a^2}=1
			      \qquad(a>b>0),
		      \]
		      只需令
		      \[
			      \alpha=b^2,\qquad \beta=a^2.
		      \]

		\item[\textbf{\textcolor{CStatement}{判别式与位置关系}}]
		      设消元后的一元二次方程为
		      \[
			      pt^2+qt+r=0,
		      \]
		      其中$t$表示$x$或$y$，判别式为
		      \[
			      \Delta=q^2-4pr.
		      \]
		      则
		      \[
			      \Delta>0
		      \]
		      等价于直线与椭圆有两个不同交点；
		      \[
			      \Delta=0
		      \]
		      等价于直线与椭圆相切；
		      \[
			      \Delta<0
		      \]
		      等价于直线与椭圆没有公共点.
	\end{description}
\end{statementbox}
