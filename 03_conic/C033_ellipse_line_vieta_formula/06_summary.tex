\begin{summarybox}
\textbf{一句话：} 先确认有实交点，选择合法消元方向，再用根的和与积替代逐个求根。

\textbf{速查一：} $x^2/\alpha+y^2/\beta=1$，$Ax+By+C=0$，
$\alpha,\beta>0$，$A^2+B^2>0$，记 $D=A^2\alpha+B^2\beta,H=D-C^2$。
\[
 H>0:\text{两不同交点};\quad H=0:\text{相切};\quad H<0:\text{无公共点}.
\]
\textbf{速查二（$B\ne0$，$H\ge0$）：}
\[
 x_1+x_2=-\frac{2AC\alpha}{D},\qquad
 x_1x_2=\frac{\alpha(C^2-B^2\beta)}{D}.
\]
\textbf{速查三（$A\ne0$，$H\ge0$）：}
\[
 y_1+y_2=-\frac{2BC\beta}{D},\qquad
 y_1y_2=\frac{\beta(C^2-A^2\alpha)}{D}.
\]
\textbf{速查四（$H>0$，所有方向）：}
\[
 M=\left(-\frac{AC\alpha}{D},-\frac{BC\beta}{D}\right),\qquad
 PQ=\frac{2\sqrt{\alpha\beta(A^2+B^2)H}}{D}.
\]
\textbf{最后四查：}
\begin{itemize}[leftmargin=*]
\item $\alpha,\beta$ 是否读为各自分母，即半轴长的平方？
\item $A,B,C$ 的符号是否正确？消元时所除系数非零吗？
\item 实弦要求 $H>0$，是否漏掉相切或无交点情形？
\item 根差和弦长分清了吗？判别式是否与方程系数匹配？
\end{itemize}
\end{summarybox}
