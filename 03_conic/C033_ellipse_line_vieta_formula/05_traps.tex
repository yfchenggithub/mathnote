\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（消元条件忽略）}}] 使用消$y$公式时未检查$B\neq0$，使用消$x$公式时未检查$A\neq0$，直接套用导致无意义.
		\item[\textbf{\textcolor{CTrap}{正确理解（先验前提）}}] 必须先确认所消变量的系数非零. 消去$y$要能由直线解出$y$，所以要求$B\neq0$；消去$x$要能由直线解出$x$，所以要求$A\neq0$.
		\item[\textbf{\textcolor{CTrap}{错因分析（机械记忆）}}] 学生只记忆了公式形式，忽略公式的适用前提，缺乏对推导过程的理解.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（焦点轴混淆）}}] 将焦点在$x$轴与$y$轴的公式混淆，使用错误的$a,b$位置.
		\item[\textbf{\textcolor{CTrap}{正确理解（统一分母）}}] 不必死背两套公式. 先把椭圆写成
		      \[
			      \frac{x^2}{\alpha}+\frac{y^2}{\beta}=1.
		      \]
		      焦点在$x$轴时$\alpha=a^2,\beta=b^2$；焦点在$y$轴时$\alpha=b^2,\beta=a^2$. 套公式时只看$x^2$与$y^2$各自的分母.
		\item[\textbf{\textcolor{CTrap}{错因分析（乱用公式）}}] 未养成先确定$x^2,y^2$分母的习惯，导致$a,b$位置代错.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号与参数）}}] 将直线方程写成非一般式时，参数$A,B,C$对应错误，如将$y=kx+m$写成$kx+y+m=0$等，导致符号错误.
		\item[\textbf{\textcolor{CTrap}{正确理解（严格转换）}}] 使用硬解定理前必须将直线化为
		      \[
			      Ax+By+C=0
		      \]
		      的形式. 例如
		      \[
			      y=kx+m
		      \]
		      应改写为
		      \[
			      kx-y+m=0,
		      \]
		      此时
		      \[
			      A=k,\qquad B=-1,\qquad C=m.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（形式不敏感）}}] 对一般式与斜截式转换不熟练，未注意每一项的符号.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（弦长公式误用）}}] 看到弦长就直接使用
		      \[
			      |AB|=\sqrt{1+k^2}\cdot\frac{\sqrt{\Delta}}{|p|}
		      \]
		      但没有检查直线斜率$k$是否存在.
		\item[\textbf{\textcolor{CTrap}{正确理解（先看是否竖直）}}] 该公式适用于$B\neq0$且采用消去$y$得到关于$x$的二次方程时. 此时
		      \[
			      k=-\frac{A}{B},
			      \qquad
			      |AB|=\sqrt{1+k^2}\cdot\frac{\sqrt{\Delta_x}}{|p|}.
		      \]
		      若$B=0$，直线为竖直线，斜率不存在，不能使用含$k$的弦长公式，应改用纵坐标根差或单独处理.
		\item[\textbf{\textcolor{CTrap}{错因分析（公式套用过快）}}] 只记住弦长模板，忽略了斜率存在的前提.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点五（相交与相切混说）}}] 将$\Delta>0$简单说成“相交”，容易把相切也混进去.
		\item[\textbf{\textcolor{CTrap}{正确理解（判别式三分法）}}] 消元后若判别式为$\Delta$，则
		      \[
			      \Delta>0
		      \]
		      表示两个不同交点；
		      \[
			      \Delta=0
		      \]
		      表示相切；
		      \[
			      \Delta<0
		      \]
		      表示没有公共点.
		\item[\textbf{\textcolor{CTrap}{错因分析（语言不严谨）}}] “相交”在口语中有时包含“相切”，但做题时必须区分“两个不同交点”和“一个切点”.
	\end{description}
\end{trapbox}
