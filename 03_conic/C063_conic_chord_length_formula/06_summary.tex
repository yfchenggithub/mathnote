\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{弦长公式 $|AB|=\sqrt{1+k^2}\,|x_1-x_2|$（$k$存在时），是直线与圆锥曲线相交弦长的直接计算公式，可极大简化运算。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（斜率存在）：} 直线斜率 $k$ 存在。
		\item \textbf{条件 2（两个交点）：} 直线与圆锥曲线相交于两个不同点，联立后判别式大于0。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（纵版本条件）：} 纵坐标版本仅当 $k\neq0$ 时可用。
		\item \textbf{检查项（绝对值不可省）：} 计算 $|x_1-x_2|$ 时需通过平方开方或韦达定理取得正值。
	\end{itemize}
\end{summarybox}
