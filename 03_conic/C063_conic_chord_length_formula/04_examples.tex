\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $y=x+1$ 与圆 $x^2+y^2=5$ 相交于 $A,B$ 两点，求弦长 $|AB|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（联立消元）：}] 将 $y=x+1$ 代入圆方程，得 $x^2+(x+1)^2=5$，化简得 $2x^2+2x-4=0$，即 $x^2+x-2=0$。
			\par\textbf{理由：} 交点坐标满足两个方程，消去 $y$ 得关于 $x$ 的方程。
		\item[\textbf{第二步（韦达求差）：}] 由 $x^2+x-2=0$ 得 $x_1+x_2=-1$，$x_1x_2=-2$，于是
			\[
				\begin{aligned}
					|x_1-x_2| &= \sqrt{(x_1+x_2)^2-4x_1x_2} \\
					&= \sqrt{1+8} \\
					&= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用两根和积计算差的绝对值。
		\item[\textbf{第三步（横坐标公式）：}] 已知斜率 $k=1$，代入
			\[
				\begin{aligned}
					|AB| &= \sqrt{1+k^2}\,|x_1-x_2| \\
					&= \sqrt{2}\times3 \\
					&= 3\sqrt{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接应用弦长公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|AB|=3\sqrt{2}$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=2x$，直线 $y=2x-1$ 交抛物线于 $A,B$ 两点，求 $|AB|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（改直线形式）：}] 由 $y=2x-1$ 解出 $x=\frac{y+1}{2}$。
			\par\textbf{理由：} 将直线改写为用 $y$ 表示 $x$，便于代入抛物线。
		\item[\textbf{第二步（联立得 $y$ 方程）：}] 代入 $y^2=2x$，得 $y^2=2\times\frac{y+1}{2}=y+1$，即 $y^2-y-1=0$。
			\par\textbf{理由：} 消去 $x$，得到关于 $y$ 的二次方程。
		\item[\textbf{第三步（韦达求纵差）：}] 由 $y^2-y-1=0$ 得 $y_1+y_2=1$，$y_1y_2=-1$，所以
			\[
				\begin{aligned}
					|y_1-y_2| &= \sqrt{(y_1+y_2)^2-4y_1y_2} \\
					&= \sqrt{1+4} \\
					&= \sqrt{5}.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用韦达定理求纵坐标差的绝对值。
		\item[\textbf{第四步（纵坐标公式）：}] 已知 $k=2$，代入
			\[
				\begin{aligned}
					|AB| &= \sqrt{1+\frac{1}{k^2}}\,|y_1-y_2| \\
					&= \sqrt{1+\frac14}\times\sqrt{5} \\
					&= \frac{\sqrt5}{2}\times\sqrt5 \\
					&= \frac{5}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接应用纵坐标形式的弦长公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|AB|=\frac{5}{2}$}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^2}{2}+y^2=1$，斜率为 $1$ 的直线 $l:y=x+m$ 与椭圆交于 $A,B$ 两点，求弦长 $|AB|$ 的最大值，并求此时 $m$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（联立方程）：}] 将 $y=x+m$ 代入椭圆方程，得 $\frac{x^2}{2}+(x+m)^2=1$，化简得 $\frac32x^2+2mx+m^2-1=0$，两边乘 $2$ 得 $3x^2+4mx+2(m^2-1)=0$。
			\par\textbf{理由：} 联立消 $y$ 得到关于 $x$ 的二次方程。
		\item[\textbf{第二步（判别式约束）：}] 要求直线与椭圆有两个交点。
			\[
				\begin{aligned}
					\Delta &= 16m^2 - 4 \cdot 3 \cdot 2(m^2-1) \\
					&= 16m^2 - 24(m^2-1) \\
					&= 24 - 8m^2 .
			\end{aligned}
			\]
			由 $\Delta > 0$，得 $24 - 8m^2 > 0$，解得 $-\sqrt{3}<m<\sqrt{3}$。
			\par\textbf{理由：} 判别式大于0确保有两个不同的交点。
		\item[\textbf{第三步（韦达定理）：}] $x_1+x_2=-\frac{4m}{3}$，$x_1x_2=\frac{2(m^2-1)}{3}$。
			\par\textbf{理由：} 二次方程根与系数的关系。
		\item[\textbf{第四步（弦长表达式）：}] 由弦长公式和韦达定理，代入得：
			\[
				\begin{aligned}
					|AB|&=\sqrt2\,\sqrt{\left(-\frac{4m}{3}\right)^2-4\cdot\frac{2(m^2-1)}{3}} \\
					&=\sqrt2\,\sqrt{\frac{16m^2}{9}-\frac{8(m^2-1)}{3}} \\
					&=\sqrt2\,\sqrt{\frac{24-8m^2}{9}} \\
					&=\frac{4}{3}\sqrt{3-m^2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入弦长公式并化简。
		\item[\textbf{第五步（求最值）：}] 当 $m=0$ 时，$3-m^2$ 取最大值 $3$，所以
			\[
				\begin{aligned}
					|AB|_{\max} &= \frac{4}{3}\sqrt{3} \\
					&= \frac{4\sqrt3}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 被开方数是 $m$ 的二次函数，在 $m=0$ 处取得最大值（在 $m$ 范围内）。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最大弦长为$\frac{4\sqrt3}{3}$，此时 $m=0$。}
\end{examplebox}
