\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{在椭圆中，长半轴 $a$ 的平方等于短半轴 $b$ 的平方与半焦距 $c$ 的平方之和：$a^{2}=b^{2}+c^{2}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（椭圆标准方程）：} 椭圆方程为 $\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$ 或 $\frac{y^{2}}{a^{2}}+\frac{x^{2}}{b^{2}}=1$，且 $a>b>0$。
		\item \textbf{条件 2（参数定义）：} $a$ 为长半轴长，$b$ 为短半轴长，$c$ 为半焦距。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分双曲线）：} 椭圆中 $a^{2}=b^{2}+c^{2}$，双曲线中 $c^{2}=a^{2}+b^{2}$，两者关系不同，需严格区分。
		\item \textbf{检查项（开方取正）：} 由 $c^{2}=a^{2}-b^{2}$ 求 $c$ 时，开方只取算术平方根 $c=\sqrt{a^{2}-b^{2}}$，因为 $c>0$。
	\end{itemize}
\end{summarybox}
