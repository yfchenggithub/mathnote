	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（标准方程形式）：}}] 椭圆方程为 $\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$ 或 $\frac{y^{2}}{a^{2}}+\frac{x^{2}}{b^{2}}=1$，且 $a>b>0$。
		\item[\textbf{\textcolor{CStatement}{条件 2（参数定义）：}}] 设 $a$ 为长半轴长，$b$ 为短半轴长，$c$ 为半焦距。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（主关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在椭圆中，长半轴的平方等于短半轴平方与半焦距平方之和。
			\[
				a^{2}=b^{2}+c^{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（变形公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知任意两个参数可求第三个（注意开方取正值）。
			\[
				c=\sqrt{a^{2}-b^{2}},\quad b=\sqrt{a^{2}-c^{2}},\quad a=\sqrt{b^{2}+c^{2}}.
			\]
	\end{description}

\end{statementbox}
