\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\dfrac{x^{2}}{25}+\dfrac{y^{2}}{9}=1$，求 $a,b,c$ 并验证 $a^{2}=b^{2}+c^{2}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 由标准方程得 $a^{2}=25$，$b^{2}=9$，故 $a=5$，$b=3$。
			\par\textbf{理由：} 在椭圆标准方程 $\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$ 中，分母分别为 $a^{2}$ 和 $b^{2}$。
		\item[\textbf{第二步（求半焦距）：}] 利用 $c^{2}=a^{2}-b^{2}=25-9=16$，得 $c=4$（取正值）。
			\par\textbf{理由：} 椭圆参数关系 $a^{2}=b^{2}+c^{2}$ 的变形。
		\item[\textbf{第三步（验证关系）：}] 计算 $a^{2}=25$，$b^{2}+c^{2}=9+16=25$，两者相等。
			\par\textbf{理由：} 直接代入数值检验 $a^{2}=b^{2}+c^{2}$ 是否成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=5,\;b=3,\;c=4$，且满足 $a^{2}=b^{2}+c^{2}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆的长轴长为 $10$，短轴长为 $6$，求焦距 $2c$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定长半轴和短半轴）：}] 长轴 $2a=10$ 得 $a=5$；短轴 $2b=6$ 得 $b=3$。
			\par\textbf{理由：} 长轴长为 $2a$，短轴长为 $2b$。
		\item[\textbf{第二步（求半焦距）：}] 由 $c^{2}=a^{2}-b^{2}=25-9=16$，得 $c=4$。
			\par\textbf{理由：} 应用 $a^{2}=b^{2}+c^{2}$。
		\item[\textbf{第三步（求焦距）：}] 焦距 $2c=8$。
			\par\textbf{理由：} 焦距是 $2c$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{焦距为 $8$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1$ 的离心率 $e=\dfrac{1}{2}$，且短半轴长 $b=2\sqrt{3}$，求 $a$ 和 $c$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列出方程）：}] 由 $e=\dfrac{c}{a}=\dfrac{1}{2}$ 得 $c=\dfrac{a}{2}$。又由 $a^{2}=b^{2}+c^{2}$ 和 $b=2\sqrt{3}$ 得 $a^{2}=12+\dfrac{a^{2}}{4}$。
			\par\textbf{理由：} 离心率定义 $e=\frac{c}{a}$，参数关系 $a^{2}=b^{2}+c^{2}$。
		\item[\textbf{第二步（解方程）：}] 整理 $a^{2}-\dfrac{a^{2}}{4}=12$，即 $\dfrac{3}{4}a^{2}=12$，解得 $a^{2}=16$，故 $a=4$（取正值）。
			\par\textbf{理由：} 代数运算，注意 $a>0$。
		\item[\textbf{第三步（求半焦距）：}] 由 $c=\dfrac{a}{2}=2$。
			\par\textbf{理由：} 回代 $c=\frac{a}{2}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=4$，$c=2$，且满足 $a^{2}=b^{2}+c^{2}$（$b^{2}=12$）。}
\end{examplebox}
