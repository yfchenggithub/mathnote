\begin{summarybox}
\textbf{C094\quad 共焦椭圆与双曲线的四个公共点及坐标：快记／快懂／快用}

\textbf{快记（两曲线、一公式、四个点）：}
\[
\begin{gathered}
E:\ \frac{x^2}{a_E^2}+\frac{y^2}{a_E^2-c^2}=1,\qquad
H:\ \frac{x^2}{a_H^2}-\frac{y^2}{c^2-a_H^2}=1,\\[-1pt]
a_E>c>a_H>0,\qquad
\boxed{P=\left(\pm\frac{a_Ea_H}{c},\;
\pm\frac{\sqrt{(a_E^2-c^2)(c^2-a_H^2)}}{c}\right)}.
\end{gathered}
\]
两个符号独立选取，所以\textbf{必有且仅有四个不同的实交点}。

\textbf{快懂（从焦点定义走向坐标）：}
\[
PF_1+PF_2=2a_E,\quad |PF_1-PF_2|=2a_H
\quad\Longrightarrow\quad PF_1,PF_2=a_E\pm a_H.
\]
对右上交点，两个勾股式\emph{相减}：
\[
4cX=PF_1^2-PF_2^2=4a_Ea_H
\quad\Longrightarrow\quad X=\frac{a_Ea_H}{c}.
\]
两个勾股式\emph{相加}：
\[
X^2+Y^2=a_E^2+a_H^2-c^2
\quad\Longrightarrow\quad
Y=\frac{\sqrt{(a_E^2-c^2)(c^2-a_H^2)}}c.
\]
\textbf{核心认知：}相减消去高度 $Y^2$，找横坐标；
相加消去左右方向项，找离原点的距离，再求高度。

\textbf{快用（考场四步）：}
\begin{enumerate}
\item 先核查\emph{相同两焦点}（中心、焦点轴都需对齐）。
\item 算 $c,a_E,a_H$，核查 $a_E>c>a_H>0$。
\item 写 $X=a_Ea_H/c$，$Y=b_Eb_H/c$，分别取两处正负号。
\item 回代一个点；其余三点由坐标轴对称性确定。
\end{enumerate}

\textbf{两个有价值的延伸：}
\[
\boxed{X=\frac{c}{e_Ee_H}},\qquad
\boxed{k_Ek_H=-1\quad\text{（四点处切线正交）}}.
\]
四点的焦点张角相同（联系 C090），但不等于切线夹角 $90^\circ$。

\textbf{一句话：}\emph{「共焦定和差；平方差求横坐标；
平方和求高度；独立正负出四点。」}
\end{summarybox}
