\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{焦点三角形面积等于短半轴平方乘以顶角半角正切：$S = b^{2}\tan\frac{\theta}{2}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（椭圆标准型）：} 椭圆方程$\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$（$a>b>0$），焦点在$x$轴（其他方向可类比）。
		\item \textbf{条件2（P点位置）：} $P$为椭圆上非长轴端点的任意一点，$\theta=\angle F_{1}PF_{2}$在$(0,\pi)$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式混淆）：} 面积公式$S=b^{2}\tan\frac{\theta}{2}$，不是$a^{2}$或$c^{2}$。
		\item \textbf{检查项（退化情形）：} 确保$P$不在长轴端点，否则三角形退化但公式仍成立（面积为0），需注意。
	\end{itemize}
\end{summarybox}
