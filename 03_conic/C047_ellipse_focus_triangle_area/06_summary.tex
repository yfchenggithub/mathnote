\begin{summarybox}
\textbf{C047\quad 考场速记：一幅三角形、两种面积、三个检查点。}

\[
 \boxed{S=\tfrac12\cdot2c\cdot|y|=c|y|}
 \qquad\Longleftrightarrow\qquad
 \boxed{S=b^2\tan\frac\theta2}.
\]
\[
 \boxed{S_{\max}=cb},\qquad
 \boxed{0<\theta\le2\arctan\frac cb},\qquad
 \boxed{\tan\frac\theta2\le\frac cb}.
\]

\textbf{如何选方法（从快到慢）：}

\textbf{方法一·最快证明：}设 $m=PF_1,n=PF_2$，抓住
\[
 m+n=2a\ \Longrightarrow\ mn(1+\cos\theta)=2b^2
 \ \Longrightarrow\ S=\tfrac12mn\sin\theta=b^2\tan(\theta/2).
\]
\textbf{方法二·最低门槛：}画内切圆；从同一点作切线段相等，
顶点切线段 $PD=a-c$；$r=(a-c)\tan(\theta/2)$；
面积 $S=r(a+c)=(a-c)(a+c)\tan(\theta/2)=b^2\tan(\theta/2)$。

\textbf{方法三·常规坐标法：}设 $P(x,y)$，先算 $S=c|y|$，
再用焦半径、向量内积与行列式证明 $\tan(\theta/2)=c|y|/b^2$。

\medskip
\textbf{做题时最快的选择：}
已知 $\theta$ 用半角；已知 $y$ 用 $c|y|$；问最大值用 $cb$；
给出面积或角度却怀疑无点可取，先验 $S\le cb$。

\textbf{记忆核心：}椭圆定义给 $2a$，焦距给 $2c$，
$(a-c)(a+c)=b^2$；\textbf{角看一半，高看 $|y|$，还要验存在性。}
\end{summarybox}
