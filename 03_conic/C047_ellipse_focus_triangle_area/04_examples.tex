\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆$\frac{x^{2}}{25}+\frac{y^{2}}{16}=1$，$P$为椭圆上一点，$\angle F_{1}PF_{2}=60^{\circ}$，求$\triangle PF_{1}F_{2}$的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 椭圆$a^{2}=25$，$b^{2}=16$，$\theta=60^{\circ}$。
			\par\textbf{理由：}标准方程中分母$a^{2}=25$，$b^{2}=16$，顶角已知。
		\item[\textbf{第二步（套用公式）：}] 由公式得
			\[
				\begin{aligned}
					S &= b^{2}\tan\frac{\theta}{2}\\
					&= 16\times\tan30^{\circ}\\
					&= 16\times\frac{1}{\sqrt3}\\
					&= \frac{16\sqrt3}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}直接应用焦点三角形面积公式，代入$b^{2}=16$，$\theta=60^{\circ}$。
		\item[\textbf{第三步（得出结果）：}] 所以$S = \frac{16\sqrt3}{3}$。
			\par\textbf{理由：}化简完成。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\frac{16\sqrt3}{3}$}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆$\frac{x^{2}}{10}+\frac{y^{2}}{5}=1$，$\triangle PF_{1}F_{2}$的面积为$5$，求$\angle F_{1}PF_{2}$的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取b方）：}] 由椭圆方程得$b^{2}=5$。
			\par\textbf{理由：}短半轴平方为分母小项。
		\item[\textbf{第二步（反解正切）：}] 由$S=b^{2}\tan\frac{\theta}{2}$得
			\[
				\begin{aligned}
					\tan\frac{\theta}{2} &= \frac{S}{b^{2}}\\
					&= \frac{5}{5}\\
					&= 1.
			\end{aligned}
			\]
			\par\textbf{理由：}面积公式变形，代入$S=5$，$b^{2}=5$。
		\item[\textbf{第三步（求角度）：}] $\frac{\theta}{2}=45^{\circ}$，故$\theta=90^{\circ}$。
			\par\textbf{理由：}$\frac{\theta}{2}\in(0,\frac{\pi}{2})$，$\tan\frac{\theta}{2}=1$对应$45^{\circ}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\theta=90^{\circ}$}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆中心在原点，焦点在$x$轴上，焦距为$4$，过椭圆上一点$P$，$\angle F_{1}PF_{2}=60^{\circ}$，焦点三角形面积为$4\sqrt3$，求椭圆的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设定方程）：}] 设标准方程$\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$（$a>b>0$），由焦距$2c=4$得$c=2$。
			\par\textbf{理由：}椭圆中心在原点，焦点在x轴，标准形式。
		\item[\textbf{第二步（代入面积）：}] 由 $S = b^{2}\tan\frac{\theta}{2}$，$\theta = 60^{\circ}$，得
			\[
				S = b^{2}\tan30^{\circ} = \frac{b^{2}}{\sqrt3}.
			\]
			已知 $\triangle PF_{1}F_{2}$ 的面积为 $4\sqrt3$，故
			\[
				\frac{b^{2}}{\sqrt3} = 4\sqrt3,
			\]
			解得 $b^{2} = 12$。
			\par\textbf{理由：}面积公式代入 $\theta = 60^{\circ}$ 和 $S = 4\sqrt3$，解方程。
		\item[\textbf{第三步（求a方）：}] 由$a^{2}=b^{2}+c^{2}=12+4=16$，得椭圆方程$\frac{x^{2}}{16}+\frac{y^{2}}{12}=1$。
			\par\textbf{理由：}椭圆参数关系$a^{2}=b^{2}+c^{2}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\frac{x^{2}}{16}+\frac{y^{2}}{12}=1$}
\end{examplebox}
