\begin{trapbox}
\begin{minipage}{\linewidth}
\textbf{易错一：把圆周当成圆盘，误把包含情形的距离算成零。}
下图固定 $E:x^2/4+y^2=1$、$\Gamma:x^2+y^2=9$，两图比例相同。
左边的目标只有橙色边界；右边的目标还包括浅橙色内部。
\input{assets/tikz/C001_circle_vs_disk.tex}
\end{minipage}
\textbf{错因：}目标集合不同。到圆周的最近距离是 $3-2=1$；
到圆盘则可取 $P=Q$，最近距离为0。一般地，
\[
 D_{\min}^{\text{圆周}}=\min_{t\in[d_{\min},d_{\max}]}|t-r|,
 \qquad D_{\min}^{\text{圆盘}}=\max\{d_{\min}-r,0\}.
\]
\end{trapbox}

\begin{trapbox}
\begin{minipage}{\linewidth}
\textbf{易错二：直接写 $D_{\min}=|d_{\min}-r|$。}
下图使用例2的两条曲线：$E:x^2/4+y^2=1$，$\Gamma:(x-1)^2+y^2=4$。
\input{assets/tikz/C001_intersection_distance.tex}
\end{minipage}
\textbf{错因：}只看距离范围的下端点，漏掉了中间可以取到的距离。
图中 $A$ 是公共点，取 $P=Q=A$ 即得 $D_{\min}=0$；
错误公式却给出 $|\sqrt{2/3}-2|=2-\sqrt{2/3}>0$。
\textbf{正确做法：}先求\textbf{整个区间}，当 $r$ 落在其中时，最小距离为零。
\end{trapbox}

\begin{trapbox}
\begin{minipage}{\linewidth}
\textbf{易错三：把最远距离也写成差。}
下图使用例1的两条曲线。对离圆心最远的椭圆点 $P_2$，
应选圆心另一侧的 $Q_2$，而不是与 $P_2$ 同侧的 $Q_1$。
\input{assets/tikz/C001_external_distance.tex}
\end{minipage}
\textbf{错因：}把“离圆心最远”误当成“再减去半径”。
$P_2Q_1=5-2=3$ 只是这一配对的距离；图中还有更远的配对
$P_2Q_2=5+2=7$。
\textbf{正确做法：}选取 $P_2,M,Q_2$ 依次共线，得到 $D_{\max}=d_{\max}+r$。
\end{trapbox}

\begin{trapbox}
\begin{minipage}{\linewidth}
\textbf{易错四：把 $r<d_{\min}$ 一律当成外离。}
下面固定同一椭圆 $E:x^2/4+y^2=1$ 和同一半径 $r=1/2$，
只把圆心从 $M(0,0)$ 移到 $M(4,0)$；两幅图比例相同。
\input{assets/tikz/C001_trap_inside_vs_outside.tex}
\end{minipage}
\textbf{错因：}把“无公共点”误当成“圆一定在椭圆外”。
左图小圆完全在椭圆内，右图两曲线外离，但两图都满足 $r<d_{\min}$。
\textbf{正确做法：}先由不等式判断两条边界无公共点，再看圆心及圆周的具体位置。
\end{trapbox}

\begin{trapbox}
\begin{minipage}{\linewidth}
\textbf{易错五：只检查 $F'(\theta)=0$ 就求圆周距离。}
下图使用例2的 $F(\theta)=3\cos^2\theta-4\cos\theta+2$ 和 $r=2$。
\input{assets/tikz/C001_stationary_vs_intersection.tex}
\end{minipage}
\textbf{错因：}圆心距离平方的极值点，不等同于两条曲线之间最近距离的全部候选点。
红点处 $F=r^2=4$，距离 $|\sqrt F-r|=0$，但曲线在这里仍在上升或下降，
并非 $F$ 的驻点。
\textbf{正确做法：}先求 $F$ 的完整值域，再对 $|\sqrt F-r|$ 求最小值；
半径落在距离范围内时不能漏掉零距离。
\end{trapbox}

\begin{trapbox}
\begin{minipage}{\linewidth}
\textbf{边界补充：圆心在椭圆上，也不能把固定点到圆周的距离当成零。}
取 $E:x^2/4+y^2=1$、$M(2,0)$，圆周半径为1。
\input{assets/tikz/C001_trap_center_on_ellipse.tex}
\end{minipage}
$M\in E$ 只说明 $d_{\min}=0$。若固定 $P=M$，圆周上任意点 $Q$ 都满足
$PQ=r$，故 $|PM-r|=PM+r=r$，不是零。
\textbf{注意：}这说的是固定 $P=M$ 后的距离；整条椭圆与圆周的距离仍须按完整公式求解。
\end{trapbox}
