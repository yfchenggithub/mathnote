\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1(标准方程)：}}]
			双曲线方程为 $\frac{x^2}{a^2} - \frac{y^2}{b^2}=1$ 或 $\frac{y^2}{a^2} - \frac{x^2}{b^2}=1$，其中 $a>0,b>0$，焦点 $F_1,F_2$ 在坐标轴上。
		\item[\textbf{\textcolor{CStatement}{条件2(点P位置)：}}]
			点 $P$ 在双曲线上，且不在实轴端点（即 $P$ 不是顶点）。
		\item[\textbf{\textcolor{CStatement}{条件3(夹角定义)：}}]
			记 $\angle F_1PF_2 = \theta$，其中 $0<\theta<\pi$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一(面积公式)：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点三角形面积等于虚半轴平方乘以半角余切。
			\[
				S_{\triangle F_1PF_2} = b^2 \cot\frac{\theta}{2}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论(等价形式)：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积也可写为虚半轴平方除以半角正切，便于已知正切时使用。
			\[
				S_{\triangle F_1PF_2} = \frac{b^2}{\tan\frac{\theta}{2}}
			\]
	\end{description}

\end{statementbox}
