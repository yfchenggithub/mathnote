\begin{examplebox}
	\textbf{\textcolor{CExample}{例1(基础)}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $\frac{x^2}{4}-\frac{y^2}{5}=1$ 上一点 $P$（非顶点），$\angle F_1PF_2=60^\circ$，求 $\triangle F_1PF_2$ 的面积。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(识别参数)：}]
			由双曲线方程得 $a^2=4$，$b^2=5$，焦点在 $x$ 轴。
			\par\textbf{理由：} 标准方程形式，分母分别为 $a^2,b^2$。
		\item[\textbf{第二步(计算半角余切)：}]
			$\theta=60^\circ$，则 $\frac{\theta}{2}=30^\circ$，$\cot30^\circ=\sqrt{3}$。
			\par\textbf{理由：} 特殊角三角函数值。
		\item[\textbf{第三步(套用公式)：}]
			$S=b^2\cot\frac{\theta}{2}=5\times\sqrt{3}=5\sqrt{3}$。
			\par\textbf{理由：} 双曲线焦点三角形面积公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S=5\sqrt{3}$。}

	\medskip
	\textbf{\textcolor{CExample}{例2(稍变形)}}
	\textbf{\textcolor{CExample}{题目：}}
	双曲线 $\frac{x^2}{9}-\frac{y^2}{16}=1$ 的焦点三角形面积为 $16$，求 $\angle F_1PF_2$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(确定b²)：}]
			由双曲线方程得 $b^2=16$。
			\par\textbf{理由：} 标准方程中 $y^2$ 分母为 $b^2$。
		\item[\textbf{第二步(建立方程)：}]
			设所求角为 $\theta$，由 $S=b^2\cot\frac{\theta}{2}$，代入 $S=16,b^2=16$，得 $\cot\frac{\theta}{2}=1$。
			\par\textbf{理由：} 面积公式直接代入。
		\item[\textbf{第三步(解出角度)：}]
			因为 $\cot\frac{\theta}{2}=1$ 且 $0<\theta<\pi$，得 $\frac{\theta}{2}=45^\circ$，$\theta=90^\circ$。
			\par\textbf{理由：} 反余切函数，注意角度范围。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\angle F_1PF_2=90^\circ$。}

	\medskip
	\textbf{\textcolor{CExample}{例3(综合应用)}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$ 的焦点三角形面积为 $8\sqrt{3}$，$\angle F_1PF_2=60^\circ$，焦距为 $10$，求双曲线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(面积反求b²)：}]
			由 $\theta=60^\circ$ 得 $\frac{\theta}{2}=30^\circ$，$\cot30^\circ=\sqrt{3}$。根据 $S=b^2\cot\frac{\theta}{2}$，得 $b^2=\frac{S}{\cot\frac{\theta}{2}}=\frac{8\sqrt{3}}{\sqrt{3}}=8$。
			\par\textbf{理由：} 面积公式变形。
		\item[\textbf{第二步(求半焦距)：}]
			焦距 $2c=10$，所以 $c=5$。
			\par\textbf{理由：} 焦距定义。
		\item[\textbf{第三步(求参写方程)：}]
			在双曲线中 $c^2=a^2+b^2$，故 $a^2=c^2-b^2=25-8=17$。双曲线焦点在 $x$ 轴，方程为 $\frac{x^2}{17}-\frac{y^2}{8}=1$。
			\par\textbf{理由：} 双曲线参数关系及标准形式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\frac{x^2}{17}-\frac{y^2}{8}=1$。}
