\begin{statementbox}
\textbf{\textcolor{CStatement}{C018｜椭圆上一点的切线方程}}

\medskip
\textbf{\textcolor{CStatement}{条件}}
\begin{itemize}[leftmargin=*, itemsep=0.35em]
    \item 椭圆 $E:\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$，$a>b>0$。
    \item 已知切点 $P(x_0,y_0)$ \textbf{在椭圆上}：
    \[
    \frac{x_0^2}{a^2}+\frac{y_0^2}{b^2}=1.
    \]
\end{itemize}

\textbf{\textcolor{CStatement}{核心结论（所有切点都适用）}}
\[
\boxed{\frac{x_0x}{a^2}+\frac{y_0y}{b^2}=1.}
\]
\textbf{速记：} $x^2\mapsto x_0x$，$y^2\mapsto y_0y$，分母和右边的 $1$ 都不变。

\medskip
\textbf{\textcolor{CStatement}{等价形式及特殊位置}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{一般式（无需讨论斜率）：}]
    \[
      b^2x_0x+a^2y_0y=a^2b^2.
    \]
    \item[\textbf{斜率（仅 $y_0\ne0$）：}]
    \[
      k=-\frac{b^2x_0}{a^2y_0},\qquad
      y-y_0=k(x-x_0).
    \]
    \item[\textbf{左右顶点（$y_0=0$）：}]
    $P=(\pm a,0)$，切线为 $x=\pm a$（竖直，斜率不存在）。
    \item[\textbf{上下顶点（$x_0=0$）：}]
    $P=(0,\pm b)$，切线为 $y=\pm b$（水平，斜率为 $0$）。
    \item[\textbf{截距（仅 $x_0y_0\ne0$）：}]
    \[
      \frac{x}{a^2/x_0}+\frac{y}{b^2/y_0}=1,
      \quad S_{\triangle OAB}=\frac{a^2b^2}{2|x_0y_0|}.
    \]
\end{description}

\textbf{\textcolor{CStatement}{关联结论（过渡到直线切线判定）}}

对于非竖直直线 $y=kx+m$，它与椭圆相切的充要条件为
\[
\boxed{m^2=a^2k^2+b^2.}
\]
此时 $m\ne0$，切点为
\[
\boxed{P\left(-\frac{a^2k}{m},\frac{b^2}{m}\right).}
\]
注意：这个条件只用于非竖直直线；竖直切线另为 $x=\pm a$。
\end{statementbox}
