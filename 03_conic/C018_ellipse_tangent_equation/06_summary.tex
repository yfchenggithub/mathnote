\begin{summarybox}
\textbf{\textcolor{CSummary}{C018｜30 秒复盘}}

\textbf{条件：} $P(x_0,y_0)$ 在椭圆
$\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$ 上，$a>b>0$。

\textbf{一号公式（必记）：}
\[
\boxed{\frac{x_0x}{a^2}+\frac{y_0y}{b^2}=1.}
\]
\textbf{一句记忆：} 曲线的 $x^2,y^2$ 换成 $x_0x,y_0y$，其余不变。

\medskip
\textbf{\textcolor{CSummary}{最值得掌握的三种推导}}
\begin{itemize}[leftmargin=*, itemsep=0.4em]
 \item \textbf{最快：柯西不等式。}
 $\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2}\le1$（$P,Q$ 均在椭圆上），等号仅在 $Q=P$ 时成立，所以等号直线就是切线。
 \item \textbf{最直观：椭圆压成圆。}
 $u=x/a,\ v=y/b$；圆切线 $u_0u+v_0v=1$；再换回原坐标。
 \item \textbf{最通用的解析入口：法向量。}
 $\boldsymbol n=(x_0/a^2,y_0/b^2)$，过点 $P$ 写
 $\boldsymbol n\cdot\overrightarrow{PQ}=0$。
\end{itemize}

\medskip
\textbf{\textcolor{CSummary}{二号公式（已知斜率求切线）}}

非竖直直线 $y=kx+m$：
\[
 \boxed{m^2=a^2k^2+b^2},\qquad
 \boxed{P=\left(-\frac{a^2k}{m},\frac{b^2}{m}\right)}.
\]
一般式 $Ax+By=C$（$A,B$ 不同时为零）：
\[
 \boxed{C^2=a^2A^2+b^2B^2}.
\]
竖直切线单独记 $x=\pm a$；水平切线 $y=\pm b$。

\medskip
\textbf{\textcolor{CSummary}{考场选法}}

\textbf{给切点} $\to$ 直接替换；
\textbf{给斜率／一般式} $\to$ 用相切判定；
\textbf{给外点} $\to$ 过外点设直线并用判别式；
\textbf{问几何本质} $\to$ 想圆的切线与坐标伸缩。

\textbf{最后核查：} 切点必须在椭圆上；斜率公式仅 $y_0\ne0$；
椭圆法线一般\textbf{不是} $OP$ 方向。
\end{summarybox}
