\begin{statementbox}
\textbf{C007｜椭圆斜率式切线判定}\par
设椭圆
\[
C:\ \frac{x^2}{a^2}+\frac{y^2}{b^2}=1\qquad(a>b>0),
\]
直线 $l:y=kx+m$，其中 $k,m\in\mathbb R$，即\textbf{直线斜率存在}。记
\[
S=a^2k^2+b^2>0.
\]
\begin{enumerate}
    \item \textbf{核心充要条件（必记）}：
    \[
    \boxed{l\text{ 与椭圆相切}\iff m^2=a^2k^2+b^2.}
    \]

    \item \textbf{完整位置关系}：
    \[
    \begin{array}{ccl}
    m^2<S&\iff&\text{相交（两个不同交点）},\\
    m^2=S&\iff&\text{相切（一个交点）},\\
    m^2>S&\iff&\text{相离（没有交点）}.
    \end{array}
    \]

    \item \textbf{斜率已知，直接写两条切线}：
    \[
    \boxed{y=kx\pm\sqrt{a^2k^2+b^2}.}
    \]
    两条切线互相平行，且关于原点中心对称。

    \item \textbf{相切时的切点坐标}：若已知切线 $y=kx+m$ 满足上述相切条件，则 $m\ne0$，唯一切点为
    \[
    \boxed{T\left(-\frac{a^2k}{m},\frac{b^2}{m}\right).}
    \]

    \item \textbf{通用升级版（包括竖直切线）}：对任意直线 $Ax+By+C=0$，其中 $(A,B)\ne(0,0)$，有
    \[
    \boxed{\text{相切}\iff C^2=a^2A^2+b^2B^2.}
    \]
    把等号换成 $<$ 或 $>$，分别判定相交或相离。特别地，斜率不存在的切线为 $x=\pm a$。
\end{enumerate}
\end{statementbox}
