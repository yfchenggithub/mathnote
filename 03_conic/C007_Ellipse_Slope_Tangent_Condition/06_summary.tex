\begin{summarybox}
\textbf{必背一行}\par
\[
\boxed{\frac{x^2}{a^2}+\frac{y^2}{b^2}=1,\quad
 y=kx+m\text{ 相切}\iff m^2=a^2k^2+b^2.}
\]
\textbf{最快记忆}：\textbf{“截距平方＝横半轴平方乘斜率平方＋纵半轴平方”。} 斜率 $k$ 与\textbf{横半轴} $a$ 配对；常数项是\textbf{纵半轴} $b^2$。

\medskip
\textbf{一眼判三种位置}\par
设 $S=a^2k^2+b^2$：$m^2<S$ 是\textbf{两交点}，$m^2=S$ 是\textbf{切点}，$m^2>S$ 是\textbf{无交点}。

\medskip
\textbf{三条秒用公式}\par
\begin{enumerate}
    \item \textbf{已知斜率求切线}：$\displaystyle y=kx\pm\sqrt{a^2k^2+b^2}$。
    \item \textbf{已知切线求切点}：$\displaystyle T\left(-\frac{a^2k}{m},\frac{b^2}{m}\right)$。
    \item \textbf{任意直线通式（包括竖直）}：$\displaystyle Ax+By+C=0\text{ 相切}\iff\boxed{C^2=a^2A^2+b^2B^2}$。
\end{enumerate}

\textbf{最快推导（10 秒复现）}\par
记住\textbf{椭圆可以变成单位圆}：$X=x/a,\ Y=y/b$，此时直线变成 $-akX+bY=m$；与单位圆相切意味着圆心到直线距离为 $1$，即
\[
\frac{|m|}{\sqrt{a^2k^2+b^2}}=1
\quad\Longrightarrow\quad\boxed{m^2=a^2k^2+b^2}.
\]
也可直接记：$m=y-kx=(-ak)\cos\theta+b\sin\theta$，相切对应取最大或最小值，其绝对值为 $\sqrt{a^2k^2+b^2}$。

\medskip
\textbf{考场决策}\par
看到“斜率、相切、过定点、平行切线”，优先用本结论；看到“竖直切线”，改用通式或 $x=\pm a$。\textbf{不要忘记}：双曲线的相似公式有额外条件，渐近线不是切线。
\end{summarybox}
