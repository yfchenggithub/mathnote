\textbf{\textcolor{CStatement}{条件}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
    \item[\textbf{\textcolor{CStatement}{条件1（椭圆与旋转轴）：}}] 椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$（$a>0, b>0$）绕 $y$ 轴旋转一周。
\end{description}

\textbf{\textcolor{CStatement}{结论}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{\textcolor{CStatement}{结论一（旋转椭球体积）：}}]
    \textbf{\textcolor{CStatement}{直观描述：}} 以 $a$ 为赤道半径、$b$ 为极半径的旋转椭球，体积与 $a^2b$ 成正比。
    \[
    V_y = \frac{4}{3}\pi a^2 b ,
    \]
    其中 $a$ 是椭圆在 $x$ 轴方向的半轴长，$b$ 是椭圆在 $y$ 轴方向的半轴长。

    \item[\textbf{\textcolor{CStatement}{推广结论（绕x轴体积）：}}]
    \textbf{\textcolor{CStatement}{直观描述：}} 绕 $x$ 轴旋转时，体积与 $ab^2$ 成正比，公式呈对称形式。
    \[
    V_x = \frac{4}{3}\pi a b^2
    \]
\end{description}

\end{statementbox}
