\textbf{\textcolor{CExample}{例1（基础）}}

\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{25}+\frac{y^2}{9}=1$ 绕 $y$ 轴旋转一周，求所得旋转体的体积。

\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{第一步（识别半轴）：}] 由椭圆方程得 $a^2=25$, $b^2=9$，所以 $a=5$, $b=3$。
    \[
    a=5,\quad b=3
    \]
    \par\textbf{理由：} 椭圆标准方程 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ 中分母对应半轴平方。

    \item[\textbf{第二步（套用体积公式）：}] 旋转体积公式 $V_y=\frac{4}{3}\pi a^2 b$，代入数值：
    \[
    \begin{aligned}
    V_y &= \frac{4}{3}\pi a^2 b \\
    &= \frac{4}{3}\pi \cdot 25 \cdot 3 \\
    &= \frac{4}{3}\pi \cdot 75 \\
    &= 100\pi
    \end{aligned}
    \]
    \par\textbf{理由：} 直接应用椭圆绕 $y$ 轴旋转的体积公式。

    \item[\textbf{第三步（得出结论）：}] 旋转体的体积为 $V_y = 100\pi$。
    \[
    V_y = 100\pi
    \]
    \par\textbf{理由：} 计算无误，结果为最简形式。
\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{旋转椭球体积为 $100\pi$。}

\textbf{\textcolor{CExample}{例2（稍变形）}}

\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{a^2}+\frac{y^2}{9}=1$ 绕 $y$ 轴旋转所得体积为 $48\pi$，求 $a$ 的值（$a>0$）。

\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{第一步（确定已知半轴）：}] 由椭圆方程得 $b^2=9$，所以 $b=3$；设 $a$ 为未知正数。
    \[
    b=3
    \]
    \par\textbf{理由：} 椭圆标准方程中 $y^2$ 项分母为 $b^2$。

    \item[\textbf{第二步（写出体积表达式）：}] 利用公式 $V_y=\frac{4}{3}\pi a^2 b$，代入 $b=3$ 得 $V_y = \frac{4}{3}\pi a^2 \cdot 3 = 4\pi a^2$。
    \[
    V_y = 4\pi a^2
    \]
    \par\textbf{理由：} $\frac{4}{3} \times 3 = 4$，简化系数。

    \item[\textbf{第三步（列方程并求解）：}] 由已知 $V_y=48\pi$ 得：
    \[
    4\pi a^2 = 48\pi
    \]
    解得：
    \[
    \begin{aligned}
    a^2 &= 12, \\
    a &= 2\sqrt{3}
    \end{aligned}
    \]
    \par\textbf{理由：} 代入体积值，解一元二次方程，取正根。
\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{$a = 2\sqrt{3}$。}

\textbf{\textcolor{CExample}{例3（综合应用）}}

\textbf{\textcolor{CExample}{题目：}} 椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$（$a>b>0$）绕 $y$ 轴旋转的体积是绕 $x$ 轴旋转体积的 $2$ 倍，求椭圆的离心率 $e$。

\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{第一步（写出两个体积公式）：}] 根据椭圆绕坐标轴旋转的体积公式：
    \[
    \begin{aligned}
    V_y &= \frac{4}{3}\pi a^2 b, \\
    V_x &= \frac{4}{3}\pi a b^2
    \end{aligned}
    \]
    \par\textbf{理由：} 绕 $y$ 轴时 $x$ 轴半轴 $a$ 起平方作用，绕 $x$ 轴时 $b$ 起平方作用。

    \item[\textbf{第二步（根据条件列方程）：}] 由 $V_y = 2V_x$ 得：
    \[
    \frac{4}{3}\pi a^2 b = 2 \cdot \frac{4}{3}\pi a b^2
    \]
    \par\textbf{理由：} 直接代入条件。

    \item[\textbf{第三步（化简得 $a$ 与 $b$ 关系）：}] 两边同时除以 $\frac{4}{3}\pi a b$（$a,b>0$）：
    \[
    a = 2b
    \]
    \par\textbf{理由：} 除法化简，$a\neq0,\ b\neq0$。

    \item[\textbf{第四步（求离心率）：}] 椭圆离心率 $e = \frac{c}{a}$，其中 $c = \sqrt{a^2 - b^2}$。先计算 $c$：
    \[
    \begin{aligned}
    c &= \sqrt{a^2 - b^2} \\
    &= \sqrt{(2b)^2 - b^2} \\
    &= \sqrt{4b^2 - b^2} \\
    &= \sqrt{3}\,b
    \end{aligned}
    \]
    再计算 $e$：
    \[
    \begin{aligned}
    e &= \frac{c}{a} \\
    &= \frac{\sqrt{3}\,b}{2b} \\
    &= \frac{\sqrt{3}}{2}
    \end{aligned}
    \]
    \par\textbf{理由：} 离心率定义，代入 $a=2b$ 计算。
\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{椭圆的离心率 $e = \frac{\sqrt{3}}{2}$。}
