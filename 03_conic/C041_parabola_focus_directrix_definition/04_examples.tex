\begin{examplebox}
\textbf{例 1：焦点 $(3,0)$，准线 $x=-3$，求方程。}
先定顶点 $(0,0)$，再算 $p=6$，故 $\boxed{y^2=12x}$。
回验：$P=(3,6)$ 到焦点距离为 $6$，到准线的距离也为 $6$。

\medskip
\textbf{例 2：$y^2=8x$ 上 $PF=6$，求准距。}
定义直接给出准距为 $\boxed6$，不用先求 $p$。
若需回验，$p=4$，$F=(2,0)$、$\ell:x=-2$，
$x+2=6$ 得 $x=4$，再得 $y=\pm4\sqrt2$，实际有两点满足条件。
\end{examplebox}

\begin{examplebox}
\textbf{补充辨析：给定的焦半径一定能取吗？}
仍取 $y^2=8x$。$PF=x+2\ge2$，所以焦半径 $1$ 不可取，$2$ 只在顶点取得，$6$ 对应上下两个点。
\input{assets/tikz/C041_focal_radius_range}
一般地，给定焦半径 $L$，先用 $x=L-p/2$。
$L<p/2$ 无解；$L=p/2$ 得顶点；$L>p/2$ 时
$y=\pm\sqrt{2p(L-p/2)}$，恰有两点。

\medskip
\textbf{例 3：点 $A(1,2)$ 在 $y^2=2px$ 上，求 $p,AF$。}
代入得 $4=2p$，所以 $\boxed{p=2}$。
准线 $x=-1$，由定义 $AF=1+1=\boxed2$。
回验：$F=(1,0)$，$AF=\sqrt{(1-1)^2+2^2}=2$。
\end{examplebox}

\begin{examplebox}
\textbf{例 4（方向判断）：$F=(3,0)$，$\ell:x=5$。}
\textbf{方法 A｜直观最低门槛：}焦点在准线左侧，中点顶点 $V=(4,0)$，
焦准距 $p=|3-5|=2$，应向左开口：
\[\boxed{y^2=-4(x-4)}.\]
\textbf{方法 B｜最快列式：}
\[
(x-3)^2+y^2=(x-5)^2
\iff y^2=-4x+16=-4(x-4).
\]
\textbf{回验：} $P=(3,2)$ 在曲线上，$PF=2$，到 $x=5$ 的垂距也是 $2$。
\textbf{触发点：}焦点 $x$ 坐标为正，并不意味着向右开口；要看准线在焦点哪侧。
\end{examplebox}

\begin{examplebox}
\textbf{例 5（斜准线）：焦点 $F=(1,1)$，准线 $\ell:x+y=0$。}
\input{assets/tikz/C041_oblique_directrix_geometry}
\textbf{最容易操作的方法：}设 $P=(x,y)$，点到直线的垂距为 $|x+y|/\sqrt2$，因此
\[
\begin{aligned}
\sqrt{(x-1)^2+(y-1)^2}&=\frac{|x+y|}{\sqrt2},\\
2\bigl[(x-1)^2+(y-1)^2\bigr]&=(x+y)^2,\\
&\boxed{(x-y)^2=4(x+y-1)}.
\end{aligned}
\]
\textbf{几何回看：}如图，先作垂足 $D$、再取中点 $V$；对称轴与准线垂直。
\textbf{数值回验：} $P=(2,0)$ 满足方程，$PF=\sqrt2$，
$d(P,\ell)=2/\sqrt2=\sqrt2$。
\textbf{要点：}准线倾斜时，切勿直接套未旋转的四种标准式。
\end{examplebox}

\begin{examplebox}
\textbf{例 6（考场速算焦半径）：$P=(x,y)\in y^2=8x$，已知 $PF=7$，求 $P$。}
方程给 $2p=8$，即 $p=4$；因此准线 $x=-2$。
\textbf{最快：}用 $PF=x+2$，一眼得 $x=5$，再代回抛物线：
\[
y^2=8\times5=40\quad\Longrightarrow\quad
\boxed{P=(5,\pm2\sqrt{10})}.
\]
\textbf{慢但可行：}设 $F=(2,0)$，列 $\sqrt{(x-2)^2+y^2}=7$，
再与 $y^2=8x$ 联立，结果相同，但多做了一次平方展开。
\textbf{见到什么就想到它：}“抛物线 + 到焦点的距离”优先联想“准线垂距”。
\end{examplebox}

\begin{examplebox}
\textbf{例 7（换皮题）：圆经过 $F=(1,0)$，且与直线 $x=-1$ 相切，圆半径为 $3$；求圆心。}
设圆心为 $P$。圆经过 $F$ 表示 $PF=3$；圆与准线相切表示
$d(P,\ell)=3$。因此圆心 $P$ 在由 $F,\ell$ 确定的抛物线上。
其焦准距 $p=2$，轨迹是 $y^2=4x$；再由焦半径公式
\[
PF=x+1=3\Longrightarrow x=2,
\qquad y^2=8\Longrightarrow
\boxed{P=(2,\pm2\sqrt2)}.
\]
\textbf{完整回验：}两个圆心到 $F$ 的距离均为
$\sqrt{(2-1)^2+(2\sqrt2)^2}=3$；到 $x=-1$ 的距离也都是 $3$。
\textbf{真正考点：}题目不一定出现“抛物线”，只要是\textbf{经过定点、切定直线的圆心轨迹}就该想到焦准等距。
\end{examplebox}

\begin{examplebox}
\textbf{例 8（无需坐标的最值）：已知焦点到准线距离 $p=10$，动点 $P$ 在对应抛物线上，求 $PF+PH$ 的最小值。}
由定义 $PF=PH$，所以 $PF+PH=2PF$。
设 $D$ 是焦点到准线的垂足。以 $H$ 为 $P$ 到准线的垂足，
\[
10=FD\le FH\le FP+PH=2FP=PF+PH.
\]
因此最小值为 $\boxed{10}$；当 $P$ 是 $FD$ 中点（顶点）时取得。
\textbf{最低门槛：}只用了“垂线段最短”和“三角形两边之和不小于第三边”，根本不用抛物线方程。

\medskip
\textbf{例 9（方向互换）：$F=(0,2)$、$\ell:y=-2$，求轨迹方程。}
顶点 $(0,0)$，$p=4$，开口向上，故 $\boxed{x^2=8y}$。
直接列 $(x-0)^2+(y-2)^2=(y+2)^2$，也立得 $x^2=8y$。
\end{examplebox}
