\begin{examplebox}
\textbf{例 1：焦点 $(3,0)$，准线 $x=-3$，求方程。}
先定顶点 $(0,0)$，再算 $p=6$，故 $\boxed{y^2=12x}$。
回验：$P=(3,6)$ 到焦点距离为 $6$，到准线的距离也为 $6$。

\medskip
\textbf{例 2：$y^2=8x$ 上 $PF=6$，求准距。}
定义直接给出准距为 $\boxed6$，不用先求 $p$。
若需回验，$p=4$，$F=(2,0)$、$\ell:x=-2$，
$x+2=6$ 得 $x=4$，再得 $y=\pm4\sqrt2$，实际有两点满足条件。
\end{examplebox}

\begin{examplebox}
\textbf{补充辨析：给定的焦半径一定能取吗？}
仍取 $y^2=8x$。$PF=x+2\ge2$，所以焦半径 $1$ 不可取，$2$ 只在顶点取得，$6$ 对应上下两个点。
\input{assets/tikz/C041_focal_radius_range}
一般地，给定焦半径 $L$，先用 $x=L-p/2$。
$L<p/2$ 无解；$L=p/2$ 得顶点；$L>p/2$ 时
$y=\pm\sqrt{2p(L-p/2)}$，恰有两点。

\medskip
\textbf{例 3：点 $A(1,2)$ 在 $y^2=2px$ 上，求 $p,AF$。}
代入得 $4=2p$，所以 $\boxed{p=2}$。
准线 $x=-1$，由定义 $AF=1+1=\boxed2$。
回验：$F=(1,0)$，$AF=\sqrt{(1-1)^2+2^2}=2$。
\end{examplebox}
