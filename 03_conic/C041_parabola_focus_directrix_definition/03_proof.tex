\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过建立适当的坐标系，将几何条件$|PF| = d(P,l)$转化为坐标关系，经代数变形得到标准方程$y^2 = 2px$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（坐标系设定）：}] 以焦点$F$到准线$l$的垂线段中点为原点，$F$在$x$轴正半轴上。
			\[
				F\bigl(\frac{p}{2},0\bigr),\quad l:x = -\frac{p}{2},\quad (p>0).
			\]
			\par\textbf{理由：}
			选择对称坐标系可简化代数推导。

		\item[\textbf{步骤二（距离表达式）：}] 设动点$P(x,y)$，写出距离。
			\[
				|PF| = \sqrt{\bigl(x-\frac{p}{2}\bigr)^2 + y^2},\quad
				d(P,l) = \bigl|x + \frac{p}{2}\bigr|.
			\]
			\par\textbf{理由：}
			平面上两点距离公式及点到直线距离公式。

		\item[\textbf{步骤三（代入等式）：}] 根据定义$|PF| = d(P,l)$，两边平方去根号。
			\[
				\bigl(x-\frac{p}{2}\bigr)^2 + y^2 = \bigl(x+\frac{p}{2}\bigr)^2.
			\]
			\par\textbf{理由：}
			距离等式，两边平方去掉绝对值。

		\item[\textbf{步骤四（展开化简）：}] 展开并合并。
			\[
				\begin{aligned}
					x^2 - p x + \frac{p^2}{4} + y^2 &= x^2 + p x + \frac{p^2}{4} \\ -p x + y^2 &= p x \\ y^2 &= 2p x.
			\end{aligned}
			\]
			\par\textbf{理由：}
			移项合并同类项。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{所得方程$y^2 = 2px$即为焦点在$x$轴正半轴时的抛物线标准方程，满足原定义条件。}
\end{proofbox}
