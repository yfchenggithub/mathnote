\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略条件）：}}] 忽略焦点$F$不在准线$l$上的条件，认为任意情况下定义都成立。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件约束）：}}] 若$F$在$l$上，则到$F$和$l$距离相等的点的轨迹是过$F$且垂直于$l$的直线，不是抛物线。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件漏看）：}}] 未注意定义中$F\notin l$的限制，导致轨迹判断错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（参数混淆）：}}] 认为标准方程$y^2=2px$中焦点是$(p,0)$，或准线是$x=-p$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确形式）：}}] 焦点应为$(\frac{p}{2},0)$，准线是$x=-\frac{p}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}] 混淆$p$与焦点横坐标的关系，误将$p$当作焦点到原点的距离。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（比例乱用）：}}] 在椭圆或双曲线问题中错误地使用到焦点和准线距离相等的条件。
		\item[\textbf{\textcolor{CTrap}{正确理解（区别定义）：}}] 抛物线第二定义中离心率$e=1$，仅适用于抛物线；椭圆、双曲线中$e\neq1$，距离比为$e$。
		\item[\textbf{\textcolor{CTrap}{错因分析（统一定义混淆）：}}] 未区分圆锥曲线统一定义的离心率，误认为所有圆锥曲线中距离比都为$1$。
	\end{description}
\end{trapbox}
