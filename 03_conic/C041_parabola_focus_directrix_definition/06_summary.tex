\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{抛物线是到焦点距离等于到准线距离的点的轨迹。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（定点定线）：} 焦点$F$不在准线$l$上。
		\item \textbf{条件 2（等距关系）：} 动点$P$满足$|PF| = d(P,l)$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（验证条件）：} 使用定义前务必检查$F\notin l$。
		\item \textbf{检查项（参数对应）：} 在标准方程$y^2=2px$中，焦点是$(\frac{p}{2},0)$，准线$x=-\frac{p}{2}$。
	\end{itemize}
\end{summarybox}
