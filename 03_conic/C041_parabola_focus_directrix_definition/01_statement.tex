\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（定点定线）：}}] 定点$F$（焦点）不在定直线$l$（准线）上。
		\item[\textbf{\textcolor{CStatement}{条件二（距离相等）：}}] 动点$P$到$F$的距离等于$P$到$l$的距离。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（定义关联）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			到焦点距离等于到准线距离的轨迹是抛物线。
			\[
				\frac{|PF|}{d(P,l)} = 1.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（标准方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点在$x$轴正半轴时，标准方程为$y^2 = 2px$（$p>0$）。
			\[
				y^2 = 2px,\quad p>0.
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{推广结论（开口方向）：}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	焦点在$x$轴负半轴时，标准方程为$y^2 = -2px$。
	\[
		y^2 = -2px,\quad p>0.
	\]
\end{statementbox}
