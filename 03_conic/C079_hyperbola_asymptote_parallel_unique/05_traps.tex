\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（漏截距条件）：}}]
			只注意到直线与渐近线平行，就认为与双曲线只有一个交点，漏掉 $c\neq0$ 的限制。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须排除重合）：}}]
			只有当直线不与渐近线重合（$c\neq0$）时，才保证一个交点；若 $c=0$，则无交点。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			将“平行”等同于“唯一交点”，未考虑重合（渐近线本身）的例外情况。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（焦点轴混淆）：}}]
			当双曲线焦点在 $y$ 轴时，仍然使用 $y = \pm\frac{b}{a}x + c$ 的形式，导致错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（对应渐近线形式）：}}]
			对于 $\frac{y^2}{a^2} - \frac{x^2}{b^2}=1$，渐近线为 $y = \pm\frac{a}{b}x$，平行直线应为 $y = \pm\frac{a}{b}x + c$。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式遗忘）：}}]
			死记 $y=\pm\frac{b}{a}x$，没有根据双曲线标准形式调整 $a,b$ 的位置。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（类比失误）：}}]
			受椭圆和抛物线影响，认为平行于渐近线的直线可能与双曲线有两个交点或相切。
		\item[\textbf{\textcolor{CTrap}{正确理解（代数特征决定）：}}]
			联立后二次项系数为零，方程退化为一次，因此必定只有一个交点，且不是相切（斜率不等于渐近线斜率时才可能相切）。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
			未从代数角度分析，仅凭几何直觉臆断；忽略了双曲线特有的渐近线性质。
	\end{description}

\end{trapbox}
