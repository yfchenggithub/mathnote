\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	求证：直线 $y = \frac{3}{2}x + 1$ 与双曲线 $\frac{x^2}{4} - \frac{y^2}{9} = 1$ 有且仅有一个交点。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（联立方程）：}]
			将直线代入双曲线，消去 $y$。
			\[
				\frac{x^2}{4} - \frac{(\frac{3}{2}x + 1)^2}{9} = 1.
			\]
			\par\textbf{理由：} 交点坐标满足两个方程。
		\item[\textbf{第二步（化简消元）：}]
			展开并合并，二次项抵消。
			\[
				\frac{x^2}{4} - \frac{\frac{9}{4}x^2 + 3x + 1}{9} = 1 \;\Rightarrow\; -\frac{x}{3} - \frac{1}{9} = 1.
			\]
			\par\textbf{理由：} 展开后 $\frac{x^2}{4}$ 项消失，方程变为一次。
		\item[\textbf{第三步（求解结论）：}]
			解出 $x$ 并回代，得唯一交点。
			\[
				x = -\frac{10}{3},
				\qquad
				y = -4.
			\]
			\par\textbf{理由：} 一次方程有唯一解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	交点为 $\left(-\frac{10}{3},-4\right)$，有且仅有一个交点。

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	判断直线 $y = -2x$ 与双曲线 $\frac{y^2}{16} - \frac{x^2}{4} = 1$ 的交点个数。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（渐近线方程）：}]
			由双曲线方程写出渐近线。
			\[
				\frac{y^2}{16} - \frac{x^2}{4} = 0 \;\Rightarrow\; y = \pm 2x.
			\]
			\par\textbf{理由：} 焦点在 $y$ 轴时渐近线为 $y = \pm\frac{a}{b}x$。
		\item[\textbf{第二步（比较直线）：}]
			直线 $y = -2x$ 与渐近线 $y = -2x$ 比较。
			\[
				y = -2x.
			\]
			\par\textbf{理由：} 斜率相同且截距为 $0$，故直线与渐近线重合。
		\item[\textbf{第三步（代入验证）：}]
			将直线代入双曲线检验。
			\[
				\begin{aligned}
					\frac{(-2x)^2}{16} - \frac{x^2}{4} &= \frac{4x^2}{16} - \frac{x^2}{4} \\ &= \frac{x^2}{4} - \frac{x^2}{4} \\ &= 0 \neq 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入后左边为 $0$，右边为 $1$，等式不成立，故无交点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	直线与双曲线的交点个数为 $0$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $\frac{x^2}{4} - \frac{y^2}{9} = 1$，直线 $l$ 过点 $P(1,2)$ 且与一条渐近线平行，且与双曲线只有一个公共点，求 $l$ 的方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（渐近线斜率）：}]
			写出渐近线方程。
			\[
				\frac{x^2}{4} - \frac{y^2}{9} = 0 \;\Rightarrow\; y = \pm\frac{3}{2}x.
			\]
			\par\textbf{理由：} 渐近线斜率 $k = \pm\frac{3}{2}$。
		\item[\textbf{第二步（设直线求截距）：}]
			设直线 $y = \frac{3}{2}x + b$ 或 $y = -\frac{3}{2}x + b$，代入点 $P$。
			\[
				\begin{cases}
					2 = \frac{3}{2} + b \;\Rightarrow\; b = \frac{1}{2},\\ 2 = -\frac{3}{2} + b \;\Rightarrow\; b = \frac{7}{2}.
			\end{cases}
			\]
			\par\textbf{理由：} 点在直线上，代入求解 $b$。
		\item[\textbf{第三步（应用结论）：}]
			两条直线均满足斜率等于渐近线斜率且 $b\neq0$，故交点唯一。
			\[
				y = \frac{3}{2}x + \frac{1}{2},
				\qquad
				y = -\frac{3}{2}x + \frac{7}{2}.
			\]
			\par\textbf{理由：} 根据二级结论，平行于渐近线且不重合的直线与双曲线只有一个交点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	所求直线方程为 $y = \frac{3}{2}x + \frac{1}{2}$ 或 $y = -\frac{3}{2}x + \frac{7}{2}$。
\end{examplebox}
