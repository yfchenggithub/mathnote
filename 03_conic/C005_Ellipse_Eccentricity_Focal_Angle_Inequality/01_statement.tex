\begin{statementbox}
设椭圆
\[
C:\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\qquad(a>b>0),
\]
左右焦点为 $F_1(-c,0)$、$F_2(c,0)$，其中
\[
c^2=a^2-b^2,\qquad e=\frac ca\quad(0<e<1).
\]
任取椭圆上的点 $P$，记焦点张角 $\angle F_1PF_2=\theta$，则：
\begin{enumerate}
    \item \textbf{核心不等式：}
    \[
    \boxed{\sin\frac{\theta}{2}\le e.}
    \]

    \item \textbf{取等条件：}当且仅当 $PF_1=PF_2$，即
    $P=(0,b)$ 或 $P=(0,-b)$（短轴两端点）时，取等号。

    \item \textbf{焦点张角的准确范围：}
    \[
    \boxed{0\le\theta\le\theta_{\max}=2\arcsin e<\pi.}
    \]
    在长轴端点 $P=(\pm a,0)$ 时，$\theta=0$；在短轴端点时，$\theta$ 最大。

    \item \textbf{存在性判定：}对给定的 $0\le\theta_0<\pi$，椭圆上存在一点 $P$ 使
    $\angle F_1PF_2\ge\theta_0$，当且仅当
    \[
    \boxed{e\ge\sin\frac{\theta_0}{2}.}
    \]
    特别地，存在焦点张角\emph{恰好等于} $\theta_0$ 的点，条件也相同。
\end{enumerate}
\end{statementbox}
