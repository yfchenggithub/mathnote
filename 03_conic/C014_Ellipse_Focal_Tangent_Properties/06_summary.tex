\begin{summarybox}
\textbf{C014 一页速记：记住“一根绳、一次反射、一个主圆、一次点幂”。}

\textbf{起点（椭圆定义）}：
\[
PF_1+PF_2=2a,\qquad c^2=a^2-b^2.
\]

\textbf{最值钱的几何证明链}：
\[
\begin{gathered}
F_1\xrightarrow{\text{关于切线反射}}Q,\quad
F_2,P,Q\text{ 共线},\quad F_2Q=2a;\\
O,H_1\text{ 是三角形两边中点}
\Longrightarrow OH_1=a;\\
K=-H_2,\quad H_1,F_1,K\text{ 共线}
\Longrightarrow d_1d_2=a^2-c^2=b^2.
\end{gathered}
\]

\textbf{五条结论（含适用条件）}：
\begin{enumerate}
\item \textbf{距离积}：$\boxed{d_1d_2=b^2}$，任意切点成立。
\item \textbf{共圆}：$\boxed{OH_1=OH_2=a}$，垂足位于主圆 $x^2+y^2=a^2$。
\item \textbf{光学}：非长轴顶点时，法线是焦点角内角平分线，切线是外角平分线。
\item \textbf{截距积}：若 $x_0y_0\ne0$，$\boxed{x_M=c^2x_0/a^2}$、$\boxed{x_N=a^2/x_0}$，故 $\boxed{x_Mx_N=c^2}$。
\item \textbf{面积最大值（分类）}：
\[
\boxed{S_{\max}=\begin{cases}2bc,&c\le b,\\a^2,&c\ge b.\end{cases}}
\]
\end{enumerate}

\textbf{面积的最快判断}：
\[
S=\frac12(d_1+d_2)H_1H_2,\qquad
(d_1+d_2)^2+H_1H_2^2=4a^2.
\]
因此 $S\le a^2$；只有 $c\ge b$ 才能取等，否则最大值为 $2bc$。

\textbf{圆与椭圆的对应记忆}：
圆的切线垂直半径 $\longleftrightarrow$ 椭圆法线平分焦点角；
圆的中位线与点幂 $\longleftrightarrow$ 椭圆的垂足共圆和距离积。

\textbf{考场选法}：
\begin{itemize}
\item 求 $OH_i$：直接主圆，\textbf{一步}。
\item 求 $d_1d_2$：直接 $b^2$；需要证明时优先\textbf{共圆+点幂}。
\item 求截距积：切线 $x_N=a^2/x_0$，法线 $x_M=c^2x_0/a^2$。
\item 求面积最值：\textbf{先比较 $c$ 与 $b$}，再写分段结论；绝不漏取等可行性。
\end{itemize}

\textbf{一句话本质}：\textbf{把椭圆的焦点反射问题，变成初中圆的中位线、点幂与直角三角形最值问题。}
\end{summarybox}
