\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知$F_1(-4,0),F_2(4,0)$为双曲线两焦点，动点$P$满足$\bigl| |PF_1|-|PF_2| \bigr|=6$，求$P$的轨迹方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由定义，$2a=6$，所以$a=3$；焦距$|F_1F_2|=8$，所以$2c=8$，$c=4$。
			\par\textbf{理由：}距离差绝对值常数等于$2a$；两焦点距离等于$2c$。
		\item[\textbf{第二步（计算b）：}]
			利用$b^2=c^2-a^2=16-9=7$，得$b=\sqrt{7}$。
			\par\textbf{理由：}标准方程中参数关系。
		\item[\textbf{第三步（写方程）：}]
			焦点在$x$轴，标准方程为$\displaystyle\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$，即$\frac{x^2}{9}-\frac{y^2}{7}=1$。
			\par\textbf{理由：}根据焦点位置确定方程形式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	轨迹方程为$\displaystyle\frac{x^2}{9}-\frac{y^2}{7}=1$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线焦点为$F_1(0,-3),F_2(0,3)$，且双曲线上的点到两焦点距离之差的绝对值为$4$，求双曲线标准方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由定义，$2a=4$，所以$a=2$；焦距$|F_1F_2|=6$，所以$2c=6$，$c=3$。
			\par\textbf{理由：}距离差绝对值常数等于$2a$；两焦点距离等于$2c$。
		\item[\textbf{第二步（计算b）：}]
			利用$b^2=c^2-a^2=9-4=5$，得$b=\sqrt{5}$。
			\par\textbf{理由：}标准方程中参数关系。
		\item[\textbf{第三步（写方程）：}]
			焦点在$y$轴，标准方程为$\displaystyle\frac{y^2}{a^2}-\frac{x^2}{b^2}=1$，即$\frac{y^2}{4}-\frac{x^2}{5}=1$。
			\par\textbf{理由：}根据焦点位置确定方程形式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	标准方程为$\displaystyle\frac{y^2}{4}-\frac{x^2}{5}=1$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	双曲线$\frac{x^2}{9}-\frac{y^2}{16}=1$的左右焦点分别为$F_1,F_2$，点$P$在双曲线上，且$|PF_1|=7$，求$|PF_2|$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由方程得$a^2=9$，$c^2=a^2+b^2=9+16=25$，所以$a=3$，$c=5$。
			\par\textbf{理由：}标准方程参数关系。
		\item[\textbf{第二步（应用定义）：}]
			由双曲线定义，$\bigl| |PF_1|-|PF_2| \bigr|=2a=6$。代入$|PF_1|=7$得$|7-|PF_2||=6$，解得$|PF_2|=1$或$13$。
			\par\textbf{理由：}定义直接建立方程。
		\item[\textbf{第三步（检验舍根）：}]
			检验三角形两边之和：$|PF_1|+|PF_2|>|F_1F_2|=10$。若$|PF_2|=1$，则$7+1=8<10$，矛盾；若$|PF_2|=13$，则$7+13=20>10$，符合。所以$|PF_2|=13$。
			\par\textbf{理由：}实际轨迹上点需满足构成三角形，否则不存在。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$|PF_2|=13$。
\end{examplebox}
