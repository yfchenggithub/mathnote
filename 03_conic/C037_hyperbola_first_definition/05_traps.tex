\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略绝对值）}}]
			学生直接写$|PF_1|-|PF_2|=2a$，忽略了绝对值，导致只表示双曲线一支。
		\item[\textbf{\textcolor{CTrap}{正确理解（定义需绝对值）：}}]
			定义是$\bigl| |PF_1|-|PF_2| \bigr|=2a$，包括正负两种情况，对应双曲线两支。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆模糊）：}}]
			对双曲线定义记忆不完整，与椭圆定义混淆（椭圆强调距离和）。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（a,c关系颠倒）}}]
			学生认为$a>c$或$a=c$，导致轨迹判断错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（0<a<c）：}}]
			双曲线要求$2a<2c$，即$a<c$；若$a=c$则轨迹为两条射线，若$a>c$则无轨迹。
		\item[\textbf{\textcolor{CTrap}{错因分析（与椭圆混淆）：}}]
			椭圆中$a>c$，故学生容易套用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（焦点位置与方程符号错位）}}]
			焦点在$x$轴时方程$\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$，在$y$轴时方程$\frac{y^2}{a^2}-\frac{x^2}{b^2}=1$，学生常写反。
		\item[\textbf{\textcolor{CTrap}{正确理解（焦点所在轴对应项为正）：}}]
			焦点所在轴（实轴）对应项系数为正。
		\item[\textbf{\textcolor{CTrap}{错因分析（形式记忆不牢）：}}]
			死记硬背，不结合图形理解。
	\end{description}

\end{trapbox}
