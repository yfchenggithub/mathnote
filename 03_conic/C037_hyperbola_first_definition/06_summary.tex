\begin{summarybox}
\textbf{一句话：} 距离差取绝对值，正常范围是 $0<2a<2c$，两支一起保留。
\[
|PF_1-PF_2|=2a,\qquad c^2=a^2+b^2.
\]
焦点 $(\pm c,0)$ 对应 $x^2/a^2-y^2/b^2=1$；纵向交换坐标。
右支 $r_1-r_2=2a$，左支 $r_1-r_2=-2a$。

\textbf{四项检查。}
\begin{enumerate}[leftmargin=*]
\item 距离差是否严格介于 $0$ 与焦距之间？
\item 是否保留绝对值或明确分支？
\item 正项方向是否对应焦点轴？是否使用 $c^2=a^2+b^2$？
\item 求出的距离能否同时实现，是否代回距离公式检查？
\end{enumerate}
\end{summarybox}

\begin{summarybox}
\textbf{考场速写：先看差，再看轴，最后看平方。}
\[
\boxed{0<2a<2c},\qquad
\boxed{b^2=c^2-a^2},\qquad
\boxed{\text{焦点方向}=\text{标准方程的正项方向}}.
\]
\textbf{焦点在横轴：} $(\pm c,0)$，顶点 $(\pm a,0)$，
$\displaystyle\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$；
\textbf{焦点在纵轴：} $(0,\pm c)$，顶点 $(0,\pm a)$，
$\displaystyle\frac{y^2}{a^2}-\frac{x^2}{b^2}=1$。

\par\medskip\noindent
\textbf{反推标准方程的四步（右支）：}
\[
\begin{gathered}
r_1-r_2=2a,\qquad r_1^2-r_2^2=4cx,\\
(r_1-r_2)(r_1+r_2)=4cx
\ \Longrightarrow\ r_1+r_2=\frac{2cx}{a},\\
r_2=\frac{cx}{a}-a
\ \xrightarrow[\text{距离平方}]{\text{代入}}
\ \frac{x^2}{a^2}-\frac{y^2}{b^2}=1.
\end{gathered}
\]
\textbf{两支统一的焦半径核验式}（只在该双曲线上）：
\[
PF_1=\left|\frac{cx}{a}+a\right|,\qquad
PF_2=\left|\frac{cx}{a}-a\right|.
\]
\textbf{牢记数值锚点：} $a=3,b=4,c=5$，两焦点 $\pm5$，
两顶点 $\pm3$，右顶点两焦半径 $8,2$，差 $6$。

\textbf{忘了 $a<c$ 时，回忆三角形不等式：}
椭圆“和大于焦距”，双曲线“差小于焦距”。
若题目给焦半径候选值，再检验
$|PF_1-PF_2|\le2c\le PF_1+PF_2$。
\end{summarybox}
