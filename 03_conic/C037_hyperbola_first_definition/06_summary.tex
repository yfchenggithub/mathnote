\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	双曲线第一定义：$\bigl| |PF_1|-|PF_2| \bigr| = 2a$（$0<a<c$）。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（两焦点已知）：} 需要明确两焦点坐标或距离$2c$。
		\item \textbf{条件2（距离差常数）：} 已知$\bigl| |PF_1|-|PF_2| \bigr|$为常数$2a$，且$2a<2c$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（绝对值符号）：} 定义中必须带绝对值，否则仅表示一支。
		\item \textbf{检查项（a,b,c关系）：} $c^2=a^2+b^2$，注意$a,c$大小关系（$a<c$）。
	\end{itemize}

\end{summarybox}
