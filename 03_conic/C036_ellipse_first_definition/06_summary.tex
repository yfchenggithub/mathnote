\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{椭圆定义：到两定点距离之和为常数（大于两定点距离）的点的轨迹；其标准方程有 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ 和 $\frac{y^2}{a^2}+\frac{x^2}{b^2}=1$ 两种形式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（常数与焦距）：} 常数 $2a$ 必须大于两定点距离 $2c$（$2a>2c>0$）。
		\item \textbf{条件2（定点固定）：} 两定点 $F_1,F_2$ 固定，位置确定。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（范围限制）：} 忽略 $2a>2c$ 会导致轨迹误判，$2a=2c$ 为线段，$2a<2c$ 无轨迹。
		\item \textbf{检查项（关系正确）：} 椭圆中 $b^2 = a^2 - c^2$，不要与双曲线关系混淆；焦点在长轴，分母大者对应长轴。
	\end{itemize}
\end{summarybox}
