	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（距离和与焦距）：}}] 平面内动点 $P$ 到两定点 $F_1,F_2$ 的距离之和等于常数 $2a$，$|F_1F_2| = 2c$，且 $2a>2c>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（椭圆轨迹）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			到两定点距离和等于固定值的点集构成椭圆。
			\[
				|PF_1|+|PF_2|=2a
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（标准方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当椭圆中心在原点、焦点在坐标轴上时，可用标准方程表示。
			\[
				\frac{x^2}{a^2}+\frac{y^2}{b^2}=1 \quad (b^2=a^2-c^2)
			\]
			\[
				\frac{y^2}{a^2}+\frac{x^2}{b^2}=1 \quad (b^2=a^2-c^2)
			\]
	\end{description}

\end{statementbox}
