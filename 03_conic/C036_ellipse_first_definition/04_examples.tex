\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知两定点 $F_1(-3,0)$、$F_2(3,0)$，动点 $P$ 满足 $|PF_1|+|PF_2|=10$，求点 $P$ 的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证定义）：}] 由题设，$|PF_1|+|PF_2|=10$，且 $|F_1F_2|=6$，满足 $10>6$，符合椭圆定义。
			\par\textbf{理由：}椭圆定义要求常数大于焦距。
		\item[\textbf{第二步（求参数）：}] 设 $2a=10$，得 $a=5$；$2c=6$，得 $c=3$。
			\[
				a=5,\quad c=3
			\]
			\par\textbf{理由：}根据定义列出参数。
		\item[\textbf{第三步（写方程）：}] 由 $b^2=a^2-c^2=25-9=16$，焦点在 $x$ 轴，故轨迹方程为
			\[
				\frac{x^2}{25}+\frac{y^2}{16}=1
			\]
			\par\textbf{理由：}标准方程形式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{轨迹是以 $F_1(-3,0),F_2(3,0)$ 为焦点，长轴为 $10$，短轴为 $8$ 的椭圆，方程为 $\dfrac{x^2}{25}+\dfrac{y^2}{16}=1$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	椭圆 $\dfrac{x^2}{25}+\dfrac{y^2}{9}=1$ 上一点 $P$ 到左焦点 $F_1$ 的距离为 $4$，求点 $P$ 到右焦点 $F_2$ 的距离。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定 a）：}] 从椭圆方程可知 $a^2=25$，故 $a=5$。
			\par\textbf{理由：}在标准方程中 $a$ 是长半轴。
		\item[\textbf{第二步（应用定义）：}] 由椭圆定义，$|PF_1|+|PF_2|=2a=10$。
			\[
				|PF_1|+|PF_2|=10
			\]
			\par\textbf{理由：}椭圆上任意一点到两焦点距离和为常数。
		\item[\textbf{第三步（代入求解）：}] 已知 $|PF_1|=4$，所以 $|PF_2|=10-4=6$。
			\[
				|PF_2|=6
			\]
			\par\textbf{理由：}由定义式直接解出。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{该点到右焦点的距离为 $6$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	椭圆 $\dfrac{x^2}{25}+\dfrac{y^2}{16}=1$ 上一点 $P$ 与两个焦点 $F_1,F_2$ 构成 $\triangle PF_1F_2$，且 $\angle F_1PF_2=60^\circ$，求 $\triangle PF_1F_2$ 的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}] 由椭圆方程 $a=5$，$b=4$，则 $c=\sqrt{a^2-b^2}=3$，焦距 $2c=6$。
			\[
				a=5,\; c=3,\; |F_1F_2|=6
			\]
			\par\textbf{理由：}参数是后续计算的基础。

		\item[\textbf{第二步（设距离列方程）：}] 设 $|PF_1|=m$，$|PF_2|=n$，则由椭圆定义 $m+n=10$，在 $\triangle PF_1F_2$ 中由余弦定理：
			\[
				|F_1F_2|^2=m^2+n^2-2mn\cos60^\circ
				\Rightarrow 36=m^2+n^2-mn
			\]
			\par\textbf{理由：}利用定义和余弦定理建立 $m,n$ 的关系。

		\item[\textbf{第三步（求面积）：}] 将 $m+n=10$ 两边平方得 $m^2+n^2+2mn=100$，减去余弦式得 $3mn=64$，所以 $mn=\dfrac{64}{3}$。则面积
			\[
				S=\frac12 mn\sin60^\circ=\frac12\cdot\frac{64}{3}\cdot\frac{\sqrt3}{2}=\frac{16\sqrt3}{3}
			\]
			\par\textbf{理由：}三角形面积公式 $S=\frac12ab\sin C$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\triangle PF_1F_2$ 的面积为 $\dfrac{16\sqrt3}{3}$。}
\end{examplebox}
