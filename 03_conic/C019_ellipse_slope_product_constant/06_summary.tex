\begin{summarybox}
\textbf{\textcolor{CSummary}{30 秒记忆卡：圆 $\to$ 椭圆}}
\[
 \underbrace{k_{OP}k_l=-1}_{\text{圆：切线垂直半径}}
 \quad\xrightarrow[\text{纵向 }b]{\text{横向 }a}\quad
 \boxed{k_{OP}k_l=-\left(\frac ba\right)^2=e^2-1}.
\]
\textbf{直觉：} 两条斜率都乘 $b/a$，所以乘积乘 $(b/a)^2$。

\medskip
\textbf{\textcolor{CSummary}{条件必看}}
横轴标准椭圆
$\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1\ (a>b>0)$，
$P(x_0,y_0)$ 必须在椭圆上，且 $x_0y_0\ne0$，
才能谈两个普通斜率的乘积。

\medskip
\textbf{\textcolor{CSummary}{考场最快三行}}
已知切点式
$\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2}=1$，所以
\[
 k_l=-\frac{b^2x_0}{a^2y_0},\qquad
 k_{OP}=\frac{y_0}{x_0}
 \quad\Longrightarrow\quad
 \boxed{k_{OP}k_l=-\frac{b^2}{a^2}=e^2-1}.
\]

\medskip
\textbf{\textcolor{CSummary}{使用与反用}}
\begin{itemize}[leftmargin=*, itemsep=0.35em, topsep=0.3em]
\item 已知径向斜率和离心率：$k_l=(e^2-1)/k_{OP}$。
\item 已知两斜率：$e=\sqrt{1+k_{OP}k_l}$，要检查乘积在 $(-1,0)$。
\item 求整条切线或遇到轴点：用
$\dfrac{x_0x}{a^2}+\dfrac{y_0y}{b^2}=1$，它在四个顶点也成立。
\end{itemize}

\medskip
\textbf{\textcolor{CSummary}{三大防错}}
\begin{itemize}[leftmargin=*, itemsep=0.3em, topsep=0.3em]
\item $OP$ 从\textbf{中心}出发，不是从焦点出发。
\item 椭圆切线一般\textbf{不垂直} $OP$；只有圆具有普遍垂直性。
\item 横轴椭圆积为负；标准横轴双曲线积为正，不能混淆。
\end{itemize}
\end{summarybox}
