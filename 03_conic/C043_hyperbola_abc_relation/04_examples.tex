\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	求双曲线 $9x^2-16y^2=144$ 的实半轴长 $a$、虚半轴长 $b$、半焦距 $c$，并指出焦点位置。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化为标准形）：}]
			方程两边除以 $144$ 得 $\frac{x^2}{16} - \frac{y^2}{9} = 1$。
			\par\textbf{理由：} 使右边为 $1$，确定标准形式。
		\item[\textbf{第二步（读取参数）：}]
			由标准方程得 $a^2=16$，$b^2=9$，所以 $a=4$，$b=3$（正数）。
			\par\textbf{理由：} 分母对应 $a^2$ 和 $b^2$。
		\item[\textbf{第三步（计算 $c$ 与焦点）：}]
			由 $c^2=a^2+b^2=16+9=25$ 得 $c=5$。因为 $x^2$ 项系数为正，焦点在 $x$ 轴，坐标为 $(\pm5,0)$。
			\par\textbf{理由：} 关系式以及焦点位置判断。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=4$，$b=3$，$c=5$，焦点在 $x$ 轴。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线的一个焦点为 $(5,0)$，实半轴长 $a=3$，求双曲线的标准方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定焦点轴）：}]
			焦点坐标 $(5,0)$ 在 $x$ 轴，故设双曲线方程为 $\frac{x^2}{a^2} - \frac{y^2}{b^2}=1$。
			\par\textbf{理由：} 焦点位置决定方程形式。
		\item[\textbf{第二步（代入求 $b$）：}]
			$c=5$，$a=3$，由 $c^2=a^2+b^2$ 得 $25=9+b^2$，所以 $b^2=16$。
			\par\textbf{理由：} 直接使用双曲线基本关系。
		\item[\textbf{第三步（写出方程）：}]
			将 $a^2=9$，$b^2=16$ 代入，得 $\frac{x^2}{9} - \frac{y^2}{16}=1$。
			\par\textbf{理由：} 标准方程形式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{双曲线方程为 $\frac{x^2}{9} - \frac{y^2}{16}=1$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线与椭圆 $\frac{x^2}{25} + \frac{y^2}{9}=1$ 有相同的焦点，且实轴长为 $6$，求双曲线的标准方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求椭圆焦点）：}]
			椭圆 $a_{\text{椭}}^2=25$，$b_{\text{椭}}^2=9$，所以 $c_{\text{椭}}^2 = a_{\text{椭}}^2 - b_{\text{椭}}^2 = 25-9=16$，故焦点 $(\pm4,0)$。
			\par\textbf{理由：} 椭圆中 $c^2=a^2-b^2$。
		\item[\textbf{第二步（确定双曲线参数）：}]
			双曲线与椭圆同焦点，所以 $c=4$；实轴长 $2a=6$，所以 $a=3$。
			\par\textbf{理由：} 焦点相同即 $c$ 相等，实轴长给出 $a$。
		\item[\textbf{第三步（求 $b$ 写方程）：}]
			由 $c^2=a^2+b^2$ 得 $16=9+b^2$，$b^2=7$。因为焦点在 $x$ 轴，方程为 $\frac{x^2}{9} - \frac{y^2}{7}=1$。
			\par\textbf{理由：} 代入关系求 $b$，写标准方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{双曲线方程为 $\frac{x^2}{9} - \frac{y^2}{7}=1$。}
\end{examplebox}
