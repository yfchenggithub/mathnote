\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（标准方程形式）：}}]
			双曲线方程为 $\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$ 或 $\frac{y^2}{a^2} - \frac{x^2}{b^2} = 1$，其中 $a>0, b>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平方和关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			半焦距 $c$ 的平方等于实半轴 $a$ 的平方与虚半轴 $b$ 的平方之和。
			\[
				c^2 = a^2 + b^2.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（离心率范围）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			离心率 $e = c/a$ 恒大于 $1$，且 $e$ 越大双曲线开口越开阔。
			\[
				e = \frac{c}{a} > 1.
			\]
	\end{description}

\end{statementbox}
