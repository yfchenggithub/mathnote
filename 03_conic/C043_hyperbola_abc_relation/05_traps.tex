\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆椭圆双曲）：}}]
			学生常常将双曲线中 $a,b,c$ 的关系写成 $c^2 = a^2 - b^2$，错误地套用椭圆结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（双曲加法关系）：}}]
			双曲线中半焦距满足 $c^2 = a^2 + b^2$，且 $c>a$，$c>b$。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式记忆模糊）：}}]
			没有从定义推导理解，仅凭记忆导致使用错误公式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略焦点位置）：}}]
			对于方程 $\frac{y^2}{9} - \frac{x^2}{4} = 1$，仍认为 $a^2=9,b^2=4$ 但焦点在 $x$ 轴，导致后续计算错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（先定焦点轴）：}}]
			先看正项对应的变量，正项分母是 $a^2$，焦点在该轴上；关系 $c^2 = a^2 + b^2$ 仍适用，但 $a$ 必须对应正项分母。
		\item[\textbf{\textcolor{CTrap}{错因分析（标准意识不足）：}}]
			未养成先化标准方程再提取参数的习惯。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开方符号疏漏）：}}]
			在计算 $c$ 时，忽略 $c>0$，或认为 $c$ 可能小于 $a$（受椭圆影响）。
		\item[\textbf{\textcolor{CTrap}{正确理解（c恒大于a）：}}]
			双曲线中 $c^2 = a^2 + b^2$，所以 $c > a$ 且 $c > 0$，开方只取正。
		\item[\textbf{\textcolor{CTrap}{错因分析（受椭圆影响）：}}]
			椭圆中 $c < a$，学生将这种大小关系错误套用到双曲线。
	\end{description}

\end{trapbox}
