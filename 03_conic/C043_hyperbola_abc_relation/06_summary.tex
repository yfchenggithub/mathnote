\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{双曲线标准形式下，半焦距平方等于实半轴平方加虚半轴平方：$c^2=a^2+b^2$，且 $c>a$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（方程标准化）：} 双曲线方程已化为 $\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$ 或 $\frac{y^2}{a^2} - \frac{x^2}{b^2} = 1$，$a>0,b>0$。
		\item \textbf{条件 2（参数含义明确）：} $a$ 为实半轴长，$b$ 为虚半轴长，$c$ 为半焦距。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分椭圆双曲）：} 牢记 $c^2=a^2+b^2$，而不是 $c^2=a^2-b^2$。
		\item \textbf{检查项（确认焦点轴）：} 使用前先确定焦点所在坐标轴，并使 $a$ 对应正项分母。
	\end{itemize}

\end{summarybox}
