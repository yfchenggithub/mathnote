\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 \(\frac{x^2}{4}+\frac{y^2}{3}=1\)，求以点 \(M(1,1)\) 为中点的弦所在的直线方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定公式）：}] 椭圆标准方程 \(a^2=4, b^2=3\)，中点 \(M(1,1)\)，使用椭圆中点弦公式 \(k = -\frac{b^2 x_0}{a^2 y_0}\)。
			\par\textbf{理由：} 直接应用公式的前提。
		\item[\textbf{第二步（代入计算）：}] 代入 \(x_0=1, y_0=1\) 得 \(k = -\frac{3\times1}{4\times1} = -\frac{3}{4}\)。
			\par\textbf{理由：} 代数运算。
		\item[\textbf{第三步（写直线方程）：}] 由点斜式 \(y-1 = -\frac{3}{4}(x-1)\)，整理得 \(3x+4y-7=0\)。
			\par\textbf{理由：} 得到所求直线方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$3x+4y-7=0$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 \(\frac{x^2}{4}-\frac{y^2}{5}=1\)，求以点 \(M(4,3)\) 为中点的弦所在的直线方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定公式）：}] 双曲线标准方程 \(a^2=4, b^2=5\)，中点 \(M(4,3)\)，使用双曲线中点弦公式 \(k = \frac{b^2 x_0}{a^2 y_0}\)。
			\par\textbf{理由：} 直接应用公式的前提。
		\item[\textbf{第二步（代入计算）：}] 代入 \(x_0=4, y_0=3\) 得 \(k = \frac{5\times4}{4\times3} = \frac{20}{12} = \frac{5}{3}\)。
			\par\textbf{理由：} 代数运算。
		\item[\textbf{第三步（写直线方程）：}] 由点斜式 \(y-3 = \frac{5}{3}(x-4)\)，整理得 \(5x-3y-11=0\)。
			\par\textbf{理由：} 得到所求直线方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$5x-3y-11=0$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	斜率为 \(2\) 的直线与抛物线 \(y^2=4x\) 交于 \(A,B\) 两点，求弦 \(AB\) 中点 \(M\) 的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（点差设关系）：}] 设 \(A(x_1,y_1), B(x_2,y_2)\)，中点 \(M(x_0,y_0)\)。对 \(y^2=4x\) 作差得 \((y_1-y_2)(y_1+y_2)=4(x_1-x_2)\)，即 \(k\cdot y_0 = 2\)（因为 \(y_1+y_2=2y_0\)）。
			\par\textbf{理由：} 点差法步骤。
		\item[\textbf{第二步（代斜率求中点）：}] 已知 \(k=2\)，代入 \(2\cdot y_0 = 2\)，解得 \(y_0=1\)。
			\par\textbf{理由：} 解方程。
		\item[\textbf{第三步（得轨迹方程）：}] 中点 \(M\) 的纵坐标恒为 \(1\)，轨迹方程为 \(y=1\)（位于抛物线内的一段）。
			\par\textbf{理由：} 点差法直接得到纵坐标与斜率关系。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y=1$}
\end{examplebox}
