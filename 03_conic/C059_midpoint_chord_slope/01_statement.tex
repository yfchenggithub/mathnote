	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（两交点存在）：}}] 弦与圆锥曲线有两个交点。
		\item[\textbf{\textcolor{CStatement}{条件二（中点非零纵坐标）：}}] 中点 \(M(x_0,y_0)\) 满足 \(y_0\neq0\)。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（椭圆中点弦）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆弦斜率与中点坐标的关系。
			\[
				k = -\frac{b^2 x_0}{a^2 y_0}
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（双曲线中点弦）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			双曲线弦斜率与中点坐标的关系。
			\[
				k = \frac{b^2 x_0}{a^2 y_0}
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（抛物线中点弦）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			抛物线弦斜率与中点纵坐标的关系。
			\[
				k = \frac{p}{y_0}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（椭圆退化为圆）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当椭圆 \(a=b=r\) 时，公式简化为圆的中点弦斜率。
			\[
				k = -\frac{x_0}{y_0}
			\]
	\end{description}

\end{statementbox}
