\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知抛物线 $y^2=4x$（$p=2$）与直线 $x+2y-2=0$ 相交，求弦长 $|AB|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别系数）：}]
		      直线化为
		      \[
			      1\cdot x+2\cdot y-2=0,
		      \]
		      即
		      \[
			      A=1,\qquad B=2,\qquad C=-2.
		      \]
		      抛物线 $y^2=2px$ 中 $p=2$。
		      \par\textbf{理由：} 抛物线开口向右，且 $A=1\neq0$，可以使用结论一；又 $B=2\neq0$，斜率存在，可使用斜率弦长公式。

		\item[\textbf{第二步（套用韦达）：}]
		      由结论一，关于 $x$ 的二次方程为
		      \[
			      A^2x^2+(2AC-2B^2p)x+C^2=0.
		      \]
		      代入得
		      \[
			      x^2+\bigl(2\cdot1\cdot(-2)-2\cdot2^2\cdot2\bigr)x+(-2)^2=0,
		      \]
		      即
		      \[
			      x^2-20x+4=0.
		      \]
		      故
		      \[
			      x_1+x_2=20,\qquad x_1x_2=4.
		      \]
		      \par\textbf{理由：} 直接代入系数，避免完整联立解方程组。

		\item[\textbf{第三步（计算弦长）：}]
		      直线斜率为
		      \[
			      k=-\frac{A}{B}=-\frac12.
		      \]
		      又
		      \[
			      |x_1-x_2|
			      =\sqrt{(x_1+x_2)^2-4x_1x_2}
			      =\sqrt{20^2-4\cdot4}
			      =\sqrt{384}
			      =8\sqrt6.
		      \]
		      所以
		      \[
			      |AB|
			      =\sqrt{1+k^2}\,|x_1-x_2|
			      =\sqrt{1+\frac14}\cdot8\sqrt6
			      =\frac{\sqrt5}{2}\cdot8\sqrt6
			      =4\sqrt{30}.
		      \]
		      \par\textbf{理由：} 本题不是竖直弦，斜率弦长公式适用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 弦长 $|AB|=4\sqrt{30}$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 抛物线 $x^2=8y$（$p=4$），直线 $2x-y+1=0$，求弦的中点坐标。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别系数）：}]
		      直线
		      \[
			      2x-y+1=0
		      \]
		      中
		      \[
			      A=2,\qquad B=-1,\qquad C=1.
		      \]
		      抛物线 $x^2=2py$ 开口向上，$p=4$。
		      \par\textbf{理由：} 开口向上应使用结论三；且 $B=-1\neq0$，关于 $x$ 的方程是二次方程。

		\item[\textbf{第二步（套用韦达）：}]
		      由结论三，关于 $x$ 的二次方程为
		      \[
			      Bx^2+2Apx+2Cp=0.
		      \]
		      代入得
		      \[
			      -x^2+2\cdot2\cdot4x+2\cdot1\cdot4=0,
		      \]
		      即
		      \[
			      -x^2+16x+8=0.
		      \]
		      两边同乘 $-1$，得
		      \[
			      x^2-16x-8=0.
		      \]
		      故
		      \[
			      x_1+x_2=16,\qquad x_0=\frac{x_1+x_2}{2}=8.
		      \]
		      \par\textbf{理由：} 用韦达定理直接得到中点横坐标。

		\item[\textbf{第三步（求中点纵坐标）：}]
		      由直线
		      \[
			      2x-y+1=0
		      \]
		      得
		      \[
			      y=2x+1.
		      \]
		      代入 $x_0=8$，得
		      \[
			      y_0=2\cdot8+1=17.
		      \]
		      \par\textbf{理由：} 弦的中点仍在弦所在直线上。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 中点坐标为 $(8,17)$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 抛物线 $y^2=4x$，直线 $y=kx+1$ 与抛物线交于 $A,B$ 两点，且弦 $AB$ 中点在直线 $x=1$ 上，求 $k$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（排除退化并识别系数）：}]
		      若 $k=0$，则直线为 $y=1$，与 $y^2=4x$ 只有一个交点，不符合题意。
		      因此 $k\neq0$。
		      直线化为
		      \[
			      kx-y+1=0,
		      \]
		      所以
		      \[
			      A=k,\qquad B=-1,\qquad C=1.
		      \]
		      抛物线 $y^2=4x$ 即 $y^2=2px$，故 $p=2$。
		      \par\textbf{理由：} 抛物线开口向右，且 $A=k\neq0$，可以使用结论一。

		\item[\textbf{第二步（中点横坐标表达式）：}]
		      由结论一，
		      \[
			      x_1+x_2
			      =-\frac{2AC-2B^2p}{A^2}
			      =-\frac{2k-2\cdot1\cdot2}{k^2}
			      =-\frac{2k-4}{k^2}.
		      \]
		      因此中点横坐标为
		      \[
			      x_0=\frac{x_1+x_2}{2}
			      =-\frac{k-2}{k^2}.
		      \]
		      \par\textbf{理由：} 用韦达定理直接表示弦中点横坐标。

		\item[\textbf{第三步（利用中点条件）：}]
		      因为弦 $AB$ 的中点在直线 $x=1$ 上，所以
		      \[
			      x_0=1.
		      \]
		      于是
		      \[
			      -\frac{k-2}{k^2}=1.
		      \]
		      整理得
		      \[
			      k^2+k-2=0.
		      \]
		      解得
		      \[
			      k=1\quad\text{或}\quad k=-2.
		      \]
		      \par\textbf{理由：} 中点横坐标条件转化为关于 $k$ 的方程。

		\item[\textbf{第四步（检验两交点条件）：}]
		      当 $k=1$ 时，直线为 $y=x+1$。
		      联立
		      \[
			      (x+1)^2=4x
		      \]
		      得
		      \[
			      x^2-2x+1=0,
		      \]
		      此时 $\Delta_x=0$，直线与抛物线相切，不符合“两点”要求。

		      当 $k=-2$ 时，直线为 $y=-2x+1$。
		      联立
		      \[
			      (-2x+1)^2=4x
		      \]
		      得
		      \[
			      4x^2-8x+1=0,
		      \]
		      此时
		      \[
			      \Delta_x=(-8)^2-4\cdot4\cdot1=48>0,
		      \]
		      符合题意。
		      \par\textbf{理由：} 本题不是竖直弦，$\Delta_x>0$ 可以判断两个交点横坐标不同。

	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $k=-2$。
\end{examplebox}
