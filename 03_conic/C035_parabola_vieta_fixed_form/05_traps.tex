\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（二次项系数为零）}}]
		      直接套公式时，只看“直线与抛物线联立”，却没有检查所得方程是否真的是二次方程。

		\item[\textbf{\textcolor{CTrap}{正确理解（先看开口，再看系数）：}}]
		      对于左右开口抛物线
		      \[
			      y^2=2px,\qquad y^2=-2px,
		      \]
		      本文公式得到的是
		      \[
			      A^2x^2+\cdots=0,
		      \]
		      因此需要 $A\neq0$。

		      对于上下开口抛物线
		      \[
			      x^2=2py,\qquad x^2=-2py,
		      \]
		      本文公式得到的是
		      \[
			      Bx^2+\cdots=0,
		      \]
		      因此需要 $B\neq0$。

		\item[\textbf{\textcolor{CTrap}{错因分析（把条件写成一个）：}}]
		      不能简单写成“$A\neq0$”或“$B\neq0$”适用于所有开口。
		      左右开口和上下开口的二次项系数不同，前提也不同。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（把 $\Delta_x>0$ 等同于所有两点相交）}}]
		      很多同学会直接认为“直线与抛物线有两个交点”等价于关于 $x$ 的方程满足 $\Delta_x>0$。

		\item[\textbf{\textcolor{CTrap}{正确理解（$\Delta_x$ 判断的是两个不同横坐标）：}}]
		      对于本文整理出的关于 $x$ 的二次方程，$\Delta_x>0$ 表示两个交点的横坐标不同。
		      如果两个交点的横坐标相同，则关于 $x$ 的方程可能表现为重根。

		\item[\textbf{\textcolor{CTrap}{典型特例（左右开口的竖直弦）：}}]
		      例如
		      \[
			      y^2=2px
		      \]
		      与竖直线 $x=x_0$ 相交。
		      当 $x_0>0$ 时，交点为
		      \[
			      (x_0,\sqrt{2px_0}),\qquad (x_0,-\sqrt{2px_0}).
		      \]
		      这两个交点不同，但
		      \[
			      x_1=x_2=x_0.
		      \]
		      此时关于 $x$ 的方程表现为重根，不能简单用 $\Delta_x>0$ 来概括。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（斜率弦长公式乱用）}}]
		      看到弦长就直接使用
		      \[
			      |AB|=\sqrt{1+k^2}\,|x_1-x_2|
		      \]
		      而没有检查直线斜率是否存在。

		\item[\textbf{\textcolor{CTrap}{正确理解（斜率存在才可用）：}}]
		      若直线
		      \[
			      Ax+By+C=0
		      \]
		      中 $B\neq0$，则斜率存在，
		      \[
			      k=-\frac{A}{B},
		      \]
		      可以使用
		      \[
			      |AB|=\sqrt{1+k^2}\,|x_1-x_2|.
		      \]
		      若 $B=0$，直线为竖直线，斜率不存在，必须直接计算纵坐标差。

		\item[\textbf{\textcolor{CTrap}{竖直弦处理：}}]
		      对于
		      \[
			      y^2=2px,\qquad x=x_0,\qquad x_0>0,
		      \]
		      弦长应直接写成
		      \[
			      |AB|=2\sqrt{2px_0}.
		      \]
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（开口方向符号混淆）}}]
		      不同开口下公式中 $p$ 的符号和系数位置容易记混。

		\item[\textbf{\textcolor{CTrap}{正确理解（开口与符号对应）：}}]
		      开口向右：
		      \[
			      y^2=2px
		      \]
		      开口向左：
		      \[
			      y^2=-2px
		      \]
		      开口向上：
		      \[
			      x^2=2py
		      \]
		      开口向下：
		      \[
			      x^2=-2py
		      \]

		\item[\textbf{\textcolor{CTrap}{错因分析（只背一个形式）：}}]
		      左右开口主要是 $y^2=\pm2px$ 的符号变化；
		      上下开口则变成 $x^2=\pm2py$，变量位置也变了，不能只说“符号变一下”。
	\end{description}
\end{trapbox}
