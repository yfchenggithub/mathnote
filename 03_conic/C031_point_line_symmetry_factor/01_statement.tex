\begin{statementbox}
\textbf{\textcolor{CStatement}{条件与结论}}

给定点 $P(x_0,y_0)$ 和直线 $l:Ax+By+C=0$，其中 $A,B$ 不同时为零。
令对称因子
\[
 t=\frac{Ax_0+By_0+C}{A^2+B^2}.
\]
则 $P$ 关于 $l$ 的对称点为
\[
 \boxed{P'(x_0-2At,\ y_0-2Bt)}.
\]
\input{assets/tikz/C031_reflection_overview}

\textbf{\textcolor{CStatement}{先求垂足，再求对称点}}
\[
 H(x_0-At,\ y_0-Bt),\qquad P'=2H-P.
\]
这里 $H$ 是 $P$ 到 $l$ 的垂足；若 $P$ 已在直线上，则 $t=0$，$H=P'=P$。

\medskip
\textbf{\textcolor{CStatement}{展开式（与上式等价）}}
\[
\begin{cases}
 x'=x_0-\dfrac{2A(Ax_0+By_0+C)}{A^2+B^2},\\[7pt]
 y'=y_0-\dfrac{2B(Ax_0+By_0+C)}{A^2+B^2}.
\end{cases}
\]
$t$ 可以为正、负或零；它是位移的比例因子，不能直接当作距离。
\end{statementbox}
