\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（直线一般式）：}}]
		      给定点 $P(x_0,y_0)$ 和直线 $l:Ax+By+C=0$，其中 $A,B$ 不同时为零。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对称点公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 通过对称因子 $t$ 直接计算对称点坐标。
		      \[
			      \begin{cases}
				      x' = x_0 - 2At, \\
				      y' = y_0 - 2Bt,
			      \end{cases}
		      \]
		      其中
		      \[
			      t = \dfrac{Ax_0+By_0+C}{A^2+B^2}.
		      \]

		\item[\textbf{\textcolor{CStatement}{等价形式（分式展开式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 将 $t$ 代入展开为分式形式，便于直接代值；这只是同一公式的等价写法，不是新的条件或新的结论。
		      \[
			      \begin{cases}
				      x' = x_0 - \dfrac{2A(Ax_0+By_0+C)}{A^2+B^2}, \\[10pt]
				      y' = y_0 - \dfrac{2B(Ax_0+By_0+C)}{A^2+B^2}.
			      \end{cases}
		      \]
	\end{description}
\end{statementbox}
