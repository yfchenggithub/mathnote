\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对称因子
		\[
			t = \dfrac{Ax_0+By_0+C}{A^2+B^2}
		\]
		将点关于直线的对称问题转化为代入计算，避免解方程组。}

	\medskip

	\textbf{\textcolor{CSummary}{使用条件}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（一般式直线）：} 直线方程必须写成 $Ax+By+C=0$ 形式，且 $A,B$ 不全为零。
		\item \textbf{条件2（坐标已知）：} 点 $P(x_0,y_0)$ 的坐标已知。
	\end{itemize}

	\medskip

	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{符号提醒：} 公式为
		      \[
			      x' = x_0 - 2At,\qquad y' = y_0 - 2Bt.
		      \]
		      其中 $t$ 可正可负，实际移动方向由 $Ax_0+By_0+C$ 的符号决定，不能把减号机械理解成固定向某一侧移动。

		\item \textbf{分母提醒：} 对称因子的分母是
		      \[
			      A^2+B^2,
		      \]
		      不要误写成 $A+B$，也不要与点到直线距离公式中的 $\sqrt{A^2+B^2}$ 混淆。

		\item \textbf{方程形式：} 确保直线方程已化为
		      \[
			      Ax+By+C=0
		      \]
		      的一般式，并正确识别 $A,B,C$。

		\item \textbf{结果检验：} 若求得 $P'(x',y')$，则中点
		      \[
			      M\left(\frac{x_0+x'}{2},\frac{y_0+y'}{2}\right)
		      \]
		      必须在直线 $l$ 上，即
		      \[
			      A\cdot\frac{x_0+x'}{2}
			      +B\cdot\frac{y_0+y'}{2}
			      +C=0.
		      \]
		      同时，
		      \[
			      (x'-x_0,\ y'-y_0)
		      \]
		      应与法向量 $(A,B)$ 平行。
	\end{itemize}
\end{summarybox}
