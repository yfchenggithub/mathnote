\begin{summarybox}
\textbf{\textcolor{CSummary}{一句话核心}}

沿垂线到 $H$，再走同样远到 $P'$；$H$ 是中点，$P'=2H-P$。

\medskip
\textbf{\textcolor{CSummary}{一般式速查}}
\[
 t=\frac{Ax_0+By_0+C}{A^2+B^2},\quad
 H=(x_0-At,y_0-Bt),\quad
 P'=(x_0-2At,y_0-2Bt).
\]
条件：$l:Ax+By+C=0$，且 $A,B$ 不同时为零。

\medskip
\textbf{\textcolor{CSummary}{斜截式速查}}
\[
 \Delta=y_0-kx_0-b,\qquad
 PH=\frac{|\Delta|}{\sqrt{1+k^2}},\qquad s=\frac{\Delta}{1+k^2}=-t,
\]
\[
 H=(x_0+ks,y_0-s),\qquad P'=(x_0+2ks,y_0-2s).
\]
直线为 $x=c$ 时，直接用 $P'=(2c-x_0,y_0)$。

\medskip
\textbf{\textcolor{CSummary}{最后三项检查}}
\begin{itemize}[leftmargin=*,itemsep=.4em]
\item 区分竖直差、实际距离和带符号的比例；求长度时保留绝对值。
\item 一般式系数读对；斜截式中 $B=-1$，并注意 $s=-t$。
\item 检查中点在线上、位移与法向量平行；点在线上时结果应保持不动。
\end{itemize}
\end{summarybox}
