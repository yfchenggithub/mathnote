\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用“点差法”：设出弦的两端点坐标，分别代入双曲线方程，作差后引入中点坐标和弦的斜率，直接消去未知点，得到斜率与中点坐标的关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设点代方程）：}] 设弦 $AB$ 的端点坐标为 $A(x_1,y_1)$、$B(x_2,y_2)$，中点为 $M(x_0,y_0)$，斜率 $k=\frac{y_1-y_2}{x_1-x_2}$。由 $A,B$ 在双曲线上，有
			\[
				\frac{x_1^2}{a^2} - \frac{y_1^2}{b^2} = 1,
				\qquad
				\frac{x_2^2}{a^2} - \frac{y_2^2}{b^2} = 1.
			\]
			\par\textbf{理由：} 直接根据双曲线标准方程写出 $A,B$ 满足的条件。

		\item[\textbf{步骤二（作差因式分解）：}] 将两式相减，得
			\[
				\frac{x_1^2 - x_2^2}{a^2} - \frac{y_1^2 - y_2^2}{b^2} = 0.
			\]
			利用平方差公式分解：
			\[
				\frac{(x_1 - x_2)(x_1 + x_2)}{a^2} - \frac{(y_1 - y_2)(y_1 + y_2)}{b^2} = 0.
			\]
			\par\textbf{理由：} $x_1^2 - x_2^2 = (x_1-x_2)(x_1+x_2)$，对 $y$ 同理。

		\item[\textbf{步骤三（代入斜率关系）：}] 因为 $x_1 \neq x_2$（斜率存在），将 $y_1 - y_2 = k(x_1 - x_2)$ 代入上式，得
			\[
				\frac{(x_1 - x_2)(x_1 + x_2)}{a^2} - \frac{k(x_1 - x_2)(y_1 + y_2)}{b^2}=0.
			\]
			提取公因式 $(x_1 - x_2)$：
			\[
				(x_1 - x_2)\left[\frac{x_1 + x_2}{a^2} - \frac{k(y_1 + y_2)}{b^2}\right]=0.
			\]
			由于 $x_1\neq x_2$，两边除以 $x_1 - x_2$，得
			\[
				\frac{x_1 + x_2}{a^2} - \frac{k(y_1 + y_2)}{b^2}=0.
			\]
			\par\textbf{理由：} 利用平方差结构引入斜率；除法由 $x_1\neq x_2$ 保证。

		\item[\textbf{步骤四（中点替换）：}] 由中点公式 $x_1+x_2=2x_0$，$y_1+y_2=2y_0$，代入上式：
			\[
				\frac{2x_0}{a^2} - \frac{k \cdot 2y_0}{b^2}=0.
			\]
			约去 $2$ 后整理：
			\[
				\frac{x_0}{a^2} = \frac{k y_0}{b^2}.
			\]
			\par\textbf{理由：} 中点坐标关系直接代换。

		\item[\textbf{步骤五（解出斜率）：}] 由 $y_0 \neq 0$，从 $\frac{x_0}{a^2} = \frac{k y_0}{b^2}$ 解出斜率：
			\[
				k = \frac{b^2 x_0}{a^2 y_0}.
			\]
			\par\textbf{理由：} 代数变形直接导出公式。注意，$y_0 \neq 0$ 是条件，否则分母为零无意义；不等式 $\frac{x_0^2}{a^2} - \frac{y_0^2}{b^2} < 1$ 保证弦与双曲线有两个交点（中点弦存在）；$k$ 不与渐近线平行确保直线与双曲线交于两点（而非渐近方向）。这些条件缺一不可。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	通过点差法，我们直接建立了双曲线中点弦斜率与中点坐标的关系：$k = \frac{b^2 x_0}{a^2 y_0}$。该公式将复杂的弦与中点关系转化为简洁的代数表达式，是解决中点弦问题的有力工具。
\end{proofbox}
