\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（标准双曲线）：}}] 双曲线方程为 $\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$，其中 $a,b>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（中点非顶点）：}}] 弦 $AB$ 的中点为 $M(x_0,y_0)$，且 $y_0 \neq 0$。
		\item[\textbf{\textcolor{CStatement}{条件3（内点限制）：}}] 点 $M$ 满足 $\frac{x_0^2}{a^2} - \frac{y_0^2}{b^2} < 1$（保证中点弦存在）。
		\item[\textbf{\textcolor{CStatement}{条件4（斜率存在）：}}] 弦的斜率 $k$ 存在，且不与双曲线的渐近线 $y=\pm\frac{b}{a}x$ 平行。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中点弦斜率公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			弦的斜率等于半短轴平方乘横坐标除以半长轴平方乘纵坐标。
			\[
				k = \frac{b^2 x_0}{a^2 y_0}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（纵轴双曲线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当双曲线焦点在 $y$ 轴时，公式变为半长轴平方乘横坐标除以半短轴平方乘纵坐标。
			\[
				k = \frac{a^2 x_0}{b^2 y_0}.
			\]
			此时 $a$ 对应 $y$ 轴半轴，$b$ 对应 $x$ 轴半轴，条件类似（$y_0\neq0$，$\frac{y_0^2}{a^2}-\frac{x_0^2}{b^2}<1$）。
	\end{description}
\end{statementbox}
