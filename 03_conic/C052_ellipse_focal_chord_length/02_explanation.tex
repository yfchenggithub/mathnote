\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n只需知道椭圆的半长轴 $a$、半短轴 $b$ 和直线的倾斜角 $\theta$，直接代入公式即可得到焦点弦长，无需联立方程组。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（直线参数化）：}]
			过焦点的直线用倾斜角 $\theta$ 表示参数方程，代入椭圆方程。\n
		\item[\textbf{要点二（韦达定理）：}]
			利用韦达定理求出弦长公式，消去参数简化得到 $\frac{2ab^2}{a^2-c^2\cos^2\theta}$。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（离心率关联）}}\n\n焦点弦长 $L$ 依赖于 $a$、$b$ 和 $\cos^2\theta$，当 $\theta=90^\circ$ 时取得最小值通径 $2b^2/a$；当 $\theta=0^\circ$ 时，分母最小，弦长最长。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（三角分式）}}\n\n公式可视为关于 $\cos^2\theta$ 的分式函数，其取值范围对应弦长的变化规律。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（弦长与最值）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（直接代入）：} 已知椭圆方程和焦点弦倾斜角，直接代入公式计算弦长。\n
		\item \textbf{考法二（最值问题）：} 由 $\cos^2\theta$ 范围 $[0,1]$ 求弦长最值，常结合离心率。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n公式分母 $a^2-c^2\cos^2\theta$ 与椭圆极坐标方程 $\rho=ep/(1-e\cos\theta)$ 密切相关，理解后可打通圆锥曲线统一性。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（直接求弦长）：} 题目给出倾斜角或斜率，求焦点弦长。\n
		\item \textbf{场景二（反求参数）：} 已知弦长或其范围，反求倾斜角或离心率。\n
	\end{itemize}
	\n
\end{explanationbox}
