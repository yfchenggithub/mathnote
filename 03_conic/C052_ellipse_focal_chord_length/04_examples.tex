\textbf{\textcolor{CExample}{例1（基础：已知角求焦点弦长）}}


\textbf{题目：}椭圆 $x^2/4+y^2/3=1$，过右焦点的弦倾斜角为 $60^\circ$，求弦长。
\textbf{解：}$a=2,b^2=3,c^2=1$，故
$L=12/(4-1/4)=\boxed{16/5}$。
\textbf{核验：}主图两段 $6/5+2=16/5$，且通径 $3<L<4$（长轴弦）。
\par\medskip
\textbf{\textcolor{CExample}{例2（反求角：注意余弦的正负）}}

\textbf{题目：}上述椭圆的某条焦点弦长为 $16/5$，求倾斜角 $\theta\in[0,\pi)$。

\begin{center}\input{assets/tikz/C052_supplementary_angles.tex}\end{center}
\textbf{解：}
\[
 \frac{16}{5}=\frac{12}{4-\cos^2\theta}
 \ \Longrightarrow\ \cos^2\theta=\frac14
 \ \Longrightarrow\ \cos\theta=\pm\frac12.
\]
故 $\boxed{\theta=60^\circ\ \text{或}\ 120^\circ}$。
\textbf{陷阱：}已知的是 $\cos^2\theta$，\textbf{不是} $\cos\theta$。

\medskip
\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例3（反求椭圆参数：保留原题）}}

\textbf{题目：}椭圆离心率 $e=1/2$，一条倾斜角为 $60^\circ$ 的焦点弦长为 $16/5$，求椭圆方程（长轴在 $x$ 轴）。

\textbf{解：}由 $c=a/2$、$b^2=a^2-c^2=3a^2/4$，
\[
 L=\frac{2a(3a^2/4)}{a^2-(a^2/4)(1/4)}
 =\frac{8a}{5}=\frac{16}{5}.
\]
得到 $a=2$、$b^2=3$，所以
$\boxed{\dfrac{x^2}{4}+\dfrac{y^2}{3}=1}$。

\medskip
\textbf{\textcolor{CExample}{例4（给斜率：不用算角度）}}

\textbf{题目：}椭圆 $\dfrac{x^2}{25}+\dfrac{y^2}{9}=1$，过右焦点的直线斜率 $k=1$，求焦点弦长。

\textbf{解：}$a=5$，$b^2=9$，$c^2=16$，
$\cos^2\theta=\dfrac1{1+k^2}=\dfrac12$，故
\[
 L=\frac{2\cdot5\cdot9}{25-16\cdot\frac12}
 =\boxed{\frac{90}{17}}.
\]
\textbf{捷径：}直接写 $\displaystyle L=\frac{2ab^2(1+k^2)}{a^2(1+k^2)-c^2}$。

\medskip
\textbf{\textcolor{CExample}{例5（特例：不用背通径，现场算）}}

\textbf{题目：}椭圆 $\dfrac{x^2}{25}+\dfrac{y^2}{9}=1$，过右焦点 $F(4,0)$ 作竖直线，求弦长。

\textbf{解：}令 $x=4$，则 $\dfrac{16}{25}+\dfrac{y^2}{9}=1$，
所以 $y=\pm\dfrac95$，弦长
$\boxed{L=\dfrac{18}{5}}$。
\textbf{核验：}通径公式 $2b^2/a=18/5$。

\medskip
\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例6（最值：不用求导）}}

\textbf{题目：}在椭圆 $\dfrac{x^2}{25}+\dfrac{y^2}{9}=1$ 中，焦点弦长 $L$ 的范围是什么？

\textbf{解：}$0\le\cos^2\theta\le1$，且
$b^2\le a^2-c^2\cos^2\theta\le a^2$。
故
\[
 \boxed{\frac{18}{5}\le L\le10}.
\]
竖直焦点弦取最小值 $18/5$；沿长轴的焦点弦取最大值 $10$。

\medskip
\textbf{\textcolor{CExample}{例7（辨析：普通竖直弦不是通径）}}

\textbf{题目：}椭圆 $\dfrac{x^2}{4}+\dfrac{y^2}{3}=1$，比较直线 $x=0$ 与 $x=1$ 的截弦长度。

\textbf{解：}焦点为 $(\pm1,0)$。
\[
 x=0:\ y=\pm\sqrt3\ \Longrightarrow\ L_0=\boxed{2\sqrt3};\qquad
 x=1:\ y=\pm\frac32\ \Longrightarrow\ L_1=\boxed{3}.
\]
两条都是竖直弦，\textbf{只有 $x=1$ 经过焦点}，因此只有它是通径。

\medskip
\end{examplebox}

\begin{examplebox}
\textbf{\textcolor{CExample}{例8（推广：长轴转到 $y$ 轴）}}

\textbf{题目：}椭圆 $\dfrac{x^2}{9}+\dfrac{y^2}{25}=1$，过焦点的直线与 $x$ 轴夹角 $30^\circ$，求弦长。

\begin{center}\input{assets/tikz/C052_vertical_major_axis_angle.tex}\end{center}
\textbf{解：}现在长轴在 $y$ 轴，$a=5,b^2=9,c^2=16$。
直线与长轴的锐角为 $\varphi=60^\circ$，
故 $\cos^2\varphi=1/4$，
\[
 L=\frac{2\cdot5\cdot9}{25-16/4}
 =\boxed{\frac{30}{7}}.
\]
\textbf{陷阱：}此题若把与 $x$ 轴夹角的 $\cos^230^\circ$ 硬套进原公式，必错。
