\textbf{\textcolor{CExample}{例 1（基础）}}

\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{4}+\frac{y^2}{3}=1$，过右焦点且倾斜角为 $60^\circ$ 的直线交椭圆于 $A,B$ 两点，求弦 $AB$ 的长。

\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由椭圆方程得 $a=2$，$b^2=3$，$c=\sqrt{a^2-b^2}=1$，倾斜角 $\theta=60^\circ$，$\cos\theta=\frac12$。

			\par\textbf{理由：} 代入公式需要这些量。

		\item[\textbf{第二步（代入公式）：}]
			焦点弦长公式 $L=\frac{2ab^2}{a^2-c^2\cos^2\theta}$，代入得
			\[
				\begin{aligned}
					L &= \frac{2\times2\times3}{4-1\times\left(\frac12\right)^2} \\
					&= \frac{12}{4-\frac14} \\
					&= \frac{12}{\frac{15}{4}} \\
					&= \frac{48}{15} \\
					&= \frac{16}{5}.
				\end{aligned}
			\]

			\par\textbf{理由：} 直接计算即得。

		\item[\textbf{第三步（给出结论）：}]
			所以 $|AB|=\frac{16}{5}$。

			\par\textbf{理由：} 根据计算得到最终答案。
	\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{$|AB| = \dfrac{16}{5}$}

\textbf{\textcolor{CExample}{例 2（稍变形）}}

\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{4}+\frac{y^2}{3}=1$，过右焦点且倾斜角为 $\theta$ 的焦点弦长等于 $\frac{16}{5}$，求 $\cos\theta$ 的值。

\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入公式）：}]
			由公式 $L=\frac{2ab^2}{a^2-c^2\cos^2\theta}$，得 $\frac{16}{5} = \frac{12}{4-\cos^2\theta}$。

			\par\textbf{理由：} 已知 $a=2,b^2=3,c=1$，代入化简。

		\item[\textbf{第二步（解方程）：}]
			分母乘过去：
			\[
				\begin{aligned}
					16(4-\cos^2\theta) &= 60 \\
					64 - 16\cos^2\theta &= 60 \\
					16\cos^2\theta &= 4 \\
					\cos^2\theta &= \frac14 \\
					\cos\theta &= \pm\frac12.
				\end{aligned}
			\]

			\par\textbf{理由：} 代数变形解出 $\cos^2\theta$，注意开方有正负。

		\item[\textbf{第三步（结论）：}]
			所以 $\cos\theta = \pm\frac12$。

			\par\textbf{理由：} 倾斜角 $\theta$ 范围 $[0,\pi)$，余弦可正可负，均能使弦长成立。
	\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\cos\theta = \pm\dfrac12$}

\textbf{\textcolor{CExample}{例 3（综合应用）}}

\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$（$a>b>0$）的离心率 $e=\frac12$，过右焦点且倾斜角为 $60^\circ$ 的直线被椭圆截得的弦长等于 $\frac{16}{5}$，求椭圆方程。

\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（利用离心率）：}]
			由 $e=\frac{c}{a}=\frac12$ 得 $c=\frac a2$，从而 $b^2=a^2-c^2=a^2-\frac{a^2}{4}=\frac{3a^2}{4}$。

			\par\textbf{理由：} 离心率给出 $a,b,c$ 的关系。

		\item[\textbf{第二步（代入弦长公式）：}]
			$\theta=60^\circ$，$\cos\theta=\frac12$，代入公式 $L=\frac{2ab^2}{a^2-c^2\cos^2\theta}$ 得
			\[
				\begin{aligned}
					L &= \frac{2a\cdot\frac{3a^2}{4}}{a^2-\frac{a^2}{4}\cdot\frac14} \\
					&= \frac{\frac{3a^3}{2}}{a^2-\frac{a^2}{16}} \\
					&= \frac{\frac{3a^3}{2}}{\frac{15a^2}{16}} \\
					&= \frac{3a^3}{2}\cdot\frac{16}{15a^2} \\
					&= \frac{8a}{5}.
				\end{aligned}
			\]

			\par\textbf{理由：} 利用 $a$ 表达 $L$，化简为 $8a/5$。

		\item[\textbf{第三步（求 $a$）：}]
			令 $\frac{8a}{5}=\frac{16}{5}$，解得 $a=2$。于是 $c=1$，$b^2=3$。

			\par\textbf{理由：} 由待定系数法求出 $a$，再回代得 $b$。

		\item[\textbf{第四步（结论）：}]
			椭圆方程为 $\frac{x^2}{4}+\frac{y^2}{3}=1$。

			\par\textbf{理由：} 代入标准形式即得。
	\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\dfrac{x^2}{4}+\dfrac{y^2}{3}=1$}

\end{examplebox}
