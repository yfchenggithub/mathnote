\begin{summarybox}

\textbf{\textcolor{CSummary}{一眼触发}}

\textbf{``椭圆＋过焦点的弦＋角度/斜率'' $\Longrightarrow$ 焦点弦长。}
椭圆 $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$，$a>b>0$，$c^2=a^2-b^2$。
\[
 \boxed{L=\frac{2ab^2}{a^2-c^2\cos^2\theta}}
 \qquad\left(\theta\text{ 是弦与长轴的夹角}\right).
\]

\textbf{\textcolor{CSummary}{最低记忆负担（不想背就这样现推）}}

\[
 \underbrace{FA}_{r}=\frac{b^2}{a+c\cos\theta},\qquad
 \underbrace{FB}_{s}=\frac{b^2}{a-c\cos\theta},\qquad
 \boxed{AB=FA+FB}.
\]
\textbf{同分子，两分母一加一减}，相加就是平方差。
若会焦半径 $PF_2=a-\frac ca x$，只需令
$x_A=c+r\cos\theta$、$x_B=c-s\cos\theta$，各列一个一次方程即可。

\medskip
\textbf{\textcolor{CSummary}{不用背的两个端点检查}}

\begin{itemize}[leftmargin=*,itemsep=0.35em,topsep=0.35em]
 \item \textbf{横着穿长轴：}$\theta=0^\circ$，$L_{\max}=2a$。
 \item \textbf{竖着过焦点：}$\theta=90^\circ$，$L_{\min}=2b^2/a$（通径）。
\end{itemize}

\medskip
\textbf{\textcolor{CSummary}{斜率直接版／反求版}}

\[
 \cos^2\theta=\frac1{1+k^2}\quad(k\text{ 为有限斜率}),\qquad
 L=\frac{2ab^2(1+k^2)}{a^2(1+k^2)-c^2}.
\]
\[
 \cos^2\theta=\frac{a^2-2ab^2/L}{c^2}\qquad(c>0).
\]
逆求倾斜角\textbf{注意余弦正负}；长轴在 $y$ 轴时将角度改为\textbf{相对长轴}的角。

\medskip
\textbf{\textcolor{CSummary}{考场防错一句话}}

\textbf{只有过焦点的弦才能用；两段相加才是整条弦；始终用与长轴的夹角。}

\end{summarybox}
