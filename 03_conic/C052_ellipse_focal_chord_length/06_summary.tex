\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{焦点弦长公式：已知倾斜角直接求过焦点的弦长，避免联立方程组。}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（椭圆标准）：} 焦点在 $x$ 轴上的椭圆，$a>b>0$。\n
		\item \textbf{条件 2（直线条件）：} 弦过椭圆焦点，且倾斜角 $\theta$ 为直线与 $x$ 轴正方向夹角。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（公式前提）：} 检查椭圆焦点位置和直线是否过焦点；分母是 $a^2-c^2\cos^2\theta$，注意 $\cos^2\theta$。\n
		\item \textbf{检查项（通径检验）：} 令 $\theta=90^\circ$ 得 $L=2b^2/a$，可验证公式是否正确。\n
	\end{itemize}
	\n
\end{summarybox}
