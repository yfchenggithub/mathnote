	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（椭圆方程）：}}]
			焦点在 $x$ 轴上的椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$（$a>b>0$），半焦距 $c=\sqrt{a^2-b^2}$。\n
		\item[\textbf{\textcolor{CStatement}{条件2（焦点弦）：}}]
			弦过椭圆的一个焦点，且弦所在直线的倾斜角为 $\theta$（与 $x$ 轴正方向夹角）。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（焦点弦长）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点弦长由半长轴、半短轴和倾斜角决定。\n
			\[
				\n L = \frac{2ab^2}{a^2 - c^2\cos^2\theta},\n
			\]
			\n 其中 $c = \sqrt{a^2-b^2}$。\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（通径特例）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			当弦与 $x$ 轴垂直时，弦长等于通径长。\n
			\[
				\n L = \frac{2b^2}{a}.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
