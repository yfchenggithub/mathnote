\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（抛物线标准形）：}}]
			抛物线方程为$y^2 = 2px$，其中$p>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（弦过焦点）：}}]
			弦$AB$经过焦点$F\left(\frac{p}{2},0\right)$，且$A$、$B$均在抛物线上。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（倒数之和）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			抛物线焦点弦被焦点分成的两段长度倒数之和为常数。
			\[
				\frac{1}{|AF|} + \frac{1}{|BF|} = \frac{2}{p}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（倾斜角形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点弦长可由倾斜角$\theta$表示，倒数和仍为$\frac{2}{p}$。
			\[
				|AF| = \frac{p}{1 - \cos\theta}, \quad |BF| = \frac{p}{1 + \cos\theta},\quad \frac{1}{|AF|} + \frac{1}{|BF|} = \frac{2}{p}.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{适用范围（标准抛物线）：}}
	该结论仅适用于抛物线$y^2=2px\,(p>0)$，且弦$AB$必须过焦点$F$。
\end{statementbox}
