\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	抛物线$y^2=4x$的焦点为$F$，过$F$的弦$AB$满足$|AF|=3$，求$|BF|$的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			将$y^2=4x$与标准式$y^2=2px$对比，得$2p=4$，$p=2$。
			\[
				2p=4 \Rightarrow p=2.
			\]
			\par\textbf{理由：} 直接对比系数。
		\item[\textbf{第二步（套用结论）：}]
			由$\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{p}$，代入$p=2$，$|AF|=3$，得
			\[
				\frac{1}{3}+\frac{1}{|BF|}=1.
			\]
			\par\textbf{理由：} 结论将$p$和$|AF|$关联。
		\item[\textbf{第三步（解方程）：}]
			解$\frac{1}{|BF|}=1-\frac{1}{3}=\frac{2}{3}$，所以$|BF|=\frac{3}{2}$。
			\[
				\frac{1}{|BF|}=\frac{2}{3} \;\Rightarrow\; |BF|=\frac32.
			\]
			\par\textbf{理由：} 取倒数即可。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|BF|=\frac{3}{2}$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	抛物线$y^2=8x$的焦点为$F$，过$F$的弦$AB$的倾斜角为$60^\circ$，求$\frac{1}{|AF|}+\frac{1}{|BF|}$的值以及$|AB|$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由$y^2=8x$得$2p=8$，$p=4$。
			\[
				2p=8 \Rightarrow p=4.
			\]
			\par\textbf{理由：} 对比标准式。
		\item[\textbf{第二步（求倒数和）：}]
			由结论，$\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{p}=\frac{2}{4}=\frac12$。
			\[
				\frac{1}{|AF|}+\frac{1}{|BF|}=\frac12.
			\]
			\par\textbf{理由：} 结论与倾斜角无关。
		\item[\textbf{第三步（倾斜角求长）：}]
			由倾斜角形式，$|AF|=\frac{p}{1-\cos\theta}$，$|BF|=\frac{p}{1+\cos\theta}$，其中$\theta=60^\circ$，$\cos60^\circ=\frac12$。
			\[
				|AF|=\frac{4}{1-\frac12}=8,\quad |BF|=\frac{4}{1+\frac12}=\frac{8}{3}.
			\]
			因此$|AB|=|AF|+|BF|=8+\frac{8}{3}=\frac{32}{3}$。
			\par\textbf{理由：} 也可由焦点弦长公式$|AB|=\frac{2p}{\sin^2\theta}$直接计算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\frac{1}{|AF|}+\frac{1}{|BF|}=\frac12$}，\textbf{$|AB|=\frac{32}{3}$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线$y^2=4x$的焦点为$F$，过$F$的弦$AB$满足$|AF|:|BF|=3:1$，求$\frac{1}{|AF|}+\frac{1}{|BF|}$的值以及$|AB|$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由$y^2=4x$得$p=2$，所以$\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{p}=1$。
			\[
				\frac{1}{|AF|}+\frac{1}{|BF|}=1.
			\]
			\par\textbf{理由：} 直接由结论得倒数和定值。
		\item[\textbf{第二步（引入参数）：}]
			由比例$|AF|:|BF|=3:1$，设$|AF|=3t$，$|BF|=t$（$t>0$）。
			\par\textbf{理由：} 引入参数简化。
		\item[\textbf{第三步（求解参数）：}]
			代入倒数和$\frac{1}{3t}+\frac{1}{t}=1$，即$\frac{1}{3t}+\frac{3}{3t}=\frac{4}{3t}=1$，解得$t=\frac{4}{3}$。从而$|AF|=4$，$|BF|=\frac{4}{3}$，$|AB|=|AF|+|BF|=4+\frac{4}{3}=\frac{16}{3}$。
			\[
				\frac{1}{3t}+\frac{1}{t}=1 \Rightarrow \frac{4}{3t}=1 \Rightarrow t=\frac43.
			\]
			\par\textbf{理由：} 解分式方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\frac{1}{|AF|}+\frac{1}{|BF|}=1$}，\textbf{$|AB|=\frac{16}{3}$}
\end{examplebox}
