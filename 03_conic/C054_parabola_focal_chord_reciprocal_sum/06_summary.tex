\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{抛物线焦点弦被焦点分成的两段长度倒数之和恒为$\frac{2}{p}$，与弦的倾斜角无关。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（抛物线标准式）：} 抛物线方程必须为$y^2=2px$（$p>0$）。
		\item \textbf{条件2（弦过焦点）：} 弦$AB$必须经过焦点$F$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（通径验证）：} 在使用结论时，务必确保已考虑斜率不存在的情况（通径）。
		\item \textbf{检查项（参数正负）：} 仅当$p>0$时结论成立，注意抛物线开口方向。
	\end{itemize}

\end{summarybox}
