\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $C:\frac{x^2}{16}+\frac{y^2}{9}=1$，$F_1,F_2$ 分别为左、右焦点，过 $F_1$ 的弦 $PQ$ 与 $F_2$ 构成三角形，求 $\triangle PF_2Q$ 的周长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定长半轴）：}]
		      由标准方程知 $a^2=16$，故 $a=4$。
		      \par\textbf{理由：} 椭圆标准方程中，长半轴 $a$ 是较大分母的平方根。

		\item[\textbf{第二步（套用结论）：}]
		      由焦点弦三边长度和定值结论，
		      \[
			      \begin{aligned}
				      C_{\triangle PF_2Q} & = 4a         \\
				                          & = 4 \times 4 \\
				                          & = 16.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 过 $F_1$ 的弦与另一焦点 $F_2$ 构成的三边长度和恒为 $4a$。

		\item[\textbf{第三步（得出答案）：}]
		      所以 $\triangle PF_2Q$ 的周长为 $16$。
		      \par\textbf{理由：} 代入 $a=4$ 即可。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$16$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\ (a>b>0)$ 的焦点为 $F_1,F_2$，过 $F_1$ 的弦 $PQ$ 与 $F_2$ 构成的三边长度和为 $20$，求椭圆的长半轴 $a$ 及长轴长 $2a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列出等式）：}]
		      由已知，三边长度和为 $20$，即
		      \[
			      C_{\triangle PF_2Q}=20.
		      \]
		      \par\textbf{理由：} 题目给出了焦点弦与另一焦点构成的三边长度和。

		\item[\textbf{第二步（套用结论）：}]
		      由椭圆焦点弦三边长度和定值结论知
		      \[
			      C_{\triangle PF_2Q}=4a,
		      \]
		      故
		      \[
			      4a=20.
		      \]
		      \par\textbf{理由：} 过焦点的弦与另一焦点构成的三边长度和恒为 $4a$。

		\item[\textbf{第三步（解出参数）：}]
		      解得
		      \[
			      a=5,
		      \]
		      所以
		      \[
			      2a=10.
		      \]
		      \par\textbf{理由：} $a$ 是长半轴，$2a$ 才是长轴长。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$a=5$}，长轴长为 \textbf{$10$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 椭圆 $E$ 的方程为 $9x^2+25y^2=225$，$F_1,F_2$ 为其左、右焦点，过 $F_1$ 的直线 $l$ 交 $E$ 于 $P,Q$ 两点，求 $\triangle PF_2Q$ 的周长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化为标准式）：}]
		      将 $9x^2+25y^2=225$ 化为
		      \[
			      \frac{x^2}{25}+\frac{y^2}{9}=1,
		      \]
		      故 $a^2=25$，$a=5$。
		      \par\textbf{理由：} 椭圆标准形式为 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\ (a>b>0)$。

		\item[\textbf{第二步（判断焦点与弦）：}]
		      由于 $25>9$，焦点在 $x$ 轴上，故 $F_1(-c,0)$，$F_2(c,0)$。直线 $l$ 过 $F_1$，满足焦点弦条件。
		      \par\textbf{理由：} 椭圆长轴在 $x$ 轴上，所以左焦点为 $F_1$。

		\item[\textbf{第三步（套用结论）：}]
		      由焦点弦三边长度和为 $4a$，得
		      \[
			      \begin{aligned}
				      C_{\triangle PF_2Q} & = 4 \times 5 \\
				                          & = 20.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 过 $F_1$ 的弦与另一焦点 $F_2$ 构成的三边长度和恒为 $4a$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$20$}。
\end{examplebox}
