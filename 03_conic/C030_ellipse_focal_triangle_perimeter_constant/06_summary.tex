\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	过椭圆焦点的弦与另一焦点构成的三边长度和恒为
	\[
		4a,
	\]
	其中 $a$ 是椭圆的长半轴。若三点不共线，它就是普通三角形周长；若三点共线，则按退化三角形的三边长度和理解。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准椭圆）：} 椭圆方程为
		      \[
			      \frac{x^2}{a^2}+\frac{y^2}{b^2}=1\quad (a>b>0)
		      \]
		      或可化为该形式，此时焦点在 $x$ 轴上。

		\item \textbf{条件 2（焦点弦）：} 弦 $PQ$ 过焦点 $F_1$，另一焦点 $F_2$ 参与构成三边长度和。

		\item \textbf{条件 3（方向推广）：} 若椭圆长轴在 $y$ 轴上，例如
		      \[
			      \frac{x^2}{b^2}+\frac{y^2}{a^2}=1\quad (a>b>0),
		      \]
		      则焦点应写为 $F_1(0,-c)$、$F_2(0,c)$，证明完全相同，结论仍为 $4a$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（弦长分解）：} 务必利用 $F_1$ 在线段 $PQ$ 上，将
		      \[
			      |PQ|
		      \]
		      写成
		      \[
			      |PF_1|+|QF_1|.
		      \]

		\item \textbf{检查项（焦点位置）：} 确认焦点在椭圆长轴上，且弦确实过焦点，而不是过一般点。

		\item \textbf{检查项（参数含义）：} $a$ 是长半轴，$2a$ 是长轴长；已知周长 $C$ 只能直接得到
		      \[
			      a=\frac{C}{4},
		      \]
		      不能单独由周长确定离心率。

		\item \textbf{检查项（退化情形）：} 若 $PQ$ 是长轴弦，则 $P,F_2,Q$ 三点共线，不是普通三角形，但三边长度和仍为 $4a$。
	\end{itemize}
\end{summarybox}
