\begin{statementbox}
\textbf{条件：}双曲线 $C:\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1$，$a,b>0$；
$A_1(-a,0),A_2(a,0)$ 为\textbf{实轴端点（不是焦点）}。
取 $P(x_0,y_0)\in C\setminus\{A_1,A_2\}$，则 $|x_0|>a$、$y_0\ne0$，两斜率均存在。
\begin{center}\input{assets/tikz/C065_statement_overview.tex}\end{center}
\textbf{斜率积定值：}记 $k_1=k_{PA_1},\ k_2=k_{PA_2}$，则
\[
\boxed{k_1k_2=\frac{b^2}{a^2}=e^2-1>0},\qquad
\boxed{e=\sqrt{1+k_1k_2}}.
\]
其中 $c^2=a^2+b^2$，$e=c/a>1$；定值不随动点的位置改变。

\textbf{逆向轨迹：}固定两端点，给定 $m>0$，当两斜率均存在时，
\[
 k_1k_2=m\iff \frac{x^2}{a^2}-\frac{y^2}{a^2m}=1\quad(x\ne\pm a).
\]
轨迹须去掉两个顶点。

\textbf{纵轴型：}若 $\dfrac{y^2}{a^2}-\dfrac{x^2}{b^2}=1$，
定点取 $B_1(0,-a),B_2(0,a)$，非顶点处
\[
\boxed{k_{PB_1}k_{PB_2}=\frac{a^2}{b^2}=\frac1{e^2-1}}.
\]

\medskip
\textbf{从圆衔接的统一模型（拓展）：}固定同样的 $A_1,A_2$，若两斜率均有定义，
令 $k_{PA_1}k_{PA_2}=m$，则
\[
\boxed{y^2=m(x^2-a^2)}\qquad(x\ne\pm a).
\]
$m<0$ 为椭圆（$m=-1$ 时是以 $A_1A_2$ 为直径的圆），
$m>0$ 为双曲线，$m=0$ 为去掉两定点的 $x$ 轴；
以上轨迹均需\textbf{排除 $A_1,A_2$}。
椭圆若 $m\ne-1$，其长轴方向并不一定沿 $x$ 轴。

\end{statementbox}
