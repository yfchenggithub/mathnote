	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（标准双曲线）：}}]
			焦点在$x$轴上的标准双曲线 $\dfrac{x^{2}}{a^{2}}-\dfrac{y^{2}}{b^{2}}=1$。
		\item[\textbf{\textcolor{CStatement}{条件2（实轴端点）：}}]
			设 $A_{1}(-a,0),\;A_{2}(a,0)$。
		\item[\textbf{\textcolor{CStatement}{条件3（P点限制）：}}]
			$P$ 为双曲线上异于 $A_{1},A_{2}$ 的任意一点。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（斜率积定值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			双曲线上任意一点到两个实轴端点连线的斜率之积为固定正值。
			\[
				k_{PA_{1}}\cdot k_{PA_{2}} = \frac{b^{2}}{a^{2}}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（离心率形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将定值用离心率 $e$ 表示，更便于与离心率综合题衔接。
			\[
				k_{PA_{1}}\cdot k_{PA_{2}} = e^{2}-1
			\]
	\end{description}

\end{statementbox}
