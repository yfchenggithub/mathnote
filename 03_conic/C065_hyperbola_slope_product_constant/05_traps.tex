\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（符号记忆）}}]
			误认为 $k_{PA_1}k_{PA_2} = -\dfrac{b^2}{a^2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号为正）：}}]
			双曲线第三定义结果为正值 $\dfrac{b^2}{a^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（负号混淆）：}}]
			与椭圆第三定义 $-\dfrac{b^2}{a^2}$ 混淆；双曲线离心率 $e>1$，故 $b^2/a^2 = e^2-1 >0$，应为正。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（端点纳入）}}]
			计算时忽略“$P$ 异于 $A_1,A_2$”的条件，直接将 $P$ 取为端点 $A_1$ 或 $A_2$ 导致分母为零。
		\item[\textbf{\textcolor{CTrap}{正确理解（排除端点）：}}]
			当 $P$ 与 $A_1$ 或 $A_2$ 重合时，斜率不存在，结论不适用，必须满足 $x_P\neq \pm a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）：}}]
			没有仔细审题，忽略了 $P$ 的限制条件，直接套用公式得错误结果。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（方程形式）}}]
			对于焦点在 $y$ 轴的双曲线 $\dfrac{y^2}{a^2} - \dfrac{x^2}{b^2}=1$，仍套用 $k_{PA_1}k_{PA_2} = \dfrac{b^2}{a^2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分形式）：}}]
			此类双曲线实轴端点为 $(0,\pm a)$，利用相同方法推导得 $k_{PA_1}k_{PA_2} = \dfrac{a^2}{b^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（位置颠倒）：}}]
			焦点轴变化导致 $a$ 与 $b$ 的地位互换，应准确记忆不同标准方程下的结论。
	\end{description}

\end{trapbox}
