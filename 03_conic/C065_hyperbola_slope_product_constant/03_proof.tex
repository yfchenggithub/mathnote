\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	用坐标法：设$P(x_0,y_0)$，写出$k_{PA_1},k_{PA_2}$，然后利用$P$在双曲线上满足的方程消去$y_0$，最终得到与$x_0$无关的常数。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（斜率表示）：}]
			设$P(x_0,y_0)$，其中$x_0 \neq \pm a$（因为$P$异于$A_1,A_2$）。则
			\[
				k_{PA_1}=\frac{y_0}{x_0+a},
				\qquad
				k_{PA_2}=\frac{y_0}{x_0-a}.
			\]
			所以
			\[
				k_{PA_1}\cdot k_{PA_2} = \frac{y_0^2}{(x_0+a)(x_0-a)} = \frac{y_0^2}{x_0^2-a^2}.
			\]
			\par\textbf{理由：} 直接利用斜率公式$k = \dfrac{y_2-y_1}{x_2-x_1}$，整理分母为平方差。
		\item[\textbf{步骤二（曲线代换）：}]
			因为$P$在双曲线上，所以$\dfrac{x_0^2}{a^2} - \dfrac{y_0^2}{b^2} = 1$。移项得
			\[
				\frac{y_0^2}{b^2} = \frac{x_0^2}{a^2} - 1 = \frac{x_0^2-a^2}{a^2},
			\]
			即
			\[
				y_0^2 = \frac{b^2}{a^2}(x_0^2-a^2).
			\]
			\par\textbf{理由：} 利用双曲线标准方程解出$y_0^2$，为下一步代入做准备。
		\item[\textbf{步骤三（代入化简）：}]
			将$y_0^2$的表达式代入步骤一中的斜率积：
			\[
				k_{PA_1}\cdot k_{PA_2} = \frac{\frac{b^2}{a^2}(x_0^2-a^2)}{x_0^2-a^2} = \frac{b^2}{a^2}.
			\]
			\par\textbf{理由：} 这里$x_0^2-a^2$是公因子，直接约去（$x_0\neq\pm a$保证分母非零），即得定值。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{由此证得：对于任意异于$A_1,A_2$的点$P$，总有$k_{PA_1}\cdot k_{PA_2} = \frac{b^2}{a^2}$。若用离心率$e$表示，则$\frac{b^2}{a^2} = e^2-1$，因此$k_{PA_1}k_{PA_2} = e^2-1$。}
\end{proofbox}
