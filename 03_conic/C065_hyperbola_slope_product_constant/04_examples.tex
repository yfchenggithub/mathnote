\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $C: \frac{x^2}{4} - \frac{y^2}{5} = 1$，实轴端点为 $A_1(-2,0), A_2(2,0)$，点 $P$ 为 $C$ 上一点（异于 $A_1,A_2$），求 $k_{PA_1}\cdot k_{PA_2}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由双曲线方程得 $a^2=4$, $b^2=5$。
			\par\textbf{理由：} 将方程与标准形式 $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$ 比对即可。
		\item[\textbf{第二步（使用结论）：}]
			由双曲线第三定义，直接得 $k_{PA_1}\cdot k_{PA_2} = \frac{b^2}{a^2}$。
			\par\textbf{理由：} 该结论即为本课内容，适用于焦点在 $x$ 轴的标准双曲线。
		\item[\textbf{第三步（代入计算）：}]
			代入 $a^2=4$, $b^2=5$，得 $k_{PA_1}\cdot k_{PA_2} = \frac{5}{4}$。
			\par\textbf{理由：} 只需将数值代入分数即得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$k_{PA_1}\cdot k_{PA_2} = \frac{5}{4}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $C: \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$（$a>0,b>0$）的实轴端点为 $A_1(-a,0),A_2(a,0)$，点 $P$ 在 $C$ 上且异于 $A_1,A_2$，若 $k_{PA_1}\cdot k_{PA_2}=3$，求 $C$ 的离心率 $e$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（乘积转化）：}]
			由双曲线第三定义，得 $\frac{b^2}{a^2}=3$。
			\par\textbf{理由：} 第三定义：$k_{PA_1}k_{PA_2}=\frac{b^2}{a^2}$，代入已知值得。
		\item[\textbf{第二步（建立关系）：}]
			因为 $b^2=c^2-a^2$，所以 $\frac{c^2-a^2}{a^2}=3$，整理得 $c^2=4a^2$。
			\par\textbf{理由：} 双曲线参数关系 $c^2=a^2+b^2$ 及已知条件。
		\item[\textbf{第三步（求离心率）：}]
			$e=\frac{c}{a}$，由 $c^2=4a^2$ 得 $e=2$。
			\par\textbf{理由：} 离心率定义，取正值（$e>1$）。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$e=2$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线 $C: \frac{x^2}{4} - \frac{y^2}{5} = 1$ 的左右顶点为 $A_1(-2,0), A_2(2,0)$，点 $P$ 为 $C$ 上异于 $A_1,A_2$ 的一点，且直线 $PA_1$ 与 $PA_2$ 的斜率之和为 $3$，求点 $P$ 的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（定乘积）：}]
			由 $C:\frac{x^2}{4}-\frac{y^2}{5}=1$ 得 $a^2=4$, $b^2=5$。由双曲线第三定义，
			\[
				k_{PA_1}\cdot k_{PA_2} = \frac{b^2}{a^2} = \frac{5}{4}.
			\]
			\par\textbf{理由：} 直接应用第三定义结论。
		\item[\textbf{第二步（解斜率）：}]
			设 $k_1=k_{PA_1}$, $k_2=k_{PA_2}$，已知 $k_1+k_2=3$，$k_1k_2=\frac{5}{4}$。由韦达定理，$k_1,k_2$ 是二次方程 $t^2-3t+\frac{5}{4}=0$ 的两根。解方程得
			\[
				t=\frac{3\pm\sqrt{9-5}}{2}=\frac{3\pm2}{2},
			\]
			即 $t_1=\frac12$, $t_2=\frac52$。
			\par\textbf{理由：} 韦达定理及一元二次方程求根公式。
		\item[\textbf{第三步（求坐标）：}]
			不妨取 $k_1=\frac12$, $k_2=\frac52$。由斜率定义：
			\[
				k_1=\frac{y_0}{x_0+2}=\frac12,
				\qquad
				k_2=\frac{y_0}{x_0-2}=\frac52.
			\]
			由第一式得 $y_0=\frac{x_0+2}{2}$，代入第二式：
			\[
				\frac{(x_0+2)/2}{x_0-2}=\frac52\;\Longrightarrow\; \frac{x_0+2}{2(x_0-2)}=\frac52.
			\]
			交叉相乘得 $x_0+2=5(x_0-2)$，解得 $x_0=3$。进而 $y_0=\frac{3+2}{2}=\frac52$。检验：将 $P(3,\frac52)$ 代入 $C$ 方程，
			\[
				\frac{3^2}{4}-\frac{(5/2)^2}{5}=\frac94-\frac{25/4}{5}=\frac94-\frac54=1,
			\]
			符合。由对称性，另一解 $P(3,-\frac52)$ 对应 $k_1=\frac52,k_2=\frac12$（或互换）。
			\par\textbf{理由：} 利用斜率关系联立方程组，解出坐标并验证曲线方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{点 $P$ 的坐标为 $(3,\frac52)$ 或 $(3,-\frac52)$。}
\end{examplebox}
