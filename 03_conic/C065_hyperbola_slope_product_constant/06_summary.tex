\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{双曲线第三定义：双曲线上动点$P$与实轴端点$A_1,A_2$连线的斜率之积恒为$\dfrac{b^2}{a^2}$（即$e^2-1$）。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（标准方程）：} 双曲线必须是标准形式 $\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1$，$a>0,b>0$。
		\item \textbf{条件2（非端点）：} 点$P$在双曲线上且横坐标 $x_P \neq \pm a$（即异于实轴端点）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号为正）：} 结果为正值$b^2/a^2$，不要与椭圆的$-\dfrac{b^2}{a^2}$混淆。
		\item \textbf{检查项（焦点位置）：} 若双曲线焦点在$y$轴，结论变为$\dfrac{a^2}{b^2}$，使用前需确认标准方程。
	\end{itemize}

\end{summarybox}
