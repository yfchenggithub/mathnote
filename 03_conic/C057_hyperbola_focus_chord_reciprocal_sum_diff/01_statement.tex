\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（标准双曲线）：}}] 双曲线 $\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1\ (a>0,b>0)$，焦点 $F(\pm c,0)$，其中 $c=\sqrt{a^2+b^2}$。
		\item[\textbf{\textcolor{CStatement}{条件2（焦点弦）：}}] 过焦点 $F$ 的弦 $AB$ 与双曲线相交。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（同支情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同一支上焦点弦的倒数之和为常数。
			\[
				\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2a}{b^2}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（异支情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两支上焦点弦的倒数之差的绝对值为同一常数。
			\[
				\left|\frac{1}{|AF|}-\frac{1}{|BF|}\right|=\frac{2a}{b^2}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（离心率表达）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			常数用离心率 $e$ 表示，结论形式更统一。
			\[
				\frac{2a}{b^2} = \frac{2}{a(e^2-1)},\quad e=\frac{c}{a},\ b^2=a^2(e^2-1).
			\]
			因此同支情形可写为 $\displaystyle\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{a(e^2-1)}$，异支情形可写为 $\displaystyle\left|\frac{1}{|AF|}-\frac{1}{|BF|}\right|=\frac{2}{a(e^2-1)}$。
	\end{description}
\end{statementbox}
