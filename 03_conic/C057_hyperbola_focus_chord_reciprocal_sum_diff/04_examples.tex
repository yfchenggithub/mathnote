\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	双曲线 $\frac{x^2}{9}-\frac{y^2}{16}=1$ 的右焦点为 $F$，过 $F$ 的弦 $AB$ 与右支交于 $A,B$，且 $|AF|=3$，求 $|BF|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（使用定理）：}] 因为 $A,B$ 在同一支，由焦点分弦定理有 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2a}{b^2}$。对本题 $a=3,b^2=16$，故 $\frac{2a}{b^2}=\frac{6}{16}$，即 $\frac{3}{8}$。
			\par\textbf{理由：} 直接套用同支公式。
		\item[\textbf{第二步（代入已知）：}] 将 $|AF|=3$ 代入得 $\frac{1}{3}+\frac{1}{|BF|}=\frac{3}{8}$。
			\par\textbf{理由：} 建立关于 $|BF|$ 的方程。
		\item[\textbf{第三步（求解长度）：}] 移项得
			\[
				\begin{aligned}
					\frac{1}{|BF|}
					&= \frac{3}{8}-\frac{1}{3} \\
					&= \frac{9}{24}-\frac{8}{24} \\
					&= \frac{1}{24},
			\end{aligned}
			\]
			因此 $|BF|=24$。
			\par\textbf{理由：} 解一次方程即可。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{|BF|=24}$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	双曲线 $\frac{x^2}{4}-\frac{y^2}{16}=1$ 的右焦点为 $F$，过 $F$ 的弦 $AB$ 分别交双曲线于左、右两支，且 $A$ 在右支，$|AF|=3$，求 $|BF|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判断公式）：}] 由 $A,B$ 分别在两支，应用异支公式 $\left|\frac{1}{|AF|}-\frac{1}{|BF|}\right|=\frac{2a}{b^2}$。本题 $a=2,b^2=16$，故 $\frac{2a}{b^2}=\frac{4}{16}$，即 $\frac{1}{4}$。
			\par\textbf{理由：} 注意异支情况使用差的绝对值。
		\item[\textbf{第二步（列方程）：}] 结合图形，右支上的点 $A$ 离右焦点更近，左支上的点 $B$ 更远，故 $|AF|<|BF|$，从而 $\frac{1}{|AF|}>\frac{1}{|BF|}$，所以 $\frac{1}{|AF|}-\frac{1}{|BF|}=\frac{1}{4}$。代入 $|AF|=3$ 得 $\frac{1}{3}-\frac{1}{|BF|}=\frac{1}{4}$。
			\par\textbf{理由：} 利用几何关系确定差的符号，去掉绝对值。
		\item[\textbf{第三步（求解长度）：}] 整理得
			\[
				\begin{aligned}
					\frac{1}{|BF|}
					&= \frac{1}{3}-\frac{1}{4} \\
					&= \frac{4}{12}-\frac{3}{12} \\
					&= \frac{1}{12},
			\end{aligned}
			\]
			故 $|BF|=12$。
			\par\textbf{理由：} 代数求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{|BF|=12}$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	双曲线 $\frac{x^2}{9}-\frac{y^2}{16}=1$ 的右焦点 $F$，过 $F$ 的直线 $l$ 与双曲线右支交于 $A,B$ 两点，且 $|AF|=2|BF|$。求直线 $l$ 的斜率 $k$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（利用定理求长度）：}] 由同支公式 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2a}{b^2}=\frac{3}{8}$。设 $|BF|=t$，则 $|AF|=2t$，代入得 $\frac{1}{2t}+\frac{1}{t}=\frac{3}{2t}$，于是 $\frac{3}{2t}=\frac{3}{8}$，解得 $t=4$。故 $|AF|=8,\ |BF|=4$。
			\par\textbf{理由：} 先利用已知比例和定理确定两段的具体长度。
		\item[\textbf{第二步（焦半径公式求横坐标）：}] 双曲线中 $a=3,b=4,c=5,e=\frac{5}{3}$。对于右支上的点 $P$，$|FP|=ex_P-a$。因此
			\[
				\begin{aligned}
					|AF| &= \frac{5}{3}x_A-3 \\
					&= 8 \;\Rightarrow\; x_A=\frac{33}{5},\\
					|BF| &= \frac{5}{3}x_B-3 \\
					&= 4 \;\Rightarrow\; x_B=\frac{21}{5}.
			\end{aligned}
			\]
			\par\textbf{理由：} 焦半径公式将焦点弦长度转化为横坐标。
		\item[\textbf{第三步（求纵坐标与斜率）：}] 将 $x_A,x_B$ 代入双曲线方程：
			\[
				\begin{aligned}
					\frac{x_A^2}{9}-\frac{y_A^2}{16} &= 1 \\
					\Rightarrow\; y_A^2 &= 16\left(\frac{x_A^2}{9}-1\right) \\
					&= \frac{13824}{225} \\
					&= \frac{1536}{25},
			\end{aligned}
			\]
			\[
				\begin{aligned}
					\frac{x_B^2}{9}-\frac{y_B^2}{16} &= 1 \\
					\Rightarrow\; y_B^2 &= 16\left(\frac{x_B^2}{9}-1\right) \\
					&= \frac{3456}{225} \\
					&= \frac{384}{25}.
			\end{aligned}
			\]
			开方得 $|y_A|=\frac{16\sqrt{6}}{5}$，$|y_B|=\frac{8\sqrt{6}}{5}$。由 $A,F,B$ 共线，向量 $\overrightarrow{FA}$ 与 $\overrightarrow{FB}$ 成比例，即 $(x_A-5):(x_B-5)=y_A:y_B$，计算得 $y_A:y_B=-2:1$，故 $y_A$ 与 $y_B$ 异号。取 $y_A=\frac{16\sqrt{6}}{5},\ y_B=-\frac{8\sqrt{6}}{5}$（或相反符号），则斜率
			\[
				\begin{aligned}
					k &= \frac{y_A-y_B}{x_A-x_B} \\
					&= \frac{\frac{16\sqrt{6}}{5}-\left(-\frac{8\sqrt{6}}{5}\right)}{\frac{33}{5}-\frac{21}{5}} \\
					&= \frac{\frac{24\sqrt{6}}{5}}{\frac{12}{5}} \\
					&= 2\sqrt{6}.
			\end{aligned}
			\]
			同理取相反符号得 $k=-2\sqrt{6}$。故所求斜率为 $\pm 2\sqrt{6}$。
			\par\textbf{理由：} 利用双曲线方程解出纵坐标，再由斜率公式计算，注意纵坐标的符号需满足共线条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	直线 $l$ 的斜率为 $\boxed{\pm 2\sqrt{6}}$。
\end{examplebox}
