% C057 | 对应条件、代数、几何、可行性四类检查
\begin{trapbox}
  \begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.5em]
    \item[\textbf{\textcolor{CTrap}{易错1：没判同支异支就背公式。}}]
    \textbf{改法：} 看 $x_Ax_B$ 的符号（标准左右开口双曲线）：$x_Ax_B>0$ 同支，\textbf{倒数相加}；$x_Ax_B<0$ 异支，\textbf{倒数相减并取绝对值}。统一常数 $2a/b^2$。

    \item[\textbf{\textcolor{CTrap}{易错2：以为“异支焦点弦”必须把焦点夹在中间。}}]
    \textbf{改法：} 对右焦点而言，同支顺序 $A-F-B$，异支顺序 $F-A-B$（$A$ 在右支）。\textbf{异支时 $F$ 在 $AB$ 的延长线上}；这里“过焦点”说的是\textbf{所在直线}，不是线段内部。

    \item[\textbf{\textcolor{CTrap}{易错3：异支去绝对值，盲目决定符号。}}]
    \textbf{改法：} 若 $F=F_2(c,0)$，$A$ 在右支、$B$ 在左支，\textbf{严格成立}
    $\dfrac1{|AF|}-\dfrac1{|BF|}=\dfrac{2a}{b^2}>0$，即 $|AF|<|BF|$；换成左焦点时同理按\textbf{焦点所在支较近}排列。不明确标签时保留绝对值。

    \item[\textbf{\textcolor{CTrap}{易错4：右焦点的焦半径公式对左右两支都写成 $ex-a$。}}]
    \textbf{改法：} 右焦点必须用 $|PF|=|ex-a|$；右支取 $ex-a$，左支取 $a-ex$。不是所有点都能去掉绝对值。

    \item[\textbf{\textcolor{CTrap}{易错5：把双曲线写成椭圆的 $b^2=a^2-c^2$。}}]
    \textbf{改法：} 双曲线 $c^2=a^2+b^2$，所以
    $\dfrac{2a}{b^2}=\dfrac2{a(e^2-1)}$；$b$ 是虚半轴长，别与椭圆混淆。

    \item[\textbf{\textcolor{CTrap}{易错6：只要直线过焦点，就断言必有两交点。}}]
    \textbf{改法：} 直线 $y=k(x-c)$ 若 $k=\pm b/a$，联立后的二次项为零，\textbf{只有一个有限交点}；斜率不存在的 $x=c$ 则有\textbf{两个同支交点}，不能一并排除。

    \item[\textbf{\textcolor{CTrap}{易错7：公式算出负长度仍继续解。}}]
    \textbf{改法：} 设 $K=2a/b^2$。同支若已知 $r_A$，另一段满足
    $\dfrac1{r_B}=K-\dfrac1{r_A}>0$，所以\textbf{必要条件}是 $r_A>1/K=b^2/(2a)$。等号对应趋向无穷远的极限，而不是有限交点。

    \item[\textbf{\textcolor{CTrap}{易错8：常规韦达中把 $ex-a$ 的有符号值当距离。}}]
    \textbf{改法：} 可以先算 $u_i=ex_i-a$ 的代数倒数之和；\textbf{最后}再依支别还原 $r_i=|u_i|$。右支 $u_i>0$，左支 $u_i<0$，否则异支的“差”会推错。


    \item[\textbf{\textcolor{CTrap}{易错9：左支点必须用左焦点，右支点必须用右焦点。}}]
    \textbf{改法：} \textbf{不是点和焦点配对，而是焦点和准线配对。}
    双曲线上任意点 $P$（左支或右支）都同时满足
    $|PF_1|=e\,d(P,\ell_1)$ 和 $|PF_2|=e\,d(P,\ell_2)$；
    其中 $\ell_1:x=-a^2/c$、$\ell_2:x=a^2/c$。
    例如 C057 选右焦点 $F_2$，左支交点也能统一使用\textbf{右准线}。

    \item[\textbf{\textcolor{CTrap}{易错10：焦点与准线交叉配对，或把距离比写成 $1$。}}]
    \textbf{改法：}必须\textbf{左焦点配左准线，右焦点配右准线}，
    $\dfrac{|PF_i|}{d(P,\ell_i)}=e=\dfrac ca>1$，\textbf{不是 $1$}。
    若在左右支展开 $|PF_2|=|ex-a|$，只改变绝对值的符号，不改变所选准线。

  \end{description}\end{trapbox}\clearpage\begin{trapbox}\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.5em]

    \item[\textbf{\textcolor{CTrap}{易错11：把直线倾斜角 $\alpha$ 当作射线方向角 $\varphi$。}}]
    \textbf{改法：}倾斜角 $\alpha\in[0,\pi)$ 描述的是\textbf{整条直线}；
    从焦点 $F$ 出发到某交点的射线方向，可能是 $\alpha$，也可能是
    $\alpha+\pi$（模 $2\pi$）。例如水平线倾斜角 $0$，
    但从 $F(5,0)$ 指向 $(3,0)$ 或 $(-3,0)$ 的射线方向角均为 $\pi$。

    \item[\textbf{\textcolor{CTrap}{易错12：只看 $b^2/(a-c\cos\varphi)$，不检查分母符号。}}]
    \textbf{改法：}对\textbf{右焦点的右支交点}，才有
    $\displaystyle r=\frac{b^2}{a-c\cos\varphi}>0$，
    要求 $a-c\cos\varphi>0$。若交点在\textbf{左支}，
    必须换用 $\displaystyle r=-\frac{b^2}{a+c\cos\varphi}>0$，
    要求 $a+c\cos\varphi<0$。
    不能把错误公式的负结果直接取绝对值充当同一个点的焦半径。

    \item[\textbf{\textcolor{CTrap}{易错13：把有向参数 $t<0$ 理解成“负的点间距离”。}}]
    \textbf{改法：}设 $P=F+t(\cos\alpha,\sin\alpha)$，其中 $t\in\mathbb R$。
    负号只代表交点在选定方向的\textbf{反向}，而实际距离永远为
    $|FP|=|t|>0$。水平线 $F=(5,0)$ 取向右为正时，
    $t=-8$ 对应左顶点 $(-3,0)$，距离是 $8$，不是 $-8$。

    \item[\textbf{\textcolor{CTrap}{易错14：看到有向倒数式就直接把 $t_i$ 换成长度。}}]
    \textbf{改法：}若两个有限交点有向参数满足
    $\displaystyle\frac1{t_1}-\frac1{t_2}=\frac{2a}{b^2}$，
    则同支时 $t_1>0>t_2$，才变成\textbf{两段倒数相加}；
    异支时两个 $t_i$ 同号，才变成\textbf{较近段倒数减较远段倒数}。
    先看正负，再换成 $r_i=|t_i|$。
  \end{description}
\end{trapbox}
