\begin{examplebox}

	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=4x$ 的焦点为 $F$，过 $F$ 的直线交抛物线于 $A,B$．若 $A$ 的纵坐标为 $-2$，求 $B$ 的纵坐标及横坐标．

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（定参数）：}] 由 $y^2=4x$ 得 $2p=4$，所以 $p=2$，焦点 $F(1,0)$．
			\par\textbf{理由：} 将标准方程与 $y^2=2px$ 对比，找出 $p$ 值．
		\item[\textbf{第二步（纵积结论）：}] 由焦点弦性质 $y_1y_2 = -p^2$，代入 $p=2$ 得 $y_1y_2 = -4$．再代入 $y_1=-2$，得 $y_2=2$．
			\par\textbf{理由：} 直接使用 $y_1y_2=-p^2$，无需联立．
		\item[\textbf{第三步（求横坐标）：}] 将 $y_2=2$ 代入 $y^2=4x$，得 $x_2 = \dfrac{y_2^2}{4}$，计算得 $x_2 = 1$．
			\par\textbf{理由：} 点在抛物线上，满足方程．
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$B(1,2)$}．

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=8x$ 的焦点为 $F$，过 $F$ 的弦 $AB$ 满足 $|AF|=5$，求 $|BF|$．

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（定参数）：}] 由 $y^2=8x$ 得 $2p=8$，即 $p=4$，焦点 $F(2,0)$．
			\par\textbf{理由：} 对比标准方程得出 $p$ 值．
		\item[\textbf{第二步（横积结论）：}] 由 $x_1x_2 = \dfrac{p^2}{4}$，代入 $p=4$ 得 $x_1x_2 = 4$．
			\par\textbf{理由：} 直接套用焦点弦横坐标积结论．
		\item[\textbf{第三步（焦半径公式）：}] 焦点半径公式：$|AF| = x_1 + \dfrac{p}{2}$，代入 $p=4$ 得 $x_1+2=5$，所以 $x_1=3$．
			\par\textbf{理由：} 抛物线上的点到焦点距离等于横坐标加 $\dfrac{p}{2}$．
		\item[\textbf{第四步（求另一横坐标）：}] 由 $x_1x_2=4$ 及 $x_1=3$，得 $x_2 = \dfrac{4}{3}$．
			\par\textbf{理由：} 代入横坐标积结论求解．
		\item[\textbf{第五步（求 $|BF|$）：}] 由焦半径公式 $|BF| = x_2 + \dfrac{p}{2}$，代入得 $|BF| = \dfrac{10}{3}$．
			\par\textbf{理由：} 再次使用焦半径公式．
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$|BF| = \dfrac{10}{3}$}．

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=4x$ 的焦点为 $F$，过 $F$ 的直线 $l$ 与抛物线交于 $A(x_1,y_1)$，$B(x_2,y_2)$．求证：过 $A,B$ 两点的切线互相垂直，并求两切线交点的坐标．

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设线联立）：}] 设直线 $l: x=my+1$（$m$ 为实数），代入 $y^2=4x$ 得
			\[
				y^2 = 4(my+1) \;\Longrightarrow\; y^2 - 4my - 4 = 0.
			\]
			由韦达定理得 $y_1+y_2 = 4m$，$y_1y_2 = -4$．
			\par\textbf{理由：} 焦点 $F(1,0)$，直线方程避开斜率分类，联立后得到纵坐标关系．
		\item[\textbf{第二步（切线斜率）：}] 抛物线在点 $A(x_1,y_1)$ 处的切线方程为 $y_1y = 2(x+x_1)$，斜率 $k_1 = \dfrac{2}{y_1}$；同理 $k_2 = \dfrac{2}{y_2}$．则
			\[
				\begin{aligned}
					k_1k_2 &= \frac{4}{y_1y_2} \\
					&= \frac{4}{-4} \\
					&= -1.
			\end{aligned}
			\]
			故两切线互相垂直．
			\par\textbf{理由：} 利用 $y_1y_2=-4$ 直接得到斜率积为 $-1$，完成垂直证明．
		\item[\textbf{第三步（交点纵坐标）：}] 联立两切线方程：
			\[
				\begin{cases}
					y_1y = 2(x+x_1),\\
					y_2y = 2(x+x_2).
			\end{cases}
			\]
			两式相减得 $(y_1-y_2)y = 2(x_1-x_2)$．由 $y^2=4x$ 得 $x_i = \dfrac{y_i^2}{4}$，故
			\[
				\begin{aligned}
					x_1-x_2 &= \frac{y_1^2-y_2^2}{4} \\
					&= \frac{(y_1-y_2)(y_1+y_2)}{4}.
			\end{aligned}
			\]
			代入得 $(y_1-y_2)y = \dfrac{(y_1-y_2)(y_1+y_2)}{2}$．因 $y_1\neq y_2$，约去得 $y = \dfrac{y_1+y_2}{2}$，由韦达定理 $y_1+y_2=4m$，故 $y = 2m$．
			\par\textbf{理由：} 通过代数运算将交点纵坐标用 $y_1+y_2$ 表示，利用韦达定理得出具体值．
		\item[\textbf{第四步（交点横坐标）：}] 将 $y = \dfrac{y_1+y_2}{2}$ 代入第一条切线方程：
			\[
				y_1\cdot\frac{y_1+y_2}{2} = 2(x+x_1).
			\]
			整理得
			\[
				\begin{aligned}
					2x &= \frac{y_1^2+y_1y_2}{2} - 2x_1 \\
					&= \frac{y_1^2+y_1y_2}{2} - \frac{y_1^2}{2} \quad (\text{利用 } x_1 = \frac{y_1^2}{4}) \\
					&= \frac{y_1y_2}{2}.
			\end{aligned}
			\]
			故
			\[
				\begin{aligned}
					x &= \frac{y_1y_2}{4} \\
					&= \frac{-4}{4} \\
					&= -1.
			\end{aligned}
			\]
			\par\textbf{理由：} 回代并利用 $y_1y_2=-4$ 得到横坐标为定值 $-1$，即准线方程．
		\item[\textbf{第五步（结论）：}] 因此过 $A,B$ 的切线互相垂直，交点坐标为 $(-1,\,2m)$，恒在准线 $x=-1$ 上．
			\par\textbf{理由：} 综合前四步得出最终结论．
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线垂直，交点 $(-1,2m)$ 在准线 $x=-1$ 上}．
