\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	设直线 $x=my+\dfrac{p}{2}$ 与抛物线 $y^2=2px$ 联立，利用韦达定理先得 $y_1y_2$，再通过 $x=\dfrac{y^2}{2p}$ 转化为 $x_1x_2$．

	\medskip

	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设线消参）：}] 设过焦点 $F(\frac{p}{2},0)$ 的直线 $l$ 的方程为 $x=my+\dfrac{p}{2}$（$m\in\mathbb{R}$），代入 $y^2=2px$．
			\[
				y^2 = 2p\left(my+\frac{p}{2}\right).
			\]
			\par\textbf{理由：} 这种设线方式可直接避开讨论斜率不存在的情况，且包含所有过焦点的直线．

		\item[\textbf{步骤二（化简方程）：}] 展开整理得
			\[
				y^2 = 2pmy + p^2 \;\Longrightarrow\; y^2 - 2pmy - p^2 = 0.
			\]
			\par\textbf{理由：} 该方程是关于 $y$ 的一元二次方程，其两根即为 $A,B$ 的纵坐标 $y_1,y_2$．

		\item[\textbf{步骤三（韦达定理）：}] 由韦达定理，
			\[
				y_1 + y_2 = 2pm,
				\qquad
				y_1y_2 = -p^2.
			\]
			\par\textbf{理由：} 二次项系数为 $1$，一次项系数为 $-2pm$，常数项为 $-p^2$，故两根之积为 $-p^2$，与 $m$ 无关．（注：$y_1y_2=-p^2$ 是核心定值．）

		\item[\textbf{步骤四（横积转化）：}] 利用抛物线方程 $y^2=2px$ 得 $x=\dfrac{y^2}{2p}$，从而
			\[
				\begin{aligned}
					x_1x_2 &= \frac{y_1^2}{2p}\cdot\frac{y_2^2}{2p} \\
					&= \frac{(y_1y_2)^2}{4p^2}.
			\end{aligned}
			\]
			代入 $y_1y_2=-p^2$，
			\[
				\begin{aligned}
					x_1x_2 &= \frac{(-p^2)^2}{4p^2} \\
					&= \frac{p^4}{4p^2} \\
					&= \frac{p^2}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用抛物线方程实现坐标转化，将 $x_1x_2$ 用 $y_1y_2$ 表示，再由已知 $y_1y_2$ 求得 $x_1x_2$．

		\item[\textbf{步骤五（回扣结论）：}] 联立已得
			\[
				x_1x_2 = \frac{p^2}{4},
				\qquad
				y_1y_2 = -p^2.
			\]
			\par\textbf{理由：} 两个定值同时成立，且与直线参数 $m$（即倾斜角）无关．
	\end{description}

	\medskip

	\textbf{\textcolor{CProof}{结论回扣}}

	焦点弦端点坐标乘积仅与抛物线焦准距 $p$ 有关，与弦的倾斜程度无关；这一结论省去了每次重新联立计算的时间，是抛物线中非常重要的二级结论．
\end{proofbox}
