\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	抛物线 $y^2=2px\ (p>0)$ 的焦点弦端点横坐标之积为 $\dfrac{p^2}{4}$，纵坐标之积为 $-p^2$．

	\medskip

	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准形式）：} 抛物线必须是 $y^2=2px\ (p>0)$，其他标准形式结论不同．
		\item \textbf{条件 2（焦点位置）：} 弦必须过焦点 $F\bigl(\frac{p}{2},0\bigr)$，且端点 $A,B$ 在抛物线上．
	\end{itemize}

	\medskip

	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号与形式）：} 注意 $y_1y_2=-p^2$ 中的负号，且结论仅适用于 $y^2=2px$．
		\item \textbf{检查项（直线设法）：} 使用 $x=my+\dfrac{p}{2}$ 设线可避免分类讨论，保证结论正确性．
	\end{itemize}
\end{summarybox}
