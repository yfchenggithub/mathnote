\begin{summarybox}
\textbf{\textcolor{CSummary}{15 秒快记}}\par\smallskip
\[
 \begin{gathered}
 y^2=2px\ (p>0),\quad F\!\left(\frac p2,0\right),\quad AB\text{ 过焦点},\\[3pt]
 \boxed{\ y_1y_2=-p^2\ },\qquad
 \boxed{\ x_1x_2=\frac{p^2}{4}\ },\\[3pt]
 \boxed{\ k_Ak_B=-1\ },\qquad
 \boxed{\ x_T=-\frac p2\ },\\[3pt]
 \boxed{\ \tan\alpha\tan\beta=4\ }
 \quad(\alpha=\angle AOF,\ \beta=\angle FOB).
 \end{gathered}
\]
\textbf{轴上定点弦的通式：}若 $P(a,0)$ 且 $a>0$，
过 $P$ 的弦端点分居轴两侧，令 $\alpha=\angle AOP$、$\beta=\angle POB$，则
\[
 \boxed{y_1y_2=-2pa},\quad
 \boxed{x_1x_2=a^2},\quad
 \boxed{\tan\alpha\tan\beta=\frac{2p}{a}}
 \quad\left(a=\frac p2\Rightarrow4\right).
\]
\textbf{口诀：}\quad \textbf{轴上正点 $a$，纵积负 $2pa$、横积 $a$ 平方、正切积 $2p/a$；焦点取 $p/2$，正切积四，两切成直角，交点落准线。}

\medskip
\textbf{\textcolor{CSummary}{证明记忆链（按推荐顺序）}}\par\smallskip
\begin{enumerate}[leftmargin=*,itemsep=0.3em]
 \item \textbf{最快：}设 $x=my+\frac p2$；代入后常数项 $-p^2$；韦达得纵积；平方转成横积。
 \item \textbf{最低门槛：}先用相似 $h/u=k/v$，代入坐标得 $(u+v)(uv-p^2)=0$。也可设点 $A(\frac{a^2}{2p},a),B(\frac{b^2}{2p},b)$；三点共线 $\Rightarrow(a-b)(ab+p^2)=0$。
 \item \textbf{常规：}设 $y=k(x-\frac p2)$；联立对 $x$ 用韦达；竖直通径另验。
\end{enumerate}

\medskip
\textbf{\textcolor{CSummary}{推广的三步证明（不额外设斜率）}}\par\smallskip
设 $x=my+a$（覆盖竖直弦），代入得
$y^2-2pmy-2pa=0$；韦达得 $y_1y_2=-2pa$；
由 $x_i=\frac{y_i^2}{2p}$ 得 $x_1x_2=a^2$，
所以 $\tan\alpha\tan\beta=-\frac{y_1y_2}{x_1x_2}=\frac{2p}{a}$。

\end{summarybox}

\begin{summarybox}
\textbf{\textcolor{CSummary}{做题入手信号}}\par\smallskip
看见\textbf{焦点弦 $AB$}，先写 $y_1y_2=-p^2$；要求横坐标就用
$x_1x_2=\frac{p^2}{4}$；涉及切线就用 $k_i=\frac p{y_i}$；涉及焦半径就用
$|AF|=x_1+\frac p2$。
若题目从顶点 $O$ 看\textbf{过轴上定点 $P(a,0)$ 的弦}（$a>0$），
先写\textbf{两侧正角的正切积 $\frac{2p}{a}$}；当 $P$ 恰为焦点 $F$ 时，
再化成 $4$。两条顶点连线的\textbf{带符号斜率积}则是 $-\frac{2p}{a}$（焦点特例为 $-4$）。

\medskip
\textbf{\textcolor{CSummary}{最后五项检查}}\par\smallskip
\begin{itemize}[leftmargin=*,itemsep=0.25em]
 \item 弦是否\textbf{经过题设轴上定点}，并确有两个不同交点？要套焦点弦定值，需核对定点是否为焦点。
 \item 是 $y^2=\pm2px$ 还是 $x^2=\pm2py$？上下开口要\textbf{交换横纵}。
 \item $y_1y_2$ 的\textbf{负号}与焦点坐标 $\frac p2$ 是否正确？
 \item 是轴上一般定点 $a>0$（正切积 $2p/a$），还是焦点 $a=p/2$（正切积 $4$）？
 \item 张角正切积取正，而两条顶点连线的\textbf{带符号斜率积取负}，有没有混淆？
\end{itemize}
\end{summarybox}
