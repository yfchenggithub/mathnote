\begin{summarybox}
\textbf{C016｜一页速记：四心坐标、最快证明与迁移}

\textbf{前提：} $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$，
$c^2=a^2+b^2$，$e=c/a$，$F_1(-c,0)$、$F_2(c,0)$，
$P(x_0,y_0)$ 为\textbf{非顶点}，$y_0\ne0$。
令 $\varepsilon=\operatorname{sgn}(x_0)$。

\textbf{四心公式（考试直接用）：}
\begin{align*}
\boxed{G}&=\left(\frac{x_0}{3},\frac{y_0}{3}\right),\\
\boxed{I}&=\left(\varepsilon a,\frac{ay_0}{a+|x_0|}\right)
=\left(\varepsilon a,\frac{cy_0}{c+e|x_0|}\right),\\
\boxed{H}&=\left(x_0,\frac{c^2-x_0^2}{y_0}\right),\\
\boxed{O'}&=\left(0,\frac{x_0^2+y_0^2-c^2}{2y_0}\right).
\end{align*}

\textbf{最少步骤：四心横坐标先秒出}
\begin{itemize}
  \item $G_x=x_0/3$：三点坐标平均。
  \item $I_x=(PF_1-PF_2)/2=\varepsilon a$：内切圆切线长 $+$ 双曲线定义。
  \item $H_x=x_0$：过 $P$ 的高线竖直。
  \item $O'_x=0$：底边中垂线是 $y$ 轴。
\end{itemize}

\textbf{纵坐标最快求法：}
\begin{itemize}
  \item 内心：$S=rs$，$S=c|y_0|$、$s=c+e|x_0|$，再加 $y_I$ 的正负号。
  \item 外心：设 $(0,t)$，令 $O'F_2^2=O'P^2$，一次平方相等。
  \item 垂心：可联立两条高线，\textbf{或者}先求外心，再用 $H=3G-2O'$。
\end{itemize}

\textbf{三条极易迁移的母结论：}
\[
\boxed{x_I=\frac{PF_1-PF_2}{2}},\qquad
\boxed{S_{\triangle IF_1F_2}/S_{\triangle PF_1F_2}
=\frac{a}{a+|x_0|}},
\]
\[
\boxed{H=3G-2O'},\qquad
\boxed{x_I=\pm a,\quad0<|y_I|<b}.
\]

\textbf{与椭圆对比：} 椭圆\textbf{和定}使 $y_I=cy_0/(a+c)$；
双曲线\textbf{差定}使 $x_I=\pm a$。

\textbf{一句口诀：} “重心平均、内心差定、垂心随点、外心居中；
先定横标，再算纵标；左右取绝对值，顶点必须排除。”
\end{summarybox}
