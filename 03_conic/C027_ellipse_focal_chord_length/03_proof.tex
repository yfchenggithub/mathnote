\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	先处理过右焦点且斜率存在的情形。
	通过联立直线与椭圆方程，利用韦达定理和弦长公式，推导出焦点弦长公式。
	垂直直线再单独验证，左焦点由对称性得到同样结论。

	\medskip

	\textbf{\textcolor{CProof}{正式推导}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设直线方程）：}]
		      先考虑过右焦点 $F(c,0)$ 且不垂直于 $x$ 轴的直线。
		      设直线方程为
		      \[
			      y=k(x-c).
		      \]
		      \par\textbf{理由：}
		      直线过点 $(c,0)$，且斜率为 $k$。

		\item[\textbf{步骤二（联立椭圆）：}]
		      代入椭圆
		      \[
			      \frac{x^2}{a^2}+\frac{y^2}{b^2}=1
		      \]
		      得
		      \[
			      \frac{x^2}{a^2}+\frac{k^2(x-c)^2}{b^2}=1.
		      \]
		      两边乘 $a^2b^2$，整理得
		      \[
			      (a^2k^2+b^2)x^2-2a^2k^2c\,x+(a^2k^2c^2-a^2b^2)=0.
		      \]
		      \par\textbf{理由：}
		      代入后去分母，并合并 $x$ 的同次项。

		\item[\textbf{步骤三（韦达定理）：}]
		      设两个交点的横坐标为 $x_1,x_2$，则
		      \[
			      x_1+x_2=\frac{2a^2k^2c}{a^2k^2+b^2},
			      \qquad
			      x_1x_2=\frac{a^2k^2c^2-a^2b^2}{a^2k^2+b^2}.
		      \]
		      \par\textbf{理由：}
		      二次方程 $Ax^2+Bx+C=0$ 的两根满足
		      \[
			      x_1+x_2=-\frac BA,\qquad x_1x_2=\frac CA.
		      \]

		\item[\textbf{步骤四（计算 $|x_1-x_2|^2$）：}]
		      \[
			      \begin{aligned}
				      (x_1-x_2)^2
				       & = (x_1+x_2)^2-4x_1x_2                               \\[0.3em]
				       & = \frac{4a^4k^4c^2}{(a^2k^2+b^2)^2}
				      -\frac{4(a^2k^2c^2-a^2b^2)}{a^2k^2+b^2}                \\[0.3em]
				       & = \frac{4a^4k^4c^2-4(a^2k^2c^2-a^2b^2)(a^2k^2+b^2)}
				      {(a^2k^2+b^2)^2}                                       \\[0.3em]
				       & = \frac{4a^2b^2(a^2k^2+b^2-k^2c^2)}
				      {(a^2k^2+b^2)^2}.
			      \end{aligned}
		      \]
		      又因为
		      \[
			      c^2=a^2-b^2,
		      \]
		      所以
		      \[
			      \begin{aligned}
				      a^2k^2+b^2-k^2c^2
				       & =a^2k^2+b^2-k^2(a^2-b^2) \\
				       & =b^2(1+k^2).
			      \end{aligned}
		      \]
		      因此
		      \[
			      (x_1-x_2)^2
			      =
			      \frac{4a^2b^4(1+k^2)}{(a^2k^2+b^2)^2}.
		      \]
		      \par\textbf{理由：}
		      代入韦达结果，通分后利用 $c^2=a^2-b^2$ 化简。

		\item[\textbf{步骤五（使用弦长公式）：}]
		      由于直线斜率为 $k$，两交点间距离满足
		      \[
			      |PQ|=\sqrt{1+k^2}\,|x_1-x_2|.
		      \]
		      由上式开方得
		      \[
			      |x_1-x_2|
			      =
			      \frac{2ab^2\sqrt{1+k^2}}{a^2k^2+b^2}.
		      \]
		      所以
		      \[
			      \begin{aligned}
				      |PQ|
				       & =\sqrt{1+k^2}\cdot
				      \frac{2ab^2\sqrt{1+k^2}}{a^2k^2+b^2} \\[0.3em]
				       & =\frac{2ab^2(1+k^2)}{a^2k^2+b^2}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：}
		      弦长公式与韦达结果结合即可得到焦点弦长公式。
	\end{description}

	\medskip

	\textbf{\textcolor{CProof}{垂直情形验证}}
	当过右焦点的直线垂直于 $x$ 轴时，直线为
	\[
		x=c.
	\]
	代入椭圆方程得
	\[
		\frac{c^2}{a^2}+\frac{y^2}{b^2}=1.
	\]
	因为 $c^2=a^2-b^2$，所以
	\[
		\frac{y^2}{b^2}=1-\frac{c^2}{a^2}
		=1-\frac{a^2-b^2}{a^2}
		=\frac{b^2}{a^2}.
	\]
	故
	\[
		y=\pm\frac{b^2}{a}.
	\]
	于是通径长度为
	\[
		|PQ|=\frac{b^2}{a}-\left(-\frac{b^2}{a}\right)
		=\frac{2b^2}{a}.
	\]

	\medskip

	\textbf{\textcolor{CProof}{左焦点同理}}
	以上推导针对右焦点 $F(c,0)$。
	若直线过左焦点 $F(-c,0)$，可设直线为
	\[
		y=k(x+c).
	\]
	同样联立椭圆方程，或直接利用椭圆关于 $y$ 轴对称，可知所得弦长公式完全相同。

	\medskip

	\textbf{\textcolor{CProof}{角度形式推导}}
	若过焦点的直线与 $x$ 轴夹角为 $\theta$，且 $\cos\theta\ne0$，则
	\[
		k=\tan\theta.
	\]
	代入斜率公式：
	\[
		\begin{aligned}
			|PQ|
			 & =\frac{2ab^2(1+\tan^2\theta)}{a^2\tan^2\theta+b^2} \\[0.3em]
			 & =\frac{2ab^2\sec^2\theta}
			{a^2\frac{\sin^2\theta}{\cos^2\theta}+b^2}            \\[0.3em]
			 & =\frac{2ab^2}{a^2\sin^2\theta+b^2\cos^2\theta}.
		\end{aligned}
	\]
	当 $\theta=\frac{\pi}{2}$ 时，该式直接给出
	\[
		|PQ|=\frac{2ab^2}{a^2}=\frac{2b^2}{a},
	\]
	与通径公式一致。

	\medskip

	\textbf{\textcolor{CProof}{结论回扣}}
	因此，对于焦点在 $x$ 轴的椭圆
	\[
		\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\qquad (a>b>0),
	\]
	过焦点且斜率为 $k$ 的非垂直直线对应的焦点弦长为
	\[
		|PQ|=\frac{2ab^2(1+k^2)}{a^2k^2+b^2};
	\]
	过焦点且垂直于 $x$ 轴的直线对应通径，弦长为
	\[
		|PQ|=\frac{2b^2}{a}.
	\]
\end{proofbox}

\begin{proofbox}
    \textbf{\textcolor{CProof}{补充证明：椭圆化圆法}}

    \textbf{\textcolor{CProof}{思路提示}}
    只做三件事：椭圆变圆，利用圆的弦长公式求对应弦，再把长度还原到椭圆。
    这里要区分\textbf{斜率的变换}与\textbf{弦长的变换}：
    二者的比例并不相同。

    \medskip

    \textbf{\textcolor{CProof}{正式推导}}
    对于椭圆
    \[
        \frac{x^2}{a^2}+\frac{y^2}{b^2}=1,
        \qquad a>b>0,\qquad c^2=a^2-b^2,
    \]
    设过右焦点 $F(c,0)$ 的非垂直直线与椭圆交于 $P,Q$，斜率为 $k$。

    \begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
        \item[\textbf{步骤一（椭圆化圆，得到新斜率）：}]
              令
              \[
                  X=x,\qquad Y=\frac ab y.
              \]
              即横坐标不变、纵坐标放大 $\frac ab$ 倍，原椭圆变为
              \[
                  X^2+Y^2=a^2.
              \]
              原焦点 $F(c,0)$ 的位置不变，\textbf{但它不是圆心}。
              原焦点弦所在直线
              \[
                  y=k(x-c)
              \]
              在圆中变为
              \[
                  Y=\frac ab k(X-c)=K(X-c),
                  \qquad \boxed{K=\frac ab k}.
              \]
              记该直线与圆的交点为 $P',Q'$，分别对应原椭圆的 $P,Q$。
              \par\textbf{理由：}
              纵坐标拉伸 $\frac ab$ 倍，同一条对应直线的斜率也变为 $\frac ab$ 倍。

        \item[\textbf{步骤二（在圆中求弦长）：}]
              圆的半径为 $a$，圆心为 $O(0,0)$。
              直线 $Y=K(X-c)$ 即
              \[
                  KX-Y-Kc=0.
              \]
              圆心到这条直线的距离为
              \[
                  d=\frac{|Kc|}{\sqrt{1+K^2}}.
              \]
              由半径、半弦长与弦心距构成的直角三角形，得
              \[
                  \begin{aligned}
                      |P'Q'|
                       &=2\sqrt{a^2-d^2}\\[0.3em]
                       &=2\sqrt{a^2-\frac{K^2c^2}{1+K^2}}\\[0.3em]
                       &=2\sqrt{\frac{a^2+(a^2-c^2)K^2}{1+K^2}}\\[0.3em]
                       &=\boxed{2\sqrt{\frac{a^2+b^2K^2}{1+K^2}}}.
                  \end{aligned}
              \]
              \par\textbf{理由：}
              用圆的弦长公式 $2\sqrt{a^2-d^2}$，并代入 $a^2-c^2=b^2$。

        \item[\textbf{步骤三（从圆弦长还原椭圆弦长）：}]
              由于变换中 $X=x$，对应两条弦的\textbf{横坐标差相同}。
              记这个差的绝对值为 $|\Delta x|$。
              原椭圆中直线斜率为 $k$，圆中对应直线斜率为 $K$，因此
              \[
                  |PQ|=|\Delta x|\sqrt{1+k^2},\qquad
                  |P'Q'|=|\Delta x|\sqrt{1+K^2}.
              \]
              两式相除，得\textbf{长度还原公式}：
              \[
                  \boxed{\frac{|PQ|}{|P'Q'|}
                    =\sqrt{\frac{1+k^2}{1+K^2}}}.
              \]
              代入 $K=\frac ab k$，可分别化简为
              \[
                  |P'Q'|
                  =\frac{2ab\sqrt{1+k^2}}{\sqrt{a^2k^2+b^2}},
                  \qquad
                  \sqrt{\frac{1+k^2}{1+K^2}}
                  =\frac{b\sqrt{1+k^2}}{\sqrt{a^2k^2+b^2}}.
              \]
              最后相乘：
              \[
                  \begin{aligned}
                      |PQ|
                       &=|P'Q'|\sqrt{\frac{1+k^2}{1+K^2}}\\[0.3em]
                       &=\frac{2ab\sqrt{1+k^2}}{\sqrt{a^2k^2+b^2}}
                         \cdot\frac{b\sqrt{1+k^2}}{\sqrt{a^2k^2+b^2}}\\[0.3em]
                       &=\boxed{\frac{2ab^2(1+k^2)}{a^2k^2+b^2}}.
                  \end{aligned}
              \]
              \par\textbf{理由：}
              对应弦的水平投影长度不变；实际弦长与水平投影的比值是
              $\sqrt{1+\text{斜率}^2}$，\textbf{不是斜率本身}。
    \end{description}

    \medskip

    \textbf{\textcolor{CProof}{垂直情形（直接用圆验证）}}
    若原直线为 $x=c$，则变换后仍为 $X=c$。
    圆内对应弦长为
    \[
        |P'Q'|=2\sqrt{a^2-c^2}=2b.
    \]
    该弦竖直，返回椭圆时纵向长度缩小为 $\frac ba$ 倍，所以
    \[
        \boxed{|PQ|=\frac ba\cdot 2b=\frac{2b^2}{a}}.
    \]
    过左焦点的情况由椭圆关于 $y$ 轴的对称性得到同样结论。

    \medskip

    \textbf{\textcolor{CProof}{方法总结}}
    \[
        \begin{aligned}
            &\text{第一步：}\quad K=\frac ab k;\\
            &\text{第二步：}\quad |P'Q'|=2\sqrt{a^2-d^2};\\
            &\text{第三步：}\quad |PQ|=|P'Q'|\sqrt{\frac{1+k^2}{1+K^2}}.
        \end{aligned}
    \]
\end{proofbox}

% 原韦达证明与椭圆化圆证明均原样保留。

\begin{proofbox}
    \textbf{\textcolor{CProof}{新增证明一：焦半径公式法（最快推导，首选）}}

    \textbf{\textcolor{CProof}{核心思路}}
    将焦点弦拆成两条焦半径，分别求长再相加。
    不必设直线方程联立二次方程，也不必计算交点坐标。

    \medskip

    设椭圆焦点为 $F_1(-c,0),F_2(c,0)$，$c^2=a^2-b^2$。
    过右焦点 $F_2$ 的直线与椭圆交于 $P,Q$。
    选取上半平面的交点作为 $P$（若弦为水平长轴，则取右端点为 $P$），
    令射线 $F_2P$ 与 $x$ 轴正方向的夹角为 $\theta\in[0,\pi]$；
    $\overrightarrow{F_2Q}$ 的方向角即 $\theta+\pi$。

    \begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.55em]
        \item[\textbf{第一步（用右焦点焦半径公式）：}]
              对椭圆上任意点 $P(x_P,y_P)$，右焦点焦半径为
              \[
                  \boxed{|F_2P|=a-\frac ca x_P}.
              \]
              记 $r_+=|F_2P|$。由于从 $F_2(c,0)$ 出发沿方向 $\theta$ 走 $r_+$，
              \[
                  x_P=c+r_+\cos\theta.
              \]
              代回焦半径公式：
              \[
                  \begin{aligned}
                      r_+&=a-\frac ca(c+r_+\cos\theta),\\
                      r_+(a+c\cos\theta)&=a^2-c^2=b^2,
                  \end{aligned}
              \]
              因此
              \[
                  \boxed{r_+=\frac{b^2}{a+c\cos\theta}}.
              \]
              \textbf{理由：}以焦点为起点写横坐标，未知数仅有焦半径 $r_+$。

        \item[\textbf{第二步（方向相反，只换余弦符号）：}]
              令 $r_-=|F_2Q|$。由于 $Q$ 位于焦点另一侧，
              \[
                  x_Q=c-r_-\cos\theta.
              \]
              同理得到
              \[
                  \boxed{r_-=\frac{b^2}{a-c\cos\theta}}.
              \]
              \textbf{理由：}反方向角为 $\theta+\pi$，满足
              $\cos(\theta+\pi)=-\cos\theta$。

        \item[\textbf{第三步（相加、化为角度或斜率式）：}]
              因为 $P,F_2,Q$ 共线且焦点位于两交点之间，所以
              \[
                  \begin{aligned}
                  |PQ| &= r_++r_-\\
                       &=\frac{b^2}{a+c\cos\theta}
                         +\frac{b^2}{a-c\cos\theta}\\
                       &=\frac{2ab^2}{a^2-c^2\cos^2\theta}\\
                       &=\boxed{\frac{2ab^2}{a^2\sin^2\theta+b^2\cos^2\theta}}.
                  \end{aligned}
              \]
              这里用到 $a^2-c^2\cos^2\theta
              =a^2\sin^2\theta+(a^2-c^2)\cos^2\theta$。
              非竖直时 $k=\tan\theta$，于是
              \[
                  \cos^2\theta=\frac{1}{1+k^2},\qquad
                  \sin^2\theta=\frac{k^2}{1+k^2},
              \]
              最终得到
              \[
                  \boxed{|PQ|=\frac{2ab^2(1+k^2)}{a^2k^2+b^2}}.
              \]
    \end{description}

    \medskip

    \textbf{\textcolor{CProof}{焦半径公式若忘记：两行即时恢复}}
    由椭圆定义 $|F_1P|+|F_2P|=2a$，记 $|F_2P|=r$，则
    $|F_1P|=2a-r$。两距离平方之差一方面等于
    \[
        |F_1P|^2-|F_2P|^2
        =((x_P+c)^2+y_P^2)-((x_P-c)^2+y_P^2)=4cx_P,
    \]
    另一方面又等于
    \[
        (2a-r)^2-r^2=4a^2-4ar.
    \]
    比较两式即 $\boxed{r=a-\frac ca x_P}$，无需硬背。
    过左焦点的情形由关于 $y$ 轴的对称性得到相同弦长。

    \medskip

    \textbf{\textcolor{CProof}{两条特殊焦点弦（直接带入角度）}}
    \[
        \begin{array}{c|c|c}
        \text{方向}&\text{角度}&\text{焦点弦长}\\ \hline
        \text{水平长轴}&\theta=0&\boxed{2a}\\[0.35em]
        \text{竖直通径}&\theta=\frac{\pi}{2}&\boxed{\frac{2b^2}{a}}
        \end{array}
    \]
    角度公式在竖直方向仍成立；斜率公式不能直接给竖直直线赋值。
\end{proofbox}

\begin{proofbox}
    \textbf{\textcolor{CProof}{新增证明二：椭圆定义与余弦定理法（从零恢复）}}

    \textbf{\textcolor{CProof}{核心思路}}
    只用椭圆定义 $PF_1+PF_2=2a$ 与初中熟悉的余弦定理，
    先求右焦点到一个交点的距离，再求反方向，最后相加。

    仍令 $F_1(-c,0),F_2(c,0)$，取椭圆上点 $P$，记
    \[
        PF_2=r,\qquad PF_1=2a-r,\qquad F_1F_2=2c.
    \]
    取上半平面的交点为 $P$（水平时取右端点），
    设射线 $F_2P$ 与 $x$ 轴正方向的夹角为 $\theta\in[0,\pi]$。
    \textbf{特别注意：}在三角形 $F_1F_2P$ 中，
    $\overrightarrow{F_2F_1}$ 指向 $x$ 轴负方向，
    故 $F_2$ 处的夹角余弦为 $\cos(\pi-\theta)=-\cos\theta$（方向角按模 $2\pi$ 理解）。

    \begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.55em]
        \item[\textbf{第一步（在焦点三角形中用余弦定理）：}]
              \[
                  \begin{aligned}
                  (2a-r)^2
                  &=(2c)^2+r^2-2(2c)r\cos(\pi-\theta)\\
                  &=4c^2+r^2+4cr\cos\theta.
                  \end{aligned}
              \]
              \textbf{理由：}与水平正方向夹角是 $\theta$，与指向左焦点的射线夹角不同，负号不能漏。

        \item[\textbf{第二步（消掉平方，求单段焦半径）：}]
              \[
                  \begin{aligned}
                  4a^2-4ar+r^2&=4c^2+r^2+4cr\cos\theta,\\
                  r(a+c\cos\theta)&=a^2-c^2=b^2.
                  \end{aligned}
              \]
              因此
              \[
                  \boxed{PF_2=r=\frac{b^2}{a+c\cos\theta}}.
              \]
              \textbf{理由：}两侧 $r^2$ 正好消去，解的是一次方程。

        \item[\textbf{第三步（对向点换角、两段相加）：}]
              另一交点 $Q$ 的方向角为 $\theta+\pi$，从而
              \[
                  QF_2=\frac{b^2}{a-c\cos\theta}.
              \]
              所以
              \[
                  \begin{aligned}
                  |PQ|&=PF_2+QF_2\\
                      &=\frac{2ab^2}{a^2-c^2\cos^2\theta}\\
                      &=\frac{2ab^2}{a^2\sin^2\theta+b^2\cos^2\theta}.
                  \end{aligned}
              \]
              当 $k=\tan\theta$ 存在时，这又化为
              \[
                  \boxed{|PQ|=\frac{2ab^2(1+k^2)}{a^2k^2+b^2}}.
              \]
    \end{description}

    \medskip
    \textbf{\textcolor{CProof}{考试速记}}
    \[
        PF_1+PF_2=2a
        \ \xrightarrow{\ \text{余弦定理}\ }
        \ \frac{b^2}{a+c\cos\theta},\ \frac{b^2}{a-c\cos\theta}
        \ \xrightarrow{\ \text{相加}\ }\ |PQ|.
    \]
    两种新证明最终是同一条\textbf{焦半径求和路线}：
    已记住焦半径公式时，方法一更快；完全忘记公式时，方法二更稳。
\end{proofbox}
