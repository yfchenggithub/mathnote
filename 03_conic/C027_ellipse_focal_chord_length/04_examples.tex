\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	椭圆
	\[
		\frac{x^2}{4}+\frac{y^2}{3}=1
	\]
	的右焦点为 $F(1,0)$，过 $F$ 的直线斜率为 $2$，求该直线被椭圆截得的弦长 $|PQ|$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
		      由椭圆方程得
		      \[
			      a^2=4,\qquad b^2=3,\qquad c^2=a^2-b^2=1.
		      \]
		      因此 $a=2$，$c=1$。
		      \par\textbf{理由：}
		      椭圆为焦点在 $x$ 轴的标准形式。

		\item[\textbf{第二步（代入公式）：}]
		      因为直线过焦点且斜率 $k=2$ 存在，所以可用焦点弦长公式
		      \[
			      |PQ|=\frac{2ab^2(1+k^2)}{a^2k^2+b^2}.
		      \]
		      代入 $a=2$，$b^2=3$，$k=2$：
		      \[
			      |PQ|=\frac{2\cdot2\cdot3\cdot(1+4)}{4\cdot4+3}.
		      \]
		      \par\textbf{理由：}
		      直接套用非垂直焦点弦长公式。

		\item[\textbf{第三步（计算结果）：}]
		      \[
			      |PQ|=\frac{60}{19}.
		      \]
		      \par\textbf{理由：}
		      算术化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	所求弦长为
	\[
		\boxed{\frac{60}{19}}.
	\]

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	椭圆
	\[
		\frac{x^2}{16}+\frac{y^2}{12}=1
	\]
	的右焦点为 $F(2,0)$，过 $F$ 的直线被椭圆截得的弦长为 $\frac{15}{2}$，求直线的斜率 $k$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出公式）：}]
		      由椭圆方程知
		      \[
			      a=4,\qquad b^2=12.
		      \]
		      焦点弦长公式为
		      \[
			      |PQ|=\frac{2ab^2(1+k^2)}{a^2k^2+b^2}.
		      \]
		      \par\textbf{理由：}
		      椭圆焦点在 $x$ 轴，且题目要求的是过焦点直线的斜率。

		\item[\textbf{第二步（列方程）：}]
		      代入 $|PQ|=\frac{15}{2}$ 得
		      \[
			      \frac{2\cdot4\cdot12(1+k^2)}{16k^2+12}
			      =
			      \frac{15}{2}.
		      \]
		      即
		      \[
			      \frac{96(1+k^2)}{16k^2+12}
			      =
			      \frac{15}{2}.
		      \]
		      \par\textbf{理由：}
		      根据已知弦长建立方程。

		\item[\textbf{第三步（解方程）：}]
		      两边乘 $2(16k^2+12)$ 得
		      \[
			      192(1+k^2)=15(16k^2+12).
		      \]
		      展开得
		      \[
			      192+192k^2=240k^2+180.
		      \]
		      移项得
		      \[
			      12=48k^2,
		      \]
		      所以
		      \[
			      k^2=\frac14,\qquad k=\pm\frac12.
		      \]
		      \par\textbf{理由：}
		      代数运算解方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	直线斜率为
	\[
		\boxed{k=\frac12\quad\text{或}\quad k=-\frac12}.
	\]

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	设椭圆
	\[
		\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\qquad (a>b>0)
	\]
	的右焦点为 $F$，过 $F$ 且斜率为 $1$ 的直线被椭圆截得的弦长为 $\frac{4a}{3}$，求椭圆的离心率 $e$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入公式）：}]
		      因为直线过焦点且斜率 $k=1$，所以
		      \[
			      |PQ|=\frac{2ab^2(1+k^2)}{a^2k^2+b^2}.
		      \]
		      代入 $k=1$ 得
		      \[
			      |PQ|
			      =
			      \frac{2ab^2(1+1)}{a^2\cdot1^2+b^2}
			      =
			      \frac{4ab^2}{a^2+b^2}.
		      \]
		      \par\textbf{理由：}
		      使用非垂直焦点弦长公式。

		\item[\textbf{第二步（列方程）：}]
		      由题设 $|PQ|=\frac{4a}{3}$，因此
		      \[
			      \frac{4ab^2}{a^2+b^2}
			      =
			      \frac{4a}{3}.
		      \]
		      \par\textbf{理由：}
		      把已知弦长代入。

		\item[\textbf{第三步（化简求 $e$）：}]
		      两边除以 $4a$（$a>0$）得
		      \[
			      \frac{b^2}{a^2+b^2}=\frac13.
		      \]
		      所以
		      \[
			      3b^2=a^2+b^2,
		      \]
		      即
		      \[
			      a^2=2b^2.
		      \]
		      又因为
		      \[
			      c^2=a^2-b^2,
		      \]
		      所以
		      \[
			      c^2=b^2,\qquad c=b.
		      \]
		      于是
		      \[
			      e=\frac ca=\frac{b}{\sqrt2\,b}=\frac{\sqrt2}{2}.
		      \]
		      \par\textbf{理由：}
		      代入离心率公式 $e=\frac ca$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	椭圆的离心率为
	\[
		\boxed{\frac{\sqrt2}{2}}.
	\]
\end{examplebox}
