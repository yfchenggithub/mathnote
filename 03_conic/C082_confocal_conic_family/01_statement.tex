\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（椭圆参数）：}}]
			焦点在 $x$ 轴上的椭圆 $\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$（$a>b>0$），焦距 $c=\sqrt{a^{2}-b^{2}}$。
		\item[\textbf{\textcolor{CStatement}{条件2（参数范围）：}}]
			实数 $\lambda$ 满足 $\lambda<a^{2}$ 且 $\lambda\neq b^{2}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（曲线系方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			参数 $\lambda$ 变动时曲线类型改变，但焦点始终相同。
			\[
				\frac{x^{2}}{a^{2}-\lambda}+\frac{y^{2}}{b^{2}-\lambda}=1
			\]
			\textbf{\textcolor{CStatement}{分类规则：}}
			当 $\lambda<b^{2}$ 时为椭圆；当 $b^{2}<\lambda<a^{2}$ 时为双曲线。
		\item[\textbf{\textcolor{CStatement}{推广结论（焦点恒定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有曲线共享相同的焦点 $(\pm c,0)$。
			\[
				c=\sqrt{a^{2}-b^{2}}
			\]
	\end{description}

\end{statementbox}
