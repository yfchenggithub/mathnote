\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\medskip
	\textbf{共焦点曲线系的核心是：焦点恒为 $(\pm c,0)$，曲线类型由 $\lambda$ 与 $b^{2}$ 的大小关系决定。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*,itemsep=0.5em,topsep=0.4em]
		\item \textbf{条件1（椭圆参数）：} 存在焦点在 $x$ 轴上的椭圆 $\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$ ($a>b>0$)，$c=\sqrt{a^{2}-b^{2}}$。
		\item \textbf{条件2（λ参数限制）：} $\lambda<a^{2}$ 且 $\lambda\neq b^{2}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*,itemsep=0.5em,topsep=0.4em]
		\item \textbf{易错点（检查λ范围）：} 使用曲线系必须先验证 $\lambda$ 是否满足条件。
		\item \textbf{检查项（分母符号）：} 根据 $b^{2}-\lambda$ 的正负确定曲线类型。
	\end{itemize}

\end{summarybox}
