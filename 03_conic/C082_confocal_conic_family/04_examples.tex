\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^{2}}{16}+\frac{y^{2}}{9}=1$，写出与其共焦点的二次曲线系方程，并求当 $\lambda=5$ 时曲线的类型。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由椭圆方程得 $a^{2}=16,b^{2}=9,c^{2}=7$。
			\par\textbf{理由：} 椭圆 $a^{2}>b^{2}$，焦距 $c^{2}=a^{2}-b^{2}$。
		\item[\textbf{第二步（写曲线系）：}]
			共焦点曲线系方程为 $\frac{x^{2}}{16-\lambda}+\frac{y^{2}}{9-\lambda}=1$。
			\par\textbf{理由：} 共焦点曲线系公式代入。
		\item[\textbf{第三步（判断类型）：}]
			将 $\lambda=5$ 代入，分母均正且 $\lambda<9$，故曲线为椭圆。
			\par\textbf{理由：} $\lambda<b^{2}$ 时为椭圆。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{曲线系 $\frac{x^{2}}{16-\lambda}+\frac{y^{2}}{9-\lambda}=1$，$\lambda=5$ 时曲线为椭圆。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	已知二次曲线与椭圆 $\frac{x^{2}}{16}+\frac{y^{2}}{9}=1$ 共焦点，且经过点 $P(0,4)$，求该曲线的方程并判断类型。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（设曲线系）：}]
			设曲线方程为 $\frac{x^{2}}{16-\lambda}+\frac{y^{2}}{9-\lambda}=1$（$\lambda<16,\lambda\neq9$）。
			\par\textbf{理由：} 与椭圆共焦点。
		\item[\textbf{第二步（代入求λ）：}]
			代入 $x=0,y=4$ 得 $\frac{16}{9-\lambda}=1$，解得 $\lambda=-7$。
			\par\textbf{理由：} 化简一次方程。
		\item[\textbf{第三步（写方程并判断）：}]
			$\lambda=-7<9$，故为椭圆，方程 $\frac{x^{2}}{23}+\frac{y^{2}}{16}=1$。
			\par\textbf{理由：} $\lambda<b^{2}$ 时为椭圆。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{曲线方程为 $\frac{x^{2}}{23}+\frac{y^{2}}{16}=1$，该曲线是椭圆。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	已知双曲线与椭圆 $\frac{x^{2}}{8}+\frac{y^{2}}{4}=1$ 共焦点，且经过点 $A(3,\sqrt{2})$，求双曲线的标准方程。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（设曲线系）：}]
			设共焦点曲线系 $\frac{x^{2}}{8-\lambda}+\frac{y^{2}}{4-\lambda}=1$，因为双曲线，所以 $4<\lambda<8$。
			\par\textbf{理由：} 双曲线需要分母一正一负。
		\item[\textbf{第二步（代入求λ）：}]
			代入 $A(3,\sqrt{2})$ 得 $\frac{9}{8-\lambda}+\frac{2}{4-\lambda}=1$。化简得 $\lambda^{2}-\lambda-20=0$，解得 $\lambda=5$ 或 $\lambda=-4$（舍去）。
			\par\textbf{理由：} 分式方程化整式，检验根的范围。
		\item[\textbf{第三步（写方程）：}]
			$\lambda=5$ 代入得 $8-\lambda=3$, $4-\lambda=-1$，故双曲线方程为 $\frac{x^{2}}{3}-\frac{y^{2}}{1}=1$，即 $\frac{x^{2}}{3}-y^{2}=1$。
			\par\textbf{理由：} 双曲线标准形式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{双曲线方程为 $\frac{x^{2}}{3}-y^{2}=1$。}
\end{examplebox}
