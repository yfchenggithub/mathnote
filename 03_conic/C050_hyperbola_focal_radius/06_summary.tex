\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{双曲线焦半径公式将点到焦点的距离转化为横坐标的线性绝对值表达式，是快速解题的重要工具。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（焦点在$x$轴）：} 双曲线标准方程 $\frac{x^2}{a^2}-\frac{y^2}{b^2}=1$（$a,b>0$），$e=c/a>1$，焦点 $F_1(-c,0),F_2(c,0)$。
		\item \textbf{条件2（点在曲线上）：} 已知点 $P(x_0,y_0)$ 满足双曲线方程，且 $|x_0|\ge a$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（绝对值处理）：} 公式中的绝对值不能随意去掉，必须根据 $x_0$ 的范围（左支或右支）确定符号。
		\item \textbf{检查项（定义域验证）：} 使用公式前应先确认 $|x_0|\ge a$，否则点不在双曲线上。
	\end{itemize}

\end{summarybox}
