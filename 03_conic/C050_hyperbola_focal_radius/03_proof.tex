\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	从距离公式出发，利用双曲线方程消去 $y_0$，将 $|PF_1|^2$ 和 $|PF_2|^2$ 整理成完全平方 $(ex_0\pm a)^2$，再开方即得。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设点代入曲线）：}]
			\[
				\begin{aligned}
					|PF_1|^2 &= (x_0+c)^2 + y_0^2,\\
					y_0^2 &= \frac{b^2}{a^2}(x_0^2 - a^2).
			\end{aligned}
			\]
			\par\textbf{理由：} 两点间距离公式，以及双曲线方程变形得到 $y_0^2$ 表达式。

		\item[\textbf{步骤二（合并化为完全平方）：}]
			\[
				\begin{aligned}
					|PF_1|^2 &= x_0^2+2cx_0+c^2 + \frac{b^2}{a^2}(x_0^2 - a^2) \\
					&= \left(1+\frac{b^2}{a^2}\right)x_0^2 + 2c x_0 + c^2 - b^2 \\
					&= \frac{a^2+b^2}{a^2}x_0^2 + 2c x_0 + (c^2-b^2) \\
					&= \frac{c^2}{a^2}x_0^2 + 2c x_0 + a^2 \\
					&= (e x_0)^2 + 2a e x_0 + a^2 \\
					&= (e x_0 + a)^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用 $c^2=a^2+b^2$ 和 $e=c/a$ 逐步简化，发现是完全平方。

		\item[\textbf{步骤三（开方取绝对值）：}]
			\[
				\begin{aligned}
					|PF_1| &= \sqrt{(e x_0 + a)^2} \\
					&= |e x_0 + a|.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术平方根是非负的，必须加绝对值。

		\item[\textbf{步骤四（计算 $|PF_2|^2$）：}]
			\[
				\begin{aligned}
					|PF_2|^2 &= (x_0-c)^2 + y_0^2 \\
					&= (e x_0 - a)^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 将 $c$ 换为 $-c$，重复步骤一和二的简化过程。

		\item[\textbf{步骤五（开方得 $|PF_2|$）：}]
			\[
				\begin{aligned}
					|PF_2| &= \sqrt{(e x_0 - a)^2} \\
					&= |e x_0 - a|.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术平方根非负，取绝对值。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	因此，双曲线焦半径公式为 $|PF_1| = |a + e x_0|$，$|PF_2| = |a - e x_0|$。进一步，当 $x_0 \ge a$ 时 $|PF_1|=e x_0+a$，$|PF_2|=e x_0-a$；当 $x_0 \le -a$ 时 $|PF_1| = -(e x_0+a)$，$|PF_2| = -(e x_0-a)$。

\end{proofbox}
