	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（抛物线标准形式）：}}] 抛物线为 $y^2=2px$ 或 $y^2=-2px$，其中 $p>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（焦点弦倾斜角）：}}] 过焦点的弦所在直线倾斜角为 $\theta$，且 $\theta \neq 0,\pi$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦点弦长公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点弦长度由倾斜角的正弦平方决定，与 $p$ 成正比。
			\[
				L = \frac{2p}{\sin^2\theta}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（通径特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当弦垂直于 $x$ 轴时取得最小长度 $2p$，称为通径。
			\[
				\theta = \frac{\pi}{2} \;\Rightarrow\; L_{\min} = 2p.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{适用范围（两种标准形式）}}
	\[
		y^2 = 2px \quad\text{或}\quad y^2 = -2px \quad (p>0).
	\]

\end{statementbox}
