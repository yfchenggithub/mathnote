\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=4x$ 的焦点为 $F$，过 $F$ 且倾斜角为 $30^\circ$ 的直线交抛物线于 $A,B$ 两点，求 $|AB|$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 抛物线 $y^2=4x$ 为标准形式 $y^2=2px$，所以 $2p=4$，$p=2$。
			\par\textbf{理由：}
			套用焦点弦长公式前，需确定 $p$ 的值。
		\item[\textbf{第二步（代入公式）：}] 倾斜角 $\theta=30^\circ$，$\sin30^\circ=\dfrac12$，$\sin^2\theta=\dfrac14$。代入公式 $L=\dfrac{2p}{\sin^2\theta}$ 得 $L=\dfrac{2\times2}{\frac{1}{4}}=\dfrac{4}{\frac{1}{4}}=4\times4=16$。
			\par\textbf{理由：}
			$L$ 仅与 $p$ 和 $\theta$ 有关，直接代入即可。
		\item[\textbf{第三步（写出答案）：}] 所以 $|AB|=16$。
			\par\textbf{理由：}
			得到最终结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{焦点弦长 $L$ 由公式 $L=\dfrac{2p}{\sin^2\theta}$ 直接计算得到，$p=2$，$\theta=30^\circ$，得 $L=16$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=8x$ 的焦点弦长为 $16$，求该焦点弦所在直线的倾斜角 $\theta$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}] 由 $y^2=8x$ 得 $2p=8$，$p=4$。
			\par\textbf{理由：}
			公式需要 $p$ 值。
		\item[\textbf{第二步（列方程）：}] 由 $L=\dfrac{2p}{\sin^2\theta}$，$L=16$，得 $16=\dfrac{2\times4}{\sin^2\theta}=\dfrac{8}{\sin^2\theta}$，所以 $\sin^2\theta=\dfrac{8}{16}=\dfrac12$。
			\par\textbf{理由：}
			将已知量代入公式，反解 $\sin^2\theta$。
		\item[\textbf{第三步（求倾斜角）：}] 因为 $\theta\in(0,\pi)$ 且 $\theta\neq0,\pi$，$\sin\theta>0$，所以 $\sin\theta=\dfrac{\sqrt2}{2}$，得 $\theta=\dfrac{\pi}{4}$ 或 $\theta=\dfrac{3\pi}{4}$。
			\par\textbf{理由：}
			在 $(0,\pi)$ 内正弦值为 $\sqrt2/2$ 对应两个角度。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{利用公式反解 $\sin^2\theta$，注意倾斜角的范围得到两个可能值 $\theta=\dfrac{\pi}{4},\dfrac{3\pi}{4}$。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	抛物线 $y^2=2px\;(p>0)$ 的焦点为 $F$，过 $F$ 的直线 $l$ 与抛物线交于 $A,B$ 两点，$l$ 的倾斜角为 $60^\circ$，且 $|AF|+|BF|=8$，求 $p$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（转换弦长）：}] 由于 $A,F,B$ 三点共线且 $F$ 在 $A,B$ 之间，根据抛物线定义，$|AF|=x_A+\dfrac{p}{2}$，$|BF|=x_B+\dfrac{p}{2}$，所以 $|AF|+|BF|=(x_A+x_B)+p$。又因为 $A,B$ 是焦点弦，有 $|AB|=|AF|+|BF|$（焦点在弦上），故 $|AB|=8$。
			\par\textbf{理由：}
			抛物线上的点到焦点距离等于到准线距离，焦半径之和恰为弦长。
		\item[\textbf{第二步（代入公式）：}] 已知 $\theta=60^\circ$，$\sin60^\circ=\dfrac{\sqrt3}{2}$，$\sin^2\theta=\dfrac34$。由焦点弦长公式 $L=\dfrac{2p}{\sin^2\theta}$ 得 $8=\dfrac{2p}{\frac{3}{4}}=\dfrac{8p}{3}$。
			\par\textbf{理由：}
			将 $L=8$ 和 $\theta$ 代入公式建立关于 $p$ 的方程。
		\item[\textbf{第三步（解出参数）：}] 解方程 $8=\dfrac{8p}{3}$，得 $p=3$。
			\par\textbf{理由：}
			等式两边乘以3再除以8即得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{由 $|AF|+|BF|=|AB|$ 得到弦长，再套用 $L=\dfrac{2p}{\sin^2\theta}$ 列出方程解得 $p=3$。}
\end{examplebox}
