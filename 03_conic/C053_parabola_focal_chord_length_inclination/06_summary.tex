\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{抛物线焦点弦长等于 $2p$ 除以倾斜角正弦的平方，其取值只与 $p$ 和 $\theta$ 有关。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（抛物线标准形式）：} 抛物线为 $y^2=\pm2px$（$p>0$），即开口向左或向右。
		\item \textbf{条件2（倾斜角限制）：} 倾斜角 $\theta\in(0,\pi)$ 且 $\theta\neq0,\pi$，保证直线与抛物线有两个交点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母为零）：} $\theta=0$ 或 $\pi$ 时公式无意义，此时直线与抛物线的对称轴平行，不相交于两点。
		\item \textbf{检查项（$p$ 的取值）：} 确认 $p>0$，若抛物线方程为 $y^2=2px$，$p$ 为正常数；若为 $y^2=-2px$，$p$ 同样取正值，公式相同。
	\end{itemize}
\end{summarybox}
