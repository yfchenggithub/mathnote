\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（双曲线方程）：}}] 双曲线 $\dfrac{x^{2}}{a^{2}}-\dfrac{y^{2}}{b^{2}}=1$（$a>0$, $b>0$）。
		\item[\textbf{\textcolor{CStatement}{条件2（点在曲线上）：}}] 点 $P(x_{0},y_{0})$ 在该双曲线上。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（切线方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将双曲线方程中的 $x^{2}$ 换成 $x_{0}x$，$y^{2}$ 换成 $y_{0}y$ 即得切线方程。
			\[
				\frac{x_{0}x}{a^{2}}-\frac{y_{0}y}{b^{2}}=1.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（焦点在y轴）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若双曲线为 $\dfrac{y^{2}}{a^{2}}-\dfrac{x^{2}}{b^{2}}=1$，类似地，将 $y^{2}$ 换为 $y_{0}y$，$x^{2}$ 换为 $x_{0}x$。
			\[
				\frac{y_{0}y}{a^{2}}-\frac{x_{0}x}{b^{2}}=1.
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒：}}
	该切线公式仅适用于点 $P$ 在双曲线上的情形。若点 $P$ 在双曲线外部或内部，该公式不表示切线。
\end{statementbox}
