\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2 = 4x$ 及外一点 $P(2,3)$，过 $P$ 引抛物线的两条切线，求切点弦所在直线方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别方程形式）：}] 抛物线 $y^2=4x$ 对应标准形式 $y^2=2px$，故 $2p=4$，$p=2$。
			\par\textbf{理由：} 确定参数 $p$。

		\item[\textbf{第二步（套用公式）：}] 切点弦方程为 $y_0 y = p(x+x_0)$，代入 $p=2$, $x_0=2$, $y_0=3$ 得 $3y = 2(x+2)$。
			\par\textbf{理由：} 直接应用结论 $y_0 y = p(x+x_0)$。

		\item[\textbf{第三步（化简）：}] 整理得 $2x - 3y + 4 = 0$。
			\par\textbf{理由：} 化为一般式，便于后续计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	切点弦方程为 $2x - 3y + 4 = 0$。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $x^2 = 6y$ 及外一点 $P(-1, 1)$，求切点弦方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别方程）：}] $x^2=6y$ 对应 $x^2=2py$，故 $2p=6$，$p=3$。
			\par\textbf{理由：} 确定 $p$ 值。

		\item[\textbf{第二步（套用公式）：}] 切点弦方程为 $x_0 x = p(y+y_0)$，代入 $x_0=-1$, $y_0=1$, $p=3$ 得 $-x = 3(y+1)$。
			\par\textbf{理由：} 应用推广结论。

		\item[\textbf{第三步（化简）：}] 整理得 $x+3y+3=0$。
			\par\textbf{理由：} 一般式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	切点弦方程为 $x+3y+3=0$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=2x$ 及外一点 $P(1,2)$。
	(1) 求切点弦方程；
	(2) 若切点弦分别交 $x$ 轴、$y$ 轴于点 $M,N$，求 $\triangle OMN$ 的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求切点弦）：}] 由 $y^2=2x$ 知 $p=1$。代入 $x_0=1$, $y_0=2$ 于 $y_0 y = p(x+x_0)$ 得 $2y = 1(x+1)$，即 $x-2y+1=0$。
			\par\textbf{理由：} 直接套公式。

		\item[\textbf{第二步（求坐标轴交点）：}] 令 $y=0$ 得 $x=-1$，即 $M(-1,0)$；令 $x=0$ 得 $y=\frac12$，即 $N(0,\frac12)$。
			\par\textbf{理由：} 直线与坐标轴交点计算。

		\item[\textbf{第三步（求面积）：}] $\triangle OMN$ 面积为 $\frac12 \times |OM| \times |ON| = \frac12 \times 1 \times \frac12 = \frac14$。
			\par\textbf{理由：} 三角形面积公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	(1) 切点弦方程为 $x-2y+1=0$；(2) $\triangle OMN$ 面积为 $\frac14$。
