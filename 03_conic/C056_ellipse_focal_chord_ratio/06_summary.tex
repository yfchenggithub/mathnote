\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{过椭圆右焦点的弦，两段焦半径的长度及其比值均可由倾斜角余弦简洁表示为 $\frac{b^2}{a-c\cos\theta}$、$\frac{b^2}{a+c\cos\theta}$ 和 $\frac{1+e\cos\theta}{1-e\cos\theta}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（椭圆标准右焦）：} 椭圆方程 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ 中 $a>b>0$，焦点在 $x$ 轴，取右焦点 $F(c,0)$，$e=c/a$。
		\item \textbf{条件2（倾斜角定义）：} 直线过 $F$，倾斜角 $\theta\in[0,\pi)$，且规定点 $A$ 对应极角 $\theta$，点 $B$ 对应 $\theta+\pi$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分左右焦点）：} 左焦点弦公式的余弦符号与右焦点相反，不可不加区分地套用。
		\item \textbf{检查项（分母恒正验证）：} 由于 $a>c$，$|c\cos\theta|\le c<a$，故公式所有分母均为正，可直接使用。
	\end{itemize}
\end{summarybox}
