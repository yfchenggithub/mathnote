\begin{examplebox}

	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^2}{25}+\frac{y^2}{9}=1$，过右焦点 $F(4,0)$ 的直线倾斜角为 $60^\circ$，求 $\frac{|AF|}{|BF|}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取基本参数）：}] 由椭圆方程得 $a=5$, $b=3$, $c=4$, 离心率 $e=\frac{c}{a}=\frac{4}{5}$；倾斜角 $\theta=60^\circ$，所以 $\cos\theta=\frac12$。
			\par\textbf{理由：}
			确定椭圆基本量是应用公式的前提。
		\item[\textbf{第二步（代入比例公式）：}] 将 $e$, $\cos\theta$ 代入焦点分弦比例公式
			\[
				\frac{|AF|}{|BF|} = \frac{1+e\cos\theta}{1-e\cos\theta}.
			\]
			\par\textbf{理由：}
			该公式直接给出了比值与 $e,\cos\theta$ 的关系。
		\item[\textbf{第三步（化简得结果）：}]
			\[
				\begin{aligned}
					\frac{|AF|}{|BF|}
					&= \frac{1+\frac45\cdot\frac12}{1-\frac45\cdot\frac12} \\
					&= \frac{1+\frac{2}{5}}{1-\frac{2}{5}} \\
					&= \frac{\frac{7}{5}}{\frac{3}{5}} \\
					&= \frac{7}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			完成分式运算并化为最简分数。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\displaystyle\frac{|AF|}{|BF|} = \frac{7}{3}$。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	椭圆 $\frac{x^2}{25}+\frac{y^2}{9}=1$，过右焦点的直线斜率为 $1$，求 $\frac{|AF|}{|BF|}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由斜率得倾斜角）：}] 直线斜率 $k=1$，故倾斜角 $\theta = 45^\circ$，$\cos\theta = \frac{\sqrt2}{2}$。
			\par\textbf{理由：}
			斜率与倾斜角的关系：$k=\tan\theta$。
		\item[\textbf{第二步（代入比例公式）：}] 已知 $e=\frac{4}{5}$，代入
			\[
				\frac{|AF|}{|BF|} = \frac{1+e\cos\theta}{1-e\cos\theta}
				= \frac{1+\frac45\cdot\frac{\sqrt2}{2}}{1-\frac45\cdot\frac{\sqrt2}{2}}.
			\]
			\par\textbf{理由：}
			直接应用焦点分弦比例公式。
		\item[\textbf{第三步（化简表达式）：}]
			\[
				\begin{aligned}
					\frac{|AF|}{|BF|}
					&= \frac{1+\frac45\cdot\frac{\sqrt2}{2}}{1-\frac45\cdot\frac{\sqrt2}{2}} \\
					&= \frac{1+\frac{2\sqrt2}{5}}{1-\frac{2\sqrt2}{5}} \\
					&= \frac{5+2\sqrt2}{5-2\sqrt2}.
			\end{aligned}
			\]
			可进一步有理化得到 $\frac{33+20\sqrt2}{17}$，但保留分式形式即可。
			\par\textbf{理由：}
			完成代数运算，结果保留简洁形式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\displaystyle\frac{|AF|}{|BF|} = \frac{5+2\sqrt2}{5-2\sqrt2}$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	椭圆 $\frac{x^2}{4}+\frac{y^2}{3}=1$ 的右焦点为 $F(1,0)$，过 $F$ 的直线交椭圆于 $A,B$，且 $|AF|:|BF|=2:1$，求直线方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由比例反设方程）：}] 设直线倾斜角为 $\theta$，离心率 $e=\frac{c}{a}=\frac{1}{2}$。由已知 $|AF|:|BF|=2:1$，即 $\frac{|AF|}{|BF|}=2$。
			\par\textbf{理由：}
			将比例转化为比值，以便使用公式。
		\item[\textbf{第二步（解出 $\cos\theta$）：}] 代入比例公式
			\[
				\frac{1+e\cos\theta}{1-e\cos\theta}=2,
			\]
			解得 $1+e\cos\theta = 2-2e\cos\theta$，即 $3e\cos\theta=1$，故 $\cos\theta = \frac{1}{3e} = \frac{1}{3\times\frac12}=\frac{2}{3}$。
			\par\textbf{理由：}
			利用比例公式建立关于 $\cos\theta$ 的方程并求解。
		\item[\textbf{第三步（求斜率得直线方程）：}]
			$\sin\theta = \sqrt{1-\cos^2\theta} = \sqrt{1-\frac{4}{9}} = \frac{\sqrt5}{3}$，
			斜率 $k=\tan\theta = \frac{\sin\theta}{\cos\theta} = \frac{\sqrt5}{2}$。
			过 $F(1,0)$，直线方程为 $y = \frac{\sqrt5}{2}(x-1)$。
			\par\textbf{理由：}
			由 $\cos\theta,\sin\theta$ 求出斜率，再根据点斜式写方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	直线方程为 $y = \frac{\sqrt5}{2}(x-1)$。
