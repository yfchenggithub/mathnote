\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（椭圆标准方程）：}}] 椭圆 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ 中 $a>b>0$，右焦点 $F(c,0)$，离心率 $e=\frac{c}{a}$。
		\item[\textbf{\textcolor{CStatement}{条件2（倾斜角定义）：}}] 过 $F$ 的弦 $AB$ 的倾斜角 $\theta\in[0,\pi)$，且点 $A$ 对应极角 $\theta$，点 $B$ 对应 $\theta+\pi$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两焦半径长度由 $e$ 和 $\cos\theta$ 决定，分母分别为 $a-c\cos\theta$ 与 $a+c\cos\theta$。
			\[
				|AF|=\frac{b^2}{a-c\cos\theta},
				\qquad
				|BF|=\frac{b^2}{a+c\cos\theta}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（分弦比例公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两段焦半径的比值仅由 $e$ 和 $\cos\theta$ 决定。
			\[
				\frac{|AF|}{|BF|}=\frac{1+e\cos\theta}{1-e\cos\theta},
				\qquad
				|AF|:|BF|=(1+e\cos\theta):(1-e\cos\theta).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（焦点弦长度）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			焦点弦总长可由焦半径求和得到，分母为平方差形式。
			\[
				\begin{aligned}
					|AB| &= |AF| + |BF| \\
					&= \frac{2ab^2}{a^2 - c^2\cos^2\theta} \\
					&= \frac{2ab^2}{a^2(1 - e^2\cos^2\theta)} .
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
