\begin{statementbox}
设椭圆 $x^2/a^2+y^2/b^2=1$，$a>b>0$，$c^2=a^2-b^2$，$e=c/a\in(0,1)$；取右焦点 $F=(c,0)$。倾斜角 $0\le\theta<\pi$ 从 $x$ 轴正向逆时针量起。
\begin{center}\input{assets/tikz/C056_statement_overview.tex}\end{center}
\textbf{方向约定：}$A=F+r(\cos\theta,\sin\theta)$，$B=F-s(\cos\theta,\sin\theta)$，$r=AF>0,s=BF>0$。先定方向，再写比值：
\[
 \boxed{\frac{AF}{BF}=\frac{a-c\cos\theta}{a+c\cos\theta}=\frac{1-e\cos\theta}{1+e\cos\theta}}.
\]
交换 $A,B$ 后，比值必须取倒数。
\end{statementbox}\clearpage\begin{statementbox}
\textbf{两段长度与整条弦长}
\[
 \boxed{AF=\frac{b^2}{a+c\cos\theta}},\qquad
 \boxed{BF=\frac{b^2}{a-c\cos\theta}}.
\]
\[
 \boxed{AB=AF+BF=\frac{2ab^2}{a^2-c^2\cos^2\theta}}.
\]
$a>c>0$ 且 $|\cos\theta|\le1$，全部分母恒正。两段都是正长度，不是有向线段；焦点在弦内，所以弦长用两段相加。

\medskip
\textbf{方向与边界速查}
\[
 \theta=0:\ AF=a-c,\ BF=a+c;\qquad
 \theta=\frac\pi2:\ AF=BF=\frac{b^2}{a}.
\]
当 $\theta$ 为钝角时，正向端 $A$ 位于向左的射线，$AF>BF$。

\end{statementbox}
