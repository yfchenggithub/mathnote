\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知定点$F(2,0)$和定直线$\ell:x=-2$，动点$P$满足$PF$等于点$P$到$\ell$的距离。求动点$P$的轨迹方程，并指出轨迹名称。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设点列式）：}] 设$P(x,y)$，则$PF=\sqrt{(x-2)^2+y^2}$，$d(P,\ell)=|x+2|$。由条件得
			\[
				\sqrt{(x-2)^2+y^2}=|x+2|.
			\]
			\par\textbf{理由：} 将几何条件翻译成代数等式，利用两点间距离公式和点到直线距离公式。

		\item[\textbf{第二步（化简方程）：}] 两边平方并整理：
			\[
				\begin{aligned}
					(x-2)^2+y^2 &= (x+2)^2,\\
					x^2-4x+4+y^2 &= x^2+4x+4,\\
					y^2 &= 8x.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方消去根号和绝对值，得到最简方程。

		\item[\textbf{第三步（判定轨迹）：}] 轨迹方程为$y^2=8x$，是以$F(2,0)$为焦点、$\ell:x=-2$为准线的抛物线（开口向右）。
			\par\textbf{理由：} 标准形式$y^2=2px$对应$p=4$，符合抛物线定义。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y^2=8x$}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知点$F(0,1)$和直线$\ell:y=-1$，动点$P$满足$PF$等于点$P$到$\ell$的距离。求点$P$的轨迹方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设点列式）：}] 设$P(x,y)$，则$PF=\sqrt{x^2+(y-1)^2}$，$d(P,\ell)=|y+1|$。条件为
			\[
				\sqrt{x^2+(y-1)^2}=|y+1|.
			\]
			\par\textbf{理由：} 用坐标表示距离。

		\item[\textbf{第二步（平方化简）：}] 平方并整理：
			\[
				\begin{aligned}
					x^2+(y-1)^2 &= (y+1)^2,\\
					x^2+y^2-2y+1 &= y^2+2y+1,\\
					x^2 &= 4y.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方后展开，消去同类项。

		\item[\textbf{第三步（结论）：}] 轨迹方程为$x^2=4y$，是焦点在$y$轴正半轴、开口向上的抛物线。
			\par\textbf{理由：} 标准形式$x^2=4py$对应$p=1$，符合抛物线定义。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x^2=4y$}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线$y^2=4x$的焦点为$F$，定点$A(3,2)$，$P$为抛物线上的动点，求$|PA|+|PF|$的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定基本量）：}] 由$y^2=4x$得$p=2$，焦点$F(1,0)$，准线$\ell:x=-1$。
			\par\textbf{理由：} 标准方程$y^2=2px$，对应$2p=4$。

		\item[\textbf{第二步（定义转化）：}] 根据抛物线定义，$|PF|$等于点$P$到准线$\ell$的距离$d(P,\ell)$。故
			\[
				|PA|+|PF| = |PA|+d(P,\ell).
			\]
			\par\textbf{理由：} 利用定义将焦半径$|PF|$换为到准线的距离，便于用几何法求最值。

		\item[\textbf{第三步（几何最值）：}] 过点$A$作准线$\ell$的垂线，垂足为$H(-1,2)$。当$P$为线段$AH$与抛物线的交点时，$|PA|+d(P,\ell)$取最小值，等于$|AH|$。计算得
			\[
				\begin{aligned}
					|AH| &= |3-(-1)| \\
					&= 4.
			\end{aligned}
			\]
			所以$|PA|+|PF|$的最小值为$4$。
			\par\textbf{理由：} 点到直线的垂线段最短。$A$到准线的距离即为$|AH|$，此时$P$在$AH$与抛物线的交点处（该交点存在且唯一）。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$4$}
\end{examplebox}
