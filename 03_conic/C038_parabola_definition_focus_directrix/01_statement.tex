\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定点与定线）：}}] 给定平面内一定点$F$和一定直线$\ell$，且$F\notin\ell$。
		\item[\textbf{\textcolor{CStatement}{条件2（动点约束）：}}] 动点$P$满足$PF=PH$，其中$H$为点$P$在直线$\ell$上的垂足。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（充要条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点$P$在抛物线上的充要条件是$PF$等于$P$到$\ell$的距离。
			\[
				PF=PH.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（离心率为1）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			抛物线上任意一点到焦点的距离与到准线的距离之比恒为1。
			\[
				\frac{PF}{PH}=1,
			\]
			即离心率$e=1$。
	\end{description}
\end{statementbox}
