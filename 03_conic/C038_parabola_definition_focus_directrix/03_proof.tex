\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过建立恰当的直角坐标系，将几何条件转化为代数等式，化简得到抛物线标准方程，从而证明等价性。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（建系设点）：}] 以$F$到$\ell$的垂线段的中点为原点$O$，以从$F$到$\ell$的方向为$x$轴正方向建立直角坐标系。
			\[
				F\!\left(\frac{p}{2},0\right),
				\qquad
				\ell:x=-\frac{p}{2},
			\]
			其中$p>0$为$F$到$\ell$的距离。
			\par\textbf{理由：}
			对称的坐标系使抛物线方程最简。

		\item[\textbf{步骤二（等距条件）：}] 设动点$P(x,y)$，垂足$H(-\frac{p}{2},y)$。由定义$PF=PH$得
			\[
				\sqrt{\left(x-\frac{p}{2}\right)^2+y^2}=\left|x+\frac{p}{2}\right|.
			\]
			\par\textbf{理由：}
			点$P$到直线$x=-\frac{p}{2}$的水平距离为$|x+\frac{p}{2}|$。

		\item[\textbf{步骤三（平方化简）：}] 两边平方并展开：
			\[
				\begin{aligned}
					\left(x-\frac{p}{2}\right)^2+y^2 &= \left(x+\frac{p}{2}\right)^2,\\
					x^2-px+\frac{p^2}{4}+y^2 &= x^2+px+\frac{p^2}{4},\\
					-px + y^2 &= px.
			\end{aligned}
			\]
			\par\textbf{理由：}
			平方消去根号和绝对值的分类讨论。

		\item[\textbf{步骤四（标准形式）：}] 移项合并得
			\[
				y^2 = 2px.
			\]
			该方程是焦点在$x$轴正半轴的标准抛物线方程。
			\par\textbf{理由：}
			方程$y^2=2px$表示顶点在原点、开口向右的抛物线。

		\item[\textbf{步骤五（逆推验证）：}] 反之，若点$P$在抛物线$y^2=2px$上，则
			\[
				\begin{aligned}
					PF &= \sqrt{\left(x-\frac{p}{2}\right)^2+y^2} \\
					&= \sqrt{\left(x-\frac{p}{2}\right)^2+2px} \\
					&= \sqrt{\left(x+\frac{p}{2}\right)^2} \\
					&= \left|x+\frac{p}{2}\right| \\
					&= PH.
			\end{aligned}
			\]
			\par\textbf{理由：}
			代入$y^2=2px$可消去$y$，并回代得到与$PH$相等的表达式。

	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，$PF=PH$等价于点$P$在抛物线$y^2=2px$上，即原命题成立。}
\end{proofbox}
