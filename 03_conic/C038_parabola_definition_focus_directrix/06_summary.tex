\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{抛物线是到定点（焦点）与定直线（准线）距离相等的点的轨迹。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（定点与定线）：} 平面内存在一个定点$F$和一条定直线$\ell$，且$F\notin\ell$。
		\item \textbf{条件2（等距条件）：} 动点$P$满足$PF$等于$P$到$\ell$的垂线段长度。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略前提）：} 使用定义前必须验证$F\notin\ell$，否则轨迹是直线而非抛物线。
		\item \textbf{检查项（定义转化）：} 利用定义将抛物线上的点到焦点的距离转化为到准线的距离时，需确认抛物线标准方程中$p$的值及焦点、准线坐标。
	\end{itemize}
\end{summarybox}
