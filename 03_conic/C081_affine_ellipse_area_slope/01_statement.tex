\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（椭圆标准形）：}}]
			椭圆方程 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$ 满足 $a>b>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（仿射变换）：}}]
			变换 $\varphi: x'=x,\ y'=\frac{a}{b}y$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（化圆）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆经仿射变换后成为半径为 $a$ 的圆。
			\[
				x'^2+y'^2=a^2
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（面积关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			变换后图形面积为原面积的 $a/b$ 倍。
			\[
				S=\frac{b}{a}S'
			\]
			（或等价 $S'=\frac{a}{b}S$）
		\item[\textbf{\textcolor{CStatement}{结论三（斜率关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			变换后直线斜率变为原斜率的 $a/b$ 倍。
			\[
				k=\frac{b}{a}k'
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（不变性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			仿射变换保持直线平行关系和共线点分线段的比例不变。
			\[
				l_1\parallel l_2 \;\Rightarrow\; l'_1\parallel l'_2,
				\qquad
				\frac{AP}{PB}=\lambda \;\Rightarrow\; \frac{A'P'}{P'B'}=\lambda .
			\]
	\end{description}

\end{statementbox}
