\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（面积方向错误）}}]
			认为 $S' = \frac{b}{a} S$，混淆缩放方向。
		\item[\textbf{\textcolor{CTrap}{正确理解（面积缩放方向）：}}]
			变换后 $y$ 伸长，面积变大，故 $S' = \frac{a}{b} S$，因此 $S = \frac{b}{a} S'$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略拉伸）：}}]
			未理解拉伸方向，误认为面积缩小。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（斜率公式记反）}}]
			认为 $k' = \frac{b}{a}k$ 或 $k = \frac{a}{b}k'$。
		\item[\textbf{\textcolor{CTrap}{正确理解（斜率缩放方向）：}}]
			$y'$ 拉伸，斜率 $k' = \frac{a}{b}k$，故 $k = \frac{b}{a} k'$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆变量）：}}]
			混淆了 $k$ 与 $k'$ 的关系，未从变换定义推导。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（滥用垂直关系）}}]
			误以为圆中垂直弦在椭圆中仍垂直。
		\item[\textbf{\textcolor{CTrap}{正确理解（保平行不保角）：}}]
			仿射变换不保角，仅保平行和比例，因此圆中垂直关系在椭圆中不再成立（除非特殊方向）。
		\item[\textbf{\textcolor{CTrap}{错因分析（类推错误）：}}]
			忽略仿射变换的角度扭曲，错误地将圆中的垂直结论推广到椭圆。
	\end{description}

\end{trapbox}
