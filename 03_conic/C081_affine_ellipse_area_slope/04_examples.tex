\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $C:\frac{x^2}{9}+\frac{y^2}{4}=1$，作仿射变换 $\varphi: x'=x,\ y'=\frac{3}{2}y$。 (1) 求变换后的曲线方程； (2) 若椭圆 $C$ 内有一三角形面积为 $6$，求变换后该三角形面积。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			由椭圆方程得 $a=3,b=2$，所以 $\frac{a}{b}=\frac{3}{2}$，变换 $\varphi$ 符合条件。
			\par\textbf{理由：} 确定 $a,b$ 以应用公式。
		\item[\textbf{第二步（化圆方程）：}]
			代入变换：由 $x'=x,\ y'=\frac{3}{2}y$ 得 $x=x',\ y=\frac{2}{3}y'$，代入椭圆方程：
			\[
				\begin{aligned}
					\frac{x'^2}{9}+\frac{(\frac{2}{3}y')^2}{4}=1 &\Rightarrow \frac{x'^2}{9}+\frac{y'^2}{9}=1 \\ &\Rightarrow x'^2+y'^2=9
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 代入后化简得到圆方程。
		\item[\textbf{第三步（面积计算）：}]
			由面积关系 $S=\frac{b}{a}S'$，现 $S=6$，故
			\[
				\begin{aligned}
					S'&=\frac{a}{b}S \\ &=\frac{3}{2}\times6 \\ &=9
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 直接利用面积缩放公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{变换后曲线为圆 $x'^2+y'^2=9$，变换后三角形面积为 $9$。}
\end{examplebox}

\begin{examplebox}
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 已知椭圆 $C:\frac{x^2}{16}+\frac{y^2}{9}=1$，直线 $l: y=\frac{1}{2}x+1$，作仿射变换 $\varphi: x'=x,\ y'=\frac{4}{3}y$。求直线 $l$ 变换后的方程及其斜率。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变换参数）：}]
			$a=4,b=3$，所以 $\frac{a}{b}=\frac{4}{3}$，变换 $\varphi: y'=\frac{4}{3}y$。
			\par\textbf{理由：} 确定缩放因子。
		\item[\textbf{第二步（代入变换）：}]
			由 $y'=\frac{4}{3}y$ 得 $y=\frac{3}{4}y'$，代入 $l$ 方程：$\frac{3}{4}y'=\frac{1}{2}x+1$，解得 $y'=\frac{2}{3}x+\frac{4}{3}$。又 $x'=x$，故 $l'$ 方程为 $y'=\frac{2}{3}x'+\frac{4}{3}$。
			\par\textbf{理由：} 消去 $x,y$ 得到 $x',y'$ 关系。
		\item[\textbf{第三步（读取斜率）：}]
			$l'$ 斜率为 $\frac{2}{3}$。
			\par\textbf{理由：} 斜率为 $x'$ 系数。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{变换后直线方程 $y'=\frac{2}{3}x'+\frac{4}{3}$，斜率 $\frac{2}{3}$。}
\end{examplebox}

\begin{examplebox}
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 求椭圆 $C:\frac{x^2}{4}+\frac{y^2}{3}=1$ 内接三角形面积的最大值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作仿射变换）：}]
			取变换 $\varphi: x'=x,\ y'=\frac{a}{b}y=\frac{2}{\sqrt{3}}y$，椭圆化为圆 $x'^2+y'^2=4$（$a=2,b=\sqrt{3}$）。
			\par\textbf{理由：} 利用仿射变换将椭圆变圆，且面积关系已知。
		\item[\textbf{第二步（圆内接三角形最值）：}]
			圆 $x'^2+y'^2=4$ 的内接三角形中，等边三角形的面积最大，最大面积
			\[
				\begin{aligned}
					S'_{\max}&=\frac{3\sqrt{3}}{4}R^2 \\ &=\frac{3\sqrt{3}}{4}\times4 \\ &=3\sqrt{3}
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 圆内接正三角形面积公式。
		\item[\textbf{第三步（反求椭圆内面积）：}]
			由面积关系 $S=\frac{b}{a}S'$，得椭圆内接三角形最大面积
			\[
				\begin{aligned}
					S_{\max}&=\frac{b}{a}S'_{\max} \\ &=\frac{\sqrt{3}}{2}\times3\sqrt{3} \\ &=\frac{9}{2}
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 利用面积缩放公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最大面积为 $\frac{9}{2}$。}
\end{examplebox}
