\begin{examplebox}

	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2 = 8x$ 的焦点为 $F$，过 $F$ 的弦 $AB$ 满足 $|AF| = 6$，求 $|BF|$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}] 由 $y^2=8x$ 知 $2p=8$，故 $p=4$，焦点 $F(2,0)$。
			\par\textbf{理由：} 确定 $p$ 是运用结论 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{p}$ 的前提。
		\item[\textbf{第二步（套用结论）：}] 由焦点弦倒数和定值及 $p=4$，得 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac12$。代入 $|AF|=6$，有 $\frac16+\frac{1}{|BF|}=\frac12$。
			\par\textbf{理由：} 结论直接给出了焦半径倒数和与 $p$ 的关系，无需重新推导。
		\item[\textbf{第三步（解出结果）：}] 移项得 $\frac{1}{|BF|}=\frac13$，故 $|BF|=3$。
			\par\textbf{理由：} 简单分式运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$|BF|=3$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	设抛物线 $y^2=2px$（$p>0$）的焦点为 $F$，过 $F$ 的弦 $AB$ 满足 $|AF|=3$，$|BF|=6$，求 $p$ 的值及抛物线方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入定值公式）：}] 由焦点弦性质得 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{p}$，代入 $|AF|=3$，$|BF|=6$：
			\[
				\begin{aligned}
					\frac13+\frac16 &= \frac12, \\
					\frac{2}{p} &= \frac12.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接利用结论，将所给线段长度代入，建立关于 $p$ 的方程。
		\item[\textbf{第二步（求解参数 $p$）：}] 解 $\frac{2}{p}=\frac12$，交叉相乘得 $p=4$。
			\par\textbf{理由：} 一次分式方程，直接求解。
		\item[\textbf{第三步（写出方程）：}] 由 $p=4$ 得抛物线方程为 $y^2=8x$。
			\par\textbf{理由：} 标准方程形式 $y^2=2px$ 代入即可。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$p=4$，抛物线方程为 $y^2=8x$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知抛物线 $y^2=4x$ 的焦点为 $F$，过 $F$ 的弦 $AB$ 与抛物线交于 $A,B$ 两点，求 $|AF|+|BF|$ 的最小值，并求此时直线 $AB$ 的方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（利用定值准备）：}] 由抛物线 $y^2=4x$ 得 $p=2$，故 $\frac{1}{|AF|}+\frac{1}{|BF|}=1$。
			\par\textbf{理由：} 先得到倒数和为 $1$，这是使用均值不等式的定量前提。
		\item[\textbf{第二步（均值不等式求最值）：}] 因为 $|AF|>0,\,|BF|>0$，由均值不等式得
			\[
				(|AF|+|BF|)\left(\frac{1}{|AF|}+\frac{1}{|BF|}\right)\ge4,
			\]
			等号当 $|AF|=|BF|$ 时成立。代入已知 $\frac{1}{|AF|}+\frac{1}{|BF|}=1$，得 $|AF|+|BF|\ge4$。故最小值为 $4$。
			\par\textbf{理由：} 利用倒数定值与均值不等式联用，是解析几何求最值的典型技巧。
		\item[\textbf{第三步（取等条件与弦方程）：}] 当 $|AF|=|BF|$ 时取等，此时 $\frac{2}{|AF|}=1$，解得 $|AF|=2$，从而 $|BF|=2$。因此 $A,B$ 关于 $x$ 轴对称，直线 $AB$ 垂直于 $x$ 轴，方程为 $x=1$（因为焦点为 $(1,0)$）。
			\par\textbf{理由：} 由抛物线对称性，当 $|AF|=|BF|$ 时弦垂直于 $x$ 轴，过焦点且垂直 $x$ 轴的直线即通径所在直线。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$|AF|+|BF|$ 的最小值为 $4$，此时直线 $AB$ 的方程为 $x=1$。
\end{examplebox}
