\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{抛物线过焦点的弦，其焦半径 $|AF|,|BF|$ 的倒数之和等于常数 $\dfrac{2}{p}$，与弦的斜率无关。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准方程）：} 抛物线必须是开口向右的标准形式 $y^2 = 2px$（$p>0$），焦点 $F(\frac{p}{2},0)$。
		\item \textbf{条件 2（弦过焦点）：} 弦 $AB$ 经过焦点 $F$，且 $A,B$ 为抛物线上两个不同的点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式与开口）：} 易将 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{2}{p}$ 记成 $\frac{1}{|AF|}+\frac{1}{|BF|}=\frac{p}{2}$ 或 $|AF|+\frac{1}{|BF|}=\frac{2}{p}$；易套用到开口向左或上下的抛物线。
		\item \textbf{检查项（斜率不存在）：} 使用直线点斜式设 $y=k(x-\frac{p}{2})$ 时，必须单独验证 $k$ 不存在（即 $AB\perp x$ 轴）的情况，此时结论依然成立。
	\end{itemize}
\end{summarybox}
