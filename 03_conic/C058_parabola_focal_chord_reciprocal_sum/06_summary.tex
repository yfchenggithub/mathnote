% C058：考前 30 秒复习清单。
\begin{summarybox}
\textbf{\textcolor{CSummary}{30 秒核心卡：一个条件、一个公式、一个样板}}

\textbf{条件：}$y^2=2px$（$p>0$），$AB$ 为\textbf{真正}经过 $F(\frac p2,0)$ 的两端点焦点弦（$F$ 在 $A,B$ 中间）。
\[
 \boxed{\frac1{|AF|}+\frac1{|BF|}=\frac2p}.
\]
\textbf{样板：}通径 $x=\frac p2$，$|AF|=|BF|=p$，所以 $1/p+1/p=2/p$。

\medskip
\textbf{\textcolor{CSummary}{最快重建（角度法，2 行）}}

取 $r=|AF|,s=|BF|$，$FA$ 与正 $x$ 轴夹角 $\theta$，焦点两侧方向相反：
\[
 r=p+r\cos\theta,\qquad s=p-s\cos\theta.
\]
\[
 \frac1r=\frac{1-\cos\theta}p,\qquad
 \frac1s=\frac{1+\cos\theta}p\quad\Longrightarrow\quad\frac1r+\frac1s=\frac2p.
\]

\medskip
\textbf{\textcolor{CSummary}{最低门槛重建（不用三角函数）}}

焦半径给出 $x_A=r-\frac p2$、$x_B=s-\frac p2$；相反方向的横向投影比互为相反数：
\[
 \frac{x_A-\frac p2}r=-\frac{x_B-\frac p2}s
 \ \Longrightarrow\ \frac{r-p}r=-\frac{s-p}s
 \ \Longrightarrow\ \boxed{\frac1r+\frac1s=\frac2p}.
\]

\medskip
\textbf{\textcolor{CSummary}{联动结论}}
\[
 rs=\frac p2(r+s)=\frac p2|AB|,\qquad
 x_1x_2=\frac{p^2}{4},\qquad |AB|\ge2p.
\]
等号在\textbf{通径} $x=\frac p2$ 处成立。四种轴向开口（$p>0$）倒数和形式相同。

\medskip
\textbf{\textcolor{CSummary}{最容易犯的三种错}}
\begin{itemize}[leftmargin=*,itemsep=0.35em]
 \item \textbf{记错参数：}$y^2=4ax$ 时 $p=2a$，故倒数和为 $1/a$。
 \item \textbf{漏解／退化：}竖直通径属于焦点弦；水平线 $y=0$ 仅有一个有限交点，\textbf{不属于}焦点弦。
 \item \textbf{误判逆命题：}倒数和等于 $2/p$ 或 $x_1x_2=p^2/4$ \textbf{都不单独保证} $AB$ 过焦点。
\end{itemize}

\textbf{最终口诀：}\textbf{“准线定焦半径，焦点两侧反方向；倒数相加正负消，通径一验就不会忘。”}
\end{summarybox}
