\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	在平面内，到两定点距离之比为常数（$\lambda>0,\lambda\neq1$）的点的轨迹是圆，称为阿波罗尼斯圆；圆心在两定点所在的直线上.

	\medskip

	\textbf{\textcolor{CSummary}{标准坐标公式}}

	若
	\[
		A(-a,0),\qquad B(a,0),\qquad a>0,
	\]
	且
	\[
		\frac{PA}{PB}=\lambda,\qquad \lambda>0,\quad \lambda\neq1,
	\]
	则轨迹方程为
	\[
		\left(x-\frac{\lambda^2+1}{\lambda^2-1}a\right)^2+y^2
		=
		\left(\frac{2a\lambda}{\lambda^2-1}\right)^2.
	\]
	圆心与半径分别为
	\[
		C\left(\frac{\lambda^2+1}{\lambda^2-1}a,\;0\right),
		\qquad
		R=\left|\frac{2a\lambda}{\lambda^2-1}\right|.
	\]

	\medskip

	\textbf{\textcolor{CSummary}{一般坐标公式}}

	若
	\[
		A(x_1,y_1),\qquad B(x_2,y_2),\qquad A\neq B,
	\]
	则可先写距离方程
	\[
		(x-x_1)^2+(y-y_1)^2
		=
		\lambda^2\bigl[(x-x_2)^2+(y-y_2)^2\bigr].
	\]
	当 $\lambda>0$ 且 $\lambda\neq1$ 时，整理后仍为圆，其圆心和半径为
	\[
		C\left(
		\frac{x_1-\lambda^2x_2}{1-\lambda^2},
		\frac{y_1-\lambda^2y_2}{1-\lambda^2}
		\right),
		\qquad
		R=
		\frac{\lambda\sqrt{(x_1-x_2)^2+(y_1-y_2)^2}}{|1-\lambda^2|}.
	\]

	\medskip

	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点一（$\lambda=1$ 例外）：} $\lambda=1$ 时轨迹是线段 $AB$ 的中垂线，不可使用阿波罗尼斯圆公式.
		\item \textbf{易错点二（半径符号）：} 半径必须取正值，表达式
		      \[
			      \left|\frac{2a\lambda}{\lambda^2-1}\right|
		      \]
		      不能漏掉绝对值.
		\item \textbf{易错点三（非标准坐标）：} 两定点不是 $(-a,0)$ 和 $(a,0)$ 时，不要机械套标准公式，应先标准化或直接用一般坐标方程.
		\item \textbf{易错点四（空间问题）：} 平面内轨迹是圆；空间任意动点的对应轨迹一般是球面，不能混淆.
	\end{itemize}
\end{summarybox}
