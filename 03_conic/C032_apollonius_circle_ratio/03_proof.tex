\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	设点 $P(x,y)$，利用 $PA^2=\lambda^2PB^2$ 列出方程，整理后配方即得圆方程.

	\medskip

	\textbf{\textcolor{CProof}{正式推导}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设点列式）：}]
		      \[
			      PA=\sqrt{(x+a)^2+y^2},\qquad
			      PB=\sqrt{(x-a)^2+y^2}.
		      \]
		      \par\textbf{理由：} 两点距离公式.

		\item[\textbf{步骤二（平方去根号）：}]
		      由
		      \[
			      \frac{PA}{PB}=\lambda
		      \]
		      得
		      \[
			      (x+a)^2+y^2
			      =
			      \lambda^2\bigl[(x-a)^2+y^2\bigr].
		      \]
		      \par\textbf{理由：} 比值 $\dfrac{PA}{PB}$ 有意义，所以 $PB\neq0$；又因为 $\lambda>0$，满足条件的点不可能使 $PA=0$. 因此两边均为非负量，可以平方，且不会引入异号问题.

		\item[\textbf{步骤三（展开移项）：}]
		      \[
			      \begin{aligned}
				      x^2+2ax+a^2+y^2
				       & =
				      \lambda^2(x^2-2ax+a^2+y^2) \\
				       & =
				      \lambda^2x^2-2a\lambda^2x+\lambda^2a^2+\lambda^2y^2.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 展开平方项.

		\item[\textbf{步骤四（合并同类项）：}]
		      将所有项移至左边，得
		      \[
			      x^2-\lambda^2x^2+2ax+2a\lambda^2x+a^2-\lambda^2a^2+y^2-\lambda^2y^2=0.
		      \]
		      合并同类项：
		      \[
			      (1-\lambda^2)x^2+2a(1+\lambda^2)x+(1-\lambda^2)y^2+(1-\lambda^2)a^2=0.
		      \]
		      \par\textbf{理由：} 将同类项合并，得到关于 $x,y$ 的二次方程.

		\item[\textbf{步骤五（除以系数并配方）：}]
		      因为 $\lambda\neq1$，所以 $1-\lambda^2\neq0$. 两边除以 $1-\lambda^2$，得
		      \[
			      x^2+\frac{2a(1+\lambda^2)}{1-\lambda^2}x+y^2+a^2=0.
		      \]
		      注意
		      \[
			      \frac{1+\lambda^2}{1-\lambda^2}
			      =
			      -\frac{\lambda^2+1}{\lambda^2-1}.
		      \]
		      令
		      \[
			      M=\frac{\lambda^2+1}{\lambda^2-1}a,
		      \]
		      则原方程可写为
		      \[
			      x^2-2Mx+y^2+a^2=0.
		      \]
		      对 $x$ 项配方：
		      \[
			      x^2-2Mx=(x-M)^2-M^2.
		      \]
		      因此
		      \[
			      (x-M)^2+y^2=M^2-a^2.
		      \]
		      计算右边：
		      \[
			      \begin{aligned}
				      M^2-a^2
				       & =
				      \left(\frac{\lambda^2+1}{\lambda^2-1}\right)^2a^2-a^2 \\
				       & =
				      a^2\left[
					      \frac{(\lambda^2+1)^2-(\lambda^2-1)^2}{(\lambda^2-1)^2}
				      \right]                                               \\
				       & =
				      a^2\cdot\frac{4\lambda^2}{(\lambda^2-1)^2}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 移项、配方，并利用平方差公式
		      \[
			      (\lambda^2+1)^2-(\lambda^2-1)^2=4\lambda^2.
		      \]

		\item[\textbf{步骤六（写出标准方程）：}]
		      所以
		      \[
			      \left(x-\frac{\lambda^2+1}{\lambda^2-1}a\right)^2+y^2
			      =
			      \left(\frac{2a\lambda}{\lambda^2-1}\right)^2.
		      \]
		      因此圆心为
		      \[
			      C\left(\frac{\lambda^2+1}{\lambda^2-1}a,\;0\right),
		      \]
		      半径为
		      \[
			      R=\left|\frac{2a\lambda}{\lambda^2-1}\right|.
		      \]
		      \par\textbf{理由：} 与圆的标准方程
		      \[
			      (x-h)^2+(y-k)^2=r^2
		      \]
		      对比即可.
	\end{description}

	\medskip

	\textbf{\textcolor{CProof}{结论回扣}}

	综上，到两定点 $A(-a,0),B(a,0)$ 距离之比为常数 $\lambda>0,\lambda\neq1$ 的点 $P$ 的轨迹，是以
	\[
		C\left(\frac{\lambda^2+1}{\lambda^2-1}a,\;0\right)
	\]
	为圆心，以
	\[
		\left|\frac{2a\lambda}{\lambda^2-1}\right|
	\]
	为半径的圆. 由于 $a>0,\lambda>0,\lambda\neq1$，该半径严格为正.
\end{proofbox}
