\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $A(-1,0),\;B(1,0)$，动点 $P$ 满足 $\dfrac{PA}{PB}=2$，求点 $P$ 的轨迹方程.

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
		      由 $A(-1,0),B(1,0)$ 可知 $a=1$；由 $\dfrac{PA}{PB}=2$ 可知 $\lambda=2$.
		      \par\textbf{理由：} 定点坐标与比例常数直接读出.

		\item[\textbf{第二步（算圆心半径）：}]
		      \[
			      x_C=\frac{\lambda^2+1}{\lambda^2-1}a
			      =
			      \frac{4+1}{4-1}\times1
			      =
			      \frac53,
		      \]
		      \[
			      R=\left|\frac{2a\lambda}{\lambda^2-1}\right|
			      =
			      \left|\frac{2\times1\times2}{3}\right|
			      =
			      \frac43.
		      \]
		      \par\textbf{理由：} 直接代入标准坐标下的阿波罗尼斯圆公式.

		\item[\textbf{第三步（写方程）：}]
		      \[
			      \left(x-\frac53\right)^2+y^2=\left(\frac43\right)^2.
		      \]
		      \par\textbf{理由：} 圆的标准方程形式.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 所求轨迹是圆心 $\left(\dfrac53,0\right)$、半径 $\dfrac43$ 的圆.

	\medskip

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $A(-2,0),\;B(2,0)$，动点 $P$ 满足 $PA:PB=1:2$，求点 $P$ 的轨迹方程.

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（转化比值）：}]
		      由 $PA:PB=1:2$ 得
		      \[
			      \frac{PA}{PB}=\frac12.
		      \]
		      故
		      \[
			      a=2,\qquad \lambda=\frac12.
		      \]
		      \par\textbf{理由：} 比值转化为标准形式.

		\item[\textbf{第二步（算圆心半径）：}]
		      \[
			      x_C=
			      \frac{\lambda^2+1}{\lambda^2-1}a
			      =
			      \frac{\frac14+1}{\frac14-1}\times2
			      =
			      \frac{5/4}{-3/4}\times2
			      =
			      -\frac{10}{3}.
		      \]
		      \[
			      R=
			      \left|\frac{2a\lambda}{\lambda^2-1}\right|
			      =
			      \left|\frac{2\times2\times\frac12}{-\frac34}\right|
			      =
			      \frac83.
		      \]
		      \par\textbf{理由：} 代入公式，注意 $\lambda^2-1<0$，但半径必须取正值.

		\item[\textbf{第三步（写方程）：}]
		      \[
			      \left(x+\frac{10}{3}\right)^2+y^2=\left(\frac83\right)^2.
		      \]
		      \par\textbf{理由：} 圆的标准方程.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 圆心 $\left(-\dfrac{10}{3},0\right)$，半径 $\dfrac83$ 的圆.

	\medskip

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $A(-3,0),\;B(3,0)$，点 $P$ 满足 $\dfrac{PA}{PB}=2$ 且点 $P$ 在直线 $y=x$ 上，求点 $P$ 的坐标.

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求轨迹方程）：}]
		      由阿波罗尼斯圆公式，$a=3,\lambda=2$，得
		      \[
			      x_C=\frac{4+1}{4-1}\times3=5,
			      \qquad
			      R=\left|\frac{2\times3\times2}{3}\right|=4.
		      \]
		      所以轨迹方程为
		      \[
			      (x-5)^2+y^2=16.
		      \]
		      \par\textbf{理由：} 先由距离比定出圆的方程.

		\item[\textbf{第二步（联立直线）：}]
		      点 $P$ 在 $y=x$ 上，代入圆方程得
		      \[
			      (x-5)^2+x^2=16.
		      \]
		      \par\textbf{理由：} 消去 $y$.

		\item[\textbf{第三步（解方程求坐标）：}]
		      展开得
		      \[
			      x^2-10x+25+x^2=16,
		      \]
		      即
		      \[
			      2x^2-10x+9=0.
		      \]
		      解得
		      \[
			      x=\frac{5\pm\sqrt7}{2}.
		      \]
		      又 $y=x$，故
		      \[
			      P\left(\frac{5+\sqrt7}{2},\frac{5+\sqrt7}{2}\right)
			      \quad\text{或}\quad
			      P\left(\frac{5-\sqrt7}{2},\frac{5-\sqrt7}{2}\right).
		      \]
		      \par\textbf{理由：} 解一元二次方程.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 存在两个满足条件的点 $P$，坐标分别为
	\[
		\left(\frac{5+\sqrt7}{2},\frac{5+\sqrt7}{2}\right)
		\quad\text{和}\quad
		\left(\frac{5-\sqrt7}{2},\frac{5-\sqrt7}{2}\right).
	\]

	\medskip

	\textbf{\textcolor{CExample}{例 4（一般坐标版）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $A(0,0),\;B(4,0)$，动点 $P$ 满足 $\dfrac{PA}{PB}=2$，求点 $P$ 的轨迹方程.

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（不要硬套标准公式）：}]
		      这里两定点不是 $(-a,0)$ 与 $(a,0)$ 的形式，所以不应直接套标准坐标公式. 设 $P(x,y)$，直接由距离比列式：
		      \[
			      \sqrt{x^2+y^2}
			      =
			      2\sqrt{(x-4)^2+y^2}.
		      \]
		      \par\textbf{理由：} 非对称坐标下，直接用距离公式更稳.

		\item[\textbf{第二步（平方整理）：}]
		      两边平方得
		      \[
			      x^2+y^2=4\bigl[(x-4)^2+y^2\bigr].
		      \]
		      展开并整理：
		      \[
			      x^2+y^2=4x^2-32x+64+4y^2,
		      \]
		      即
		      \[
			      3x^2+3y^2-32x+64=0.
		      \]
		      \par\textbf{理由：} 将根号方程转化为二次方程.

		\item[\textbf{第三步（配方成圆）：}]
		      两边除以 $3$：
		      \[
			      x^2+y^2-\frac{32}{3}x+\frac{64}{3}=0.
		      \]
		      配方得
		      \[
			      \left(x-\frac{16}{3}\right)^2+y^2
			      =
			      \frac{64}{9}.
		      \]
		      \par\textbf{理由：} 将二次方程化为圆的标准方程.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 所求轨迹是圆心 $\left(\dfrac{16}{3},0\right)$、半径 $\dfrac83$ 的圆.
\end{examplebox}
