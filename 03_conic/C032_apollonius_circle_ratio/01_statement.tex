\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（标准坐标下的两定点）：}}]
		      平面上给定两点 $A(-a,0)$ 和 $B(a,0)$，其中 $a>0$.

		\item[\textbf{\textcolor{CStatement}{条件 2（距离比）：}}]
		      动点 $P$ 满足
		      \[
			      \frac{PA}{PB}=\lambda,
		      \]
		      其中常数 $\lambda>0$ 且 $\lambda\neq1$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（轨迹形状）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 点 $P$ 的轨迹是一个圆，圆心在 $x$ 轴上.
		      \[
			      \text{圆心 } C\left(\frac{\lambda^2+1}{\lambda^2-1}a,\;0\right),
			      \qquad
			      \text{半径 } R=\left|\frac{2a\lambda}{\lambda^2-1}\right|.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（标准轨迹方程）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 将距离比平方后化简并配方，可得到圆的标准方程.
		      \[
			      \left(x-\frac{\lambda^2+1}{\lambda^2-1}a\right)^2+y^2
			      =
			      \left(\frac{2a\lambda}{\lambda^2-1}\right)^2.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论三（一般坐标版）：}}]
		      若两定点为 $A(x_1,y_1)$，$B(x_2,y_2)$，且 $A\neq B$，动点 $P(x,y)$ 满足
		      \[
			      \frac{PA}{PB}=\lambda,\qquad \lambda>0,\quad \lambda\neq1,
		      \]
		      则轨迹方程可先写为
		      \[
			      (x-x_1)^2+(y-y_1)^2
			      =
			      \lambda^2\bigl[(x-x_2)^2+(y-y_2)^2\bigr].
		      \]
		      整理后仍为圆，其圆心和半径为
		      \[
			      C\left(
			      \frac{x_1-\lambda^2x_2}{1-\lambda^2},
			      \frac{y_1-\lambda^2y_2}{1-\lambda^2}
			      \right),
			      \qquad
			      R=
			      \frac{\lambda\sqrt{(x_1-x_2)^2+(y_1-y_2)^2}}{|1-\lambda^2|}.
		      \]
	\end{description}
\end{statementbox}
