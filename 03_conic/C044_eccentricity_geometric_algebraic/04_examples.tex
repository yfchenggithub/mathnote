\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知椭圆$C:\frac{x^{2}}{25}+\frac{y^{2}}{16}=1$，求其离心率$e$。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（识别参数）：}]
			从方程中找到$a^{2}=25$，$b^{2}=16$，所以$a=5$，$b=4$。\n
			\par\textbf{理由：}\n 椭圆标准方程$\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$对应项比较。\n\n
		\item[\textbf{第二步（计算半焦距）：}]
			由$c^{2}=a^{2}-b^{2}$，得$c^{2}=25-16$，所以$c^{2}=9$，故$c=3$。\n
			\par\textbf{理由：}\n 椭圆参数关系$c^{2}=a^{2}-b^{2}$。\n\n
		\item[\textbf{第三步（求离心率）：}]
			由公式$e=\frac{c}{a}$，得$e=\frac{3}{5}$。\n
			\par\textbf{理由：}\n 离心率代数公式$e=\frac{c}{a}$。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 椭圆$C$的离心率为$\frac{3}{5}$。\n\n

	\n\textbf{\textcolor{CExample}{例2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知双曲线中心在原点，焦点在$x$轴上，实轴长为$8$，离心率$e=\frac{5}{4}$，求双曲线的标准方程。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求实半轴$a$）：}]
			实轴长$2a=8$，所以$a=4$。\n
			\par\textbf{理由：}\n 实轴长定义。\n\n
		\item[\textbf{第二步（求半焦距$c$）：}]
			由$e=\frac{c}{a}$得$c = a e$。代入$a=4$，$e=\frac{5}{4}$，得$c = 4 \times \frac{5}{4}$，故$c=5$。\n
			\par\textbf{理由：}\n 离心率代数公式变形。\n\n
		\item[\textbf{第三步（求$b$并写方程）：}]
			由$c^{2}=a^{2}+b^{2}$得$b^{2}=c^{2}-a^{2}$。代入得$b^{2}=25-16$，所以$b^{2}=9$，$b=3$。故标准方程为$\frac{x^{2}}{16}-\frac{y^{2}}{9}=1$。\n
			\par\textbf{理由：}\n 双曲线参数关系；焦点在$x$轴时方程形式。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 双曲线方程为$\dfrac{x^{2}}{16}-\dfrac{y^{2}}{9}=1$。\n\n

	\n\textbf{\textcolor{CExample}{例3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知椭圆中心在原点，焦点在$x$轴上，椭圆上一点$P$到右焦点$F$的距离为$4$，到右准线$l$的距离为$5$。若椭圆的长轴长为$20$，求椭圆的标准方程。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（利用定义求$e$）：}]
			由离心率定义 $e=\frac{|PF|}{d(P,l)}$，代入得 $e=\frac{4}{5}$。\n
			\par\textbf{理由：}\n 定义直接应用。\n\n
		\item[\textbf{第二步（求$a,c$）：}]
			长轴长$2a=20$，所以$a=10$。由$e=\frac{c}{a}$得$c = a e$。代入得 $c = 10 \times \frac{4}{5}$，故 $c=8$。\n
			\par\textbf{理由：}\n 离心率代数关系。\n\n
		\item[\textbf{第三步（求$b$并得方程）：}]
			由$c^{2}=a^{2}-b^{2}$得$b^{2}=a^{2}-c^{2}$。代入得 $b^{2}=100-64$，所以 $b^{2}=36$，$b=6$。椭圆标准方程为 $\frac{x^{2}}{100}+\frac{y^{2}}{36}=1$。\n
			\par\textbf{理由：}\n 椭圆参数关系代入。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 椭圆方程为$\dfrac{x^{2}}{100}+\dfrac{y^{2}}{36}=1$。\n
\end{examplebox}
