	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知椭圆$C:\frac{x^{2}}{25}+\frac{y^{2}}{16}=1$，求其离心率$e$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]

		\item[\textbf{第一步（识别参数）：}]
		      从方程中找到$a^{2}=25$，$b^{2}=16$，所以$a=5$，$b=4$。

		      \par\textbf{理由：}
		      椭圆标准方程$\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1$对应项比较。

		\item[\textbf{第二步（计算半焦距）：}]
		      由$c^{2}=a^{2}-b^{2}$，得$c^{2}=25-16=9$，所以$c=3$。

		      \par\textbf{理由：}
		      椭圆参数关系$c^{2}=a^{2}-b^{2}$。

		\item[\textbf{第三步（求离心率）：}]
		      由公式$e=\frac{c}{a}$，得$e=\frac{3}{5}$。

		      \par\textbf{理由：}
		      离心率代数公式$e=\frac{c}{a}$。

	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 椭圆$C$的离心率为$\frac{3}{5}$。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知双曲线中心在原点，焦点在$x$轴上，实轴长为$8$，离心率$e=\frac{5}{4}$，求双曲线的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]

		\item[\textbf{第一步（求实半轴$a$）：}]
		      实轴长$2a=8$，所以$a=4$。

		      \par\textbf{理由：}
		      实轴长定义。

		\item[\textbf{第二步（求半焦距$c$）：}]
		      由$e=\frac{c}{a}$得$c=ae$。代入$a=4$，$e=\frac{5}{4}$，得$c=5$。

		      \par\textbf{理由：}
		      离心率代数公式变形。

		\item[\textbf{第三步（求$b$并写方程）：}]
		      由$c^{2}=a^{2}+b^{2}$得$b^{2}=c^{2}-a^{2}$。代入得$b^{2}=25-16=9$。故标准方程为

		      \[
			      \frac{x^{2}}{16}-\frac{y^{2}}{9}=1.
		      \]

		      \par\textbf{理由：}
		      双曲线参数关系；焦点在$x$轴时方程形式为$\frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1$。

	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 双曲线方程为$\dfrac{x^{2}}{16}-\dfrac{y^{2}}{9}=1$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知椭圆中心在原点，焦点在$x$轴上，椭圆上一点$P$到右焦点$F$的距离为$4$，到对应右准线$l$的距离为$5$。若椭圆的长轴长为$20$，求椭圆的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]

		\item[\textbf{第一步（利用定义求$e$）：}]
		      由离心率定义 $e=\frac{|PF|}{d(P,l)}$，代入得

		      \[
			      e=\frac{4}{5}.
		      \]

		      \par\textbf{理由：}
		      焦点和准线对应，可以直接使用离心率定义。

		\item[\textbf{第二步（求$a,c$）：}]
		      长轴长$2a=20$，所以$a=10$。由$e=\frac{c}{a}$得$c=ae$。代入得

		      \[
			      c=10\times\frac{4}{5}=8.
		      \]

		      \par\textbf{理由：}
		      离心率代数关系。

		\item[\textbf{第三步（求$b$并得方程）：}]
		      由$c^{2}=a^{2}-b^{2}$得$b^{2}=a^{2}-c^{2}$。代入得

		      \[
			      b^{2}=100-64=36.
		      \]

		      所以椭圆标准方程为

		      \[
			      \frac{x^{2}}{100}+\frac{y^{2}}{36}=1.
		      \]

		      \par\textbf{理由：}
		      椭圆参数关系代入。

	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 椭圆方程为$\dfrac{x^{2}}{100}+\dfrac{y^{2}}{36}=1$。
