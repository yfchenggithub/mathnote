\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]

		\item[\textbf{\textcolor{CStatement}{条件1（曲线类型）：}}]
		      适用于高中常见圆锥曲线：椭圆、双曲线、抛物线。其中公式$e=\frac{c}{a}$只适用于椭圆和双曲线；抛物线的离心率由定义直接得到$e=1$。

		\item[\textbf{\textcolor{CStatement}{条件2（对应关系）：}}]
		      使用定义$e=\frac{|PF|}{d(P,l)}$时，焦点$F$必须和准线$l$对应。

	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]

		\item[\textbf{\textcolor{CStatement}{结论一（几何定义）：}}]

		      \textbf{\textcolor{CStatement}{直观描述：}}
		      圆锥曲线上任意一点到对应焦点的距离，与到对应准线的距离之比是一个常数，这个常数叫离心率。

		      \[
			      e = \frac{|PF|}{d(P,l)}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（椭圆、双曲线代数形式）：}}]

		      \textbf{\textcolor{CStatement}{直观描述：}}
		      对椭圆和双曲线，离心率等于半焦距$c$与半实轴长或半长轴$a$之比。

		      \[
			      e = \frac{c}{a}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论三（取值范围）：}}]

		      \textbf{\textcolor{CStatement}{直观描述：}}
		      椭圆、抛物线、双曲线可以用离心率大小区分。

		      \[
			      \text{椭圆：}0<e<1,\qquad
			      \text{抛物线：}e=1,\qquad
			      \text{双曲线：}e>1.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（与$a,b$关系）：}}]

		      \textbf{\textcolor{CStatement}{直观描述：}}
		      椭圆离心率可写为$\sqrt{1-\frac{b^{2}}{a^{2}}}$，双曲线离心率可写为$\sqrt{1+\frac{b^{2}}{a^{2}}}$。

		      \[
			      \text{椭圆：}e = \sqrt{1-\frac{b^{2}}{a^{2}}},
			      \qquad
			      \text{双曲线：}e = \sqrt{1+\frac{b^{2}}{a^{2}}}.
		      \]

	\end{description}

\end{statementbox}
