	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（曲线类型）：}}]
			曲线必须为椭圆、双曲线或抛物线等非退化二次曲线。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（几何定义与代数关系）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			离心率是点到焦点与准线距离的比值，也等于半焦距与半长轴之比。\n
			\[
				\n
				\begin{aligned}
					\n e &= \frac{|PF|}{d(P,l)} \\ &= \frac{c}{a}.\n
			\end{aligned}
				\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{结论二（取值范围）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆离心率在0到1之间，双曲线大于1，抛物线等于1。\n
			\[
				\n 0<e<1,\quad e>1,\quad e=1.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（与a,b关系）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			椭圆离心率可写为$\sqrt{1-\frac{b^{2}}{a^{2}}}$，双曲线为$\sqrt{1+\frac{b^{2}}{a^{2}}}$。\n
			\[
				\n e = \sqrt{1-\frac{b^{2}}{a^{2}}},
				\qquad
				e = \sqrt{1+\frac{b^{2}}{a^{2}}}.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
