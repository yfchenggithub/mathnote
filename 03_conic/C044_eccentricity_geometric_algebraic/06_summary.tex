\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{离心率是连接焦点、准线、标准方程参数的统一度量，几何定义$e=\frac{|PF|}{d(P,l)}$与代数形式$e=\frac{c}{a}$等价。}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（曲线类型）：} 适用于椭圆、双曲线、抛物线。\n
		\item \textbf{条件2（标准形式）：} 代数形式$e=\frac{c}{a}$仅适用于椭圆和双曲线的标准方程；抛物线直接使用定义。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（参数关系）：} 椭圆$c^{2}=a^{2}-b^{2}$，双曲线$c^{2}=a^{2}+b^{2}$，不可混淆。\n
		\item \textbf{检查项（定义优先）：} 涉及焦点、准线的距离问题，首先联想定义$e=\frac{|PF|}{d(P,l)}$。\n
	\end{itemize}
	\n
\end{summarybox}
