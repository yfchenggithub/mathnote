\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	这里以椭圆为例证明。由椭圆标准方程出发，计算点$P$到右焦点$F$的距离，以及点$P$到对应右准线$l$的距离，最后证明二者比值恒等于$\frac{c}{a}$。

	\medskip

	\textbf{\textcolor{CProof}{正式推导}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]

		\item[\textbf{步骤一（设定方程与元素）：}]
		      设椭圆
		      \[
			      E:\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1
			      \quad (a>b>0),
		      \]
		      右焦点为$F(c,0)$，对应右准线为
		      \[
			      l:x=\frac{a^{2}}{c},
		      \]
		      其中
		      \[
			      c^{2}=a^{2}-b^{2}.
		      \]

		      \par\textbf{理由：}
		      椭圆焦点在$x$轴上时，$c^{2}=a^{2}-b^{2}$，右焦点对应右准线。

		\item[\textbf{步骤二（距离表示）：}]
		      设$P(x,y)$为椭圆上任意一点，则

		      \[
			      |PF| = \sqrt{(x-c)^{2}+y^{2}},
			      \qquad
			      d(P,l) = \left|x-\frac{a^{2}}{c}\right|.
		      \]

		      \par\textbf{理由：}
		      前者是两点间距离公式；后者是点到竖直直线$x=\frac{a^{2}}{c}$的距离。

		\item[\textbf{步骤三（消去$y^{2}$）：}]
		      由椭圆方程得

		      \[
			      y^{2}=b^{2}-\frac{b^{2}}{a^{2}}x^{2}.
		      \]

		      代入$|PF|^{2}$中：

		      \[
			      \begin{aligned}
				      |PF|^{2}
				       & = (x-c)^{2}+y^{2}                                          \\
				       & = (x-c)^{2}+b^{2}-\frac{b^{2}}{a^{2}}x^{2}                 \\
				       & = \left(1-\frac{b^{2}}{a^{2}}\right)x^{2}-2cx+c^{2}+b^{2}.
			      \end{aligned}
		      \]

		      \par\textbf{理由：}
		      用椭圆方程把$y^{2}$转化成含$x$的表达式。

		\item[\textbf{步骤四（化简$|PF|$）：}]
		      因为$c^{2}=a^{2}-b^{2}$，所以

		      \[
			      1-\frac{b^{2}}{a^{2}}=\frac{c^{2}}{a^{2}},
			      \qquad
			      c^{2}+b^{2}=a^{2}.
		      \]

		      于是

		      \[
			      \begin{aligned}
				      |PF|^{2}
				       & = \frac{c^{2}}{a^{2}}x^{2}-2cx+a^{2} \\
				       & = \left(a-\frac{c}{a}x\right)^{2}.
			      \end{aligned}
		      \]

		      因为椭圆上有$-a\le x\le a$，且$0<c<a$，所以

		      \[
			      a-\frac{c}{a}x>0.
		      \]

		      因此

		      \[
			      |PF| = a-\frac{c}{a}x.
		      \]

		      \par\textbf{理由：}
		      配方后开方，要根据椭圆上$x$的取值范围判断正负。

		\item[\textbf{步骤五（计算比值）：}]
		      因为$0<c<a$，所以

		      \[
			      \frac{a^{2}}{c}>a.
		      \]

		      而椭圆上$x\le a$，所以

		      \[
			      d(P,l)=\frac{a^{2}}{c}-x.
		      \]

		      于是

		      \[
			      \begin{aligned}
				      \frac{|PF|}{d(P,l)}
				       & = \frac{a-\frac{c}{a}x}{\frac{a^{2}}{c}-x}.
			      \end{aligned}
		      \]

		      又因为

		      \[
			      a-\frac{c}{a}x
			      =
			      \frac{c}{a}\left(\frac{a^{2}}{c}-x\right),
		      \]

		      所以

		      \[
			      \begin{aligned}
				      \frac{|PF|}{d(P,l)}
				       & = \frac{\frac{c}{a}\left(\frac{a^{2}}{c}-x\right)}{\frac{a^{2}}{c}-x} \\
				       & = \frac{c}{a}.
			      \end{aligned}
		      \]

		      \par\textbf{理由：}
		      分子与分母有相同因式，约分后得到常数$\frac{c}{a}$。

	\end{description}

	\medskip

	\textbf{\textcolor{CProof}{结论回扣}}

	因此，对椭圆而言，

	\[
		e=\frac{|PF|}{d(P,l)}=\frac{c}{a}.
	\]

	双曲线也可类似证明，只是参数关系改为$c^{2}=a^{2}+b^{2}$。抛物线的定义本身就是曲线上任意一点到焦点的距离等于到准线的距离，所以抛物线的离心率为

	\[
		e=1.
	\]

\end{proofbox}
