\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	对于焦点在 $x$ 轴上的标准椭圆
	\[
		\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\quad (a>b>0),
	\]
	若焦点为
	\[
		F_1(-c,0),\qquad F_2(c,0),
	\]
	且
	\[
		c^2=a^2-b^2,\qquad e=\frac{c}{a},
	\]
	则点 $P(x,y)$ 到两焦点的距离为
	\[
		PF_1=a+ex,\qquad PF_2=a-ex.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准椭圆）：} 椭圆方程为
		      \[
			      \frac{x^2}{a^2}+\frac{y^2}{b^2}=1\quad (a>b>0),
		      \]
		      焦点在 $x$ 轴。
		\item \textbf{条件 2（点在椭圆上）：} 点 $P(x,y)$ 满足椭圆方程，因此
		      \[
			      -a\le x\le a.
		      \]
		\item \textbf{条件 3（焦点命名固定）：} 默认
		      \[
			      F_1(-c,0),\qquad F_2(c,0).
		      \]
		      若焦点命名改变，公式中的符号也要相应改变。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{常用推论}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{焦半径之和：}
		      \[
			      PF_1+PF_2=2a.
		      \]
		\item \textbf{焦半径有向差：}
		      \[
			      PF_1-PF_2=2ex.
		      \]
		\item \textbf{焦半径差的绝对值：}
		      \[
			      |PF_1-PF_2|=2e|x|.
		      \]
		\item \textbf{焦半径范围：}
		      \[
			      a-c\le PF_1\le a+c,\qquad a-c\le PF_2\le a+c.
		      \]
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{符号规律：} 左焦点距离用加号，右焦点距离用减号，即“左加右减”。
		\item \textbf{差的理解：}
		      \[
			      PF_1-PF_2=2ex
		      \]
		      是有向差；若题目问非负的“两焦半径之差”，要加绝对值。
		\item \textbf{坐标范围：} 由焦半径反解出 $x$ 后，必须验证
		      \[
			      |x|\le a.
		      \]
		\item \textbf{$y$ 轴焦点情形：} 若椭圆为
		      \[
			      \frac{x^2}{b^2}+\frac{y^2}{a^2}=1\quad (a>b>0),
		      \]
		      焦点为
		      \[
			      F_1(0,-c),\qquad F_2(0,c),
		      \]
		      则公式变为
		      \[
			      PF_1=a+ey,\qquad PF_2=a-ey.
		      \]
	\end{itemize}
\end{summarybox}
