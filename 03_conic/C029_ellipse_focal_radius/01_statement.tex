\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（椭圆标准式）：}}] 点 $P(x,y)$ 在椭圆
		      \[
			      \frac{x^2}{a^2}+\frac{y^2}{b^2}=1\quad (a>b>0)
		      \]
		      上，焦点为 $F_1(-c,0)$，$F_2(c,0)$，其中
		      \[
			      c^2=a^2-b^2,\qquad e=\frac{c}{a}.
		      \]
		      本文中 $PF_1,PF_2$ 均表示线段长度。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦半径公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 左焦半径等于 $a+ex$，右焦半径等于 $a-ex$。
		      \[
			      PF_1 = a+ex,\qquad PF_2 = a-ex.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（焦半径和差）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 两焦半径之和为常数 $2a$，两焦半径的有向差与横坐标 $x$ 成正比。
		      \[
			      PF_1+PF_2 = 2a,\qquad PF_1-PF_2 = 2ex.
		      \]
		      注意：这里的 $PF_1-PF_2$ 是有向差，可能为正、为零或为负。若题目问“两焦半径之差的绝对值”，则
		      \[
			      |PF_1-PF_2| = 2e|x|.
		      \]
	\end{description}
\end{statementbox}
