\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 椭圆 $C:\frac{x^2}{9}+\frac{y^2}{5}=1$ 的焦点为 $F_1,F_2$，点 $P(2,\frac{5}{3})$ 在椭圆 $C$ 上。利用焦半径公式求 $PF_1$ 和 $PF_2$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}] 由椭圆方程知
		      \[
			      a^2=9,\qquad b^2=5,
		      \]
		      则
		      \[
			      a=3,\qquad c=\sqrt{a^2-b^2}=2,\qquad e=\frac{c}{a}=\frac{2}{3}.
		      \]
		      \par\textbf{理由：} 焦半径公式需要 $a$ 和 $e$。

		\item[\textbf{第二步（代入公式）：}] 点 $P$ 的横坐标为 $x=2$，代入焦半径公式：
		      \[
			      PF_1=a+ex=3+\frac{2}{3}\times 2
			      =3+\frac{4}{3}
			      =\frac{13}{3},
		      \]
		      \[
			      PF_2=a-ex=3-\frac{2}{3}\times 2
			      =3-\frac{4}{3}
			      =\frac{5}{3}.
		      \]
		      \par\textbf{理由：} 直接套用左加右减公式。

		\item[\textbf{第三步（验证结果）：}] 利用椭圆定义
		      \[
			      PF_1+PF_2=2a=6.
		      \]
		      而
		      \[
			      \frac{13}{3}+\frac{5}{3}=6,
		      \]
		      与椭圆定义相符。
		      \par\textbf{理由：} 用 $PF_1+PF_2=2a$ 验证计算正确性。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		PF_1=\frac{13}{3},\qquad PF_2=\frac{5}{3}.
	\]

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 椭圆 $C:\frac{x^2}{9}+\frac{y^2}{5}=1$ 的焦点为 $F_1,F_2$，点 $P$ 在椭圆上且 $PF_2=2$。求 $PF_1$ 和点 $P$ 的横坐标 $x$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设横坐标 $x$）：}] 设 $P(x,y)$。由焦半径公式
		      \[
			      PF_2=a-ex,
		      \]
		      其中
		      \[
			      a=3,\qquad e=\frac{2}{3}.
		      \]
		      \par\textbf{理由：} 利用已知 $PF_2=2$ 建立方程。

		\item[\textbf{第二步（解方程求 $x$）：}]
		      \[
			      2=3-\frac{2}{3}x
			      \Rightarrow \frac{2}{3}x=1
			      \Rightarrow x=\frac{3}{2}.
		      \]
		      \par\textbf{理由：} 解得 $x=\frac{3}{2}$，并且
		      \[
			      -3\le \frac{3}{2}\le 3,
		      \]
		      所以横坐标有效。

		\item[\textbf{第三步（求 $PF_1$）：}] 由焦半径公式
		      \[
			      PF_1=a+ex
			      =3+\frac{2}{3}\times\frac{3}{2}
			      =3+1
			      =4.
		      \]
		      也可以由椭圆定义得到
		      \[
			      PF_1=2a-PF_2=6-2=4.
		      \]
		      \par\textbf{理由：} 两种方法相互验证。

		\item[\textbf{第四步（说明点 $P$ 不唯一）：}] 若题目进一步要求点 $P$ 的坐标，则由
		      \[
			      \frac{x^2}{9}+\frac{y^2}{5}=1,\qquad x=\frac{3}{2},
		      \]
		      得
		      \[
			      \frac{y^2}{5}=1-\frac{(3/2)^2}{9}
			      =1-\frac{1}{4}
			      =\frac{3}{4}.
		      \]
		      所以
		      \[
			      y^2=\frac{15}{4},\qquad y=\pm\frac{\sqrt{15}}{2}.
		      \]
		      因此点 $P$ 有两个：
		      \[
			      P\left(\frac{3}{2},\frac{\sqrt{15}}{2}\right)
			      \quad\text{或}\quad
			      P\left(\frac{3}{2},-\frac{\sqrt{15}}{2}\right).
		      \]
		      \par\textbf{理由：} 椭圆关于 $x$ 轴对称；本题只求横坐标 $x$ 和 $PF_1$，所以上下两个点对应结果相同。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		x=\frac{3}{2},\qquad PF_1=4.
	\]

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 椭圆 $C:\frac{x^2}{9}+\frac{y^2}{5}=1$ 的焦点为 $F_1,F_2$，点 $P$ 在椭圆上运动。求 $PF_1\cdot PF_2$ 的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（表达乘积）：}] 由焦半径公式，
		      \[
			      PF_1=a+ex,\qquad PF_2=a-ex.
		      \]
		      所以
		      \[
			      PF_1\cdot PF_2=(a+ex)(a-ex)=a^2-e^2x^2.
		      \]
		      代入
		      \[
			      a=3,\qquad e=\frac{2}{3},
		      \]
		      得
		      \[
			      PF_1\cdot PF_2=9-\frac{4}{9}x^2.
		      \]
		      \par\textbf{理由：} 利用焦半径公式将乘积转化为关于 $x$ 的函数。

		\item[\textbf{第二步（确定 $x$ 范围）：}] 点 $P$ 在椭圆上，所以横坐标满足
		      \[
			      |x|\le a=3,
		      \]
		      即
		      \[
			      x\in[-3,3].
		      \]
		      \par\textbf{理由：} 椭圆长轴端点的横坐标为 $\pm a$。

		\item[\textbf{第三步（求最值）：}] 作为 $x$ 的函数，
		      \[
			      PF_1\cdot PF_2=9-\frac{4}{9}x^2
		      \]
		      是开口向下的二次函数。也可以令
		      \[
			      t=x^2,\qquad 0\le t\le 9,
		      \]
		      则
		      \[
			      PF_1\cdot PF_2=9-\frac{4}{9}t.
		      \]
		      这是关于 $t$ 的一次减函数，所以当 $t$ 最小时乘积最大。
		      \[
			      t_{\min}=0\Rightarrow x=0.
		      \]
		      因此
		      \[
			      \max(PF_1\cdot PF_2)=9-\frac{4}{9}\times 0=9.
		      \]
		      \par\textbf{理由：} $x^2$ 的最小值为 $0$，此时 $x=0$，对应点 $P$ 在短轴端点。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		PF_1\cdot PF_2
	\]
	的最大值为
	\[
		9.
	\]
	此时 $x=0$，点 $P$ 在短轴端点。
\end{examplebox}
