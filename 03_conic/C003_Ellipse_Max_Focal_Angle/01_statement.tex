\begin{statementbox}
设椭圆
\[
C:\quad \frac{x^2}{a^2}+\frac{y^2}{b^2}=1,
\qquad a>b>0,\qquad c=\sqrt{a^2-b^2},\qquad e=\frac ca.
\]
左、右焦点分别为 $F_1(-c,0)$、$F_2(c,0)$。任取椭圆上一点
$P(x_0,y_0)$，记 $\theta=\angle F_1PF_2$，
$r_1=PF_1$、$r_2=PF_2$，则有以下结论：
\begin{enumerate}
  \item \textbf{焦角半角公式（任意点成立）：}
  \[
  \boxed{\tan\frac{\theta}{2}=\frac{c|y_0|}{b^2}}.
  \]
  这里 $|y_0|$ 是点 $P$ 到焦点所在直线（$x$ 轴）的距离。

  \item \textbf{焦角最大值：}当且仅当 $P$ 为短轴端点
  $(0,b)$ 或 $(0,-b)$ 时，焦角取得最大值，并且
  \[
  \boxed{\tan\frac{\theta_{\max}}2=\frac cb},\qquad
  \boxed{\theta_{\max}=2\arctan\frac cb=2\arcsin e}.
  \]
  因而 $0\leq\theta\leq\theta_{\max}<180^\circ$；
  当 $P=(\pm a,0)$ 时，$\theta=0$。

  \item \textbf{余弦定理形式（任意点成立）：}
  \[
  \boxed{\cos\theta=\frac{r_1^2+r_2^2-4c^2}{2r_1r_2}
  =\frac{b^4-c^2y_0^2}{b^4+c^2y_0^2}}.
  \]
  特别地，在短轴端点处
  \[
  \boxed{\cos\theta_{\max}=\frac{b^2-c^2}{b^2+c^2}
  =1-2e^2}.
  \]
\end{enumerate}
\textbf{适用性提醒：}上述公式要求 $P$ 在该椭圆上，且长轴位于
$x$ 轴；其中 $\tan(\theta/2)=c/|y_0|$ \textbf{不是}任意点公式，
只在 $|y_0|=b$（短轴端点）时与正确公式数值相同。
\end{statementbox}
