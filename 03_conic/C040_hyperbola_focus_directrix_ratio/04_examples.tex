\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\par\textbf{\textcolor{CExample}{题目：}} 已知双曲线 $\dfrac{x^{2}}{16}-\dfrac{y^{2}}{9}=1$ 上一点 $P$ 到右焦点 $F$ 的距离 $|PF|=10$，求点 $P$ 到右准线 $l$ 的距离 $|PH|$（$PH\perp l$）。
	\par\textbf{\textcolor{CExample}{解题步骤：}}\par
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求基本量）：}]
			由双曲线方程得 $a=4,\;b=3$，计算 $c=\sqrt{a^{2}+b^{2}}=5$，离心率 $e=\dfrac{c}{a}=\dfrac{5}{4}$。
			\par\textbf{理由：} 标准方程参数关系。
		\item[\textbf{第二步（应用定义）：}]
			根据双曲线第二定义，$\dfrac{|PF|}{|PH|}=e$，所以 $|PH|=\dfrac{|PF|}{e}$。
			\par\textbf{理由：} 统一定义直接套用。
		\item[\textbf{第三步（计算）：}]
			$|PH|=\dfrac{10}{5/4}=10\times\dfrac{4}{5}=8$。
			\par\textbf{理由：} 代数运算。
	\end{description}
	\par\textbf{\textcolor{CExample}{关键结论：}} \textbf{点 $P$ 到右准线的距离为 $8$}。\par

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\par\textbf{\textcolor{CExample}{题目：}} 已知双曲线 $\dfrac{x^{2}}{4}-\dfrac{y^{2}}{12}=1$ 上一点 $P$ 的横坐标为 $5$，求点 $P$ 到右焦点 $F$ 的距离 $|PF|$ 和到右准线 $l$ 的距离 $|PH|$。
	\par\textbf{\textcolor{CExample}{解题步骤：}}\par
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求基本量）：}]
			$a=2,\;c=\sqrt{4+12}=4,\;e=\dfrac{c}{a}=2$，右准线 $x=\dfrac{a^{2}}{c}=1$。
			\par\textbf{理由：} 标准方程与参数关系。
		\item[\textbf{第二步（求 $|PH|$）：}]
			由 $P$ 横坐标 $5$，准线 $x=1$，得 $|PH|=|5-1|=4$。
			\par\textbf{理由：} 点到直线的距离（横坐标差绝对值）。
		\item[\textbf{第三步（求 $|PF|$）：}]
			由第二定义 $\dfrac{|PF|}{|PH|}=e$，得 $|PF|=e\cdot|PH|=2\times4=8$。
			\par\textbf{理由：} 统一定义。
	\end{description}
	\par\textbf{\textcolor{CExample}{关键结论：}} \textbf{$|PH|=4$，$|PF|=8$}。\par

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\par\textbf{\textcolor{CExample}{题目：}} 已知双曲线 $\dfrac{x^{2}}{4}-\dfrac{y^{2}}{5}=1$ 的右焦点为 $F$，右准线为 $l$，点 $P$ 在双曲线上，$PH\perp l$ 于 $H$，且 $|PF|=6$。求点 $P$ 的横坐标 $x_{0}$。
	\par\textbf{\textcolor{CExample}{解题步骤：}}\par
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求参数与准线）：}]
			由方程得 $a=2,\;b=\sqrt{5}$，则 $c=\sqrt{a^{2}+b^{2}}=3$，离心率 $e=\dfrac{c}{a}=\dfrac{3}{2}$，右准线 $x=\dfrac{a^{2}}{c}=\dfrac{4}{3}$。
			\par\textbf{理由：} 双曲线标准方程的基本量。
		\item[\textbf{第二步（应用定义列方程）：}]
			由第二定义 $\dfrac{|PF|}{|PH|}=e$，且 $|PH|=|x_{0}-\dfrac{4}{3}|$，得
			\[
				6 = \frac{3}{2}\Bigl|x_{0}-\frac{4}{3}\Bigr|.
			\]
			\par\textbf{理由：} 统一定义及坐标表示。
		\item[\textbf{第三步（解绝对值方程）：}]
			$\Bigl|x_{0}-\dfrac{4}{3}\Bigr| = 4$，故 $x_{0}=\dfrac{4}{3}\pm4$，即 $x_{0}=\dfrac{16}{3}$ 或 $x_{0}=-\dfrac{8}{3}$。
			\par\textbf{理由：} 绝对值方程求解，考虑正负两解。
	\end{description}
	\par\textbf{\textcolor{CExample}{关键结论：}} \textbf{点 $P$ 的横坐标为 $\dfrac{16}{3}$ 或 $-\dfrac{8}{3}$（若点 $P$ 在右支，应取 $\dfrac{16}{3}$）}。
\end{examplebox}
