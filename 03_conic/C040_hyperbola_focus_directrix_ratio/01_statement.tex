	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（统一定义）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			双曲线上的点到焦点距离与到准线距离之比恒为离心率.
			\[
				\frac{|PF|}{|PH|}=e \quad (e>1).
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（标准形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			标准方程下焦点坐标与准线方程由半实轴 $a$ 与半焦距 $c$ 决定.
			\[
				e=\frac{c}{a},\quad c^{2}=a^{2}+b^{2}.
			\]
			\[
				F(\pm c,0),\quad x=\pm\frac{a^{2}}{c}.
			\]
	\end{description}
