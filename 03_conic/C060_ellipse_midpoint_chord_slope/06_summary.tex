\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{中点弦斜率公式：$k = -\dfrac{b^2 x_0}{a^2 y_0}$，给出椭圆标准方程下弦的斜率与中点坐标的直接关系。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准方程）：} 椭圆为中心在原点的标准方程 $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$（$a>b>0$）。
		\item \textbf{条件 2（中点内部）：} 弦中点 $M(x_0,y_0)$ 位于椭圆内部（$\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2}<1$）。
		\item \textbf{条件 3（分母非零）：} $y_0\neq 0$，且弦不与 $y$ 轴平行（斜率存在）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母为零）：} 当 $y_0=0$ 时公式无意义，此时若 $x_0\neq0$ 弦垂直于 $x$ 轴（斜率不存在）；若 $x_0=0$（中点为中心），弦为任意过原点的直线。
		\item \textbf{检查项（中点在内）：} 使用公式前务必验证中点坐标满足 $\dfrac{x_0^2}{a^2}+\dfrac{y_0^2}{b^2}<1$，这是点差法适用的前提。
		\item \textbf{检查项（系数对应）：} 牢记公式是 $-\dfrac{b^2 x_0}{a^2 y_0}$，不要混淆 $a$ 和 $b$。
	\end{itemize}
\end{summarybox}
