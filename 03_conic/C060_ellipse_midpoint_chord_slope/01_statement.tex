\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（标准方程）：}}] 椭圆方程为 $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$，其中 $a>b>0$。
		\item[\textbf{\textcolor{CStatement}{条件 2（中点约束）：}}] 弦的中点 $M(x_0,y_0)$ 位于椭圆内部，且 $y_0\neq 0$。
		\item[\textbf{\textcolor{CStatement}{条件 3（斜率存在）：}}] 弦不与 $y$ 轴平行，即弦的斜率存在。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中点弦斜率）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			弦所在直线的斜率等于 $-\dfrac{b^2}{a^2}$ 乘以中点横纵坐标之比 $\dfrac{x_0}{y_0}$。
			\[
				\boxed{
				\begin{aligned}
					k &= -\frac{b^2}{a^2}\cdot\frac{x_0}{y_0} \\
					&= -\frac{b^2 x_0}{a^2 y_0}
			\end{aligned}
				}\qquad (y_0\neq 0).
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（焦点在y轴）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若椭圆长轴在 $y$ 轴上，方程形如 $\dfrac{x^2}{b^2}+\dfrac{y^2}{a^2}=1$，则比例系数变为 $\dfrac{a^2}{b^2}$。
			\[
				\boxed{
				\begin{aligned}
					k &= -\frac{a^2}{b^2}\cdot\frac{x_0}{y_0} \\
					&= -\frac{a^2 x_0}{b^2 y_0}
			\end{aligned}
				}\qquad (y_0\neq 0).
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{特殊情形（公式边界）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{当 $y_0=0$ 时：}}] 中点纵坐标为零，公式分母为零，不适用。若 $x_0\neq0$，弦垂直于 $x$ 轴（斜率不存在）；若 $x_0=0$，弦为过原点的任意直线（此时中点即椭圆中心）。
		\item[\textbf{\textcolor{CStatement}{当 $x_0=0$ 且 $y_0\neq0$ 时：}}] 中点横坐标为零，代入得 $k=0$，弦平行于 $x$ 轴（水平弦）。
	\end{description}
\end{statementbox}
