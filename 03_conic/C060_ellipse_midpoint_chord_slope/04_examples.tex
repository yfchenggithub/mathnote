\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	椭圆 $\frac{x^2}{16}+\frac{y^2}{12}=1$ 的一条弦的中点为 $M(2,1)$，求该弦所在直线的斜率及方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 代入中点弦斜率公式 $k = -\dfrac{b^2 x_0}{a^2 y_0}$，其中 $a^2=16$, $b^2=12$, $x_0=2$, $y_0=1$，得
			\[
				\begin{aligned}
					k &= -\frac{12\times 2}{16\times 1} \\
					&= -\frac{24}{16} \\
					&= -\frac{3}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 已知椭圆方程和中点坐标，可直接套用公式求斜率。
		\item[\textbf{第二步（点斜式方程）：}] 利用点斜式 $y-y_0 = k(x-x_0)$ 写出弦所在直线的方程：
			\[
				y-1 = -\frac{3}{2}(x-2).
			\]
			\par\textbf{理由：} 直线上一点和斜率可确定直线方程。
		\item[\textbf{第三步（化简方程）：}] 两边乘以 $2$ 并整理：
			\[
				\begin{aligned}
					2(y-1) &= -3(x-2) \\
					\Longrightarrow 2y-2 &= -3x+6 \\
					\Longrightarrow 3x+2y-8 &= 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 化为一般式便于书写和检查。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{斜率 $-\dfrac{3}{2}$，弦所在直线方程 $3x+2y-8=0$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^2}{9}+\frac{y^2}{4}=1$，一条弦所在直线方程为 $2x-3y+1=0$，求该弦的中点坐标。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取条件）：}] 从直线方程 $2x-3y+1=0$ 得斜率 $k=\dfrac{2}{3}$。设中点 $M(x_0,y_0)$，则中点满足直线方程：$2x_0-3y_0+1=0$。
			\par\textbf{理由：} 中点位于弦上，故坐标满足直线方程。
		\item[\textbf{第二步（建立斜率方程）：}] 由中点弦斜率公式 $k = -\dfrac{b^2 x_0}{a^2 y_0}$，代入 $a^2=9$, $b^2=4$, $k=\dfrac{2}{3}$：
			\[
				\frac{2}{3} = -\frac{4x_0}{9y_0}.
			\]
			\par\textbf{理由：} 中点弦公式给出了斜率与中点坐标之间的约束关系。
		\item[\textbf{第三步（联立求解）：}] 由 $\dfrac{2}{3} = -\dfrac{4x_0}{9y_0}$ 两边交叉相乘得 $2\cdot 9y_0 = -4x_0\cdot 3$，即 $18y_0 = -12x_0$，化简得 $2x_0+3y_0=0$。联立
			\[
				\begin{cases}
					2x_0-3y_0+1=0,\\
					2x_0+3y_0=0,
			\end{cases}
			\]
			解得 $x_0 = -\dfrac{1}{4}$，$y_0 = \dfrac{1}{6}$。
			\par\textbf{理由：} 两个方程解两个未知数，得中点坐标。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{弦的中点坐标为 $\left(-\dfrac{1}{4},\dfrac{1}{6}\right)$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $\frac{x^2}{12}+\frac{y^2}{8}=1$ 的一条弦的斜率为 $2$，弦中点的横坐标为 $1$，求该弦中点的纵坐标和弦所在直线的方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设中点坐标）：}] 设弦的中点坐标为 $M(1,y_0)$，其中 $y_0\neq 0$。
			\par\textbf{理由：} 横坐标已知，纵坐标为待求未知数。
		\item[\textbf{第二步（代入中点弦公式）：}] 由中点弦斜率公式 $k = -\dfrac{b^2 x_0}{a^2 y_0}$，代入 $a^2=12$, $b^2=8$, $x_0=1$, $k=2$：
			\[
				\begin{aligned}
					2 &= -\frac{8\cdot 1}{12\cdot y_0} \\
					&= -\frac{8}{12y_0} \\
					&= -\frac{2}{3y_0}.
			\end{aligned}
			\]
			\par\textbf{理由：} 公式直接关联斜率和中点坐标。
		\item[\textbf{第三步（求解 $y_0$）：}] 方程 $2 = -\dfrac{2}{3y_0}$ 两边乘 $3y_0$ 得 $6y_0 = -2$，所以 $y_0 = -\dfrac{1}{3}$。
			\par\textbf{理由：} 简单代数方程求解。
		\item[\textbf{第四步（写直线方程）：}] 由点斜式 $y - y_0 = k(x - x_0)$ 得
			\[
				y + \frac{1}{3} = 2(x - 1).
			\]
			化简：
			\[
				\begin{aligned}
					y &= 2x - 2 - \frac{1}{3} \\
					&= 2x - \frac{7}{3},
			\end{aligned}
			\]
			即 $6x - 3y - 7 = 0$。
			\par\textbf{理由：} 已知斜率和一点可确定直线方程，化为一般式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{中点纵坐标为 $-\dfrac{1}{3}$，弦所在直线方程为 $6x-3y-7=0$。}
