\begin{summarybox}
\textbf{一句话：}截点在两条不同渐近线上，原点三角形的面积等于
\textbf{斜率绝对值乘两个横坐标乘积的绝对值}。
\[
\boxed{S_{OAB}=k|x_1x_2|=\frac ba|x_1x_2|},\qquad k=\frac ba>0.
\]
\textbf{先检查三件事：}
\begin{itemize}[leftmargin=*,itemsep=.25em]
\item $A\in\{y=kx\}$、$B\in\{y=-kx\}$；截点唯一、不同，且 $x_1x_2\ne0$。
\item $\ell$ 可以水平或竖直，$x_1,x_2$ 可以同号或异号。
\item 面积取绝对值；主公式\textbf{不再乘 $1/2$}，也不普遍等于 $ab$。
\end{itemize}
\textbf{由截线方程速求面积：}若 $\ell:y=mx+n$，$n\ne0,m\ne\pm k$，则
\[
 x_1=\frac{n}{k-m},\quad x_2=-\frac{n}{k+m},\quad
 \boxed{S=\frac{kn^2}{|m^2-k^2|}}.
\]
\textbf{两条过原点直线的推广：}
\[
\boxed{S=\frac{|m_1-m_2|}{2}|x_1x_2|}.
\]
\textbf{切线条件给出面积定值：}
\[
\boxed{\ell\text{ 为双曲线切线}\ \Longrightarrow\ x_1x_2=a^2\ \Longrightarrow\ S=ab}.
\]
\textbf{六种证明怎么选：}
\begin{itemize}[leftmargin=*,itemsep=.18em]
\item 行列式最快：平行四边形面积的一半，两份 $kx_1x_2$ 抵消 $1/2$。
\item 初中法：竖直截线先得 $kx_1^2$，共底面积比再乘 $|x_2/x_1|$。
\item 横轴分块法：同号时先求 $c=2x_1x_2/(x_1+x_2)$，再用 $OC$ 为底将上下两块相加；水平截线单独算。
\item 伸缩法：$Y=y/k$ 使两斜线垂直，原面积为新面积的 $k$ 倍。
\item 常规法：取 $OA$ 作底，用 $B$ 到 $OA$ 的垂直距离作高。
\item 推广法：用两条直线的斜率差代替 $2k$。
\end{itemize}
\textbf{迁移时分清模型：}$xy=q$ 的 $|q|/2$ 属于单点与坐标轴所围三角形；
一般变换 $X=x/a+y/b,Y=x/a-y/b$ 下，面积须乘逆变换因子 $ab/2$。

\textbf{自测：}$a=3,b=2$，截线 $x=6$ 给 $S=24\ne ab=6$。
\end{summarybox}
