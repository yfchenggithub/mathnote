\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（基准椭圆）：}}]
			给定椭圆标准方程 $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$（$a>b>0$），焦点在 $x$ 轴上，半焦距 $c=\sqrt{a^2-b^2}$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（统一方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有与该椭圆共焦点的椭圆或双曲线可表示为统一方程，参数 $\lambda$ 的取值决定曲线类型。
			\[
				\boxed{\frac{x^2}{a^2+\lambda}+\frac{y^2}{b^2+\lambda}=1},
			\]
			其中 $\lambda$ 为实参数，满足 $a^2+\lambda>0$，$b^2+\lambda\neq 0$。
			当 $\lambda>-b^2$ 时表示椭圆系，当 $-a^2<\lambda<-b^2$ 时表示双曲线系。

		\item[\textbf{\textcolor{CStatement}{推广结论（焦点不变）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有共焦点曲线的焦距恒为 $2c$，焦点坐标保持 $(\pm c,0)$ 不变。
			\[
				(a^2+\lambda)-(b^2+\lambda)=a^2-b^2=c^2.
			\]
	\end{description}

\end{statementbox}
