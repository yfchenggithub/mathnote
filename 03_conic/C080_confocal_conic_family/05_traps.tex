\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（范围混淆）}}] 求共焦点的双曲线时，误用椭圆范围 $\lambda>-b^2$，导致参数选择错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（区间记忆）}}] 双曲线参数满足 $-a^2<\lambda<-b^2$，此时 $x^2$ 分母为正，$y^2$ 分母为负。
		\item[\textbf{\textcolor{CTrap}{错因分析（分母符号）}}] 未注意曲线类型由分母符号决定，死记统一形式而忽略符号要求。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分母为零）}}] 已知曲线过定点，解得 $\lambda$ 却未检查分母非零，导致出现荒谬方程。
		\item[\textbf{\textcolor{CTrap}{正确理解（检验分母）}}] 无论曲线类型，必须确保 $a^2+\lambda>0$ 且 $b^2+\lambda\neq0$，否则方程无意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义域）}}] 将曲线系方程视为万能形式，忽略了分式方程的分母非零约束。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（焦点位置）}}] 将指定焦点在 $x$ 轴的结论直接套用到焦点在 $y$ 轴的椭圆，导致方程错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（交换分母）}}] 若基准椭圆焦点在 $y$ 轴，应将标准方程写成 $\dfrac{x^2}{b^2+\lambda}+\dfrac{y^2}{a^2+\lambda}=1$，保持 $c^2=a^2-b^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（形式类比）}}] 忽略焦点轴的位置，默认 $x$ 轴分母对应 $a^2$，缺乏灵活调整。
	\end{description}
\end{trapbox}
