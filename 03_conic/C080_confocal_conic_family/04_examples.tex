\begin{examplebox}

	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $C:\dfrac{x^2}{9}+\dfrac{y^2}{5}=1$，求与该椭圆共焦点且经过点 $P(0,4)$ 的椭圆方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算焦距）：}] 由 $a^2=9$，$b^2=5$，得 $c^2=a^2-b^2=4$。
			\par\textbf{理由：} 椭圆焦点在 $x$ 轴，其半焦距平方 $c^2=a^2-b^2$。
		\item[\textbf{第二步（设曲线系方程）：}] 依共焦点曲线系，设所求椭圆方程为
			\[
				\frac{x^2}{9+\lambda}+\frac{y^2}{5+\lambda}=1,
			\]
			其中 $\lambda>-5$（确保分母都正且为椭圆）。
			\par\textbf{理由：} 共焦点的椭圆系可统一表示为此形式，$\lambda>-5$ 保证 $5+\lambda>0$，从而为椭圆。
		\item[\textbf{第三步（代入点求 $\lambda$）：}] 将 $P(0,4)$ 代入方程：
			\[
				0+\frac{16}{5+\lambda}=1\;\Longrightarrow\;5+\lambda=16\;\Longrightarrow\;\lambda=11.
			\]
			$\lambda=11$ 满足 $\lambda>-5$，故符合椭圆条件。
			\par\textbf{理由：} 点在曲线上，其坐标满足方程。
		\item[\textbf{第四步（写出方程）：}] 将 $\lambda=11$ 代入方程得
			\[
				\frac{x^2}{20}+\frac{y^2}{16}=1.
			\]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{所求椭圆方程为 $\dfrac{x^2}{20}+\dfrac{y^2}{16}=1$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $C:\dfrac{x^2}{9}+\dfrac{y^2}{5}=1$，求与该椭圆共焦点且经过点 $Q(3,\sqrt{2})$ 的双曲线方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算焦距）：}] 同前，$c^2=4$。
			\par\textbf{理由：} 同上。
		\item[\textbf{第二步（设曲线系方程）：}] 设所求双曲线方程为
			\[
				\frac{x^2}{9+\lambda}+\frac{y^2}{5+\lambda}=1,
			\]
			此时 $\lambda$ 应满足 $-9<\lambda<-5$。
			\par\textbf{理由：} 双曲线要求 $y^2$ 的分母为负，即 $5+\lambda<0$，且 $x^2$ 分母为正，则 $9+\lambda>0$，故 $-9<\lambda<-5$。
		\item[\textbf{第三步（代入点求 $\lambda$）：}] 将 $Q(3,\sqrt{2})$ 代入：
			\[
				\frac{9}{9+\lambda}+\frac{2}{5+\lambda}=1.
			\]
			通分并整理：
			\[
				\begin{aligned}
					\frac{9(5+\lambda)+2(9+\lambda)}{(9+\lambda)(5+\lambda)} &= 1 \\
					45+9\lambda+18+2\lambda &= (9+\lambda)(5+\lambda).
			\end{aligned}
			\]
			即 $63+11\lambda=45+14\lambda+\lambda^2$，移项得 $\lambda^2+3\lambda-18=0$，解得 $\lambda=-6$ 或 $\lambda=3$。
			\par\textbf{理由：} 整理方程为一元二次方程求解。
		\item[\textbf{第四步（检验并确定 $\lambda$）：}] $\lambda=3$ 不满足 $-9<\lambda<-5$，舍去；$\lambda=-6$ 在区间 $(-9,-5)$ 内，故取 $\lambda=-6$。
			\par\textbf{理由：} 必须同时满足双曲线条件。
		\item[\textbf{第五步（写出方程）：}] 代入 $\lambda=-6$：
			\[
				\frac{x^2}{3}+\frac{y^2}{-1}=1\quad\Longrightarrow\quad\frac{x^2}{3}-y^2=1.
			\]
			故所求双曲线方程为 $\dfrac{x^2}{3}-y^2=1$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{双曲线方程为 $\dfrac{x^2}{3}-y^2=1$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知椭圆 $C:\dfrac{x^2}{16}+\dfrac{y^2}{7}=1$，共焦点曲线系 $C_{\lambda}$ 的方程为 $\dfrac{x^2}{16+\lambda}+\dfrac{y^2}{7+\lambda}=1$（$\lambda$ 为实参数，$16+\lambda>0$，$7+\lambda\neq0$）。若 $C_{\lambda}$ 的离心率为 $\dfrac{3}{2}$，求 $\lambda$ 的值并判断曲线类型。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算焦距）：}] 基准椭圆中 $a^2=16$，$b^2=7$，故 $c^2=16-7=9$，$c=3$。
			\par\textbf{理由：} 椭圆 $a^2-b^2=c^2$。
		\item[\textbf{第二步（离心率公式）：}] 对曲线系 $C_{\lambda}$，离心率表示为 $e=\dfrac{c}{\sqrt{16+\lambda}}$（焦点在 $x$ 轴，$c=3$）。
			\par\textbf{理由：} 在标准位置，椭圆 $e=c/a'$，双曲线 $e=c/a'$，其中 $a'=\sqrt{16+\lambda}$。
		\item[\textbf{第三步（建立方程求 $\lambda$）：}] 由 $e=3/2$ 得
			\[
				\begin{aligned}
					\frac{3}{\sqrt{16+\lambda}} &= \frac{3}{2} \\
					\sqrt{16+\lambda} &= 2 \\
					16+\lambda &= 4 \\
					\lambda &= -12.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入离心率公式解方程。
		\item[\textbf{第四步（判断曲线类型）：}] 检查 $\lambda=-12$ 的范围：椭圆要求 $\lambda>-7$，双曲线要求 $-16<\lambda<-7$。由于 $-16<-12<-7$，故 $\lambda=-12$ 在双曲线范围内，对应曲线为双曲线。
			\par\textbf{理由：} 参数范围直接区分曲线类型。
		\item[\textbf{第五步（写出标准方程）：}] 将 $\lambda=-12$ 代入曲线系方程：
			\[
				\frac{x^2}{4}+\frac{y^2}{-5}=1\;\Longrightarrow\;\frac{x^2}{4}-\frac{y^2}{5}=1.
			\]
			故所求曲线为双曲线，方程为 $\dfrac{x^2}{4}-\dfrac{y^2}{5}=1$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\lambda=-12$，该曲线是双曲线，方程为 $\dfrac{x^2}{4}-\dfrac{y^2}{5}=1$。}
