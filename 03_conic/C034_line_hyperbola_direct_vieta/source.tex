
\[
\text{圆锥曲线硬解定理}
\]

\[
\text{2. 直线与双曲线}
\]

\[
\text{（1）双曲线焦点在 }x\text{ 轴}
\]

联立
\[
\begin{cases}
Ax+By+C=0,\quad (A\ne 0),\\[4pt]
\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1,\quad (a>0,\ b>0),
\end{cases}
\]
得：
\[
(B^2b^2-A^2a^2)x^2-2ACa^2x-C^2a^2-B^2a^2b^2=0
\]
或
\[
(B^2b^2-A^2a^2)y^2+2BCb^2y+(C^2b^2-A^2a^2b^2)=0.
\]

\[
x_1+x_2=\frac{2ACa^2}{B^2b^2-A^2a^2},
\qquad
x_1x_2=\frac{-C^2a^2-B^2a^2b^2}{B^2b^2-A^2a^2}
\]
或
\[
y_1+y_2=-\frac{2BCb^2}{B^2b^2-A^2a^2},
\qquad
y_1y_2=\frac{C^2b^2-A^2a^2b^2}{B^2b^2-A^2a^2}.
\]

\[
\text{（2）双曲线焦点在 }y\text{ 轴}
\]

联立
\[
\begin{cases}
Ax+By+C=0,\quad (A\ne 0),\\[4pt]
\dfrac{y^2}{a^2}-\dfrac{x^2}{b^2}=1,\quad (a>0,\ b>0),
\end{cases}
\]
得：
\[
(A^2b^2-B^2a^2)x^2+2ACb^2x+(C^2b^2-B^2a^2b^2)=0
\]
或
\[
(A^2b^2-B^2a^2)y^2-2BCa^2y-C^2a^2-A^2a^2b^2=0.
\]

\[
x_1+x_2=-\frac{2ACb^2}{A^2b^2-B^2a^2},
\qquad
x_1x_2=\frac{C^2b^2-B^2a^2b^2}{A^2b^2-B^2a^2}
\]
或
\[
y_1+y_2=\frac{2BCa^2}{A^2b^2-B^2a^2},
\qquad
y_1y_2=\frac{-C^2a^2-A^2a^2b^2}{A^2b^2-B^2a^2}.
\]
