\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（直线方程）：}}] 直线 $l: Ax+By+C=0$，其中 $A^{2}+B^{2}>0$。
		
		若要消去 $x$，通常需要 $A\neq0$，可写成
		\[
			x=-\frac{By+C}{A}.
		\]
		若要消去 $y$，通常需要 $B\neq0$，可写成
		\[
			y=-\frac{Ax+C}{B}.
		\]
		本结论中关于 $y$ 的方程默认 $A\neq0$；关于 $x$ 的方程默认 $B\neq0$。若遇到竖直线或水平线，应优先选择不退化的坐标关系。

		\item[\textbf{\textcolor{CStatement}{条件2（双曲线标准）：}}] 双曲线为 $\dfrac{x^{2}}{a^{2}}-\dfrac{y^{2}}{b^{2}}=1$（焦点在 $x$ 轴）或 $\dfrac{y^{2}}{a^{2}}-\dfrac{x^{2}}{b^{2}}=1$（焦点在 $y$ 轴），且 $a>0,\,b>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（焦点在 $x$ 轴）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 对于双曲线
		      \[
			      \frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1,
		      \]
		      直线与双曲线联立后，消元方程的二次项系数为
		      \[
			      M=B^{2}b^{2}-A^{2}a^{2}.
		      \]
		      当 $M\neq0$ 时，可使用二次韦达公式。

		      消去 $y$ 得：
		      \[
			      (B^{2}b^{2}-A^{2}a^{2})x^{2}-2ACa^{2}\,x-C^{2}a^{2}-B^{2}a^{2}b^{2}=0.
		      \]
		      消去 $x$ 得：
		      \[
			      (B^{2}b^{2}-A^{2}a^{2})y^{2}+2BCb^{2}\,y+C^{2}b^{2}-A^{2}a^{2}b^{2}=0.
		      \]
		      韦达定理（$x$ 关系）：
		      \[
			      x_{1}+x_{2}=\frac{2ACa^{2}}{B^{2}b^{2}-A^{2}a^{2}},\qquad
			      x_{1}x_{2}=\frac{-C^{2}a^{2}-B^{2}a^{2}b^{2}}{B^{2}b^{2}-A^{2}a^{2}}.
		      \]
		      韦达定理（$y$ 关系）：
		      \[
			      y_{1}+y_{2}=-\frac{2BCb^{2}}{B^{2}b^{2}-A^{2}a^{2}},\qquad
			      y_{1}y_{2}=\frac{C^{2}b^{2}-A^{2}a^{2}b^{2}}{B^{2}b^{2}-A^{2}a^{2}}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（焦点在 $y$ 轴）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 对于双曲线
		      \[
			      \frac{y^{2}}{a^{2}}-\frac{x^{2}}{b^{2}}=1,
		      \]
		      直线与双曲线联立后，消元方程的二次项系数为
		      \[
			      N=A^{2}b^{2}-B^{2}a^{2}.
		      \]
		      当 $N\neq0$ 时，可使用二次韦达公式。

		      消去 $y$ 得：
		      \[
			      (A^{2}b^{2}-B^{2}a^{2})x^{2}+2ACb^{2}\,x+C^{2}b^{2}-B^{2}a^{2}b^{2}=0.
		      \]
		      消去 $x$ 得：
		      \[
			      (A^{2}b^{2}-B^{2}a^{2})y^{2}-2BCa^{2}\,y-C^{2}a^{2}-A^{2}a^{2}b^{2}=0.
		      \]
		      韦达定理（$x$ 关系）：
		      \[
			      x_{1}+x_{2}=-\frac{2ACb^{2}}{A^{2}b^{2}-B^{2}a^{2}},\qquad
			      x_{1}x_{2}=\frac{C^{2}b^{2}-B^{2}a^{2}b^{2}}{A^{2}b^{2}-B^{2}a^{2}}.
		      \]
		      韦达定理（$y$ 关系）：
		      \[
			      y_{1}+y_{2}=\frac{2BCa^{2}}{A^{2}b^{2}-B^{2}a^{2}},\qquad
			      y_{1}y_{2}=\frac{-C^{2}a^{2}-A^{2}a^{2}b^{2}}{A^{2}b^{2}-B^{2}a^{2}}.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{推广结论（零系数情形）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{推广结论（渐近线平行或退化）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 当二次项系数 $B^{2}b^{2}-A^{2}a^{2}=0$（焦点在 $x$ 轴）或 $A^{2}b^{2}-B^{2}a^{2}=0$（焦点在 $y$ 轴）时，直线方向与双曲线的一条渐近线方向平行。

		      此时不能使用二次韦达公式，应单独讨论：
		      \[
			      C\neq0
			      \quad\Longrightarrow\quad
			      \text{消元方程退化为一次方程，通常只有一个有限交点；}
		      \]
		      \[
			      C=0
			      \quad\Longrightarrow\quad
			      \text{直线本身就是双曲线的一条渐近线，与双曲线没有有限交点。}
		      \]
	\end{description}
\end{statementbox}
