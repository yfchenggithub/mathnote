\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知直线 $l: y=x+2$ 与双曲线 $C: \dfrac{x^{2}}{4}-y^{2}=1$ 相交于 $A,B$ 两点。
	\begin{enumerate}[label=(\alph*)]
		\item 利用硬解定理直接写出二次方程系数及韦达定理。
		\item 求弦长 $|AB|$。
	\end{enumerate}

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别与赋值）：}] 双曲线 $\dfrac{x^{2}}{4}-y^{2}=1$，焦点在 $x$ 轴，$a^{2}=4,\;b^{2}=1$。直线化为 $x-y+2=0$，得
		      \[
			      A=1,\qquad B=-1,\qquad C=2.
		      \]
		      此时 $B\neq0$，可以使用关于 $x$ 的韦达关系和斜率弦长公式。
		      \par\textbf{理由：} 准备套用硬解公式所需的参数，并检查消元方向是否适合。

		\item[\textbf{第二步（套用硬解公式）：}] 焦点在 $x$ 轴，二次项系数为
		      \[
			      B^{2}b^{2}-A^{2}a^{2}=(-1)^{2}\cdot1-1^{2}\cdot4=-3\neq0.
		      \]
		      所以可以使用二次韦达公式：
		      \[
			      x_{1}+x_{2}=\frac{2ACa^{2}}{B^{2}b^{2}-A^{2}a^{2}}
			      =\frac{2\cdot1\cdot2\cdot4}{(-1)^{2}\cdot1-1^{2}\cdot4}
			      =\frac{16}{-3}
			      =-\frac{16}{3},
		      \]
		      \[
			      x_{1}x_{2}=\frac{-C^{2}a^{2}-B^{2}a^{2}b^{2}}{B^{2}b^{2}-A^{2}a^{2}}
			      =\frac{-2^{2}\cdot4-(-1)^{2}\cdot4\cdot1}{-3}
			      =\frac{-16-4}{-3}
			      =\frac{20}{3}.
		      \]
		      \par\textbf{理由：} 先确认分母非零，再代入硬解定理中的韦达公式。

		\item[\textbf{第三步（求弦长）：}] 直线 $y=x+2$ 的斜率为 $k=1$，因此
		      \[
			      |AB|=\sqrt{1+k^{2}}\;\sqrt{(x_{1}+x_{2})^{2}-4x_{1}x_{2}}
			      =\sqrt{2}\;\sqrt{\left(-\frac{16}{3}\right)^{2}-4\cdot\frac{20}{3}}
		      \]
		      \[
			      =\sqrt{2}\;\sqrt{\frac{256}{9}-\frac{80}{3}}
			      =\sqrt{2}\;\sqrt{\frac{256-240}{9}}
			      =\frac{4\sqrt{2}}{3}.
		      \]
		      \par\textbf{理由：} 因为 $B\neq0$，直线不是竖直线，可以使用 $x$ 坐标形式的弦长公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $x_{1}+x_{2}=-\dfrac{16}{3},\; x_{1}x_{2}=\dfrac{20}{3},\; |AB|=\dfrac{4\sqrt{2}}{3}$.

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知直线 $y=2x+1$ 与双曲线 $\dfrac{y^{2}}{5}-\dfrac{x^{2}}{4}=1$ 相交于 $A,B$ 两点。试利用硬解定理求 $x_{1}+x_{2}$ 与 $y_{1}y_{2}$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别与赋值）：}] 双曲线 $\dfrac{y^{2}}{5}-\dfrac{x^{2}}{4}=1$，焦点在 $y$ 轴，$a^{2}=5,\;b^{2}=4$。直线化为 $2x-y+1=0$，得
		      \[
			      A=2,\qquad B=-1,\qquad C=1.
		      \]
		      \par\textbf{理由：} 确定焦点位置以选用对应公式。

		\item[\textbf{第二步（检查二次项系数）：}] 焦点在 $y$ 轴，核心二次项系数为
		      \[
			      A^{2}b^{2}-B^{2}a^{2}
			      =
			      2^{2}\cdot4-(-1)^{2}\cdot5
			      =
			      16-5
			      =
			      11\neq0.
		      \]
		      \par\textbf{理由：} 分母非零，说明可以使用二次韦达公式。

		\item[\textbf{第三步（套用焦点在 $y$ 轴公式）：}] 使用焦点在 $y$ 轴的韦达公式：
		      \[
			      x_{1}+x_{2}=-\frac{2ACb^{2}}{A^{2}b^{2}-B^{2}a^{2}}
			      =-\frac{2\cdot2\cdot1\cdot4}{2^{2}\cdot4-(-1)^{2}\cdot5}
			      =-\frac{16}{11},
		      \]
		      \[
			      y_{1}y_{2}=\frac{-C^{2}a^{2}-A^{2}a^{2}b^{2}}{A^{2}b^{2}-B^{2}a^{2}}
			      =\frac{-1^{2}\cdot5-2^{2}\cdot5\cdot4}{11}
			      =\frac{-5-80}{11}
			      =-\frac{85}{11}.
		      \]
		      \par\textbf{理由：} 直接代入硬解定理，但必须注意焦点在 $y$ 轴时分母是 $A^{2}b^{2}-B^{2}a^{2}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $x_{1}+x_{2}=-\dfrac{16}{11},\; y_{1}y_{2}=-\dfrac{85}{11}$.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知直线 $l: y=x+m$ 与双曲线 $C: x^{2}-2y^{2}=2$ 相交于 $A,B$ 两点，且 $|AB|=4\sqrt{3}$，求实数 $m$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（标准化与参数赋值）：}] 双曲线化为标准形
		      \[
			      \frac{x^{2}}{2}-\frac{y^{2}}{1}=1,
		      \]
		      焦点在 $x$ 轴，$a^{2}=2,\;b^{2}=1$。直线化为 $x-y+m=0$，得
		      \[
			      A=1,\qquad B=-1,\qquad C=m.
		      \]
		      \par\textbf{理由：} 准备参数，同时 $B\neq0$，可以使用 $x$ 坐标弦长公式。

		\item[\textbf{第二步（检查二次项系数）：}] 焦点在 $x$ 轴，核心二次项系数为
		      \[
			      B^{2}b^{2}-A^{2}a^{2}
			      =
			      (-1)^{2}\cdot1-1^{2}\cdot2
			      =
			      -1\neq0.
		      \]
		      \par\textbf{理由：} 分母非零，可以使用二次韦达公式。

		\item[\textbf{第三步（利用硬解定理求韦达）：}] 焦点在 $x$ 轴，消去 $y$ 的韦达公式：
		      \[
			      x_{1}+x_{2}=\frac{2ACa^{2}}{B^{2}b^{2}-A^{2}a^{2}}
			      =\frac{2\cdot1\cdot m\cdot2}{(-1)^{2}\cdot1-1^{2}\cdot2}
			      =\frac{4m}{1-2}
			      =-4m,
		      \]
		      \[
			      x_{1}x_{2}=\frac{-C^{2}a^{2}-B^{2}a^{2}b^{2}}{B^{2}b^{2}-A^{2}a^{2}}
			      =\frac{-m^{2}\cdot2-1\cdot2\cdot1}{-1}
			      =2m^{2}+2.
		      \]
		      \par\textbf{理由：} 代入硬解定理公式。

		\item[\textbf{第四步（利用弦长列方程）：}] 斜率 $k=1$，弦长公式为
		      \[
			      |AB|=\sqrt{1+k^{2}}\;\sqrt{(x_{1}+x_{2})^{2}-4x_{1}x_{2}}.
		      \]
		      因此
		      \[
			      |AB|
			      =
			      \sqrt{2}\;\sqrt{(-4m)^{2}-4(2m^{2}+2)}
			      =
			      \sqrt{2}\;\sqrt{16m^{2}-8m^{2}-8}
		      \]
		      \[
			      =
			      \sqrt{2}\;\sqrt{8(m^{2}-1)}
			      =
			      4\sqrt{m^{2}-1}.
		      \]
		      已知 $|AB|=4\sqrt{3}$，故
		      \[
			      4\sqrt{m^{2}-1}=4\sqrt{3},
		      \]
		      解得
		      \[
			      m^{2}=4,
			      \qquad
			      m=\pm2.
		      \]
		      \par\textbf{理由：} 弦长公式与已知条件联立。

		\item[\textbf{第五步（检验）：}] 当 $m=\pm2$ 时，
		      \[
			      (x_{1}+x_{2})^{2}-4x_{1}x_{2}
			      =
			      8(m^{2}-1)
			      =
			      24>0.
		      \]
		      因此直线与双曲线有两个不同交点，满足题意。
		      \par\textbf{理由：} 题目要求 $A,B$ 两点，必须保证不是相切或无交点。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $m=2$ 或 $m=-2$.
\end{examplebox}
