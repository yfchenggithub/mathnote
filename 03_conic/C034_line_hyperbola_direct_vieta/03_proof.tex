\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	本证明以焦点在 $x$ 轴的双曲线
	\[
		\frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1
	\]
	为例。若 $A\neq0$，可由直线方程解出 $x$，代入双曲线后得到关于 $y$ 的方程；若 $B\neq0$，也可解出 $y$，得到关于 $x$ 的方程。二次项系数非零时使用韦达定理；二次项系数为零时，方程退化，需单独讨论。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（解出 $x$）：}] 由 $Ax+By+C=0$，若 $A\neq0$，移项得
		      \[
			      x=-\frac{By+C}{A}.
		      \]
		      \par\textbf{理由：} 将直线写成 $x$ 关于 $y$ 的表达式，便于代入双曲线消去 $x$。

		\item[\textbf{步骤二（代入双曲线）：}] 将 $x$ 代入
		      \[
			      \frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1,
		      \]
		      得
		      \[
			      \frac{1}{a^{2}}\left(-\frac{By+C}{A}\right)^{2}-\frac{y^{2}}{b^{2}}=1.
		      \]
		      \par\textbf{理由：} 代入消元，转化为只含 $y$ 的方程。

		\item[\textbf{步骤三（去分母整理）：}] 两边同时乘以 $a^{2}A^{2}b^{2}$，得
		      \[
			      b^{2}(By+C)^{2}-A^{2}a^{2}y^{2}=A^{2}a^{2}b^{2}.
		      \]
		      \par\textbf{理由：} 消除分母，将方程化为整式。

		\item[\textbf{步骤四（展开移项）：}] 展开 $(By+C)^{2}=B^{2}y^{2}+2BCy+C^{2}$，并移项合并：
		      \[
			      (b^{2}B^{2}-A^{2}a^{2})y^{2}+2b^{2}BCy+b^{2}C^{2}-A^{2}a^{2}b^{2}=0.
		      \]
		      \par\textbf{理由：} 合并 $y^{2}$、$y$ 项及常数项，整理成关于 $y$ 的方程。

		\item[\textbf{步骤五（写出二次方程）：}] 因此消去 $x$ 后的方程为
		      \[
			      (B^{2}b^{2}-A^{2}a^{2})y^{2}+2BCb^{2}\,y+(C^{2}b^{2}-A^{2}a^{2}b^{2})=0.
		      \]
		      记
		      \[
			      M=B^{2}b^{2}-A^{2}a^{2}.
		      \]
		      \par\textbf{理由：} $M$ 是关于 $y$ 的方程的二次项系数，也是判断能否使用二次韦达公式的关键。

		\item[\textbf{步骤六（使用韦达定理）：}] 若 $M\neq0$，则方程为真正的二次方程。由韦达定理得
		      \[
			      y_{1}+y_{2}=-\frac{2BCb^{2}}{M},\qquad
			      y_{1}y_{2}=\frac{C^{2}b^{2}-A^{2}a^{2}b^{2}}{M}.
		      \]
		      \par\textbf{理由：} 对二次方程 $py^{2}+qy+r=0$，有 $y_{1}+y_{2}=-\dfrac{q}{p}$，$y_{1}y_{2}=\dfrac{r}{p}$。

		\item[\textbf{步骤七（判别式意义）：}] 在 $M\neq0$ 时，设该二次方程判别式为 $\Delta$。则
		      \[
			      \Delta>0
		      \]
		      表示直线与双曲线有两个不同实交点；
		      \[
			      \Delta=0
		      \]
		      表示直线与双曲线相切，只有一个实交点；
		      \[
			      \Delta<0
		      \]
		      表示没有实交点。
		      \par\textbf{理由：} 韦达公式是代数关系，几何上是否对应两个不同交点，还要由判别式决定。

		\item[\textbf{步骤八（二次项系数为零）：}] 若 $M=0$，则关于 $y$ 的方程退化，不能使用二次韦达公式。此时直线方向与双曲线的一条渐近线方向平行。

		      若 $C\neq0$，退化后通常得到一次方程，对应一个有限交点；若 $C=0$，直线就是双曲线的一条渐近线，与双曲线没有有限交点。
		      \par\textbf{理由：} 二次项消失后，原来的“两根之和、两根之积”结构已经不存在。

		\item[\textbf{步骤九（消去 $y$ 的情形）：}] 若 $B\neq0$，同样可从直线解出
		      \[
			      y=-\frac{Ax+C}{B},
		      \]
		      代入双曲线，得到关于 $x$ 的方程及其韦达关系，结果与定理中公式一致。
		      \par\textbf{理由：} 推导过程完全类似，只是所选消元变量不同。若 $B=0$，直线为竖直线，不宜用关于 $x$ 的方程判定两个交点，应改用关于 $y$ 的方程。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	由此推导验证了焦点在 $x$ 轴时硬解定理中关于 $y$ 的公式。关于 $x$ 的公式，以及焦点在 $y$ 轴时的公式，都可通过同样的代数消元得到。使用时必须记住三件事：先看焦点位置，再看二次项系数是否为零，最后用判别式判断实际交点个数。
\end{proofbox}
