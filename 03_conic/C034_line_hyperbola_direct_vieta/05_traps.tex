\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（分母为零）：}}] 未检查二次项系数 $B^{2}b^{2}-A^{2}a^{2}$（或 $A^{2}b^{2}-B^{2}a^{2}$）是否为零，直接使用韦达公式，导致分母为零错误或错误结论。

		\item[\textbf{\textcolor{CTrap}{正确理解（退化讨论）：}}] 必须首先判断该系数是否非零。若该系数为零，则直线方向与双曲线的一条渐近线方向平行，二次韦达公式不适用，应单独讨论：
		      \[
			      C\neq0
			      \quad\Longrightarrow\quad
			      \text{消元方程退化为一次方程，通常只有一个有限交点；}
		      \]
		      \[
			      C=0
			      \quad\Longrightarrow\quad
			      \text{直线就是双曲线的一条渐近线，与双曲线没有有限交点。}
		      \]

		\item[\textbf{\textcolor{CTrap}{错因分析（记忆不全）：}}] 只强记公式，忽略了使用前提，尤其容易漏掉“直线本身就是渐近线”的特殊情形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公式混淆）：}}] 混淆焦点在 $x$ 轴与 $y$ 轴时的韦达公式，特别是 $x_{1}+x_{2}$ 的符号和分母中 $A,B$ 的位置。

		\item[\textbf{\textcolor{CTrap}{正确理解（焦点区分）：}}] 根据双曲线标准方程
		      \[
			      \frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1
		      \]
		      或
		      \[
			      \frac{y^{2}}{a^{2}}-\frac{x^{2}}{b^{2}}=1
		      \]
		      选用对应公式组。焦点在 $x$ 轴时，核心分母是 $B^{2}b^{2}-A^{2}a^{2}$；焦点在 $y$ 轴时，核心分母是 $A^{2}b^{2}-B^{2}a^{2}$。

		\item[\textbf{\textcolor{CTrap}{错因分析（死记硬背）：}}] 未理解公式来源，机械记忆导致张冠李戴。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（判别条件）：}}] 使用韦达定理前未区分 $\Delta>0$、$\Delta=0$、$\Delta<0$，在相切或无交点时仍把结果当作“两交点”处理。

		\item[\textbf{\textcolor{CTrap}{正确理解（判别条件）：}}] 当消元后是真正的二次方程时：
		      \[
			      \Delta>0
		      \]
		      表示两个不同实交点；
		      \[
			      \Delta=0
		      \]
		      表示相切，只有一个实交点，韦达仍有代数意义；
		      \[
			      \Delta<0
		      \]
		      表示没有实交点。

		      若题目要求两点弦长，则必须有 $\Delta>0$。

		\item[\textbf{\textcolor{CTrap}{错因分析（逻辑跳跃）：}}] 把“能写韦达公式”误认为“一定有两个不同交点”，忽略了判别式的几何含义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（竖直线弦长）：}}] 看到弦长就直接使用
		      \[
			      |AB|=\sqrt{1+k^{2}}\sqrt{(x_{1}+x_{2})^{2}-4x_{1}x_{2}},
		      \]
		      但忘记该公式要求直线斜率 $k$ 存在。

		\item[\textbf{\textcolor{CTrap}{正确理解（换坐标计算）：}}] 若 $B\neq0$，直线可写成 $y=kx+n$，可以使用 $x$ 坐标弦长公式。若 $B=0$，直线为竖直线，斜率不存在，应改用 $y$ 坐标：
		      \[
			      |AB|
			      =
			      \frac{\sqrt{A^{2}+B^{2}}}{|A|}
			      \sqrt{(y_{1}+y_{2})^{2}-4y_{1}y_{2}}.
		      \]

		\item[\textbf{\textcolor{CTrap}{错因分析（套公式过快）：}}] 没有先判断直线是否为竖直线，导致斜率公式被误用。
	\end{description}
\end{trapbox}
