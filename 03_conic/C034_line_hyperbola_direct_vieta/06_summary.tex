\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	直线与双曲线联立后，二次项系数及韦达关系可以由公式直接写出；但使用前必须先检查焦点位置、消元方向、二次项系数和判别式。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（直线标准）：} 直线方程为
		      \[
			      Ax+By+C=0,\qquad A^{2}+B^{2}>0.
		      \]
		      若使用关于 $y$ 的方程，通常要求 $A\neq0$；若使用关于 $x$ 的方程，通常要求 $B\neq0$。

		\item \textbf{条件 2（双曲线标准）：} 双曲线为标准方程
		      \[
			      \frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1
		      \]
		      或
		      \[
			      \frac{y^{2}}{a^{2}}-\frac{x^{2}}{b^{2}}=1,
		      \]
		      其中 $a>0,\;b>0$。

		\item \textbf{条件 3（二次项非零）：} 焦点在 $x$ 轴时，核心二次项系数为
		      \[
			      B^{2}b^{2}-A^{2}a^{2}.
		      \]
		      焦点在 $y$ 轴时，核心二次项系数为
		      \[
			      A^{2}b^{2}-B^{2}a^{2}.
		      \]
		      只有该系数非零时，才能直接使用二次韦达公式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{提醒 1（分母为零）：} 分母为零时，直线方向与渐近线方向平行。若 $C\neq0$，通常退化为一次方程；若 $C=0$，直线本身就是渐近线，与双曲线没有有限交点。

		\item \textbf{提醒 2（判别式）：} 真正的二次方程中，$\Delta>0$ 表示两个不同实交点，$\Delta=0$ 表示相切，$\Delta<0$ 表示无实交点。题目要求两点弦长时，必须有 $\Delta>0$。

		\item \textbf{提醒 3（弦长公式）：} 若 $B\neq0$，可用
		      \[
			      |AB|=\sqrt{1+k^{2}}\sqrt{(x_{1}+x_{2})^{2}-4x_{1}x_{2}},
			      \qquad k=-\frac{A}{B}.
		      \]
		      若 $B=0$，直线为竖直线，斜率不存在，应改用 $y$ 坐标弦长公式。

		\item \textbf{提醒 4（不要死套）：} 先判断焦点位置，再判断是否退化，最后结合判别式确认几何意义。
	\end{itemize}
\end{summarybox}
