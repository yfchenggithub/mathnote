\begin{summarybox}
\textbf{\textcolor{CSummary}{C020｜30 秒记忆卡}}

\textbf{适用：}椭圆 $x^2/a^2+y^2/b^2=1$，$a>b>0$；$P$ 在椭圆上，$l$ 为 $P$ 处切线；$r_1,r_2$ 是 $P$ 到焦点的距离，$d$ 是\textbf{中心}到切线的垂距。

\[
\boxed{\sqrt{r_1r_2}\,d=ab}
\qquad\Longleftrightarrow\qquad
\boxed{r_1r_2d^2=a^2b^2}.
\]

\textbf{最快几何复原（两步）：}
\[
\begin{aligned}
\text{角平分线 + }r_1+r_2=2a:
&\quad d=a\cos\frac\theta2,\\
\text{余弦定理 + }c^2=a^2-b^2:
&\quad r_1r_2\cos^2\frac\theta2=b^2.
\end{aligned}
\]
\textbf{最快代数复原（同一个因子）：}
\[
\boxed{r_1r_2=a^2-\frac{c^2x_0^2}{a^2}},\qquad
\boxed{\frac1{d^2}=\frac{r_1r_2}{a^2b^2}}.
\]

\textbf{两个端点记忆：}
$P=(\pm a,0):\ (d,\sqrt{r_1r_2})=(a,b)$；
$P=(0,\pm b):\ (d,\sqrt{r_1r_2})=(b,a)$。

\textbf{重要范围：}
\[
\boxed{b\le d\le a},\qquad
\boxed{b\le\sqrt{r_1r_2}\le a}.
\]

\textbf{知识地图联结：}
椭圆定义 $r_1+r_2=2a$ $\to$ 切线光学性质 $\to$ 余弦定理；
以及焦点到切线垂距 $h_1h_2=b^2$ 与 $h_i/r_i=d/a$。
圆 $a=b=R$ 的退化特例是 $r_1=r_2=d=R$。

\textbf{考场顺序：}先检查 $P$ 是切点、$d$ 是中心距离且 $b\le d\le a$；
有焦点夹角优先几何法，有 $x_0$ 优先焦半径一次式，有切线方程优先点到直线距离。
\end{summarybox}
