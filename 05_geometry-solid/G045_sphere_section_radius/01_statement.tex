	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（球半径已知）：}}] 球的半径为$R$。
		\item[\textbf{\textcolor{CStatement}{条件 2（距离范围）：}}] 球心到平面的距离$d$满足$0\le d<R$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（截面半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			球心到截面垂线、截面半径、球半径构成直角三角形，勾股定理导出。
			\[
				r = \sqrt{R^{2} - d^{2}}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（逆用公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若已知球半径 $R$ 和截面圆半径 $r$（$0\le r\le R$），则球心到平面的距离 $d$ 可通过代数变形求得。
			\[
				d = \sqrt{R^{2} - r^{2}}.
			\]
	\end{description}
\end{statementbox}
