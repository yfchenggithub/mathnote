\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{球截面半径$r$、球心到平面的距离$d$与球半径$R$满足$r=\sqrt{R^{2}-d^{2}}$，该关系源于直角三角形勾股定理。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（球半径已知）：} 需要已知球的半径$R$。
		\item \textbf{条件 2（距离范围）：} 球心到平面的距离$d$必须满足$0\le d<R$，否则截面不存在。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略距离范围）：} 使用公式前必检$d<R$，避免根号内非正。
		\item \textbf{检查项（区分直角边）：} $d$和$r$都是直角边，$R$是斜边，不要混淆。
	\end{itemize}
\end{summarybox}
