\begin{trapbox}

	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略绝对值）}}]
			直接使用$\cos\theta=\frac{S_{\text{投影}}}{S_{\text{原}}}$，未考虑二面角可能为钝角。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式带绝对值）：}}]
			射影面积法核心是$|\cos\theta|=\frac{S_{\text{投影}}}{S_{\text{原}}}$，再根据二面角锐钝决定$\cos\theta$的正负。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视钝角）：}}]
			习惯性认为二面角为锐角，忽略符号讨论，导致结论符号错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（投影非正投）}}]
			将非正投影（如斜投影）当作正投影使用公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（正投影前提）：}}]
			公式要求投影是正投影（投影线垂直于投影面），否则面积比不等于$|\cos\theta|$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			未区分正投影与斜投影，误以为任意投影下面积关系都成立。
	\end{description}

	\medskip
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（图形重叠）}}]
			原平面图形有折叠或重叠部分时直接套用公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（图形无重叠）：}}]
			公式推导基于图形无折叠重叠，图形必须是简单的平面区域。
		\item[\textbf{\textcolor{CTrap}{错因分析（复杂图形处理）：}}]
			将交叉、重叠的图形当作简单图形，忽略了面积重复计算。
	\end{description}

\end{trapbox}
