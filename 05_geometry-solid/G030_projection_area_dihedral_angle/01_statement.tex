\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（图形无重叠）：}}]
			平面图形$S_{\text{原}}$无折叠与重叠部分。
		\item[\textbf{\textcolor{CStatement}{条件2（垂直投影）：}}]
			该图形在另一平面上的正投影面积为$S_{\text{投影}}$，投影方向垂直于投影面。
		\item[\textbf{\textcolor{CStatement}{条件3（角度范围）：}}]
			两平面所成二面角的平面角$\theta$满足$0<\theta<\pi$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（绝对值关系）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			投影面积与原面积之比等于二面角余弦的绝对值。
			\[
				\boxed{|\cos\theta|=\frac{S_{\text{投影}}}{S_{\text{原}}}}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（符号判定）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			锐二面角时直接取正，钝二面角时取负。
			\[
				\cos\theta=
				\begin{cases}
					\dfrac{S_{\text{投影}}}{S_{\text{原}}},&0<\theta<\dfrac{\pi}{2},\
					\[
						1.2em]-\dfrac{S_{\text{投影}}}{S_{\text{原}}},&\dfrac{\pi}{2}<\theta<\pi.
				\end{cases}
				\]
		\end{description}

	\end{statementbox}
