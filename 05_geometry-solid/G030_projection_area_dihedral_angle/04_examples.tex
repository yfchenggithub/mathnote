\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}在棱长为1的正方体$ABCD-A_1B_1C_1D_1$中，求二面角$A_1-BD-A$的余弦值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（计算原面积）：}]
			$\triangle A_1BD$是边长为$\sqrt{2}$的等边三角形，面积$S_{\text{原}}=\frac{\sqrt{3}}{4}(\sqrt{2})^2=\frac{\sqrt{3}}{2}$。
			\par\textbf{理由：}$A_1B=A_1D=BD=\sqrt{2}$。
		\item[\textbf{第二步（计算投影面积）：}]
			$\triangle A_1BD$在底面$ABCD$上的正投影为$\triangle ABD$，$AB=AD=1$，$S_{\text{投影}}=\frac12\times1\times1=\frac12$。
			\par\textbf{理由：}$A_1$投影为$A$，$B,D$在底面。
		\item[\textbf{第三步（求二面角余弦）：}]
			由射影面积法，$|\cos\theta|=\frac{S_{\text{投影}}}{S_{\text{原}}}=\frac{1/2}{\sqrt{3}/2}=\frac{1}{\sqrt{3}}=\frac{\sqrt{3}}{3}$。二面角$A_1-BD-A$为锐角，所以$\cos\theta=\frac{\sqrt{3}}{3}$。
			\par\textbf{理由：}公式$|\cos\theta|=S_{\text{投影}}/S_{\text{原}}$，并由直观判断锐钝。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\cos\theta=\frac{\sqrt{3}}{3}$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}在$60^\circ$的二面角$\alpha-l-\beta$中，平面$\alpha$内矩形$ABCD$的边$AB\parallel l$，$AB=4$，$BC=3$。求矩形$ABCD$在平面$\beta$上的正投影面积。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（求原面积）：}]
			矩形面积$S_{\text{原}}=4\times3=12$。
			\par\textbf{理由：}边长已知，面积公式。
		\item[\textbf{第二步（分析投影边长）：}]
			$AB$平行于棱$l$，投影后长度不变仍为$4$；$BC$垂直于$l$，投影后长度变为$3\cos60^\circ=1.5$。
			\par\textbf{理由：}$AB\parallel l$且$l\subset\beta$，故$AB\parallel\beta$，投影长度不变；$BC$与$\beta$夹角等于二面角$\theta=60^\circ$，投影长度缩短为$BC\cos\theta$。
		\item[\textbf{第三步（求投影面积）：}]
			投影后仍为矩形，面积$S_{\text{投影}}=4\times1.5=6$。
			\par\textbf{理由：}矩形面积公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{正投影面积为$6$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}已知四棱锥$P-ABCD$，底面$ABCD$是边长为2的正方形，$PA\perp$底面，且$PA=2$，求侧面$PCD$与底面$ABCD$所成二面角的余弦值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（求投影面积）：}]
			$P$在底面投影为$A$，故$\triangle PCD$在底面正投影为$\triangle ACD$。$S_{\text{投影}}=S_{\triangle ACD}=\frac12\times2\times2=2$。
			\par\textbf{理由：}正方形面积一半。
		\item[\textbf{第二步（求侧面$PCD$面积）：}]
			由$CD\perp AD$和$CD\perp PA$，得$CD\perp$平面$PAD$，故$CD\perp PD$。所以$\triangle PCD$是直角三角形，$CD=2$，$PD=\sqrt{PA^2+AD^2}=2\sqrt{2}$，$S_{\text{原}}=\frac12\times2\times2\sqrt{2}=2\sqrt{2}$。
			\par\textbf{理由：}线面垂直得线线垂直，用勾股定理求直角边。
		\item[\textbf{第三步（求二面角余弦）：}]
			由射影面积法，$|\cos\theta|=\frac{S_{\text{投影}}}{S_{\text{原}}}=\frac{2}{2\sqrt{2}}=\frac{\sqrt{2}}{2}$。结合图形知二面角为锐角，故$\cos\theta=\frac{\sqrt{2}}{2}$。
			\par\textbf{理由：}公式$|\cos\theta|=S_{\text{投影}}/S_{\text{原}}$，并注意锐钝。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\cos\theta=\frac{\sqrt{2}}{2}$。}
\end{examplebox}
