\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一个直圆柱的底面半径为 $3$ 厘米，高为 $5$ 厘米，求该圆柱的体积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（找条件）：}] 明确圆柱的底面半径 $r=3$，高 $h=5$。
			\par\textbf{理由：}题目直接给出。
			\[
				r=3,\quad h=5.
			\]
		\item[\textbf{第二步（算底面积）：}] 由圆面积公式，底面面积为
			\[
				\begin{aligned}
					S &= \pi r^2 \\
					&= \pi \times 3^2 \\
					&= 9\pi.
			\end{aligned}
			\]
			\par\textbf{理由：}圆面积公式 $S=\pi r^2$。
		\item[\textbf{第三步（代公式）：}] 代入柱体体积公式 $V=Sh$，得
			\[
				\begin{aligned}
					V &= 9\pi \times 5 \\
					&= 45\pi.
			\end{aligned}
			\]
			\par\textbf{理由：}柱体体积公式 $V=Sh$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	该圆柱的体积为 $45\pi$ 立方厘米。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	斜三棱柱 $ABC$-$A'B'C'$，底面是边长为 $2$ 的正三角形，侧棱 $AA'=4$，$AA'$ 与底面 $ABC$ 所成角为 $60^\circ$，求该斜三棱柱的体积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（算底面积）：}] 底面正三角形的面积为
			\[
				\begin{aligned}
					S &= \frac{\sqrt{3}}{4}a^2 \\
					&= \frac{\sqrt{3}}{4} \times 4 \\
					&= \sqrt{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}正三角形面积公式 $S = \frac{\sqrt{3}}{4}a^2$。
		\item[\textbf{第二步（求垂直高）：}] 高 $h$ 由侧棱长与夹角确定：
			\[
				\begin{aligned}
					h &= AA' \cdot \sin 60^\circ \\
					&= 4 \times \frac{\sqrt{3}}{2} \\
					&= 2\sqrt{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}侧棱与底面所成角的正弦等于垂直高与侧棱长之比。
		\item[\textbf{第三步（代公式求体积）：}] 由柱体体积公式 $V=Sh$ 得
			\[
				\begin{aligned}
					V &= \sqrt{3} \times 2\sqrt{3} \\
					&= 6.
			\end{aligned}
			\]
			\par\textbf{理由：}柱体体积公式 $V=Sh$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	该斜三棱柱的体积为 $6$（立方单位）。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一个直六棱柱的底面是正六边形，侧棱长为 $l$，它的侧面展开图是一个正方形，求该六棱柱的体积（用 $l$ 表示）。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（建立边长关系）：}] 设正六边形边长为 $a$，则底面周长 $C=6a$。侧面展开图的边长分别为 $C$ 和侧棱长 $l$，因为展开图是正方形，所以 $C=l$，即 $6a=l$，解得 $a=\dfrac{l}{6}$。
			\par\textbf{理由：}棱柱侧面展开图是矩形，一组边长等于底面周长，另一组等于高（侧棱长）。正方形要求两边相等。
		\item[\textbf{第二步（计算底面积）：}] 正六边形的面积公式 $S = \dfrac{3\sqrt{3}}{2}a^2$，代入 $a$ 得
			\[
				\begin{aligned}
					S &= \dfrac{3\sqrt{3}}{2}a^2 \\
					&= \dfrac{3\sqrt{3}}{2}\left(\dfrac{l}{6}\right)^2 \\
					&= \dfrac{3\sqrt{3}}{2} \cdot \dfrac{l^2}{36} \\
					&= \dfrac{\sqrt{3}}{24}\,l^2.
			\end{aligned}
			\]
			\par\textbf{理由：}正六边形面积公式。
		\item[\textbf{第三步（求体积）：}] 直棱柱的体积 $V = S h$，高 $h=l$，所以
			\[
				\begin{aligned}
					V &= \dfrac{\sqrt{3}}{24}\,l^2 \cdot l \\
					&= \dfrac{\sqrt{3}}{24}\,l^3.
			\end{aligned}
			\]
			\par\textbf{理由：}柱体体积公式 $V=Sh$，此处高就是侧棱长（因为直棱柱）。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	该直六棱柱的体积为 $\dfrac{\sqrt{3}}{24}\,l^3$。
\end{examplebox}
