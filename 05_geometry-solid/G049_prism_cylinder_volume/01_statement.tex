\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（柱体类型）：}}] 所给几何体为柱体（直棱柱、斜棱柱、直圆柱、斜圆柱），即有两个平行且全等的底面，侧面由平行于侧棱或母线的直线段构成。
		\item[\textbf{\textcolor{CStatement}{条件 2（底面积给定）：}}] 底面面积为 $S$。
		\item[\textbf{\textcolor{CStatement}{条件 3（高为垂直距离）：}}] 两底面间的垂直距离为 $h$，不是侧棱长度。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（体积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			柱体的体积等于底面积乘以垂直高度。
			\[
				V = S h.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（等底等高性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			底面积相等且高相等的任意两个柱体体积相同。
			\[
				\begin{aligned}
					S_1 &= S_2,\\
					h_1 &= h_2,\\
					\therefore V_1 &= V_2.
			\end{aligned}
			\]
	\end{description}
\end{statementbox}
