\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{正四面体的高等于棱长的$\frac{\sqrt{6}}{3}$倍，记为$h = \frac{\sqrt{6}}{3}a$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（正四面体）：}多面体必须是正四面体，即所有棱长相等。
		\item \textbf{条件2（已知棱长）：}已知棱长$a$，直接求高；若已知高，可反求棱长。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（记错系数）：}容易错误记成$\frac{\sqrt{3}}{2}a$或$\frac{\sqrt{2}}{3}a$，需记忆$\frac{\sqrt{6}}{3}a$。
		\item \textbf{检查项（底面中心定位）：}高线的垂足是底面正三角形的中心，不是任意一点。
	\end{itemize}
\end{summarybox}
