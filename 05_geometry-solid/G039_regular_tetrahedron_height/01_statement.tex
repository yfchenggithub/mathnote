	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（棱长相等）：}}]正四面体所有棱长均为$a$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（高公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			高等于棱长乘以常数$\frac{\sqrt{6}}{3}$.
			\[
				h = \frac{\sqrt{6}}{3} a
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（体积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			体积等于棱长立方乘常数$\frac{\sqrt{2}}{12}$.
			\[
				V = \frac{\sqrt{2}}{12} a^3
			\]
	\end{description}
\end{statementbox}
