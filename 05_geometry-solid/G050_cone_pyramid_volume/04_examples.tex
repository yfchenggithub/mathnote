\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	一个圆锥的底面半径$r=3$，高$h=4$，求这个圆锥的体积$V$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面积）：}]
			$S=\pi r^2=9\pi$。
			\par\textbf{理由：}圆的面积公式。
		\item[\textbf{第二步（代公式）：}]
			$V=\frac{1}{3}Sh=\frac{1}{3}\times 9\pi\times 4 =12\pi$。
			\par\textbf{理由：}锥体体积公式$V=\frac{1}{3}Sh$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$V=12\pi$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	一个正三棱锥的底面是边长为$2\sqrt{3}$的正三角形，它的体积是$4\sqrt{3}$，求这个棱锥的高$h$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面积）：}]
			$S=\frac{\sqrt{3}}{4}a^2=\frac{\sqrt{3}}{4}(2\sqrt{3})^2=3\sqrt{3}$。
			\par\textbf{理由：}正三角形面积公式$S=\frac{\sqrt{3}}{4}a^2$。
		\item[\textbf{第二步（逆用公式）：}]
			由$V=\frac{1}{3}Sh$得$h=\frac{3V}{S}$。
			\par\textbf{理由：}等式性质。
		\item[\textbf{第三步（代入求值）：}]
			$h=\frac{3\times 4\sqrt{3}}{3\sqrt{3}}=4$。
			\par\textbf{理由：}代入数值并化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{高 $h=4$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	一个圆锥的体积为$V$，底面半径为$r$，试用$V$和$r$表示它的侧面积$S_{\text{l}}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（体积求高）：}]
			由$V=\frac{1}{3}\pi r^2 h$，得$h=\frac{3V}{\pi r^2}$。
			\par\textbf{理由：}锥体体积公式变形。
		\item[\textbf{第二步（求母线）：}]
			母线$l=\sqrt{r^2+h^2}=\sqrt{r^2+\frac{9V^2}{\pi^2 r^4}}=\frac{\sqrt{\pi^2 r^6+9V^2}}{\pi r^2}$。
			\par\textbf{理由：}勾股定理。
		\item[\textbf{第三步（求侧面积）：}]
			$S_{\text{l}}=\pi r l=\pi r\cdot\frac{\sqrt{\pi^2 r^6+9V^2}}{\pi r^2}=\frac{\sqrt{\pi^2 r^6+9V^2}}{r}$。
			\par\textbf{理由：}圆锥侧面积公式$S_{\text{l}}=\pi r l$，并化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_{\text{l}}=\frac{\sqrt{\pi^2 r^6+9V^2}}{r}$。}
\end{examplebox}
