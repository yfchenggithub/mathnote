\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（锥体定义）：}}]
			底面为多边形（棱锥）或圆（圆锥），顶点在底面所在平面外的几何体。
		\item[\textbf{\textcolor{CStatement}{条件 2（已知量）：}}]
			底面积为$S$，高为$h$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（体积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			锥体体积等于底面积与高乘积的三分之一。
			\[
				V = \frac{1}{3} S h
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（柱体比例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			等底等高的柱体（棱柱或圆柱）体积是锥体体积的三倍。
			\[
				V_c = 3 V
			\]
	\end{description}

\end{statementbox}
