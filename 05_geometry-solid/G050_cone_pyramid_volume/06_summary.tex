\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{$V=\frac{1}{3}Sh$，锥体体积是等底等高柱体体积的三分之一。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（锥体形状）：}几何体必须是棱锥（底面为多边形）或圆锥（底面为圆）。
		\item \textbf{条件 2（已知量）：}需已知底面积$S$和高$h$，两者单位一致。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（系数与垂高）：}不可漏写$\frac{1}{3}$，且$h$为顶点到底面的垂直距离，绝非斜高或母线。
		\item \textbf{检查项（底面积复核）：}代公式前应验算底面积计算是否准确（尤其当底面为非圆形时）。
	\end{itemize}

\end{summarybox}
