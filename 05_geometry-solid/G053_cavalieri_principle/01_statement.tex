\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（夹持等高）：}}] 两个几何体 $A,B$ 夹在两平行平面之间，高均为 $H$。
		\item[\textbf{\textcolor{CStatement}{条件2（截面相等）：}}] 对于任意高度 $h\in[0,H]$，平行于底面的平面截得的截面面积 $S_A(h)=S_B(h)$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（体积相等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两几何体在相同高度范围内体积相同。
			\[
				V_A = V_B.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（截面成比例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若截面面积恒成比例，则体积也成相同比例。
			\[
				S_A(h) = k\,S_B(h)\ \ (\forall h\in[0,H])\ \Longrightarrow\ V_A = k\,V_B.
			\]
	\end{description}

\end{statementbox}
