\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{体积等于截面积沿高方向的累积，若两几何体处处截面积相等，则体积相等。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（夹持等高）：} 两个几何体夹在两平行平面之间，高相等。
		\item \textbf{条件2（截面相等）：} 所有平行于底面的截面面积对应相等。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（检查截面平行）：} 截面必须平行于底面，不能使用斜截面。
		\item \textbf{检查项（验证全高度）：} 须确保截面积函数在全部高度上处处相等，不可只验几点。
	\end{itemize}
\end{summarybox}
