\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略直棱柱条件）}}] 错误地将公式用到斜棱柱或无外接圆的棱柱上。
		\item[\textbf{\textcolor{CTrap}{正确理解（直棱柱才适用）}}] 公式
		      \[
			      R = \sqrt{r^{2}+\left(\frac{h}{2}\right)^{2}}
		      \]
		      只适用于侧棱垂直底面且底面存在外接圆的直棱柱。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺少垂直条件）}}] 对于一般斜棱柱，上、下底面外心连线通常不垂直于底面，球心无法按“底面外接圆半径 $r$ 与半高 $h/2$”构造直角三角形。因此即使底面有外接圆，也不能直接套用直棱柱公式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（半高系数遗漏）}}] 记错公式为
		      \[
			      R = \sqrt{r^{2} + h^{2}},
		      \]
		      遗漏了 $\frac{h}{2}$ 中的除以 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（半高代入公式）}}] 球心在上、下底面外心连线的中点，到任一底面的距离都是 $h/2$，公式必须使用半高 $h/2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（对称性忽视）}}] 只关注了底面外接圆半径，未考虑球心在棱柱高度方向的中间位置。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（半径计算陷阱）}}] 求底面外接圆半径时，错误地用内切圆半径、边长、高或其他长度替代。
		\item[\textbf{\textcolor{CTrap}{正确理解（依形状选用公式）}}] 常见底面外接圆半径公式如下：
		      \[
			      \text{正三角形： } r=\frac{a}{\sqrt{3}},\qquad
			      \text{正方形： } r=\frac{a}{\sqrt{2}},\qquad
			      \text{直角三角形： } r=\frac{c}{2}.
		      \]
		      其中 $a$ 表示边长，$c$ 表示直角三角形的斜边。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）}}] 未区分外接圆与内切圆，或记错多边形外接圆半径公式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（反求参数不检查）}}] 已知 $R$ 和 $h$ 反求 $r$ 时，直接写
		      \[
			      r=\sqrt{R^2-\left(\frac{h}{2}\right)^2},
		      \]
		      但没有检查根号内是否非负。
		\item[\textbf{\textcolor{CTrap}{正确理解（先检查存在性）}}] 必须先检查
		      \[
			      R\geq \frac{h}{2}.
		      \]
		      若底面非退化，则应有
		      \[
			      R>\frac{h}{2}.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略几何约束）}}] 外接球半径不可能小于球心到任一底面的距离。若 $R<h/2$，则不存在满足条件的直棱柱。
	\end{description}
\end{trapbox}
