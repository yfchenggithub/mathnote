\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知直三棱柱的底面是边长为 $6$ 的正三角形，侧棱长（高）为 $4$，求该直三棱柱外接球的半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求外接圆半径）：}]
		      正三角形外接圆半径
		      \[
			      r = \frac{a}{\sqrt{3}}.
		      \]
		      代入 $a=6$ 得
		      \[
			      r = \frac{6}{\sqrt{3}} = 2\sqrt{3}.
		      \]
		      \par\textbf{理由：} 正三角形外接圆半径公式 $r = a/\sqrt{3}$。

		\item[\textbf{第二步（代入外接球公式）：}]
		      直棱柱外接球半径公式为
		      \[
			      R = \sqrt{r^{2} + \left(\frac{h}{2}\right)^{2}}.
		      \]
		      代入 $r = 2\sqrt{3}, h=4$ 得
		      \[
			      \begin{aligned}
				      R & = \sqrt{(2\sqrt{3})^{2} + \left(\frac{4}{2}\right)^{2}} \\
				        & = \sqrt{12 + 4}                                         \\
				        & = 4.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 代入已知数据并化简。

		\item[\textbf{第三步（给出答案）：}]
		      所以外接球半径
		      \[
			      R = 4.
		      \]
		      \par\textbf{理由：} 计算结果即为最终答案。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$R = 4$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知直四棱柱（底面为正方形）的外接球半径为 $5$，侧棱长（高）为 $6$，求底面正方形的边长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（检查存在性）：}]
		      已知 $R=5,h=6$，先检查
		      \[
			      R\geq \frac{h}{2}.
		      \]
		      因为
		      \[
			      5>\frac{6}{2}=3,
		      \]
		      所以可以反求底面外接圆半径，且底面非退化。
		      \par\textbf{理由：} 反求 $r$ 时必须保证 $R^2-(h/2)^2\geq 0$。

		\item[\textbf{第二步（反求外接圆半径）：}]
		      由
		      \[
			      R = \sqrt{r^{2} + \left(\frac{h}{2}\right)^{2}}
		      \]
		      得
		      \[
			      r^{2} = R^{2} - \left(\frac{h}{2}\right)^{2}.
		      \]
		      代入 $R=5, h=6$ 得
		      \[
			      \begin{aligned}
				      r^{2} & = 5^{2} - \left(\frac{6}{2}\right)^{2} \\
				            & = 25 - 9                               \\
				            & = 16.
			      \end{aligned}
		      \]
		      因此
		      \[
			      r = 4.
		      \]
		      \par\textbf{理由：} 代数变形。

		\item[\textbf{第三步（正方形边长公式）：}]
		      正方形外接圆半径
		      \[
			      r = \frac{a}{\sqrt{2}}.
		      \]
		      所以
		      \[
			      a = \sqrt{2}\, r = 4\sqrt{2}.
		      \]
		      \par\textbf{理由：} 正方形对角线等于 $2r$，边长 $a = \sqrt{2}\, r$。

		\item[\textbf{第四步（给出答案）：}]
		      底面正方形边长为
		      \[
			      4\sqrt{2}.
		      \]
		      \par\textbf{理由：} 计算结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$a = 4\sqrt{2}$}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知直三棱柱的底面三角形三边长分别为 $3,4,5$，侧棱长（高）为 $8$，求该直三棱柱外接球的表面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求外接圆半径）：}]
		      因为
		      \[
			      3^{2}+4^{2}=5^{2},
		      \]
		      所以底面是直角三角形。直角三角形外接圆半径等于斜边的一半：
		      \[
			      r = \frac{5}{2}.
		      \]
		      \par\textbf{理由：} 直角三角形的外心是斜边中点，因此外接圆半径 $r=c/2$，其中 $c$ 为斜边。

		\item[\textbf{第二步（求外接球半径）：}]
		      代入公式
		      \[
			      R = \sqrt{r^{2} + \left(\frac{h}{2}\right)^{2}},
		      \]
		      其中 $r=\frac{5}{2}, h=8$，得
		      \[
			      \begin{aligned}
				      R & = \sqrt{\left(\frac{5}{2}\right)^{2} + \left(\frac{8}{2}\right)^{2}} \\
				        & = \sqrt{\frac{25}{4} + 16}                                           \\
				        & = \sqrt{\frac{25}{4}+\frac{64}{4}}                                   \\
				        & = \sqrt{\frac{89}{4}}                                                \\
				        & = \frac{\sqrt{89}}{2}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 直接代入公式，化简根式。

		\item[\textbf{第三步（求外接球表面积）：}]
		      球的表面积公式为
		      \[
			      S = 4\pi R^{2}.
		      \]
		      由
		      \[
			      R^{2} = \frac{89}{4},
		      \]
		      得
		      \[
			      \begin{aligned}
				      S & = 4\pi R^{2}              \\
				        & = 4\pi \cdot \frac{89}{4} \\
				        & = 89\pi.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 表面积公式及代数化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$S = 89\pi$}
\end{examplebox}
