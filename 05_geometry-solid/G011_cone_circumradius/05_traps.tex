\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（对象不明确）：}}] 只写“圆锥”而不说明是直圆锥，容易把斜圆锥也误套公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（适用对象）：}}] 本公式适用于直圆锥（正圆锥），其顶点在底面圆心的正上方，满足$l^2=r^2+h^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件省略）：}}] 公式依赖轴截面为等腰三角形；若是一般斜圆锥，不能直接用$R=\dfrac{r^2+h^2}{2h}$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（记忆混淆）：}}] 误以为外接球半径等于$\dfrac{h}{2}+r$（高的一半加底面半径）。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确形式）：}}] 正确公式为$R = \dfrac{h}{2} + \dfrac{r^2}{2h}$，第二项是$\dfrac{r^2}{2h}$而不是$r$。
		\item[\textbf{\textcolor{CTrap}{错因分析（类比错误）：}}] 错误地认为球心在高的中点处，忽略了底面圆周到球心距离也必须等于球半径。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式选择不当）：}}] 在已知$l$和$h$时，仍去先求$r$再代入$R=\dfrac{r^2+h^2}{2h}$，增加计算量。
		\item[\textbf{\textcolor{CTrap}{正确理解（灵活选用）：}}] 已知$l$和$h$时直接使用$R=\dfrac{l^2}{2h}$更便捷。
		\item[\textbf{\textcolor{CTrap}{错因分析（表达式不熟悉）：}}] 只记住一种形式，不熟悉等价变换，导致步骤冗余。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（球心位置误解）：}}] 认为球心一定在圆锥内部（锥体内部），当$h<r$时怀疑公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（位置可变）：}}] 公式$R=\dfrac{r^2+h^2}{2h}$对任意$r>0,h>0$均成立。若以底面圆心为原点、向顶点方向为正方向，则球心坐标$d=\dfrac{h^2-r^2}{2h}$；当$h<r$时$d<0$，球心在底面下方。
		\item[\textbf{\textcolor{CTrap}{错因分析（定性想象不足）：}}] 缺乏对空间位置的分析，只考虑常见情形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点五（外接球与最小容纳球混淆）：}}] 看到“圆锥放入球中，求最小球半径”，就直接套外接球公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（先看题意）：}}] 若题意是“外接球”，球面必须经过顶点和底面圆周，使用$R=\dfrac{r^2+h^2}{2h}$；若只是“能容纳圆锥的最小球”，则
		      \[
			      R_{\min}=\begin{cases}
				      r,                   & 0<h\le r, \\[4pt]
				      \dfrac{r^2+h^2}{2h}, & h\ge r.
			      \end{cases}
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混用）：}}] 外接球要求顶点和底面圆周都在球面上；最小容纳球只要求整个圆锥在球内，两者条件不同。
	\end{description}
\end{trapbox}
