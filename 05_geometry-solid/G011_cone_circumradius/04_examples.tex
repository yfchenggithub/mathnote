	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}已知直圆锥底面半径$r=3$，高$h=4$，求圆锥的外接球半径$R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求母线长）：}] 由勾股定理，母线长$l = \sqrt{r^2+h^2} = \sqrt{3^2+4^2}=5$。
		      \par\textbf{理由：}直圆锥的母线、底面半径、高构成直角三角形。
		\item[\textbf{第二步（套公式）：}] 将$l=5$，$h=4$代入$R = \dfrac{l^2}{2h}$得：
		      \[
			      R = \frac{5^2}{2\times4} = \frac{25}{8}.
		      \]
		      \par\textbf{理由：}直圆锥外接球半径公式可直接使用。
		\item[\textbf{第三步（计算结果）：}] $\dfrac{25}{8} = 3.125$，故外接球半径为$3.125$（或写为$\dfrac{25}{8}$）。
		      \par\textbf{理由：}已获得最终数值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{外接球半径 $R = \dfrac{l^2}{2h} = \dfrac{25}{8}$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}已知直圆锥母线长$l=10$，高$h=8$，求外接球半径$R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面半径）：}] 由$l^2 = r^2 + h^2$得$r^2 = l^2 - h^2 = 100 - 64 = 36$，故$r=6$。
		      \par\textbf{理由：}直圆锥基本关系。
		\item[\textbf{第二步（选择公式）：}] 可用$R = \dfrac{l^2}{2h} = \dfrac{100}{16} = \dfrac{25}{4}$；或$R = \dfrac{r^2+h^2}{2h} = \dfrac{36+64}{16} = \dfrac{100}{16}$，结果一致。
		      \[
			      R = \frac{100}{16} = \frac{25}{4}.
		      \]
		      \par\textbf{理由：}两种形式等价。
		\item[\textbf{第三步（得答案）：}] 因此外接球半径$R = \dfrac{25}{4} = 6.25$。
		      \par\textbf{理由：}完成计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$R = \dfrac{l^2}{2h} = \dfrac{25}{4}$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}已知直圆锥的外接球半径$R=5$，高$h=8$，求圆锥底面半径$r$和母线长$l$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（检查存在性）：}] 反求底面半径时，由
		      \[
			      r^2=2Rh-h^2
		      \]
		      可知若要求$r>0$，需满足$0<h<2R$。本题$0<8<10$，满足条件。
		      \par\textbf{理由：}先确认反求出的底面半径为正数。

		\item[\textbf{第二步（代入公式）：}] 由$R = \dfrac{r^2+h^2}{2h}$得：
		      \[
			      5 = \frac{r^2+8^2}{2\times8} = \frac{r^2+64}{16}.
		      \]
		      \par\textbf{理由：}直接代入已知数据。
		\item[\textbf{第三步（解方程）：}] 两边乘以$16$得$80 = r^2+64$，所以$r^2=16$，即$r=4$（$r>0$）。
		      \[
			      r^2 = 80-64 = 16 \quad\Rightarrow\quad r=4.
		      \]
		      \par\textbf{理由：}解一元二次方程，取正根。
		\item[\textbf{第四步（求母线长）：}] 由$l^2 = r^2+h^2 = 16+64=80$，故$l = \sqrt{80} = 4\sqrt{5}$。
		      \[
			      l = \sqrt{80} = 4\sqrt{5}.
		      \]
		      \par\textbf{理由：}勾股定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{底面半径 $r=4$，母线长 $l=4\sqrt{5}$。}
