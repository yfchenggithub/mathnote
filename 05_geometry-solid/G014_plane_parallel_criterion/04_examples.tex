\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在正方体 $ABCD-A_1B_1C_1D_1$ 中，证明平面 $A_1BD\parallel$ 平面 $B_1CD_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线线平行）：}] 连接 $A_1B,\;A_1D,\;D_1C,\;B_1C$。
			由 $A_1D_1\parallel BC$ 且 $A_1D_1=BC$，得四边形 $A_1BCD_1$ 是平行四边形，故 $A_1B\parallel D_1C$。
			同理 $A_1D\parallel B_1C$。
			\[
				A_1B\parallel D_1C,\quad A_1D\parallel B_1C
			\]
			\par\textbf{理由：}
			平行四边形对边平行。

		\item[\textbf{第二步（线面平行）：}]
			因为 $A_1B\parallel D_1C$，且 $D_1C\subset$ 平面 $B_1CD_1$，$A_1B\not\subset$ 平面 $B_1CD_1$，所以 $A_1B\parallel$ 平面 $B_1CD_1$。
			同理 $A_1D\parallel$ 平面 $B_1CD_1$。
			\[
				A_1B\parallel B_1CD_1,\quad A_1D\parallel B_1CD_1
			\]
			\par\textbf{理由：}
			线面平行判定定理：若平面外一直线平行于平面内一直线，则线面平行。

		\item[\textbf{第三步（面面平行）：}]
			由于 $A_1B\subset$ 平面 $A_1BD$，$A_1D\subset$ 平面 $A_1BD$，且 $A_1B\cap A_1D=A_1$，$A_1B\parallel$ 平面 $B_1CD_1$，$A_1D\parallel$ 平面 $B_1CD_1$，由面面平行判定定理得
			\[
				A_1BD\parallel B_1CD_1.
			\]
			\par\textbf{理由：}
			面面平行判定定理（本结论）：若一个平面内的两条相交直线都平行于另一个平面，则两平面平行。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面 $A_1BD$ 与平面 $B_1CD_1$ 平行，即 $A_1BD\parallel B_1CD_1$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在四棱锥 $P-ABCD$ 中，底面 $ABCD$ 是正方形，$E,F,G$ 分别是 $PA,PB,PC$ 的中点。证明平面 $EFG\parallel$ 平面 $ABCD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（中位线平行）：}]
			在 $\triangle PAB$ 中，$E,F$ 分别为 $PA,PB$ 中点，所以 $EF\parallel AB$。
			在 $\triangle PBC$ 中，$F,G$ 分别为 $PB,PC$ 中点，所以 $FG\parallel BC$。
			\[
				EF\parallel AB,\quad FG\parallel BC
			\]
			\par\textbf{理由：}
			三角形中位线平行于底边。

		\item[\textbf{第二步（线面平行）：}]
			因为 $EF\parallel AB$，$AB\subset$ 平面 $ABCD$，$EF\not\subset$ 平面 $ABCD$，所以 $EF\parallel$ 平面 $ABCD$。
			同理 $FG\parallel$ 平面 $ABCD$。
			\[
				EF\parallel ABCD,\quad FG\parallel ABCD
			\]
			\par\textbf{理由：}
			线面平行判定定理。

		\item[\textbf{第三步（面面平行）：}]
			由 $EF\subset$ 平面 $EFG$，$FG\subset$ 平面 $EFG$，$EF\cap FG=F$，
			且 $EF\parallel$ 平面 $ABCD$，$FG\parallel$ 平面 $ABCD$，
			根据面面平行判定定理得
			\[
				EFG\parallel ABCD.
			\]
			\par\textbf{理由：}
			面面平行判定定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面 $EFG$ 平行于平面 $ABCD$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三棱柱 $ABC-A_1B_1C_1$ 中，$D,E,F$ 分别是 $AB,AC,A_1B_1$ 的中点。证明平面 $DEF\parallel$ 平面 $BCC_1B_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线线平行）：}]
			在 $\triangle ABC$ 中，$D,E$ 是 $AB,AC$ 中点，所以 $DE\parallel BC$。
			连接 $DF$。在平行四边形 $AA_1B_1B$ 中，$D,F$ 是 $AB,A_1B_1$ 中点，所以 $DF\parallel AA_1$。
			又因为 $AA_1\parallel BB_1$（三棱柱侧棱平行），故 $DF\parallel BB_1$。
			\[
				DE\parallel BC,\quad DF\parallel BB_1
			\]
			\par\textbf{理由：}
			三角形中位线平行于底边；平行四边形中，一组对边中点连线平行于另一组对边；侧棱互相平行。

		\item[\textbf{第二步（线面平行）：}]
			因为 $DE\parallel BC$，$BC\subset$ 平面 $BCC_1B_1$，$DE\not\subset$ 平面 $BCC_1B_1$，所以 $DE\parallel$ 平面 $BCC_1B_1$。
			同理 $DF\parallel$ 平面 $BCC_1B_1$（因为 $DF\parallel BB_1$）。
			\[
				DE\parallel BCC_1B_1,\quad DF\parallel BCC_1B_1
			\]
			\par\textbf{理由：}
			线面平行判定定理。

		\item[\textbf{第三步（面面平行）：}]
			由 $DE\subset$ 平面 $DEF$，$DF\subset$ 平面 $DEF$，$DE\cap DF=D$，
			且 $DE\parallel$ 平面 $BCC_1B_1$，$DF\parallel$ 平面 $BCC_1B_1$，
			根据面面平行判定定理得
			\[
				DEF\parallel BCC_1B_1.
			\]
			\par\textbf{理由：}
			面面平行判定定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面 $DEF$ 平行于平面 $BCC_1B_1$。}
\end{examplebox}
