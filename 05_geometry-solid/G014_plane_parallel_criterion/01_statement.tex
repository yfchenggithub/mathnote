\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（平面内相交线）：}}] 平面 $\alpha$ 内有两条相交直线 $a,\;b$，即 $a \subset \alpha,\; b \subset \alpha,\; a \cap b = P$。
		\item[\textbf{\textcolor{CStatement}{条件2（均平行于 $\beta$）：}}] $a \parallel \beta,\; b \parallel \beta$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面面平行）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两条相交直线同时平行于同一平面 $\beta$，则 $\alpha$ 与 $\beta$ 无公共点。
			\[
				\boxed{\left.\begin{array}{l}
				a \subset \alpha,\; b \subset \alpha\\
				a \cap b = P\\
				a \parallel \beta,\; b \parallel \beta
		\end{array}\right\} \Longrightarrow \alpha \parallel \beta}
		\]

	\item[\textbf{\textcolor{CStatement}{推广结论（线线平行推面面）：}}]
		\textbf{\textcolor{CStatement}{直观描述：}}
		两平面内各有一对相交直线对应平行，则两平面平行。
		\[
			\boxed{\left.\begin{array}{l}
			a \subset \alpha,\; b \subset \alpha,\; a \cap b = P\\
			c \subset \beta,\; d \subset \beta,\; c \cap d = Q\\
			a \not\subset \beta,\; b \not\subset \beta\\
			a \parallel c,\; b \parallel d
	\end{array}\right\} \Longrightarrow \alpha \parallel \beta}
	\]
\end{description}
\end{statementbox}
