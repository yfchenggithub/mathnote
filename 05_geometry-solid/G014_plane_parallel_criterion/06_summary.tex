\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{通过两条相交直线的线面平行推出面面平行。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（两线共面且相交）：} 两条直线 $a,b$ 都在平面 $\alpha$ 内，且 $a\cap b\neq\emptyset$。
		\item \textbf{条件2（均平行于另一平面）：} $a\parallel\beta$ 且 $b\parallel\beta$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（相交条件）：} 若 $a\parallel b$，即使都平行于 $\beta$，也不能推出 $\alpha\parallel\beta$。
		\item \textbf{检查项（直线归属）：} 确认 $a,b$ 确实是 $\alpha$ 内的直线，且 $a,b$ 不在 $\beta$ 内。
	\end{itemize}
\end{summarybox}
