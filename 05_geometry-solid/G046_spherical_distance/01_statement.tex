\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（经纬度表示）：}}]
			球半径 $R$；两点 $A(\varphi_1,\lambda_1)$、$B(\varphi_2,\lambda_2)$，纬度 $\varphi$、经度 $\lambda$ 以弧度计，北纬、东经为正；经度差 $\Delta\lambda = \lambda_2-\lambda_1$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（大圆劣弧长）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			球面上两点沿大圆的最短弧长等于球半径乘以球心角的弧度。
			\[
				l = R\arccos\!\bigl( \sin\varphi_1\sin\varphi_2 + \cos\varphi_1\cos\varphi_2\cos(\Delta\lambda) \bigr).
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（同经线特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当两点在同一经线上时（$\Delta\lambda=0$），球面距离简化为 $l = R\,|\varphi_1-\varphi_2|$。
			\[
				l = R\,|\varphi_1-\varphi_2| \quad (\Delta\lambda=0).
			\]
	\end{description}

\end{statementbox}
