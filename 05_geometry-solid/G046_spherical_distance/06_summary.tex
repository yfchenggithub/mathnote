\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\medskip
	\textbf{$l = R\arccos\bigl(\sin\varphi_1\sin\varphi_2 + \cos\varphi_1\cos\varphi_2\cos\Delta\lambda\bigr)$}.

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（球半径已知）：} 需已知球半径 $R$。
		\item \textbf{条件2（经纬度规范）：} 两点纬度、经度以弧度制表示，北纬东经取正，南纬西经取负。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（弧度检查）：} 确保经纬度已转为弧度，否则公式失效。
		\item \textbf{检查项（经度差范围）：} $\Delta\lambda$ 直接采用 $\lambda_2-\lambda_1$，公式自动处理余弦符号。
	\end{itemize}

\end{summarybox}
