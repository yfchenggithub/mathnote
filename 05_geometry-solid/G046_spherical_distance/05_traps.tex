\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（弧度制混淆）}}]
			直接使用度数代替弧度代入三角函数公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（弧度要求）：}}]
			三角函数参数必须为弧度，角度需先乘以 $\pi/180$ 转换为弧度。
		\item[\textbf{\textcolor{CTrap}{错因分析（度数直接代入）：}}]
			忘记角度与弧度的换算，导致计算出的余弦值错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（经纬度符号）}}]
			未将南纬或西经转换为负弧度，直接将度数代入，导致 $\varphi$ 或 $\lambda$ 的符号错误，从而使公式计算错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（统一符号）：}}]
			北纬、东经取正弧度；南纬、西经取负弧度。确保 $\varphi,\lambda$ 的符号正确，再代入公式。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略方向）：}}]
			习惯将经纬度视为正数，未意识到球面距离公式依赖正负号区分南北、东西方向。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（纬线距离混淆）}}]
			误以为沿纬线的距离是球面最短距离，直接计算纬线弧长。
		\item[\textbf{\textcolor{CTrap}{正确理解（大圆劣弧）：}}]
			球面两点间最短距离是大圆劣弧，只有赤道和经线是大圆，其他纬线不是大圆。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆纬线大圆）：}}]
			忽视大圆定义，认为纬线是圆即是最短路径，实际上纬线曲率半径小于球半径，弧长更大。
	\end{description}

\end{trapbox}
