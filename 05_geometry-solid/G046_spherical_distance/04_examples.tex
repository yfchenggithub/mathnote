\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	已知单位球面上两点 $A(0^\circ,0^\circ)$、$B(0^\circ,60^\circ\mathrm{E})$，求球面距离 $l$。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化弧度）：}]
			将度数化为弧度：$\varphi_1 = 0$，$\varphi_2 = 0$，$\Delta\lambda = \frac{\pi}{3}$。
			\par\textbf{理由：} $60^\circ = \pi/3$ 弧度。
		\item[\textbf{第二步（求球心角余弦）：}]
			代入公式：
			\[
				\begin{aligned}
					\cos\theta &= \sin 0\cdot\sin 0 + \cos 0\cdot\cos 0\cdot\cos\frac{\pi}{3} \\ &= 0 + 1\cdot1\cdot\frac12 \\ &= \frac12.
			\end{aligned}
			\]
			\par\textbf{理由：} 应用球面距离公式。
		\item[\textbf{第三步（求弧长）：}]
			$\theta = \arccos\frac12$，所以 $\theta = \frac{\pi}{3}$。因此 $l = R\theta = \frac{\pi}{3}$（$R=1$）。
			\par\textbf{理由：} 弧长公式 $l=R\theta$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{l = \dfrac{\pi}{3}}$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	已知单位球面上两点 $A(30^\circ\mathrm{N},0^\circ)$、$B(30^\circ\mathrm{N},90^\circ\mathrm{E})$，求球面距离 $l$。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化弧度）：}]
			化 $30^\circ$ 为 $\pi/6$，$\Delta\lambda = \pi/2$。
			\par\textbf{理由：} $90^\circ=\pi/2$。
		\item[\textbf{第二步（代入公式）：}]
			\[
				\begin{aligned}
					\cos\theta &= \sin\frac{\pi}{6}\sin\frac{\pi}{6} + \cos\frac{\pi}{6}\cos\frac{\pi}{6}\cos\frac{\pi}{2} \\ &= \left(\frac12\right)^2 + \left(\frac{\sqrt3}{2}\right)^2\cdot 0 \\ &= \frac14.
			\end{aligned}
			\]
			\par\textbf{理由：} $\sin(\pi/6)=1/2$，$\cos(\pi/6)=\sqrt3/2$。
		\item[\textbf{第三步（求弧长）：}]
			$\theta = \arccos\frac14$，故 $l = \arccos\frac14$。
			\par\textbf{理由：} 弧长公式。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{l = \arccos\dfrac14}$（数值约 $1.318$）。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	在单位球面上，已知 $A(30^\circ\mathrm{N},30^\circ\mathrm{E})$、$B(30^\circ\mathrm{S},150^\circ\mathrm{E})$，求球面距离 $l$。（$\mathrm{S}$ 表示南纬，取负纬度）

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（处理符号与弧度）：}]
			北纬为正，南纬为负：$\varphi_1 = \frac{\pi}{6}$，$\varphi_2 = -\frac{\pi}{6}$；经度差 $\Delta\lambda = 150^\circ-30^\circ$，得 $120^\circ$，即 $\frac{2\pi}{3}$ 弧度。
			\par\textbf{理由：} 注意南纬取负弧度。
		\item[\textbf{第二步（代入公式）：}]
			\[
				\begin{aligned}
					\cos\theta
					&= \sin\frac{\pi}{6}\sin\left(-\frac{\pi}{6}\right) + \cos\frac{\pi}{6}\cos\left(-\frac{\pi}{6}\right)\cos\frac{2\pi}{3}\\
					&= \frac12\cdot\left(-\frac12\right) + \frac{\sqrt3}{2}\cdot\frac{\sqrt3}{2}\cdot\left(-\frac12\right)\\
					&= -\frac14 + \frac34\cdot\left(-\frac12\right)\\
					&= -\frac14 - \frac38\\
					&= -\frac58.
			\end{aligned}
			\]
			\par\textbf{理由：} $\sin(-\alpha)=-\sin\alpha$，$\cos(-\alpha)=\cos\alpha$；$\cos(2\pi/3)=-1/2$。
		\item[\textbf{第三步（求弧长）：}]
			$\theta = \arccos\left(-\frac58\right)$，且 $\theta<\pi$，故 $l = \arccos\left(-\frac58\right)$。
			\par\textbf{理由：} 大圆劣弧对应球心角在 $(0,\pi)$ 内。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{l = \arccos\left(-\dfrac58\right)}$。
\end{examplebox}
