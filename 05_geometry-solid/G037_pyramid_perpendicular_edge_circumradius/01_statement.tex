	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（垂直侧棱）：}}] 棱锥有一条侧棱垂直于底面。
		\item[\textbf{\textcolor{CStatement}{条件 2（有外接圆）：}}] 底面多边形有外接圆。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外接球球心在底面外心的正上方，距离底面高度为侧棱长的一半，半径由底面外接圆半径和半侧棱长组成。
			\[
				R = \sqrt{r^{2} + \left(\frac{h}{2}\right)^{2}}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（正三角形底面）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若底面是正三角形，边长为$a$，则外接圆半径$r = a/\sqrt{3}$，代入即得直接用边长计算的公式。
			\[
				R = \sqrt{\frac{a^{2}}{3} + \left(\frac{h}{2}\right)^{2}}.
			\]
	\end{description}

\end{statementbox}
