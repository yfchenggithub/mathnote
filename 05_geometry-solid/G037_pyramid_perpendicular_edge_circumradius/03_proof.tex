\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	设底面外心为$O'$，球心$O$在过$O'$且垂直于底面的直线上，设$OO'=d$，利用$OA=OS$建立方程，解出$d$后代入求$R$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（设定变量）：}] 记侧棱$SA$垂直于底面于点$A$，$O'$为底面外心，球心$O$在过$O'$且垂直底面的直线上。设$OO' = d$，底面外接圆半径$O'A = r$，垂直侧棱长$SA = h$。
			\[
				\begin{aligned}
					OA^{2} &= O'A^{2} + OO'^{2} \\
					&= r^{2} + d^{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} $O'$是底面外心，$A$在底面圆周上，故$O'A = r$；$O$在$O'$正上方，所以$OA$是直角三角形的斜边。

		\item[\textbf{步骤二（顶点距离）：}] 顶点$S$在$A$正上方，因此球心$O$到$S$的水平距离等于$O'A = r$，竖直距离为$SA - OO' = h-d$。
			\[
				OS^{2} = r^{2} + (h-d)^{2}.
			\]
			\par\textbf{理由：} 由勾股定理，$OS$的平方等于水平距离平方加竖直距离平方。

		\item[\textbf{步骤三（等量关系）：}] 由球心$O$到顶点$S$和底面顶点$A$距离相等（$OA = OS$）得：
			\[
				r^{2} + d^{2} = r^{2} + (h-d)^{2}.
			\]
			\par\textbf{理由：} 外接球球心到所有顶点距离相等。

		\item[\textbf{步骤四（解方程）：}] 上式消去$r^{2}$得$d^{2} = (h-d)^{2}$，两边开方（取正）得$d = h/2$。
			\[
				d = \frac{h}{2}.
			\]
			\par\textbf{理由：} $d$与$h-d$均非负，直接比较得$d = h-d$。

		\item[\textbf{步骤五（求半径）：}] 将$d = h/2$代入$OA^{2}$：
			\[
				R^{2} = r^{2} + \left(\frac{h}{2}\right)^{2}.
			\]
			\[
				\Rightarrow\quad R = \sqrt{r^{2} + \left(\frac{h}{2}\right)^{2}}.
			\]
			\par\textbf{理由：} $R = OA$，且$d$已求。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	至此我们证明了：当棱锥有一条侧棱垂直于底面且底面有外接圆时，外接球半径等于$\sqrt{r^{2} + (h/2)^{2}}$，球心位于底面外心的正上方$h/2$处。该结论将空间问题转化为平面勾股定理，极大简化计算。

\end{proofbox}
