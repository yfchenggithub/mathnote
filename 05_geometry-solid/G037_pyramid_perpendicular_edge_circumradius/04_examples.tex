\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三棱锥$P-ABC$中，$PA\perp$底面$ABC$，底面三角形外接圆半径$r=3$，侧棱$PA=8$，求外接球半径$R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 该棱锥满足有一条侧棱垂直底面且底面有外接圆，$r=3$，$h=8$。
			\par\textbf{理由：} 题目直接给出，符合结论适用条件。

		\item[\textbf{第二步（代入公式）：}] 根据结论 $R = \sqrt{r^{2} + (h/2)^{2}}$，代入数值：
			\[
				\begin{aligned}
					R &= \sqrt{3^{2} + \left(\frac{8}{2}\right)^{2}} \\
					&= \sqrt{9 + 16} \\
					&= \sqrt{25} \\
					&= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 将$r$和$h$代入公式直接计算。

		\item[\textbf{第三步（写出答案）：}] 外接球半径$R = 5$。
			\par\textbf{理由：} 计算得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	所求外接球半径为 $5$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三棱锥$P-ABC$中，$PA\perp$底面$ABC$，底面是正三角形，边长为$6$，$PA=4$，求外接球半径$R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面半径）：}] 正三角形外接圆半径
			\[
				\begin{aligned}
					r &= \frac{a}{\sqrt{3}} \\
					&= \frac{6}{\sqrt{3}} \\
					&= 2\sqrt{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 正三角形外接圆半径公式$r = a/\sqrt{3}$。

		\item[\textbf{第二步（代入公式）：}] 由结论 $R = \sqrt{r^{2} + (h/2)^{2}}$，其中$h=PA=4$：
			\[
				\begin{aligned}
					R &= \sqrt{(2\sqrt{3})^{2} + \left(\frac{4}{2}\right)^{2}} \\
					&= \sqrt{12 + 4} \\
					&= \sqrt{16} \\
					&= 4.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用公式计算。

		\item[\textbf{第三步（写出答案）：}] 外接球半径$R = 4$。
			\par\textbf{理由：} 计算得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	所求外接球半径为 $4$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三棱锥$P-ABC$中，$PA\perp$底面$ABC$，底面$\triangle ABC$中，$AB=AC=2$，$\angle BAC=120^{\circ}$，$PA=4$，求外接球半径$R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面外接圆半径）：}] 在$\triangle ABC$中，由余弦定理得
			\[
				\begin{aligned}
					BC^{2} &= AB^{2}+AC^{2}-2\cdot AB\cdot AC\cdot\cos 120^{\circ} \\
					&= 4+4-2\cdot2\cdot2\cdot\left(-\frac12\right) \\
					&= 12,
			\end{aligned}
			\]
			所以 $BC=2\sqrt{3}$。由正弦定理，外接圆半径
			\[
				\begin{aligned}
					r &= \frac{BC}{2\sin\angle BAC} \\
					&= \frac{2\sqrt{3}}{2\cdot\sin 120^{\circ}} \\
					&= \frac{2\sqrt{3}}{2\cdot\frac{\sqrt{3}}{2}} \\
					&= 2.
			\end{aligned}
			\]
			\par\textbf{理由：} 正弦定理 $a/\sin A = 2r$，代入求$r$。

		\item[\textbf{第二步（代入棱锥公式）：}] 由结论$R = \sqrt{r^{2} + (h/2)^{2}}$，其中$h=PA=4$：
			\[
				\begin{aligned}
					R &= \sqrt{2^{2} + \left(\frac{4}{2}\right)^{2}} \\
					&= \sqrt{4 + 4} \\
					&= \sqrt{8} \\
					&= 2\sqrt{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接代入公式。

		\item[\textbf{第三步（写出答案）：}] 外接球半径$R = 2\sqrt{2}$。
			\par\textbf{理由：} 计算得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	所求外接球半径为 $2\sqrt{2}$。
