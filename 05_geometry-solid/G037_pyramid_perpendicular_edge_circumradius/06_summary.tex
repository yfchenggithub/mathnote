\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{若棱锥有一条侧棱垂直于底面且底面有外接圆，则外接球半径 $R = \sqrt{r^{2} + \left(\frac{h}{2}\right)^{2}}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（垂直侧棱）：} 有一条侧棱垂直于底面。
		\item \textbf{条件 2（有外接圆）：} 底面多边形有外接圆。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件验证）：} 使用前必须确保底面有外接圆，且垂直侧棱明确。
		\item \textbf{检查项（变量区分）：} 确认$h$是垂直侧棱的长，底面顶点都在外接圆上。
	\end{itemize}
\end{summarybox}
