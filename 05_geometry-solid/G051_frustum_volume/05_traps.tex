\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记忆错误）：}}]
		      常见错误是写成 $V=\dfrac{h}{2}(S_1+S_2)$ 或 $V=\dfrac{h}{3}(S_1+S_2)$，漏掉 $\sqrt{S_1S_2}$。

		\item[\textbf{\textcolor{CTrap}{正确理解（记清完整公式）：}}]
		      台体体积公式为 $V=\dfrac{h}{3}(S_1+S_2+\sqrt{S_1S_2})$，不能删除几何平均项。

		\item[\textbf{\textcolor{CTrap}{错因分析（与梯形混淆）：}}]
		      平面梯形面积公式是 $(a+b)h/2$，学生容易类比到台体时遗漏几何平均项，并把系数 $3$ 误写为 $2$。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（面积与半径混淆）：}}]
		      题目给出半径时，直接把 $r_1,r_2$ 当作 $S_1,S_2$ 代入。

		\item[\textbf{\textcolor{CTrap}{正确理解（先求面积）：}}]
		      圆台问题必须先计算 $S_1=\pi r_1^2,\\ S_2=\pi r_2^2$，再代入台体体积公式。

		\item[\textbf{\textcolor{CTrap}{错因分析（忽略量纲）：}}]
		      公式中的 $S_1,S_2$ 是面积，不是长度；若混用长度，量纲不对，结果必然错误。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略适用条件）：}}]
		      对上下底面不平行或不相似的几何体也直接使用此公式。

		\item[\textbf{\textcolor{CTrap}{正确理解（验证条件）：}}]
		      公式只适用于棱台、圆台以及锥体或柱体的相应退化情形，必须保证上下底面平行且相似。

		\item[\textbf{\textcolor{CTrap}{错因分析（条件不明）：}}]
		      没有检查几何体是否满足“平行且相似”的性质，误把一般拟柱体的体积问题当作台体问题。
	\end{description}

\end{trapbox}
