	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记忆错误）}}]
			常见的错误是写成 $V = \dfrac{h}{2}(S_1+S_2)$ 或 $V = \dfrac{h}{3}(S_1+S_2)$（漏掉 $\sqrt{S_1S_2}$）。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（记清完整公式）：}}]
			台体体积公式为 $V = \dfrac{h}{3}(S_1+S_2+\sqrt{S_1S_2})$，不能删除几何平均项。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（与梯形混淆）：}}]
			平面梯形面积公式是 $(a+b)h/2$，学生容易类比到台体时遗漏几何平均项并把系数 $3$ 误写为 $2$。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（面积与半径混淆）}}]
			题目给出半径时直接代入 $r_1,r_2$ 而不是计算面积。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（先求面积）：}}]
			必须先计算 $S_1 = \pi r_1^2,\ S_2 = \pi r_2^2$（圆台）或直接用已知面积值，再代入公式。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略量纲）：}}]
			公式中的 $S_1,S_2$ 是面积，若混用长度则结果错误。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略适用条件）}}]
			对上下底面不平行或不相似的几何体（例如一般的拟柱体）也使用此公式。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（验证条件）：}}]
			公式只适用于棱台或圆台，必须保证上下底面平行且相似。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（条件不明）：}}]
			没有检查几何体是否满足“平行且相似”的性质，误将一般拟柱体适用公式当作台体公式。\n
	\end{description}
