\begin{summarybox}
	\n\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{台体体积等于高乘以上底面积、下底面积与两者几何平均数之和的三分之一。}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（上下底面平行）：} 几何体必须是棱台或圆台，上下底面平行且相似。\n
		\item \textbf{条件 2（面积与高已知）：} 需要知道上底面积 $S_1$、下底面积 $S_2$ 和高 $h$（或能间接求出）。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（公式混淆）：} 不要与梯形面积公式混淆，台体体积公式有三项，系数为 $3$。\n
		\item \textbf{检查项（单位一致）：} 确保 $S_1$ 和 $S_2$ 使用相同的面积单位，长度单位统一。\n
	\end{itemize}
	\n\n
\end{summarybox}
