\begin{statementbox}
	\n\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件 1（上下底面平行）：}}]
			上底面与下底面平行且相似，面积分别为 $S_1, S_2$，高为 $h$。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（台体体积公式）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			体积等于高乘以上底加下底再加几何平均，除以三。\n
			\[
				\n V = \frac{h}{3}\bigl(S_1 + S_2 + \sqrt{S_1S_2}\bigr).\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（柱锥退化情形）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			当上底等于下底时变为柱体，当上底为零时变为锥体。\n
			\[
				\n S_1 = S_2 \Rightarrow V = S_1 h,
				\qquad
				S_1 = 0 \Rightarrow V = \frac13 S_2 h.\n
			\]
			\n
	\end{description}
	\n\n
\end{statementbox}
