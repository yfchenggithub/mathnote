\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（上下底面平行且相似）：}}]
		      台体的上底面与下底面平行且相似，面积分别为 $S_1,S_2$，高为 $h$，其中 $h>0$，$S_1,S_2\ge 0$，且 $S_1,S_2$ 不同时为 $0$。
	\end{description}

	\medskip

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（台体体积公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      体积等于高乘以上底面积、下底面积以及两者几何平均数之和，再除以三。
		      \[
			      V = \frac{h}{3}\bigl(S_1 + S_2 + \sqrt{S_1S_2}\bigr).
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（柱锥退化情形）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      当上底面积等于下底面积时变为柱体体积公式；当一个底面积为 $0$ 时变为锥体体积公式。
		      \[
			      S_1 = S_2 \Rightarrow V = S_1h,
			      \qquad
			      S_1 = 0 \Rightarrow V = \frac13 S_2h.
		      \]
	\end{description}

\end{statementbox}
