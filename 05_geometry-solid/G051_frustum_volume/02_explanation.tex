\begin{explanationbox}
	\n\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n台体体积公式可看作是梯形面积公式向三维空间的推广：梯形的面积是 $\frac{a+b}{2}h$，台体体积是 $\frac{S_1+S_2+\sqrt{S_1S_2}}{3}h$。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（三部分求和）：}]
			公式由三部分组成：上底面积 $S_1$、下底面积 $S_2$ 和几何平均数 $\sqrt{S_1S_2}$。\n
		\item[\textbf{要点二（几何平均意义）：}]
			$\sqrt{S_1S_2}$ 反映上下底面积的中间缩放水平，当两底越接近时，该项越接近 $S_1$ 或 $S_2$。\n
		\item[\textbf{要点三（祖暅原理）：}]
			该公式可通过祖暅原理推导，台体截面面积随高度二次变化，积分即得。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（大锥减小锥）}}\n\n将台体补成一个大圆锥（或棱锥），在上底面处截去一个小锥。设大锥高 $H$，小锥高 $h'$，台高 $h = H - h'$，由相似比 $\frac{S_1}{S_2} = \left(\frac{h'}{H}\right)^2$ 可推导出公式。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（对称结构）}}\n\n公式中 $S_1$ 与 $S_2$ 地位对称，互换 $S_1$ 和 $S_2$ 公式不变，与台体放置方向无关。\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（考法分析）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（直接求体积）：} 已知上底面积、下底面积和高，直接代入公式计算体积。\n
		\item \textbf{考法二（逆向求参数）：} 已知体积、高和其中一个底面积，或已知体积和面积关系，列方程求另一底面积。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n\textbf{记公式时类比梯形面积：梯形面积是 $\frac{a+b}{2}h$，台体体积是 $\frac{S_1+S_2+\sqrt{S_1S_2}}{3}h$，系数从 $2$ 变 $3$，多了一项几何平均。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（直接计算）：} 在立体几何题中直接给出棱台或圆台的上下底面积和高。\n
		\item \textbf{场景二（组合体分解）：} 将一个复杂几何体分割成柱、锥、台等基本部分，分别用相应体积公式计算再求和。\n
	\end{itemize}
	\n\n
\end{explanationbox}
