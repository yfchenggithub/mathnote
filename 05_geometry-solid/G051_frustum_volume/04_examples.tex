\begin{examplebox}
	\n\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知圆台的上底面半径 $r_1 = 2$，下底面半径 $r_2 = 4$，高 $h = 6$，求该圆台的体积。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求上下底面积）：}]
			\n $S_1 = \pi r_1^2 = \pi \cdot 2^2 = 4\pi$，$S_2 = \pi r_2^2 = \pi \cdot 4^2 = 16\pi$。\n
			\par\textbf{理由：} 圆面积公式 $S = \pi r^2$。\n\n
		\item[\textbf{第二步（代入公式化简）：}]
			\n
			\[
				\n
				\begin{aligned}
					\n V &= \frac{h}{3}(S_1 + S_2 + \sqrt{S_1 S_2}) \\\n &= \frac{6}{3}(4\pi + 16\pi + \sqrt{4\pi \cdot 16\pi}) \\\n &= 2(20\pi + \sqrt{64\pi^2}) \\\n &= 2(20\pi + 8\pi).\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 台体体积公式；注意 $\sqrt{4\pi \cdot 16\pi} = \sqrt{64\pi^2} = 8\pi$。\n\n
		\item[\textbf{第三步（得体积）：}]
			\n $V = 2 \cdot 28\pi = 56\pi$。\n
			\par\textbf{理由：} 合并同类项。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 该圆台的体积为 $56\pi$。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知圆台的体积为 $19\pi$，高为 $3$，且上下底面积之和为 $13\pi$，求圆台的上底半径和下底半径。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（设未知数并建立方程）：}]
			\n 设上底面积为 $x\pi$，下底面积为 $y\pi$（$x>0,y>0$），则 $x+y=13$。由体积公式\n
			\[
				\n
				\begin{aligned}
					\n 19\pi &= \frac{3}{3}\bigl(x\pi + y\pi + \pi\sqrt{xy}\bigr) \\\n &= \pi(x+y+\sqrt{xy}).\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 代入 $h=3$，$S_1=x\pi$，$S_2=y\pi$，化简。\n\n
		\item[\textbf{第二步（求 $\sqrt{xy}$）：}]
			\n 两边除以 $\pi$ 得 $19 = x+y+\sqrt{xy}$，再代入 $x+y=13$ 得\n
			\[
				\n 19 = 13 + \sqrt{xy} \quad\Rightarrow\quad \sqrt{xy} = 6.\n
			\]
			\n 平方得 $xy = 36$。\n
			\par\textbf{理由：} 移项后平方。\n\n
		\item[\textbf{第三步（求半径）：}]
			\n 解方程组 $
			\begin{cases}
				x+y=13 \\ xy=36
		\end{cases}
			$，得 $(x,y)=(4,9)$ 或 $(9,4)$。取 $x=4,y=9$（上底较小），故\n
			\[
				\n S_1 = 4\pi,\ r_1 = \sqrt{4}=2,\quad S_2 = 9\pi,\ r_2 = \sqrt{9}=3.\n
			\]
			\n
			\par\textbf{理由：} 韦达定理，方程 $t^2-13t+36=0$ 的解为 $4$ 和 $9$。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 上底半径为 $2$，下底半径为 $3$，上底面积为 $4\pi$，下底面积为 $9\pi$。\n\n

	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知圆锥的底面半径为 $6$，高为 $8$。现用平行于底面的平面在距底面 $3$ 高处截去一个小圆锥，求剩余圆台的体积。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求小锥底面半径）：}]
			\n 截去的小圆锥高为 $h_1 = 8-3 = 5$，大圆锥高 $H=8$。由相似性，小锥底面半径 $r_1$ 满足 $\displaystyle\frac{r_1}{6} = \frac{5}{8}$，故 $r_1 = 6 \times \frac{5}{8} = \frac{15}{4} = 3.75$。\n
			\par\textbf{理由：} 平行于底面的截面与底面相似，半径比等于高比。\n\n
		\item[\textbf{第二步（求上下底面积）：}]
			\n 圆台上底面积 $S_1 = \pi r_1^2 = \pi\left(\frac{15}{4}\right)^2 = \frac{225\pi}{16}$，下底面积 $S_2 = \pi \cdot 6^2 = 36\pi$，台高 $h=3$。\n
			\par\textbf{理由：} 圆面积公式，截面即为圆台上底面。\n\n
		\item[\textbf{第三步（代入台体体积公式）：}]
			\n
			\[
				\n
				\begin{aligned}
					\n V
					&= \frac{h}{3}(S_1 + S_2 + \sqrt{S_1S_2})\\
					\n
					&= \frac{3}{3}\left( \frac{225\pi}{16} + 36\pi + \sqrt{ \frac{225\pi}{16} \cdot 36\pi } \right)\\
					\n
					&= \frac{225\pi}{16} + 36\pi + \sqrt{ \frac{225 \times 36\pi^2}{16} }\\
					\n
					&= \frac{225\pi}{16} + 36\pi + \frac{15 \times 6\pi}{4}\\
					\n
					&= \frac{225\pi}{16} + 36\pi + \frac{90\pi}{4}\\
					\n
					&= \frac{225\pi}{16} + 36\pi + \frac{360\pi}{16}\\
					\n
					&= \frac{225\pi+360\pi}{16} + 36\pi\\
					\n
					&= \frac{585\pi}{16} + 36\pi\\
					\n
					&= \frac{585\pi}{16} + \frac{576\pi}{16}\\
					\n
					&= \frac{1161\pi}{16}.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：} 台体体积公式，分数运算，$\sqrt{225\times36}=15\times6$，$\sqrt{16}=4$，将各项化为同分母相加。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 剩余圆台的体积为 $\displaystyle \frac{1161\pi}{16}$。\n\n
\end{examplebox}
