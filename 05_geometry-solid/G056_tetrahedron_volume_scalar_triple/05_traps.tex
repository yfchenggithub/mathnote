\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（起点须统一）：}}] 若以不同顶点为起点取向量（如用 $\overrightarrow{BA},\overrightarrow{CA},\overrightarrow{DA}$），直接代入公式将得到错误结果。
		\item[\textbf{\textcolor{CTrap}{正确理解（同一起点要求）：}}] 三条向量必须从同一顶点出发，通常选择 $A$ 作为起点。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导基础）：}}] 公式的推导基于共起点的向量，否则底面和高关系不成立。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（缺失绝对值）：}}] 直接写 $V=\frac16(\vec{a}\times\vec{b})\cdot\vec{c}$ 可能导致负值，而体积不能为负。
		\item[\textbf{\textcolor{CTrap}{正确理解（取绝对值运算）：}}] 必须使用 $V=\frac16\bigl|(\vec{a}\times\vec{b})\cdot\vec{c}\bigr|$，取绝对值。
		\item[\textbf{\textcolor{CTrap}{错因分析（物理意义）：}}] 混合积的正负取决于向量方向，体积应为正值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（系数记错）：}}] 容易记成 $V=\frac13|(\vec{a}\times\vec{b})\cdot\vec{c}|$ 或 $V=|(\vec{a}\times\vec{b})\cdot\vec{c}|$。
		\item[\textbf{\textcolor{CTrap}{正确理解（六分之一系数）：}}] 正确公式为 $V=\frac16|(\vec{a}\times\vec{b})\cdot\vec{c}|$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}] 三角形面积公式中有 $1/2$，对应到空间常忽略 $1/6$ 的来源。
	\end{description}
\end{trapbox}
