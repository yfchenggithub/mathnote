\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{四面体体积等于从同一顶点出发的三条棱向量混合积绝对值的六分之一。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（共起点向量）：} 已知从同一顶点出发的三条棱的向量或坐标。
		\item \textbf{条件 2（非共面保证）：} 确保四点不共面（否则体积为零）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（必取绝对值）：} 公式中必须有绝对值，防止体积为负。
		\item \textbf{检查项（混合积为零）：} 若混合积为零，则四点共面，体积为零。
	\end{itemize}
\end{summarybox}
