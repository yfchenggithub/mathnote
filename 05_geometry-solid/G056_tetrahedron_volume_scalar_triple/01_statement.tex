	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（起始顶点向量）：}}] 设四面体 $ABCD$，以顶点 $A$ 为起点，记 $\vec{a}=\overrightarrow{AB}$, $\vec{b}=\overrightarrow{AC}$, $\vec{c}=\overrightarrow{AD}$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（体积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			四面体体积等于以 $AB,AC,AD$ 为邻边的平行六面体体积的六分之一。
			\[
				V = \frac{1}{6}\bigl|(\vec{a}\times\vec{b})\cdot\vec{c}\bigr|.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（共面判据）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当混合积为零时，$A,B,C,D$ 四点共面。
			\[
				(\vec{a}\times\vec{b})\cdot\vec{c}=0.
			\]
	\end{description}

\end{statementbox}
