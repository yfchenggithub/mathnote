\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l$ 与平面 $\alpha$ 交于点 $P$，$Q$ 是 $l$ 上一点，$Q$ 到平面 $\alpha$ 的距离 $QH=4$，$PQ=5$，求直线 $l$ 与平面 $\alpha$ 所成角 $\theta$ 的正弦值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作垂线）：}] 过 $Q$ 作 $QH \perp \alpha$ 于 $H$，连接 $PH$，得直角三角形 $QPH$。
			\par\textbf{理由：} $QH \perp \alpha$，$QH$ 即点到面的距离。
		\item[\textbf{第二步（确定线面角）：}] 由定义，$\angle QPH$ 即为线面角 $\theta$。
			\par\textbf{理由：} $PH$ 是 $l$ 在 $\alpha$ 内的射影。
		\item[\textbf{第三步（应用公式）：}] 在 $\mathrm{Rt}\triangle QPH$ 中，
			\[
				\begin{aligned}
					\sin\theta &= \frac{QH}{PQ} \\
					&= \frac{4}{5}.
			\end{aligned}
			\]
			\par\textbf{理由：} 锐角正弦等于对边比斜边。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin\theta = \dfrac{4}{5}$}。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在长方体 $ABCD-A_1B_1C_1D_1$ 中，$AB=3$，$AD=4$，$AA_1=5$，求体对角线 $A_1C$ 与底面 $ABCD$ 所成角 $\theta$ 的正弦值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作垂线）：}] 由长方体性质，$A_1A \perp$ 底面 $ABCD$，垂足为 $A$，所以 $A_1$ 到底面距离为 $AA_1=5$。
			\par\textbf{理由：} 长方体侧棱垂直于底面。
		\item[\textbf{第二步（确定射影）：}] $A_1C$ 在底面 $ABCD$ 内的射影为 $AC$，故 $\angle A_1CA$ 就是线面角 $\theta$。
			\par\textbf{理由：} 过垂足 $A$ 与 $C$ 的连线 $AC$ 为射影。
		\item[\textbf{第三步（计算正弦）：}]
			计算相关长度：
			\[
				\begin{aligned}
					AC &= \sqrt{AB^2+AD^2}=5,\\
					A_1C &= \sqrt{AA_1^2+AC^2}=5\sqrt{2}.
			\end{aligned}
			\]
			在 $\mathrm{Rt}\triangle A_1AC$ 中，
			\[
				\begin{aligned}
					\sin\theta &= \frac{AA_1}{A_1C} \\
					&= \frac{5}{5\sqrt{2}} \\
					&= \frac{\sqrt{2}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 在 $\mathrm{Rt}\triangle A_1AC$ 中，$\angle A_1CA$ 的对边是 $A_1A$，斜边是 $A_1C$，由正弦定义得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin\theta = \dfrac{\sqrt{2}}{2}$}。

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三棱锥 $P-ABC$ 中，$PA \perp$ 底面 $ABC$，$AB \perp BC$，$AB=3$，$BC=4$，$PA=5$，求 $PC$ 与底面 $ABC$ 所成角 $\theta$ 的正弦值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作垂线）：}] 由 $PA \perp$ 底面 $ABC$，得 $P$ 到底面距离为 $PA=5$，垂足为 $A$。
			\par\textbf{理由：} $PA$ 垂直于底面，$A$ 为垂足。
		\item[\textbf{第二步（确定射影）：}] $PC$ 在底面 $ABC$ 内的射影为 $AC$，故 $\angle PCA$ 为线面角 $\theta$。
			\par\textbf{理由：} 过垂足 $A$ 与 $C$ 的连线 $AC$ 为射影。
		\item[\textbf{第三步（计算正弦）：}]
			计算相关长度：
			\[
				\begin{aligned}
					AC &= \sqrt{AB^2+BC^2}=5,\\
					PC &= \sqrt{PA^2+AC^2}=5\sqrt{2}.
			\end{aligned}
			\]
			在 $\mathrm{Rt}\triangle PAC$ 中，
			\[
				\begin{aligned}
					\sin\theta &= \frac{PA}{PC} \\
					&= \frac{5}{5\sqrt{2}} \\
					&= \frac{\sqrt{2}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 在 $\mathrm{Rt}\triangle PAC$ 中，$\angle PCA$ 的对边是 $PA$，斜边是 $PC$，由正弦定义得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin\theta = \dfrac{\sqrt{2}}{2}$}。
\end{examplebox}
