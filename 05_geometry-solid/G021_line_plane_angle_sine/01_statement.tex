\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（线面相交）：}}] 直线 $l$ 与平面 $\alpha$ 相交于点 $P$。$Q$ 为 $l$ 上异于 $P$ 的任意一点，$QH\perp\alpha$ 于 $H$，连接 $PH$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正弦公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			线面角的正弦等于垂线段长与斜线段长的比值。
			\[
				\sin\theta = \frac{QH}{PQ},
				\qquad
				0^\circ \le \theta \le 90^\circ .
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（向量公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			设直线 $l$ 的方向向量为 $\vec{d}$，平面 $\alpha$ 的法向量为 $\vec{n}$，则
			\[
				\sin\theta = \frac{|\vec{d} \cdot \vec{n}|}{|\vec{d}|\,|\vec{n}|}.
			\]
	\end{description}
\end{statementbox}
