\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忘记取绝对值）}}]
			直接使用$\cos\theta=\frac{m^2+n^2-d^2}{2mn}$，当分子为负时得到负值，但异面直线所成角$\theta\in(0^\circ,90^\circ]$，余弦应为非负。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须取绝对值）：}}]
			公式应为$\cos\theta=\frac{|m^2+n^2-d^2|}{2mn}$，保证余弦非负。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视角度范围）：}}]
			未注意异面直线所成角定义为锐角或直角，余弦非负；若平移后角为钝角，其余弦为负，应取其绝对值对应锐角。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（对应边混淆）}}]
			错误地将$m,n,d$与三角形三边乱匹配，例如将$d$取为连接两交点的线段，或取错了边。
		\item[\textbf{\textcolor{CTrap}{正确理解（明确符号定义）：}}]
			$m,n$为平移后两交线段的长度，$d$为连接两非交端点的线段长度（三角形中与$m,n$所夹边相对的一边）。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号混淆）：}}]
			对结论中每一线段具体对应哪一条边理解不清，导致代入错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（三角形不可解）}}]
			平移后三角形三边未知或无法求出，仍然尝试套用公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（确保三角形可解）：}}]
			公式使用的前提是平移后三角形可解（三边长度已知或可求）。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件缺失）：}}]
			忽略条件“三角形可解”，在几何图形中未正确计算三边就代入。
	\end{description}

\end{trapbox}
