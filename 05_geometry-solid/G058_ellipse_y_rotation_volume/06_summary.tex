\begin{summarybox}

\textbf{\textcolor{CSummary}{一句话核心}}

\textbf{椭圆绕 $y$ 轴旋转形成的旋转椭球体积公式为 $V_y = \frac{4}{3}\pi a^2 b$。}

\medskip
\textbf{\textcolor{CSummary}{使用条件}}
\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
    \item \textbf{条件1（椭圆标准形式）：} 椭圆方程为 $\frac{x^2}{a^2}+\frac{y^2}{b^2}=1$，且有 $a>0,\ b>0$。
    \item \textbf{条件2（绕 $y$ 轴旋转）：} 旋转轴为 $y$ 轴，旋转一周。
\end{itemize}

\medskip
\textbf{\textcolor{CSummary}{关键提醒}}
\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
    \item \textbf{易错点（区分旋转轴）：} 绕 $y$ 轴用 $V_y = \frac{4}{3}\pi a^2 b$，绕 $x$ 轴时用 $V_x = \frac{4}{3}\pi a b^2$。
    \item \textbf{检查项（半轴与坐标对应）：} 注意 $a$ 是 $x$ 轴半轴长，$b$ 是 $y$ 轴半轴长，不要混用。
\end{itemize}


\medskip
\textbf{\textcolor{CSummary}{三种证明一眼复原}}
\begin{itemize}[leftmargin=*, itemsep=0.45em, topsep=0.4em]
    \item \textbf{沿 $y$ 轴切圆：} $A(y)=\pi a^2\left(1-\frac{y^2}{b^2}\right)$，从 $-b$ 积到 $b$。
    \item \textbf{沿 $x$ 轴切椭圆：} $A(x)=\pi ab\left(1-\frac{x^2}{a^2}\right)$，从 $-a$ 积到 $a$。
    \item \textbf{球体伸缩（最快）：} 半径 $b$ 的球沿 $x,z$ 同时乘 $a/b$，故 $V_y=\frac43\pi b^3\left(\frac ab\right)^2$。
\end{itemize}
\[
   \boxed{V_y=\frac43\pi a^2b}
   \qquad(\text{三法结果一致}).
\]

\end{summarybox}
