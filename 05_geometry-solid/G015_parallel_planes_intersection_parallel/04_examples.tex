\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知平面$\alpha\parallel\beta$，平面$\gamma$与$\alpha$、$\beta$分别交于直线$a,b$。求证$a\parallel b$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（共面确认）：}] 由于$\gamma\cap\alpha=a$，$\gamma\cap\beta=b$，所以$a\subset\gamma$，$b\subset\gamma$，即$a$与$b$共面。
			\par\textbf{理由：} 两平面相交，交线在所交平面内。

		\item[\textbf{第二步（无公共点证明）：}] 因为$\alpha\parallel\beta$，所以$\alpha$与$\beta$无公共点。如果$a$与$b$相交于点$P$，则$P$既在$\alpha$又在$\beta$，矛盾。故$a$与$b$无公共点。
			\par\textbf{理由：} 面面平行定义。

		\item[\textbf{第三步（平行结论）：}] 在平面$\gamma$中，两条直线$a,b$无公共点，因此$a\parallel b$。
			\par\textbf{理由：} 平面内不相交则平行。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a\parallel b$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知平面$\alpha\parallel\beta$，点$P$在$\alpha$外，过$P$的两条直线分别交$\alpha$于$A,C$，交$\beta$于$B,D$，且$A,B,C,D$四点共面。求证$AC\parallel BD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定交线）：}] 因为$A,C\in\alpha$，$B,D\in\beta$，且$A,B,C,D$共面于$\gamma$，所以$\gamma\cap\alpha=AC$，$\gamma\cap\beta=BD$。
			\par\textbf{理由：} $A,C$是$\alpha,\gamma$的公共点，所以交线是直线$AC$；同理交线是$BD$。

		\item[\textbf{第二步（应用性质定理）：}] 由于$\alpha\parallel\beta$，$\gamma$与$\alpha,\beta$分别交于$AC,BD$，根据面面平行性质定理得$AC\parallel BD$。
			\par\textbf{理由：} 面面平行性质定理。

		\item[\textbf{第三步（得出结论）：}] 因此$AC\parallel BD$成立。
			\par\textbf{理由：} 推导已完成。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AC\parallel BD$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知平面$\alpha\parallel\beta$，平面$\gamma\parallel\delta$，且$\gamma\cap\alpha=a$，$\gamma\cap\beta=b$，$\delta\cap\alpha=c$，$\delta\cap\beta=d$。求证$a\parallel c$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线线平行推导）：}] 因为$\alpha\parallel\beta$，$\gamma$与$\alpha,\beta$交于$a,b$，所以$a\parallel b$；同理，$\delta$与$\alpha,\beta$交于$c,d$，得$c\parallel d$。
			\par\textbf{理由：} 面面平行性质定理（两次）。

		\item[\textbf{第二步（再次应用定理）：}] 因为$\gamma\parallel\delta$，且$\alpha$与$\gamma,\delta$分别交于$a,c$，所以$a\parallel c$。
			\par\textbf{理由：} 面面平行性质定理，视$\alpha$为第三平面。

		\item[\textbf{第三步（得出结论）：}] 因此$a\parallel c$。
			\par\textbf{理由：} 综合步骤一、二。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a\parallel c$}
