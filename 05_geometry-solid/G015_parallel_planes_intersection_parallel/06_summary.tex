\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{两个平行平面与第三个平面相交，所得的两条交线互相平行。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（两平面平行）：} 必须存在两个平面 $\alpha$ 和 $\beta$ 满足 $\alpha\parallel\beta$。
		\item \textbf{条件 2（第三平面截）：} 必须有第三个平面 $\gamma$ 同时与 $\alpha$、$\beta$ 相交。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略平行前提）：} 使用定理前务必确认两个平面平行，否则结论不成立。
		\item \textbf{检查项（识别对应交线）：} 找出 $\gamma$ 与 $\alpha$、$\beta$ 的交线，这两条线即为所证。
	\end{itemize}
\end{summarybox}
