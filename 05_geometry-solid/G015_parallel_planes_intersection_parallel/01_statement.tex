	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（交线互相平行）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个平行平面被第三个平面所截，所得的两条交线方向相同。
			\[
				a\parallel b.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（共面线线平行）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个平行平面内的直线，若共面，则必平行。
			\[
				a\parallel b.
			\]
	\end{description}
