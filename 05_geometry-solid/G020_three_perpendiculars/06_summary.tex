\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{面内直线垂直射影等价于垂直斜线，关键是利用垂线$PO\perp\alpha$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（垂线存在）：} $PO\perp\alpha$已知，提供基础垂直关系。
		\item \textbf{条件2（直线在面内）：} $l\subset\alpha$，确保定理适用。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（射影正确）：} 射影是$OA$（垂足与斜足连线），不能弄错。
		\item \textbf{检查项（条件核对）：} 使用前确认$l\subset\alpha$且$PO\perp\alpha$。
	\end{itemize}
\end{summarybox}
