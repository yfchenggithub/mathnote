\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	利用线面垂直的判定与性质，将面内直线与射影的垂直关系转化为与斜线的垂直关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（线面垂直基础）：}]
			\[
				PO\perp\alpha,\; l\subset\alpha \;\Rightarrow\; PO\perp l.
			\]
			\par\textbf{理由：}线面垂直定义：若一条直线垂直于平面，则它垂直于平面内所有直线。故 $PO\perp l$ 恒成立。

		\item[\textbf{步骤二（正向证明）：}]
			若 $l\perp OA$，结合步骤一的 $PO\perp l$ 以及 $PO\cap OA=O$，则有
			\[
				l\perp OA,\; PO\perp l,\; PO\cap OA=O \;\Rightarrow\; l\perp PA.
			\]
			\par\textbf{理由：} $l$ 垂直于平面 $POA$ 内的两条相交直线 $PO$ 和 $OA$，由线面垂直判定定理知 $l\perp$ 平面 $POA$，再由线面垂直性质（定义）得 $l\perp PA$。

		\item[\textbf{步骤三（逆向证明）：}]
			若 $l\perp PA$，同样结合 $PO\perp l$ 以及 $PO\cap PA=P$，可得
			\[
				l\perp PA,\; PO\perp l,\; PO\cap PA=P \;\Rightarrow\; l\perp OA.
			\]
			\par\textbf{理由：} $l$ 垂直于平面 $POA$ 内的两条相交直线 $PO$ 和 $PA$，故 $l\perp$ 平面 $POA$，从而 $l\perp OA$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{步骤二和步骤三分别证明了 $l\perp OA\Rightarrow l\perp PA$ 和 $l\perp PA\Rightarrow l\perp OA$，因此 $l\perp OA\iff l\perp PA$ 成立。}
\end{proofbox}
