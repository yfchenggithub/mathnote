	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（线面垂直）：}}] $PO\perp\alpha$ 于 $O$。
		\item[\textbf{\textcolor{CStatement}{条件2（斜线射影）：}}] $PA$ 交 $\alpha$ 于 $A$（$A\neq O$），$OA$ 为 $PA$ 在 $\alpha$ 内的射影。
		\item[\textbf{\textcolor{CStatement}{条件3（面内直线）：}}] $l\subset\alpha$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（充要条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面内直线垂直射影当且仅当垂直斜线。
			\[
				l\perp OA \iff l\perp PA
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（三垂线定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面内直线垂直射影则必垂直斜线。
			\[
				l\perp OA \Rightarrow l\perp PA
			\]
	\end{description}

\end{statementbox}
