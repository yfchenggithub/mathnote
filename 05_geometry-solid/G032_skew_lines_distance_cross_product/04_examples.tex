\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两条异面直线$l_1$过点$A(1,0,0)$，方向向量$\vec{u}=(1,1,0)$；$l_2$过点$B(0,2,1)$，方向向量$\vec{v}=(0,1,1)$。求$l_1$与$l_2$的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求连接向量）：}]
			$\overrightarrow{AB}=(-1,2,1)$。
			\par\textbf{理由：}向量减法。
		\item[\textbf{第二步（求叉积）：}]
			\[
				\begin{aligned}
					\vec{n} &= \vec{u}\times\vec{v} \\ &= (1\cdot1-0\cdot1,\,0\cdot0-1\cdot1,\,1\cdot1-1\cdot0) \\ &= (1,-1,1).
			\end{aligned}
			\]
			\par\textbf{理由：}叉积计算公式。
		\item[\textbf{第三步（代公式计算）：}]
			\[
				\begin{aligned}
					|\vec{n}| &= \sqrt{1^2+(-1)^2+1^2} \\ &= \sqrt{3},
			\end{aligned}
			\]
			\[
				\begin{aligned}
					\overrightarrow{AB}\cdot\vec{n} &= (-1)\cdot1+2\cdot(-1)+1\cdot1 \\ &= -2.
			\end{aligned}
			\]
			故
			\[
				\begin{aligned}
					d &= \frac{|-2|}{\sqrt{3}} \\ &= \frac{2}{\sqrt{3}} \\ &= \frac{2\sqrt{3}}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}代入异面直线距离公式$d=\dfrac{|\overrightarrow{AB}\cdot(\vec{u}\times\vec{v})|}{|\vec{u}\times\vec{v}|}$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d=\dfrac{2\sqrt{3}}{3}$}。

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线$l_1$的参数方程为$
	\begin{cases}
		x=2+t\\ y=1-t\\ z=3t
	\end{cases}
	$（$t$为参数），直线$l_2$过点$B(4,2,1)$且方向向量$\vec{v}=(1,1,2)$。求$l_1$与$l_2$的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取点与方向）：}]
			从$l_1$参数方程得点$A(2,1,0)$，方向向量$\vec{u}=(1,-1,3)$。连接向量$\overrightarrow{AB}=(2,1,1)$。
			\par\textbf{理由：}参数方程中$x,y,z$关于$t$的系数为方向向量，常数项为定点。
		\item[\textbf{第二步（求叉积与模）：}]
			\[
				\begin{aligned}
					\vec{n} &= \vec{u}\times\vec{v} \\ &= ((-1)\cdot2-3\cdot1,\,3\cdot1-1\cdot2,\,1\cdot1-(-1)\cdot1) \\ &= (-5,1,2),
			\end{aligned}
			\]
			\[
				\begin{aligned}
					|\vec{n}| &= \sqrt{(-5)^2+1^2+2^2} \\ &= \sqrt{30}.
			\end{aligned}
			\]
			\par\textbf{理由：}叉积行列式运算。
		\item[\textbf{第三步（代公式计算）：}]
			\[
				\begin{aligned}
					\overrightarrow{AB}\cdot\vec{n} &= 2\cdot(-5)+1\cdot1+1\cdot2 \\ &= -10+1+2 \\ &= -7,
			\end{aligned}
			\]
			故
			\[
				\begin{aligned}
					d &= \frac{|-7|}{\sqrt{30}} \\ &= \frac{7}{\sqrt{30}} \\ &= \frac{7\sqrt{30}}{30}.
			\end{aligned}
			\]
			\par\textbf{理由：}代入异面直线距离公式$d=\dfrac{|\overrightarrow{AB}\cdot(\vec{u}\times\vec{v})|}{|\vec{u}\times\vec{v}|}$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d=\dfrac{7\sqrt{30}}{30}$}。

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在棱长为2的正方体$ABCD-A_1B_1C_1D_1$中，求直线$A_1B$与$AC_1$的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（建系与坐标）：}]
			以$A$为原点，$AB$为$x$轴，$AD$为$y$轴，$AA_1$为$z$轴建立空间直角坐标系。则$A(0,0,0)$，$B(2,0,0)$，$A_1(0,0,2)$，$C_1(2,2,2)$。
			\par\textbf{理由：}正方体顶点坐标可直接写出。
		\item[\textbf{第二步（确定向量）：}]
			$l_1=A_1B$过点$A_1(0,0,2)$，方向向量$\vec{u}=(2,0,-2)$（简化为$(1,0,-1)$）；$l_2=AC_1$过点$A(0,0,0)$，方向向量$\vec{v}=(2,2,2)$（简化为$(1,1,1)$）。取连接向量$\overrightarrow{AA_1}=(0,0,2)$。
			\par\textbf{理由：}由直线上两点确定方向。
		\item[\textbf{第三步（代公式计算）：}]
			\[
				\begin{aligned}
					\vec{n} &= \vec{u}\times\vec{v} \\ &=
					\begin{vmatrix}
						\mathbf{i} & \mathbf{j} & \mathbf{k} \\ 1 & 0 & -1 \\ 1 & 1 & 1
				\end{vmatrix}
					\\ &= (1,-2,1),
			\end{aligned}
			\]
			\[
				\begin{aligned}
					|\vec{n}| &= \sqrt{1^2+(-2)^2+1^2} \\ &= \sqrt{6},
			\end{aligned}
			\]
			$\overrightarrow{AA_1}\cdot\vec{n}=2$，故
			\[
				\begin{aligned}
					d &= \frac{2}{\sqrt{6}} \\ &= \frac{\sqrt{6}}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}代入距离公式$d=\dfrac{|\overrightarrow{AB}\cdot(\vec{u}\times\vec{v})|}{|\vec{u}\times\vec{v}|}$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d=\dfrac{\sqrt{6}}{3}$}。
\end{examplebox}
