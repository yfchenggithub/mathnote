\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\medskip
	\textbf{此公式将空间直线与平面交点问题转化为向量点积比值，通过参数$t$判断交点是否在线段上。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（棱与截面不平行）：} $\vec{n}\cdot\vec{v}\neq0$，确保分母非零且直线与平面有唯一交点。
		\item \textbf{条件 2（截面由不共线三点确定）：} 法向量 $\vec{n}\neq\mathbf{0}$，平面定义有效。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（检查分母非零）：} 使用前必须验证 $\vec{n}\cdot\vec{v}\neq0$，否则公式无效。
		\item \textbf{检查项（验证$t$范围）：} 只有 $t\in[0,1]$ 才说明交点在棱上，避免无效连接。
	\end{itemize}

\end{summarybox}
