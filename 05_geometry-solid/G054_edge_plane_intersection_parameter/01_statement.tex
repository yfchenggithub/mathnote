\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（棱存在方向向量）：}}]
			棱 $AB$ 存在方向向量 $\vec{v}=\overrightarrow{AB}$，且 $A$ 点位置向量 $\vec{a}=\overrightarrow{OA}$。
		\item[\textbf{\textcolor{CStatement}{条件 2（截面法向量非零）：}}]
			截面 $\pi$ 由不共线三点 $P,Q,R$ 确定，法向量 $\vec{n}=\overrightarrow{PQ}\times\overrightarrow{PR}\neq\mathbf{0}$。
		\item[\textbf{\textcolor{CStatement}{条件 3（不平行于截面）：}}]
			$\vec{n}\cdot\vec{v}\neq 0$，即 $AB$ 不平行于 $\pi$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（交点参数公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用向量方程组求出直线与平面交点的比例参数。
			\[
				t = \frac{\vec{n}\cdot(\vec{p}-\vec{a})}{\vec{n}\cdot\vec{v}}.
			\]
			交点位于线段 $AB$ 上当且仅当 $t\in[0,1]$，此时交点坐标为 $\vec{a}+t\vec{v}$。
		\item[\textbf{\textcolor{CStatement}{推广结论（截面多边形构建）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对几何体每条棱求 $t$，取 $t\in[0,1]$ 的点并连接得截面。交点坐标为 $\vec{a}+t\vec{v}$，判断条件 $t\in[0,1]$。
	\end{description}

\end{statementbox}
