\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视分母为零）}}]
			直接使用公式未检查 $\vec{n}\cdot\vec{v}\neq0$，导致计算无意义。
		\item[\textbf{\textcolor{CTrap}{正确理解（先验条件）：}}]
			使用前必须核算 $\vec{n}\cdot\vec{v}\neq0$，否则棱与截面平行或棱在截面内。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			将直线与平面交点的存在性与公式机械套用混淆，忽略了几何前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分子向量顺序颠倒）}}]
			计算分子时误用 $\vec{n}\cdot(\vec{a}-\vec{p})$ 而不是 $\vec{n}\cdot(\vec{p}-\vec{a})$，导致符号错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（统一符号）：}}]
			分子应为 $\vec{n}\cdot(\vec{p}-\vec{a})$，即“平面一点向量减棱起点”。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆不清）：}}]
			没有从联立方程推导，仅仅记忆公式，忽略了移项符号。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（$t$范围判断错误）}}]
			求出 $t$ 后直接认为交点在棱上，不检查 $t$ 是否满足 $0\le t\le1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（几何意义）：}}]
			$t$ 是位置比例，只有 $[0,1]$ 对应线段内部，否则在延长线上。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏直观）：}}]
			没有把 $t$ 与线段端点对应起来，$t=0$ 是 $A$，$t=1$ 是 $B$，超出范围自然不在线段上。
	\end{description}

\end{trapbox}
