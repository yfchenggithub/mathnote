\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（正方体棱长）：}}]
		      正方体棱长为 $a$，$a>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 球心位于正方体中心，半径等于正方体中心到顶点的距离。
		      \[
			      R=\frac{\sqrt{3}}{2}a
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（体对角线关系）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 正方体外接球的直径等于正方体体对角线长度。
		      \[
			      2R=\sqrt{3}a
		      \]

		\item[\textbf{\textcolor{CStatement}{常见变形（反求棱长）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 已知外接球半径 $R$ 时，可反求正方体棱长。
		      \[
			      a=\frac{2R}{\sqrt{3}}=\frac{2\sqrt{3}}{3}R
		      \]
	\end{description}
\end{statementbox}
