\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{正方体外接球半径等于其体对角线的一半，即}
	\[
		R=\frac{\sqrt{3}}{2}a.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{核心关系}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{体对角线：}
		      \[
			      d=\sqrt{3}a.
		      \]

		\item \textbf{外接球直径：}
		      \[
			      2R=d=\sqrt{3}a.
		      \]

		\item \textbf{外接球半径：}
		      \[
			      R=\frac{\sqrt{3}}{2}a.
		      \]

		\item \textbf{反求棱长：}
		      \[
			      a=\frac{2R}{\sqrt{3}}=\frac{2\sqrt{3}}{3}R.
		      \]
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（几何体条件）：} 研究对象必须是正方体。
		\item \textbf{条件2（棱长为正）：} 正方体棱长为 $a$，且 $a>0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式混淆）：} 不要记成
		      \[
			      R=\frac{\sqrt{2}}{2}a.
		      \]
		      这是平面正方形外接圆半径，不是空间正方体外接球半径。

		\item \textbf{易错点（内外混淆）：} 不要把外接球半径与内切球半径混淆。正方体内切球半径是
		      \[
			      r=\frac{a}{2},
		      \]
		      而外接球半径是
		      \[
			      R=\frac{\sqrt{3}}{2}a.
		      \]

		\item \textbf{检查项（直径验证）：} 检查所得半径是否满足
		      \[
			      2R=\sqrt{3}a.
		      \]
		      如果满足，说明外接球直径等于正方体体对角线，结果通常是正确的。
	\end{itemize}
\end{summarybox}
