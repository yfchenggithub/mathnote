\begin{examplebox}\textbf{\textcolor{CExample}{例1（基础）}}\textbf{\textcolor{CExample}{题目：}} 若正方体的棱长为$2$，求其外接球半径$R$。\textbf{\textcolor{CExample}{解题步骤：}}\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]\item[\textbf{第一步（写半径公式）：}] 根据正方体外接球半径公式，有\[ R = \frac{\sqrt{3}}{2}a. \]\par\textbf{理由：} 直接套用结论。\item[\textbf{第二步（代棱长值）：}] 代入$a=2$，\[ R = \frac{\sqrt{3}}{2} \times 2. \]\par\textbf{理由：} 已知棱长数值。\item[\textbf{第三步（化简结果）：}] 计算得\[ R = \sqrt{3}. \]\par\textbf{理由：} 分数乘法。\end{description}\textbf{\textcolor{CExample}{关键结论：}} $R = \sqrt{3}$。\medskip\textbf{\textcolor{CExample}{例2（稍变形）}}\textbf{\textcolor{CExample}{题目：}} 已知正方体的外接球半径为$\sqrt{6}$，求该正方体的棱长$a$。\textbf{\textcolor{CExample}{解题步骤：}}\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]\item[\textbf{第一步（写半径公式）：}] 由正方体外接球性质，有\[ R = \frac{\sqrt{3}}{2}a. \]\par\textbf{理由：} 结论公式。\item[\textbf{第二步（反解棱长）：}] 将公式变形为\[ a = \frac{2R}{\sqrt{3}}. \]\par\textbf{理由：} 等式两边同乘$\frac{2}{\sqrt{3}}$。\item[\textbf{第三步（代入数据）：}] 代入$R=\sqrt{6}$，\[ \begin{aligned} a &= \frac{2\sqrt{6}}{\sqrt{3}} \\ &= 2\sqrt{2}. \end{aligned} \]\par\textbf{理由：} $\frac{\sqrt{6}}{\sqrt{3}} = \sqrt{2}$。\end{description}\textbf{\textcolor{CExample}{关键结论：}} $a = 2\sqrt{2}$。\medskip\textbf{\textcolor{CExample}{例3（综合应用）}}\textbf{\textcolor{CExample}{题目：}} 若正方体的外接球体积为$4\sqrt{3}\pi$，求正方体的棱长$a$。\textbf{\textcolor{CExample}{解题步骤：}}\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]\item[\textbf{第一步（写体积关系）：}] 球的体积公式$V = \frac{4}{3}\pi R^3$，因此\[ R^3 = \frac{3V}{4\pi}. \]\par\textbf{理由：} 反解$R^3$。\item[\textbf{第二步（代入体积值）：}] 代入$V=4\sqrt{3}\pi$，\[ \begin{aligned} R^3 &= \frac{3\cdot 4\sqrt{3}\pi}{4\pi} \\ &= 3\sqrt{3}. \end{aligned} \]\par\textbf{理由：} 化简系数，$\pi$约去。\item[\textbf{第三步（求$R$）：}] 开立方得\[ \begin{aligned} R &= \sqrt[3]{3\sqrt{3}} \\ &= \sqrt[3]{3^{3/2}} \\ &= 3^{1/2} \\ &= \sqrt{3}. \end{aligned} \]\par\textbf{理由：} 指数运算，$3\sqrt{3}=3^{1}\cdot 3^{1/2}=3^{3/2}$，开三次方得$3^{(3/2)\cdot (1/3)} = 3^{1/2}$。\item[\textbf{第四步（利用半径求棱长）：}] 由$R = \frac{\sqrt{3}}{2}a$，得\[ \begin{aligned} a &= \frac{2R}{\sqrt{3}} \\ &= \frac{2\sqrt{3}}{\sqrt{3}} \\ &= 2. \end{aligned} \]\par\textbf{理由：} 代入$R=\sqrt{3}$并化简。\end{description}\textbf{\textcolor{CExample}{关键结论：}} 棱长$a=2$。\end{examplebox}
