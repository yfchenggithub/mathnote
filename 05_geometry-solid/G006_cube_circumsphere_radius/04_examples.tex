\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 若正方体的棱长为 $2$，求其外接球半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写半径公式）：}]
		      根据正方体外接球半径公式，有
		      \[
			      R=\frac{\sqrt{3}}{2}a.
		      \]
		      \par\textbf{理由：} 直接套用结论。

		\item[\textbf{第二步（代入棱长）：}]
		      代入 $a=2$，得
		      \[
			      R=\frac{\sqrt{3}}{2}\times 2.
		      \]
		      \par\textbf{理由：} 已知正方体棱长为 $2$。

		\item[\textbf{第三步（化简结果）：}]
		      计算得
		      \[
			      R=\sqrt{3}.
		      \]
		      \par\textbf{理由：} 分数乘法化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $R=\sqrt{3}$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 已知正方体的外接球半径为 $\sqrt{6}$，求该正方体的棱长 $a$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写半径公式）：}]
		      由正方体外接球性质，有
		      \[
			      R=\frac{\sqrt{3}}{2}a.
		      \]
		      \par\textbf{理由：} 正方体外接球半径等于体对角线的一半。

		\item[\textbf{第二步（反解棱长）：}]
		      由
		      \[
			      R=\frac{\sqrt{3}}{2}a
		      \]
		      两边同乘 $\frac{2}{\sqrt{3}}$，得
		      \[
			      a=\frac{2R}{\sqrt{3}}.
		      \]
		      \par\textbf{理由：} 解关于 $a$ 的一次方程，注意是乘以 $\frac{2}{\sqrt{3}}$，不是开平方。

		\item[\textbf{第三步（代入数据）：}]
		      代入 $R=\sqrt{6}$，得
		      \[
			      \begin{aligned}
				      a
				       & =\frac{2\sqrt{6}}{\sqrt{3}} \\
				       & =2\sqrt{2}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：}
		      \[
			      \frac{\sqrt{6}}{\sqrt{3}}
			      =\sqrt{\frac{6}{3}}
			      =\sqrt{2}.
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $a=2\sqrt{2}$。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 若正方体的外接球体积为 $4\sqrt{3}\pi$，求正方体的棱长 $a$。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写体积关系）：}]
		      球的体积公式为
		      \[
			      V=\frac{4}{3}\pi R^3,
		      \]
		      因此
		      \[
			      R^3=\frac{3V}{4\pi}.
		      \]
		      \par\textbf{理由：} 由球体积公式反解 $R^3$。

		\item[\textbf{第二步（代入体积值）：}]
		      代入 $V=4\sqrt{3}\pi$，得
		      \[
			      \begin{aligned}
				      R^3
				       & =\frac{3\cdot 4\sqrt{3}\pi}{4\pi} \\
				       & =3\sqrt{3}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 化简系数，$\pi$ 约去。

		\item[\textbf{第三步（求半径 $R$）：}]
		      由
		      \[
			      R^3=3\sqrt{3}
		      \]
		      可得
		      \[
			      3\sqrt{3}=3\cdot 3^{\frac12}=3^{\frac32}.
		      \]
		      所以
		      \[
			      \begin{aligned}
				      R
				       & =\sqrt[3]{3\sqrt{3}}                \\
				       & =\left(3^{\frac32}\right)^{\frac13} \\
				       & =3^{\frac12}                        \\
				       & =\sqrt{3}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 先把 $3\sqrt{3}$ 化为指数形式 $3^{\frac32}$，再开三次方。

		\item[\textbf{第四步（利用半径求棱长）：}]
		      由
		      \[
			      R=\frac{\sqrt{3}}{2}a
		      \]
		      得
		      \[
			      a=\frac{2R}{\sqrt{3}}.
		      \]
		      代入 $R=\sqrt{3}$，得
		      \[
			      \begin{aligned}
				      a
				       & =\frac{2\sqrt{3}}{\sqrt{3}} \\
				       & =2.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 外接球半径已求出，再反解正方体棱长。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} 棱长 $a=2$。
\end{examplebox}
