\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（系数遗漏）}}] 使用公式$h=\dfrac{V}{S}$，忘记乘以$3$。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式本源）：}}] 体积公式$V=\dfrac13Sh$，所以$h=\dfrac{3V}{S}$，系数$3$不可缺少。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}] 将锥体体积与柱体体积$V=Sh$混淆，误以为高是$V/S$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略非零）}}] 应用时直接除$S$，未考虑$S=0$的情形。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件约束）：}}] 底面面积$S>0$（实际几何图形中一定成立），但计算时需确认$S\neq0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏检验）：}}] 未养成分母不为零的验证习惯。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（底面对应错误）}}] 等体积法求高时，选错底面导致高与底面积不匹配。
		\item[\textbf{\textcolor{CTrap}{正确理解（唯一对应）：}}] 高必须垂直于所选底面，且底面积必须是该底面的面积。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 对“底面”与“高”的对应关系理解不清，常用于点面距离中顶点与平面不对应。
	\end{description}
\end{trapbox}
