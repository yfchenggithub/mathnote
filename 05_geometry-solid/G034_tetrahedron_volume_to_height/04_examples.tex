	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一个三棱锥的体积$V=12$，底面面积$S=4$，求该底面对应的高$h$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 写出等体积法公式：$h=\dfrac{3V}{S}$。
			\par\textbf{理由：}根据三棱锥体积公式变形得到。
		\item[\textbf{第二步（代入数值）：}] 代入$V=12$，$S=4$，得$h=\dfrac{3\times12}{4}$。
			\par\textbf{理由：}直接代入已知数据。
		\item[\textbf{第三步（计算结果）：}] 计算得$h=9$。
			\par\textbf{理由：}算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{所求高为$\mathbf{9}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	一个三棱锥的体积$V=6\sqrt{3}$，底面是边长为$2$的正三角形，求该底面对应的高$h$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面积）：}] 正三角形面积公式$S=\dfrac{\sqrt{3}}{4}a^2$，代入$a=2$，得$S=\sqrt{3}$。
			\par\textbf{理由：}面积公式直接计算。
		\item[\textbf{第二步（套用公式）：}] 由等体积法$h=\dfrac{3V}{S}$，代入$V=6\sqrt{3}$，$S=\sqrt{3}$，得$h=\dfrac{3\times6\sqrt{3}}{\sqrt{3}}$。
			\par\textbf{理由：}应用公式，代入已知。
		\item[\textbf{第三步（简化结果）：}] 化简得$h=18$。
			\par\textbf{理由：}约去$\sqrt{3}$，算出数值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{所求高为$\mathbf{18}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在棱长为$1$的正方体$ABCD-A_1B_1C_1D_1$中，求点$A_1$到平面$AB_1D_1$的距离$h$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求三棱锥体积）：}] 考虑三棱锥$A_1-AB_1D_1$，以$\triangle A_1B_1D_1$为底面（面积为$\frac12$），高为$AA_1=1$。由体积公式得$V = \frac13 \times \frac12 \times 1$。计算得$V = \frac16$。
			\par\textbf{理由：}体积计算选择适当底面和高。

		\item[\textbf{第二步（求底面面积）：}] 底面$\triangle AB_1D_1$是边长为$\sqrt{2}$的正三角形。由正三角形面积公式$S = \frac{\sqrt{3}}{4} a^2$，代入$a = \sqrt{2}$得$S = \frac{\sqrt{3}}{4} \times 2$。所以$S = \frac{\sqrt{3}}{2}$。
			\par\textbf{理由：}正三角形面积公式。

		\item[\textbf{第三步（等体积法求高）：}] 由体积公式$V=\frac13 S h$得$h = \frac{3V}{S}$。代入$V = \frac16$，$S = \frac{\sqrt{3}}{2}$。先计算$3V = \frac12$，然后$h = \frac12 \Big/ \frac{\sqrt{3}}{2}$，化简得$h = \frac{1}{\sqrt{3}}$。有理化得$h = \frac{\sqrt{3}}{3}$。
			\par\textbf{理由：}代入已知量，化简得结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{所求距离为$\mathbf{\dfrac{\sqrt{3}}{3}}$。}
