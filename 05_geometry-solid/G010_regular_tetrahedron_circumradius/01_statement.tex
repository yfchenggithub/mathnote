	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（几何体限制）：}}] 几何体必须是正四面体，即六条棱长全部相等。
		\item[\textbf{\textcolor{CStatement}{条件 2（防止误用）：}}] 若只是一般三棱锥或正三棱锥，不能直接套用本结论；只有当它的所有棱长都相等时，才是正四面体。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 正四面体外接球半径与棱长成正比，系数为 $\frac{\sqrt{6}}{4}$。
		      \[
			      R = \frac{\sqrt{6}}{4}a.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（棱高关系）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 正四面体外接球半径等于正四面体高的 $\frac{3}{4}$。
		      \[
			      R = \frac{3}{4}h,\quad h = \frac{\sqrt{6}}{3}a.
		      \]
	\end{description}

\end{statementbox}
