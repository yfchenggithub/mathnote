\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（球心位置）}}] 认为外接球球心在底面正三角形的中心。
		\item[\textbf{\textcolor{CTrap}{正确理解（球心在高线）：}}] 外接球球心在正四面体的高线上，距离底面为 $\frac{1}{4}h$，距离顶点为 $\frac{3}{4}h$，所以外接球半径为 $R=\frac{3}{4}h$。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏空间对称分析）：}}] 只考虑了底面 $BCD$ 的对称性，误将底面外心当成空间外接球球心。实际上球心必须同时到 $A,B,C,D$ 四个顶点距离相等。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（球心远近判断）}}] 误以为外接球球心更靠近正四面体的顶点。
		\item[\textbf{\textcolor{CTrap}{正确理解（更靠近底面）：}}] 若正四面体的高为 $h$，则球心到底面中心的距离为 $\frac{1}{4}h$，到顶点的距离为 $\frac{3}{4}h$。因此球心更靠近底面，而不是更靠近顶点。
		\item[\textbf{\textcolor{CTrap}{错因分析（把半径方向看反）：}}] 外接球半径 $R=\frac{3}{4}h$ 是球心到顶点的距离，不是球心到底面的距离。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（中心对称误解）}}] 把正四面体说成中心对称图形。
		\item[\textbf{\textcolor{CTrap}{正确理解（高度对称而非中心对称）：}}] 正四面体具有很强的空间对称性，但不是中心对称图形。在正四面体中，外接球球心、内切球球心、重心和垂心重合。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混用）：}}] “对称性强”不等于“中心对称”。中心对称要求图形关于某一点反演后仍与自身重合，正四面体不满足这一点。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（公式记忆）}}] 误记外接球半径公式为 $R = \frac{\sqrt{6}}{3}a$ 或 $R = \frac{\sqrt{6}}{2}a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数核对）：}}] 正确公式为
		      \[
			      R = \frac{\sqrt{6}}{4}a.
		      \]
		      可结合高
		      \[
			      h = \frac{\sqrt{6}}{3}a
		      \]
		      以及关系
		      \[
			      R = \frac{3}{4}h
		      \]
		      辅助记忆。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆高与半径）：}}] 高公式的系数是 $\frac{\sqrt{6}}{3}$，外接球半径系数是 $\frac{\sqrt{6}}{4}$，两者容易混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点五（适用范围）}}] 将公式套用到一般三棱锥或正三棱锥。
		\item[\textbf{\textcolor{CTrap}{正确理解（只限正四面体）：}}] 该公式仅适用于所有棱长都相等的正四面体。正三棱锥不一定是正四面体，只有当底面三边和三条侧棱全部相等时，才可使用
		      \[
			      R=\frac{\sqrt{6}}{4}a.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（条件忽略）：}}] “底面是正三角形”或“顶点在底面中心正上方”还不够，必须检查六条棱是否全部相等。
	\end{description}
\end{trapbox}
