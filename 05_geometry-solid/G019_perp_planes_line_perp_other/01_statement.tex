	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（面面垂直）：}}] 平面 $\alpha$ 与 $\beta$ 满足 $\alpha\perp\beta$，且交线为 $l$。
		\item[\textbf{\textcolor{CStatement}{条件 2（线在面内垂交）：}}] 直线 $a$ 在平面 $\alpha$ 内，且 $a\perp l$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线面垂直）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在垂直平面内且垂直于交线的直线，垂直于另一个平面。
			\[
				a\perp\beta.
			\]

		\item[\textbf{\textcolor{CStatement}{关联性质（线面垂直推出线线垂直）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若直线垂直于平面，则它垂直于平面内的任何直线，特别地垂直于交线。
			\[
				a\perp\beta,\ l\subset\beta \;\Longrightarrow\; a\perp l.
			\]
	\end{description}

\end{statementbox}
