\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{面面垂直时，面内垂交得线面垂直。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（面面垂直）：} 平面 $\alpha\perp\beta$，且交线为 $l$。
		\item \textbf{条件 2（线在面内垂交）：} 直线 $a\subset\alpha$ 且 $a\perp l$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（缺线在面内）：} 使用定理时，不能忘记 $a$ 必须在 $\alpha$ 内。
		\item \textbf{检查项（找对交线）：} 每次应用前，先核实两个平面及其交线。
	\end{itemize}
\end{summarybox}
