\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知正三棱柱底面边长 $a=2$，高 $h=3$，求其外接球半径 $R$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认条件）：}] $a=2$, $h=3$.
		      \par\textbf{理由：} 题目明确给出正三棱柱，且底面边长和高均为正数，可以直接套用公式.

		\item[\textbf{第二步（套用公式）：}] 代入外接球半径公式.
		      \[
			      R = \sqrt{\frac{2^{2}}{3} + \frac{3^{2}}{4}}.
		      \]
		      \par\textbf{理由：} 正三棱柱外接球公式为
		      \[
			      R=\sqrt{\frac{a^{2}}{3}+\frac{h^{2}}{4}}.
		      \]

		\item[\textbf{第三步（计算化简）：}] 通分计算.
		      \[
			      \begin{aligned}
				      R
				       & = \sqrt{\frac{4}{3} + \frac{9}{4}}     \\
				       & = \sqrt{\frac{16}{12} + \frac{27}{12}} \\
				       & = \sqrt{\frac{43}{12}}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 分数通分，合并同类项.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$R = \sqrt{\dfrac{43}{12}}$，也可写为 $\dfrac{\sqrt{129}}{6}$.}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知正三棱柱的外接球半径 $R = \sqrt{3}$，高 $h=2$，求底面边长 $a$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认条件）：}] $R = \sqrt{3}$, $h = 2$.
		      \par\textbf{理由：} 已知外接球半径和高，要求底面边长，属于反求参数问题.

		\item[\textbf{第二步（存在性检查）：}] 反求 $a$ 时，需要满足
		      \[
			      R^{2}>\frac{h^{2}}{4}.
		      \]
		      本题中
		      \[
			      3>\frac{2^{2}}{4}=1,
		      \]
		      所以存在满足条件的正三棱柱.
		      \par\textbf{理由：} 因为
		      \[
			      a^{2}=3\left(R^{2}-\frac{h^{2}}{4}\right),
		      \]
		      且 $a>0$，所以必须有 $R^{2}>h^{2}/4$.

		\item[\textbf{第三步（构造方程）：}] 由公式
		      \[
			      R^{2} = \frac{a^{2}}{3} + \frac{h^{2}}{4}
		      \]
		      代入已知得
		      \[
			      3 = \frac{a^{2}}{3} + 1.
		      \]
		      \par\textbf{理由：} 代入 $R=\sqrt{3}$, $h=2$，得 $3 = a^{2}/3 + 1$.

		\item[\textbf{第四步（解方程）：}] 移项化简，开平方取正根.
		      \[
			      \begin{aligned}
				      \frac{a^{2}}{3} & = 2,        \\
				      a^{2}           & = 6,        \\
				      a               & = \sqrt{6}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 底面边长为正数，所以开平方取正根.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$a = \sqrt{6}$.}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 正三棱柱的外接球表面积为 $8\pi$，底面边长 $a=2$，求高 $h$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由表面积求半径）：}] 球的表面积
		      \[
			      S = 4\pi R^{2}.
		      \]
		      由 $S=8\pi$ 得
		      \[
			      \begin{aligned}
				      4\pi R^{2} & = 8\pi, \\
				      R^{2}      & = 2.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 外接球表面积公式为 $S=4\pi R^{2}$.

		\item[\textbf{第二步（存在性检查）：}] 反求 $h$ 时，需要满足
		      \[
			      R^{2}>\frac{a^{2}}{3}.
		      \]
		      本题中
		      \[
			      2>\frac{2^{2}}{3}=\frac43,
		      \]
		      所以存在满足条件的正三棱柱.
		      \par\textbf{理由：} 因为
		      \[
			      h^{2}=4\left(R^{2}-\frac{a^{2}}{3}\right),
		      \]
		      且 $h>0$，所以必须有 $R^{2}>a^{2}/3$.

		\item[\textbf{第三步（代入外接球公式）：}] 将 $R^{2}=2$, $a=2$ 代入
		      \[
			      R^{2} = \frac{a^{2}}{3} + \frac{h^{2}}{4}.
		      \]
		      得
		      \[
			      2 = \frac{4}{3} + \frac{h^{2}}{4}.
		      \]
		      \par\textbf{理由：} 代入 $a=2$ 得 $a^{2}/3=4/3$.

		\item[\textbf{第四步（解高 $h$）：}] 移项后逐步计算.
		      \[
			      \begin{aligned}
				      2 - \frac{4}{3} & = \frac{2}{3},         \\
				      \frac{h^{2}}{4} & = \frac{2}{3},         \\
				      h^{2}           & = \frac{8}{3},         \\
				      h               & = \frac{2\sqrt{6}}{3}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 高为正数，所以开平方取正根.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$h = \dfrac{2\sqrt{6}}{3}$.}
\end{examplebox}
