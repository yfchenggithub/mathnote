\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（形状条件）：}}] 底面为正三角形，且侧棱垂直于底面的三棱柱，即正三棱柱.
		\item[\textbf{\textcolor{CStatement}{条件2（已知参数）：}}] 底面正三角形边长为 $a$，棱柱高为 $h$，其中 $a>0$, $h>0$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 外接球半径由底面边长和高直接算出.
		      \[
			      R = \sqrt{\frac{a^{2}}{3} + \frac{h^{2}}{4}}.
		      \]

		\item[\textbf{\textcolor{CStatement}{常见变形（平方拆分）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 半径平方等于底面外接圆半径平方加上高的一半的平方.
		      \[
			      R^{2} = r_{\triangle}^{2} + \left(\frac{h}{2}\right)^{2},\quad
			      r_{\triangle} = \frac{a}{\sqrt{3}}.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{参数限制：}} 本结论中的 $R=\sqrt{a^{2}/3+h^{2}/4}$ 只适用于正三棱柱；其中 $a$ 必须是底面正三角形的边长，$h$ 必须是棱柱的高，且 $a>0$, $h>0$.
\end{statementbox}
