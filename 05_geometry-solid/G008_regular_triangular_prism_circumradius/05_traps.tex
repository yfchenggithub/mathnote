\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（外接圆半径混淆）}}] 将底面正三角形的外接圆半径错误记为 $a/(2\sqrt{3})$ 或 $a/\sqrt{2}$ 等.
		\item[\textbf{\textcolor{CTrap}{正确理解（外接圆公式）：}}] 正三角形外接圆半径
		      \[
			      r=\frac{a}{\sqrt3},
		      \]
		      内切圆半径为
		      \[
			      r_{\text{内}}=\frac{a}{2\sqrt3}.
		      \]
		      两者不同.
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}] 混淆了中心到顶点与中心到边的距离；外接圆半径是中心到顶点，内切圆半径是中心到边.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（高的一半遗漏）}}] 代入公式时写成
		      \[
			      R^{2}=\frac{a^{2}}{3}+h^{2},
		      \]
		      忘记将 $h$ 除以 $2$.
		\item[\textbf{\textcolor{CTrap}{正确理解（球心在中点）：}}] 球心在上下底面中心连线的中点，所以从球心到任一底面的竖直距离为 $h/2$，正确项为
		      \[
			      \left(\frac h2\right)^{2}.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（几何误解）：}}] 误以为球心在某一个底面上，未意识到球心是上下底面对称后的中点.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（条件误用）}}] 把
		      \[
			      R=\sqrt{\frac{a^{2}}{3}+\frac{h^{2}}{4}}
		      \]
		      直接用于底面不是正三角形的直三棱柱，或直接用于斜棱柱.
		\item[\textbf{\textcolor{CTrap}{正确理解（正三棱柱前提）：}}] 本公式中的 $a/\sqrt3$ 来自底面正三角形的外接圆半径，所以只有正三棱柱才可直接套用
		      \[
			      R=\sqrt{\frac{a^{2}}{3}+\frac{h^{2}}{4}}.
		      \]
		      对于一般直三棱柱，若底面三角形外接圆半径为 $R_{\triangle}$，则可用
		      \[
			      R^{2}=R_{\triangle}^{2}+\left(\frac h2\right)^{2}.
		      \]
		      但不能把一般三角形的外接圆半径误写成 $a/\sqrt3$.
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视限制）：}}] 只记住了形式，没有确认底面是否为正三角形，也没有确认侧棱是否垂直底面.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（反求参数不验存在性）}}] 已知 $R$ 和一个参数时，直接解方程，却不判断另一个参数是否能取正数.
		\item[\textbf{\textcolor{CTrap}{正确理解（先看差值是否为正）：}}] 若反求底面边长 $a$，由
		      \[
			      a^{2}=3\left(R^{2}-\frac{h^{2}}{4}\right)
		      \]
		      可知必须满足
		      \[
			      R^{2}>\frac{h^{2}}{4}.
		      \]
		      若反求高 $h$，由
		      \[
			      h^{2}=4\left(R^{2}-\frac{a^{2}}{3}\right)
		      \]
		      可知必须满足
		      \[
			      R^{2}>\frac{a^{2}}{3}.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（只算不判）：}}] 只关注代数方程，没有结合 $a>0$, $h>0$ 的几何意义；若差值不为正，就不存在这样的正三棱柱.
	\end{description}
\end{trapbox}
