\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{正三棱柱外接球半径由底面正三角形边长和棱柱高决定：}
	\[
		\boxed{R=\sqrt{\frac{a^{2}}{3}+\frac{h^{2}}{4}}}.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（几何形状）：} 底面为正三角形，且侧棱垂直于底面，即正三棱柱.
		\item \textbf{条件 2（参数正数）：} 底面正三角形边长 $a>0$，棱柱高 $h>0$.
		\item \textbf{条件 3（参数含义）：} $a$ 是底面正三角形的边长，不是底面外接圆半径；$h$ 是上下底面之间的距离.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{记忆方式}}

	先记底面正三角形外接圆半径
	\[
		r=\frac a{\sqrt3},
	\]
	再记球心到任一底面的竖直距离
	\[
		\frac h2.
	\]
	于是由勾股定理得
	\[
		R^{2}=r^{2}+\left(\frac h2\right)^{2}
		=\frac{a^{2}}{3}+\frac{h^{2}}{4}.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式记忆）：} 牢记
		      \[
			      R^{2}=\frac{a^{2}}{3}+\left(\frac h2\right)^{2},
		      \]
		      不要遗漏 $h/2$，也不要把正三角形外接圆半径误写成 $a/(2\sqrt3)$.

		\item \textbf{检查项（前提确认）：} 使用前必须确认几何体是正三棱柱；如果底面不是正三角形，不能直接使用 $a/\sqrt3$.

		\item \textbf{反求参数（存在性检查）：} 若反求 $a$，需
		      \[
			      R^{2}>\frac{h^{2}}{4};
		      \]
		      若反求 $h$，需
		      \[
			      R^{2}>\frac{a^{2}}{3}.
		      \]
		      否则不存在满足条件的正三棱柱.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{拓展但不乱用}}

	对于一般直三棱柱，若底面三角形外接圆半径为 $R_{\triangle}$，则有
	\[
		R^{2}=R_{\triangle}^{2}+\left(\frac h2\right)^{2}.
	\]
	正三棱柱只是其中底面外接圆半径满足 $R_{\triangle}=a/\sqrt3$ 的特殊情况；斜棱柱不能直接套用本结论.
\end{summarybox}
