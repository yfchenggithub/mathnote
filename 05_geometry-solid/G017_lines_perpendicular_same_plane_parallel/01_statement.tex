\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（线面垂直条件）：}}] 直线 $a\perp\alpha$，直线 $b\perp\alpha$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两直线平行）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			垂直于同一平面的两条直线互相平行。
			\[
				a\parallel b
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（平行则垂直）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平行线中一条垂直于平面，另一条也垂直于该平面。
			\[
				a\parallel b,\ a\perp\alpha \Rightarrow b\perp\alpha
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{使用提醒：}}
	常用于立体几何中证明两条直线平行，使用时需明确两条直线都与同一平面垂直。
\end{statementbox}
