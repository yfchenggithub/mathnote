\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在长方体$ABCD-A_1B_1C_1D_1$中，$AA_1\perp$平面$ABCD$，$BB_1\perp$平面$ABCD$，求证$AA_1\parallel BB_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别垂线）：}] 由已知条件可知，$AA_1\perp$平面$ABCD$，$BB_1\perp$平面$ABCD$。
			\par\textbf{理由：}
			已知直接给出。

		\item[\textbf{第二步（套用定理）：}] 因为$AA_1$和$BB_1$都垂直于同一平面$ABCD$，根据线面垂直的性质定理，$AA_1\parallel BB_1$。
			\par\textbf{理由：}
			垂直于同一平面的两条直线平行。

		\item[\textbf{第三步（写出结论）：}] 因此$AA_1\parallel BB_1$成立。
			\par\textbf{理由：}
			由第二步得到。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AA_1\parallel BB_1$}。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在四棱锥$P-ABCD$中，$PA\perp$平面$ABCD$，过点$C$作$CQ\perp$平面$ABCD$，垂足为$C$，求证$PA\parallel CQ$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定垂线）：}] 由$PA\perp$平面$ABCD$和$CQ\perp$平面$ABCD$，可知$PA$与$CQ$都是平面$ABCD$的垂线。
			\par\textbf{理由：}
			已知条件，注意垂足$A$与$C$不同，但均满足线面垂直。

		\item[\textbf{第二步（应用定理）：}] 因为$PA$与$CQ$垂直于同一平面$ABCD$，所以$PA\parallel CQ$（垂直于同一平面的两条直线平行）。
			\par\textbf{理由：}
			线面垂直的性质定理。

		\item[\textbf{第三步（确认结论）：}] 故$PA\parallel CQ$。
			\par\textbf{理由：}
			由第二步直接得出。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA\parallel CQ$}。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知平面$\alpha\parallel\beta$，直线$a\perp\alpha$，直线$b\perp\beta$，求证$a\parallel b$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平行平面性质）：}] 由$\alpha\parallel\beta$及$a\perp\alpha$，根据“若一条直线垂直于两个平行平面中的一个，则它也垂直于另一个”，得$a\perp\beta$。
			\par\textbf{理由：}
			平行平面的垂线性质（教材定理）。

		\item[\textbf{第二步（再次应用定理）：}] 因为$a\perp\beta$且$b\perp\beta$，所以$a\parallel b$（垂直于同一平面的两条直线平行）。
			\par\textbf{理由：}
			线面垂直的性质定理。

		\item[\textbf{第三步（结论完成）：}] 因此$a\parallel b$。
			\par\textbf{理由：}
			由第二步得到。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a\parallel b$}。
\end{examplebox}
