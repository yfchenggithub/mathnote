\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{垂直于同一平面的两条直线互相平行。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（线面垂直条件）：} 直线 $a\perp\alpha$。
		\item \textbf{条件2（同一平面条件）：} 直线 $b\perp\alpha$（注意必须是同一个$\alpha$）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（同平面检查）：} 使用前要确认两条直线垂直于同一个平面，不能分别垂直于不同平面。
		\item \textbf{检查项（垂足位置）：} 垂足是否相同不影响平行结论，只需垂直同一平面即可。
	\end{itemize}
\end{summarybox}
