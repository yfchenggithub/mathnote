\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用过空间一点有且只有一条直线垂直于已知平面的唯一性，通过作平行线导出两直线重合。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{第一步（作平行线）：}]
			设 $a\cap\alpha=A$，$b\cap\alpha=B$，在直线 $a$ 上任取一点 $P$（$P\neq A$），过 $P$ 作直线 $c\parallel b$。
			\[
				a\cap\alpha=A,\quad b\cap\alpha=B,\quad P\in a\ (P\neq A),\quad c\parallel b.
			\]
			\par\textbf{理由：}
			构造与 $b$ 平行且经过 $a$ 上一点的直线 $c$，为后续使用唯一性做准备。

		\item[\textbf{第二步（证$c\perp\alpha$）：}]
			由 $b\perp\alpha$ 和 $c\parallel b$，根据“平行线垂直传递”定理（若一条直线垂直于平面，则其平行线也垂直于该平面），得 $c\perp\alpha$。
			\[
				b\perp\alpha,\ c\parallel b \Rightarrow c\perp\alpha.
			\]
			\par\textbf{理由：}
			线面垂直的判定定理（平行线垂直传递）。

		\item[\textbf{第三步（唯一性导出重合）：}]
			因为 $a\perp\alpha$，$c\perp\alpha$，且 $a$ 和 $c$ 均过点 $P$，由“过空间一点有且只有一条直线与已知平面垂直”可知，直线 $a$ 与 $c$ 重合。
			\[
				a\perp\alpha,\ c\perp\alpha,\ P\in a,\ P\in c\Rightarrow a\equiv c.
			\]
			\par\textbf{理由：}
			过一点作平面的垂线唯一。

		\item[\textbf{第四步（结论）：}]
			由 $c\parallel b$ 及 $a\equiv c$，得 $a\parallel b$。
			\[
				c\parallel b,\ a\equiv c\Rightarrow a\parallel b.
			\]
			\par\textbf{理由：}
			平行线的传递性。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，垂直于同一平面的两条直线互相平行。该定理是证明空间中两条直线平行的重要依据，常用于立体几何推理。}
\end{proofbox}
