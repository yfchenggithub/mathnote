\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（符号设定）：}}]
			设直线 $a,b$，平面 $\alpha,\beta,\gamma$；若 $a$ 与 $b$ 相交，记交点为 $P$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线面平行判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若线线平行且一条在面外、一条在面内，则线面平行。
			\[
				a\parallel b,\ a\not\subset\alpha,\ b\subset\alpha \Rightarrow a\parallel\alpha
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（面面平行判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若一个平面内两条相交直线分别平行于另一平面，则两平面平行。
			\[
				a,b\subset\alpha,\ a\cap b=P,\ a\parallel\beta,\ b\parallel\beta \Rightarrow \alpha\parallel\beta
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（面面平行性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两平行平面被同平面截，所得截线平行。
			\[
				\alpha\parallel\beta,\ \gamma\cap\alpha=a,\ \gamma\cap\beta=b \Rightarrow a\parallel b
			\]
		\item[\textbf{\textcolor{CStatement}{结论四（线面平行性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若线面平行，过该直线作平面与已知平面相交，则交线与该直线平行。
			\[
				a\parallel\alpha,\ a\subset\beta,\ \alpha\cap\beta=b \Rightarrow a\parallel b
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（面面平行推线面）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两平面平行时，一个平面内的任意直线都平行于另一平面。
			\[
				\alpha\parallel\beta,\ a\subset\alpha \Rightarrow a\parallel\beta
			\]
	\end{description}

\end{statementbox}
