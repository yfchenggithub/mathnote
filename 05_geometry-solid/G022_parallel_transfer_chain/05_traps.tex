\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（漏判线在面外）：}}]
			在应用线面平行判定时，只注意线线平行，忽略检查直线是否在平面外。
		\item[\textbf{\textcolor{CTrap}{正确理解（需找外在条件）：}}]
			必须同时确认直线不在平面内（$a\not\subset\alpha$）。若$a\subset\alpha$，则$a$与$\alpha$有无数公共点，不成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视包含关系）：}}]
			直观上认为“线线平行则线面平行”，遗漏了“线在面外”的前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（相交直线遗漏）：}}]
			在面面平行判定中，只找出两条平行线分别平行于另一平面，但未检查两线是否相交。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须两线相交）：}}]
			定理要求一个平面内的两条相交直线都平行于另一平面。若两条直线平行，它们不能确定该平面，无法保证面面平行。
		\item[\textbf{\textcolor{CTrap}{错因分析（误用平行直线）：}}]
			平行直线不能代表整个平面，即使它们都平行于另一平面，两平面仍可能相交。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略截平面条件）：}}]
			在应用面面平行性质时，直接由$\alpha\parallel\beta$及$a\subset\alpha$、$b\subset\beta$推出$a\parallel b$。
		\item[\textbf{\textcolor{CTrap}{正确理解（需第三平面截）：}}]
			性质定理要求$\gamma\cap\alpha=a$，$\gamma\cap\beta=b$，即两条交线是由同一平面$\gamma$截得的；缺少截面$\gamma$时，$a$与$b$可能异面不平行。
		\item[\textbf{\textcolor{CTrap}{错因分析（误解平行关系）：}}]
			误认为两平行平面内的任意直线都互相平行，实际上它们可能方向不一致，只有通过同一截平面才能保证共面且平行。
	\end{description}

\end{trapbox}
