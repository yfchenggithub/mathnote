\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	如图，在空间四边形 $ABCD$ 中，$E$、$F$ 分别是 $AB$、$AD$ 的中点。求证：$EF\parallel$ 平面 $BCD$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（中位线找平行）：}]
			\[
				EF\parallel BD
			\]
			\par\textbf{理由：} 在 $\triangle ABD$ 中，$E,F$ 分别为 $AB,AD$ 中点，故 $EF\parallel BD$。
		\item[\textbf{第二步（确认条件）：}]
			\[
				BD\subset BCD,\quad EF\not\subset BCD
			\]
			\par\textbf{理由：} $BD$ 在平面 $BCD$ 内，$EF$ 不在平面 $BCD$ 内（显然）。
		\item[\textbf{第三步（套用判定）：}]
			\[
				EF\parallel BD,\ BD\subset BCD,\ EF\not\subset BCD \Rightarrow EF\parallel BCD
			\]
			\par\textbf{理由：} 根据线面平行判定定理，如果平面外一条直线与平面内一条直线平行，则该直线与平面平行。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$EF\parallel$ 平面 $BCD$.}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在正方体 $ABCD-A_1B_1C_1D_1$ 中，$E,F,G$ 分别是棱 $AA_1,A_1B_1,A_1D_1$ 的中点。求证：平面 $EFG\parallel$ 平面 $ABCD$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（中位线得平行）：}]
			\[
				EF\parallel A_1B_1\parallel AB,\quad EG\parallel A_1D_1\parallel AD
			\]
			\par\textbf{理由：} $E,F$ 分别是 $AA_1$、$A_1B_1$ 中点，得 $EF\parallel A_1B_1$，又 $A_1B_1\parallel AB$，故 $EF\parallel AB$；同理 $EG\parallel AD$。
		\item[\textbf{第二步（证线面平行）：}]
			\[
				EF\parallel AB,\ AB\subset ABCD,\ EF\not\subset ABCD \Rightarrow EF\parallel ABCD,
			\]
			\[
				EG\parallel AD,\ AD\subset ABCD,\ EG\not\subset ABCD \Rightarrow EG\parallel ABCD.
			\]
			\par\textbf{理由：} 由线面平行判定定理，$EF\parallel$ 平面 $ABCD$，$EG\parallel$ 平面 $ABCD$。
		\item[\textbf{第三步（证面面平行）：}]
			\[
				EF\parallel ABCD,\ EG\parallel ABCD,\ EF\cap EG=E \Rightarrow EFG\parallel ABCD.
			\]
			\par\textbf{理由：} 根据面面平行判定定理，如果一个平面内的两条相交直线都平行于另一个平面，那么这两个平面平行。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面 $EFG\parallel$ 平面 $ABCD$.}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在四棱锥 $P-ABCD$ 中，底面 $ABCD$ 是平行四边形，$E,F$ 分别是 $AB,PC$ 的中点。求证：$EF\parallel$ 平面 $PAD$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造中位线）：}]
			取 $CD$ 中点 $G$，连接 $EG,FG$，则
			\[
				EG\parallel AD,\quad FG\parallel PD.
			\]
			\par\textbf{理由：} 在平行四边形 $ABCD$ 中，$E,G$ 为 $AB,CD$ 中点，故 $EG\parallel AD$；在 $\triangle PCD$ 中，$F,G$ 为 $PC,CD$ 中点，故 $FG\parallel PD$。
		\item[\textbf{第二步（证面面平行）：}]
			\[
				EG\parallel AD,\ AD\subset PAD,\ EG\not\subset PAD \Rightarrow EG\parallel PAD,
			\]
			\[
				FG\parallel PD,\ PD\subset PAD,\ FG\not\subset PAD \Rightarrow FG\parallel PAD,
			\]
			且 $EG\cap FG=G$，故平面 $EFG\parallel$ 平面 $PAD$。
			\par\textbf{理由：} 由线面平行判定得 $EG\parallel$ 平面 $PAD$，$FG\parallel$ 平面 $PAD$，再由面面平行判定得平面 $EFG\parallel$ 平面 $PAD$。
		\item[\textbf{第三步（面面平行性质）：}]
			\[
				EF\subset EFG,\ EFG\parallel PAD \Rightarrow EF\parallel PAD.
			\]
			\par\textbf{理由：} 根据面面平行的性质定理，如果两个平面平行，那么一个平面内的直线平行于另一个平面。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$EF\parallel$ 平面 $PAD$.}
\end{examplebox}
