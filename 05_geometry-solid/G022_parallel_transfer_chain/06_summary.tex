\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{平行转化链通过判定定理和性质定理实现线线、线面、面面平行的双向转化，其中线面平行是中心枢纽。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（线面平行判定）：} 需要线线平行、线在面外、线在面内。
		\item \textbf{条件 2（面面平行判定）：} 需要面内两条相交直线分别平行于另一平面。
		\item \textbf{条件 3（面面平行性质）：} 需要第三个平面同时截两平行平面。
		\item \textbf{条件 4（线面平行性质）：} 需要过已知直线的平面与已知平面相交。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（判定与性质混淆）：} 判定定理是从线线平行推线面或面面平行；性质定理是从平行推线线平行。注意方向不能反。
		\item \textbf{检查项（条件完整性）：} 使用任何一条定理时，务必检查所有条件（如线在面外、线相交、截面存在），缺一不可。
	\end{itemize}

\end{summarybox}
