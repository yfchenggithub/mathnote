	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 在正方体$ABCD\text{-}A_1B_1C_1D_1$中，求证：平面$A_1BD\perp$平面$ACC_1A_1$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线线垂直到线面）：}]
			因为$AC\perp BD$（正方形对角线垂直），$AA_1\perp$底面$ABCD$，所以$AA_1\perp BD$。$AC$与$AA_1$相交于$A$，所以$BD\perp$平面$ACC_1A_1$。
			\[
				AC\perp BD,\; AA_1\perp BD,\; AC\cap AA_1=A \;\Rightarrow\; BD\perp ACC_1A_1.
			\]
			\par\textbf{理由：}线面垂直判定定理。
		\item[\textbf{第二步（线面垂直到面面）：}]
			因为$BD\subset$平面$A_1BD$，且$BD\perp$平面$ACC_1A_1$，所以平面$A_1BD\perp$平面$ACC_1A_1$。
			\[
				BD\perp ACC_1A_1,\; BD\subset A_1BD \;\Rightarrow\; A_1BD\perp ACC_1A_1.
			\]
			\par\textbf{理由：}面面垂直判定定理。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面$A_1BD\perp$平面$ACC_1A_1$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 在三棱锥$P\text{-}ABC$中，$PA\perp$底面$ABC$，$AB\perp BC$，求证：平面$PAB\perp$平面$PBC$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线面垂直到线线）：}]
			因为$PA\perp$底面$ABC$，$BC\subset$底面$ABC$，所以$PA\perp BC$。
			\[
				PA\perp ABC,\; BC\subset ABC \;\Rightarrow\; PA\perp BC.
			\]
			\par\textbf{理由：}线面垂直性质。
		\item[\textbf{第二步（线线垂直到线面）：}]
			因为$PA\perp BC$（已证），$AB\perp BC$（已知），且$PA,AB\subset$平面$PAB$，$PA\cap AB=A$，所以$BC\perp$平面$PAB$。
			\[
				PA\perp BC,\; AB\perp BC,\; PA,AB\subset PAB,\; PA\cap AB=A \;\Rightarrow\; BC\perp PAB.
			\]
			\par\textbf{理由：}线面垂直判定定理（垂直两条相交直线）。
		\item[\textbf{第三步（线面垂直到面面）：}]
			因为$BC\subset$平面$PBC$，且$BC\perp$平面$PAB$，所以平面$PAB\perp$平面$PBC$。
			\[
				BC\perp PAB,\; BC\subset PBC \;\Rightarrow\; PAB\perp PBC.
			\]
			\par\textbf{理由：}面面垂直判定定理。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面$PAB\perp$平面$PBC$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 在三棱锥$P\text{-}ABC$中，$PA\perp$底面$ABC$，$AB=AC$，$D$是$BC$中点，求证：平面$PBC\perp$平面$PAD$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（线面垂直到线线）：}]
			因为$PA\perp$底面$ABC$，$BC\subset$底面$ABC$，所以$PA\perp BC$。
			\[
				PA\perp ABC,\; BC\subset ABC \;\Rightarrow\; PA\perp BC.
			\]
			\par\textbf{理由：}线面垂直性质。
		\item[\textbf{第二步（线线垂直到线面）：}]
			由$AB=AC$，$D$是$BC$中点得$AD\perp BC$；结合$PA\perp BC$（已证），$PA,AD\subset$平面$PAD$，$PA\cap AD=A$，所以$BC\perp$平面$PAD$。
			\[
				PA\perp BC,\; AD\perp BC,\; PA,AD\subset PAD,\; PA\cap AD=A \;\Rightarrow\; BC\perp PAD.
			\]
			\par\textbf{理由：}线面垂直判定定理；$AD\perp BC$由等腰三角形三线合一得出。
		\item[\textbf{第三步（线面垂直到面面）：}]
			因为$BC\subset$平面$PBC$，且$BC\perp$平面$PAD$，所以平面$PBC\perp$平面$PAD$。
			\[
				BC\perp PAD,\; BC\subset PBC \;\Rightarrow\; PBC\perp PAD.
			\]
			\par\textbf{理由：}面面垂直判定定理。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面$PBC\perp$平面$PAD$。}
