\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两相交线垂直）：}}]
			若直线 $l$ 垂直于平面 $\alpha$ 内的两条相交直线，则 $l\perp\alpha$。
		\item[\textbf{\textcolor{CStatement}{条件2（平面含垂线）：}}]
			若平面 $\beta$ 包含直线 $l$ 且 $l\perp\alpha$，则 $\alpha\perp\beta$。
		\item[\textbf{\textcolor{CStatement}{条件3（交线垂线）：}}]
			若 $\alpha\perp\beta$，交线为 $l$，且直线 $m\subset\alpha$，$m\perp l$，则 $m\perp\beta$。
		\item[\textbf{\textcolor{CStatement}{条件4（线面垂直性质）：}}]
			若直线 $m\perp\beta$，则 $m$ 垂直于 $\beta$ 内任意直线。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（垂直转化链）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			垂直关系可在直线与平面间通过判定和性质双向转化。
			\[
				\boxed{\text{线线垂直}\;\xrightarrow{\text{判定}}\;\text{线面垂直}\;\xrightarrow{\text{判定}}\;\text{面面垂直}}
				\qquad
				\boxed{\text{面面垂直}\;\xrightarrow{\text{性质}}\;\text{线面垂直}\;\xrightarrow{\text{性质}}\;\text{线线垂直}}
			\]
			即
			\[
				\boxed{\text{线线垂直}\;\underset{\text{性质}}{\overset{\text{判定}}{\rightleftarrows}}\;\text{线面垂直}\;\underset{\text{性质}}{\overset{\text{判定}}{\rightleftarrows}}\;\text{面面垂直}}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（线面垂直性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若直线垂直平面，则它垂直该平面内任意直线。
			\[
				l\perp\alpha \Rightarrow l\perp m\ (\forall m\subset\alpha).
			\]
	\end{description}

\end{statementbox}
