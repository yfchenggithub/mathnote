\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知球的半径 $R=3$，求球的体积 $V$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（回忆公式）：}] 球的体积公式为
			\[
				V = \frac{4}{3}\pi R^3.
			\]
			\par\textbf{理由：}
			此公式是标准结论.

		\item[\textbf{第二步（代入数值）：}] 将 $R=3$ 代入公式：
			\[
				V = \frac{4}{3}\pi \cdot 3^3.
			\]
			\par\textbf{理由：}
			直接代入已知半径.

		\item[\textbf{第三步（计算体积）：}] 计算 $3^3=27$，得
			\[
				\begin{aligned}
					V &= \frac{4}{3}\pi \cdot 27 \\
					&= 36\pi
			\end{aligned}
			\]
			\par\textbf{理由：}
			完成乘法运算.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$V=36\pi$.}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知球的体积 $V=36\pi$，求球的半径 $R$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（建立方程）：}] 由体积公式得
			\[
				36\pi = \frac{4}{3}\pi R^3.
			\]
			\par\textbf{理由：}
			代入体积公式.

		\item[\textbf{第二步（化简方程）：}] 两边同除以 $\frac{4}{3}\pi$，得
			\[
				R^3 = 27.
			\]
			\par\textbf{理由：}
			等式性质.

		\item[\textbf{第三步（开立方）：}] 取立方根，得
			\[
				\begin{aligned}
					R &= \sqrt[3]{27} \\
					&= 3
			\end{aligned}
			\]
			\par\textbf{理由：}
			半径应为正数.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=3$.}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	一个球的体积与一个正方体的体积相等，正方体的棱长为 $a$，球的半径为 $R$，求 $a$ 与 $R$ 的比值.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出体积）：}] 球的体积 $V_b = \dfrac{4}{3}\pi R^3$，正方体体积 $V_c = a^3$.
			\[
				V_b = \frac{4}{3}\pi R^3,
				\qquad
				V_c = a^3.
			\]
			\par\textbf{理由：}
			运用球体积公式和正方体体积公式.

		\item[\textbf{第二步（等量关系）：}] 由 $V_b = V_c$，得
			\[
				\frac{4}{3}\pi R^3 = a^3.
			\]
			\par\textbf{理由：}
			体积相等.

		\item[\textbf{第三步（求比值）：}] 改写得 $\dfrac{a^3}{R^3} = \dfrac{4}{3}\pi$，两边开立方：
			\[
				\frac{a}{R} = \sqrt[3]{\frac{4\pi}{3}}.
			\]
			\par\textbf{理由：}
			立方根运算.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\dfrac{a}{R} = \sqrt[3]{\dfrac{4\pi}{3}}$.}
\end{examplebox}
