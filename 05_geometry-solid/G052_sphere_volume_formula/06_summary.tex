\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{球的体积 $V$ 与半径 $R$ 的立方成正比：$V = \frac{4}{3}\pi R^3$.}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（已知半径）：} 知道球的半径 $R$.
		\item \textbf{条件 2（球体假设）：} 物体可近似为球体，且 $R>0$.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（系数与次方）：} 勿将 $\frac{4}{3}$ 写错，勿将 $R^3$ 写成 $R^2$.
		\item \textbf{检查项（单位与反算）：} 计算过程注意单位一致；反求半径时需开立方.
	\end{itemize}
\end{summarybox}
