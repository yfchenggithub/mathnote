\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（系数混淆）}}] 错误将球的体积系数写为 $\frac{3}{4}$ 或 $4$，如 $V = \frac{3}{4}\pi R^3$.
		\item[\textbf{\textcolor{CTrap}{正确理解（标准系数）：}}] 球的体积系数是 $\frac{4}{3}$，来源于积分或几何推导.
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}] 与表面积公式 $4\pi R^2$ 混淆，或记反系数.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（次方混淆）}}] 误将 $R^3$ 写为 $R^2$，即 $V = \frac{4}{3}\pi R^2$.
		\item[\textbf{\textcolor{CTrap}{正确理解（立方关系）：}}] 体积与半径的立方成正比，必须用 $R^3$.
		\item[\textbf{\textcolor{CTrap}{错因分析（面积干扰）：}}] 受圆的面积 $S=\pi R^2$ 或球表面积 $4\pi R^2$ 的影响，习惯性使用平方.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开方类型混淆）}}] 已知体积 $V$ 求半径 $R$ 时，错误地使用开平方：$R = \sqrt{\frac{3V}{4\pi}}$.
		\item[\textbf{\textcolor{CTrap}{正确理解（开立方运算）：}}] 应取立方根：$R = \sqrt[3]{\frac{3V}{4\pi}}$.
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性思维）：}}] 解方程 $R^3 = M$ 时误用平方根，忽略三次方要求开立方.
	\end{description}
\end{trapbox}
