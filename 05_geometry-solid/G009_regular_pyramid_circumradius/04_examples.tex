	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知正四棱锥的高为 $4$，底面正方形边长为 $6$，求该棱锥的外接球半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求外接圆半径）：}]
		      正方形外接圆半径等于对角线的一半：
		      \[
			      r=\frac{\sqrt2}{2}a=\frac{\sqrt2}{2}\times 6=3\sqrt2.
		      \]
		      \par\textbf{理由：} 正方形对角线长为 $\sqrt2a$，一半即外接圆半径。

		\item[\textbf{第二步（套用公式）：}]
		      代入 $h=4,\ r=3\sqrt2$：
		      \[
			      R
			      =
			      \frac{h^2+r^2}{2h}
			      =
			      \frac{4^2+(3\sqrt2)^2}{2\times4}
			      =
			      \frac{16+18}{8}
			      =
			      \frac{34}{8}
			      =
			      \frac{17}{4}.
		      \]
		      \par\textbf{理由：} 直接使用正棱锥外接球半径公式
		      \[
			      R=\frac{h^2+r^2}{2h}.
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $\mathbf{R=\dfrac{17}{4}}$。

	\textbf{\textcolor{CExample}{例 2（反求高，注意双解）}}

	\textbf{\textcolor{CExample}{题目：}} 已知正三棱锥的外接球半径 $R=5$，底面外接圆半径 $r=4$，求棱锥的高 $h$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]

		\item[\textbf{第一步（列出方程）：}]
		      由公式
		      \[
			      R=\frac{h^2+r^2}{2h}
		      \]
		      可得
		      \[
			      2Rh=h^2+r^2.
		      \]
		      整理为关于 $h$ 的一元二次方程：
		      \[
			      h^2-2Rh+r^2=0.
		      \]

		      \par\textbf{理由：} 已知 $R$ 和 $r$，反求 $h$ 时，通常会得到一元二次方程。

		\item[\textbf{第二步（写出一般解并检查条件）：}]
		      由
		      \[
			      h^2-2Rh+r^2=0
		      \]
		      可得
		      \[
			      h=R\pm\sqrt{R^2-r^2}.
		      \]
		      因此必须满足
		      \[
			      R^2-r^2\ge0,
		      \]
		      即
		      \[
			      R\ge r.
		      \]
		      本题中
		      \[
			      R=5>4=r,
		      \]
		      所以存在两个正高度。

		\item[\textbf{第三步（代入数值）：}]
		      \[
			      h=5\pm\sqrt{5^2-4^2}
			      =5\pm\sqrt{25-16}
			      =5\pm3.
		      \]
		      因此
		      \[
			      h=2
		      \]
		      或
		      \[
			      h=8.
		      \]

		      \par\textbf{理由：} 两个解都为正，且都满足几何存在条件。

		\item[\textbf{第四步（理解两个高度的差异）：}]
		      球心高度为
		      \[
			      z=\frac{h^2-r^2}{2h}.
		      \]
		      当 $h=2$ 时，
		      \[
			      z=\frac{2^2-4^2}{2\times2}
			      =
			      \frac{4-16}{4}
			      =-3,
		      \]
		      球心在底面下方。

		      当 $h=8$ 时，
		      \[
			      z=\frac{8^2-4^2}{2\times8}
			      =
			      \frac{64-16}{16}
			      =3,
		      \]
		      球心在底面上方。

		      \par\textbf{理由：} 同一个外接球半径和同一个底面外接圆半径，可能对应两个不同高度的正棱锥。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $\mathbf{h=2}$ 或 $\mathbf{h=8}$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知正六棱锥的高 $h=8$，侧棱长 $l=10$，求外接球半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（检查侧棱关系）：}]
		      正棱锥中，侧棱 $l$、高 $h$、底面外接圆半径 $r$ 满足
		      \[
			      l^2=h^2+r^2.
		      \]
		      本题中
		      \[
			      l=10>8=h,
		      \]
		      满足非退化条件。

		\item[\textbf{第二步（求底面外接圆半径）：}]
		      \[
			      r^2=l^2-h^2=10^2-8^2=100-64=36,
		      \]
		      所以
		      \[
			      r=6.
		      \]
		      \par\textbf{理由：} 顶点、底面外心、底面顶点构成直角三角形。

		\item[\textbf{第三步（套用公式）：}]
		      代入 $h=8,\ r=6$：
		      \[
			      R
			      =
			      \frac{h^2+r^2}{2h}
			      =
			      \frac{64+36}{16}
			      =
			      \frac{100}{16}
			      =
			      \frac{25}{4}.
		      \]

		      \par\textbf{理由：} 使用正棱锥外接球半径公式。

		\item[\textbf{另解（直接用侧棱形式）：}]
		      因为
		      \[
			      R=\frac{l^2}{2h},
		      \]
		      所以
		      \[
			      R=\frac{10^2}{2\times8}
			      =
			      \frac{100}{16}
			      =
			      \frac{25}{4}.
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} $\mathbf{R=\dfrac{25}{4}}$。
