\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	正棱锥的外接球半径 $R$ 等于高 $h$ 与底面外接圆半径 $r$ 的平方和除以 $2h$，即
	\[
		R=\frac{h^2+r^2}{2h}.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{形状条件：} 棱锥为正棱锥，即底面为正多边形，顶点在底面外心正上方。
		\item \textbf{参数条件：} 高 $h>0$，底面外接圆半径 $r>0$。
		\item \textbf{推广条件：} 只要底面顶点共圆，且顶点投影为底面外接圆圆心，公式同样成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{常用变形}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{反求 $r$：}
		      \[
			      r^2=2Rh-h^2,\qquad r=\sqrt{2Rh-h^2}.
		      \]
		      需满足
		      \[
			      2Rh-h^2\ge0.
		      \]
		      若底面非退化，则需
		      \[
			      2Rh-h^2>0.
		      \]

		\item \textbf{反求 $h$：}
		      \[
			      h^2-2Rh+r^2=0,
		      \]
		      所以
		      \[
			      h=R\pm\sqrt{R^2-r^2}.
		      \]
		      需满足
		      \[
			      R\ge r.
		      \]
		      当 $R>r$ 时，通常有两个正高度。

		\item \textbf{侧棱形式：} 若侧棱长为 $l$，则
		      \[
			      l^2=h^2+r^2,
		      \]
		      因此
		      \[
			      R=\frac{l^2}{2h}.
		      \]
		      非退化情形下应有
		      \[
			      l>h>0.
		      \]
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{公式记忆：} 分母是 $2h$，不是 $h$，也不是 $2$。
		\item \textbf{参数识别：} $r$ 是底面外接圆半径，不一定是底面边长。
		\item \textbf{球心位置：} 球心一定在高所在直线上，但不一定在棱锥内部。其高度为
		      \[
			      z=\frac{h^2-r^2}{2h}.
		      \]
		      当 $h<r$ 时，球心在底面下方。
		\item \textbf{适用范围：} 不能把公式随意用于一般斜棱锥；必须保证顶点投影在底面外接圆圆心处。
	\end{itemize}

\end{summarybox}
