\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	利用正棱锥的几何对称性，将外接球球心放在高所在直线上。再建立空间直角坐标系，用“球心到顶点距离等于球心到底面顶点距离”列方程求解。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]

		\item[\textbf{步骤一（建坐标系）：}]
		      设底面外心为 $C$。以 $C$ 为原点，底面所在平面为 $xy$ 平面，$z$ 轴垂直底面向上。

		      因为正棱锥顶点在底面外心正上方，所以顶点 $S$ 的坐标为
		      \[
			      S(0,0,h).
		      \]
		      取底面外接圆上的一个顶点 $P$，不妨设
		      \[
			      P(r,0,0).
		      \]

		      \par\textbf{理由：} 底面各顶点在同一个外接圆上，外接圆半径为 $r$；正棱锥的高经过底面外心。

		\item[\textbf{步骤二（设球心坐标）：}]
		      设外接球球心为 $O$。由于 $O$ 到底面各顶点距离相等，所以 $O$ 在过底面外心 $C$ 且垂直于底面的直线上。于是可设
		      \[
			      O(0,0,z).
		      \]

		      \par\textbf{理由：} 空间中到同一圆上各顶点距离相等的点，在该圆圆心的垂线上；对正棱锥而言，这条垂线就是棱锥高所在直线。

		\item[\textbf{步骤三（列距离方程）：}]
		      外接球经过顶点 $S$ 和底面顶点 $P$，因此
		      \[
			      OS=OP.
		      \]
		      由两点间距离公式得
		      \[
			      |z-h|=\sqrt{r^2+z^2}.
		      \]

		      \par\textbf{理由：} 外接球球心到所有顶点的距离都等于外接球半径。

		\item[\textbf{步骤四（平方化简）：}]
		      两边平方并展开：
		      \[
			      (z-h)^2=r^2+z^2.
		      \]
		      即
		      \[
			      z^2-2hz+h^2=r^2+z^2.
		      \]

		      \par\textbf{理由：} 两边均为非负量，平方不会改变等价性。

		\item[\textbf{步骤五（解出球心高度 $z$）：}]
		      消去 $z^2$，得
		      \[
			      -2hz+h^2=r^2.
		      \]
		      移项得
		      \[
			      2hz=h^2-r^2.
		      \]
		      因为 $h>0$，所以
		      \[
			      z=\frac{h^2-r^2}{2h}.
		      \]

		      \par\textbf{理由：} 这是关于 $z$ 的一次方程。注意这里要求棱锥高 $h>0$。

		\item[\textbf{步骤六（求外接球半径 $R$）：}]
		      外接球半径
		      \[
			      R=OS=|z-h|.
		      \]
		      代入
		      \[
			      z=\frac{h^2-r^2}{2h}
		      \]
		      可得
		      \[
			      R
			      =
			      \left|\frac{h^2-r^2}{2h}-h\right|
			      =
			      \left|\frac{h^2-r^2-2h^2}{2h}\right|
			      =
			      \left|\frac{-h^2-r^2}{2h}\right|.
		      \]
		      由于
		      \[
			      h>0,\qquad h^2+r^2>0,
		      \]
		      所以
		      \[
			      R=\frac{h^2+r^2}{2h}.
		      \]

		      \par\textbf{理由：} 外接球半径必须为正数，故取绝对值后的正值。

	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{球心位置补充}}

	由
	\[
		z=\frac{h^2-r^2}{2h}
	\]
	可知：

	\[
		\begin{array}{c|c}
			\text{条件} & \text{球心位置}     \\ \hline
			h>r       & \text{球心在底面上方}  \\[3pt]
			h=r       & \text{球心在底面外心处} \\[3pt]
			h<r       & \text{球心在底面下方}
		\end{array}
	\]

	因此，球心一定在高所在直线上，但不一定在棱锥内部。

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	因此，正棱锥的外接球半径公式为
	\[
		R=\frac{h^2+r^2}{2h},
	\]
	其中 $h$ 为棱锥的高，$r$ 为底面外接圆半径。

	若侧棱长为 $l$，则
	\[
		l^2=h^2+r^2,
	\]
	所以也可写成
	\[
		R=\frac{l^2}{2h}.
	\]
	非退化情形下有
	\[
		h>0,\qquad r>0,\qquad l>h.
	\]

\end{proofbox}
