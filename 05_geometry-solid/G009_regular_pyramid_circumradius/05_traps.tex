\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆内切球）：}}] 认为外接球半径与内切球半径相同，或使用内切球公式计算。
		\item[\textbf{\textcolor{CTrap}{正确理解（内外区分）：}}] 外接球球心到各顶点距离相等；内切球球心到各面距离相等。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）：}}] 混淆“外接”与“内切”的几何意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（漏掉分母 $2$）：}}] 直接写成
		      \[
			      R=\frac{h^2+r^2}{h},
		      \]
		      忘记分母中的 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式结构）：}}] 正确公式为
		      \[
			      R=\frac{h^2+r^2}{2h},
		      \]
		      分母是 $2h$。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导错误）：}}] 在由
		      \[
			      2hR=h^2+r^2
		      \]
		      解出 $R$ 时，遗漏了除以 $2h$ 的步骤。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（条件误用）：}}] 将公式直接用于一般斜棱锥，或用于顶点投影不在底面外心的棱锥。
		\item[\textbf{\textcolor{CTrap}{正确理解（适用条件）：}}] 本公式对正棱锥一定成立。更一般地，只要底面顶点共圆，且顶点投影为底面外接圆圆心，公式也成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏对称）：}}] 若顶点投影不在底面外心，外接球球心一般不在棱锥高所在直线上，不能直接套用
		      \[
			      R=\frac{h^2+r^2}{2h}.
		      \]
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（把底面边长当成 $r$）：}}] 看到底面边长为 $a$，就直接令 $r=a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（先求外接圆半径）：}}] 公式中的 $r$ 是底面外接圆半径，不一定等于底面边长。
		      \[
			      \begin{array}{c|c}
				      \text{底面图形}      & \text{外接圆半径 } r    \\ \hline
				      \text{正三角形边长 } a & \dfrac{\sqrt3}{3}a \\[6pt]
				      \text{正方形边长 } a  & \dfrac{\sqrt2}{2}a \\[6pt]
				      \text{正六边形边长 } a & a
			      \end{array}
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（参数认错）：}}] 题目给的往往是底面边长，而公式需要的是底面外接圆半径。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点五（反求参数不检查存在条件）：}}] 已知 $R$ 和 $h$ 时，直接写
		      \[
			      r=\sqrt{2Rh-h^2},
		      \]
		      却不检查根号内是否非负。
		\item[\textbf{\textcolor{CTrap}{正确理解（存在条件）：}}] 必须满足
		      \[
			      2Rh-h^2\ge0.
		      \]
		      若底面非退化，则应有
		      \[
			      2Rh-h^2>0.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视几何意义）：}}] $r$ 表示底面外接圆半径，不能是虚数；非退化底面还要求 $r>0$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点六（以为球心一定在棱锥内部）：}}] 看到“球心在高线上”，就误以为球心一定在顶点和底面之间。
		\item[\textbf{\textcolor{CTrap}{正确理解（位置分情况）：}}] 球心高度为
		      \[
			      z=\frac{h^2-r^2}{2h}.
		      \]
		      当 $h<r$ 时，$z<0$，球心在底面下方；当 $h=r$ 时，球心在底面外心处；当 $h>r$ 时，球心在底面上方。
		\item[\textbf{\textcolor{CTrap}{错因分析（把直线当线段）：}}] “球心在高所在直线上”说的是一条直线，不是说一定在线段内部。
	\end{description}

\end{trapbox}
