\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（几何特征）：}}] 正棱锥的顶点在底面外心的正上方，即高线经过底面外心。
		\item[\textbf{\textcolor{CStatement}{条件 2（已知参数）：}}] 已知棱锥的高 $h$ 和底面外接圆半径 $r$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}外接球球心在棱锥高所在直线上，由距离相等列方程可得。
		      \[
			      R = \frac{h^2 + r^2}{2h}.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（侧棱表示）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}若侧棱长为 $l$，则
		      \[
			      l^2 = h^2 + r^2,
		      \]
		      代入上式可得
		      \[
			      R = \frac{l^2}{2h}.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{参数限制：}}
	\[
		h>0,\qquad r>0.
	\]
	若使用侧棱形式，则非退化情形下还需满足
	\[
		l^2=h^2+r^2,\qquad l>h>0.
	\]

	\medskip
	\textbf{\textcolor{CStatement}{适用范围补充：}}

	本结论对正棱锥一定成立。事实上，只要底面顶点共圆，且顶点投影为底面外接圆圆心，公式同样成立；不能直接套用于一般斜棱锥。

\end{statementbox}
