\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（球内接几何体）：}}] 几何体的所有顶点均位于同一球面上（即存在外接球），球心未指定但存在。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正方体）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正方体外接球直径等于棱长的 $\sqrt{3}$ 倍。
			\[
				2R = a\sqrt{3}
			\]
			(其中 $a$ 为正方体棱长)

		\item[\textbf{\textcolor{CStatement}{结论二（长方体）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			长方体外接球直径等于体对角线长度。
			\[
				2R = \sqrt{a^{2} + b^{2} + c^{2}}
			\]
			(其中 $a,b,c$ 分别为长方体的长、宽、高)

		\item[\textbf{\textcolor{CStatement}{推广结论（正四面体）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正四面体外接球直径与棱长之比为确定常数。
			\[
				2R = \frac{\sqrt{6}}{2}a
			\]
			(其中 $a$ 为正四面体棱长)
	\end{description}
\end{statementbox}
