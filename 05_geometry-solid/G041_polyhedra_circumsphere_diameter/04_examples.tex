\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知正方体棱长 $a=2$，求该正方体外接球的直径 $2R$ 和球体积 $V$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用正方体公式）：}]
			正方体外接球直径公式：$2R = a\sqrt{3}$。代入 $a=2$，得
			\[
				2R = 2\sqrt{3}.
			\]
			\par\textbf{理由：} 直接使用结论。

		\item[\textbf{第二步（求半径）：}]
			由 $2R = 2\sqrt{3}$，得 $R = \sqrt{3}$。
			\par\textbf{理由：} 简单除法。

		\item[\textbf{第三步（求球体积）：}]
			球体积公式 $V = \frac{4}{3}\pi R^3$，代入 $R = \sqrt{3}$：
			\[
				V = \frac{4}{3}\pi (\sqrt{3})^3 = \frac{4}{3}\pi \cdot 3\sqrt{3} = 4\sqrt{3}\pi.
			\]
			\par\textbf{理由：} 体积公式直接计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$2R = 2\sqrt{3}$，$V = 4\sqrt{3}\pi$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知长方体长、宽、高分别为 $3,\ 4,\ 5$，求其外接球半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求体对角线）：}]
			长方体体对角线长 $l = \sqrt{3^2 + 4^2 + 5^2} = \sqrt{9+16+25} = \sqrt{50} = 5\sqrt{2}$。
			\par\textbf{理由：} 长方体体对角线公式 $l = \sqrt{a^2+b^2+c^2}$。

		\item[\textbf{第二步（套用长方体公式）：}]
			外接球直径 $2R = l$，所以 $2R = 5\sqrt{2}$，从而
			\[
				R = \frac{5\sqrt{2}}{2}.
			\]
			\par\textbf{理由：} 长方体结论直接使用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$R = \dfrac{5\sqrt{2}}{2}$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	同一球内接一个正方体和一个正四面体（所有顶点均在球面上）。若正方体棱长为 $a$，正四面体棱长为 $b$，求 $a$ 与 $b$ 之间的比例关系。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（正方体关系）：}]
			对于内接正方体，有 $2R = a\sqrt{3}$。
			\par\textbf{理由：} 正方体结论。

		\item[\textbf{第二步（正四面体关系）：}]
			对于内接正四面体，有 $2R = \frac{\sqrt{6}}{2}b$。
			\par\textbf{理由：} 正四面体结论。

		\item[\textbf{第三步（等量代换）：}]
			球是同一个，所以直径相等：
			\[
				a\sqrt{3} = \frac{\sqrt{6}}{2}b.
			\]
			解出比例：
			\[
				\frac{a}{b} = \frac{\sqrt{6}}{2\sqrt{3}} = \frac{\sqrt{2}}{2}
				\quad\text{或}\quad
				a = \frac{\sqrt{2}}{2}b.
			\]
			\par\textbf{理由：} 等式两边同除以 $b\sqrt{3}$ 并化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$a = \dfrac{\sqrt{2}}{2}b$。
\end{examplebox}
