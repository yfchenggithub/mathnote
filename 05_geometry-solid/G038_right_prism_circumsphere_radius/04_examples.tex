\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	一个圆柱的底面半径为 $3$，高为 $8$，求其外接球半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 圆柱底面半径 $r=3$，高 $h=8$，满足直柱体条件。
			\par\textbf{理由：} 圆柱母线垂直于底面，且底面是圆，符合公式使用条件。
		\item[\textbf{第二步（套用公式）：}]
			\[
				\begin{aligned}
					R^2 &= r^2 + \left(\frac{h}{2}\right)^2 \\
					&= 3^2 + \left(\frac{8}{2}\right)^2 \\
					&= 9 + 16 \\
					&= 25.
			\end{aligned}
			\]
			\par\textbf{理由：} 直圆柱外接球半径公式。
		\item[\textbf{第三步（求半径）：}]
			\[
				\begin{aligned}
					R &= \sqrt{25} \\
					&= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 开平方取正值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	外接球半径为 $\mathbf{5}$.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	一个直三棱柱的底面为直角三角形，两直角边长分别为 $3$ 和 $4$，高为 $10$，求其外接球半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面外接圆半径）：}] 由直角三角形性质，斜边长 $c = \sqrt{3^2+4^2}$，故 $c=5$；外接圆半径 $r = \frac{c}{2}$，即 $r = 2.5$.
			\par\textbf{理由：} 直角三角形斜边即为外接圆直径，所以 $r = \frac{\text{斜边}}{2}$.
		\item[\textbf{第二步（套用公式）：}]
			\[
				\begin{aligned}
					R^2 &= r^2 + \left(\frac{h}{2}\right)^2 \\
					&= (2.5)^2 + \left(\frac{10}{2}\right)^2 \\
					&= 6.25 + 25 \\
					&= 31.25 \\
					&= \frac{125}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 直棱柱外接球半径公式。
		\item[\textbf{第三步（求半径）：}]
			\[
				\begin{aligned}
					R &= \sqrt{\frac{125}{4}} \\
					&= \frac{\sqrt{125}}{2} \\
					&= \frac{5\sqrt{5}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 化简根式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	外接球半径为 $\mathbf{\dfrac{5\sqrt{5}}{2}}$.

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	一个直三棱柱的底面是正三角形，边长为 $6$，高为 $10$，求其外接球半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求底面外接圆半径）：}] 正三角形外接圆半径 $r = \frac{a}{\sqrt{3}}$，代入 $a=6$ 得 $r = \frac{6}{\sqrt{3}}$，化简得 $r = 2\sqrt{3}$.
			\par\textbf{理由：} 正三角形外接圆半径公式 $r = \frac{a}{\sqrt{3}}$（或 $r = \frac{a}{2\sin 60^\circ}$）.
		\item[\textbf{第二步（套用公式）：}]
			\[
				\begin{aligned}
					R^2 &= r^2 + \left(\frac{h}{2}\right)^2 \\
					&= (2\sqrt{3})^2 + \left(\frac{10}{2}\right)^2 \\
					&= 12 + 25 \\
					&= 37.
			\end{aligned}
			\]
			\par\textbf{理由：} 直棱柱外接球半径公式。
		\item[\textbf{第三步（求半径）：}] $R = \sqrt{37}$.
			\par\textbf{理由：} 开平方得最简根式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	外接球半径为 $\mathbf{\sqrt{37}}$.
\end{examplebox}
