\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（底面外接圆）：}}] 几何体为直棱柱或圆柱，底面存在外接圆，半径记为 $r$。
		\item[\textbf{\textcolor{CStatement}{条件2（侧棱垂直底面）：}}] 所有侧棱（或母线）均垂直于底面，高记为 $h$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外接球半径等于底面外接圆半径和半高的平方和开方。
			\[
				R^2 = r^2 + \left(\frac{h}{2}\right)^2.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（长方体特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			长方体对角线的一半即为其外接球半径。
			\[
				R^2 = \frac{a^2+b^2+c^2}{4}, \quad \text{其中 } a,b,c \text{ 为长方体的长,宽,高}.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{适用范围：}}
	适用于所有底面有外接圆的直棱柱（含长方体、正方体）和圆柱；对于圆柱，底面圆半径 $r$ 即为外接圆半径。
\end{statementbox}
