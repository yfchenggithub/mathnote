\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（法向量与角）：}}]
		      已知平面 $\alpha,\beta$ 的法向量分别为 $\vec{n}_1,\vec{n}_2$，$\theta$ 为二面角的平面角，且 $0\le\theta\le\pi$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（分段公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      二面角的余弦值等于法向量夹角余弦的绝对值，符号由二面角锐钝决定。
		      \[
			      \cos\theta =
			      \begin{cases}
				      \bigl|\cos\langle\vec{n}_1,\vec{n}_2\rangle\bigr|
				      = \dfrac{|\vec{n}_1\cdot\vec{n}_2|}{|\vec{n}_1|\,|\vec{n}_2|},  & \text{若二面角为锐角或直角}, \\[0.8em]
				      -\bigl|\cos\langle\vec{n}_1,\vec{n}_2\rangle\bigr|
				      = -\dfrac{|\vec{n}_1\cdot\vec{n}_2|}{|\vec{n}_1|\,|\vec{n}_2|}, & \text{若二面角为钝角}.
			      \end{cases}
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（绝对值式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      无论二面角是锐角还是钝角，其余弦的绝对值恒等于法向量夹角余弦的绝对值。
		      \[
			      |\cos\theta|
			      = \bigl|\cos\langle\vec{n}_1,\vec{n}_2\rangle\bigr|
			      = \dfrac{|\vec{n}_1\cdot\vec{n}_2|}{|\vec{n}_1|\,|\vec{n}_2|}.
		      \]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒：}}
	二面角的锐钝需结合几何图形或题目条件判断，不能仅由任意选取的法向量内积正负直接确定 $\cos\theta$ 的符号。
\end{statementbox}
