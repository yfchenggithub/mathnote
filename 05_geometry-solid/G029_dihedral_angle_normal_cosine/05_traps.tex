\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略绝对值）}}]
			不计算绝对值，直接使用 $\dfrac{\vec{n}_1\cdot\vec{n}_2}{|\vec{n}_1||\vec{n}_2|}$ 作为 $\cos\theta$。
		\item[\textbf{\textcolor{CTrap}{正确理解（先绝对后符号）：}}]
			二面角 $\theta$ 的余弦绝对值等于法向量夹角余弦的绝对值，符号需单独判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（关系混淆）：}}]
			误认为法向量夹角即等于二面角，没有考虑互补情形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忘记符号判定）}}]
			算出 $|\cos\theta|$ 后直接作为答案，没有根据二面角锐钝确定正负。
		\item[\textbf{\textcolor{CTrap}{正确理解（必检锐钝）：}}]
			算完绝对值后，必须结合图形或已知条件判断 $\theta$ 是锐角（$\cos\theta>0$）还是钝角（$\cos\theta<0$），再定符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
			习惯于算夹角就取正值，忽略了二面角范围包含钝角。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号与内积混淆）}}]
			认为 $\vec{n}_1\cdot\vec{n}_2>0$ 时二面角为锐角，$\vec{n}_1\cdot\vec{n}_2<0$ 时二面角为钝角。
		\item[\textbf{\textcolor{CTrap}{正确理解（图形判定）：}}]
			内积的正负取决于法向量的选取方向，不直接对应二面角锐钝；只能通过绝对值公式得到 $|\cos\theta|$，锐钝需借图形判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（方向忽视）：}}]
			忽略了法向量方向可反向，导致内积符号与二面角锐钝关系不确定。
	\end{description}

\end{trapbox}
