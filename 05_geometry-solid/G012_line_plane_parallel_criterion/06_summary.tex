\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{平面外的一条直线，如果平行于平面内的一条直线，那么该直线与这个平面平行。}

	\medskip
	\textbf{\textcolor{CSummary}{标准判定格式}}

	\[
		\left.
		\begin{aligned}
			b & \subset \alpha,     \\
			a & \not\subset \alpha, \\
			a & \parallel b
		\end{aligned}
		\right\}
		\quad\Longrightarrow\quad
		a\parallel\alpha
	\]

	\textbf{\textcolor{CSummary}{三个必要条件}}

	\begin{enumerate}[leftmargin=*, itemsep=0.4em, topsep=0.35em]
		\item 参照直线在目标平面内：
		      \[
			      b\subset\alpha;
		      \]

		\item 目标直线不在目标平面内：
		      \[
			      a\not\subset\alpha;
		      \]

		\item 目标直线与参照直线平行：
		      \[
			      a\parallel b.
		      \]
	\end{enumerate}

	\medskip
	\textbf{\textcolor{CSummary}{证明题三步法}}

	\begin{enumerate}[leftmargin=*, itemsep=0.4em, topsep=0.35em]
		\item 在目标平面内找一条参照直线 $b$；
		\item 证明目标直线 $a\parallel b$；
		\item 说明 $a\not\subset\alpha$，然后引用判定定理。
	\end{enumerate}

	\medskip
	\textbf{\textcolor{CSummary}{最容易漏掉的条件}}

	只证明
	\[
		a\parallel b
		\qquad\text{和}\qquad
		b\subset\alpha
	\]
	还不够。

	必须再确认
	\[
		a\not\subset\alpha.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{记忆口诀}}

	\[
		\boxed{\text{面内找平行，面外验不含；三条件齐全，线面才平行。}}
	\]
\end{summarybox}
