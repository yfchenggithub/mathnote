\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{平面外一条直线如果平行于平面内的一条直线，那么该直线与平面平行。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（线面位置）：} 直线$a$不在平面$\alpha$内（$a\not\subset\alpha$）。
		\item \textbf{条件2（面内直线）：} 直线$b$在平面$\alpha$内（$b\subset\alpha$）。
		\item \textbf{条件3（两线平行）：} $a\parallel b$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件完整）：} 三个条件缺一不可，尤其不能忘$a\not\subset\alpha$。
		\item \textbf{检查项（验证平行）：} 确认$a$与$b$的确平行（同一平面内），且$b$确实在$\alpha$内。
	\end{itemize}
\end{summarybox}
