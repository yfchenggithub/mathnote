\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（线在面内）：}}] 直线$b$在平面$\alpha$内，即$b \subset \alpha$。
		\item[\textbf{\textcolor{CStatement}{条件2（线不在面内）：}}] 直线$a$不在平面$\alpha$内，即$a \not\subset \alpha$。
		\item[\textbf{\textcolor{CStatement}{条件3（两线平行）：}}] $a \parallel b$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（判定定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直线$a$与平面$\alpha$内一条直线$b$平行，且$a$在$\alpha$外部，则$a$不会与$\alpha$相交。
			\[
				a \parallel \alpha
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（符号表达）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用符号简洁表示判定定理的条件与结论。
			\[
				a \not\subset \alpha,\; b \subset \alpha,\; a \parallel b \;\Longrightarrow\; a \parallel \alpha
			\]
	\end{description}
\end{statementbox}
