\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在空间四边形$ABCD$中，$E,F$分别是$AB,AD$的中点。求证：$EF\parallel$平面$BCD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（连接$BD$）：}] 连接$BD$。
			\par\textbf{理由：}
			构造平面$BCD$内的一条直线$BD$作为参照。
		\item[\textbf{第二步（证明$EF\parallel BD$）：}] 在$\triangle ABD$中，$E,F$分别为$AB,AD$中点，由中位线定理得$EF\parallel BD$。
			\par\textbf{理由：}
			三角形的中位线平行于第三边。
		\item[\textbf{第三步（判定条件检验）：}] $EF$不在平面$BCD$内（否则$A$在平面$BCD$内，与空间四边形矛盾），$BD\subset$平面$BCD$，$EF\parallel BD$，所以$EF\parallel$平面$BCD$。
			\par\textbf{理由：}
			满足线面平行的判定定理：平面外一条直线平行于平面内一条直线。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$EF\parallel$平面$BCD$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，在四棱锥$P-ABCD$中，底面$ABCD$为平行四边形，$E$为$PC$中点。求证：$PA\parallel$平面$BDE$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（连接$AC$交$BD$于$O$）：}] 连接$AC$交$BD$于点$O$，则$O$为$AC$中点。
			\par\textbf{理由：}
			平行四边形对角线互相平分。
		\item[\textbf{第二步（连接$EO$，证明$EO\parallel PA$）：}] 在$\triangle PAC$中，$E$为$PC$中点，$O$为$AC$中点，所以$EO\parallel PA$。
			\par\textbf{理由：}
			三角形的中位线平行于第三边。
		\item[\textbf{第三步（判定条件检验）：}] $PA$不在平面$BDE$内（因为$A$不在平面$BDE$内），$EO\subset$平面$BDE$，$PA\parallel EO$，所以$PA\parallel$平面$BDE$。
			\par\textbf{理由：}
			满足线面平行的判定定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA\parallel$平面$BDE$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，在直三棱柱$ABC-A_1B_1C_1$中，$D$为$BC$中点。求证：$A_1B\parallel$平面$ADC_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（连接$A_1C$交$AC_1$于$O$）：}] 连接$A_1C$，因为$ACC_1A_1$为矩形，所以$A_1C$与$AC_1$互相平分，记$O$为$A_1C$中点（同时也是$AC_1$中点）。
			\par\textbf{理由：}
			直三棱柱侧面为矩形，对角线互相平分。
		\item[\textbf{第二步（连接$OD$，证明$OD\parallel A_1B$）：}] 在$\triangle A_1BC$中，$D$为$BC$中点，$O$为$A_1C$中点，所以$OD\parallel A_1B$（中位线）。
			\par\textbf{理由：}
			三角形的中位线平行于第三边。
		\item[\textbf{第三步（判定条件检验）：}] $A_1B$不在平面$ADC_1$内（因为$A_1$不在平面$ADC_1$内），$OD\subset$平面$ADC_1$，$A_1B\parallel OD$，所以$A_1B\parallel$平面$ADC_1$。
			\par\textbf{理由：}
			满足线面平行的判定定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$A_1B\parallel$平面$ADC_1$。}
\end{examplebox}
