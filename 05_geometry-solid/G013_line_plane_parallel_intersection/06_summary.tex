\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{线面平行 $\Rightarrow$ 线线平行（与过该直线的平面交线平行）。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（线面平行）：} $a\parallel\alpha$.
		\item \textbf{条件 2（线在面内）：} $a\subset\beta$.
		\item \textbf{条件 3（两平面相交）：} $\alpha\cap\beta=l$.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（必须同时满足三条件）：} 缺一不可，尤其是“直线在平面内”容易遗漏。
		\item \textbf{检查项（交线与直线共面）：} 应用时检查 $a$ 和 $l$ 是否同在 $\beta$ 内（已经保证），以及 $\alpha\cap\beta=l$ 确实为一条直线。
	\end{itemize}
\end{summarybox}
