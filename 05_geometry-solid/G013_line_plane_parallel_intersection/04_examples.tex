\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在长方体 $ABCD-A_1B_1C_1D_1$ 中，$E$ 为 $AB$ 中点，$F$ 为 $A_1B_1$ 中点。已知 $EF\parallel$ 平面 $BCC_1B_1$，$EF\subset$ 平面 $ABB_1A_1$，平面 $ABB_1A_1\cap$ 平面 $BCC_1B_1=BB_1$。求证：$EF\parallel BB_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件符号化）：}] 令 $\alpha$ 表示平面 $BCC_1B_1$，$\beta$ 表示平面 $ABB_1A_1$，$l=BB_1$。由题意得
			\[
				EF\parallel\alpha,\quad EF\subset\beta,\quad \alpha\cap\beta=l.
			\]
			\par\textbf{理由：}
			将文字条件转化为符号，正好是定理的三条前提。

		\item[\textbf{第二步（套用定理）：}] 由线面平行性质定理，若 $a\parallel\alpha$，$a\subset\beta$，$\alpha\cap\beta=l$，则 $a\parallel l$。应用至 $a=EF$，得
			\[
				\Rightarrow EF\parallel l.
			\]
			\par\textbf{理由：}
			直接代入定理，前提满足，结论成立。

		\item[\textbf{第三步（回代结论）：}] 将 $l=BB_1$ 代回，即得
			\[
				EF\parallel BB_1.
			\]
			\par\textbf{理由：}
			结论中交线即 $BB_1$，故 $EF\parallel BB_1$.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$EF\parallel BB_1$}.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在四棱锥 $P-ABCD$ 中，底面 $ABCD$ 是正方形，$O$ 为 $AC$ 与 $BD$ 的交点，$E$ 为 $PC$ 中点，$PA\perp$ 底面 $ABCD$。已知 $PA\parallel$ 平面 $EBD$，$PA\subset$ 平面 $PAC$，平面 $PAC\cap$ 平面 $EBD=EO$。求证：（1）$PA\parallel EO$；（2）$EO\perp$ 平面 $ABCD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件符号化）：}] 设 $\alpha$ 为平面 $EBD$，$\beta$ 为平面 $PAC$，$l=EO$，$\gamma$ 为底面 $ABCD$。则已知条件可写为
			\[
				PA\parallel\alpha,\quad PA\subset\beta,\quad \alpha\cap\beta=l,\quad PA\perp\gamma.
			\]
			\par\textbf{理由：}
			将文字语言翻译为符号语言，便于使用定理。

		\item[\textbf{第二步（证第 (1) 问）：}] 由线面平行性质定理，由 $PA\parallel\alpha,\ PA\subset\beta,\ \alpha\cap\beta=l$ 得
			\[
				\Rightarrow PA\parallel l.
			\]
			\par\textbf{理由：}
			定理条件齐全，直接推出 $PA\parallel l$，即 $PA\parallel EO$。

		\item[\textbf{第三步（证第 (2) 问）：}] 由 $PA\perp\gamma$ 且 $PA\parallel l$，根据“若一条直线垂直于平面，则其平行线也垂直于该平面”得
			\[
				PA\perp\gamma,\ PA\parallel l \ \Rightarrow\ l\perp\gamma.
			\]
			\par\textbf{理由：}
			线面垂直的平行传递性，所以 $l\perp\gamma$，即 $EO\perp$ 底面 $ABCD$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{(1) $PA\parallel EO$; (2) $EO\perp$ 平面 $ABCD$}.
