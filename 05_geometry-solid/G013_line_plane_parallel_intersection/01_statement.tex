\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（线面平行）：}}]
			直线 $a\parallel$ 平面 $\alpha$.
		\item[\textbf{\textcolor{CStatement}{条件2（线在面内）：}}]
			直线 $a\subset$ 平面 $\beta$.
		\item[\textbf{\textcolor{CStatement}{条件3（两面相交）：}}]
			平面 $\alpha\cap\beta=l$.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线线平行）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过平行线的平面与已知平面的交线与原直线平行。
			\[
				\left.
				\begin{array}
					{c} a\parallel\alpha \\ a\subset\beta \\ \alpha\cap\beta=l
			\end{array}
				\right\} \Rightarrow a\parallel l.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（线面平行判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若直线与两平面交线平行且不在已知平面内，则线面平行。
			\[
				\left.
				\begin{array}
					{c} a\subset\beta \\ \alpha\cap\beta=l \\ a\parallel l \\ a\not\subset\alpha
			\end{array}
				\right\} \Rightarrow a\parallel\alpha.
			\]
	\end{description}

\end{statementbox}
