\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知直线$l$的方向向量$\vec{l}=(1,2,1)$，平面$\alpha$的法向量$\vec{n}=(0,1,0)$。求直线$l$与平面$\alpha$所成角$\theta$的正弦值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（计算点积）：}]
			计算$\vec{l}\cdot\vec{n}=1\times0+2\times1+1\times0=2$。\n
			\par\textbf{理由：}\n 点积公式$\vec{l}\cdot\vec{n}=l_1n_1+l_2n_2+l_3n_3$。\n
		\item[\textbf{第二步（计算模长）：}]
			$|\vec{l}|=\sqrt{1^2+2^2+1^2}=\sqrt{6}$，$|\vec{n}|=\sqrt{0^2+1^2+0^2}=1$。\n
			\par\textbf{理由：}\n 向量模长等于各分量的平方和再开方。\n
		\item[\textbf{第三步（代入公式）：}]
			\n
			\[
				\n
				\begin{aligned}
					\n \sin\theta &= \frac{|\vec{l}\cdot\vec{n}|}{|\vec{l}||\vec{n}|} \\\n &= \frac{2}{\sqrt{6}\times1} \\\n &= \frac{\sqrt{6}}{3}.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n 直接套用线面角向量公式。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 该直线与平面所成角的正弦为$\frac{\sqrt{6}}{3}$。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 直线$l$经过两点$A(0,1,1),B(1,3,2)$，平面$\alpha$的法向量为$\vec{n}=(1,-1,2)$。求直线$l$与平面$\alpha$所成角的正弦值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求方向向量）：}]
			$\vec{l}=\overrightarrow{AB}=(1-0,3-1,2-1)=(1,2,1)$。\n
			\par\textbf{理由：}\n 方向向量取终点坐标减起点坐标。\n
		\item[\textbf{第二步（计算点积和模）：}]
			$\vec{l}\cdot\vec{n}=1\times1+2\times(-1)+1\times2=1-2+2=1$；$|\vec{l}|=\sqrt{1^2+2^2+1^2}=\sqrt{6}$，$|\vec{n}|=\sqrt{1^2+(-1)^2+2^2}=\sqrt{6}$。\n
			\par\textbf{理由：}\n 使用向量点积和模的计算公式。\n
		\item[\textbf{第三步（代入公式）：}]
			\n
			\[
				\n
				\begin{aligned}
					\n \sin\theta &= \frac{|\vec{l}\cdot\vec{n}|}{|\vec{l}||\vec{n}|} \\\n &= \frac{1}{\sqrt{6}\times\sqrt{6}} \\\n &= \frac{1}{6}.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n 直接套用线面角向量公式。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 直线$l$与平面$\alpha$所成角的正弦值为$\frac{1}{6}$。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 平面$\alpha$经过三点$A(1,0,1),B(0,1,1),C(0,0,2)$，直线$l$的方向向量为$\vec{l}=(1,2,1)$。求直线$l$与平面$\alpha$所成角的正弦值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求平面法向量）：}]
			由$A,B,C$得$\overrightarrow{AB}=(-1,1,0)$，$\overrightarrow{AC}=(-1,0,1)$。法向量$\vec{n}=\overrightarrow{AB}\times\overrightarrow{AC}$。\n
			\[
				\n \vec{n} =
				\begin{vmatrix}
					\n \mathbf{i} & \mathbf{j} & \mathbf{k} \\\n -1 & 1 & 0 \\\n -1 & 0 & 1\n
			\end{vmatrix}
				= (1\cdot1-0\cdot0,\; 0\cdot(-1)-(-1)\cdot1,\; (-1)\cdot0-1\cdot(-1)) = (1,1,1).\n
			\]
			\n
			\par\textbf{理由：}\n 法向量垂直于平面内两条相交直线，由行列式计算。\n
		\item[\textbf{第二步（计算点积和模）：}]
			$\vec{l}\cdot\vec{n}=1\cdot1+2\cdot1+1\cdot1=4$；$|\vec{l}|=\sqrt{6}$，$|\vec{n}|=\sqrt{3}$。\n
			\par\textbf{理由：}\n 点积与模的定义。\n
		\item[\textbf{第三步（代入公式）：}]
			\n
			\[
				\n
				\begin{aligned}
					\n \sin\theta &= \frac{|\vec{l}\cdot\vec{n}|}{|\vec{l}||\vec{n}|} \\\n &= \frac{4}{\sqrt{6}\sqrt{3}} \\\n &= \frac{4}{\sqrt{18}} \\\n &= \frac{4}{3\sqrt{2}} \\\n &= \frac{2\sqrt{2}}{3}.\n
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n 利用公式计算正弦值。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} 直线$l$与平面$\alpha$所成角的正弦值为$\frac{2\sqrt{2}}{3}$。\n
\end{examplebox}
