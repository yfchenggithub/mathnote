	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆正弦与余弦）}}]
			直接将法向量与方向向量夹角的余弦值当作线面角正弦值，忘记取绝对值或直接使用$\cos\theta$公式。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（互余关系）：}}]
			线面角正弦等于法向量与方向向量夹角余弦的绝对值：$\sin\theta = |\cos\langle\vec{n},\vec{l}\rangle|$。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（定义混淆）：}}]
			学生误以为线面角就是法向量与直线的夹角，实际后者是线面角的余角（或其补角），故正弦等于余弦绝对值。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（向量为零向量）}}]
			使用公式时未注意法向量或方向向量是否为零向量，直接代入导致计算无意义。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（向量非零条件）：}}]
			公式要求$\vec{n}\neq\vec{0}$且$\vec{l}\neq\vec{0}$，否则模为零无法计算。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略前提）：}}]
			零向量方向不定，不能代表直线方向或平面法向；为零时公式分母为零，失去意义。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略特殊角）}}]
			计算出$\sin\theta=0$或$1$时，未对应到几何位置关系（平行或垂直）。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（几何对应）：}}]
			$\sin\theta=0$且$\vec{l}\cdot\vec{n}=0$对应$l\parallel\alpha$或$l\subset\alpha$；$\sin\theta=1$对应$l\perp\alpha$。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏几何直观）：}}]
			只进行代数计算，没有将结果还原为立体几何情境，导致无法判定具体位置关系。\n
	\end{description}
	\n
\end{trapbox}
