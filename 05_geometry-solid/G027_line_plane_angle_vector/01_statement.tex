\begin{statementbox}
	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（位置关系）：}}]
			直线$l$与平面$\alpha$相交；当$l\parallel\alpha$或$l\subset\alpha$时，约定$\theta=0^\circ$。\n
		\item[\textbf{\textcolor{CStatement}{条件2（向量非零）：}}]
			平面$\alpha$的法向量$\vec{n}\neq\vec{0}$，直线$l$的方向向量$\vec{l}\neq\vec{0}$。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（线面角正弦公式）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			直线与平面所成角的正弦等于法向量与方向向量夹角余弦的绝对值。\n
			\[
				\n \sin\theta = \lvert\cos\langle\vec{n},\vec{l}\rangle\rvert = \frac{\lvert\vec{n}\cdot\vec{l}\rvert}{\lvert\vec{n}\rvert\,\lvert\vec{l}\rvert}.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（长度无关性）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			公式结果与所取法向量、方向向量的长度无关，改变长度比值不变。\n
			\[
				\n \frac{\lvert(k\vec{n})\cdot(k'\vec{l})\rvert}{\lvert k\vec{n}\rvert\,\lvert k'\vec{l}\rvert} = \frac{\lvert\vec{n}\cdot\vec{l}\rvert}{\lvert\vec{n}\rvert\,\lvert\vec{l}\rvert},\quad k,k'\in\mathbb{R}\setminus\{0\}.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
