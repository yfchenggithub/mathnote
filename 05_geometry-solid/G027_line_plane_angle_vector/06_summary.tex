\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n线面角的正弦值等于法向量与方向向量夹角余弦的绝对值，本质是将几何角度转化为向量点积运算。\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（向量非零）：} 法向量$\vec{n}\neq\vec{0}$，方向向量$\vec{l}\neq\vec{0}$。\n
		\item \textbf{条件2（位置限定）：} 直线$l$与平面$\alpha$相交（若$l\parallel\alpha$或$l\subset\alpha$，则约定$\theta=0^\circ$）。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（取绝对值）：} 公式中分子必须取绝对值，保证正弦非负；若直接使用$\frac{\vec{n}\cdot\vec{l}}{|\vec{n}||\vec{l}|}$会得到有符号余弦，需取绝对值得正弦。\n
		\item \textbf{检查项（特殊情形）：} 当$\vec{n}\cdot\vec{l}=0$时$\theta=0^\circ$；当$\vec{n}\parallel\vec{l}$时$\theta=90^\circ$。\n
	\end{itemize}
	\n
\end{summarybox}
