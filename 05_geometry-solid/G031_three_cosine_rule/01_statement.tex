	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（斜线射影条件）：}}] 斜线 $l$ 与平面 $\alpha$ 斜交，$l$ 在 $\alpha$ 内的射影为 $l'$。
		\item[\textbf{\textcolor{CStatement}{条件2（任意直线条件）：}}] $m$ 是 $\alpha$ 内过斜足的任意直线。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（三余弦关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			斜线与平面内直线所成角的余弦值等于线面角余弦与射影角余弦的乘积。
			\[
				\cos\theta = \cos\theta_1 \cdot \cos\theta_2.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（最小角性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			斜线与平面内所有直线所成角中，线面角最小，且当直线 $m$ 与射影 $l'$ 重合时取等。
			\[
				\theta \ge \theta_1.
			\]
	\end{description}
\end{statementbox}
