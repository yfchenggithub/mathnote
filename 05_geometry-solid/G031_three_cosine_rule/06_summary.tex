\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{斜线与平面内直线所成角的余弦值等于线面角余弦与射影角余弦的乘积，且线面角是最小角。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（斜线射影条件）：} 斜线$l$与平面$\alpha$斜交且射影$l'$确定。
		\item \textbf{条件2（直线过斜足）：} 平面内直线$m$经过斜足$A$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（取角范围）：} $\theta_1,\theta_2,\theta$均取锐角，$\theta_2$为射影与$m$的锐角夹角。
		\item \textbf{检查项（确认直线过斜足）：} 使用前检查$m$是否过斜足，不过则平移。
	\end{itemize}
\end{summarybox}
