\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直圆锥的底面半径 $r=3$，高 $h=4$，求内切球半径 $R$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求母线）：}]
			由勾股定理，$l=\sqrt{r^2+h^2}$。
			\[
				l=\sqrt{3^2+4^2}=5
			\]
			\par\textbf{理由：} 圆锥底面半径、高、母线的关系。
		\item[\textbf{第二步（套公式）：}]
			代入公式 $R=\dfrac{rh}{r+l}$。
			\[
				R=\frac{3\times4}{3+5}=\frac{12}{8}=1.5
			\]
			\par\textbf{理由：} 圆锥内切球半径公式。
		\item[\textbf{第三步（作答）：}]
			得到内切球半径。
			\[
				R=1.5
			\]
			\par\textbf{理由：} 直接计算结果。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=1.5$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直圆锥的底面半径 $r=5$，母线 $l=13$，求内切球半径 $R$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求高）：}]
			由 $h=\sqrt{l^2-r^2}$。
			\[
				h=\sqrt{13^2-5^2}=12
			\]
			\par\textbf{理由：} 圆锥底面半径、高、母线的关系。
		\item[\textbf{第二步（套公式）：}]
			代入公式 $R=\dfrac{rh}{r+l}$ 或变形 $R=\dfrac{r(l-r)}{h}$。
			\[
				R=\frac{5\times12}{5+13}=\frac{60}{18}=\frac{10}{3};\quad\text{或}\quad R=\frac{5\times(13-5)}{12}=\frac{40}{12}=\frac{10}{3}
			\]
			\par\textbf{理由：} 圆锥内切球半径公式。
		\item[\textbf{第三步（作答）：}]
			内切球半径为 $\dfrac{10}{3}$。
			\[
				R=\frac{10}{3}
			\]
			\par\textbf{理由：} 计算结果。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=\dfrac{10}{3}$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	若圆锥的内切球半径为 $2$，底面半径为 $3$，求圆锥的高 $h$ 和母线 $l$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（选变形公式）：}]
			使用有理化变形 $R=\dfrac{r(l-r)}{h}$ 避免根号。代入 $R=2,\ r=3$ 得方程。
			\[
				2=\frac{3(l-3)}{h}\quad\Rightarrow\quad 2h=3l-9
			\]
			\par\textbf{理由：} 变形公式便于建立代数方程。
		\item[\textbf{第二步（联立勾股定理）：}]
			由 $l^2=h^2+r^2$ 和 $2h=3l-9$ 联立。将 $l=\dfrac{2h+9}{3}$ 代入 $l^2=h^2+9$。
			\[
				\left(\frac{2h+9}{3}\right)^2 = h^2+9
			\]
			\par\textbf{理由：} 母线、高、底面半径满足勾股关系。
		\item[\textbf{第三步（解方程）：}]
			展开并整理得 $5h^2-36h=0$，解得 $h=\dfrac{36}{5}=7.2$（舍去 $h=0$），回代得 $l=7.8$。
			\[
				h=7.2,\quad l=7.8
			\]
			\par\textbf{理由：} 一元二次方程求解与回代。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{高 $h=7.2$，母线 $l=7.8$}。
\end{examplebox}
