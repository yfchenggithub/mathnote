\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记忆）}}]
			误认为 $R=\dfrac{rh}{r+h}$，分母错用高 $h$ 代替母线 $l$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式）}}]
			正确公式为 $R=\dfrac{rh}{r+l}$，分母是底面半径与母线长之和。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导混淆）}}]
			未理解等面积法推导步骤，或混淆了 $l$ 与 $h$ 的几何意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（母线计算）}}]
			计算母线时直接使用 $l=r+h$ 或忽略平方关系。
		\item[\textbf{\textcolor{CTrap}{正确理解（勾股求值）}}]
			必须通过 $l=\sqrt{r^2+h^2}$ 精确计算。
		\item[\textbf{\textcolor{CTrap}{错因分析（粗心误解）}}]
			对圆锥母线定义不清，或因计算疏忽导致错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（通用混淆）}}]
			认为圆锥内切球半径等于 $\dfrac{r+h}{2}$ 等无根据形式。
		\item[\textbf{\textcolor{CTrap}{正确理解（唯一公式）}}]
			圆锥内切球半径必须用 $R=\dfrac{rh}{r+l}$ 计算。
		\item[\textbf{\textcolor{CTrap}{错因分析（直觉误判）}}]
			凭感觉类比其他几何体的内切半径，未从圆锥结构出发。
	\end{description}

\end{trapbox}
