\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（几何前提）：}}]
			直圆锥，顶点在底面圆心正上方，底面半径为 $r$，高为 $h$，母线长 $l = \sqrt{r^2 + h^2}$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（内切球半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆锥内切球半径等于底面半径乘以高除以底面半径与母线之和。
			\[
				R = \frac{rh}{r+l}
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（有理化形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过分母有理化消去分母中的根式，便于计算。
			\[
				R = \frac{r(l-r)}{h}
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{适用范围：}}
	仅适用于直圆锥，且球与圆锥底面和侧面均相切。
\end{statementbox}
