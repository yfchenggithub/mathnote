\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（截面边数范围）}}]
			认为截面多边形可以是任意超过6的边数，或认为可以切出三角形以外的2边形（线段）。
		\item[\textbf{\textcolor{CTrap}{正确理解（边数限制）：}}]
			截面边数等于相交面数，且由于正方体只有6个面，封闭截面至少与3个面相交，故边数只能是3,4,5,6。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略面数）：}}]
			忽视了正方体有6个面，以及截面封闭所需的最少面数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（边长直接代入）}}]
			计算三角形边长时，直接使用截线段长 $x,y,z$ 作为边长。
		\item[\textbf{\textcolor{CTrap}{正确理解（勾股求和）：}}]
			边长不是 $x$ 或 $y$，而是 $\sqrt{x^2+y^2}$、$\sqrt{y^2+z^2}$、$\sqrt{z^2+x^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆线段）：}}]
			误以为截面三角形的顶点与正方体顶点重合，实际截面三角形的顶点在棱上，而非正方体顶点，故需用勾股定理。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（锐角性质忽略）}}]
			认为截面三角形可能是钝角或直角三角形，从而错误判断。
		\item[\textbf{\textcolor{CTrap}{正确理解（必锐角）：}}]
			由边长平方关系可知任意两边的平方和大于第三边的平方，故三角形必为锐角三角形。
		\item[\textbf{\textcolor{CTrap}{错因分析（未用平方关系）：}}]
			没有意识到 $x,y,z$ 的正数性导致恒正，或直接用视觉直观导致误判。
	\end{description}

\end{trapbox}
