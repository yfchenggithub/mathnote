\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	正方体棱长为~4，平面过顶点~$A$，与棱~$AB,AD,AA_1$~分别交于点~$P,Q,R$，且~$AP=1,~AQ=2,~AR=3$。求~$\triangle PQR$~的三边长并判断其形状。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定边长公式）：}]
			由定理，$\triangle PQR$~三边长的平方依次为
			\[
				AP^2+AQ^2,\quad AQ^2+AR^2,\quad AR^2+AP^2.
			\]
			\par\textbf{理由：} $AP, AQ, AR$~两两垂直，故在直角三角形中由勾股定理得边长平方。
		\item[\textbf{第二步（代入数值）：}]
			代入得
			\[
				1^2+2^2=5,\quad 2^2+3^2=13,\quad 3^2+1^2=10,
			\]
			所以三边长分别为~$\sqrt{5},\;\sqrt{13},\;\sqrt{10}$。
			\par\textbf{理由：} 直接算术运算。
		\item[\textbf{第三步（判定形状）：}]
			最大边平方~$13$~小于另两边平方和~$5+10=15$，故三角形为锐角三角形。
			\par\textbf{理由：} 余弦定理：$13<5+10$ 等价于最大角为锐角。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\triangle PQR$~三边为~$\sqrt{5},\sqrt{13},\sqrt{10}$，且一定是锐角三角形。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	正方体棱长为~4，平面过顶点~$A$，与棱~$AB,AD,AA_1$~交于~$P,Q,R$，且~$AP+AQ+AR=6$，$AP:AQ:AR=1:2:3$。求~$\triangle PQR$~的三边长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}]
			设~$AP=k,\;AQ=2k,\;AR=3k$。由和条件~$k+2k+3k=6$ 得~$k=1$。故~$AP=1,\;AQ=2,\;AR=3$。
			\par\textbf{理由：} 利用比例参数设未知量，再代入和条件求解。
		\item[\textbf{第二步（边长公式）：}]
			由定理，三边平方依次为~$1^2+2^2=5$,\, $2^2+3^2=13$,\, $3^2+1^2=10$。所以三边长~$\sqrt{5},\;\sqrt{13},\;\sqrt{10}$。
			\par\textbf{理由：} 同例1，直接代入公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\triangle PQR$~三边长为~$\sqrt{5},\sqrt{13},\sqrt{10}$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	正方体棱长为~5，平面过顶点~$A$，与棱~$AB,AD,AA_1$~分别交于~$P,Q,R$，且~$AP=2,\;AQ=3,\;AR=4$。求~$\triangle PQR$~的面积。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（边长平方）：}]
			由定理，三边平方为
			\[
				2^2+3^2=13,\quad 3^2+4^2=25,\quad 4^2+2^2=20,
			\]
			故三边长分别为~$\sqrt{13},\;5,\;2\sqrt{5}$。
			\par\textbf{理由：} 定理直接给出。
		\item[\textbf{第二步（应用面积公式）：}]
			因为~$AP\perp AQ\perp AR$，三角形$PQR$的面积可由叉积得~$S=\frac12\sqrt{(AP\cdot AQ)^2+(AQ\cdot AR)^2+(AR\cdot AP)^2}$，即~$S=\frac12\sqrt{x^2y^2+y^2z^2+z^2x^2}$。代入$x=2,y=3,z=4$：
			\[
				\begin{aligned}
					S &= \frac12\sqrt{(2\cdot3)^2+(3\cdot4)^2+(4\cdot2)^2} \\ &= \frac12\sqrt{36+144+64}.
			\end{aligned}
			\]
			\par\textbf{理由：} 两两垂直时，叉积模的平方等于各坐标平方和。
		\item[\textbf{第三步（化简得结果）：}]
			\[
				\begin{aligned}
					\frac12\sqrt{244} &= \frac12\cdot2\sqrt{61} \\ &= \sqrt{61}.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\triangle PQR$~的面积为~$\sqrt{61}$。
\end{examplebox}
