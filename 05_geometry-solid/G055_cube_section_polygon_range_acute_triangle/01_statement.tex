\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（截面限制）：}}]
			平面与正方体相交，截面为封闭多边形；平面不经过正方体的顶点，也不与任何面重合。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（截面边数规则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			截面边数 $n$ 等于平面与正方体相交的面数 $m$，且 $n$ 只取 3,4,5,6 四种可能。
			\[
				n = m \in \{3,4,5,6\}.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（三角形截面性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当截面为三角形时，该三角形必为锐角三角形；三边长的平方由从同一顶点出发的三条棱上的截线段长 $x,y,z$ 决定。
			\[
				x^{2}+y^{2},\quad y^{2}+z^{2},\quad z^{2}+x^{2}.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{常见变形}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{推广结论（边长公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将三边平方公式开方即可直接得到截面三角形的三边长，便于面积或余弦计算。
			\[
				\sqrt{x^{2}+y^{2}},\quad \sqrt{y^{2}+z^{2}},\quad \sqrt{z^{2}+x^{2}}.
			\]
	\end{description}

\end{statementbox}
