\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件误用）}}] 误认为任意三棱锥的外接球球心都在过底面外心且垂直于底面的直线上。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件限定）：}}] 仅当侧棱 $SA=SB=SC$，或者等价地，顶点 $S$ 在底面上的射影为底面 $\triangle ABC$ 的外心时，该结论才成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略前提）：}}] 从正三棱锥等特殊情形类推，忽略了侧棱相等的必要条件。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公式混淆）}}] 在半径公式 $R = \dfrac{h^{2}+r^{2}}{2h}$ 中，将 $h$ 误当作侧棱长或将 $r$ 当作底面边长。
		\item[\textbf{\textcolor{CTrap}{正确理解（变量区分）：}}] $h$ 是顶点 $S$ 到底面 $ABC$ 的距离，即三棱锥的高；$r$ 是底面 $\triangle ABC$ 外接圆的半径。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混乱）：}}] 未经历公式推导，对每个参数的几何意义模糊，直接套用导致错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（位置误判）}}] 认为球心 $Q$ 一定在线段 $SO$ 上，即一定在 $S$ 与 $O$ 之间。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论）：}}] 设底面外接圆半径为 $r$，高为 $h=SO>0$，则
		      \[
			      \begin{cases}
				      h^{2}>r^{2}, & Q \text{ 在线段 } SO \text{ 上};               \\[4pt]
				      h^{2}=r^{2}, & Q=O;                                       \\[4pt]
				      h^{2}<r^{2}, & Q \text{ 在 } O \text{ 的远离 } S \text{ 的一侧}.
			      \end{cases}
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（代数盲点）：}}] 只从正棱锥的直观图形出发，未考虑解方程
		      \[
			      t=\frac{h^{2}-r^{2}}{2h}
		      \]
		      时，$t$ 可能为正、为零或为负。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（对称轴误解）}}] 认为只要 $SA=SB=SC$，直线 $SO$ 就一定是三棱锥的对称轴。
		\item[\textbf{\textcolor{CTrap}{正确理解（定位线而非必然对称轴）：}}] 直线 $SO$ 一定是球心定位线，即过底面外心且垂直于底面的直线；但当底面 $\triangle ABC$ 不是正三角形或不具有相应对称性时，它不一定是整个三棱锥的几何对称轴。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混用）：}}] 把“球心在某条垂线上”误解为“图形关于这条直线对称”，混淆了定位关系与对称关系。
	\end{description}
\end{trapbox}
