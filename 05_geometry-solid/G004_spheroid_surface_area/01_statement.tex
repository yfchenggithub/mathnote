\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（椭圆方程）：}}]
		      椭圆方程为
		      \[
			      \frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1,
			      \qquad a>0,\quad b>0.
		      \]

		\item[\textbf{\textcolor{CStatement}{条件 2（旋转轴）：}}]
		      椭圆绕 $x$ 轴旋转一周。
	\end{description}

	\medskip

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]

		\item[\textbf{\textcolor{CStatement}{结论一（长椭球情形）：}}]
		      当 $a>b$ 时，旋转椭球呈长椭球形状，表面积公式含反三角函数。

		      设
		      \[
			      e=\sqrt{1-\frac{b^{2}}{a^{2}}},
		      \]
		      则旋转椭球的表面积为
		      \[
			      S
			      =
			      2\pi b^{2}
			      +
			      \frac{2\pi ab}{e}\arcsin e.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（扁椭球情形）：}}]
		      当 $a<b$ 时，旋转椭球呈扁椭球形状，表面积公式含反双曲正切函数。

		      设
		      \[
			      e=\sqrt{1-\frac{a^{2}}{b^{2}}},
			      \qquad
			      \operatorname{artanh} e
			      =
			      \frac12\ln\frac{1+e}{1-e},
		      \]
		      则旋转椭球的表面积为
		      \[
			      S
			      =
			      2\pi b^{2}
			      +
			      \frac{2\pi a^{2}}{e}\operatorname{artanh} e.
		      \]

	\end{description}

	\medskip

	\textbf{\textcolor{CStatement}{特例}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{退化为球：}}]
		      当 $a=b=R$ 时，椭圆退化为圆，旋转椭球退化为半径为 $R$ 的球。

		      此时表面积为
		      \[
			      S=4\pi R^{2}.
		      \]
	\end{description}

\end{statementbox}
