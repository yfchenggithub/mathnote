\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（误用底面矩形）：}}] 认为外接球直径等于底面矩形对角线 $\sqrt{a^2+b^2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（体对角线）：}}] 外接球直径等于长方体体对角线 $\sqrt{a^2+b^2+c^2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略高度）：}}] 只考虑底面，忽略了高度 $c$ 的贡献。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（误用平面距离）：}}] 计算体对角线时写为 $\sqrt{a^2+b^2}+c$ 或类似错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（空间距离公式）：}}] 应使用
		      \[
			      d = \sqrt{a^2+b^2+c^2}.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（维度混淆）：}}] 空间距离不是“底面对角线长度加高度”，而是三个互相垂直方向长度的平方和再开方。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忘除二）：}}] 认为 $R = \sqrt{a^2+b^2+c^2}$，忘记除以 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（半径公式）：}}]
		      \[
			      R = \frac{1}{2}\sqrt{a^2+b^2+c^2}.
		      \]
		\item[\textbf{\textcolor{CTrap}{错因分析（定义不清）：}}] 体对角线是外接球直径，不是半径。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（补形条件不清）：}}] 看到“能补成长方体”就直接套公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（顶点与外接球）：}}] 必须确认原几何体的顶点是该长方体的部分顶点，并且长方体外接球经过原几何体所有顶点。
		\item[\textbf{\textcolor{CTrap}{错因分析（乱用补形）：}}] 补形会增加新顶点，不能说“补形不改变顶点集合”。真正需要说明的是：原来的顶点仍在同一个长方体外接球上，且原几何体的外接球由这些顶点唯一确定。
	\end{description}
\end{trapbox}
