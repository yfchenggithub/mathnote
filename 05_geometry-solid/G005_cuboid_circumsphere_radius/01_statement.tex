\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（长方体尺寸）：}}] 长方体的长、宽、高分别为 $a,b,c$，且 $a>0,b>0,c>0$。
		\item[\textbf{\textcolor{CStatement}{条件 2（补形使用）：}}] 若题目中的几何体可以补成长方体，必须保证原几何体的顶点是该长方体的部分顶点，且这些点与长方体确定的是同一个外接球。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 长方体外接球半径等于体对角线的一半。
		      \[
			      R = \frac{1}{2}\sqrt{a^2+b^2+c^2}.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（正方体特例）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 当长、宽、高相等时，即正方体棱长为 $a$ 时，
		      \[
			      R = \frac{\sqrt{3}}{2}a.
		      \]
	\end{description}
\end{statementbox}
