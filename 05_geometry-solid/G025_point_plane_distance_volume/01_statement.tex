\begin{statementbox}
	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（体积底面积已知）：}}]
			对于三棱锥 $P$-$ABC$，记其体积为 $V$，底面 $\triangle ABC$ 的面积为 $S_{\triangle ABC}$。\n
	\end{description}
	\n

	\medskip
	\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（点到面距离公式）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			点到平面的距离等于三倍体积除以底面面积。\n
			\[
				h = \frac{3V}{S_{\triangle ABC}}.
			\]
			\n
		\item[\textbf{\textcolor{CStatement}{推广结论（换底转化）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			更换三棱锥的底面，利用体积不变性可间接求距离。\n
			\[
				h_1 = \frac{3V}{S_1},\quad h_2 = \frac{3V}{S_2},\quad \dots
			\]
			\n 其中 $V$ 为体积，$S_1,S_2,\dots$ 为不同底面的面积，$h_1,h_2,\dots$ 为对应顶点到底面的距离。\n
	\end{description}
	\n
\end{statementbox}
