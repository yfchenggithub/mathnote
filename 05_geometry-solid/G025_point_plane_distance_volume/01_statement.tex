\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（体积底面积已知）：}}]
		      对于三棱锥 $P\text{-}ABC$，记其体积为 $V$，底面 $\triangle ABC$ 的面积为 $S_{\triangle ABC}$。
	\end{description}

	\medskip

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（点到面距离公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      点 $P$ 到平面 $ABC$ 的距离等于三倍体积除以底面面积。
		      \[
			      h = \frac{3V}{S_{\triangle ABC}}.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（换底转化）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      更换三棱锥的底面，利用体积不变性可间接求距离。
		      \[
			      h_1 = \frac{3V}{S_1},\quad h_2 = \frac{3V}{S_2},\quad \dots
		      \]
		      其中 $V$ 为同一个三棱锥的体积，$S_1,S_2,\dots$ 为不同底面的面积，$h_1,h_2,\dots$ 为对应顶点到底面的距离。
	\end{description}

\end{statementbox}
