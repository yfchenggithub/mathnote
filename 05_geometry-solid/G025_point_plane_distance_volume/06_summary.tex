\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{等体积法求点到平面的距离公式 $h = \dfrac{3V}{S}$，实质是将几何度量转化为数值运算，核心是利用同一个三棱锥的体积不变性。}\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（锥体体积已知）：} 需知三棱锥的体积 $V$，或能从已知条件（如垂直关系、已知高和底面积）求出 $V$。\n
		\item \textbf{条件2（底面积可求）：} 需知道或能求出顶点所对底面的面积 $S$（如已知三角形边长或坐标）。\n
	\end{itemize}
	\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（系数勿忘）：} 计算时不能遗漏系数 $3$，牢记 $h = \dfrac{3V}{S}$ 源自 $V = \dfrac13 S h$。\n
		\item \textbf{检查项（对应验证）：} 使用公式前确认底面与该顶点在同一个三棱锥中，且体积和面积对应正确。\n
	\end{itemize}
	\n
\end{summarybox}
