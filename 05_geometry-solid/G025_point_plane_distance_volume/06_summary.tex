\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{等体积法求点到平面的距离公式 $h = \dfrac{3V}{S}$，实质是将几何度量转化为数值运算，核心是利用同一个三棱锥的体积不变性。}

	\medskip

	\textbf{\textcolor{CSummary}{使用条件}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（锥体体积已知）：} 需知道三棱锥的体积 $V$，或能从已知条件，如垂直关系、已知高和底面积，求出 $V$。

		\item \textbf{条件2（底面积可求）：} 需知道或能求出顶点所对底面的面积 $S$，如已知三角形边长、坐标或面积关系。
	\end{itemize}

	\medskip

	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（系数勿忘）：} 计算时不能遗漏系数 $3$，牢记 $h = \dfrac{3V}{S}$ 源自 $V = \dfrac13 S h$。

		\item \textbf{检查项（对应验证）：} 使用公式前确认底面、顶点、体积三者属于同一个三棱锥，且面积和距离对应正确。
	\end{itemize}

\end{summarybox}
