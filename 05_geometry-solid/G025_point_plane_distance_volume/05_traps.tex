\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式漏系数）：}}]
		      误以为 $h=\dfrac{V}{S}$，忽略体积公式中的三分之一因子。

		\item[\textbf{\textcolor{CTrap}{正确理解（含三分之一）：}}]
		      三棱锥体积公式 $V=\dfrac13 S h$，故 $h=\dfrac{3V}{S}$，必须乘以 $3$。

		\item[\textbf{\textcolor{CTrap}{错因分析（混用柱体公式）：}}]
		      将锥体的体积公式与柱体混淆，柱体体积无 $\dfrac13$。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（底面对应不清）：}}]
		      随意选取一个底面面积代入公式，导致计算出的距离与所求距离不对应。

		\item[\textbf{\textcolor{CTrap}{正确理解（顶点底面匹配）：}}]
		      公式 $h=\dfrac{3V}{S}$ 中的 $S$ 必须是顶点所对底面的面积；使用等体积换底时，必须明确新底面对应的顶点。

		\item[\textbf{\textcolor{CTrap}{错因分析（忽视对应关系）：}}]
		      认为底面积只要在三棱锥中即可，忽略了距离与底面的配对关系。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（体积不等换底）：}}]
		      换底时误用不同三棱锥的体积，导致等体积关系不成立。

		\item[\textbf{\textcolor{CTrap}{正确理解（同锥体积相等）：}}]
		      所谓等体积法，必须保证三棱锥是同一个，或者明确知道两个锥体体积相等，不能在不同锥体之间随意转化。

		\item[\textbf{\textcolor{CTrap}{错因分析（混淆不同锥体）：}}]
		      没有理解“等体积”首先是指同一个三棱锥在不同选底方式下体积不变。
	\end{description}

\end{trapbox}
