\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n从三棱锥体积的定义式$V=\frac13 S h$出发，将所需距离$h$视为未知量，通过代数变形直接解出$h$。\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（列体积公式）：}]
			对于三棱锥$P$-$ABC$，记底面$\triangle ABC$面积为$S_{\triangle ABC}$，顶点$P$到底面的距离为$h$，则其体积\n
			\[
				V = \frac13 S_{\triangle ABC} \cdot h.
			\]
			\n
			\par\textbf{理由：}\n 三棱锥的体积定义是底面积乘以高的三分之一，这是立体几何中的基本公式。\n
		\item[\textbf{步骤二（等号两边同时乘以3）：}]
			\n
			\[
				3V = S_{\triangle ABC} \cdot h.
			\]
			\n
			\par\textbf{理由：}\n 等式两边同时乘以同一个非零常数，等式仍然成立（这里$3\neq0$）。\n
		\item[\textbf{步骤三（两边同时除以$S_{\triangle ABC}$）：}]
			\n
			\[
				h = \frac{3V}{S_{\triangle ABC}}.
			\]
			\n
			\par\textbf{理由：}\n 等式两边同时除以一个非零量$S_{\triangle ABC}$（底面面积不为零），等式仍然成立。\n
	\end{description}
	\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\textbf{由此得到，只要知道三棱锥的体积$V$和底面面积$S_{\triangle ABC}$，就可以直接计算出顶点$P$到底面$\triangle ABC$的距离$h$。该公式是等体积法求空间距离的基础：在同一个三棱锥中，体积$V$保持不变，更换底面即可得到不同顶点到对应底面的距离，从而将难以直接作垂线的距离转化为可求的比值。}\n
\end{proofbox}
