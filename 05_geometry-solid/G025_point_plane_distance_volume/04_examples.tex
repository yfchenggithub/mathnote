\textbf{\textcolor{CExample}{例 1（基础）}}

\textbf{\textcolor{CExample}{题目：}} 已知三棱锥 $P\text{-}ABC$ 的体积 $V=30$，底面 $\triangle ABC$ 的面积 $S=18$，求顶点 $P$ 到底面 $ABC$ 的距离 $h$。

\textbf{\textcolor{CExample}{解题步骤：}}

\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
	\item[\textbf{第一步（列条件已知量）：}]
	      体积 $V=30$，底面面积 $S=18$，待求距离 $h$。

	      \par\textbf{理由：}
	      等体积法公式 $h=\dfrac{3V}{S}$ 需要这两个量。

	\item[\textbf{第二步（代入公式）：}]
	      \[
		      h = \frac{3 \times 30}{18}.
	      \]

	      \par\textbf{理由：}
	      由等体积法公式 $h=\dfrac{3V}{S}$ 直接代入。

	\item[\textbf{第三步（计算结果）：}]
	      \[
		      \begin{aligned}
			      h & = \frac{90}{18} \\
			        & = 5.
		      \end{aligned}
	      \]

	      \par\textbf{理由：}
	      算术运算。
\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{点 $P$ 到平面 $ABC$ 的距离为 $5$。}

\textbf{\textcolor{CExample}{例 2（稍变形）}}

\textbf{\textcolor{CExample}{题目：}} 三棱锥 $A\text{-}BCD$ 中，已知底面 $BCD$ 面积 $S_1=12$，顶点 $A$ 到底面 $BCD$ 的距离 $h_1=4$；若 $\triangle ACD$ 面积 $S_2=8$，求顶点 $B$ 到底面 $ACD$ 的距离 $h_2$。

\textbf{\textcolor{CExample}{解题步骤：}}

\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
	\item[\textbf{第一步（求三棱锥体积）：}]
	      以 $BCD$ 为底面，$A$ 为顶点，则
	      \[
		      \begin{aligned}
			      V & = \frac13 S_1 h_1            \\
			        & = \frac13 \times 12 \times 4 \\
			        & = 16.
		      \end{aligned}
	      \]

	      \par\textbf{理由：}
	      三棱锥体积公式 $V=\dfrac13 S h$。

	\item[\textbf{第二步（换底建立等式）：}]
	      对于同一个三棱锥，若以 $\triangle ACD$ 为底面，$B$ 为顶点，体积仍为 $V$，设 $h_2$ 为 $B$ 到平面 $ACD$ 的距离，则
	      \[
		      V = \frac13 S_2 h_2.
	      \]

	      \par\textbf{理由：}
	      等体积法：同一个三棱锥体积不变，更换底面后对应的高改变。

	\item[\textbf{第三步（解出 $h_2$）：}]
	      \[
		      \begin{aligned}
			      h_2 & = \frac{3V}{S_2}        \\
			          & = \frac{3 \times 16}{8} \\
			          & = 6.
		      \end{aligned}
	      \]

	      \par\textbf{理由：}
	      由 $V=\dfrac13 S_2 h_2$ 解出 $h_2$。
\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{点 $B$ 到平面 $ACD$ 的距离为 $6$。}

\textbf{\textcolor{CExample}{例 3（综合应用）}}

\textbf{\textcolor{CExample}{题目：}} 三棱锥 $P\text{-}ABC$ 中，$PA,\ PB,\ PC$ 两两垂直，且 $PA=PB=PC=2$，求点 $P$ 到平面 $ABC$ 的距离。

\textbf{\textcolor{CExample}{解题步骤：}}

\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
	\item[\textbf{第一步（求三棱锥体积）：}]
	      以 $\triangle PAB$ 为底面，$PC$ 为高。因为 $PA\perp PC$，$PB\perp PC$，且 $PA$ 与 $PB$ 是平面 $PAB$ 内的两条相交直线，所以 $PC\perp$ 平面 $PAB$。因此
	      \[
		      \begin{aligned}
			      V & = \frac13 \times \frac12 \times PA \times PB \times PC \\
			        & = \frac13 \times \frac12 \times 2 \times 2 \times 2    \\
			        & = \frac{4}{3}.
		      \end{aligned}
	      \]

	      \par\textbf{理由：}
	      $PA,\ PB,\ PC$ 两两垂直，故 $PC$ 垂直于底面 $PAB$，底面 $PAB$ 为直角三角形，面积为 $\dfrac12 \cdot PA \cdot PB$。

	\item[\textbf{第二步（求底面 $\triangle ABC$ 的面积）：}]
	      由勾股定理，
	      \[
		      \begin{aligned}
			      AB & = \sqrt{PA^2+PB^2}=2\sqrt{2}, \\
			      BC & = \sqrt{PB^2+PC^2}=2\sqrt{2}, \\
			      AC & = \sqrt{PA^2+PC^2}=2\sqrt{2}.
		      \end{aligned}
	      \]
	      所以 $\triangle ABC$ 是边长为 $2\sqrt{2}$ 的等边三角形，
	      \[
		      \begin{aligned}
			      S_{\triangle ABC}
			       & = \frac{\sqrt3}{4} \times (2\sqrt{2})^2 \\
			       & = \frac{\sqrt3}{4} \times 8             \\
			       & = 2\sqrt{3}.
		      \end{aligned}
	      \]

	      \par\textbf{理由：}
	      垂直关系导出各棱长，等边三角形面积公式为 $S=\dfrac{\sqrt3}{4}a^2$。

	\item[\textbf{第三步（等体积法求距离）：}]
	      设 $h$ 为点 $P$ 到平面 $ABC$ 的距离，则由等体积法
	      \[
		      \begin{aligned}
			      h & = \frac{3V}{S_{\triangle ABC}}            \\
			        & = \frac{3 \times \dfrac{4}{3}}{2\sqrt{3}} \\
			        & = \frac{4}{2\sqrt{3}}                     \\
			        & = \frac{2}{\sqrt{3}}                      \\
			        & = \frac{2\sqrt{3}}{3}.
		      \end{aligned}
	      \]

	      \par\textbf{理由：}
	      等体积法公式 $h=\dfrac{3V}{S}$，将已求得的 $V$ 和 $S_{\triangle ABC}$ 代入。
\end{description}

\textbf{\textcolor{CExample}{关键结论：}} \textbf{点 $P$ 到平面 $ABC$ 的距离为 $\dfrac{2\sqrt{3}}{3}$。}
