	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（线面平行）：}}]
			直线 $l$ 与平面 $\alpha$ 平行。
		\item[\textbf{\textcolor{CStatement}{条件 2（面面平行）：}}]
			平面 $\beta$ 与平面 $\alpha$ 平行。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线面距公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直线 $l$ 上任一点到平面 $\alpha$ 的距离都相等，等于该点到平面的距离。
			\[
				d = \frac{|A x_0 + B y_0 + C z_0 + D|}{\sqrt{A^2 + B^2 + C^2}},
			\]
			其中 $(x_0, y_0, z_0)$ 是 $l$ 上任意取定的点，平面 $\alpha$ 方程为 $Ax+By+Cz+D=0$。
		\item[\textbf{\textcolor{CStatement}{推广结论（面面距公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平面 $\beta$ 上任一点到平面 $\alpha$ 的距离都相等，同样用点面距公式计算。
			\[
				d = \frac{|A x_0 + B y_0 + C z_0 + D|}{\sqrt{A^2 + B^2 + C^2}},
			\]
			其中 $(x_0, y_0, z_0)$ 是 $\beta$ 上任意取定的点。
	\end{description}

\end{statementbox}
