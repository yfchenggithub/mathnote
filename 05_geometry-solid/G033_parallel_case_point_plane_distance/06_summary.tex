	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{线面距和面面距均可归结为点到平面的距离，直接代入点面距公式即可。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（线面平行）：} 直线与平面平行时可用。
		\item \textbf{条件 2（面面平行）：} 两平面相互平行时可用。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式完整）：} 分母 $\sqrt{A^2+B^2+C^2}$ 和绝对值 $\left| \cdot \right|$ 不能漏。
		\item \textbf{检查项（平行确认）：} 使用前必须验证平行关系。
	\end{itemize}

\end{summarybox}
