\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l$ 平行于平面 $\alpha$，$l$ 经过点 $P(1,2,3)$，平面 $\alpha$ 的方程为 $2x - y + 2z - 1 = 0$，求直线 $l$ 到平面 $\alpha$ 的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（明确法向量）：}]
			由平面方程得法向量 $\mathbf{n}=(2,-1,2)$，模长 $|\mathbf{n}|=\sqrt{2^2+(-1)^2+2^2}=3$。
			\par\textbf{理由：} 点面距公式需要法向量和点坐标。
		\item[\textbf{第二步（代入点坐标）：}]
			将点 $P(1,2,3)$ 代入平面方程左端计算分子：$|2\cdot1 + (-1)\cdot2 + 2\cdot3 -1| = |2-2+6-1| = 5$。
			\par\textbf{理由：} 分子是 $|Ax_0+By_0+Cz_0+D|$。
		\item[\textbf{第三步（计算距离）：}]
			代入公式 $d = \frac{|\text{分子}|}{|\mathbf{n}|} = \frac{5}{3}$。
			\par\textbf{理由：} 线面距等于点面距。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直线 $l$ 到平面 $\alpha$ 的距离为 $\dfrac{5}{3}$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知平面 $\beta$ 平行于平面 $\alpha$，平面 $\alpha$ 的方程为 $x+2y-2z+3=0$，平面 $\beta$ 上一点 $M(1,1,1)$，求两平行平面间的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（读取法向量）：}]
			平面 $\alpha$ 的法向量 $\mathbf{n}=(1,2,-2)$，模长 $|\mathbf{n}|=\sqrt{1+4+4}=3$。
			\par\textbf{理由：} 点面距分母。
		\item[\textbf{第二步（代入点坐标）：}]
			将点 $M(1,1,1)$ 代入平面 $\alpha$ 方程计算分子：$|1\cdot1+2\cdot1+(-2)\cdot1+3| = |1+2-2+3| = 4$。
			\par\textbf{理由：} 分子代入点面距公式。
		\item[\textbf{第三步（计算距离）：}]
			$d = \dfrac{|4|}{3} = \dfrac{4}{3}$。
			\par\textbf{理由：} 面面距等于点面距。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{两平行平面间的距离为 $\dfrac{4}{3}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	平面 $\alpha$ 经过点 $A(1,0,0)$，$B(0,2,0)$，$C(0,0,3)$；平面 $\beta$ 平行于 $\alpha$ 且过点 $M(1,1,1)$，求平面 $\beta$ 与平面 $\alpha$ 之间的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求平面方程）：}]
			由截距式得 $\frac{x}{1}+\frac{y}{2}+\frac{z}{3}=1$，化为一般式 $6x+3y+2z-6=0$，法向量 $\mathbf{n}=(6,3,2)$，模长 $|\mathbf{n}|=\sqrt{36+9+4}=7$。
			\par\textbf{理由：} 需要平面方程才能使用点面距公式。
		\item[\textbf{第二步（确认平行并代入）：}]
			因为 $\beta\parallel\alpha$，点 $M$ 在 $\beta$ 上，所以将 $M$ 代入 $\alpha$ 的平面方程左端计算分子：$|6\cdot1+3\cdot1+2\cdot1-6| = |6+3+2-6| = 5$。
			\par\textbf{理由：} 面面距转化为点面距。
		\item[\textbf{第三步（计算距离）：}]
			$d = \dfrac{5}{7}$。
			\par\textbf{理由：} 直接应用公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{平面 $\beta$ 与平面 $\alpha$ 的距离为 $\dfrac{5}{7}$。}
\end{examplebox}
