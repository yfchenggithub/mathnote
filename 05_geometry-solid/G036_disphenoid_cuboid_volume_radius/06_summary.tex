\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对棱相等的四面体可补形为长方体，通过长方体直接求体积和外接球半径。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（对棱相等）：} 四面体三组对棱分别相等。
		\item \textbf{条件2（正数条件）：} 根号内 $\dfrac{b^2+c^2-a^2}{2}>0$ 等，确保 $x,y,z$ 为正实数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（根号内正）：} 代入公式前必须检查 $b^2+c^2>a^2$ 等，否则公式无效。
		\item \textbf{检查项（系数核对）：} 体积系数 $\frac13$，半径系数 $\frac12$，切勿混淆。
	\end{itemize}

\end{summarybox}
