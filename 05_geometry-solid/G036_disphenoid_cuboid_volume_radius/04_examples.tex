	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知四面体 $ABCD$ 对棱相等，$AB=CD=\sqrt{10}$，$AC=BD=\sqrt{10}$，$AD=BC=\sqrt{2}$。求四面体的体积 $V$ 和外接球半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（补形求解）：}] 由公式
			\[
				\begin{aligned}
					x &= \sqrt{\frac{b^2+c^2-a^2}{2}} \\
					&= \sqrt{\frac{10+2-10}{2}} \\
					&= 1,\\
					y &= \sqrt{\frac{a^2+c^2-b^2}{2}} \\
					&= \sqrt{\frac{10+2-10}{2}} \\
					&= 1,\\
					z &= \sqrt{\frac{a^2+b^2-c^2}{2}} \\
					&= \sqrt{\frac{10+10-2}{2}} \\
					&= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入 $a^2=10,\;b^2=10,\;c^2=2$ 到公式中计算。
		\item[\textbf{第二步（体积公式）：}]
			\[
				\begin{aligned}
					V &= \frac13 xyz \\
					&= \frac13\times1\times1\times3 \\
					&= 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 四面体体积等于长方体体积的三分之一。
		\item[\textbf{第三步（外接球半径）：}]
			\[
				\begin{aligned}
					R &= \frac12\sqrt{x^2+y^2+z^2} \\
					&= \frac12\sqrt{1+1+9} \\
					&= \frac{\sqrt{11}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 外接球半径等于长方体体对角线的一半。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$V=1$，$R=\dfrac{\sqrt{11}}{2}$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	四面体对棱相等，补成长方体后长宽高之比为 $1:2:3$，且长方体的体对角线长为 $\sqrt{14}$。求四面体的体积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设参数）：}] 设 $x=k,\;y=2k,\;z=3k$。由体对角线公式得
			\[
				\begin{aligned}
					\sqrt{x^2+y^2+z^2} &= \sqrt{k^2+4k^2+9k^2} \\
					&= \sqrt{14k^2} \\
					&= \sqrt{14},
			\end{aligned}
			\]
			解得 $k=1$。所以 $x=1,y=2,z=3$。
			\par\textbf{理由：} 长方体体对角线长平方等于长宽高平方和。
		\item[\textbf{第二步（体积公式）：}]
			\[
				\begin{aligned}
					V &= \frac13 xyz \\
					&= \frac13\times1\times2\times3 \\
					&= 2.
			\end{aligned}
			\]
			\par\textbf{理由：} 四面体体积公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$V=2$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	四面体对棱相等，其三组对棱长 $a,b,c$ 满足 $a^2+b^2+c^2=54$。求四面体体积的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平方和转化）：}] 由结论 $x^2+y^2+z^2=\frac{a^2+b^2+c^2}{2}$，代入 $a^2+b^2+c^2=54$ 得 $x^2+y^2+z^2=27$。
			\par\textbf{理由：} 补形后长宽高平方和与对棱平方和的关系。
		\item[\textbf{第二步（均值不等式）：}] 对 $x^2,y^2,z^2$ 应用均值不等式，$\frac{x^2+y^2+z^2}{3}\geq (x^2y^2z^2)^{1/3}$，即 $\frac{27}{3}\geq (xyz)^{2/3}$，得 $(xyz)^{2/3}\leq 9$，所以 $xyz\leq 27$。当且仅当 $x=y=z$ 时等号成立。
			\par\textbf{理由：} 算术-几何平均不等式。
		\item[\textbf{第三步（求体积最值）：}]
			\[
				\begin{aligned}
					V &= \frac13 xyz \\
					&\leq \frac13\times27 \\
					&= 9.
			\end{aligned}
			\]
			当 $x=y=z=3$ 时取得最大值。此时 $a=\sqrt{y^2+z^2}=\sqrt{18}=3\sqrt{2}$，$b=c=3\sqrt{2}$，为正四面体。
			\par\textbf{理由：} 体积公式与不等式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{体积最大值为 $9$}。
