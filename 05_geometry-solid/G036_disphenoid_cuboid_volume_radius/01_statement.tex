\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（对棱相等）：}}] 四面体 $ABCD$ 的三组对棱分别相等，即 $AB=CD=a$, $AC=BD=b$, $AD=BC=c$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（补形关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将四面体补成长方体，长方体长宽高由对棱长度确定.
			\[
				y^2+z^2=a^2,\quad x^2+z^2=b^2,\quad x^2+y^2=c^2.
			\]
			解得
			\[
				x=\sqrt{\frac{b^2+c^2-a^2}{2}},\quad
				y=\sqrt{\frac{a^2+c^2-b^2}{2}},\quad
				z=\sqrt{\frac{a^2+b^2-c^2}{2}}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（体积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			四面体体积等于长方体体积的三分之一.
			\[
				V_{\text{四面体}}=\frac13 xyz.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（外接球半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			四面体的外接球即为长方体的外接球，半径等于长方体体对角线的一半.
			\[
				\begin{aligned}
					R &= \frac12\sqrt{x^2+y^2+z^2} \\
					&= \frac12\sqrt{\frac{a^2+b^2+c^2}{2}}.
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
