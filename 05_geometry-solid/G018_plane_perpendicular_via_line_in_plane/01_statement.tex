\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（线在面内）：}}]
			直线 $l$ 在平面 $\beta$ 内，即 $l\subset\beta$。
		\item[\textbf{\textcolor{CStatement}{条件 2（线垂直于面）：}}]
			直线 $l$ 垂直于平面 $\alpha$，即 $l\perp\alpha$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面面垂直判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两平面互相垂直，记作 $\alpha\perp\beta$。
			\[
				\alpha\perp\beta.
			\]
			\medskip
		\item[\textbf{\textcolor{CStatement}{推广结论（对称判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直线所在平面与被垂直平面互换，结论仍成立。
			\[
				l\subset\alpha,\ l\perp\beta \ \Rightarrow\ \alpha\perp\beta.
			\]
	\end{description}

\end{statementbox}
