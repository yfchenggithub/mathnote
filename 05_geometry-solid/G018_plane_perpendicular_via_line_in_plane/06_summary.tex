\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	若 $l\subset\beta$ 且 $l\perp\alpha$，则 $\alpha\perp\beta$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（线面垂直）：} 存在一条直线垂直于平面 $\alpha$。
		\item \textbf{条件 2（线在面内）：} 该直线在平面 $\beta$ 内。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（缺失线在面内）：} 使用定理时，必须验证直线确实在要证明的其中一个平面内，不能忽略。
		\item \textbf{检查项（线面垂直确认）：} 确认直线垂直于整个平面，即垂直于平面内两条相交直线。
	\end{itemize}

\end{summarybox}
