\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	在三棱锥 $P-ABC$ 中，$PA\perp$ 平面 $ABC$。求证：平面 $PAB\perp$ 平面 $ABC$。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}]
			由题知 $PA\perp$ 平面 $ABC$，且 $PA\subset$ 平面 $PAB$。
			\[
				PA\perp ABC,\quad PA\subset PAB.
			\]
			\par\textbf{理由：} 已知条件直接给出了线面垂直和线在面内的关系。
		\item[\textbf{第二步（套用定理）：}]
			根据面面垂直判定定理，因 $PA\perp ABC$ 且 $PA\subset PAB$，得 $PAB\perp ABC$。
			\[
				\left.
				\begin{aligned}
					PA &\subset PAB\\PA &\perp ABC
			\end{aligned}
				\right\}\ \Rightarrow\ PAB\perp ABC.
			\]
			\par\textbf{理由：} 直接应用 $l\subset\beta,\ l\perp\alpha \Rightarrow \alpha\perp\beta$。
		\item[\textbf{第三步（得出结论）：}]
			故平面 $PAB$ 与平面 $ABC$ 垂直。
			\[
				PAB\perp ABC.
			\]
			\par\textbf{理由：} 判定定理保证了结论成立。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	证明面面垂直时，只要看到一条直线同时满足“线面垂直”和“线在面内”，即可直接判定。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	已知 $AB$ 是圆 $O$ 的直径，$PA\perp$ 圆 $O$ 所在平面，$C$ 是圆上一点（$C\neq A,B$）。求证：平面 $PAC\perp$ 平面 $PBC$。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（目标分析）：}]
			要证平面 $PAC\perp$ 平面 $PBC$，由判定定理，需在平面 $PBC$ 内找一条直线垂直于平面 $PAC$。观察发现 $BC$ 可能满足条件。
			\[
				BC\subset PBC.
			\]
			\par\textbf{理由：} 拟定 $BC$ 为桥梁线。
		\item[\textbf{第二步（证明线面垂直）：}]
			因为 $PA\perp$ 平面 $ABC$，$BC\subset$ 平面 $ABC$，所以 $PA\perp BC$。又 $AB$ 是直径，$C$ 在圆上，所以 $AC\perp BC$。从而 $BC\perp PA$ 且 $BC\perp AC$，且 $PA\cap AC=A$，所以 $BC\perp$ 平面 $PAC$。
			\[
				PA\perp BC,\quad AC\perp BC.
			\]
			\par\textbf{理由：} 线面垂直性质得 $PA\perp BC$；直径所对圆周角为直角得 $AC\perp BC$；由线面垂直判定定理得 $BC\perp$ 平面 $PAC$。
		\item[\textbf{第三步（套用定理）：}]
			由 $BC\perp$ 平面 $PAC$ 且 $BC\subset$ 平面 $PBC$，根据面面垂直判定定理，得平面 $PAC\perp$ 平面 $PBC$。
			\[
				\left.
				\begin{aligned}
					BC &\subset PBC\\BC &\perp PAC
			\end{aligned}
				\right\}\ \Rightarrow\ PAC\perp PBC.
			\]
			\par\textbf{理由：} 面面垂直判定定理。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	本题先利用圆的性质和线面垂直证明 $BC\perp$ 平面 $PAC$，再利用判定定理得到面面垂直。体现了“先证线面垂直，再得面面垂直”的常见思路。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\medskip
	\textbf{\textcolor{CExample}{题目：}}
	在四棱锥 $P-ABCD$ 中，底面 $ABCD$ 是菱形，$\angle DAB=60^\circ$，$PA\perp$ 底面 $ABCD$。求证：平面 $PBD\perp$ 平面 $PAC$。

	\medskip
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（目标分析）：}]
			要证平面 $PBD\perp$ 平面 $PAC$，观察发现在 $PBD$ 内的直线 $BD$ 可能垂直于 $PAC$。
			\[
				BD\subset PBD.
			\]
			\par\textbf{理由：} 尝试利用条件中的菱形对角线关系。
		\item[\textbf{第二步（证明线面垂直）：}]
			因为 $ABCD$ 是菱形，所以 $BD\perp AC$。又 $PA\perp$ 底面 $ABCD$，$BD\subset$ 底面，故 $PA\perp BD$。于是 $BD\perp PA$ 且 $BD\perp AC$，且 $PA\cap AC=A$，所以 $BD\perp$ 平面 $PAC$。
			\[
				BD\perp AC,\quad PA\perp BD.
			\]
			\par\textbf{理由：} 菱形对角线互相垂直；$PA\perp$ 底面得 $PA\perp BD$；由线面垂直判定定理得 $BD\perp$ 平面 $PAC$。
		\item[\textbf{第三步（套用定理）：}]
			由 $BD\perp$ 平面 $PAC$ 且 $BD\subset$ 平面 $PBD$，得平面 $PBD\perp$ 平面 $PAC$。
			\[
				\left.
				\begin{aligned}
					BD &\subset PBD\\BD &\perp PAC
			\end{aligned}
				\right\}\ \Rightarrow\ PBD\perp PAC.
			\]
			\par\textbf{理由：} 面面垂直判定定理。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	本题利用菱形的对角线垂直和 $PA\perp$ 底面，先证明 $BD\perp$ 平面 $PAC$，再推出面面垂直。
\end{examplebox}
