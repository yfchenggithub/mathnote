\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（存在内切球）：}}] 多面体存在一个球，与该多面体的所有面都相切。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（内切球半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知多面体体积和表面积，可直接求出内切球半径。
			\[
				\boxed{r = \frac{3V}{S}}
			\]。

		\item[\textbf{\textcolor{CStatement}{推广结论（变形公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知内切球半径和表面积，可求体积。
			\[
				V = \frac{1}{3} r S
			\]
	\end{description}
\end{statementbox}
