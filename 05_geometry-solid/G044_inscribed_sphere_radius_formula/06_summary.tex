\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{存在内切球的多面体，其内切球半径 $r = 3V/S$，可从体积和表面积直接求出。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（存在内切球）：} 多面体必须有一个球与所有面相切。
		\item \textbf{条件 2（已知两量）：} 已知体积 $V$ 和表面积 $S$，或者能通过其他条件求得。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（存在条件）：} 使用前必须确认多面体有内切球。
		\item \textbf{检查项（量纲一致性）：} 检查体积和表面积是否基于同一长度单位，防止数值错误。
	\end{itemize}
\end{summarybox}
