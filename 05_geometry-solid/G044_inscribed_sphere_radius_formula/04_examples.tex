\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一个正方体的棱长为 $2$，求其内切球半径 $r$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算体积）：}] 正方体体积公式 $V = a^3$（$a$ 为棱长），代入棱长 $a=2$，得
			\[
				V = 2^3
			\]。
			计算得
			\[
				V = 8
			\]。
			\par\textbf{理由：} 正方体体积公式。
		\item[\textbf{第二步（计算表面积）：}] 正方体表面积公式 $S = 6a^2$（$a$ 为棱长），代入 $a=2$，得
			\[
				S = 6 \times 4
			\]。
			计算得
			\[
				S = 24
			\]。
			\par\textbf{理由：} 正方体有 $6$ 个全等的正方形面。
		\item[\textbf{第三步（套用公式）：}] 由 $r = \dfrac{3V}{S}$，代入 $V=8$，$S=24$，得
			\[
				r = \frac{3 \times 8}{24}
			\]。
			计算得
			\[
				r = 1
			\]。
			\par\textbf{理由：} 该多面体存在内切球，满足公式使用条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$r = 1$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知某存在内切球的多面体，其内切球半径 $r = 3$，表面积 $S = 54$，求体积 $V$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变形公式）：}] 由 $r = \dfrac{3V}{S}$ 变形得 $V = \dfrac{rS}{3}$。
			\par\textbf{理由：} 等式两边同乘 $S$ 再除以 $3$。
		\item[\textbf{第二步（代入求值）：}] 将 $r=3$，$S=54$ 代入 $V = \dfrac{rS}{3}$，得
			\[
				V = \frac{3 \times 54}{3}
			\]。
			计算得
			\[
				V = 54
			\]。
			\par\textbf{理由：} 算术运算。
		\item[\textbf{第三步（结果确认）：}] 体积为 $54$，单位一致，符合预期。
			\par\textbf{理由：} 验证条件，公式适用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$V = 54$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一个正四面体（存在内切球）的棱长为 $2$，求其内切球半径 $r$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求体积）：}] 正四面体体积公式 $V = \dfrac{\sqrt{2}}{12} a^3$，代入 $a=2$，得
			\[
				V = \frac{\sqrt{2}}{12} \times 8
			\]。
			计算得
			\[
				V = \frac{2\sqrt{2}}{3}
			\]。
			\par\textbf{理由：} 正四面体体积公式。
		\item[\textbf{第二步（求表面积）：}] 正四面体每个面是等边三角形，面积 $\dfrac{\sqrt{3}}{4} a^2$，故表面积
			\[
				S = 4 \times \frac{\sqrt{3}}{4} a^2
			\]。
			化简得
			\[
				S = \sqrt{3} a^2
			\]。
			代入 $a=2$，得
			\[
				S = \sqrt{3} \times 4
			\]。
			计算得
			\[
				S = 4\sqrt{3}
			\]。
			\par\textbf{理由：} 等边三角形面积公式。
		\item[\textbf{第三步（套用公式）：}] 由 $r = \dfrac{3V}{S}$，代入 $V = \dfrac{2\sqrt{2}}{3}$，$S = 4\sqrt{3}$，得
			\[
				r = \frac{3 \times \frac{2\sqrt{2}}{3}}{4\sqrt{3}}
			\]。
			化简分子：
			\[
				r = \frac{2\sqrt{2}}{4\sqrt{3}}
			\]。
			约分：
			\[
				r = \frac{\sqrt{2}}{2\sqrt{3}}
			\]。
			有理化分母：
			\[
				r = \frac{\sqrt{6}}{6}
			\]。
			\par\textbf{理由：} 逐步代数化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$r = \dfrac{\sqrt{6}}{6}$}。
\end{examplebox}
