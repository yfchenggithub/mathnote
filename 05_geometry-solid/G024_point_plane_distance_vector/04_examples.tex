\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知平面 $\alpha$ 过点 $A(1,2,3)$，法向量 $\mathbf{n}=(2,-1,2)$，求点 $P(4,5,6)$ 到平面 $\alpha$ 的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算向量 $\vec{AP}$）：}]
			由 $A(1,2,3)$，$P(4,5,6)$，得 $\vec{AP}=(3,3,3)$。
			\par\textbf{理由：} 终点坐标减起点坐标。
		\item[\textbf{第二步（计算点积与模长）：}]
			点积：
			\[
				\begin{aligned}
					\vec{AP}\cdot\mathbf{n} &= 3\times2+3\times(-1)+3\times2 \\&= 6-3+6 \\&= 9.
			\end{aligned}
			\]
			模长：
			\[
				\begin{aligned}
					|\mathbf{n}|^2 &= 2^2+(-1)^2+2^2 \\&= 4+1+4 \\&= 9,
			\end{aligned}
			\]
			所以 $|\mathbf{n}|=3$。
			\par\textbf{理由：} 向量点积公式和模长公式。
		\item[\textbf{第三步（代入距离公式）：}]
			$d=\dfrac{|\vec{AP}\cdot\mathbf{n}|}{|\mathbf{n}|}=\dfrac{|9|}{3}=3$。
			\par\textbf{理由：} 点到平面距离的向量公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	点 $P$ 到平面 $\alpha$ 的距离为 \textbf{$3$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知平面 $\alpha$ 的方程为 $2x-2y+z+5=0$，求点 $P(1,2,3)$ 到平面 $\alpha$ 的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定法向量与平面上一点）：}]
			由平面方程得法向量 $\mathbf{n}=(2,-2,1)$。取 $x=0,y=0$ 代入方程得 $z=-5$，故平面上一点 $A(0,0,-5)$。
			\par\textbf{理由：} 法向量系数对应；平面方程特殊解。
		\item[\textbf{第二步（计算 $\vec{AP}$ 与点积）：}]
			$\vec{AP}=(1,2,8)$。点积：
			\[
				\begin{aligned}
					\vec{AP}\cdot\mathbf{n} &= 1\times2+2\times(-2)+8\times1 \\&= 2-4+8 \\&= 6.
			\end{aligned}
			\]
			模长：
			\[
				\begin{aligned}
					|\mathbf{n}|^2 &= 2^2+(-2)^2+1^2 \\&= 4+4+1 \\&= 9,
			\end{aligned}
			\]
			所以 $|\mathbf{n}|=3$。
			\par\textbf{理由：} 向量减法与点积运算。
		\item[\textbf{第三步（代入公式求距离）：}]
			$d=\dfrac{|6|}{3}=2$。
			\par\textbf{理由：} 距离公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	点 $P$ 到平面 $\alpha$ 的距离为 \textbf{$2$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在空间直角坐标系中，设平面 $\alpha$ 过三点 $A(1,0,0), B(0,1,0), C(0,0,1)$。求原点 $O(0,0,0)$ 到平面 $\alpha$ 的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求法向量）：}]
			计算 $\overrightarrow{AB}=(-1,1,0)$，$\overrightarrow{AC}=(-1,0,1)$。则法向量
			\[
				\begin{aligned}
					\mathbf{n} &= \overrightarrow{AB}\times\overrightarrow{AC} \\ &=
					\begin{vmatrix}
						\mathbf{i} & \mathbf{j} & \mathbf{k}\\ -1 & 1 & 0\\ -1 & 0 & 1
				\end{vmatrix}
					\\ &= (1,1,1).
			\end{aligned}
			\]
			\par\textbf{理由：} 利用向量积求平面的一个法向量。
		\item[\textbf{第二步（选取平面上一点并计算 $\vec{AO}$）：}]
			取 $A(1,0,0)$，$\vec{AO}=(-1,0,0)$。
			\par\textbf{理由：} $O$ 到 $A$ 的向量。
		\item[\textbf{第三步（代入公式求距离）：}]
			点积：
			\[
				\begin{aligned}
					\vec{AO}\cdot\mathbf{n} &= (-1)\times1+0\times1+0\times1 \\&= -1.
			\end{aligned}
			\]
			模长 $|\mathbf{n}|=\sqrt{3}$。代入公式：$d=\dfrac{|-1|}{\sqrt{3}}=\dfrac{\sqrt{3}}{3}$。
			\par\textbf{理由：} 距离公式，注意绝对值。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	原点 $O$ 到平面 $\alpha$ 的距离为 \textbf{$\dfrac{\sqrt{3}}{3}$}。
