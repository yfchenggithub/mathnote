\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记错）：}}] 认为正四面体的高等于$\frac{\sqrt{3}}{2}a$（等边三角形的高）。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式对比）：}}] 正四面体的高应为$h = \frac{\sqrt{6}}{3}a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆概念）：}}] 将正三角形的高与正四面体的高混淆，忘记正四面体的高需要空间勾股计算。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（比例颠倒）：}}] 误以为内切球球心将高分为$1:3$（顶点到球心:球心到底面），从而认为顶点到球心距离$=r$。
		\item[\textbf{\textcolor{CTrap}{正确理解（比例校正）：}}] 正确比例为$3:1$，球心到底面距离$=r$，顶点到球心距离$=3r$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆错误）：}}] 内切球与外接球性质混淆，或对定义理解不清。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（系数记错）：}}] 误记内切球半径$r = \frac{\sqrt{6}}{6}a$或$r = \frac{\sqrt{3}}{12}a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确系数）：}}] 正确公式为$r = \frac{\sqrt{6}}{12}a$，可由体积法推导验证。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导遗漏）：}}] 计算出错，忽略系数$\frac{1}{4}$或误用其他几何体的半径公式。
	\end{description}
\end{trapbox}
