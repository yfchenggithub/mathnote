\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（正四面体）：}}] 棱长为$a$的正四面体。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（高公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从顶点到底面中心的距离。
			\[
				h = \frac{\sqrt{6}}{3}a
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（内切球半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			内切球球心到任一面的距离。
			\[
				r = \frac{\sqrt{6}}{12}a
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（球心位置）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			球心在高上，将高分为$3:1$（上长下短）。
			\[
				\frac{h - r}{r} = 3
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{推广结论（半径与高的关系）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{推广结论一（内切球半径与高）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			内切球半径等于高的四分之一。
			\[
				r = \frac{h}{4}
			\]
	\end{description}
\end{statementbox}
