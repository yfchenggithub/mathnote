\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（圆柱参数）：}}]
			圆柱的底面半径为$R>0$，高为$h>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（充要条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆柱存在内切球当且仅当高等于底面直径。
			\[
				h = 2R.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（几何性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			内切球半径等于底面半径，且轴截面为正方形。
			\[
				r = R.
			\]
			轴截面为边长为$2R$的正方形。
		\item[\textbf{\textcolor{CStatement}{推广结论（常见公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			存在内切球时，圆柱体积和表面积可用$R$表示。
			\[
				V = 2\pi R^{3},
				\qquad
				S = 6\pi R^{2}.
			\]
	\end{description}

\end{statementbox}
