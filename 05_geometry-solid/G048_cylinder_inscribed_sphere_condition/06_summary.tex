\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	圆柱存在内切球 $\iff$ $h=2R$，且此时球半径 $r=R$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（已知参数）：} 已知圆柱底面半径 $R$ 和高 $h$。
		\item \textbf{条件2（问题类型）：} 涉及内切球存在性或相关计算。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件混淆）：} 条件是 $h=2R$，不是 $h=R$。
		\item \textbf{检查项（轴截面）：} 轴截面是否为正方形可快速判断。
	\end{itemize}

\end{summarybox}
