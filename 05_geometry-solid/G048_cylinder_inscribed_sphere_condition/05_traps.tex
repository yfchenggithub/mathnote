\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件混淆）}}]
			错误认为圆柱有内切球的条件是 $h=R$。
		\item[\textbf{\textcolor{CTrap}{正确理解（直径关系）：}}]
			正确条件应该是 $h=2R$，即高等于底面直径。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}]
			将“直径”误记为“半径”，导致等式错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（球心定位）}}]
			错误认为内切球球心可能不在圆柱轴的中点上。
		\item[\textbf{\textcolor{CTrap}{正确理解（中点对称）：}}]
			内切球同时与两底面相切，所以球心到两底等距，必在轴线中点。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略对称）：}}]
			未考虑到底面全等对称性，凭直觉认为球心可能偏上或偏下。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（矩形误解）}}]
			错误认为圆柱轴截面是矩形就有内切球。
		\item[\textbf{\textcolor{CTrap}{正确理解（正方形条件）：}}]
			必须轴截面为正方形（$h=2R$）时才有内切球。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件忽略）：}}]
			未意识到矩形存在内切圆需要邻边相等。
	\end{description}

\end{trapbox}
