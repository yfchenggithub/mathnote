\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（垂于两相交线）：}}]
			直线$l$垂直于平面$\alpha$内的两条直线$a$和$b$，且$a$与$b$相交于点$P$，即$l\perp a$，$l\perp b$，$a\cap b=P$，$a\subset\alpha$，$b\subset\alpha$.
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线面垂直判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			一条直线同时垂直于平面内两条相交直线，则该直线垂直于整个平面.\
			\[
				l\perp a,\; l\perp b,\; a\cap b=P,\; a\subset\alpha,\; b\subset\alpha \quad\Rightarrow\quad l\perp\alpha.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（向量等价表述）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直线的方向向量若与平面内两个不共线向量都垂直，则该向量垂直于平面.\
			\[
				\vec{v}\cdot\vec{a}=0,\; \vec{v}\cdot\vec{b}=0,\; \vec{a}\not\parallel\vec{b} \quad\Rightarrow\quad \vec{v}\perp\alpha.
			\]
	\end{description}

\end{statementbox}
