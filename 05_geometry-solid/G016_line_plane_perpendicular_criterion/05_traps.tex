\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略相交条件）}}]
			有些学生认为若直线$l$垂直于平面$\alpha$内的两条平行直线$m$和$n$，则可推出$l\perp\alpha$。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须相交）：}}]
			定理要求两条直线必须相交。平行直线只能确定一个方向，无法保证直线垂直于整个平面。
		\item[\textbf{\textcolor{CTrap}{错因分析（方向不完整）：}}]
			两条平行直线的方向相同，平面内其他方向可能与$l$不垂直，因此不能推出$l\perp\alpha$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（只垂一直线）}}]
			学生只证明直线$l$垂直于平面$\alpha$内的一条直线$a$，就断言$l\perp\alpha$。
		\item[\textbf{\textcolor{CTrap}{正确理解（需两相交线）：}}]
			必须证明$l$垂直于平面内两条相交直线，仅一条垂直不能确定线面垂直。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺第二个方向）：}}]
			一条直线仅确定一个方向，平面内存在与该方向平行的直线，$l$可能不与其垂直。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（误用平面外直线）}}]
			学生将平面$\alpha$外的一条直线$b$当作平面内的直线，用$l\perp b$作为条件。
		\item[\textbf{\textcolor{CTrap}{正确理解（直线必在面内）：}}]
			定理要求两条直线都必须在平面$\alpha$内，平面外的直线不能用于判定。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆面内外）：}}]
			平面外的直线可能与平面成任意角度，即使$l$垂直它，也不代表$l$与平面内所有直线垂直。
	\end{description}

\end{trapbox}
