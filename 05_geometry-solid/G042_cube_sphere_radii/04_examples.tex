	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知正方体的棱长为$4$，求它的内切球半径$r_i$、棱切球半径$r_m$、外接球半径$r_o$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套内切球公式）：}] 代入公式$r_i = \frac{a}{2}$，$a=4$，得$r_i=2$。
			\par\textbf{理由：}
			内切球半径公式直接应用。
		\item[\textbf{第二步（套棱切球公式）：}] 代入公式$r_m = \frac{\sqrt{2}}{2}a$，$a=4$，得$r_m=2\sqrt{2}$。
			\par\textbf{理由：}
			棱切球半径公式直接应用。
		\item[\textbf{第三步（套外接球公式）：}] 代入公式$r_o = \frac{\sqrt{3}}{2}a$，$a=4$，得$r_o=2\sqrt{3}$。
			\par\textbf{理由：}
			外接球半径公式直接应用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$r_i=2$，$r_m=2\sqrt{2}$，$r_o=2\sqrt{3}$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知正方体的外接球半径为$\sqrt{3}$，求正方体的棱长$a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写公式）：}] 外接球半径公式$r_o = \frac{\sqrt{3}}{2}a$，代入$r_o=\sqrt{3}$。
			\par\textbf{理由：}
			外接球半径公式。
		\item[\textbf{第二步（解方程）：}] 由$\sqrt{3} = \frac{\sqrt{3}}{2}a$，两边除以$\sqrt{3}$得$1=\frac{a}{2}$，所以$a=2$。
			\par\textbf{理由：}
			代数方程求解。
		\item[\textbf{第三步（验证）：}] 将$a=2$代回原公式，$r_o=\frac{\sqrt{3}}{2}\times2=\sqrt{3}$，与已知一致。
			\par\textbf{理由：}
			验证答案正确性。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{棱长$a=2$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	若一个正方体的内切球表面积为$4\pi$，求该正方体的外接球体积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求内切球半径）：}] 由表面积公式$S=4\pi r_i^2$，$S=4\pi$，得$4\pi r_i^2=4\pi$，所以$r_i=1$。
			\par\textbf{理由：}
			球表面积公式。
		\item[\textbf{第二步（求棱长）：}] 内切球半径与棱长关系$r_i = \frac{a}{2}$，代入$r_i=1$得$a=2$。
			\par\textbf{理由：}
			内切球半径公式反推。
		\item[\textbf{第三步（求外接球半径与体积）：}] 外接球半径
			\[
				\begin{aligned}
					r_o &= \frac{\sqrt{3}}{2}a \\
					&= \frac{\sqrt{3}}{2}\times2 \\
					&= \sqrt{3};
			\end{aligned}
			\]
			体积
			\[
				\begin{aligned}
					V &= \frac{4}{3}\pi r_o^3 \\
					&= \frac{4}{3}\pi (\sqrt{3})^3 \\
					&= 4\sqrt{3}\pi.
			\end{aligned}
			\]
			\par\textbf{理由：}
			外接球半径公式与球体积公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{外接球体积为$4\sqrt{3}\pi$}。
