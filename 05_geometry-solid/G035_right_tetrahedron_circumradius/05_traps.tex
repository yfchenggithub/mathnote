	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件遗漏）}}]
			认为任意三棱锥均适用该公式.
		\item[\textbf{\textcolor{CTrap}{正确理解（必要条件）}}]
			仅当从同一顶点出发的三条侧棱两两垂直时才成立.
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略前提）}}]
			未注意公式的严格条件，随意套用.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（球心误判）}}]
			认为球心在底面三角形 $ABC$ 的外心处.
		\item[\textbf{\textcolor{CTrap}{正确理解（球心位置）}}]
			球心是补形后长方体的体对角线中点.
		\item[\textbf{\textcolor{CTrap}{错因分析（类比错误）}}]
			将平面几何中外心概念错误推广至空间.
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式记错）}}]
			记成 $R = \sqrt{a^2+b^2+c^2}$ 或 $R = \frac12(a+b+c)$.
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式）}}]
			$R = \frac12\sqrt{a^2+b^2+c^2}$.
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆半径直径）}}]
			忽略外接球直径与体对角线的关系，或误用算术平均.
	\end{description}

\end{trapbox}
