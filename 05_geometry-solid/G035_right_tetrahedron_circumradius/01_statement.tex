\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两两垂直）：}}]
			三棱锥从同一顶点出发的三条侧棱两两垂直，分别长 $a,b,c$.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外接球半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外接球半径等于以 $a,b,c$ 为棱长的长方体体对角线的一半.
			\[
				R = \frac{1}{2}\sqrt{a^{2}+b^{2}+c^{2}}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（补形法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将墙角三棱锥补形成长方体，外接球心即为长方体中心，半径同公式.
			\[
				R = \frac{1}{2}\sqrt{a^{2}+b^{2}+c^{2}}.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{适用范围：}}
	适用于所有从同一顶点出发的三条棱两两垂直的三棱锥，常见于墙角模型、直三棱锥.
\end{statementbox}
