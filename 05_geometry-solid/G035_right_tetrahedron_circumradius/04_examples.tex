\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	若三棱锥 $P$-$ABC$ 中，$PA \perp PB,\; PB \perp PC,\; PC \perp PA$，且 $PA = 3,\; PB = 4,\; PC = 5$，求其外接球半径 $R$.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别模型）：}]
			判断三条棱两两垂直，符合墙角模型条件.
			\[
				PA \perp PB,\; PB \perp PC,\; PC \perp PA.
			\]
			\par\textbf{理由：} 题设条件直接满足，可直接应用公式.
		\item[\textbf{第二步（套用公式）：}]
			代入墙角模型外接球半径公式.
			\[
				R = \frac{1}{2}\sqrt{a^{2}+b^{2}+c^{2}}.
			\]
			\par\textbf{理由：} 墙角模型结论.
		\item[\textbf{第三步（计算数值）：}]
			以 $a=3,\;b=4,\;c=5$ 代入.
			\[
				\begin{aligned}
					R &= \frac{1}{2}\sqrt{3^{2}+4^{2}+5^{2}} \\ &= \frac{1}{2}\sqrt{9+16+25} \\ &= \frac{1}{2}\sqrt{50} \\ &= \frac{5\sqrt{2}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 依次计算平方和，开方，乘以 $\frac12$.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R = \dfrac{5\sqrt{2}}{2}$}.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知三棱锥 $P$-$ABC$ 中，$PA \perp PB,\; PB \perp PC,\; PC \perp PA$，且 $PA = 2,\; PB = 3$，外接球半径 $R = \dfrac{7}{2}$，求 $PC$ 的长.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}]
			设 $PC = x$.
			\[
				PC = x.
			\]
			\par\textbf{理由：} 未知量用 $x$ 表示.
		\item[\textbf{第二步（列方程）：}]
			由墙角模型公式，得到关于 $x$ 的方程.
			\[
				\frac{1}{2}\sqrt{2^{2}+3^{2}+x^{2}} = \frac{7}{2}.
			\]
			\par\textbf{理由：} 代入 $a=2,\;b=3,\;c=x$ 和 $R=\frac72$.
		\item[\textbf{第三步（解方程）：}]
			两边乘以 $2$ 得 $\sqrt{13+x^{2}} = 7$，再平方得 $13+x^{2}=49$，解得 $x^{2}=36$.
			\[
				\begin{aligned}
					\sqrt{13+x^{2}} &= 7 \\ 13+x^{2} &= 49 \\ x^{2} &= 36.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方去根号，注意 $x>0$，所以 $x=6$.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PC = 6$}.

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	设三棱锥 $P$-$ABC$ 的三条侧棱 $PA,PB,PC$ 两两垂直，且长度 $a,b,c$ 均为正数，若 $a+b+c = 9$，求其外接球半径 $R$ 的最小值.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（半径公式）：}]
			写出 $R$ 与 $a,b,c$ 的关系.
			\[
				R = \frac{1}{2}\sqrt{a^{2}+b^{2}+c^{2}}.
			\]
			\par\textbf{理由：} 墙角模型外接球半径公式.
		\item[\textbf{第二步（不等式放缩）：}]
			利用不等式 $a^{2}+b^{2}+c^{2} \ge \dfrac{(a+b+c)^{2}}{3}$ 放缩.
			\[
				\begin{aligned}
					R &\ge \frac{1}{2}\sqrt{\frac{(a+b+c)^{2}}{3}} \\ &= \frac{a+b+c}{2\sqrt{3}}.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方平均 $\ge$ 算术平均（或柯西不等式），等号当 $a=b=c$ 时成立.
		\item[\textbf{第三步（代入定值）：}]
			将 $a+b+c = 9$ 代入上式.
			\[
				\begin{aligned}
					R_{\min} &= \frac{9}{2\sqrt{3}} \\ &= \frac{3\sqrt{3}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入并化简，得到最小半径.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R_{\min} = \dfrac{3\sqrt{3}}{2}$（当 $a=b=c=3$ 时取得）}.
\end{examplebox}
