\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（棱上垂线）：}}] 在二面角 $\alpha\text{-}l\text{-}\beta$ 的棱 $l$ 上任取一点 $O$，分别在半平面 $\alpha$ 内作 $OA\perp l$，$\beta$ 内作 $OB\perp l$，则 $OA$、$OB$、$AB$ 为可测线段。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（定义求角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			二面角的大小等于两条垂线所夹的角。
			\[
				\theta = \angle AOB,\quad \cos\theta = \frac{OA^2 + OB^2 - AB^2}{2\cdot OA \cdot OB}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（余弦反求）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知二面角可反求棱上两点距离。
			\[
				AB^2 = OA^2 + OB^2 - 2\cdot OA\cdot OB\cdot \cos\theta .
			\]
	\end{description}
\end{statementbox}
