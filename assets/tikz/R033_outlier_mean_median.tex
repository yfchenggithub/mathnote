% 单个极端值对平均数与中位数的不同影响
% 用法：\input{assets/tikz/R033_outlier_mean_median.tex}

\begin{center}
	\begin{tikzpicture}[
		x=0.49cm,y=1cm,>=Stealth,
		line cap=round,line join=round,
		every node/.style={font=\small}
	]
		\path[use as bounding box] (0,0) rectangle (24.8,5.25);
		\fill[blue!3,rounded corners=7pt] (0.3,0.15) rectangle (24.5,5.05);
		\draw[CExplanation,rounded corners=7pt,line width=0.85pt]
			(0.3,0.15) rectangle (24.5,5.05);
		\node[font=\bfseries,text=CExplanation] at (12.4,4.55)
			{只改动一个极端值：平均数会被拉动，中位数仍由位置决定};

		% 两条数轴必须同尺度
		\draw[-Stealth,black!62,line width=0.75pt] (1,3.18)--(22.1,3.18);
		\draw[-Stealth,black!62,line width=0.75pt] (1,1.16)--(22.1,1.16);
		\foreach \x in {1,3,5,6,10,15,20}{
			\draw[black!45] (\x,3.09)--(\x,3.27);
			\draw[black!45] (\x,1.07)--(\x,1.25);
			\node[font=\scriptsize,below=3pt] at (\x,1.14) {$\x$};
		}
		\foreach \x in {1,2,3,4,5}{\fill[teal!68!black] (\x,3.50) circle (4.8pt);}
		\foreach \x in {1,2,3,4,20}{\fill[teal!68!black] (\x,1.48) circle (4.8pt);}
		\draw[orange!85!black,line width=0.9pt] (5,3.50) circle (7.2pt);
		\draw[orange!85!black,line width=0.9pt] (20,1.48) circle (7.2pt);

		\draw[red!78!black,line width=1pt] (3,3.18)--(3,3.94);
		\draw[blue!78!black,line width=1pt,dashed] (3,2.63)--(3,3.18);
		\node[anchor=west,text=black!75] at (7.1,4.03)
			{原数据 $1,2,3,4,5$};
		\node[anchor=west,text=red!78!black] at (7.1,3.56)
			{平均数 $=$ 中位数 $=3$};

		\draw[red!78!black,line width=1pt] (6,1.16)--(6,1.65);
		\draw[blue!78!black,line width=1pt,dashed] (3,1.16)--(3,1.65);
		\node[anchor=west,text=black!75] at (7.1,2.06)
			{只把 $5$ 改成 $20$};
		\node[anchor=west,text=red!78!black] at (7.1,1.61)
			{平均数移到 $6$};
		\node[anchor=west,text=blue!78!black] at (12.2,1.61)
			{中位数仍为 $3$};

		\node[font=\scriptsize,text=black!62,anchor=west] at (1.0,0.48)
			{红实线：平均数；蓝虚线：中位数。两行使用完全相同的数轴尺度。};
	\end{tikzpicture}
\end{center}
