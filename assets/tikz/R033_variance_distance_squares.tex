% 方差：离均值的距离经过平方后取平均
% 用法：\input{assets/tikz/R033_variance_distance_squares.tex}

\begin{center}
	\begin{tikzpicture}[
		x=1cm,y=1cm,>=Stealth,
		line cap=round,line join=round,
		every node/.style={font=\small}
	]
		\path[use as bounding box] (0,0) rectangle (12.5,6.45);
		\fill[teal!3,rounded corners=7pt] (0.2,0.15) rectangle (12.3,6.25);
		\draw[teal!60!black,rounded corners=7pt,line width=0.85pt]
			(0.2,0.15) rectangle (12.3,6.25);
		\node[font=\bfseries\large,text=teal!70!black] at (6.25,5.90)
			{方差把“离中心多远”变成可相加的平方面积};

		\draw[-Stealth,black!65,line width=0.8pt] (0.85,4.12)--(11.75,4.12);
		\foreach \x/\lab in {1.4/1,3.8/2,6.2/3,8.6/4,11.0/5}{
			\draw[black!50] (\x,4.02)--(\x,4.22);
			\fill[blue!70!black] (\x,4.45) circle (4.7pt);
			\node[below=4pt] at (\x,4.08) {$\lab$};
		}
		\draw[red!78!black,dashed,line width=0.9pt] (6.2,3.70)--(6.2,5.05);
		\node[font=\bfseries,text=red!78!black] at (6.2,5.31) {$\bar x=3$};
		\draw[<->,orange!85!black,line width=0.8pt] (1.4,4.83)--(6.2,4.83)
			node[midway,above=1pt,font=\scriptsize] {距离 $2$};
		\draw[<->,orange!85!black,line width=0.8pt] (6.2,4.83)--(11.0,4.83)
			node[midway,above=1pt,font=\scriptsize] {距离 $2$};

		% 平方面积与实际数值成比例：2^2 用 1.28cm 正方形，1^2 用 0.64cm 正方形。
		\foreach \x/\side/\area in {1.4/1.28/4,3.8/0.64/1,8.6/0.64/1,11.0/1.28/4}{
			\fill[orange!16] ({\x-\side/2},1.78) rectangle ({\x+\side/2},{1.78+\side});
			\draw[orange!80!black,line width=0.75pt]
				({\x-\side/2},1.78) rectangle ({\x+\side/2},{1.78+\side});
			\node[font=\bfseries,text=orange!80!black] at (\x,{1.78+\side/2}) {$\area$};
		}
		\fill[orange!80!black] (6.2,1.86) circle (2.2pt);
		\node[font=\bfseries,text=orange!80!black] at (6.2,2.30) {$0$};
		\foreach \x/\lab in {1.4/{(-2)^2},3.8/{(-1)^2},6.2/{0^2},8.6/{1^2},11.0/{2^2}}{
			\node[font=\scriptsize] at (\x,1.38) {$\lab$};
		}
		\node[font=\bfseries,text=teal!72!black] at (6.2,0.62)
			{$s^2=\dfrac{4+1+0+1+4}{5}=2$};
		\node[font=\scriptsize,text=black!62,rotate=90] at (0.58,2.42) {平方面积};
	\end{tikzpicture}
\end{center}
