% R030：非整数期望的直觉图
% 用 X^{(j)} 表示第 j 次完整抽样的结果，避免与位置指示变量 X_i 混淆。

\begin{center}
	\begin{tikzpicture}[
		x=1cm,
		y=1cm,
		>=Stealth,
		line cap=round,
		line join=round,
		every node/.style={font=\normalsize},
		target/.style={
			circle, draw=orange!85!black, fill=orange!58,
			line width=0.8pt, minimum size=0.42cm, inner sep=0pt
		},
		other/.style={
			circle, draw=black!55, fill=black!18,
			line width=0.8pt, minimum size=0.42cm, inner sep=0pt
		},
		sample/.style={
			draw=blue!70!black, fill=blue!3, rounded corners=7pt,
			line width=0.95pt
		}
	]
		\path[use as bounding box] (0,-0.50) rectangle (12.4,6.65);
		\node[font=\large\bfseries, text=blue!70!black] at (6.20,6.32)
			{四次完整抽样的示意结果（每次样本量 $n=6$）};

		% 第 1 次：1 橙 + 5 灰。
		\draw[sample] (0.12,2.80) rectangle (3.00,5.95);
		\node[font=\normalsize\bfseries] at (1.56,5.43) {第 $1$ 次};
		\node[font=\large\bfseries] at (1.56,4.85) {$X^{(1)}=1$};
		\node[target] at (0.72,4.02) {};
		\foreach \x in {1.56,2.40}{\node[other] at (\x,4.02) {};}
		\foreach \x in {0.72,1.56,2.40}{\node[other] at (\x,3.30) {};}

		% 第 2 次：2 橙 + 4 灰。
		\draw[sample] (3.22,2.80) rectangle (6.10,5.95);
		\node[font=\normalsize\bfseries] at (4.66,5.43) {第 $2$ 次};
		\node[font=\large\bfseries] at (4.66,4.85) {$X^{(2)}=2$};
		\foreach \x in {3.82,4.66}{\node[target] at (\x,4.02) {};}
		\node[other] at (5.50,4.02) {};
		\foreach \x in {3.82,4.66,5.50}{\node[other] at (\x,3.30) {};}

		% 第 3 次：1 橙 + 5 灰。
		\draw[sample] (6.32,2.80) rectangle (9.20,5.95);
		\node[font=\normalsize\bfseries] at (7.76,5.43) {第 $3$ 次};
		\node[font=\large\bfseries] at (7.76,4.85) {$X^{(3)}=1$};
		\node[target] at (6.92,4.02) {};
		\foreach \x in {7.76,8.60}{\node[other] at (\x,4.02) {};}
		\foreach \x in {6.92,7.76,8.60}{\node[other] at (\x,3.30) {};}

		% 第 4 次：3 橙 + 3 灰。
		\draw[sample] (9.42,2.80) rectangle (12.30,5.95);
		\node[font=\normalsize\bfseries] at (10.86,5.43) {第 $4$ 次};
		\node[font=\large\bfseries] at (10.86,4.85) {$X^{(4)}=3$};
		\foreach \x in {10.02,10.86,11.70}{\node[target] at (\x,4.02) {};}
		\foreach \x in {10.02,10.86,11.70}{\node[other] at (\x,3.30) {};}

		\draw[draw=blue!70!black, fill=white, rounded corners=7pt, line width=1pt]
			(0.28,-0.40) rectangle (12.12,2.20);
		\node[font=\large\bfseries, text=blue!70!black] at (6.20,1.65)
			{$(1+2+1+3)/4=7/4=1.75$};
		\node[font=\normalsize\bfseries] at (6.20,0.80)
			{每一次抽到的件数都是整数};
		\node[font=\normalsize\bfseries] at (6.20,0.10)
			{但多次整数结果的平均值可以是小数};
	\end{tikzpicture}
\end{center}
