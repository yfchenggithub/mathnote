\begin{scope}

\draw[gray,->] (0,-2.42)--(0,2.42) node[below right] {$Y$};
% S=1 is the original hyperbola in this midpoint plane.

\draw[CProof,very thick,domain=-2.5:2.5,samples=160] plot ({-sqrt(1+\x*\x)},\x);

};
