\draw[orange!85!black,very thick,domain=-1:1,samples=60] plot (\x,{\x*\x*(\x+3)});

\end{scope}
\node at (5.8,-1.7) {$0<|x|<1$ 时，$x^2>0,\ x+3>2$，故 $q(x)>q(0)$。};
