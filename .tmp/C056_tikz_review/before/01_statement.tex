\begin{statementbox}
\textbf{条件与方向约定（先定方向，再谈比值）。}
设椭圆
\[
 C:\quad \frac{x^2}{a^2}+\frac{y^2}{b^2}=1,\qquad
 a>b>0,\quad c=\sqrt{a^2-b^2},\quad e=\frac ca\in(0,1).
\]
取\textbf{右焦点} $F=(c,0)$。过 $F$ 的直线倾斜角为
$\theta\in[0,\pi)$，从 $x$ 轴正向逆时针量起。
记单位方向向量 $u=(\cos\theta,\sin\theta)$，并规定
\[
 A=F+ru,\qquad B=F-su,\qquad r=|AF|>0,\;s=|BF|>0.
\]
即 $A$ 沿 $\theta$ 方向，$B$ 沿反方向；$F$ 在线段 $AB$ 内部。
此处 $AF,BF$ 均表示\textbf{正的长度}，不是有向线段。
\par\smallskip\noindent
\textbf{核心结论：先求两段，再相除。}
\[
 \boxed{\displaystyle AF=\frac{b^2}{a+c\cos\theta}}
 \qquad
 \boxed{\displaystyle BF=\frac{b^2}{a-c\cos\theta}}
\]
\[
 \boxed{\displaystyle \frac{AF}{BF}
 =\frac{a-c\cos\theta}{a+c\cos\theta}
 =\frac{1-e\cos\theta}{1+e\cos\theta}}
\]
\textbf{连带结论：焦点弦长}
\[
 \boxed{\displaystyle AB=AF+BF
 =\frac{2ab^2}{a^2-c^2\cos^2\theta}}.
\]
由于 $a>c\ge 0$ 且 $|\cos\theta|\le1$，全部分母恒正。
交换 $A,B$ 时，$AF/BF$ 要取倒数，不能悄悄沿用旧式。
\end{statementbox}
