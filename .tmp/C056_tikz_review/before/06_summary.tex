\begin{summarybox}
\textbf{考前 15 秒快记：先端点方向，后分母正负。}
\[
 A=F+r(\cos\theta,\sin\theta),\quad
 B=F-s(\cos\theta,\sin\theta),\quad F=(c,0),\quad r,s>0.
\]
\[
 \boxed{AF=\frac{b^2}{a+c\cos\theta}},\qquad
 \boxed{BF=\frac{b^2}{a-c\cos\theta}}.
\]
\[
 \boxed{\frac{AF}{BF}=\frac{1-e\cos\theta}{1+e\cos\theta}},\qquad
 \boxed{AB=\frac{2ab^2}{a^2-c^2\cos^2\theta}}.
\]
\textbf{证明路径（按实际考试使用优先级）：}
\begin{enumerate}
 \item \textbf{最快：} $x=c\pm(\text{焦半径})\cos\theta$，
 结合 $PF=a-ex$，两式直接解出 $AF,BF$。
 \item \textbf{最低门槛：}设 $AF=r,BF=s$，
 两焦点距离之和 $2a$，各用一次余弦定理。
 \item \textbf{常规：} $P(t)=F+t(\cos\theta,\sin\theta)$，
 代入椭圆，正负根对应两个方向。
\end{enumerate}
\textbf{三重验算：}
$\theta=0$ 时 $AF=a-c,BF=a+c$；
$\theta=\pi/2$ 时 $AF=BF=b^2/a$；
$\theta$ 钝角时 $AF/BF>1$。
\par\smallskip\noindent
\textbf{题目触发词：}过焦点的弦 $+$ 倾斜角/斜率 $+$ 两段长度比、
从焦点到端点的距离、弦长最值。
\end{summarybox}
