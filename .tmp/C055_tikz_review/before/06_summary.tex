\begin{summarybox}
\textbf{\textcolor{CSummary}{15 秒快记}}\par\smallskip
\[
 \begin{gathered}
 y^2=2px\ (p>0),\quad F\!\left(\frac p2,0\right),\quad AB\text{ 过焦点},\\[3pt]
 \boxed{\ y_1y_2=-p^2\ },\qquad
 \boxed{\ x_1x_2=\frac{p^2}{4}\ },\\[3pt]
 \boxed{\ k_Ak_B=-1\ },\qquad
 \boxed{\ x_T=-\frac p2\ }.
 \end{gathered}
\]
\textbf{口诀：}\quad \textbf{纵积负 $p^2$，横积焦横平方；两切成直角，交点落准线。}

\medskip
\textbf{\textcolor{CSummary}{证明记忆链（按推荐顺序）}}\par\smallskip
\begin{enumerate}[leftmargin=*,itemsep=0.3em]
 \item \textbf{最快：}设 $x=my+\frac p2$；代入后常数项 $-p^2$；韦达得纵积；平方转成横积。
 \item \textbf{最低门槛：}设点 $A(\frac{a^2}{2p},a),B(\frac{b^2}{2p},b)$；三点共线 $\Rightarrow(a-b)(ab+p^2)=0$。
 \item \textbf{常规：}设 $y=k(x-\frac p2)$；联立对 $x$ 用韦达；竖直通径另验。
\end{enumerate}

\medskip
\textbf{\textcolor{CSummary}{做题入手信号}}\par\smallskip
看见\textbf{焦点弦 $AB$}，先写 $y_1y_2=-p^2$；要求横坐标就用
$x_1x_2=\frac{p^2}{4}$；涉及切线就用 $k_i=\frac p{y_i}$；涉及焦半径就用
$|AF|=x_1+\frac p2$。

\medskip
\textbf{\textcolor{CSummary}{最后三项检查}}\par\smallskip
\begin{itemize}[leftmargin=*,itemsep=0.25em]
 \item 弦是否\textbf{经过焦点}，并确有两个不同交点？
 \item 是 $y^2=\pm2px$ 还是 $x^2=\pm2py$？上下开口要\textbf{交换横纵}。
 \item $y_1y_2$ 的\textbf{负号}与焦点坐标 $\frac p2$ 是否正确？
\end{itemize}
\end{summarybox}
