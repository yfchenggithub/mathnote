\begin{summarybox}
\textbf{C084\quad 双曲线焦点三角形的内角平分线——快记／快懂／快用}

\textbf{快记（一句核心话）：}
\begin{center}
\textbf{“双曲线焦点角，内分线就是切线；横坐标乘横截距，恒为 $a^2$。”}
\end{center}
\[
\boxed{x_0m=a^2}\qquad
\boxed{PM:\frac{x_0x}{a^2}-\frac{y_0y}{b^2}=1}.
\]

\textbf{快懂（三条一连就出来）：}
\[
\underbrace{m=c\frac{r_1-r_2}{r_1+r_2}}_{\text{角平分线定理}},\qquad
\underbrace{r_1^2-r_2^2=4cx_0}_{\text{距离平方差}},\qquad
\underbrace{|r_1-r_2|=2a}_{\text{双曲线定义}}.
\]
把前两式相乘，立刻得到
\[
mx_0=\frac{(r_1-r_2)^2}{4}=a^2.
\]

\textbf{快用（最少步骤）：}
\begin{itemize}
  \item 求轴交点：直接 $\displaystyle m=\frac{a^2}{x_0}$；
  \item 求角平分线：直接写\textbf{过 $P$ 的切线方程}；
  \item 知交点反求点：$\displaystyle x_0=\frac{a^2}{m}$，再代回双曲线求 $y_0$；
  \item 求内切圆在焦点轴上的切点：\textbf{对应分支的顶点} $T=(\pm a,0)$；
  \item 作答前检查：$y_0\ne0$，且 $\operatorname{sgn}m=\operatorname{sgn}x_0$、$|m|<a$。
\end{itemize}

\textbf{五法怎么选：}
最低门槛\textbf{角平分线＋平方差}；
最快代数\textbf{角平分线＋焦半径}；
最快识别\textbf{已知切线性质}；
几何深化\textbf{内切圆切线长}；
拓展\textbf{半角三角函数／单位向量}。

\textbf{一眼防错：} $\boxed{T\ne M}$（非顶点）；
\textbf{椭圆}焦点内角平分线通常是\emph{法线}，
\textbf{双曲线}则是\emph{切线}。
\end{summarybox}
