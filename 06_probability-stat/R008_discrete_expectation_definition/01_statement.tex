\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（有限离散取值）：}}] 在高中常见题型中，随机变量 $\xi$ 只取有限个值 $x_1,x_2,\dots,x_n$。
		\item[\textbf{\textcolor{CStatement}{条件2（分布列性质）：}}] $P(\xi=x_i)=P_i$，且
		      \[
			      P_i\ge0,\qquad \sum_{i=1}^{n}P_i=1.
		      \]
		\item[\textbf{\textcolor{CStatement}{拓展说明（可列无穷情形）：}}] 若 $\xi$ 取可列无穷多个值 $x_1,x_2,\dots$，还需满足
		      \[
			      \sum_{i=1}^{\infty}|x_i|P_i<+\infty,
		      \]
		      此时数学期望才作为有限实数存在。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（期望定义）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 期望是以概率为权重的加权平均值，反映随机变量的长期平均水平或概率平衡位置。
		      \[
			      E\xi=\sum_{i=1}^{n}x_iP_i.
		      \]

		\item[\textbf{\textcolor{CStatement}{常见变形（线性性质）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 当相关期望存在时，期望运算满足线性性，常数可提、和可分。
		      \[
			      E(a\xi+b)=aE\xi+b,\qquad a,b\in\mathbb{R}.
		      \]
		      更一般地，若 $E\xi$ 与 $E\eta$ 都存在，则
		      \[
			      E(\xi+\eta)=E\xi+E\eta.
		      \]
		      注意：上式不要求 $\xi$ 与 $\eta$ 相互独立。
	\end{description}

\end{statementbox}
