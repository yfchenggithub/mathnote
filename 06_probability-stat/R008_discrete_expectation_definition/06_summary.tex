\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{期望是随机变量取值以概率为权重的加权平均值，反映随机变量的长期平均水平或概率平衡位置。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（有限离散已知）：} 在高中常见题型中，随机变量 $\xi$ 只取有限个值 $x_1,x_2,\dots,x_n$，且这些取值及其概率 $P_i$ 已知或可推导。
		\item \textbf{条件2（概率归一）：} 概率满足
		      \[
			      P_i\ge0,\qquad \sum_{i=1}^{n}P_i=1.
		      \]
		\item \textbf{条件3（拓展到可列无穷）：} 若 $\xi$ 取可列无穷多个值，则还需满足
		      \[
			      \sum_{i=1}^{\infty}|x_i|P_i<+\infty,
		      \]
		      此时期望才作为有限实数存在。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{核心公式}}
	\[
		E\xi=\sum_{i=1}^{n}x_iP_i.
	\]

	当 $E\xi$ 存在时，
	\[
		E(a\xi+b)=aE\xi+b,\qquad a,b\in\mathbb{R}.
	\]

	若 $E\xi$ 与 $E\eta$ 都存在，则
	\[
		E(\xi+\eta)=E\xi+E\eta.
	\]
	该性质不要求 $\xi$ 与 $\eta$ 独立。

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（前提校验）：} 计算前确认分布列已归一化，即 $\sum_{i=1}^{n}P_i=1$；若题目给的是权重而不是概率，需要先归一化。
		\item \textbf{易错点（算术平均）：} 等可能时，期望等于算术平均数；一般情况下，期望是加权平均，不能直接用算术平均数代替。
		\item \textbf{检查项（范围检查）：} 若随机变量只取有限个值，则
		      \[
			      \min_i x_i\le E\xi\le \max_i x_i.
		      \]
		      若期望超出该范围，通常说明计算或概率分布有误。
		\item \textbf{理解项（不是必然取值）：} 期望可以不是随机变量的实际取值，例如掷公平骰子的期望为 $3.5$，但骰子不会掷出 $3.5$。
	\end{itemize}
\end{summarybox}
