\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（划分与正概率）：}}] 设 $A_1,A_2,\dots,A_n$ 为样本空间的一个划分，且 $P(A_i)>0$，$i=1,\dots,n$。
		\item[\textbf{\textcolor{CStatement}{条件 2（事件正概率）：}}] 设 $B$ 为任意事件，且 $P(B)>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（后验概率公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知结果 $B$ 发生后，各原因 $A_i$ 的概率通过先验概率和似然概率加权得到。
			\[
				P(A_i \mid B) = \frac{P(A_i)P(B \mid A_i)}{\displaystyle\sum_{j=1}^{n} P(A_j)P(B \mid A_j)},
				\qquad
				i=1,\dots,n.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（两事件特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当只有两个原因事件时，后验概率公式简化为常见的二分形式。
			\[
				P(A \mid B) = \frac{P(A)P(B \mid A)}{P(A)P(B \mid A) + P(\overline{A})P(B \mid \overline{A})}.
			\]
	\end{description}

\end{statementbox}
