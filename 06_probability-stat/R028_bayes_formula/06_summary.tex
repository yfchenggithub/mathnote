\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{贝叶斯公式给出了在已知结果发生的情况下，计算各个原因发生概率的方法，实现了从先验概率到后验概率的更新。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（划分正概率）：} 事件组 $A_1,\dots,A_n$ 构成样本空间的一个划分，且每个 $P(A_i)>0$。
		\item \textbf{条件 2（事件正概率）：} 结果事件 $B$ 满足 $P(B)>0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（混淆条件）：} 不要将 $P(A_i\mid B)$ 与 $P(B\mid A_i)$ 混淆；二者方向相反。
		\item \textbf{检查项（全概率计算）：} 计算分母时必须包含所有划分事件，且概率和为1。
	\end{itemize}
\end{summarybox}
