\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{卡方独立性检验通过比较观测与期望频数的标准化偏离平方和，借助卡方分布判断两分类变量是否独立。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（期望频数足够）：}每个单元格期望频数 $\geq5$，或至少 $80\%$ 的单元格期望频数 $\geq5$。
		\item \textbf{条件2（样本独立性）：}不同个体的观测结果相互独立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式与条件）：}期望频数计算公式为 $E_{ij}=R_iC_j/n$；自由度 $k=(r-1)(c-1)$；必须提前验证条件。
		\item \textbf{检查项（计算与结论）：}计算 $\chi^2$ 后与临界值比较，在拒绝域中才拒绝 $H_0$，注意临界值与自由度对应。
	\end{itemize}
\end{summarybox}
