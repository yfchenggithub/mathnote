\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{排列数公式 $A_n^m = \dfrac{n!}{(n-m)!}$ 用于计算从 $n$ 个不同元素中取出 $m$ 个排成一列的方法总数。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（元素互异）：} $n$ 个元素各不相同。
		\item \textbf{条件 2（范围约束）：} $n,m \in \mathbb{N}$ 且 $m \le n$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分有序）：} 排列与顺序有关，不要与组合混淆。
		\item \textbf{检查项（定义域验证）：} 使用前务必检查 $m\le n$ 且 $n,m$ 为非负整数。
	\end{itemize}

\end{summarybox}
