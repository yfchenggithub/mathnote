\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（元素互异）：}}]
			$n$ 个元素互不相同。
		\item[\textbf{\textcolor{CStatement}{条件 2（数量约束）：}}]
			$n,m \in \mathbb{N}$ 且 $m \le n$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（排列计数）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从 $n$ 个不同元素中取出 $m$ 个的排列总数。
			\[
				A_n^m = \frac{n!}{(n-m)!}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（连乘展开）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			展开为连乘形式，便于直接计算。
			\[
				A_n^m = n(n-1)\cdots (n-m+1).
			\]
	\end{description}

\end{statementbox}
