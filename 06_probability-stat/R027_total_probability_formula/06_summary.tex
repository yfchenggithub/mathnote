\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{全概率公式将复杂事件的总概率分解为按原因（划分）加权的条件概率之和，实现分而治之的计算。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（完备事件组）：} 事件组$\{A_i\}_{i=1}^n$必须满足$A_i\cap A_j=\varnothing\;(i\neq j)$且$\bigcup_{i=1}^n A_i=\Omega$。
		\item \textbf{条件 2（概率正性）：} 每个$P(A_i)>0$，否则条件概率无定义。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（划分验证）：} 使用前必须确认划分的互斥性和完备性，不能想当然。
		\item \textbf{检查项（权重和）：} 验证$\sum_{i=1}^n P(A_i)=1$，确保没有遗漏或重复。
	\end{itemize}
\end{summarybox}
