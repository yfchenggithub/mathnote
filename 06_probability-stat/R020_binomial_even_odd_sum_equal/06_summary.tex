\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{二项式系数中，偶数下标项之和等于奇数下标项之和，且均为 $2^{n-1}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（正整数n）：} $n$ 必须是正整数（$n\ge 1$），否则结论不成立。
		\item \textbf{条件2（标准系数）：} 使用的是二项式展开的标准系数 $\binom{n}{k}$，未作变形。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（指数混淆）：} 指数是 $n-1$ 而非 $n$，牢记“总和一半”。
		\item \textbf{检查项（确认n范围）：} 使用前必须验证 $n\ge 1$，避免 $n=0$ 误用。
	\end{itemize}

\end{summarybox}
