\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（正整数）：}}]
			$n\in\mathbb{N}^*$，即$n\ge 1$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（奇偶和相等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			二项式系数中，偶数下标之和等于奇数下标之和，且均为$2^{n-1}$。
			\[
				\boxed{
				\begin{aligned}
					\binom{n}{0}+\binom{n}{2}+\binom{n}{4}+\cdots &= \binom{n}{1}+\binom{n}{3}+\binom{n}{5}+\cdots \\ &= 2^{\,n-1}
			\end{aligned}
				}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（交替和为零）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将奇数项系数取负号后与偶数项相加，结果为零。
			\[
				\sum_{k=0}^{n}(-1)^k\binom{n}{k}=0\quad (n\ge 1).
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{适用范围：}}
	$n$为正整数（$n\ge 1$）。当$n=0$时，$\binom{0}{0}=1$，上式不成立。
\end{statementbox}
