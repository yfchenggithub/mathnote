	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（问题特征）：}}]
			计数问题涉及元素相邻、不相邻、相同元素分配或正面计算复杂时，考虑使用以下策略。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（捆绑法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $m$ 个相邻元素捆绑为整体并内部排列，再与其余 $n-m$ 个元素全排列。
			\[
				m!\,(n-m+1)!
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（插空法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			先排 $n$ 个无限制元素，再在 $n+1$ 个空位中选择 $k$ 个插入 $k$ 个不相邻元素。
			\[
				n!\,A_{n+1}^{k}
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（隔板法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$n$ 个相同元素分给 $m$ 个不同对象，每对象至少一个，用隔板法。
			\[
				\binom{n-1}{m-1}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（排除思想）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			设总方案数为 $T$，不符合条件方案数为 $B$，则符合条件方案数为 $T-B$。
			\[
				T - B
			\]
	\end{description}

\end{statementbox}
