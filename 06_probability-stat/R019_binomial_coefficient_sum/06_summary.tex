\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{二项式系数之和 $\sum_{k=0}^{n} \binom{n}{k}$ 等于 $2^n$，整个结论基于 $(1+x)^n$ 展开后令 $x=1$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（指数为正整数）：} $n$ 必须是正整数（$n\in\mathbb{N}^*$），否则二项式系数 $\binom{n}{k}$ 可能未定义。
		\item \textbf{条件 2（求和范围完整）：} 求和必须从 $k=0$ 到 $k=n$，缺一不可。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分系数概念）：} 二项式系数和恒为 $2^n$，与 $(a+b)^n$ 中 $a,b$ 的具体值无关；不要与各项系数和混为一谈。
		\item \textbf{检查项（包含首末项）：} 求和时确认包含 $k=0$ 和 $k=n$ 两项，它们的值分别为 $1$ 和 $1$，容易遗漏。
	\end{itemize}
\end{summarybox}
