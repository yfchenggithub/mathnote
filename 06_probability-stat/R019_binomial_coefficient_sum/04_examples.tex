\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	计算 $\displaystyle\sum_{k=0}^{6}\binom{6}{k}$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结论）：}]
			该求和式是二项式系数和，直接使用公式 $\sum_{k=0}^{n}\binom{n}{k}=2^n$。
			\par\textbf{理由：}
			根据输入结论，二项式系数和恒等于 $2^n$。
		\item[\textbf{第二步（代入指数）：}]
			这里 $n=6$，代入得 $2^6$。
			\par\textbf{理由：}
			所给求和范围 $k=0$ 到 $6$，对应 $n=6$。
		\item[\textbf{第三步（计算结果）：}]
			$2^6 = 64$。
			\par\textbf{理由：}
			计算 $2$ 的 $6$ 次幂。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\displaystyle\boldsymbol{\sum_{k=0}^{6}\binom{6}{k}=64}$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sum_{k=0}^{m}\binom{m}{k}=128$，求正整数 $m$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}]
			由二项式系数和公式，有 $\sum_{k=0}^{m}\binom{m}{k}=2^m$，因此 $2^m = 128$。
			\par\textbf{理由：}
			将已知的系数和代入公式。
		\item[\textbf{第二步（指数转化）：}]
			将 $128$ 写作 $2^7$，即 $2^m = 2^7$。
			\par\textbf{理由：}
			$128$ 是 $2$ 的 $7$ 次幂。
		\item[\textbf{第三步（解得指数）：}]
			由同底数幂相等得 $m = 7$。
			\par\textbf{理由：}
			底数为 $2$ 的幂函数是单射，指数相等。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boldsymbol{m=7}$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $(2x-3)^n$ 的展开式中，所有二项式系数之和等于 $64$，求指数 $n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（区分概念）：}]
			二项式系数是指 $\binom{n}{k}$，它们只与 $n$ 有关，与 $(2x-3)$ 的具体数值无关；所有二项式系数之和为 $2^n$。
			\par\textbf{理由：}
			由二项式定理，$(a+b)^n$ 的展开式中第 $k+1$ 项的二项式系数为 $\binom{n}{k}$，其和为 $2^n$。
		\item[\textbf{第二步（建立方程）：}]
			依题意，$2^n = 64$。
			\par\textbf{理由：}
			直接使用结论：“二项式系数和等于 $2^n$”。
		\item[\textbf{第三步（解方程）：}]
			由 $2^n = 2^6$ 得 $n=6$。
			\par\textbf{理由：}
			同底数幂相等则指数相等。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boldsymbol{n=6}$。
\end{examplebox}
