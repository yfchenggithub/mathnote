\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆系数概念）}}] 将二项式系数与展开式中某项的系数混为一谈。例如求 $(2x+1)^n$ 的二项式系数和时，误算为 $(2+1)^n=3^n$。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分系数定义）：}}] 二项式系数专指 $\binom{n}{k}$，其和恒为 $2^n$，而与括号内的因子无关。$(2x+1)^n$ 的二项式系数和仍是 $2^n$，不是 $3^n$。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义不清）：}}] 没有从定义区分“二项式系数”与“系数”，误将底数 $2x+1$ 中系数 $2$ 的影响代入系数和的计算。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗漏常数项）}}] 求 $\sum_{k=1}^{n}\binom{n}{k}$ 的值时，直接写为 $2^n$，忘记 $k=0$ 项 $\binom{n}{0}=1$ 也被包含在总和里。
		\item[\textbf{\textcolor{CTrap}{正确理解（包含常数项）：}}] 二项式系数和 $\sum_{k=0}^{n}\binom{n}{k}=2^n$ 是从 $k=0$ 开始。去掉 $k=0$ 项后为 $\sum_{k=1}^{n}\binom{n}{k}=2^n-1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略首项）：}}] 没有注意到 $\binom{n}{0}=1$ 是展开式的常数项，求和时必须包括。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（幂运算错误）}}] 已知 $\sum_{k=0}^{n}\binom{n}{k}=64$，直接得出 $n=32$，错误地将 $2^n=64$ 理解为 $n=64\div2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（解指数方程）：}}] $2^n=64$ 等价于 $n=\log_2 64=6$，或通过 $64=2^6$ 比较指数得到 $n=6$。
		\item[\textbf{\textcolor{CTrap}{错因分析（运算法则混淆）：}}] 混淆了乘方运算与乘法，$2^n$ 是 $n$ 个 $2$ 相乘，不是 $2\times n$。
	\end{description}
\end{trapbox}
