\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}

	二项式系数之和就是 $2^n$，它来自 $(1+x)^n$ 中令 $x=1$。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（赋值推导）：}] 在二项式定理 $(1+x)^n=\sum_{k=0}^{n}\binom{n}{k}x^k$ 中，令所有 $x=1$，即得 $\sum_{k=0}^{n}\binom{n}{k}=2^n$。
		\item[\textbf{要点二（组合解释）：}] 从 $n$ 个不同元素中选取任意个元素（可0个可全部）组成子集，每个元素有“选”与“不选”两种状态，总共有 $2^n$ 个不同子集，这就是二项式系数和。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（子集个数）}}

	把每个组合数 $\binom{n}{k}$ 看作大小为 $k$ 的子集个数，总和就是所有子集的总数 $2^n$。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（赋值特例）}}

	$(1+x)^n$ 中 $x=1$ 的特殊值恰好等于所有系数之和；同样 $x=-1$ 得到交错系数和。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（常考题型）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接求和）：} 给定非负整数 $n$，快速计算 $\sum_{k=0}^{n}\binom{n}{k}$ 的值。
		\item \textbf{考法二（逆向求指数）：} 已知 $\sum_{k=0}^{n}\binom{n}{k} = 64$，求 $n$（即 $2^n=64$ 得 $n=6$）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}

	\textbf{把看似复杂的组合求和转化为一个简单幂运算，核心是 $(1+x)^n$ 的赋值法。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（组合计数）：} 需要计算一个集合所有子集的总数时，直接使用 $2^n$。
		\item \textbf{场景二（展开赋值）：} 在展开 $(a+b)^n$ 并求各项二项式系数和时，令 $a=b=1$ 即得 $2^n$。
	\end{itemize}
\end{explanationbox}
