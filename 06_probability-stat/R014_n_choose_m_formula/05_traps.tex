\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略条件）：}}]
			在 $m>n$ 时强行使用公式，导致分母出现负数阶乘无意义。
		\item[\textbf{\textcolor{CTrap}{正确理解（范围限制）：}}]
			公式要求 $m\le n$ 且 $m,n$ 为非负整数（含 $0$）。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义冲突）：}}]
			阶乘只对非负整数有定义，$n-m$ 为负数时无意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（混用排列）：}}]
			把 $C(n,m)$ 直接写成 $A_n^m$，忘记除以 $m!$。
		\item[\textbf{\textcolor{CTrap}{正确理解（顺序区别）：}}]
			组合不关心顺序，每个组合对应 $m!$ 个排列，必须除以 $m!$ 抵消顺序。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念模糊）：}}]
			不清楚排列和组合的本质区别，生搬公式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（计算粗心）：}}]
			在约简 $C(n,m)$ 时只约分一次就得出答案，导致结果错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（彻底约分）：}}]
			将 $n!$、$m!$、$(n-m)!$ 全部展开成乘积，逐分子分母相约，直到无法再约。
		\item[\textbf{\textcolor{CTrap}{错因分析（跳步损失）：}}]
			过度依赖心算，没有写出完整展开式，容易遗漏因子。
	\end{description}

\end{trapbox}
