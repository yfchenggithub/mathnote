\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（自然数条件）：}}]
			$n,m\in\mathbb{N}$ 且 $m\le n$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（阶乘公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过阶乘的比值直接计算组合数。
			\[
				C(n,m)=\frac{n!}{m!\,(n-m)!}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（对称性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			选取 $m$ 个与剩下 $n-m$ 个一一对应，公式交换 $m$ 与 $n-m$ 值不变。
			\[
				C(n,m)=C(n,n-m)
			\]
	\end{description}

\end{statementbox}
