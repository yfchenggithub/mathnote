\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{组合数公式 $C(n,m)=\frac{n!}{m!(n-m)!}$ 是组合计数的基本工具。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（整数限制）：}$n,m\in\mathbb{N}$ 且 $m\le n$。
		\item \textbf{条件2（无序前提）：}只用于不计顺序的选取问题。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（0! 定义）：}$0!=1$，注意 $C(n,0)=1$ 等特殊情况。
		\item \textbf{检查项（约分过程）：}将阶乘展开后逐项约分，避免漏乘。
	\end{itemize}

\end{summarybox}
