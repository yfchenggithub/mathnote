\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（事件与概率）：}}] $A,B$为同一随机试验中的两个事件，且$P(A)>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（条件概率）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在事件$A$发生的条件下，事件$B$发生的概率等于$AB$同时发生的概率除以$A$的概率。
			\[
				P(B \mid A) = \frac{P(AB)}{P(A)}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（乘法公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			事件$AB$同时发生的概率等于事件$A$的概率乘以事件$A$发生条件下$B$的条件概率。
			\[
				P(AB) = P(A) \cdot P(B \mid A)
			\]
	\end{description}
\end{statementbox}
