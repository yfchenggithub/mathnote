\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	掷一颗质地均匀的骰子。设事件$A=\{1,2,3,4\}$，事件$B=\{2,4,6\}$。求$P(B \mid A)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算基本概率）：}] 计算$P(A)$和$P(AB)$。
			\[
				\begin{aligned}
					P(A) &= \frac{4}{6} \\
					&= \frac{2}{3}, \\
					P(AB) &= \frac{2}{6} \\
					&= \frac{1}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 古典概型，概率等于有利点数除以总数。

		\item[\textbf{第二步（套用定义）：}] 代入条件概率公式。
			\[
				\begin{aligned}
					P(B \mid A) &= \frac{P(AB)}{P(A)} \\
					&= \frac{1/3}{2/3} \\
					&= \frac{1}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 条件概率的数学定义。

		\item[\textbf{第三步（缩减验证）：}] 直接利用缩减样本空间验证：$A$发生后样本空间变为$\{1,2,3,4\}$，$B$发生需为$2$或$4$，概率$\frac{2}{4}=\frac{1}{2}$。
			\[
				\frac{2}{4}=\frac{1}{2}.
			\]
			\par\textbf{理由：} 条件概率的直观含义。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P(B \mid A)=\dfrac{1}{2}$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$P(A)=0.5$，$P(B \mid A)=0.4$，求$P(AB)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变形公式）：}] 由条件概率定义变形得到乘法公式。
			\[
				P(AB)=P(A)P(B \mid A).
			\]
			\par\textbf{理由：} 条件概率定义$P(B \mid A)=P(AB)/P(A)$两边同乘$P(A)$。

		\item[\textbf{第二步（代入求值）：}] 代入已知数值。
			\[
				\begin{aligned}
					P(AB) &= 0.5 \times 0.4 \\
					&= 0.2.
			\end{aligned}
			\]
			\par\textbf{理由：} 代数乘法运算。

		\item[\textbf{第三步（给出结果）：}] 得到$P(AB)=0.2$。
			\[
				P(AB)=0.2.
			\]
			\par\textbf{理由：} 计算结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P(AB)=0.2$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$P(A)=0.4$，$P(B)=0.5$，$P(B \mid A)=0.3$，求$P(A \mid B)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求交概率）：}] 利用乘法公式计算$P(AB)$。
			\[
				\begin{aligned}
					P(AB) &= P(A)P(B \mid A) \\
					&= 0.4 \times 0.3 \\
					&= 0.12.
			\end{aligned}
			\]
			\par\textbf{理由：} 乘法公式来自条件概率定义。

		\item[\textbf{第二步（求条件概率）：}] 利用条件概率定义计算$P(A \mid B)$。
			\[
				\begin{aligned}
					P(A \mid B) &= \frac{P(AB)}{P(B)} \\
					&= \frac{0.12}{0.5} \\
					&= 0.24.
			\end{aligned}
			\]
			\par\textbf{理由：} 条件概率定义，以$B$为条件。

		\item[\textbf{第三步（得出结论）：}] 得出$P(A \mid B)=0.24$。
			\[
				P(A \mid B)=0.24.
			\]
			\par\textbf{理由：} 计算结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P(A \mid B)=0.24$}。
\end{examplebox}
