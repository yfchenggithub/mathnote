\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{乘法公式将积事件 $A_1\cdots A_n$ 的概率分解为无条件概率与条件概率的连乘积：$P(A_1\cdots A_n)=P(A_1)P(A_2|A_1)\cdots P(A_n|A_1\cdots A_{n-1})$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（分母非零）：} 每次应用公式时，所给的条件概率的分母必须大于零，即 $P(A_1\cdots A_{k-1})>0$。
		\item \textbf{条件2（顺序确定）：} 事件必须有一个明确的先后顺序或因果关系，条件概率严格按照该顺序书写。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母为零）：} 不要忽略概率为零的情况，若 $P(A)=0$ 则乘法公式不能直接使用。
		\item \textbf{检查项（条件概率正确）：} 确认每步条件概率匹配正确，如 $P(A_2|A_1)$ 不能写成 $P(A_2|A_3)$。
	\end{itemize}
\end{summarybox}
