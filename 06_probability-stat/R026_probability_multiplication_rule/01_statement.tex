\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（事件概率正）：}}] 事件 $A$ 的概率 $P(A)>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（概率积正）：}}] 对于多个事件 $A_1,\dots,A_n$，有 $P(A_1\cdots A_{n-1})>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两事件公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两事件的积事件概率等于先发事件概率乘条件概率。
			\[
				P(AB)=P(A)P(B|A)
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（多事件公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			多个事件的积事件概率等于各步条件概率的连乘积。
			\[
				P(A_1A_2\cdots A_n)=P(A_1)P(A_2|A_1)P(A_3|A_1A_2)\cdots P(A_n|A_1\cdots A_{n-1})
			\]
	\end{description}
\end{statementbox}
