\textbf{\textcolor{CExample}{例1（基础）}}
\textbf{\textcolor{CExample}{题目：}}
某校调查性别与选科偏好，数据如下：
\begin{center}

	\begin{tabular}
		{c|cc} & 理科 & 文科 \\ \hline 男生 & 20 & 10 \\ 女生 & 15 & 25
	\end{tabular}

\end{center}
计算 $\chi^2$ 统计量。
\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
	\item[\textbf{第一步（标注单元格）：}]
		记 $a=20$, $b=10$, $c=15$, $d=25$。
		\par\textbf{理由：} 列联表左上、右上、左下、右下对应$a,b,c,d$。
	\item[\textbf{第二步（计算合计数）：}]
		行合计 $R_1=30$, $R_2=40$；列合计 $C_1=35$, $C_2=35$；$n=70$。
		\par\textbf{理由：} 便于代入公式。
	\item[\textbf{第三步（代入公式计算）：}]
		\[
			\begin{aligned}
				\chi^2 &= \frac{70\times(20\times25-10\times15)^2}{30\times40\times35\times35} \\ &= \frac{70\times350^2}{30\times40\times35\times35} \\ &= \frac{8575000}{1470000} \approx 5.833.
		\end{aligned}
		\]
		\par\textbf{理由：} 直接套用 $\chi^2 = \frac{n(ad-bc)^2}{(a+b)(c+d)(a+c)(b+d)}$。
\end{description}
\textbf{\textcolor{CExample}{关键结论：}}
$\chi^2 \approx 5.833$。

\textbf{\textcolor{CExample}{例2（稍变形）}}
\textbf{\textcolor{CExample}{题目：}}
某药试验数据如下：
\begin{center}

	\begin{tabular}
		{c|cc} & 有效 & 无效 \\ \hline 用药组 & 30 & 10 \\ 对照组 & 20 & 20
	\end{tabular}

\end{center}
利用 $\chi^2 = \sum\frac{(O-E)^2}{E}$ 计算卡方值，并验证与公式 $\chi^2 = \frac{n(ad-bc)^2}{(a+b)(c+d)(a+c)(b+d)}$ 结果一致。
\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
	\item[\textbf{第一步（标注符号与合计）：}]
		$a=30$, $b=10$, $c=20$, $d=20$；行合计 $R_1=40$, $R_2=40$；列合计 $C_1=50$, $C_2=30$；$n=80$。
		\par\textbf{理由：} 建立符号。
	\item[\textbf{第二步（计算期望频数）：}]
		$E_{11} = \frac{40\times50}{80}=25$, $E_{12} = \frac{40\times30}{80}=15$, $E_{21} = \frac{40\times50}{80}=25$, $E_{22} = \frac{40\times30}{80}=15$。
		\par\textbf{理由：} 期望频数等于行合计乘以列合计除以$n$。
	\item[\textbf{第三步（用定义计算）：}]
		\[
			\begin{aligned}
				\chi^2 &= \frac{(30-25)^2}{25} + \frac{(10-15)^2}{15} + \frac{(20-25)^2}{25} + \frac{(20-15)^2}{15} \\ &= \frac{25}{25} + \frac{25}{15} + \frac{25}{25} + \frac{25}{15} \\ &= 1 + \frac{5}{3} + 1 + \frac{5}{3} \\ &= 2 + \frac{10}{3} \\ &= \frac{16}{3} \approx 5.333.
		\end{aligned}
		\]
		\par\textbf{理由：} 逐项计算后相加。
	\item[\textbf{第四步（用简洁公式验证）：}]
		\[
			\begin{aligned}
				\chi^2 &= \frac{80\times(30\times20-10\times20)^2}{40\times40\times50\times30} \\ &= \frac{80\times400^2}{40\times40\times50\times30} \\ &= \frac{12800000}{2400000} \\ &= \frac{16}{3} \approx 5.333.
		\end{aligned}
		\]
		\par\textbf{理由：} 代入简洁公式得相同结果。
\end{description}
\textbf{\textcolor{CExample}{关键结论：}}
$\chi^2 = \dfrac{16}{3} \approx 5.333$，两种方式一致。

\textbf{\textcolor{CExample}{例3（综合应用）}}
\textbf{\textcolor{CExample}{题目：}}
某研究调查性别与吸烟情况，数据如下：
\begin{center}

	\begin{tabular}
		{c|cc} & 吸烟 & 不吸烟 \\ \hline 男性 & 40 & 20 \\ 女性 & 10 & 30
	\end{tabular}

\end{center}
试在 $\alpha=0.05$ 显著性水平下（临界值 $\chi^2_{0.05}(1)=3.841$）检验性别与吸烟是否独立。
\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
	\item[\textbf{第一步（建立假设）：}]
		$H_0$：性别与吸烟独立；$H_1$：性别与吸烟有关。
		\par\textbf{理由：} 独立性检验假设。
	\item[\textbf{第二步（检查期望频数条件）：}]
		计算期望频数： $E_{11} = \frac{60\times50}{100}=30$, $E_{12} = \frac{60\times50}{100}=30$, $E_{21} = \frac{40\times50}{100}=20$, $E_{22} = \frac{40\times50}{100}=20$。所有期望频数均大于 $5$，且 $n=100>40$，条件满足。
		\par\textbf{理由：} 期望频数均 $\geqslant5$，$n\geqslant40$，可近似卡方分布。
	\item[\textbf{第三步（计算统计量）：}]
		标注 $a=40$, $b=20$, $c=10$, $d=30$，行合计 $R_1=60$, $R_2=40$；列合计 $C_1=50$, $C_2=50$；$n=100$。
		\[
			\begin{aligned}
				\chi^2 &= \frac{100\times(40\times30-20\times10)^2}{60\times40\times50\times50} \\ &= \frac{100\times1000^2}{60\times40\times50\times50} \\ &= \frac{100000000}{6000000} \\ &= \frac{100}{6} \approx 16.667.
		\end{aligned}
		\]
		\par\textbf{理由：} 代入简洁公式。
	\item[\textbf{第四步（比较与结论）：}]
		$16.667 > 3.841$，所以拒绝 $H_0$，认为性别与吸烟有关。
		\par\textbf{理由：} 卡方值大于临界值，落入拒绝域。
\end{description}
\textbf{\textcolor{CExample}{关键结论：}}
$\chi^2 \approx 16.667$，拒绝独立假设，认为吸烟与性别相关。
