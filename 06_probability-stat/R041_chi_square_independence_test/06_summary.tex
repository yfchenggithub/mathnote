\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{卡方统计量通过观测与期望的差异衡量变量间关联程度，值越大越支持相关。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（随机样本条件）：} 数据来自两个分类变量的随机样本。
		\item \textbf{条件2（期望频数条件）：} 各单元格期望频数均不小于$5$，且总频数$n\geqslant40$。
		\item \textbf{条件3（原假设条件）：} 原假设$H_0$为两分类变量相互独立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式适用条件）：} 使用前必须检查期望频数是否满足条件，否则近似失效。
		\item \textbf{检查项（单元格定位）：} 务必按列联表左上、右上、左下、右下对应$a,b,c,d$，确保$ad-bc$计算正确。
	\end{itemize}

\end{summarybox}
