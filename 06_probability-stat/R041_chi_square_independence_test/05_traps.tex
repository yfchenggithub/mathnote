\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略使用条件）：}}]
			当单元格期望频数小于5或$n<40$时，仍然盲目使用卡方检验。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件必须满足）：}}]
			只当各单元格期望频数均$\geqslant5$且$n\geqslant40$时，检验统计量才近似服从卡方分布。
		\item[\textbf{\textcolor{CTrap}{错因分析（近似前提不熟）：}}]
			忽略近似条件，可能导致错误的显著性结论。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（位置对应错误）：}}]
			将列联表单元格位置搞错，导致代入公式时$a,b,c,d$对应错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（对角乘积差）：}}]
			明确$a$为左上，$b$右上，$c$左下，$d$右下，公式中使用的是对角差$ad-bc$。
		\item[\textbf{\textcolor{CTrap}{错因分析（表格方向混淆）：}}]
			不按规范顺序填写数据，计算错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（大小判断颠倒）：}}]
			误认为卡方值越小越应该拒绝$H_0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（大则拒绝独立）：}}]
			卡方值越大，观测与期望偏差越大，越支持拒绝独立假设，即卡方值大于临界值时拒绝$H_0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（假设逻辑不清）：}}]
			对检验的拒绝域方向理解错误。
	\end{description}

\end{trapbox}
