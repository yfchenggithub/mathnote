	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（随机样本条件）：}}]
			数据来自两个分类变量$A,B$的随机样本。
		\item[\textbf{\textcolor{CStatement}{条件2（期望频数条件）：}}]
			各单元格期望频数均不小于$5$，且总频数$n\geqslant40$。
		\item[\textbf{\textcolor{CStatement}{条件3（原假设条件）：}}]
			原假设$H_0$为两个分类变量$A$与$B$相互独立。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（卡方统计量）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过观测频数与期望频数的差距衡量独立性。
			\[
				\chi^2 = \frac{n(ad-bc)^2}{(a+b)(c+d)(a+c)(b+d)}
			\]
			近似服从自由度为$1$的$\chi^2$分布。
		\item[\textbf{\textcolor{CStatement}{推广结论（等价形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			卡方值等于各单元格残差平方与期望之比的和。
			\[
				\chi^2 = \sum_{i,j} \frac{(O_{ij}-E_{ij})^2}{E_{ij}}
			\]
			其中$O_{ij}$为实际频数，$E_{ij}$为期望频数。
	\end{description}

\end{statementbox}
