\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（遗漏因子）}}]
			误以为 $D(X)=np$。
		\item[\textbf{\textcolor{CTrap}{正确理解（含概率积）：}}]
			方差应为 $D(X)=np(1-p)$，必须含有 $1-p$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}]
			将方差公式与期望公式 $E(X)=np$ 混淆，忽视方差受失败概率影响。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略独立条件）}}]
			在不满足独立同分布时，如不放回抽样，仍套用公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（独立重复前提）：}}]
			二项分布要求各次试验独立且概率 $p$ 恒定；对于不放回抽样（总体不大）应使用超几何分布。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件误判）：}}]
			未仔细审查试验是否“独立重复”就把公式生搬硬套。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（比例方差错误）}}]
			直接写 $D\!\left(\frac{X}{n}\right)=np(1-p)$。
		\item[\textbf{\textcolor{CTrap}{正确理解（方差线性）：}}]
			正确公式为 $D\!\left(\frac{X}{n}\right) = \frac{D(X)}{n^2}$，即 $\frac{p(1-p)}{n}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略平方）：}}]
			对方差性质 $D(kX)=k^2D(X)$ 不熟练，误以为与期望一样为 $k$ 倍。
	\end{description}

\end{trapbox}
