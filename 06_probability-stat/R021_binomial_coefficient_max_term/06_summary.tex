\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{二项式系数最大值位于中间位置：偶数时一项，奇数时两项相等。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（n为正整数）：} $n$ 必须是正整数，$k$ 取 $0,1,\dots,n$。
		\item \textbf{条件 2（二项式系数本身）：} 该结论仅适用于二项式系数 $\binom{n}{k}$，而非展开式的项系数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（奇偶分类）：} 务必先判断 $n$ 的奇偶性，再确定最大项的个数。
		\item \textbf{检查项（对称验证）：} 利用 $\binom{n}{k}=\binom{n}{n-k}$ 验证最值位置是否对称。
	\end{itemize}
\end{summarybox}
