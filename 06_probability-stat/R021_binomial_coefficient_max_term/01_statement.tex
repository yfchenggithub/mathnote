\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（n为正整数）：}}] $n$ 为正整数，考虑组合数 $\binom{n}{k}$，其中 $k = 0,1,\dots,n$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（偶数情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $n$ 为偶数时，正中间一个组合数最大，即 $k = n/2$ 时取得最大值。
			\[
				\max_{0\le k\le n} \binom{n}{k} = \binom{n}{n/2}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（奇数情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $n$ 为奇数时，中间两个组合数相等且同时取得最大值。
			\[
				\max_{0\le k\le n} \binom{n}{k} = \binom{n}{(n-1)/2} = \binom{n}{(n+1)/2}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（对称性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			组合数具有对称性 $\binom{n}{k} = \binom{n}{n-k}$，从而二项式系数先增后减，最大值位于中央。
			\[
				\binom{n}{k} = \binom{n}{n-k},
				\qquad
				k=0,1,\dots,n.
			\]
	\end{description}
\end{statementbox}
