\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（参数混淆）：}}]
			将 $N(\mu,\sigma^2)$ 中的 $\sigma^2$ 误认为标准差 $\sigma$，导致区间宽度错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（参数区分）：}}]
			正态分布记号第二个参数是方差 $\sigma^2$，标准差为 $\sigma$（正平方根）。例如 $N(10,4)$ 中 $\sigma=2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}]
			平时表述简写“方差为 $\sigma^2$”易导致混淆，需养成读参数先确认是方差还是标准差。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（区间对称）：}}]
			对非对称区间如 $P(\mu<X<\mu+2\sigma)$ 直接套用 $0.9545$。
		\item[\textbf{\textcolor{CTrap}{正确理解（对称前提）：}}]
			$k\sigma$ 原则只适用于对称区间 $(\mu-k\sigma,\mu+k\sigma)$。非对称区间可通过标准化转化为对称区间再计算。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）：}}]
			记忆结论时忽略区间的对称性要求，只记数值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（必然事件）：}}]
			认为 $P(\mu-3\sigma<X<\mu+3\sigma)=1$，即 $3\sigma$ 外绝对不会发生。
		\item[\textbf{\textcolor{CTrap}{正确理解（近似非精确）：}}]
			$P(\mu-3\sigma<X<\mu+3\sigma)\approx0.9973$，仍有约 $0.27\%$ 的数据落在该区间外，并非绝对。
		\item[\textbf{\textcolor{CTrap}{错因分析（结论扩张）：}}]
			将“几乎必然”误解为“必然”，忽略概率近似本质。
	\end{description}

\end{trapbox}
