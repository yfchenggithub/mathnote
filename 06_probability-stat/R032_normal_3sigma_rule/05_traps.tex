\begin{trapbox}
	\textbf{\textcolor{CTrap}{先看最常见的两个结构性错误}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.38em]
		\item[\textbf{\textcolor{CTrap}{易错点一（方差当成标准差）：}}]
			在 $N(\mu,\sigma^2)$ 中，第二个参数是方差而不是标准差。例如 $X\sim N(10,4)$ 时，$\sigma=2$，所以 $3\sigma$ 区间是 $(4,16)$，不是 $(-2,22)$。
		\item[\textbf{\textcolor{CTrap}{易错点二（对非对称区间硬套覆盖率）：}}]
			$0.6827$、$0.9545$、$0.9973$ 对应的是以 $\mu$ 为中心的对称区间。一般区间 $(a,b)$ 应先标准化两个端点：
			\[
				P(a<X<b)=
				\Phi\!\left(\frac{b-\mu}{\sigma}\right)
				-\Phi\!\left(\frac{a-\mu}{\sigma}\right).
			\]
	\end{description}

	\input{assets/tikz/R032_symmetric_asymmetric_compare.tex}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.38em]
		\item[\textbf{\textcolor{CTrap}{易错点三（把近似概率当成 $1$）：}}]
			$P(|X-\mu|<3\sigma)\approx0.9973$，并不等于 $1$。区间外仍有约 $0.27\%$ 的概率；左、右单侧各约 $0.135\%$。
	\end{description}
\end{trapbox}

\clearpage
\begin{trapbox}
	\textbf{\textcolor{CTrap}{再看模型与应用边界}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.38em]
		\item[\textbf{\textcolor{CTrap}{易错点四（忽略正态前提）：}}]
			“$68$--$95$--$99.7$”不是任意分布的通用规律。偏态、重尾或多峰数据可能有完全不同的覆盖比例；使用样本均值与样本标准差估计总体参数时，还会额外引入估计误差。
		\item[\textbf{\textcolor{CTrap}{易错点五（把控制界限当作规格界限）：}}]
			$\mu\pm3\sigma$ 常用于统计过程控制中的控制界限，但“超界”只表示过程可能出现特殊原因、值得调查；产品是否合格应依据技术规格，不能由 $3\sigma$ 原则单独判定。
	\end{description}

	\input{assets/tikz/R032_control_limits_vs_specs.tex}

	\textbf{\textcolor{CTrap}{一句话辨析}}
	控制界限描述“过程是否稳定”，规格界限描述“产品是否合格”；两者来源不同，也不必重合。
\end{trapbox}
