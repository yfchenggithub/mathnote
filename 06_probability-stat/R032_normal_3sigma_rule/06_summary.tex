\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{正态变量 $X\sim N(\mu,\sigma^{2})$ 在区间 $(\mu-k\sigma,\mu+k\sigma)$ 内的概率仅与 $k$ 有关：当 $k=1,2,3$ 时分别约为 $0.6827$、$0.9545$、$0.9973$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（分布前提）：} 随机变量 $X$ 服从正态分布 $N(\mu,\sigma^2)$。
		\item \textbf{条件 2（区间对称）：} 所求区间必须是 $(\mu-k\sigma,\mu+k\sigma)$ 形式，对称中心为 $\mu$。
		\item \textbf{条件 3（近似约定）：} 概率值为近似数，实际应用中可直接使用。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（参数混淆）：} 避免将记号中方差 $\sigma^2$ 当作标准差 $\sigma$。
		\item \textbf{检查项（区间验证）：} 使用前应校验区间端点是否恰好等于 $\mu\pm k\sigma$。
	\end{itemize}

\end{summarybox}
