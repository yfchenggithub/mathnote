\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{若 $X\sim N(\mu,\sigma^2)$，则 $P(|X-\mu|<k\sigma)=2\Phi(k)-1$；当 $k=1,2,3$ 时，覆盖概率依次约为 $68.27\%$、$95.45\%$、$99.73\%$。}

	\medskip
	\textbf{\textcolor{CSummary}{速查公式}}
	\[
		\begin{aligned}
			P(|X-\mu|<k\sigma)&=2\Phi(k)-1,\\
			P(|X-\mu|\ge k\sigma)&=2[1-\Phi(k)],\\
			P(X\ge\mu+k\sigma)&=1-\Phi(k).
		\end{aligned}
	\]

	\medskip
	\textbf{\textcolor{CSummary}{最后检查}}
	\begin{itemize}[leftmargin=*, itemsep=0.45em, topsep=0.4em]
		\item $0.9973\ne1$，$3\sigma$ 区间外仍有约 $0.27\%$ 的概率。
		\item $68$--$95$--$99.7$ 法则依赖正态分布；对一般分布不可直接套用。
		\item 连续型正态分布中，区间端点取不取等号不影响概率。
	\end{itemize}
\end{summarybox}
