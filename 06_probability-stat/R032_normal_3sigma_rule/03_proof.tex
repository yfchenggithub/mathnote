\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	先把 $X$ 标准化，再利用标准正态密度关于原点的对称性，把双侧区间概率化成一个分布函数值。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	设 $X\sim N(\mu,\sigma^{2})$，$\sigma>0$。其密度为
	\[
		f_X(x)=\frac{1}{\sigma\sqrt{2\pi}}
		\exp\!\left[-\frac{(x-\mu)^2}{2\sigma^2}\right].
	\]
	对任意 $k>0$，令 $z=(x-\mu)/\sigma$，则 $\mathrm{d}x=\sigma\,\mathrm{d}z$，于是
	\[
		\begin{aligned}
			P(\mu-k\sigma<X<\mu+k\sigma)
			&=\int_{\mu-k\sigma}^{\mu+k\sigma}f_X(x)\,\mathrm{d}x\\
			&=\int_{-k}^{k}\frac{1}{\sqrt{2\pi}}e^{-z^{2}/2}\,\mathrm{d}z\\
			&=\Phi(k)-\Phi(-k).
		\end{aligned}
	\]
	标准正态密度 $\varphi(z)=\dfrac{1}{\sqrt{2\pi}}e^{-z^2/2}$ 是偶函数，因此
	\[
		\Phi(-k)=1-\Phi(k),
	\]
	从而得到一般公式
	\[
		P(\mu-k\sigma<X<\mu+k\sigma)=2\Phi(k)-1.
	\]

	\medskip
	\textbf{\textcolor{CProof}{代入 $k=1,2,3$}}
	使用保留足够位数的标准正态分布函数值
	\[
		\Phi(1)=0.841344746\ldots,\quad
		\Phi(2)=0.977249868\ldots,\quad
		\Phi(3)=0.998650102\ldots,
	\]
	可得
	\[
		\begin{aligned}
			2\Phi(1)-1&=0.682689492\ldots\approx0.6827,\\
			2\Phi(2)-1&=0.954499736\ldots\approx0.9545,\\
			2\Phi(3)-1&=0.997300204\ldots\approx0.9973.
		\end{aligned}
	\]
	这里先使用足够精确的 $\Phi(k)$ 再统一舍入，避免用四位查表值中途计算而产生末位不一致。
\end{proofbox}

\begin{proofbox}
	\textbf{\textcolor{CProof}{尾部形式}}
	取补集并利用对称性，还得到
	\[
		P(|X-\mu|\ge k\sigma)=2\bigl[1-\Phi(k)\bigr],\qquad
		P(X\ge\mu+k\sigma)=1-\Phi(k).
	\]

	正态分布在单点处的概率为 $0$，所以上式中的 $>$ 与 $\ge$ 可以互换。
\end{proofbox}
