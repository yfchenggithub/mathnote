\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正态参数）：}}]
			随机变量 $X$ 服从正态分布 $N(\mu,\sigma^{2})$，其中 $\sigma>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（一倍标准差）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			变量落在均值附近一个标准差范围内的概率约为 $68.27\%$。
			\[
				P(\mu-\sigma<X<\mu+\sigma)\approx0.6827
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（两倍标准差）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			变量落在均值附近两个标准差范围内的概率约为 $95.45\%$。
			\[
				P(\mu-2\sigma<X<\mu+2\sigma)\approx0.9545
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（三倍标准差）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			变量落在均值附近三个标准差范围内的概率约为 $99.73\%$。
			\[
				P(\mu-3\sigma<X<\mu+3\sigma)\approx0.9973
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{常见变形（标准化变换）：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{直观描述：}}]
			作标准化 $Z=\dfrac{X-\mu}{\sigma}\sim N(0,1)$，则原区间概率化为标准正态绝对值概率。
			\[
				\begin{aligned}
					P(\mu-k\sigma<X<\mu+k\sigma) &= P(|Z|<k) \\ &= 2\Phi(k)-1
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
