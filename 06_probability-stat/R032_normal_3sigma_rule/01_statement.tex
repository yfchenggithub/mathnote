\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（分布参数）：}}]
			随机变量 $X\sim N(\mu,\sigma^{2})$，其中 $\mu\in\mathbb{R}$，$\sigma>0$；这里第二个参数 $\sigma^{2}$ 是方差。几何上，$\mu$ 给出密度曲线的对称轴与峰值位置，$\sigma$ 是中心到拐点的横向距离。
		\item[\textbf{\textcolor{CStatement}{条件 2（区间倍数）：}}]
			取 $k>0$，考察以均值 $\mu$ 为中心、半宽为 $k\sigma$ 的对称区间。
	\end{description}

	\textbf{\textcolor{CStatement}{一般结论}}
	令 $\Phi(z)=P(Z\le z)$ 为标准正态分布函数，则
	\[
		\boxed{
		P(\mu-k\sigma<X<\mu+k\sigma)
		=P\!\left(\left|\frac{X-\mu}{\sigma}\right|<k\right)
		=2\Phi(k)-1
		}.
	\]
	由于正态分布是连续型分布，把上式中的严格不等号改为非严格不等号，概率不变。

	\medskip
	\textbf{\textcolor{CStatement}{常用的 $1\sigma$、$2\sigma$、$3\sigma$ 覆盖概率}}
	\input{assets/tikz/R032_68_95_997_overview.tex}
	这三个数常合称为正态分布的“$68$--$95$--$99.7$ 法则”；其中第三行通常称为“$3\sigma$ 原则”。相应地，落在 $3\sigma$ 区间外的双侧概率约为
	\[
		P(|X-\mu|\ge 3\sigma)=2\bigl[1-\Phi(3)\bigr]
		=0.002699\ldots\approx0.27\%.
	\]
