	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（总体参数）：}}] 总体共有 $N$ 件产品，其中次品 $M$ 件。
		\item[\textbf{\textcolor{CStatement}{条件2（抽取方式）：}}] 从总体中无放回地随机抽取 $n$ 件。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论1（期望公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			不放回抽样时，抽到的次品数的期望等于抽取数乘以次品率。
			\[
				E(X)=\frac{nM}{N}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（与二项同期望）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			虽然是不放回，但期望与放回抽样的二项分布完全相同。
			\[
				E(X)=n\cdot\frac{M}{N}=np,\quad p=\frac{M}{N}.
			\]
	\end{description}
\end{statementbox}
