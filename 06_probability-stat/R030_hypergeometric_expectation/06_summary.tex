\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{不放回抽样的次品数期望等于抽取数乘以次品率：$E(X)=n\cdot\frac{M}{N}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（总体有限）：} 总体共 $N$ 件产品，次品 $M$ 件。
		\item \textbf{条件2（不放回抽样）：} 无放回地随机抽取 $n$ 件。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（混淆期望与方差）：} 期望公式与二项相同，但在方差和概率计算上仍为超几何分布，不可混用。
		\item \textbf{检查项（参数范围）：} 确保 $n\le N$ 且 $M\le N$。
	\end{itemize}
\end{summarybox}
