	\textbf{\textcolor{CTrap}{易错点一：只抄取值范围，不理解极端情形}}

	\textbf{错误做法：} 机械写出 max、min 公式，却说不清下界为什么可能大于 $0$、上界为什么同时受 $M$ 与 $n$ 限制。

	\textbf{错误原因：} 忽略了“其他个体不够用”和“目标个体、样本位置都有限”这两个容量约束。

	\input{assets/tikz/R030_support_extremes.tex}

	\textbf{正确替代：} 先分别构造“尽量少拿目标”和“尽量多拿目标”的样本，再写一般上下界；同时检查 $N,M,n$ 都是满足 $0\le M,n\le N$ 的计数参数。

	\medskip
	\textbf{\textcolor{CTrap}{易错点二：把总体比例当成计数期望}}

	\textbf{错误做法：} 写成 $E(X)=M/N$。\quad
	\textbf{错误原因：} $M/N$ 只是一个位置的命中概率，$X$ 统计 $n$ 个位置的命中总数。\quad
	\textbf{正确替代：} 先写 $p=M/N$，再用 $E(X)=np=nM/N$。

	\medskip
	\textbf{\textcolor{CTrap}{易错点三：由 $E(X)=np$ 断言 $X\sim B(n,p)$}}

	\textbf{错误做法：} 把不放回抽样直接套成二项分布。\quad
	\textbf{错误原因：} 各位置一般不独立；期望相同不能推出分布相同。\quad
	\textbf{正确替代：} 求期望可用 $np$，求具体概率仍应使用超几何分布列。

	\medskip
	\textbf{\textcolor{CTrap}{易错点四：把非整数期望当成一次抽样结果}}

	\textbf{错误做法：} 由 $E(X)=1.75$ 断言“一次会抽到 $1.75$ 个”。\quad
	\textbf{错误原因：} 期望是概率加权平均，不必属于 $X$ 的取值集合。\quad
	\textbf{正确替代：} 把 $1.75$ 解释为大量同类抽样的长期平均，并用第 $5$ 页的整数结果平均图检查直觉。
\end{trapbox}
