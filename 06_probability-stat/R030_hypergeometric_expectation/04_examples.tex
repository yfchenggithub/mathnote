\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	某批产品共20件，其中有5件次品。质检员从中无放回地随机抽取4件检测，求抽到的次品件数 $X$ 的期望。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判断分布）：}] 由题意，总体有限（20件）、无放回、关注次品数，故 $X$ 服从超几何分布。
			\par\textbf{理由：} 超几何分布的四个条件（有限总体、不放回、计数、已知参数）。
		\item[\textbf{第二步（代入参数）：}] 确定参数：$N=20$, $M=5$, $n=4$。套用公式 $E(X)=\frac{nM}{N}$。
			\par\textbf{理由：} 超几何期望公式 $E(X)=\frac{nM}{N}$。
		\item[\textbf{第三步（计算结果）：}] $E(X)=\frac{4\times5}{20}=1$。
			\par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{次品数期望为 $1$，即平均抽到 $1$ 件次品。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	某工厂生产一批产品共100件，次品率为0.1。质量部门采用不放回抽样，随机抽取10件检查，求抽到的次品件数的期望。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求次品总数）：}] 由次品率，$M=100\times0.1=10$。
			\par\textbf{理由：} 次品率 $\times$ 总数。
		\item[\textbf{第二步（判断分布）：}] $N=100$, $M=10$, $n=10$，$X$ 服从超几何分布。
			\par\textbf{理由：} 不放回抽样，总体有限。
		\item[\textbf{第三步（计算期望）：}] $E(X)=\frac{10\times10}{100}=1$。
			\par\textbf{理由：} 超几何期望公式 $E(X)=\frac{nM}{N}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{次品数期望为 $1$ 件。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知某批产品共50件，次品率为0.3。现从中不放回抽取15件，记次品数为 $X$；若有放回抽取15件，记次品数为 $Y$。求 $E(X)$ 并比较 $E(X)$ 与 $E(Y)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数与分布）：}] $M=50\times0.3=15$，$N=50$, $n=15$；$X$ 服从超几何分布，$Y$ 服从二项分布 $B(15,0.3)$。
			\par\textbf{理由：} 超几何与二项分布定义。
		\item[\textbf{第二步（计算超几何期望）：}] $E(X)=\frac{15\times15}{50}=4.5$。
			\par\textbf{理由：} 超几何期望公式。
		\item[\textbf{第三步（计算二项期望并比较）：}] $E(Y)=15\times0.3=4.5$，故 $E(X)=E(Y)=4.5$。
			\par\textbf{理由：} 二项期望公式 $E(Y)=np$；两者相等。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$E(X)=E(Y)=4.5$，超几何与二项期望相同。}
\end{examplebox}
