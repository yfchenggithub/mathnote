\begin{examplebox}
	\textbf{\textcolor{CExample}{例 3（期望相同，分布未必相同）}}

	\textbf{\textcolor{CExample}{题目：}}
	总体有 $N=10$ 个个体，其中 $M=4$ 个为目标个体。若不放回抽取全部 $n=10$ 个，目标个体数记为 $X$；若每次放回并独立抽取 $10$ 次，命中次数记为 $Y$。比较两者。

	\textbf{\textcolor{CExample}{计算：}}
	\[
		E(X)=\frac{10\times4}{10}=4,
		\qquad
		E(Y)=10\times0.4=4.
	\]

	\input{assets/tikz/R030_distribution_comparison.tex}

	\textbf{\textcolor{CExample}{答案与回验：}}
	两者期望都为 $4$；但 $X\equiv4$，而 $Y\sim B(10,0.4)$ 会在 $0$ 至 $10$ 间波动，所以二者不同分布。柱高由二项分布公式逐项计算，概率和为 $1$，加权均值为 $4$。
\end{examplebox}

\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（直接应用与非整数期望）}}

	一批零件共有 $24$ 件，其中 $7$ 件带有标记。从中等可能地、不放回抽取 $6$ 件，记带标记零件数为 $X$。则
	\[
		E(X)=\frac{nM}{N}=\frac{6\times7}{24}=\frac{7}{4}=1.75.
	\]

	\input{assets/tikz/R030_noninteger_expectation.tex}

	\textcolor{black!65}{说明：这只是一组帮助理解的示意结果，不表示每 $4$ 次抽样的平均值必然是 $1.75$。}

	\textbf{\textcolor{CExample}{答案与回验：}}
	$1.75$ 是大量同类抽样中目标个体数的长期平均，不是某一次抽样会出现的件数；且它位于 $X$ 的最小可能值与最大可能值之间。
\end{examplebox}

\begin{examplebox}
	\textbf{\textcolor{CExample}{例 2（由期望反求总体参数）}}

	总体共有 $60$ 个个体，其中 $M$ 个为目标个体；无放回抽取 $15$ 个，目标个体数为 $X$。若 $E(X)=3$，则
	\[
		3=\frac{15M}{60}=\frac{M}{4},
		\qquad M=12.
	\]
	代回得 $15\times(12/60)=3$，且 $M=12$ 是满足 $0\le M\le60$ 的整数，因此答案成立。
\end{examplebox}
