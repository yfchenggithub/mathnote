\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用指示变量和期望线性性质，利用对称性计算单次期望。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（引入指示变量）：}]
			设 $X_i$ 为第 $i$ 次是否抽到次品的指示变量，$X_i=1$ 表示第 $i$ 次抽到次品，$X_i=0$ 表示未抽到。
			则抽到的次品总数 $X=\sum_{i=1}^{n}X_i$。
			\par\textbf{理由：} 每次抽取的次品计数之和即为总次品数。

		\item[\textbf{步骤二（计算单次期望）：}]
			由定义，$E(X_i)=1\cdot P(X_i=1)+0\cdot P(X_i=0)=P(X_i=1)$。
			因不放回抽取的对称性，第 $i$ 次抽到次品的概率与第一次相同，均为 $\frac{M}{N}$。
			故
			\[
				E(X_i)=\frac{M}{N}.
			\]
			\par\textbf{理由：} 所有 $N$ 件产品在第 $i$ 次被抽中的机会均等，次品共有 $M$ 件。

		\item[\textbf{步骤三（线性求和）：}]
			利用期望的线性性质（不限独立），
			\[
				\begin{aligned}
					E(X) &= E\!\left(\sum_{i=1}^{n}X_i\right) \\
					&= \sum_{i=1}^{n}E(X_i) \\
					&= \sum_{i=1}^{n}\frac{M}{N} \\
					&= n\cdot\frac{M}{N}.
			\end{aligned}
			\]
			\par\textbf{理由：} 期望线性可加，常数求和。

		\item[\textbf{步骤四（变量回代）：}]
			将 $E(X)$ 整理为题设形式：
			\[
				E(X)=\frac{nM}{N}.
			\]
			\par\textbf{理由：} 乘法交换形式变换。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	超几何分布的期望 $E(X)=\displaystyle\frac{nM}{N}$ 得证。
\end{proofbox}
