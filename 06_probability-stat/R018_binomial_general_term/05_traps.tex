\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（项序对应错误）}}]
			求第 $3$ 项时，误把项数 $3$ 当成 $r=3$，实际得到的是 $T_{4}$，不是第 $3$ 项。
		\item[\textbf{\textcolor{CTrap}{正确理解（序数差一原则）：}}]
			通项 $T_{r+1}$ 中，第 $k$ 项对应 $r=k-1$。故第 $3$ 项应取 $r=2$，对应 $T_{3}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆起始值）：}}]
			记忆公式时忽略 $r$ 从 $0$ 开始，误以为 $r$ 就是项数序号。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（系数混淆遗漏）}}]
			认为 $(2+3x)^n$ 展开式中含 $x^{r}$ 项的系数就是 $\binom{n}{r}$，忽略了 $2$ 和 $3$ 的幂次。
		\item[\textbf{\textcolor{CTrap}{正确理解（整体系数构成）：}}]
			项的系数应为 $\binom{n}{r} \cdot 2^{n-r} \cdot 3^{r}$，因为 $a=2$, $b=3x$，注意 $b$ 的系数 $3$ 也要 $r$ 次方。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略因子系数）：}}]
			套公式时未将 $a,b$ 的整体代入，只处理了组合数部分。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（指数方程错误）}}]
			在 $(x+1/x)^n$ 中求常数项时，直接写 $r=0$ 得到第一项，未解指数方程。
		\item[\textbf{\textcolor{CTrap}{正确理解（指数归零条件）：}}]
			常数项要求变量指数为 $0$，必须从通项化简后的指数表达式 $f(r)=0$ 解出 $r$。
		\item[\textbf{\textcolor{CTrap}{错因分析（未列指数方程）：}}]
			对通项中的变量指数没有显式写出并令其为零，靠猜测判断。
	\end{description}
\end{trapbox}
