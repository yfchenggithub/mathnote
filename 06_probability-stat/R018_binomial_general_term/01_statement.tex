\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正整数与变量）：}}] $n$ 为正整数，$a, b$ 为任意实数或代数式。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（通项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			展开式第 $r+1$ 项表达式。
			\[
				T_{r+1} = \binom{n}{r} a^{n-r} b^{r},
				\qquad
				r = 0,1,\dots, n.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（系数对称）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			首尾等距的两项二项式系数相等。
			\[
				\binom{n}{r} = \binom{n}{n-r},
				\qquad
				r = 0,1,\dots,n.
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒（项序对应）：}}
	通项 $T_{r+1}$ 中的 $r$ 从 $0$ 开始，故第 $1$ 项为 $T_1 = \binom{n}{0} a^n$, 第 $n+1$ 项为 $T_{n+1} = \binom{n}{n} b^n$.
\end{statementbox}
