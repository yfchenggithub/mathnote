\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{通项公式 $T_{r+1}=\binom{n}{r}a^{n-r}b^{r}$ 直接给出 $(a+b)^n$ 展开式中任意一项}.

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（指数正整）：} $n$ 必须是正整数。
		\item \textbf{条件 2（底数任意）：} $a,b$ 可为任意实数或代数式，但代入时需整体处理。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（项序偏移）：} 第 $k$ 项对应 $r=k-1$，不要直接令 $r=k$.
		\item \textbf{检查项（系数完整）：} 系数组成为 $\binom{n}{r}$ 乘以 $a^{n-r}$ 与 $b^{r}$ 中的数字系数，缺一不可。
	\end{itemize}
\end{summarybox}
