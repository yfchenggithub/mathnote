\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（事件互斥）：}}]
		      事件$A$与$B$不能同时发生，即$A\cap B=\varnothing$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（加法公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      若两事件互斥，并集的概率等于各自概率之和。
		      \[
			      P(A\cup B)=P(A)+P(B).
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（有限加法）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      多个两两互斥的事件，并集概率等于各概率之和。
		      \[
			      P\left(\bigcup_{i=1}^{n}A_i\right)
			      =
			      \sum_{i=1}^{n}P(A_i).
		      \]
	\end{description}
\end{statementbox}
