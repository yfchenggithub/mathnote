\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（众数选高矩形）：}}] 误将频率最大的组当作众数所在组。
		\item[\textbf{\textcolor{CTrap}{正确理解（选高度最大）：}}] 应选频率/组距最大的矩形（即最高矩形）的底边中点。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆纵标）：}}] 混淆了直方图纵坐标“频率/组距”与“频率”的概念。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（F指代不清）：}}] 将中位数公式中的$F$当作包括本组的累计频率。
		\item[\textbf{\textcolor{CTrap}{正确理解（F为前组和）：}}] $F$是前$j-1$组累计频率，不含第$j$组。
		\item[\textbf{\textcolor{CTrap}{错因分析（累加错误）：}}] 未仔细区分累计频率的定义，导致代入错误数值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式误用）：}}] 错误地将公式写为$m = L + \frac{0.5-F}{p_j}$而漏乘组距$d$，或误乘组距。
		\item[\textbf{\textcolor{CTrap}{正确理解（带组距乘法）：}}] 公式为$m = L + \frac{0.5-F}{p_j} \cdot d$，比例$\frac{0.5-F}{p_j}$乘以组距$d$才是偏移量。
		\item[\textbf{\textcolor{CTrap}{错因分析（忘记比例）：}}] 忽略了线性插值中横坐标增量与频率增量成正比，需乘以区间长度。
	\end{description}
\end{trapbox}
