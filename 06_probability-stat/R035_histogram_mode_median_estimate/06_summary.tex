\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{频率分布直方图中，众数取最高矩形底边中点，中位数用面积等分加线性插值公式估计。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（直方图完整）：} 直方图各组组距和高度已知，可计算各组频率及累计频率。
		\item \textbf{条件 2（中位数落组）：} 中位数所在组需通过累计频率确定（$0.5$落在哪一组）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（众数选高不选频）：} 众数对应频率/组距最大的矩形，而非频率最大的矩形。
		\item \textbf{检查项（中位数公式）：} 使用中位数公式时，确认$F$为前$j-1$组累计频率，并乘组距$d$。
	\end{itemize}
\end{summarybox}
