\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	组合数 $\binom{n}{m}$ 在 $m$ 与 $n-m$ 处值相等，即 $\binom{n}{m}=\binom{n}{n-m}$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（范围限制）：} $n,m\in\mathbb{N}$，$0\le m\le n$。
		\item \textbf{条件 2（整数要求）：} $n,m$ 取整数，不能为负数或分数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（定义域）：} 只在 $0\le m\le n$ 时成立，超出不能使用。
		\item \textbf{检查项（符号正确）：} 确保写成 $\binom{n}{m}=\binom{n}{n-m}$，不要写错上下标。
	\end{itemize}
\end{summarybox}
