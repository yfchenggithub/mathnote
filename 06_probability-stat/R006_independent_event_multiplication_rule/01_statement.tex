\begin{statementbox}

	\textbf{\textcolor{CStatement}{记号说明}}

	下文中，$AB=A\cap B$，表示事件 $A$ 与事件 $B$ 同时发生。

	\medskip

	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（事件独立）：}}]
		      事件 $A$ 与 $B$ 相互独立。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（乘积公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      若两事件相互独立，同时发生的概率等于各自概率的乘积。
		      \[
			      P(AB)=P(A)\,P(B).
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（多个事件）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      若多个事件相互独立，则它们同时发生的概率等于各事件概率的乘积。
		      \[
			      P(A_1A_2\cdots A_n)
			      =
			      P(A_1)\,P(A_2)\,\cdots\,P(A_n).
		      \]
	\end{description}

\end{statementbox}
