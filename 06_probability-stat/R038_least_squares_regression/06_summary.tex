\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	最小二乘估计公式为 $\hat{b}=\frac{\sum (x_i-\bar{x})(y_i-\bar{y})}{\sum (x_i-\bar{x})^2}$，$\hat{a}=\bar{y}-\hat{b}\bar{x}$，所得回归直线必过样本中心 $(\bar{x},\bar{y})$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（线性相关）：} 变量 $x$ 与 $y$ 存在线性相关关系（至少大致呈直线趋势）。
		\item \textbf{条件2（x有变异）：} 自变量 $x$ 的取值不完全相同，即 $\sum(x_i-\bar{x})^2 > 0$，保证分母非零。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母非零）：} 使用公式前必须确认 $x$ 有变异，否则回归无意义。
		\item \textbf{检查项（计算顺序）：} 按“均值→离差积和→$\hat{b}$→$\hat{a}$”顺序计算，避免代错。
	\end{itemize}
\end{summarybox}
