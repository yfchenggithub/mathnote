\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（线性相关假定）：}}] 给定成对样本数据 $(x_i, y_i)\ (i=1,2,\ldots,n)$ 满足线性相关关系，且 $\displaystyle\sum_{i=1}^n (x_i-\bar{x})^2\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最小二乘公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用样本数据计算回归系数的最小二乘估计。
			\[
				\hat{b} = \frac{\sum_{i=1}^n (x_i-\bar{x})(y_i-\bar{y})}{\sum_{i=1}^n (x_i-\bar{x})^2}
			\]
			\[
				\hat{a} = \bar{y} - \hat{b}\bar{x}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（过样本中心）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			回归直线必经过数据点的均值中心。
			\[
				\bar{y} = \hat{b}\bar{x} + \hat{a}
			\]
	\end{description}
\end{statementbox}
