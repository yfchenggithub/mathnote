\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{将数据排序后，用位置 $k=\lceil\frac{p}{100}n\rceil$ 确定第 $p$ 百分位数 $P_p=x_{(k)}$，它也是满足“至少 $p\text{\%}$ 数据不超过”的最小值。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件一（数据排序）：} 原始数据必须按非降序排列，得到顺序统计量。
		\item \textbf{条件二（样本容量）：} 需要明确样本容量 $n$，且百分等级 $p\in(0,100)$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（整数位置）：} $np/100$ 为整数时，直接取第 $i$ 个数据，勿取平均。
		\item \textbf{检查项（向上取整）：} 始终使用 $k=\lceil i\rceil$ 定位，避免采用四舍五入或其他方法。
	\end{itemize}
\end{summarybox}
