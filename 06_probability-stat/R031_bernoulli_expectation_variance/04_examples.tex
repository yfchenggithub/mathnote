	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	设随机变量 $X$ 服从两点分布，且 $P(X=1)=0.2$，求 $E(X)$，$D(X)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别分布）：}] 由题知 $X$ 服从两点分布，参数 $p=0.2$。
			\par\textbf{理由：} $P(X=1)=p$，故 $p=0.2$。
		\item[\textbf{第二步（求期望）：}] 根据两点分布期望公式：
			\[
				\begin{aligned}
					E(X) &= p \\
					&= 0.2.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用 $E(X)=p$。
		\item[\textbf{第三步（求方差）：}] 根据方差公式：
			\[
				\begin{aligned}
					D(X) &= p(1-p) \\
					&= 0.2\times 0.8 \\
					&= 0.16.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入 $p=0.2$ 计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$E(X)=0.2$，$D(X)=0.16$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知随机变量 $X$ 服从两点分布，且 $E(X)=0.3$。（1）求 $p$；（2）求 $D(X)$；（3）求 $P(X=0)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（反求参数）：}] 由 $E(X)=p$ 得 $p=0.3$。
			\par\textbf{理由：} 两点分布期望即成功概率。
		\item[\textbf{第二步（计算方差）：}] 根据方差公式：
			\[
				\begin{aligned}
					D(X) &= p(1-p) \\
					&= 0.3\times 0.7 \\
					&= 0.21.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入 $p=0.3$。
		\item[\textbf{第三步（求零点概率）：}] 计算 $P(X=0)$：
			\[
				\begin{aligned}
					P(X=0) &= 1-p \\
					&= 0.7.
			\end{aligned}
			\]
			\par\textbf{理由：} 两点分布概率和为 $1$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$p=0.3$，$D(X)=0.21$，$P(X=0)=0.7$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	设 $X$ 服从两点分布，$P(X=1)=0.4$，$Y=3X+2$，求 $E(Y)$，$D(Y)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求 $X$ 数字特征）：}] 由 $p=0.4$ 得
			\[
				E(X)=0.4.
			\]
			\[
				\begin{aligned}
					D(X) &= p(1-p) \\
					&= 0.4\times 0.6 \\
					&= 0.24.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接套用两点分布期望方差公式。
		\item[\textbf{第二步（线性变换期望）：}] 利用期望线性性质：
			\[
				\begin{aligned}
					E(Y) &= E(3X+2) \\
					&= 3E(X)+2 \\
					&= 3\times 0.4 + 2 \\
					&= 1.2 + 2 \\
					&= 3.2.
			\end{aligned}
			\]
			\par\textbf{理由：} $E(aX+b)=aE(X)+b$。
		\item[\textbf{第三步（线性变换方差）：}] 利用方差线性性质：
			\[
				\begin{aligned}
					D(Y) &= D(3X+2) \\
					&= 3^2 D(X) \\
					&= 9\times 0.24 \\
					&= 2.16.
			\end{aligned}
			\]
			\par\textbf{理由：} $D(aX+b)=a^2 D(X)$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$E(Y)=3.2$，$D(Y)=2.16$。
