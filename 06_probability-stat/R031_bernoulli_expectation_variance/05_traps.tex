	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（参数对应混淆）}}] 误以为 $P(X=0)=p$，从而将期望算成 $1-p$。
		\item[\textbf{\textcolor{CTrap}{正确理解（明确参数定义）：}}] 两点分布参数 $p$ 专指 $P(X=1)$，期望 $E(X)=p$。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义不清）：}}] 未分清取值 $0$ 和 $1$ 分别对应哪个概率，盲目臆测。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（方差公式记忆错误）}}] 常见错误：$D(X)=p$ 或 $D(X)=p(1+p)$ 等。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式）：}}] 方差公式为 $D(X)=p(1-p)$，且当 $p=0.5$ 时最大 $0.25$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}] 未从定义推导，只凭印象记错；或混淆 $E(X)$ 与 $D(X)$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略参数范围）}}] 盲目使用 $p>1$ 或 $p<0$ 计算方差，得到负值。
		\item[\textbf{\textcolor{CTrap}{正确理解（范围约束）：}}] 两点分布要求 $p\in[0,1]$，方差 $p(1-p)\ge 0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略条件）：}}] 没注意参数条件，直接套公式导致数学错误。
	\end{description}

\end{trapbox}
