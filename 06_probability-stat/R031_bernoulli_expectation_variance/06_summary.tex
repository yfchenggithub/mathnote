\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{两点分布期望 $E(X)=p$，方差 $D(X)=p(1-p)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（取值二元）：} 随机变量只能取 $0$ 或 $1$。
		\item \textbf{条件 2（概率归一）：} $P(X=1)=p$，$P(X=0)=1-p$，且 $p\in[0,1]$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（参数混淆）：} $p$ 是 $P(X=1)$，不是 $P(X=0)$。
		\item \textbf{检查项（参数范围）：} 计算方差前必须检查 $p\in[0,1]$。
	\end{itemize}

\end{summarybox}
