\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（误认过样本点）}}]
			认为回归直线必然通过某个原始数据点 $(x_i,y_i)$。
		\item[\textbf{\textcolor{CTrap}{正确理解（仅过中心点）：}}]
			回归直线只保证经过样本均值点 $(\bar{x},\bar{y})$，一般不经过其他点。
		\item[\textbf{\textcolor{CTrap}{错因分析（局部代表整体）：}}]
			将回归线想象成连接散点的折线，混淆了整体趋势和个体点。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（混淆样本中心）：}}]
			误将 $(\bar{x},\bar{y})$ 当作原点或数据中某一点。
		\item[\textbf{\textcolor{CTrap}{正确理解（均值定义）：}}]
			$\bar{x},\bar{y}$ 分别是 $x$ 和 $y$ 的算术平均数，代表数据中心位置。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）：}}]
			对样本均值的定义与含义理解不到位，导致与原始坐标混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略OLS条件）：}}]
			无论回归方程如何得到，都认为必过中心点。
		\item[\textbf{\textcolor{CTrap}{正确理解（OLS保证且含截距）：}}]
			只有最小二乘法拟合的一元线性回归直线（含截距项）才一定经过样本中心。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）：}}]
			记忆定理时只记住了结论，忽略了“最小二乘法”和“含截距”的前提。
	\end{description}

\end{trapbox}
