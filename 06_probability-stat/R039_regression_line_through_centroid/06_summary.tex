\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	回归直线 $\hat{y} = \hat{b}x + \hat{a}$ 必过样本中心 $(\bar{x},\bar{y})$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（最小二乘法）：} 回归系数由最小二乘法确定。
		\item \textbf{条件2（包含截距）：} 一元线性回归模型含有常数项 $\hat{a}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（过点误区）：} 回归直线只经过中心点，一般不经过其他数据点。
		\item \textbf{检查项（中心验证）：} 求得回归方程后，代入 $\bar{x}$ 检查是否得 $\bar{y}$，以验证计算正确性。
	\end{itemize}

\end{summarybox}
