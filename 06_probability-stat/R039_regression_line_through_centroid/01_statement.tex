\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（最小二乘估计）：}}]
			回归系数由最小二乘法确定，即参数 $\hat{a},\hat{b}$ 使得残差平方和最小。
		\item[\textbf{\textcolor{CStatement}{条件2（含截距项）：}}]
			一元线性回归模型包含截距项，即形式为 $\hat{y} = \hat{b}x + \hat{a}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（过样本中心）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			样本均值点 $(\bar{x},\bar{y})$ 在回归直线上。
			\[
				\bar{y} = \hat{b}\bar{x} + \hat{a}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（点斜式方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			回归直线可写成以样本中心为起点的点斜式。
			\[
				\hat{y} = \bar{y} + \hat{b}(x - \bar{x}).
			\]
	\end{description}

\end{statementbox}
