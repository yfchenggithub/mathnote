\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{组合数递推公式 $\binom{n}{k}=\binom{n-1}{k-1}+\binom{n-1}{k}$ 与对称性 $\binom{n}{k}=\binom{n}{n-k}$ 是组合计算的两大基本工具，前者体现分类计数，后者体现补集思想。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（定义域要求）：} $n,k$ 是自然数且 $0\le k\le n$。
		\item \textbf{条件2（递推范围）：} 递推公式仅适用于 $1\le k\le n-1$，边界直接取值 $1$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽视边界）：} 在 $k=0$ 或 $k=n$ 时仍套用递推公式。
		\item \textbf{检查项（公式适用）：} 使用递推前先确认 $1\le k\le n-1$，否则单独处理边界。
	\end{itemize}

\end{summarybox}
