\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定义域范围）：}}]
			$n,k\in\mathbb{N}$ 且 $0\le k\le n$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（杨辉递推）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每个组合数等于上一行相邻两数之和。
			\[
				\binom{n}{k}=\binom{n-1}{k-1}+\binom{n-1}{k}\quad (1\le k\le n-1)
			\]
			。
		\item[\textbf{\textcolor{CStatement}{推广结论（对称性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			组合数关于中点对称。
			\[
				\binom{n}{k}=\binom{n}{n-k}\quad (0\le k\le n)
			\]
			。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{边界情形：}}
	当 $k=0$ 或 $k=n$ 时，$\binom{n}{0}=\binom{n}{n}=1$；递推公式仅适用于 $1\le k\le n-1$。
\end{statementbox}
