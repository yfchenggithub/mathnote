\begin{statementbox}
\textbf{\textcolor{CStatement}{条件}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
\item[\textbf{\textcolor{CStatement}{条件1（独立重复试验）：}}] 每次试验均考察同一事件 $A$ 是否发生，且各次试验相互独立。
\item[\textbf{\textcolor{CStatement}{条件2（概率固定）：}}] 每次试验中事件 $A$ 发生的概率均为 $p$，其中 $0\le p\le 1$。
\end{description}

\textbf{\textcolor{CStatement}{结论}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
\item[\textbf{\textcolor{CStatement}{结论一（概率计算公式）：}}]
\textbf{\textcolor{CStatement}{直观描述：}} 设随机变量 $X$ 表示事件 $A$ 在 $n$ 次试验中发生的次数，则事件 $A$ 恰好发生 $k$ 次的概率，等于发生位置的组合数与对应概率乘积之积。
\[
P_n(k)=P(X=k)=\mathrm{C}_n^k p^k (1-p)^{n-k}, \quad k=0,1,\dots,n
\]

\item[\textbf{\textcolor{CStatement}{推广结论（归一恒等式）：}}]
\textbf{\textcolor{CStatement}{直观描述：}} 事件 $A$ 的所有可能发生次数覆盖全部情形，因此对应概率之和为 $1$。
\[
\sum_{k=0}^n P(X=k)
=
\sum_{k=0}^n \mathrm{C}_n^k p^k (1-p)^{n-k}
=
1
\]
\end{description}
\end{statementbox}
