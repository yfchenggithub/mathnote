\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{按比例分配样本，使样本各层比例与总体一致，每个个体被抽中的概率相等。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（分层已知）：} 总体可划分为 $L$ 层，每层个体数 $N_i$ 已知。
		\item \textbf{条件 2（样本容量确定）：} 总样本量 $n$ 已给定。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式方向）：} 务必使用 $n_i = n \cdot \frac{N_i}{N}$，不要颠倒分子分母。
		\item \textbf{检查项（总和验证）：} 计算后验证 $\sum_{i=1}^L n_i = n$。
	\end{itemize}
\end{summarybox}
