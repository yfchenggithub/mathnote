\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（已知层数及个体数）：}}] 总体由 $L$ 层组成，第 $i$ 层个体数为 $N_i$（$i=1,\dots,L$），总个体数 $N=\sum_{i=1}^L N_i$ 已知。
		\item[\textbf{\textcolor{CStatement}{条件 2（样本与结构一致）：}}] 从总体中抽取容量为 $n$ 的样本，且要求样本结构与总体结构一致，即按各层个体数比例分配样本。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（比例分配公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			各层样本量与该层个体数占总体的比例成正比。
			\[
				n_i = n \cdot \frac{N_i}{N}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（入样概率相等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每个个体被抽入样本的概率相同，均为抽样比。
			\[
				\frac{n_i}{N_i} = \frac{n}{N}
			\]
	\end{description}
\end{statementbox}
