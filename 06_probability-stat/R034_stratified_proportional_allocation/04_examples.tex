	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	某高中高一、高二、高三学生数分别为 800 人、700 人、500 人。现用分层抽样（按比例分配）抽取样本容量为 40，求各年级应抽取的人数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算总人数）：}] 计算全校总人数 $N = 800 + 700 + 500 = 2000$。
			\par\textbf{理由：} 公式 $n_i = n \cdot N_i / N$ 需要总体总数 $N$。
		\item[\textbf{第二步（应用比例公式）：}] 利用公式 $n_i = n \cdot \frac{N_i}{N}$ 计算各年级样本量。
			\par\textbf{理由：} 按比例分配要求。
		\item[\textbf{第三步（代入数值）：}] 高一年级 $n_1 = 40 \times \frac{800}{2000} = 16$；高二年级 $n_2 = 40 \times \frac{700}{2000} = 14$；高三年级 $n_3 = 40 \times \frac{500}{2000} = 10$。
			\par\textbf{理由：} 直接计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{各年级应抽取人数分别为 16 人、14 人、10 人，总和为 40 人。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	某公司男员工 200 人。采用分层按比例抽样，抽取样本容量 50，其中男员工被抽中 20 人。求该公司女员工人数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知量）：}] 设公司总人数为 $N$，女员工人数为 $x$，则 $N = 200 + x$。
			\par\textbf{理由：} 需要建立方程。
		\item[\textbf{第二步（列方程）：}] 由男员工样本量公式 $n_{\text{男}} = n \cdot \frac{N_{\text{男}}}{N}$，代入得 $20 = 50 \cdot \frac{200}{N}$。
			\par\textbf{理由：} 男员工样本量应满足比例公式。
		\item[\textbf{第三步（解总人数）：}] 解方程 $20 = \frac{10000}{N}$，得 $N = \frac{10000}{20} = 500$。
			\par\textbf{理由：} 代数变形。
		\item[\textbf{第四步（求女员工）：}] 女员工人数 $x = N - 200 = 500 - 200 = 300$。
			\par\textbf{理由：} 总人数已知。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{该公司女员工人数为 300 人。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	某社区有 A、B、C 三个小区，人口数分别为 5000、3000、2000。现采用分层抽样（按比例分配）抽取样本容量 200 人进行收入调查。求各小区应抽取多少人，并验证每个居民被抽中的概率是否相等。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（总人口）：}] 计算总人口 $N = 5000 + 3000 + 2000 = 10000$。
			\par\textbf{理由：} 需要总体总数代入公式。
		\item[\textbf{第二步（求各层样本量）：}] 利用比例公式 $n_i = n \cdot \frac{N_i}{N}$ 分别计算：A 小区 $n_A = 200 \times \frac{5000}{10000} = 100$；B 小区 $n_B = 200 \times \frac{3000}{10000} = 60$；C 小区 $n_C = 200 \times \frac{2000}{10000} = 40$。
			\par\textbf{理由：} 按比例分配样本量。
		\item[\textbf{第三步（计算入样概率）：}] 各小区居民入样概率分别为 $p_A = \frac{n_A}{N_A} = \frac{100}{5000} = 0.02$，$p_B = \frac{60}{3000} = 0.02$，$p_C = \frac{40}{2000} = 0.02$。
			\par\textbf{理由：} 入样概率公式 $p_i = \frac{n_i}{N_i}$。
		\item[\textbf{第四步（判断相等）：}] 三个概率均为 $0.02$，故相等。
			\par\textbf{理由：} 数值比较。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{A、B、C 三小区分别抽取 100、60、40 人；每个居民被抽中的概率均为 $0.02$，说明按比例分配是等概率抽样。}
