\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆比例关系）}}] 计算时将比例公式误写成 $n_i = N_i \cdot \frac{N}{n}$ 或 $n_i = \frac{N_i}{n}$ 等形式。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式结构校正）：}}] 正确的公式为 $n_i = n \cdot \frac{N_i}{N}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（比例方向误解）：}}] 将条件“样本比例等于总体比例”错误理解为“总体比例等于样本比例”，导致分子分母颠倒。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗忘总样本量）}}] 直接使用各层人数 $N_i$ 作为样本量，即 $n_i = N_i$，或只乘以层数而非抽样比。
		\item[\textbf{\textcolor{CTrap}{正确理解（抽样比相乘）：}}] 各层样本量必须乘以总体抽样比 $n/N$，即 $n_i = \frac{n}{N} N_i$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略样本总容量）：}}] 忽略了样本容量 $n$ 的约束，认为按比例就是按层人数分配，未考虑总样本量。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（取整处理不当）}}] 当 $n_i$ 计算值为非整数时，直接四舍五入取整，导致各层样本之和偏离预定总样本量 $n$。
		\item[\textbf{\textcolor{CTrap}{正确理解（调整取整保总和）：}}] 各层样本量取整后需调整使总和等于 $n$，通常采用四舍五入后调整某一层。
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性四舍五入）：}}] 忽视各层样本量之和必须等于总样本容量这一约束，未进行总量平衡。
	\end{description}
\end{trapbox}
