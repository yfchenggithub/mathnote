\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（有限等可能）：}}]
			样本空间 $\Omega$ 是非空有限集合，且每个基本事件出现的可能性相等。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（概率公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			事件 $A$ 的概率等于 $A$ 包含的基本事件个数与样本空间总基本事件个数的比值。
			\[
				P(A)=\frac{n(A)}{n(\Omega)}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（对立事件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			事件 $A$ 与它的对立事件 $\bar{A}$ 的概率之和恒为 $1$。
			\[
				P(\bar{A})=1-P(A).
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒（等可能检验）：}}
	\begin{itemize}[leftmargin=*]
		\item 使用前必须确认试验的每个基本结果是等可能的。
		\item 计数时注意区分有序无序、有放回无放回，避免重复或遗漏。
	\end{itemize}

\end{statementbox}
