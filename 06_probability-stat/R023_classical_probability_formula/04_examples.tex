\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	投掷一枚均匀骰子一次，求出现的点数大于4的概率。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（样本空间）：}]
			\[
				n(\Omega)=6.
			\]
			\par\textbf{理由：} 骰子有 $6$ 个面，每个结果等可能，故基本事件总数为 $6$。
		\item[\textbf{第二步（事件计数）：}]
			\[
				n(A)=2.
			\]
			\par\textbf{理由：} 点数大于 $4$ 的点是 $5$ 和 $6$，共 $2$ 个。
		\item[\textbf{第三步（概率计算）：}]
			\[
				\begin{aligned}
					P(A) &= \frac{2}{6} \\ &= \frac{1}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入古典概型公式 $P(A)=n(A)/n(\Omega)$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{答案为 $\displaystyle\frac13$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	袋中装有 $3$ 个红球、$2$ 个白球，从袋中一次性任取 $2$ 个球，求取到的两个球都是红球的概率。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（样本空间）：}]
			\[
				\begin{aligned}
					n(\Omega) &= \binom{5}{2} \\ &= 10.
			\end{aligned}
			\]
			\par\textbf{理由：} 从 $5$ 个球中任取 $2$ 个（不计顺序），总取法为组合数 $\binom{5}{2}=10$。
		\item[\textbf{第二步（事件计数）：}]
			\[
				\begin{aligned}
					n(A) &= \binom{3}{2} \\ &= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 事件 $A$ 要求 $2$ 个都是红球，需从 $3$ 个红球中选 $2$ 个，取法为 $\binom{3}{2}=3$。
		\item[\textbf{第三步（概率计算）：}]
			\[
				P(A)=\frac{3}{10}.
			\]
			\par\textbf{理由：} 代入古典概型公式 $P(A)=\frac{n(A)}{n(\Omega)}$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{答案为 $\displaystyle\frac{3}{10}$。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	从数字 $1,2,3,4,5$ 中任取两个不同的数，求它们的和为偶数的概率。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（样本空间）：}]
			\[
				\begin{aligned}
					n(\Omega) &= \binom{5}{2} \\ &= 10.
			\end{aligned}
			\]
			\par\textbf{理由：} 从 $5$ 个数中任选两个（不分先后），总选法为 $\binom{5}{2}=10$。
		\item[\textbf{第二步（分类计数）：}]
			和为偶数，则两数同奇或同偶。奇数有 $1,3,5$ 共 $3$ 个，偶数有 $2,4$ 共 $2$ 个。事件 $A$ 包含的取法数为
			\[
				\begin{aligned}
					n(A) &= \binom{3}{2}+\binom{2}{2} \\ &= 3+1 \\ &= 4.
			\end{aligned}
			\]
			\par\textbf{理由：} 同奇取法 $\binom{3}{2}=3$ 种，同偶取法 $\binom{2}{2}=1$ 种，相加即得。
		\item[\textbf{第三步（概率计算）：}]
			\[
				\begin{aligned}
					P(A) &= \frac{4}{10} \\ &= \frac{2}{5}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入古典概型公式 $P(A)=n(A)/n(\Omega)$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{答案为 $\displaystyle\frac{2}{5}$。}
\end{examplebox}
