\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略等可能）}}]
			把非等可能试验当作古典概型处理，例如将不均匀硬币、不公正骰子直接用古典概型公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（等可能条件）：}}]
			古典概型要求每个基本事件等可能，应用前必须验证等可能性是否成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（误用公式）：}}]
			没有检查等可能条件而直接套用公式，导致概率计算错误，如将各选项概率不均匀的问题按等可能计算。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（计数不一致）}}]
			在计算样本空间总数和事件包含数时，使用不同计数规则。例如，分母使用排列而分子使用组合。
		\item[\textbf{\textcolor{CTrap}{正确理解（统一计数方式）：}}]
			$n(\Omega)$ 和 $n(A)$ 必须在同一规则下计数，都是有序或都是无序，都是有放回或都是无放回。
		\item[\textbf{\textcolor{CTrap}{错因分析（规则不一）：}}]
			混淆排列与组合、有放回与无放回，导致分子分母不匹配，概率计算结果错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（分母计算错）}}]
			样本空间计数时重复或遗漏，例如从含有相同元素的样本中计数时忽略元素间的区别，或者把同一类元素当作不可区分。
		\item[\textbf{\textcolor{CTrap}{正确理解（元素区分）：}}]
			在古典概型中，每个基本事件是原子结果，即使物体外观相同也应视为不同个体（例如编号），保证基本事件等可能。
		\item[\textbf{\textcolor{CTrap}{错因分析（元素同化）：}}]
			认为相同颜色的球不可区分，错误地减少总基本事件数，从而导致概率计算偏差。
	\end{description}

\end{trapbox}
