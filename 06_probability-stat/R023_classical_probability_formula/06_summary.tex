\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{古典概型概率 $P(A)$ 等于事件 $A$ 包含的基本事件个数 $n(A)$ 与样本空间总个数 $n(\Omega)$ 的比值 $n(A)/n(\Omega)$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（有限样本空间）：} 样本空间必须是非空有限集合。
		\item \textbf{条件2（等可能性）：} 每个基本事件发生的可能性相等。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（等可能验证）：} 使用公式前必须确保试验满足等可能性，否则公式不适用。
		\item \textbf{检查项（计数一致性）：} 计算 $n(\Omega)$ 和 $n(A)$ 时，必须采用相同的计数规则（排列/组合、有放回/无放回）。
	\end{itemize}

\end{summarybox}
