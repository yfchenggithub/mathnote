\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（原始数据条件）：}}] 已知数据 $x_1, x_2, \dots, x_n$（$n \ge 2$），其标准差为 $s_x$。
		\item[\textbf{\textcolor{CStatement}{条件2（线性变换形式）：}}] 设常数 $a, k \in \mathbb{R}$，定义新数据 $y_i = a + k x_i$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（标准差变换公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			新数据的标准差等于原标准差乘以系数 $|k|$。
			\[
				s_y = |k|\,s_x .
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（平移不变性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数据平移后标准差保持不变。
			\[
				s_{x+a} = s_x .
			\]
			特别地，当 $k=0$ 时，所有 $y_i$ 均等于常数 $a$，标准差 $s_y = 0$，上述公式仍然成立。
	\end{description}

\end{statementbox}
