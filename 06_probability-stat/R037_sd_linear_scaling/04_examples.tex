	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知一组数据 $x$ 的标准差 $s_x = 2$，对每个数据作线性变换 $y = 3x + 1$，求新数据 $y$ 的标准差 $s_y$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别变换系数）：}] 变换形式为 $y = a + kx$，其中 $a = 1$，$k = 3$。平移量 $a$ 不影响标准差，只需关注 $k$。
		      \par\textbf{理由：} 标准差线性变换性质 $s_y = |k| \, s_x$。
		\item[\textbf{第二步（代入公式）：}]
		      代入 $s_x = 2$，$|k| = 3$，得
		      \[
			      \begin{aligned}
				      s_y & = 3 \times 2 \\
				          & = 6 .
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 直接套用公式计算。
		\item[\textbf{第三步（得出答案）：}] 因此新数据 $y$ 的标准差为 $6$。
		      \par\textbf{理由：} 计算完成。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\mathbf{s_y = 6}$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数据 $y$ 的标准差 $s_y = 12$，且 $y$ 是由另一组数据 $x$ 经过变换 $y = -2x + 5$ 得到，求原数据 $x$ 的标准差 $s_x$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定系数）：}] 由变换 $y = -2x + 5$ 得 $k = -2$，$|k| = 2$。平移 $5$ 不影响标准差。
		      \par\textbf{理由：} $s_y = |k| s_x$。
		\item[\textbf{第二步（公式逆用）：}]
		      代入 $s_y = 12$，$|k| = 2$，得 $12 = 2 \cdot s_x$，故 $s_x = 6$。
		      \[
			      \begin{aligned}
				      s_x & = \frac{12}{2} \\
				          & = 6 .
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 解一元一次方程。
		\item[\textbf{第三步（确认答案）：}] 原数据 $x$ 的标准差为 $6$。
		      \par\textbf{理由：} 代入验证：$s_y = |-2| \times 6 = 12$，符合题意。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\mathbf{s_x = 6}$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数据 $x$ 的标准差 $s_x = 5$，经过线性变换 $y = kx - 3$ 后标准差变为 $15$。求系数 $k$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出关系式）：}] 由 $y = kx - 3$，平移量 $a = -3$ 不影响标准差，故 $s_y = |k| s_x$。
		      \par\textbf{理由：} 标准差线性变换性质。
		\item[\textbf{第二步（代入已知）：}]
		      代入 $s_y = 15$，$s_x = 5$，得
		      \[
			      15 = |k| \times 5 \quad\Rightarrow\quad |k| = 3 .
		      \]
		      \par\textbf{理由：} 解方程。
		\item[\textbf{第三步（求系数）：}] 由 $|k| = 3$ 得 $k = 3$ 或 $k = -3$。
		      \par\textbf{理由：} 绝对值方程有两个解，均可使标准差成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$k = 3$ 或 $k = -3$}。
