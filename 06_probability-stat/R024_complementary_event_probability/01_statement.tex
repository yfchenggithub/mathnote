\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（随机试验）：}}] 在任意随机试验中，设事件 $A$ 及其对立事件 $\overline{A}$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（概率公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对立事件概率之和为1。
			\[
				\boxed{P(\overline{A}) = 1 - P(A)}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（常见变形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正面难求时，反面突破。
			\[
				P(A) = 1 - P(\overline{A})
			\]
	\end{description}
\end{statementbox}
