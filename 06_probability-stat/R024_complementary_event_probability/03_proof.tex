\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用概率的规范性 $P(\Omega)=1$ 和有限可加性，由对立事件的关系 $A \cup \overline{A}=\Omega$ 与 $A \cap \overline{A}=\emptyset$ 直接推出。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（定义事件）：}] 设 $A$ 为任意事件，$\overline{A}$ 为其对立事件。则
			\[
				A \cup \overline{A} = \Omega, \quad A \cap \overline{A} = \emptyset .
			\]
			\par\textbf{理由：} 对立事件的定义要求两者互斥且覆盖所有可能结果。

		\item[\textbf{步骤二（概率公理）：}] 由概率的规范性，$P(\Omega)=1$；由有限可加性，互斥事件并的概率等于概率之和，所以
			\[
				P(A \cup \overline{A}) = P(A) + P(\overline{A}).
			\]
			\par\textbf{理由：} 概率公理是证明的基础，直接应用。

		\item[\textbf{步骤三（代入关系）：}] 将步骤一的关系代入步骤二：
			\[
				\begin{aligned}
					P(A) + P(\overline{A}) &= P(A \cup \overline{A}) \\
					&= P(\Omega) \\
					&= 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用等量代换得到概率的方程。

		\item[\textbf{步骤四（移项得公式）：}] 移项即得
			\[
				P(\overline{A}) = 1 - P(A).
			\]
			\par\textbf{理由：} 由等式两边同时减去 $P(A)$ 得到。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{公式 $P(\overline{A}) = 1 - P(A)$ 揭示了概率中“对立互补”的核心关系，是概率计算反向思维的理论依据。}
\end{proofbox}
