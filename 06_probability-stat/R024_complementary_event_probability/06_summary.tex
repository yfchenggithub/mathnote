\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对立事件概率之和为1，正面困难时先求反面。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（同一试验）：} $A$ 与 $\overline{A}$ 必须在同一个随机试验中定义。
		\item \textbf{条件2（完备互斥）：} $A\cup\overline{A}=\Omega$ 且 $A\cap\overline{A}=\emptyset$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（概念辨析）：} 注意对立事件与互斥事件的区别，避免误用。
		\item \textbf{检查项（反面计算）：} 计算 $\overline{A}$ 的概率时，确保计数准确、分类完备。
	\end{itemize}
\end{summarybox}
