\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（常数项误乘）}}] 错误做法：$E(a\xi+b)=aE(\xi)+ab$。
		\item[\textbf{\textcolor{CTrap}{正确理解（常数直接加）：}}] $E(a\xi+b)=aE(\xi)+b$，常数项 $b$ 不乘系数 $a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（分配律误用）：}}] 把 $a\xi+b$ 误看成 $a(\xi+b)$，从而错误地把 $b$ 也乘以 $a$。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分布限制误判）}}] 错误做法：认为该性质只对离散型成立，或要求 $\xi$ 服从特殊分布。
		\item[\textbf{\textcolor{CTrap}{正确理解（期望存在即可）：}}] 只要期望存在且有限，无论离散、连续还是混合分布，线性性质都成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏抽象理解）：}}] 没有理解期望本质上是关于概率分布的加权平均或积分，与具体分布形式无关。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（误以为线性组合要独立）}}] 错误做法：认为 $E(\xi+\eta)=E(\xi)+E(\eta)$ 需要 $\xi,\eta$ 相互独立。
		\item[\textbf{\textcolor{CTrap}{正确理解（期望线性不需独立）：}}] 只要相关期望存在且有限，期望的线性性质不要求随机变量相互独立。
		\item[\textbf{\textcolor{CTrap}{错因分析（与乘积期望混淆）：}}] 独立性常用于 $E(\xi\eta)=E(\xi)E(\eta)$，但不用于期望的加法线性性质。
	\end{description}

	\medskip

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（与方差混淆）}}] 错误做法：$D(a\xi+b)=aD(\xi)+b$。
		\item[\textbf{\textcolor{CTrap}{正确理解（方差平方系数）：}}] 若 $D(\xi)$ 存在，则 $D(a\xi+b)=a^2 D(\xi)$，常数项对方差无影响。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式混淆）：}}] 将期望线性性质机械套用到方差，未区分期望和方差的运算差异。
	\end{description}
\end{trapbox}
