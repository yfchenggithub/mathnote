\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（期望存在）：}}] 随机变量 $\xi$ 的期望 $E(\xi)$ 存在且有限。
		\item[\textbf{\textcolor{CStatement}{条件 2（常数）：}}] $a,b$ 为任意实数常数。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线性运算）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 常数倍提出，常数项直接加在期望后方。
		      \[
			      E(a\xi+b)=aE(\xi)+b.
		      \]

		\item[\textbf{\textcolor{CStatement}{常见变形（线性组合）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 若 $\xi_1,\xi_2,\dots,\xi_n$ 的期望均存在且有限，$a_1,a_2,\dots,a_n,b$ 为实数常数，则有限个随机变量的线性组合满足
		      \[
			      E\bigl(\sum_{i=1}^n a_i\xi_i+b\bigr)=\sum_{i=1}^n a_iE(\xi_i)+b.
		      \]
		      \textbf{\textcolor{CStatement}{特别提醒：}} 该性质不要求 $\xi_1,\xi_2,\dots,\xi_n$ 相互独立。
	\end{description}
\end{statementbox}
