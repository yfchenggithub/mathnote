\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	若随机变量 $\xi$ 的期望存在且有限，且 $a,b$ 为实数常数，则
	\[
		E(a\xi+b)=aE(\xi)+b.
	\]
	也就是说：系数提出，常数直接加。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（期望存在）：} $\xi$ 的期望 $E(\xi)$ 必须存在且有限。
		\item \textbf{条件 2（线性变换）：} $a,b$ 是任意实数常数，变换为 $a\xi+b$ 形式。
		\item \textbf{条件 3（线性组合）：} 若推广到 $\sum_{i=1}^{n}a_i\xi_i+b$，则每个 $E(\xi_i)$ 都应存在且有限。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（常数项独立）：} 常数 $b$ 直接加，不要乘 $a$。
		\item \textbf{独立性提醒：} 期望的加法线性性质不要求随机变量相互独立。
		\item \textbf{方差对比：} 若 $D(\xi)$ 存在，则 $D(a\xi+b)=a^2D(\xi)$，不是 $aD(\xi)+b$。
		\item \textbf{检查项（特殊值验证）：} 可先取特殊值验证，例如当 $a=1,b=0$ 时恒等，当 $a=0$ 时 $E(b)=b$。
	\end{itemize}
\end{summarybox}
