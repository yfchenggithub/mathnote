\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知随机变量 $\xi$ 的期望 $E(\xi)=3$，求 $E(4\xi+2)$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用性质）：}] 由 $E(a\xi+b)=aE(\xi)+b$，得 $E(4\xi+2)=4E(\xi)+2$。
		      \par\textbf{理由：} 已知 $E(\xi)$ 存在，线性变换后的期望可直接套用线性性质。
		\item[\textbf{第二步（代入已知）：}] 已知 $E(\xi)=3$，代入得 $4\times3+2$。
		      \par\textbf{理由：} 代入数值。
		\item[\textbf{第三步（计算结果）：}] $4\times3+2=12+2=14$。
		      \par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$E(4\xi+2)=14$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 设随机变量 $\xi$ 的分布列为 $P(\xi=1)=0.2,P(\xi=2)=0.3,P(\xi=3)=0.5$，求 $E(2\xi-1)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求原始期望）：}] $E(\xi)=1\times0.2+2\times0.3+3\times0.5=0.2+0.6+1.5=2.3$。
		      \par\textbf{理由：} 离散型期望定义 $\sum x_i p_i$。
		\item[\textbf{第二步（应用线性性质）：}] $E(2\xi-1)=2E(\xi)-1$。
		      \par\textbf{理由：} 线性性质 $a=2,b=-1$。
		\item[\textbf{第三步（代值计算）：}] $2\times2.3-1=4.6-1=3.6$。
		      \par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$E(\xi)=2.3,\quad E(2\xi-1)=3.6$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 已知随机变量 $\xi$ 的期望 $E(\xi)=5$，且 $E(2\xi+k)=12$，求常数 $k$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用线性性质）：}] $E(2\xi+k)=2E(\xi)+k$。
		      \par\textbf{理由：} $k$ 为常数，满足期望线性性质。
		\item[\textbf{第二步（代入已知）：}] 代入 $E(\xi)=5$，得 $2\times5+k=10+k$。
		      \par\textbf{理由：} 数值代入。
		\item[\textbf{第三步（列方程求解）：}] 由 $E(2\xi+k)=12$ 得 $10+k=12$，解得 $k=2$。
		      \par\textbf{理由：} 解一元一次方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$k=2$}
\end{examplebox}
