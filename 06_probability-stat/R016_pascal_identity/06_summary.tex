\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{组合数递推公式是杨辉三角的核心构造规则，本质是加法计数原理的体现。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（非负整数）：} $n,m$ 为非负整数且 $n\geq m$。
		\item \textbf{条件 2（边界约定）：} 当 $k<0$ 或 $k>n$ 时定义 $\binom{n}{k}=0$，保证公式对边界成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（方向混淆）：} 注意公式是前两项相加等于后一项，不要写反方向。
		\item \textbf{检查项（定义域确认）：} 应用前检查 $n,m$ 是否满足条件，必要时利用约定补充零项。
	\end{itemize}
\end{summarybox}
