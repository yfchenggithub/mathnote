\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（非负整数）：}}] 非负整数 $n,m$，满足 $n \geq m \geq 1$（或一般地 $n,m \in \mathbb{N}_0, n \geq m$），约定当 $k<0$ 或 $k>n$ 时 $\binom{n}{k}=0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（递推公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			杨辉三角中每个数等于其左上方与右上方两数之和。
			\[
				\binom{n}{m} + \binom{n}{m-1} = \binom{n+1}{m}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（降阶递推）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $n$ 替换为 $n-1$ 得到等价递推形式。
			\[
				\binom{n}{m} = \binom{n-1}{m} + \binom{n-1}{m-1}, \quad (n \geq 1, m \geq 1)
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{适用范围：}}
	非负整数 $n,m$ 且 $n \geq m$；边界由约定自动补充。
\end{statementbox}
