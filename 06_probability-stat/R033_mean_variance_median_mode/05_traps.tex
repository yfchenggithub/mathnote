	\textbf{\textcolor{CTrap}{误区一：频数表中把不同取值各算一次}}

	\input{assets/tikz/R033_frequency_weight_compare.tex}

	\noindent\textbf{错误：}把 $60,70,80$ 三种取值各算一次，得到错误平均数 $70$。
	\par\noindent\textbf{错因：}这等于假设三种成绩各出现一次，丢掉了频数。
	\par\noindent\textbf{替代：}按频数重复（或加权）可得 $\bar x=69$；计算方差也须使用同一组频数。

	\smallskip
	\textbf{\textcolor{CTrap}{误区二：不排序就取中间}}

	\input{assets/tikz/R033_median_sort_middle.tex}

	\noindent\textbf{错误：}在原书写顺序中直接取“中间”的数。
	\par\noindent\textbf{错因：}中位数的“中间”是大小次序的中间，原书写位置没有统计意义。
	\par\noindent\textbf{替代：}先排序；奇数个取正中一项，偶数个取中间两项的平均。

	\smallskip
	\textbf{\textcolor{CTrap}{误区三：把众数当成最大的数}}

	\noindent\textbf{错误：}把最大值 $9$ 当成众数。
	\par\noindent\textbf{错因：}众数比较出现次数，不比较数值大小。
	\par\noindent\textbf{替代：}$2$ 出现两次，故众数为 $2$；最高频数并列时可以有多个众数。

	\smallskip
	\textbf{\textcolor{CTrap}{误区四：混淆离差和、方差与标准差}}

	\noindent\textbf{错误：}由 $\sum(x_i-\bar x)=0$ 断言方差为 $0$，或把 $s^2$ 当成标准差。
	\par\noindent\textbf{错因：}离差会正负抵消；方差平均的是离差平方，而且方差不等于标准差。
	\par\noindent\textbf{替代：}先算 $s^2$；若题目问标准差，再取非负平方根 $s=\sqrt{s^2}$。

	\smallskip
	\textbf{\textcolor{CTrap}{误区五：见到“样本”就机械除以 $n-1$}}

	\noindent\textbf{错误：}见到“样本”就机械地把方差分母改为 $n-1$。
	\par\noindent\textbf{错因：}分母取决于定义和任务语境；$n-1$ 是无偏样本方差的口径。
	\par\noindent\textbf{替代：}本结论使用分母 $n$；若题目另给公式，则按题目定义。

\end{trapbox}
