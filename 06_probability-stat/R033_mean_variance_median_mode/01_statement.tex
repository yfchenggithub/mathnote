\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（数据集合）}}] 给定一组数据 $x_1, x_2, \dots, x_n$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平均数）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数据的平均水平。

			算术平均：$\bar{x} = \frac{1}{n}\sum_{i=1}^n x_i$。

			加权平均：$\bar{x} = \frac{\sum_{i=1}^n x_i f_i}{\sum_{i=1}^n f_i}$（$f_i$ 为权重）。

		\item[\textbf{\textcolor{CStatement}{结论二（方差与标准差）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			度量数据波动程度。

			算术方差：$s^2 = \frac{1}{n}\sum_{i=1}^n (x_i-\bar{x})^2$（$\bar{x}$ 为算术平均）。

			加权方差：$s^2 = \frac{\sum_{i=1}^n (x_i-\bar{x})^2 f_i}{\sum_{i=1}^n f_i}$（$\bar{x}$ 为加权平均）。

			标准差：$s = \sqrt{s^2}$。

		\item[\textbf{\textcolor{CStatement}{结论三（中位数）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数据中间位置的数值。

			将数据从小到大排序。若样本量 $n$ 为奇数，中位数为第 $\frac{n+1}{2}$ 个数据；若 $n$ 为偶数，中位数为第 $\frac{n}{2}$ 个和第 $\frac{n}{2}+1$ 个数据的平均数。

		\item[\textbf{\textcolor{CStatement}{结论四（众数）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			出现频率最高的数值。

			数据中出现次数最多的数值。可能存在多个众数。

		\item[\textbf{\textcolor{CStatement}{推广结论（方差等价变形）}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			方差等于数据平方的均值减去均值的平方。

			\[
				s^2 = \frac{1}{n}\sum_{i=1}^n x_i^2 - \bar{x}^2.
			\]
	\end{description}

\end{statementbox}
