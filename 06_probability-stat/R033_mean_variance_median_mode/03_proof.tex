\begin{proofbox}

	\textbf{\textcolor{CProof}{一、平均数为何是平方距离的唯一最小点}}

	对数据 $1,2,3,4,5$，把候选中心记作 $a$，平均平方距离是
	$Q(a)=2+(a-3)^2$：

	\input{assets/tikz/R033_mean_minimizer_parabola.tex}

	任取实数 $a$。由平均数定义，
	\[
		\sum_{i=1}^{n}(x_i-\bar x)
		=\sum_{i=1}^{n}x_i-n\bar x=0.
	\]
	\input{assets/tikz/R033_cross_term_balance.tex}
	图中 $1,2,6$ 并不关于 $3$ 对称，但 $-2+(-1)+3=0$；因此交叉项中的带符号离差和仍为零。
	于是
	\[
		\begin{aligned}
		\frac1n\sum_{i=1}^{n}(x_i-a)^2
		&=\frac1n\sum_{i=1}^{n}\bigl[(x_i-\bar x)+(\bar x-a)\bigr]^2\\
		&=\frac1n\sum_{i=1}^{n}(x_i-\bar x)^2
		 +\frac{2(\bar x-a)}n\sum_{i=1}^{n}(x_i-\bar x)
		 +(\bar x-a)^2\\
		&=s^2+(a-\bar x)^2\ge s^2.
		\end{aligned}
	\]
	等号成立当且仅当 $a=\bar x$。因此，平均数确实是使离差平方和最小的唯一位置。

	\medskip
	\textbf{\textcolor{CProof}{二、方差等价公式}}

	从定义展开平方，并使用 $\sum_{i=1}^{n}x_i=n\bar x$：
	\[
		\begin{aligned}
		s^2
		&=\frac1n\sum_{i=1}^{n}(x_i-\bar x)^2\\
		&=\frac1n\left(\sum_{i=1}^{n}x_i^2
		-2\bar x\sum_{i=1}^{n}x_i+n\bar x^2\right)\\
		&=\frac1n\sum_{i=1}^{n}x_i^2-2\bar x^2+\bar x^2\\
		&=\frac1n\sum_{i=1}^{n}x_i^2-\bar x^2.
		\end{aligned}
	\]
	频数表只是在每个取值前加入重复次数 $f_j$；将每项乘以 $f_j$、把 $n$ 换成 $F=\sum f_j$，同样的推导仍成立。

	\medskip
	\textbf{\textcolor{CProof}{三、零方差的等号条件}}

	\input{assets/tikz/R033_zero_variance_condition.tex}

	因为每一项 $(x_i-\bar x)^2\ge0$，这些非负数的平均为 $0$，当且仅当每一项都为 $0$。因此
	\[
		s^2=0\iff x_1=x_2=\cdots=x_n=\bar x.
	\]

\end{proofbox}
