\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	比较 $\dfrac{\ln 2}{\ln 3}$ 和 $\dfrac{\ln 4}{\ln 5}$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 观察 $2<3$，$4=2+2$，$5=3+2$，因此取 $a=2,\;b=3,\;c=2$，满足 $1<a<b$，$c>0$。
			\par\textbf{理由：} 符合对数糖水不等式的前提条件。

		\item[\textbf{第二步（套用结论）：}] 由对数糖水不等式，
			\[
				\frac{\ln 2}{\ln 3} < \frac{\ln(2+2)}{\ln(3+2)} = \frac{\ln 4}{\ln 5}.
			\]
			\par\textbf{理由：} 直接代入不等式左边部分 $\dfrac{\ln a}{\ln b}<\dfrac{\ln(a+c)}{\ln(b+c)}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{\dfrac{\ln 2}{\ln 3}<\dfrac{\ln 4}{\ln 5}}$。
\end{examplebox}
\begin{examplebox}
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	比较 $\dfrac{\ln 5}{\ln 6}$ 和 $\dfrac{\ln 6}{\ln 7}$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造增量）：}] 取 $a=5,\;b=6,\;c=1$，满足 $1<5<6$，$c=1>0$。
			\par\textbf{理由：} $6=5+1$，$7=6+1$。

		\item[\textbf{第二步（应用结论）：}] 由对数糖水不等式，
			\[
				\frac{\ln 5}{\ln 6} < \frac{\ln(5+1)}{\ln(6+1)} = \frac{\ln 6}{\ln 7}.
			\]
			\par\textbf{理由：} 直接使用不等式左边部分。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{\dfrac{\ln 5}{\ln 6}<\dfrac{\ln 6}{\ln 7}}$。
\end{examplebox}
\begin{examplebox}
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	比较 $\log_3 4$ 和 $\log_4 5$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化同底）：}] 利用换底公式，
			\[
				\log_3 4 = \frac{\ln 4}{\ln 3},
				\qquad
				\log_4 5 = \frac{\ln 5}{\ln 4}.
			\]
			\par\textbf{理由：} 换底公式 $\log_a b = \dfrac{\ln b}{\ln a}$。

		\item[\textbf{第二步（构造糖水不等式）：}] 考虑 $a=3,\;b=4,\;c=1$，则 $1<3<4$，$c=1>0$。由对数糖水不等式，
			\[
				\frac{\ln 3}{\ln 4} < \frac{\ln 4}{\ln 5}.
			\]
			\par\textbf{理由：} 代入左边部分 $\dfrac{\ln a}{\ln b}<\dfrac{\ln(a+c)}{\ln(b+c)}$。

		\item[\textbf{第三步（取倒数变形）：}] 上式两边都为正，取倒数得
			\[
				\frac{\ln 4}{\ln 3} > \frac{\ln 5}{\ln 4}.
			\]
			即 $\log_3 4 > \log_4 5$。
			\par\textbf{理由：} 不等式两边为正时取倒数不等号反向。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{\log_3 4 > \log_4 5}$。
\end{examplebox}
