\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（真数大于1）：}}] $1<a<b$，且 $c>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对数糖水不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			给两个大于1的真数加上相同正数 $c$，比值增大但仍小于1。
			\[
				\frac{\ln a}{\ln b} < \frac{\ln(a+c)}{\ln(b+c)} < 1.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（取倒数形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			取倒数后得到大于1的不等式链，方向相反。
			\[
				1 < \frac{\ln(b+c)}{\ln(a+c)} < \frac{\ln b}{\ln a}.
			\]
	\end{description}
\end{statementbox}
