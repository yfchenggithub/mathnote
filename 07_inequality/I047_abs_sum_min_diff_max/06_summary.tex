\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心（距离最值）}}
	\textbf{当 $a<b$ 时，$|x-a|+|x-b|$ 的最小值为 $|a-b|$，$\bigl| |x-a|-|x-b| \bigr|$ 的最大值为 $|a-b|$，最值点由 $x$ 相对于 $a,b$ 的位置决定。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（两个定点）：} 存在两个不同的实常数 $a,b$（不要求顺序，但大小影响区间）。
		\item \textbf{条件 2（系数为一）：} 两个绝对值项的系数均为 $1$，才能直接套用。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（取等相反）：} 和的取等区间是 $[a,b]$；差的取等区间是 $(-\infty,a]\cup[b,\infty)$，两者互补。
		\item \textbf{检查项（差的范围）：} 从 $\bigl| |x-a|-|x-b| \bigr| \le |a-b|$ 推出 $|x-a|-|x-b| \in [-|a-b|, |a-b|]$，注意符号可以取负。
	\end{itemize}

\end{summarybox}
