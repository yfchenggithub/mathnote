\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	求函数 $f(x)=|x-2|+|x+1|$ 的最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
			识别 $a=-1$, $b=2$（满足 $a<b$）。
			\par\textbf{理由：} 原式与结论形式 $|x-a|+|x-b|$ 一致。
		\item[\textbf{第二步（套用结论）：}]
			由结论，最小值为 $|a-b|$。
			\[
				\begin{aligned}
					|a-b| &= |-1-2| \\ &= 3.
			\end{aligned}
			\]
			故最小值为 $3$。
			\par\textbf{理由：} 两绝对值之和的最小值等于两点间的距离。
		\item[\textbf{第三步（写出取等）：}]
			当且仅当 $a\le x\le b$ 即 $-1\le x\le 2$ 时取到最小值。
			\par\textbf{理由：} 结论给出取等条件为 $x$ 在 $a,b$ 之间。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$f(x)$ 的最小值为 $3$，取等区间为 $[-1,2]$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	若不等式 $|x-1|+|x-4|\le a$ 的解集非空，求实数 $a$ 的最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求最小值）：}]
			令 $a_0=1$, $b_0=4$，则 $|x-1|+|x-4|$ 的最小值为 $|4-1|=3$。
			\par\textbf{理由：} 结论直接给出和的最小值。
		\item[\textbf{第二步（非空条件）：}]
			不等式解集非空等价于 $a$ 不小于左边的最小值。
			\par\textbf{理由：} 左边有最小值 $3$，若 $a<3$，则不等式不可能成立；若 $a\ge 3$，则至少存在 $x$（如 $x=2.5$）使不等式成立。
		\item[\textbf{第三步（得出参量）：}]
			所以 $a$ 的最小值为 $3$。
			\par\textbf{理由：} 由 $a\ge 3$ 且 $a$ 最小，故 $a_{\min}=3$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_{\min}=3$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	求函数 $p(x)=|x-2|-|x+1|$ 的值域。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
			$p(x)=|x-2|-|x+1|$，对应 $a=-1$, $b=2$。
			\par\textbf{理由：} 与差的形式 $|x-a|-|x-b|$ 一致。
		\item[\textbf{第二步（应用差不等式）：}]
			由结论，
			\[
				\begin{aligned}
					\bigl| |x-2|-|x+1| \bigr| &\le |2-(-1)| \\ &= 3,
			\end{aligned}
			\]
			即 $p(x) \in [-3,3]$。
			\par\textbf{理由：} 差绝对值不等式给出绝对值的上界。
		\item[\textbf{第三步（验证等号可行）：}]
			当 $x\ge 2$ 时，
			\[
				\begin{aligned}
					p(x) &= (x-2)-(x+1) \\ &= -3;
			\end{aligned}
			\]
			当 $x\le -1$ 时，
			\[
				\begin{aligned}
					p(x) &= -(x-2)-[-(x+1)] \\ &= 3.
			\end{aligned}
			\]
			\par\textbf{理由：} 取等条件要求 $x\le a$ 或 $x\ge b$，代入计算得到边界值，所以确能取到 $-3$ 和 $3$，故值域为 $[-3,3]$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$p(x)$ 的值域为 $[-3,3]$。}
\end{examplebox}
