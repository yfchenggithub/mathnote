\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略几何意义）}}]
			遇到 $|x-a|+|x-b|$ 最值时，直接展开绝对值分类讨论，计算繁琐且耗时。
		\item[\textbf{\textcolor{CTrap}{正确理解（数轴法）：}}]
			利用数轴上两点间距离直接得到最值，无需分类。
		\item[\textbf{\textcolor{CTrap}{错因分析（未抓本质）：}}]
			绝对值定义为距离，但未能转化为几何直观，导致方法低效。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（差的范围误解）}}]
			认为 $|x-a|-|x-b|$ 的最大值为 $|a-b|$，忽略它可以取负值。
		\item[\textbf{\textcolor{CTrap}{正确理解（对称区间）：}}]
			$|x-a|-|x-b|$ 的取值范围是 $[-|a-b|,|a-b|]$，即最小值为 $-|a-b|$，最大值为 $|a-b|$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忘记符号）：}}]
			只记住了绝对值不等式 $\bigl||x-a|-|x-b|\bigr|\le |a-b|$，没有转化为不等价的两边范围。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（取等条件遗忘）}}]
			应用最小值或范围时，不考虑 $x$ 的取值区间，直接套用最值。
		\item[\textbf{\textcolor{CTrap}{正确理解（区间配套）：}}]
			和的最值对应 $x$ 在 $[a,b]$ 内，差的最值对应 $x$ 在 $(-\infty,a]\cup[b,\infty)$。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件脱节）：}}]
			只记忆结论数值，忽略了条件中 $x$ 的取值范围。
	\end{description}

\end{trapbox}
