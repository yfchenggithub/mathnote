\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（两定点间距）：}}]
			实数 $a, b$ 满足 $a<b$；$x$ 为任意实数。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（和的最小值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数轴上 $x$ 到 $a$ 与 $b$ 的距离之和恒不小于 $a$ 与 $b$ 之间的距离。
			\[
				|x-a|+|x-b|\ge |a-b|.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（差的绝对值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数轴上 $x$ 到 $a$ 与 $b$ 的距离之差（的绝对值）恒不大于 $a$ 与 $b$ 之间的距离。
			\[
				\bigl| |x-a|-|x-b| \bigr| \le |a-b|.
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{等号/取等条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{取等一（区间内部）：}}]
			$|x-a|+|x-b|=|a-b|$ 当且仅当 $a\le x\le b$。
		\item[\textbf{\textcolor{CStatement}{取等二（区间外部）：}}]
			$\bigl||x-a|-|x-b|\bigr|=|a-b|$ 当且仅当 $x\le a$ 或 $x\ge b$。
	\end{description}
	\textbf{\textcolor{CStatement}{常见变形（变量推广）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{任意实数对：}}]
			若 $a,b$ 为任意实数且 $a\neq b$，以上两个不等式仍成立，只需注意 $a,b$ 大小顺序自动确定区间；若 $a>b$，则 $|x-a|+|x-b|$ 的最小值仍为 $|a-b|$，取等区间为 $[b,a]$。
	\end{description}

\end{statementbox}
