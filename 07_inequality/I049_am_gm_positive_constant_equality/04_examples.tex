\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $x>0$，$y>0$，且 $x+y=10$，求 $xy$ 的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（检查条件）：}] 变量 $x$，$y$ 均为正数，满足“一正”；和 $x+y=10$ 是定值，满足“二定”。
			\par\textbf{理由：} 基本不等式求最值的前提条件是各项为正且和或积为定值。

		\item[\textbf{第二步（套用公式）：}] 由基本不等式 $x+y\ge2\sqrt{xy}$，代入 $x+y=10$ 得
			\[
				10\ge2\sqrt{xy}.
			\]
			\par\textbf{理由：} 基本不等式对正数成立，直接代入。

		\item[\textbf{第三步（求最值并验等）：}] 由上式得 $\sqrt{xy}\le5$，故 $xy\le25$。当且仅当 $x=y=5$ 时等号成立，满足 $x+y=10$，等号可取得。
			\par\textbf{理由：} 解不等式得最大值，验证等号条件满足。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$xy$ 的最大值为 \textbf{$25$}，当 $x=y=5$ 时取得。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0$，$b>0$，且 $ab=16$，求 $a+b$ 的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（检查条件）：}] $a>0$，$b>0$，满足“一正”；积 $ab=16$ 是定值，满足“二定”。
			\par\textbf{理由：} 使用基本不等式的前提条件成立。

		\item[\textbf{第二步（套用公式）：}] 由基本不等式 $a+b\ge2\sqrt{ab}$，代入 $ab=16$ 得
			\[
				\begin{aligned}
					a+b &\ge 2\sqrt{ab} \\
					&= 2\sqrt{16} \\
					&= 8.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接代入积的数值，计算右边。

		\item[\textbf{第三步（验等得最值）：}] 等号当 $a=b$ 时成立，此时 $ab=a^2=16$，解得 $a=b=4$（正数），满足条件。故 $a+b$ 的最小值为 $8$。
			\par\textbf{理由：} 验证等号可取后，最小值即得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$a+b$ 的最小值为 \textbf{$8$}，当 $a=b=4$ 时取得。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $x>0$，$y>0$，且 $\dfrac{2}{x}+\dfrac{1}{y}=1$，求 $x+y$ 的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（配凑定值）：}] 由条件得 $1=\dfrac{2}{x}+\dfrac{1}{y}$，考虑
			\[
				x+y=(x+y)\cdot1=(x+y)\left(\dfrac{2}{x}+\dfrac{1}{y}\right).
			\]
			\par\textbf{理由：} 乘 $1$ 不改变值，目的是展开后出现定积形式。

		\item[\textbf{第二步（展开变形）：}] 展开右边：
			\[
				\begin{aligned}
					(x+y)\left(\dfrac{2}{x}+\dfrac{1}{y}\right)
					&=2+\dfrac{x}{y}+\dfrac{2y}{x}+1 \\
					&=3+\left(\dfrac{x}{y}+\dfrac{2y}{x}\right).
			\end{aligned}
			\]
			\par\textbf{理由：} 使用分配律，注意 $\dfrac{x}{y}$ 和 $\dfrac{2y}{x}$ 均为正数，且相乘为定值 $2$。

		\item[\textbf{第三步（基本不等式）：}] 对 $\dfrac{x}{y}+\dfrac{2y}{x}$ 使用基本不等式：
			\[
				\begin{aligned}
					\dfrac{x}{y}+\dfrac{2y}{x}
					&\ge 2\sqrt{\dfrac{x}{y}\cdot\dfrac{2y}{x}} \\
					&= 2\sqrt{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 两项均为正，积为常数 $2$，可用基本不等式求最小值。

		\item[\textbf{第四步（得最值并验等）：}] 从而
			\[
				x+y\ge3+2\sqrt{2}.
			\]
			等号当 $\dfrac{x}{y}=\dfrac{2y}{x}$ 时成立，结合条件 $\dfrac{2}{x}+\dfrac{1}{y}=1$ 可解得 $x=2+\sqrt{2}$，$y=1+\sqrt{2}$（均为正），等号可取。
			\par\textbf{理由：} 推导最小值，并验证等号条件同时满足题目条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$x+y$ 的最小值为 \textbf{$3+2\sqrt{2}$}，当 $x=2+\sqrt{2},\;y=1+\sqrt{2}$ 时取得。
\end{examplebox}
