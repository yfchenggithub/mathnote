\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（和为定值）：}}] 两个正数 $a,b$ 满足 $a+b = 1$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（倒数和不小）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当两个正数之和为 $1$ 时，它们的倒数之和至少为 $4$。
			\[
				\frac{1}{a}+\frac{1}{b}\geq 4,
			\]
			当且仅当 $a=b=\frac12$ 时等号成立。
		\item[\textbf{\textcolor{CStatement}{推广结论（和一般化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将和从 $1$ 推广到任意正数 $k$：若 $a,b>0$ 且 $a+b=k$，则
			\[
				\frac{1}{a}+\frac{1}{b}\geq \frac{4}{k},
			\]
			当且仅当 $a=b=\frac{k}{2}$ 时等号成立。
	\end{description}

	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	主结论中当 $a=b=\frac12$ 时取等；推广形式中当 $a=b=\frac{k}{2}$ 时取等。
\end{statementbox}
