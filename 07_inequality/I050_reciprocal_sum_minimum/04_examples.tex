\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0,\ b>0$，$a+b=1$，求 $\frac1a+\frac1b$ 的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 确认 $a>0,\ b>0$ 且 $a+b=1$，符合结论的使用条件。
			\par\textbf{理由：} 结论要求正数和为1，条件完全吻合。
		\item[\textbf{第二步（套用结论）：}] 直接由结论得
			\[
				\frac1a+\frac1b\ge4,
			\]
			因此最小值为 $4$。
			\par\textbf{理由：} 该结论已证明，可直接使用。
		\item[\textbf{第三步（取等验证）：}] 当 $a=b=\frac12$ 时，满足 $a+b=1$，且 $\frac1a+\frac1b=4$，说明最小值可以取到。
			\par\textbf{理由：} 取等条件与结论一致。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为 $4$。}
	\medskip

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0,\ b>0$，$a+b=2$，求 $\frac1a+\frac1b$ 的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（恒等变形）：}] 将条件化为 $\frac a2+\frac b2=1$，令 $x=\frac a2,\ y=\frac b2$，则 $x>0,\ y>0$，$x+y=1$。
			\par\textbf{理由：} 转化为和为1的标准形式。
		\item[\textbf{第二步（套用结论并回代）：}] 由结论知 $\frac1x+\frac1y\ge4$，即 $\frac1{a/2}+\frac1{b/2}\ge4$，从而 $\frac2a+\frac2b\ge4$，所以 $\frac1a+\frac1b\ge2$。
			\par\textbf{理由：} 利用结论并逆推回原变量。
		\item[\textbf{第三步（取等计算）：}] 当 $x=y=\frac12$ 即 $a=b=1$ 时，$\frac1a+\frac1b=2$，为最小值。
			\par\textbf{理由：} 取等条件由结论推得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为 $2$。}
	\medskip

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0,\ b>0$，$a+b=1$，求 $(1+\frac1a)(1+\frac1b)$ 的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（展开乘积）：}] 展开得
			\[
				(1+\frac1a)(1+\frac1b)=1+\frac1a+\frac1b+\frac1{ab}.
			\]
			\par\textbf{理由：} 将所求式拆分为已知结构之和。
		\item[\textbf{第二步（倒数和取值）：}] 由结论得 $\frac1a+\frac1b\ge4$。
			\par\textbf{理由：} 直接应用已知结论。
		\item[\textbf{第三步（积最值）：}] 由基本不等式 $ab\le\left(\frac{a+b}{2}\right)^2$，即 $ab\le\frac14$，且 $ab>0$，得 $\frac1{ab}\ge4$。
			\par\textbf{理由：} 和定积最大，取倒数后不等号反向。
		\item[\textbf{第四步（合并求最值）：}] 将上述结果代入展开式：
			\[
				\begin{aligned}
					1+\frac1a+\frac1b+\frac1{ab} &\ge 1+4+4 \\
					&= 9.
			\end{aligned}
			\]
			当且仅当 $\frac1a+\frac1b=4$ 且 $ab=\frac14$ 同时成立，即 $a=b=\frac12$ 时取等。
			\par\textbf{理由：} 两个不等式取等条件一致，最小值可达。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为 $9$。}
\end{examplebox}
