\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{若 $a>0,\ b>0$ 且 $a+b=S$（$S>0$），则 $\frac1a+\frac1b\ge\frac{4}{S}$，等号当 $a=b=\frac{S}{2}$ 时成立。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（正数限制）：} $a>0,\ b>0$。
		\item \textbf{条件 2（和为定值）：} $a+b$ 等于某正常数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（取等条件）：} 应用后必须验证 $a=b$ 是否在允许范围内。
		\item \textbf{检查项（前提确认）：} 应用前必须确认变量为正且和为定值。
	\end{itemize}
\end{summarybox}
