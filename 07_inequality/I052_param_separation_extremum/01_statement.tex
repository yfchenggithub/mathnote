\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（变量分离）：}}] 参数$a$已与变量$x$分离，且函数$f(x)$在区间$D$上存在最大值和最小值。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（恒成立化最大值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若要参数不小于函数值恒成立，则参数不小于函数的最大值。
			\[
				a \geq f(x) \;(\forall x\in D) \iff a \geq \max_{x\in D} f(x).
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（恒成立化最小值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若要参数不大于函数值恒成立，则参数不大于函数的最小值。
			\[
				a \leq f(x) \;(\forall x\in D) \iff a \leq \min_{x\in D} f(x).
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{常见变形（存在性问题）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{变形一（存在性 $\ge$）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			存在$x$使参数不小于函数值，等价于参数不小于函数的最小值。
			\[
				\exists x\in D,\; a \geq f(x) \iff a \geq \min_{x\in D} f(x).
			\]

		\item[\textbf{\textcolor{CStatement}{变形二（存在性 $\le$）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			存在$x$使参数不大于函数值，等价于参数不大于函数的最大值。
			\[
				\exists x\in D,\; a \leq f(x) \iff a \leq \max_{x\in D} f(x).
			\]
	\end{description}
\end{statementbox}
