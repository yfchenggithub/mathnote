\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	利用二次式的两根式分解与实数乘积符号法则，分 $\Delta>0$、$\Delta=0$、$\Delta<0$ 三种情况推导。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（因式分解）：}]
			\[
				ax^2+bx+c = a(x-x_1)(x-x_2)\quad(\Delta>0).
			\]
			\par\textbf{理由：} 若 $\Delta>0$，二次方程有两个不等实根，可写成两根式。
		\item[\textbf{步骤二（符号分析）：}]
			由于 $a>0$，只需考虑 $(x-x_1)(x-x_2)$ 的符号。
			\[
				\begin{aligned}
					x<x_1 &\Rightarrow (x-x_1)<0,\;(x-x_2)<0 \Rightarrow (x-x_1)(x-x_2)>0,\\ x_1<x<x_2 &\Rightarrow (x-x_1)>0,\;(x-x_2)<0 \Rightarrow (x-x_1)(x-x_2)<0,\\ x>x_2 &\Rightarrow (x-x_1)>0,\;(x-x_2)>0 \Rightarrow (x-x_1)(x-x_2)>0.
			\end{aligned}
			\]
			\par\textbf{理由：} 由实数乘法“同号得正，异号得负”直接得出。
		\item[\textbf{步骤三（$\Delta=0$ 情形）：}]
			\[
				ax^2+bx+c = a\left(x+\frac{b}{2a}\right)^2 \ge 0,
			\]
			等号仅当 $x=-\frac{b}{2a}$ 时成立。
			\par\textbf{理由：} 完全平方式恒非负，$a>0$ 保证整体非负。
		\item[\textbf{步骤四（$\Delta<0$ 情形）：}]
			\[
				ax^2+bx+c > 0 \quad \forall x\in\mathbb{R}.
			\]
			\par\textbf{理由：} 开口向上且无实根，函数图像全在 $x$ 轴上方。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	综合以上三种情况，即得原定理所述解集。
\end{proofbox}
