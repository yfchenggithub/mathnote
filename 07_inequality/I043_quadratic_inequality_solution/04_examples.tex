\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $x^2-5x+6>0$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别标准形式）：}]
			不等式已经是 $ax^2+bx+c>0$ 且 $a=1>0$。
			\par\textbf{理由：} 确保可用开口向上结论。
		\item[\textbf{第二步（计算判别式与根）：}]
			计算 $\Delta=(-5)^2-4\times1\times6=1>0$，求根 $x=\frac{5\pm\sqrt{1}}{2}=2,3$，得 $x_1=2$，$x_2=3$。
			\par\textbf{理由：} 判别式正时有两个实根，用求根公式。
		\item[\textbf{第三步（套结论）：}]
			因为 $a>0$ 且 $\Delta>0$，口诀“大于取两边”，解集为 $(-\infty,2)\cup(3,+\infty)$。
			\par\textbf{理由：} 直接代入结论。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	解集为 $(-\infty,2)\cup(3,+\infty)$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $-2x^2+3x+2<0$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化正系数）：}]
			原不等式两边乘 $-1$，得 $2x^2-3x-2>0$（注意不等号反转）。
			\par\textbf{理由：} 口诀只适用于二次项系数为正。
		\item[\textbf{第二步（计算判别式与根）：}]
			对于 $2x^2-3x-2>0$，$a=2>0$，$\Delta=(-3)^2-4\times2\times(-2)=9+16=25>0$，两根 $x=\frac{3\pm5}{4}$，即 $x_1=-\frac12$，$x_2=2$。
			\par\textbf{理由：} 求根公式。
		\item[\textbf{第三步（套结论）：}]
			$a>0$，$\Delta>0$，“大于取两边”，得解集 $(-\infty,-\frac12)\cup(2,+\infty)$。
			\par\textbf{理由：} 直接套用结论。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	原不等式的解集为 $(-\infty,-\frac12)\cup(2,+\infty)$。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	解关于 $x$ 的不等式 $x^2-(a+1)x+a>0$（$a$ 为常数）。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			二次项系数为 $1>0$，判别式 $\Delta=(a+1)^2-4a=(a-1)^2\ge0$。
			\par\textbf{理由：} 二次项系数为正，满足结论条件。
		\item[\textbf{第二步（求根与分类）：}]
			$x=\frac{a+1\pm|a-1|}{2}$。分类：当 $a<1$，$|a-1|=1-a$，得 $x_1=a$，$x_2=1$；当 $a=1$，得重根 $x=1$；当 $a>1$，$|a-1|=a-1$，得 $x_1=1$，$x_2=a$。
			\par\textbf{理由：} 绝对值导致根的大小顺序与 $a$ 有关。
		\item[\textbf{第三步（套结论）：}]
			由二次项系数 $>0$ 且 $\Delta\ge0$，应用“大于取两边”：$a<1$ 时解集 $(-\infty,a)\cup(1,+\infty)$；$a=1$ 时解集 $\mathbb{R}\setminus\lbrace1\rbrace$；$a>1$ 时解集 $(-\infty,1)\cup(a,+\infty)$。
			\par\textbf{理由：} 直接套用结论中对应情形。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	分类结果：$a<1$ 时 $(-\infty,a)\cup(1,+\infty)$；$a=1$ 时 $\mathbb{R}\setminus\lbrace1\rbrace$；$a>1$ 时 $(-\infty,1)\cup(a,+\infty)$。
\end{examplebox}
