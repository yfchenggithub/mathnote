\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略系数正负）}}]
			直接套用“大于取两边”，忘记检查二次项系数是否为正。
		\item[\textbf{\textcolor{CTrap}{正确理解（转换符号）：}}]
			必须先将不等式化为 $a>0$ 的标准形式；若 $a<0$，两边乘 $-1$ 并反转不等号。
		\item[\textbf{\textcolor{CTrap}{错因分析（口诀前提）：}}]
			口诀只适用于 $a>0$，否则结论完全相反。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（$\Delta<0$ 混合作答）}}]
			认为 $a>0$ 且 $\Delta<0$ 时 $ax^2+bx+c<0$ 的解集也是 $\mathbb{R}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分不等号）：}}]
			$\Delta<0$ 时 $>0$ 解集为 $\mathbb{R}$，$<0$ 解集为 $\emptyset$。
		\item[\textbf{\textcolor{CTrap}{错因分析（图像遗漏）：}}]
			只记“无解”或“恒正”，忽略了不等号方向。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（$\Delta=0$ 判断失误）}}]
			误认为 $ax^2+bx+c>0$（$a>0$）当 $\Delta=0$ 时解集为 $\mathbb{R}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（去掉等根）：}}]
			$>0$ 的解集为 $\mathbb{R}$ 除去等于根的点，$<0$ 解集为空。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略零值）：}}]
			在根处函数值等于 $0$，不满足严格大于 $0$。
	\end{description}

\end{trapbox}
