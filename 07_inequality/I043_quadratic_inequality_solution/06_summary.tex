\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{对于 $a>0$ 的一元二次不等式，解集由 $\Delta$ 与两根决定，“大于取两边，小于取中间”。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（系数为正）：} 二次项系数必须满足 $a>0$。
		\item \textbf{条件2（有根时需比较）：} 当 $\Delta>0$ 时需确定根的大小顺序 $x_1<x_2$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（检查开口方向）：} 使用口诀前务必确认 $a>0$，否则结论相反。
		\item \textbf{检查项（判别式范围）：} 先判断 $\Delta$ 正负，再根据对应规则写出解集。
	\end{itemize}

\end{summarybox}
