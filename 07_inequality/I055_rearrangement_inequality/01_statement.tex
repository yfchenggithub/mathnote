\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（两列递增）：}}]
			设有两个实数序列 $a_1 \leq a_2 \leq \cdots \leq a_n$ , $b_1 \leq b_2 \leq \cdots \leq b_n$ , 且两序列项数相同。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（顺序最大）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同序配对的和最大。\par
			\[
				\sum_{i=1}^{n} a_i b_i \geq \sum_{i=1}^{n} a_i b_{\sigma(i)} .
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（反序最小）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			反序配对的和最小。\par
			\[
				\sum_{i=1}^{n} a_i b_{n+1-i} \leq \sum_{i=1}^{n} a_i b_{\sigma(i)} .
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（n=2特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当项数 $n=2$ 时，排序不等式简化为一个常用不等式。\par
			\[
				a_1 b_1 + a_2 b_2 \geq a_1 b_2 + a_2 b_1 .
			\]
	\end{description}
	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	当且仅当所有 $a_i$ 相等或所有 $b_i$ 相等时，等号成立。
\end{statementbox}
