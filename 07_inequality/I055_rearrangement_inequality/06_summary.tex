\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{排序不等式：同序乘积和最大，反序乘积和最小，乱序和居中。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（序列递增）：} 两序列必须递增（或递减）且项数相同。
		\item \textbf{条件 2（等号条件）：} 当某序列各项全相等时等号成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（先排序）：} 应用前务必确认序列已排序。
		\item \textbf{检查项（取等判断）：} 注意序列是否可能全相等，导致等号成立。
	\end{itemize}

\end{summarybox}
