\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略递增前提）}}]
			直接套用排序不等式时不检查序列是否递增，导致结论错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（先排序）：}}]
			使用前必须确认序列已经按从小到大（或从大到小）排列，否则应先排序。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件缺失）：}}]
			学生只记忆不等式形式，忽略所需条件：序列必须有序。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（颠倒大小顺序）}}]
			错误认为反序和最大、顺序和最小。
		\item[\textbf{\textcolor{CTrap}{正确理解（牢记结论）：}}]
			顺序和最大，反序和最小。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}]
			对结论的正反关系记忆不清。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（乱序和必严格）}}]
			误认为乱序和总是严格介于顺序和与反序和之间。
		\item[\textbf{\textcolor{CTrap}{正确理解（等号情形）：}}]
			当某个序列各项全相等时，顺序和、反序和、乱序和全部相等，此时等号成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略取等）：}}]
			不理解或忘记排序不等式取等条件。
	\end{description}

\end{trapbox}
