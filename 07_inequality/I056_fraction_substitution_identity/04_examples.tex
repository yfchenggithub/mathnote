\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $x+\frac{1}{x}=3$，求 $x^2+\frac{1}{x^2}$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
		      由已知条件 $x+\frac{1}{x}=3$，所求为 $x^2+\frac{1}{x^2}$，两者满足恒等式。
		      \par\textbf{理由：} 该换元恒等式 $x^2+\frac{1}{x^2}=t^2-2$ 适合此情形。
		\item[\textbf{第二步（代入公式）：}]
		      \[
			      \begin{aligned}
				      x^2+\frac{1}{x^2}
				       & = \left(x+\frac{1}{x}\right)^2 - 2 \\
				       & = 3^2 - 2                          \\
				       & = 7.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 直接代入 $t=3$ 计算。
		\item[\textbf{第三步（写出答案）：}]
		      所以 $x^2+\frac{1}{x^2}=7$。
		      \par\textbf{理由：} 计算即得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x^2+\frac{1}{x^2}=7$}。

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $x>0$，求函数 $f(x)=x+\frac{1}{x}$ 的最小值，并指出取等条件。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定范围）：}]
		      由结论 $\left|x+\frac{1}{x}\right|\ge 2$，结合 $x>0$，得 $x+\frac{1}{x}\ge 2$。
		      \par\textbf{理由：} 当 $x>0$ 时，$x$ 与 $\frac{1}{x}$ 均为正，和为正，只能落在 $[2,+\infty)$。
		\item[\textbf{第二步（取等条件）：}]
		      当 $x=1$ 时，$f(1)=1+1=2$，达到最小值。
		      \par\textbf{理由：} 由等号成立条件 $x=1$（$x>0$）即得最小值。
		\item[\textbf{第三步（下结论）：}]
		      最小值为 $2$，取等时 $x=1$。
		      \par\textbf{理由：} 综合步骤一、二。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	最小值为 \textbf{$2$}，取等时 \textbf{$x=1$}。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $x^2+\frac{1}{x^2}=5$，求 $x+\frac{1}{x}$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（换元设定）：}]
		      令 $t=x+\frac{1}{x}$，则 $x^2+\frac{1}{x^2}=t^2-2$。
		      \par\textbf{理由：} 恒等式直接代入。
		\item[\textbf{第二步（列方程）：}]
		      由 $5=t^2-2$，解得 $t^2=7$，所以 $t=\pm\sqrt{7}$。
		      \par\textbf{理由：} 解一元二次方程，开平方时正负都要保留。
		\item[\textbf{第三步（验证范围）：}]
		      因为 $|t|=\sqrt{7}>2$，满足 $\left|x+\frac{1}{x}\right|\ge 2$，故两个值均有效。
		      \par\textbf{理由：} 反求 $x+\frac{1}{x}$ 时，需要检查是否落在 $(-\infty,-2]\cup[2,+\infty)$ 内。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x+\frac{1}{x}=\pm\sqrt{7}$}。
\end{examplebox}
