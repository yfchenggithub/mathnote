\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略定义域）：}}]
			学生容易忘记 $x\neq 0$，直接对 $x+\frac{1}{x}$ 使用换元。
		\item[\textbf{\textcolor{CTrap}{正确理解（前提条件）：}}]
			使用该恒等式前必须确认 $x\neq 0$，否则 $1/x$ 无意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定式）：}}]
			学生习惯于整式变形，忽略分式的分母非零约束。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（取等片面）：}}]
			误以为 $x+\frac{1}{x}\ge 2$ 对所有 $x\neq 0$ 恒成立。
		\item[\textbf{\textcolor{CTrap}{正确理解（严格范围）：}}]
			正确结论为 $|x+\frac{1}{x}|\ge 2$；当 $x>0$ 时 $\ge 2$，当 $x<0$ 时 $\le -2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（正负遗漏）：}}]
			受基本不等式 $a+b\ge 2\sqrt{ab}$ 要求正数的影响，忽略 $x$ 为负的情形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开方丢负）：}}]
			由 $x^2+\frac{1}{x^2}$ 反求 $x+\frac{1}{x}$ 时只取正值一个。
		\item[\textbf{\textcolor{CTrap}{正确理解（正负仍需）：}}]
			通过恒等式解得 $t^2$ 后开平方，$t$ 可取正或负，且需验证 $|t|\ge 2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（疏漏符号）：}}]
			对称思维不足，默认 $x+\frac{1}{x}$ 为正，忽略 $x$ 可以为负。
	\end{description}

\end{trapbox}
