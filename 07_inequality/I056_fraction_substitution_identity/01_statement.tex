\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（实数非零）：}}]
		      变量 $x$ 为实数，且必须满足 $x\neq 0$，保证分母有意义。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平方恒等式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      \quad 令 $t=x+\frac{1}{x}$，则 $x^2+\frac{1}{x^2}$ 可用 $t$ 简单表达。
		      \[
			      x^2+\frac{1}{x^2} = t^2-2.
		      \]
		\item[\textbf{\textcolor{CStatement}{推广结论（值域范围）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      \quad 函数 $f(x)=x+\frac{1}{x}$ 在 $\mathbb{R}\setminus\{0\}$ 上的值域以 $\pm 2$ 为边界。
		      \[
			      \left|x+\frac{1}{x}\right|\ge 2,
			      \qquad
			      f(x)\in (-\infty,-2]\cup[2,+\infty).
		      \]
		      等号成立当且仅当 $x=\pm 1$。
	\end{description}

\end{statementbox}
