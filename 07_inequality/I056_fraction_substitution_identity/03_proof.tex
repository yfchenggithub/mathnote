\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	通过完全平方公式建立 $t$ 与 $x^2+\frac{1}{x^2}$ 的联系；再利用平方的非负性得到 $|t|$ 的下界。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（平方展开）：}]
		      \[
			      \begin{aligned}
				      t^2
				       & = \left(x+\frac{1}{x}\right)^2                   \\
				       & = x^2 + 2\cdot x\cdot\frac{1}{x} + \frac{1}{x^2} \\
				       & = x^2 + 2 + \frac{1}{x^2}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 利用完全平方公式展开，中间交叉项 $2\cdot x\cdot\frac{1}{x}=2$。
		\item[\textbf{步骤二（变形恒等）：}]
		      \[
			      x^2+\frac{1}{x^2} = t^2 - 2.
		      \]
		      \par\textbf{理由：} 将步骤一等式中的常数 $2$ 移到右边即得。
		\item[\textbf{步骤三（平方非负）：}]
		      \[
			      \begin{aligned}
				      \left(x-\frac{1}{x}\right)^2
				       & = x^2 - 2 + \frac{1}{x^2} \\
				       & \ge 0.
			      \end{aligned}
		      \]
		      因此
		      \[
			      x^2+\frac{1}{x^2}\ge 2.
		      \]
		      \par\textbf{理由：} 平方项 $\left(x-\frac{1}{x}\right)^2$ 恒非负，展开后移项得到平方和的下界。
		\item[\textbf{步骤四（代入取等）：}]
		      \[
			      \begin{aligned}
				      x^2+\frac{1}{x^2} & \ge 2, \\
				      t^2-2             & \ge 2, \\
				      t^2               & \ge 4, \\
				      |t|               & \ge 2.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 将步骤二的恒等式代入步骤三的不等式，得到 $t$ 的约束。当 $x=\pm 1$ 时，$\left(x-\frac{1}{x}\right)^2=0$，且 $t=\pm 2$，等号成立；反过来若等号成立，则 $x-\frac{1}{x}=0$，所以 $x^2=1$，即 $x=\pm 1$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{综上，若 $x\in\mathbb{R}$ 且 $x\neq 0$，令 $t=x+\frac{1}{x}$，则恒有 $x^2+\frac{1}{x^2}=t^2-2$，且 $|t|\ge 2$，等号成立当且仅当 $x=\pm 1$。}
\end{proofbox}
