\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{若 $x\in\mathbb{R}$ 且 $x\neq 0$，设 $t=x+\frac{1}{x}$，则 $x^2+\frac{1}{x^2}=t^2-2$，且 $|t|\ge 2$，等号成立当且仅当 $x=\pm1$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（分母非零）：} $x\neq 0$，否则 $1/x$ 无意义。
		\item \textbf{条件2（实数范围）：} $\left|x+\frac{1}{x}\right|\ge 2$ 是实数范围内的结论。
		\item \textbf{条件3（对称结构）：} 问题中同时或隐含地出现 $x+\frac{1}{x}$ 与 $x^2+\frac{1}{x^2}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（定义域检查）：} 应用恒等式前必须检查 $x\neq 0$，不可直接套用。
		\item \textbf{检查项（范围验证）：} 通过恒等式解出 $t$ 后，务必检验 $|t|\ge 2$，否则应舍去。
		\item \textbf{符号项（正负分类）：} $x>0$ 时 $x+\frac{1}{x}\ge 2$；$x<0$ 时 $x+\frac{1}{x}\le -2$。
	\end{itemize}

\end{summarybox}
