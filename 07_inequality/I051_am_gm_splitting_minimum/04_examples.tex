\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $y=x+\frac{1}{x}\,(x>0)$ 的最小值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别正项）：}]
			识别出两个正项 $x$ 与 $\frac{1}{x}$，它们的乘积 $x\cdot\frac{1}{x}=1$ 为常数。
			\par\textbf{理由：} $x>0$ 确保了各项为正；乘积为定值 $1$。
		\item[\textbf{第二步（套用均值不等式）：}]
			由均值不等式（$n=2$），
			\[
				x+\frac{1}{x} \ge 2\sqrt{x\cdot\frac{1}{x}}=2.
			\]
			\par\textbf{理由：} 直接将正项代入不等式即得下界 $2$。
		\item[\textbf{第三步（得最小值）：}]
			等号当 $x=\frac{1}{x}$ 即 $x=1$ 时成立，故最小值为 $2$。
			\par\textbf{理由：} 等号在定义域内可以取到，所以下界即为最小值。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y=x+\frac{1}{x}\,(x>0)$ 的最小值为 $2$，当 $x=1$ 时取到。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $y=2x+\frac{3}{x}\,(x>0)$ 的最小值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别正项）：}]
			两项 $2x$ 与 $\frac{3}{x}$ 均为正，乘积 $(2x)\cdot\frac{3}{x}=6$ 为常数。
			\par\textbf{理由：} $x>0$ 保证正性；乘积与 $x$ 无关。
		\item[\textbf{第二步（套用均值不等式）：}]
			由均值不等式，
			\[
				2x+\frac{3}{x} \ge 2\sqrt{(2x)\cdot\frac{3}{x}} = 2\sqrt{6}.
			\]
			\par\textbf{理由：} 代入公式即可。
		\item[\textbf{第三步（得最小值）：}]
			等号当 $2x=\frac{3}{x}$ 即 $x=\frac{\sqrt{6}}{2}$ 时成立，故最小值为 $2\sqrt{6}$。
			\par\textbf{理由：} 等号可满足。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y=2x+\frac{3}{x}\,(x>0)$ 的最小值为 $2\sqrt{6}$，当 $x=\frac{\sqrt{6}}{2}$ 时取到。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $y=x+\frac{4}{x-1}\,(x>1)$ 的最小值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（拆项变形）：}]
			将原式改写为
			\[
				y = (x-1)+\frac{4}{x-1}+1.
			\]
			\par\textbf{理由：} 通过 $x=(x-1)+1$ 拆项，使 $(x-1)$ 与 $\frac{4}{x-1}$ 乘积为常数。
		\item[\textbf{第二步（均值求部分和）：}]
			对 $(x-1)$ 和 $\frac{4}{x-1}$ 使用均值不等式：
			\[
				(x-1)+\frac{4}{x-1} \ge 2\sqrt{(x-1)\cdot\frac{4}{x-1}} = 4.
			\]
			\par\textbf{理由：} 两项乘积为定值 $4$。
		\item[\textbf{第三步（得最小值）：}]
			因此 $y \ge 4+1 = 5$，等号当 $x-1 = \frac{4}{x-1}$ 即 $x=3$ 时成立，最小值为 $5$。
			\par\textbf{理由：} 等号可取，最小值由部分和加常数得到。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y=x+\frac{4}{x-1}\,(x>1)$ 的最小值为 $5$，当 $x=3$ 时取到。}
\end{examplebox}
