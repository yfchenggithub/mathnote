	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（正项乘积常数）：}}]
			设由原表达式通过拆项、添项或配系数得到 $n$ 个正实数 $a_1,a_2,\dots,a_n$，且乘积 $\displaystyle\prod_{i=1}^n a_i$ 为常数 $C$（$C>0$，与变量无关）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（和最小公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正项乘积为常数时，和存在最小值，当各项相等时达到。
			\[
				\sum_{i=1}^{n} a_i \ge n\sqrt[n]{\prod_{i=1}^{n} a_i}
			\]
			当且仅当 $a_1=a_2=\cdots=a_n$ 时等号成立，此时左端取得最小值 $n\sqrt[n]{\prod_{i=1}^{n} a_i}$。
		\item[\textbf{\textcolor{CStatement}{推广结论（对偶均值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正项和为常数时，乘积有最大值，各项相等时达到。
			\[
				\prod_{i=1}^{n} b_i \le \left(\frac{\sum_{i=1}^{n} b_i}{n}\right)^n
			\]
			当且仅当 $b_1=b_2=\cdots=b_n$ 时等号成立，此时右端取得最大值。
	\end{description}

\end{statementbox}
