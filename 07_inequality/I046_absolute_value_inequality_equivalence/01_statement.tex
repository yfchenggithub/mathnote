\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（参数为正）：}}] 实数 $a>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（表达式为实）：}}] $f(x)$ 是实数表达式。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（小于情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			绝对值小于正数等价于表达式介于正负数之间。
			\[
				|f(x)|<a \iff -a<f(x)<a.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（大于情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			绝对值大于正数等价于表达式小于负数或大于正数。
			\[
				|f(x)|>a \iff f(x)<-a \;\lor\; f(x)>a.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{常见变形（含等号情形）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{变形一（小于等于）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			包括等号时，边界同样包含。
			\[
				|f(x)|\leq a \iff -a\leq f(x)\leq a.
			\]
		\item[\textbf{\textcolor{CStatement}{变形二（大于等于）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			包括等号时，边界同样包含。
			\[
				|f(x)|\geq a \iff f(x)\leq -a \;\lor\; f(x)\geq a.
			\]
	\end{description}
\end{statementbox}
