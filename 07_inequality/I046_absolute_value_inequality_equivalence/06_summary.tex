\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	绝对值不等式 $|f(x)|<a$（$a>0$）等价于 $-a<f(x)<a$；$|f(x)|>a$ 等价于 $f(x)<-a$ 或 $f(x)>a$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（参数为正）：} $a>0$。
		\item \textbf{条件2（表达式为实）：} $f(x)$ 为实数表达式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（方向混淆）：} 小于取中间（且），大于取两边（或）。
		\item \textbf{检查项（前提条件）：} 使用前确认 $a>0$；若含等号，对应带上等号。
	\end{itemize}
\end{summarybox}
