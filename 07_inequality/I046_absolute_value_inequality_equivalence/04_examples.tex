\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $|2x-1|<3$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用结论）：}] 由 $a=3>0$，根据定理：
			\[
				-3<2x-1<3.
			\]
			\par\textbf{理由：} $|f(x)|<a\iff -a<f(x)<a$，这里 $f(x)=2x-1$。

		\item[\textbf{第二步（分别求解）：}] 解两个不等式：
			\[
				\begin{cases}
					2x-1>-3\;\Rightarrow\; x>-1,\\
					2x-1<3\;\Rightarrow\; x<2.
			\end{cases}
			\]
			\par\textbf{理由：} 不等式的同解变形。

		\item[\textbf{第三步（取交集）：}] 两个条件同时成立，解集为
			\[
				-1<x<2.
			\]
			\par\textbf{理由：} 逻辑连接词为“且”，故取公共部分。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	原不等式的解集为 $(-1,2)$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $|x^2-1|>2$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用结论）：}] 由 $a=2>0$，根据定理：
			\[
				x^2-1<-2 \quad\lor\quad x^2-1>2.
			\]
			\par\textbf{理由：} $|f(x)|>a\iff f(x)<-a \;\lor\; f(x)>a$。

		\item[\textbf{第二步（解二次不等式）：}]
			\[
				x^2-1<-2
			\]
			无解；
			\[
				x^2-1>2 \iff x^2>3.
			\]
			\par\textbf{理由：} $x^2-1<-2$ 即 $x^2<-1$，不可能；$x^2>3$ 解得 $x<-\sqrt{3}$ 或 $x>\sqrt{3}$。

		\item[\textbf{第三步（取并集）：}] 两个分支为或关系，故解集为
			\[
				x<-\sqrt{3} \;\lor\; x>\sqrt{3}.
			\]
			\par\textbf{理由：} 逻辑连接词为“或”，取所有满足条件的并集。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	原不等式的解集为 $(-\infty,-\sqrt{3})\cup(\sqrt{3},+\infty)$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知关于 $x$ 的不等式 $|2x+1|>m$（$m>0$）的解集为 $x<-2$ 或 $x>1$，求 $m$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用结论）：}] 由 $m>0$，根据定理：
			\[
				2x+1<-m \quad\lor\quad 2x+1>m.
			\]
			\par\textbf{理由：} $|f(x)|>m\iff f(x)<-m \;\lor\; f(x)>m$。

		\item[\textbf{第二步（解出形式）：}] 解两个线性不等式：
			\[
				2x+1<-m \;\Rightarrow\; x<\frac{-m-1}{2},
			\]
			\[
				2x+1>m \;\Rightarrow\; x>\frac{m-1}{2}.
			\]
			\par\textbf{理由：} 移项、除以正数 $2$，不等号方向不变。

		\item[\textbf{第三步（对比解集）：}] 已知解集为 $x<-2$ 或 $x>1$，因此
			\[
				\frac{-m-1}{2}=-2,\quad \frac{m-1}{2}=1.
			\]
			解得 $m=3$。
			\par\textbf{理由：} 解集的分界点对应不等式边界，联立方程求解。验证 $m=3>0$ 成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$m=3$。
\end{examplebox}
