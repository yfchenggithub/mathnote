\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（实数命题）：}}] 设 $a,b\in\mathbb{R}$，$P,Q$ 为数学命题。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（比较法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过作差符号直接比较实数大小。
			\[
				a>b \iff a-b>0,\quad a=b \iff a-b=0,\quad a<b \iff a-b<0.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（综合法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从已知条件正向推导出结论。
			\[
				P \Rightarrow Q_1 \Rightarrow Q_2 \Rightarrow \cdots \Rightarrow Q.
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（分析法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从结论反向追溯至已知条件。
			\[
				Q \Leftarrow P_1 \Leftarrow P_2 \Leftarrow \cdots \Leftarrow P.
			\]

		\item[\textbf{\textcolor{CStatement}{结论四（反证法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			假设结论否定与前提共存推出矛盾。
			\[
				(\neg Q \land P) \Rightarrow \bot \Longrightarrow P \Rightarrow Q.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{推广结论（逆否等价）：}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	反证法的逻辑基础是逆否命题与原命题等价。
	\[
		(P \Rightarrow Q) \iff (\neg Q \Rightarrow \neg P).
	\]

\end{statementbox}
