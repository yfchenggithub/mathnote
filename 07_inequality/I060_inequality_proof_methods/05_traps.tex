\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（比较法判号错误）}}] 误以为 $a>b$ 等价于 $a^2-b^2>0$ 或 $a-b$ 可以取绝对值。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须作差不转化）：}}] 比较法必须直接作差 $a-b$，判断其与0的关系，其他变形需谨慎验证等价性。
		\item[\textbf{\textcolor{CTrap}{错因分析（等价性忽略）：}}] 忽略了“作差比较”的定义基础，使用了不等价变换。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分析法方向颠倒）}}] 写作“要证 $Q$，只需证 $P$，因为 $P\Rightarrow Q$”，混淆了分析法和综合法方向。
		\item[\textbf{\textcolor{CTrap}{正确理解（执果索因逻辑）：}}] 分析法应写为“要证 $Q$，只需证 $P$，即 $Q\Leftarrow P$”，表示 $P$ 是 $Q$ 的充分条件。
		\item[\textbf{\textcolor{CTrap}{错因分析（方向逻辑混乱）：}}] 未理解分析法是逆推（执果索因），习惯性使用了综合法的正向语言。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（反证法假设不周全）}}] 假设 $a,b$ 都大于1来否定“至少有一个大于1”是不对的。
		\item[\textbf{\textcolor{CTrap}{正确理解（完全否定结论）：}}] 必须假设结论的反面全部情形。例如“至少有一个大于1”的反面是“所有数都不大于1”，即“$a\le1$ 且 $b\le1$”。
		\item[\textbf{\textcolor{CTrap}{错因分析（否定命题不完整）：}}] 对结论的否定把握不准，遗漏部分情况，导致论证不充分。
	\end{description}
\end{trapbox}
