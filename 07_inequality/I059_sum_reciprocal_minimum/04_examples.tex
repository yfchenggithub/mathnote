\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0,b>0$，且 $a+b=2$，求 $\frac{1}{a}+\frac{1}{b}$ 的最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（调用结论）：}]
			\[
				(a+b)\left(\frac{1}{a}+\frac{1}{b}\right)\ge 4.
			\]
			\par\textbf{理由：} 对正实数 $a,b$ 该不等式恒成立（已证结论）。
		\item[\textbf{第二步（代入条件）：}]
			\[
				2\left(\frac{1}{a}+\frac{1}{b}\right)\ge 4.
			\]
			\par\textbf{理由：} 将已知 $a+b=2$ 代入结论。
		\item[\textbf{第三步（求解最值）：}]
			\[
				\frac{1}{a}+\frac{1}{b}\ge 2.
			\]
			\par\textbf{理由：} 两边同除以 $2$ 即得。等号成立当 $a=b$，结合 $a+b=2$ 得 $a=b=1$，此时取最小值 $2$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为 $2$，当 $a=b=1$ 时取得。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0,b>0$，且 $a+3b=6$，求 $\frac{1}{a}+\frac{1}{3b}$ 的最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变量代换）：}]
			令 $x=a$，$y=3b$，则 $x>0,y>0$，$x+y=6$，且
			\[
				\frac{1}{a}+\frac{1}{3b} = \frac{1}{x}+\frac{1}{y}.
			\]
			\par\textbf{理由：} 代换使系数变为 $1$，便于应用结论。
		\item[\textbf{第二步（应用结论）：}]
			由结论 $(x+y)\left(\frac{1}{x}+\frac{1}{y}\right)\ge 4$ 及 $x+y=6$ 得
			\[
				6\left(\frac{1}{x}+\frac{1}{y}\right)\ge 4.
			\]
			\par\textbf{理由：} 将 $x+y=6$ 代入结论得到。
		\item[\textbf{第三步（回代求解）：}]
			\[
				\begin{aligned}
					\frac{1}{a}+\frac{1}{3b} &= \frac{1}{x}+\frac{1}{y} \\ &\ge \frac{2}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 等号当 $x=y=3$，即 $a=3,b=1$ 时成立，故最小值为 $\frac{2}{3}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为 $\frac{2}{3}$，当 $a=3,b=1$ 时取得。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $a>0,b>0$，且 $a+b=1$，求 $\left(1+\frac{1}{a}\right)\left(1+\frac{1}{b}\right)$ 的最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（展开乘积）：}]
			\[
				\left(1+\frac{1}{a}\right)\left(1+\frac{1}{b}\right)=1+\frac{1}{a}+\frac{1}{b}+\frac{1}{ab}.
			\]
			\par\textbf{理由：} 多项式乘法展开。
		\item[\textbf{第二步（求 $\frac{1}{a}+\frac{1}{b}$ 下界）：}]
			由 $a+b=1$ 及结论 $(a+b)\left(\frac{1}{a}+\frac{1}{b}\right)\ge 4$ 得
			\[
				\frac{1}{a}+\frac{1}{b}\ge 4.
			\]
			\par\textbf{理由：} 代入 $a+b=1$ 即得，等号当 $a=b$。
		\item[\textbf{第三步（求 $\frac{1}{ab}$ 下界）：}]
			由基本不等式，$ab\le \left(\frac{a+b}{2}\right)^2$。由 $a+b=1$ 得 $\left(\frac{a+b}{2}\right)^2=\frac{1}{4}$，因此 $ab\le\frac{1}{4}$，故
			\[
				\frac{1}{ab}\ge 4.
			\]
			\par\textbf{理由：} 积的最大值对应倒数最小值，等号当 $a=b$。
		\item[\textbf{第四步（合并最小值）：}]
			\[
				\begin{aligned}
					\left(1+\frac{1}{a}\right)\left(1+\frac{1}{b}\right) &\ge 1+4+4 \\ &= 9.
			\end{aligned}
			\]
			\par\textbf{理由：} 叠加不等式，等号均在 $a=b=\frac12$ 时取到，最小值 $9$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为 $9$，当 $a=b=\frac12$ 时取得。}
\end{examplebox}
