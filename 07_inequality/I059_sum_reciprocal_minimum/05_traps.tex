\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视正数条件）}}]
			部分学生不考虑 $a,b$ 的正负，直接套用 $(a+b)(\frac{1}{a}+\frac{1}{b})\ge4$，导致错误结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（正解要点）：}}]
			该不等式仅当 $a,b>0$ 时才成立；若 $a,b$ 为负，不等号方向可能改变或表达式无意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（错误根源）：}}]
			对均值不等式的前提条件“正数”记忆不牢，忽略定义域。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（遗漏等号条件）}}]
			学生只写 $(a+b)(\frac{1}{a}+\frac{1}{b})\ge4$，不说明等号何时成立。
		\item[\textbf{\textcolor{CTrap}{正确理解（正解要点）：}}]
			必须附上等号成立条件 $a=b$；在求解最值问题时，需要验证等号可以取到。
		\item[\textbf{\textcolor{CTrap}{错因分析（错误根源）：}}]
			忽略等号条件导致最值判断不严谨，可能误以为恒成立而得到错误最值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（形式套用错误）}}]
			对于 $(2a+b)(\frac{1}{a}+\frac{1}{b})$ 这类系数不匹配的式子，盲目认为最小值也是 $4$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正解要点）：}}]
			当系数不匹配时，不能直接套用原不等式；应展开后利用基本不等式求最值。
		\item[\textbf{\textcolor{CTrap}{错因分析（错误根源）：}}]
			对公式的结构特征把握不牢，忽略系数差异导致错误套用。
	\end{description}

\end{trapbox}
