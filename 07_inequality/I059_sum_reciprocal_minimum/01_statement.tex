\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正实数）：}}]
			$a,b > 0$，且 $a,b$ 均为实数。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最小值结论）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两正数的和与其倒数和之积的最小值为 $4$。
			\[
				(a+b)\left(\frac{1}{a}+\frac{1}{b}\right) \ge 4.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（恒等变形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过配方可将原式化为 $4$ 加上非负项的形式。
			\[
				\begin{aligned}
					(a+b)\left(\frac{1}{a}+\frac{1}{b}\right) &= 4 + \frac{(a-b)^2}{ab} \\ &\ge 4.
			\end{aligned}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	当且仅当 $a=b$ 时取等。
\end{statementbox}
