\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（可化标准形式）：}}] 不等式必须可以写成 $f(x) = a\prod_{i=1}^{n}(x-x_i)^{m_i} > 0$（或 $<0$），其中 $a > 0$，各零点 $x_1 < x_2 < \dots < x_n$，指数 $m_i \in \mathbb{N}^*$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（奇穿偶不穿规则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从数轴右上方起笔，遇奇次根穿过数轴，遇偶次根不穿过；曲线在 $x$ 轴上方的区间 $f(x)>0$，在下方则 $f(x)<0$，严格不等式解集为对应的开区间并集。
			\[
				f(x) = a\prod_{i=1}^{n}(x-x_i)^{m_i},\ a>0
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（非严格处理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对于 $\geq 0$ 或 $\leq 0$，解集是在严格解集的基础上并上所有根点。
			\[
				\{x\mid f(x)\geq 0\} = \{x\mid f(x)>0\} \cup \{x_1,x_2,\dots,x_n\}
			\]
	\end{description}

\end{statementbox}
