\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{穿根法：从右上方起笔，奇次根穿过，偶次根不穿，符号区间对应解集。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准形式）：} 不等式已化为 $f(x)=a\prod (x-x_i)^{m_i}>0$（或 $\ge0$ 等），$a>0$，且各根从小到大排列。
		\item \textbf{条件 2（因式分解）：} 多项式必须分解为一次因式幂乘积的形式，不可出现二次及以上无法分解的因式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（先判$a$符号）：} $a<0$ 时需先乘以 $-1$ 变号再穿根。
		\item \textbf{检查项（重根奇偶）：} 准确区分每个根的重数，偶次根不穿过。
		\item \textbf{检查项（严格与非严格）：} 严格不等式取开区间，非严格需并上根点。
	\end{itemize}
\end{summarybox}
