\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $(x-1)(x-2)(x-3)>0$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化标准形式）：}] 不等式已为因式乘积形式，$a=1>0$，根$1,2,3$，指数均为1（奇次）。
			\par\textbf{理由：} 标准形式要求$a>0$，满足。

		\item[\textbf{第二步（标根画线）：}] 在数轴上从小到大标出$1,2,3$。从右上方起笔画线，依次经过$3,2,1$，因都是奇次，每次画线均穿过数轴。
			\par\textbf{理由：} 奇穿偶不穿，奇次根穿过。

		\item[\textbf{第三步（读区间写解集）：}] 曲线在$x$轴上方的区间为$(1,2)\cup(3,+\infty)$，故原不等式解集为$(1,2)\cup(3,+\infty)$。
			\par\textbf{理由：} 曲线上方对应$f(x)>0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{解集为$(1,2)\cup(3,+\infty)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $(x+1)(x-1)^2(x-3)>0$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别根与重数）：}] 根为$x_1=-1$（奇次，$m=1$），$x_2=1$（偶次，$m=2$），$x_3=3$（奇次，$m=1$）。$a=1>0$。
			\par\textbf{理由：} 将各因式指数标出，判断奇偶。

		\item[\textbf{第二步（画线）：}] 从数轴右上方起笔，遇$3$穿过（奇次），遇$1$不穿过（偶次，曲线保持在同一侧），遇$-1$穿过（奇次）。完成曲线。
			\par\textbf{理由：} 奇穿偶不穿规则。

		\item[\textbf{第三步（确定解集）：}] 曲线位于轴上方的区间为$(-\infty,-1)$和$(3,+\infty)$（注意偶次根两侧同号，故$(-1,1)$和$(1,3)$均在轴下方）。因此$f(x)>0$的解集为$(-\infty,-1)\cup(3,+\infty)$。
			\par\textbf{理由：} 曲线上方对应$f(x)>0$，且偶次根不改变符号。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{解集为$(-\infty,-1)\cup(3,+\infty)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $\dfrac{(x-1)(x-2)}{x-3}\geq 0$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（转化为整式不等式）：}] 原不等式等价于$(x-1)(x-2)(x-3)\geq 0$且$x\neq 3$（分母不为零）。
			\par\textbf{理由：} 两边同乘$(x-3)^2>0$，不等号方向不变，得到乘积形式；但$x=3$使原分母为零必须剔除。

		\item[\textbf{第二步（穿根法求解整式不等式）：}] 考虑$(x-1)(x-2)(x-3)\geq 0$，$a=1>0$，根$1,2,3$（均为奇次）。画线从右上方起，依次穿过$3,2,1$。曲线上方区间为$(1,2)\cup(3,+\infty)$，曲线在根点值为$0$。因此整式不等式解集为$[1,2]\cup[3,+\infty)$（因$\geq0$包含根点）。
			\par\textbf{理由：} 奇穿偶不穿规则确定符号区间，非严格需并入根点。

		\item[\textbf{第三步（剔除分母根）：}] 原分式中$x=3$无定义，故在整式解集中去掉$3$即得原不等式解集为$[1,2]\cup(3,+\infty)$。
			\par\textbf{理由：} 分母不为零的限制。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{原不等式解集为$[1,2]\cup(3,+\infty)$。}
\end{examplebox}
