\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	约束条件
	\[
		\begin{cases}
			x\geq 0,\\ y\geq 0,\\ x+2y\leq 4,\\ x-y\leq 1,
		\end{cases}
	\]
	目标函数 $z=2x+y$，求 $z$ 的最大值和最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求顶点）：}]
			求可行域所有顶点。解边界方程得：
			\[
				(0,0), (1,0), (2,1), (0,2)
			\]
			\par\textbf{理由：} 约束是直线，顶点是边界直线两两交点且满足全部约束。
		\item[\textbf{第二步（代入计算）：}]
			分别将顶点坐标代入 $z=2x+y$：
			\[
				(0,0): 0,\quad (1,0): 2,\quad (2,1): 5,\quad (0,2): 2
			\]
			\par\textbf{理由：} 直接计算。
		\item[\textbf{第三步（比较最值）：}]
			最大值 $\max z = 5$，在 $(2,1)$ 取得；最小值 $\min z = 0$，在 $(0,0)$ 取得。
			\par\textbf{理由：} 比较四个数值。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\max z =5$, $\min z =0$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	约束条件同例1，目标函数 $z=2x-3y$，求最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求顶点）：}]
			同例1，顶点为 $(0,0),(1,0),(2,1),(0,2)$。
			\par\textbf{理由：} 约束相同。
		\item[\textbf{第二步（代入计算）：}]
			代入 $z=2x-3y$：
			\[
				(0,0): 0,\quad (1,0): 2,\quad (2,1): 1,\quad (0,2): -6
			\]
			\par\textbf{理由：} 直接计算。
		\item[\textbf{第三步（比较最值）：}]
			最小值 $\min z = -6$，在 $(0,2)$ 取得（最大值 $2$ 在 $(1,0)$ 取得，但此题只问最小值）。
			\par\textbf{理由：} 四个值中最小的为 $-6$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\min z = -6$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	约束条件
	\[
		\begin{cases}
			x\geq 0,\\ y\geq 0,\\ x+y\leq 4,\\ 2x+y\leq 5,
		\end{cases}
	\]
	目标函数 $z=ax+y$（$a$ 为参数），已知 $z$ 的最大值为 $5$，求 $a$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求顶点）：}]
			解边界直线交点并验证约束得顶点：
			\[
				A(0,0), B(0,4), C(1,3), D(2.5,0)
			\]
			\par\textbf{理由：} 可行域是四边形，顶点由 $x=0$,$y=0$,$x+y=4$,$2x+y=5$ 两两相交且满足全部约束的点。
		\item[\textbf{第二步（顶点函数值）：}]
			代入 $z=ax+y$：
			\[
				z_A=0,\quad z_B=4,\quad z_C=a+3,\quad z_D=2.5a
			\]
			\par\textbf{理由：} 直接代入。
		\item[\textbf{第三步（利用最值条件）：}]
			由顶点最值结论，最大值必在顶点取到，故 $\max\{0,4,a+3,2.5a\}=5$。必须同时满足：
			\[
				a+3\leq 5,\quad 2.5a\leq 5,
			\]
			且至少其中一个等于 $5$。解得 $a=2$（当 $a=2$ 时 $a+3=5$, $2.5a=5$）。
			\par\textbf{理由：} 最大值 $5$ 要求所有顶点值不超过 $5$ 且至少一个达到 $5$。
		\item[\textbf{第四步（验证）：}]
			代入 $a=2$，得 $z_C=5$, $z_D=5$, $z_B=4$, $z_A=0$，最大值为 $5$，符合。
			\par\textbf{理由：} 验证确认无误。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=2$}。
\end{examplebox}
