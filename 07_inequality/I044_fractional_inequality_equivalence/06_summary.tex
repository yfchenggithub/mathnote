\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{分式不等式 $\frac{P(x)}{Q(x)}>0$（$\ge 0$）与整式乘积不等式 $P(x)Q(x)>0$（$\ge 0$）等价，并附加分母不为零。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（分母非零）：} $Q(x)\neq 0$，转化时需标注排除分母零点。
		\item \textbf{条件 2（符号一致）：} 转化后整式乘积的符号与分式符号一致。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略分母）：} 转化后容易忘记加上分母不为零的条件，导致解集包含分母零点。
		\item \textbf{检查项（等号处理）：} $\ge 0$ 或 $\le 0$ 时等号成立需分子为零且分母非零。
	\end{itemize}

\end{summarybox}
