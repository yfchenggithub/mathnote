\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	利用实数乘除符号法则：商的正负由分子分母同号或异号决定。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（符号法则）：}]
			回顾乘法符号法则：
			\[
				a,b\neq 0,\quad ab>0 \iff a,b \text{ 同号},\quad ab<0 \iff a,b \text{ 异号}.
			\]
			\par\textbf{理由：} 同号得正，异号得负，这是处理分式不等式的基础。
		\item[\textbf{步骤二（正商转化）：}]
			证明 $\displaystyle\frac{P(x)}{Q(x)}>0\iff P(x)Q(x)>0$。
			\[
				\frac{P(x)}{Q(x)}>0 \iff P(x),Q(x)\text{ 同号}\iff P(x)Q(x)>0.
			\]
			\par\textbf{理由：} 因为 $Q(x)\neq 0$，商的正负完全由分子分母符号决定。
		\item[\textbf{步骤三（非负转化）：}]
			证明 $\displaystyle\frac{P(x)}{Q(x)}\ge 0\iff
			\begin{cases}
				P(x)Q(x)\ge 0,\\ Q(x)\neq 0.
		\end{cases}
			$
			\[
				\begin{aligned}
					&\frac{P(x)}{Q(x)}\ge 0 \\ &\iff \frac{P(x)}{Q(x)}>0 \;\text{或}\; \frac{P(x)}{Q(x)}=0 \\ &\iff (P(x)Q(x)>0) \;\text{或}\; (P(x)=0,\;Q(x)\neq 0) \\ &\iff
					\begin{cases}
						P(x)Q(x)\ge 0,\\ Q(x)\neq 0.
				\end{cases}

			\end{aligned}
			\]
			\par\textbf{理由：} 等号情形需要分子为零且分母不为零，此时积为零，满足非负。合并得证。
		\item[\textbf{步骤四（总结转化）：}]
			综上，得到等价关系：
			\[
				\frac{P(x)}{Q(x)}>0 \iff P(x)Q(x)>0,
			\]
			\[
				\frac{P(x)}{Q(x)}\ge 0 \iff
				\begin{cases}
					P(x)Q(x)\ge 0,\\ Q(x)\neq 0.
			\end{cases}
			\]
			\par\textbf{理由：} 这两组转化是解分式不等式的核心，后续只需对方程组或不等式组求解。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{分式不等式化归为整式不等式，转化后可用因式分解与数轴标根求解。}
\end{proofbox}
