\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $\frac{x+1}{x-2}>0$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等价转化）：}]
			原不等式等价于 $(x+1)(x-2)>0$。
			\par\textbf{理由：} 由分式不等式转化定理，$\frac{x+1}{x-2}>0 \iff (x+1)(x-2)>0$。
		\item[\textbf{第二步（因式分解）：}]
			本题已为因式形式，无需再分解。
			\par\textbf{理由：} 多项式已为一次因式乘积。
		\item[\textbf{第三步（穿根定号）：}]
			在数轴上标根 $-1$ 和 $2$，从右上方向下穿，得到 $x<-1$ 或 $x>2$。
			\par\textbf{理由：} 数轴标根法，奇穿偶不穿，取正区间。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{解集为 $(-\infty,-1)\cup(2,+\infty)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $\frac{x^2-1}{x+4}\ge 0$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等价转化）：}]
			原不等式等价于 $(x^2-1)(x+4)\ge 0$ 且 $x\neq -4$。
			\par\textbf{理由：} 由分式不等式转化定理，$\frac{x^2-1}{x+4}\ge 0 \iff
			\begin{cases}
				(x^2-1)(x+4)\ge 0,\\ x\neq -4.
		\end{cases}
			$
		\item[\textbf{第二步（因式分解）：}]
			将 $x^2-1$ 分解，得 $(x-1)(x+1)(x+4)\ge 0$。
			\par\textbf{理由：} 平方差公式分解因式。
		\item[\textbf{第三步（穿根定号）：}]
			在数轴上标根 $-4,-1,1$，穿根得 $[ -4,-1]\cup[1,+\infty)$，但需排除 $x=-4$，故解集为 $(-4,-1]\cup[1,+\infty)$。
			\par\textbf{理由：} 数轴标根取非负区间，并除去分母零点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{解集为 $(-4,-1]\cup[1,+\infty)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $\frac{x^2-2x-3}{x-2}\le 0$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等价转化）：}]
			原不等式等价于 $(x^2-2x-3)(x-2)\le 0$ 且 $x\neq 2$。
			\par\textbf{理由：} 由分式不等式转化定理，$\frac{x^2-2x-3}{x-2}\le 0 \iff
			\begin{cases}
				(x^2-2x-3)(x-2)\le 0,\\ x\neq 2.
		\end{cases}
			$
		\item[\textbf{第二步（因式分解）：}]
			将 $x^2-2x-3$ 分解，得 $(x-3)(x+1)(x-2)\le 0$。
			\par\textbf{理由：} 十字相乘法分解二次三项式。
		\item[\textbf{第三步（穿根定号）：}]
			在数轴上标根 $-1,2,3$，穿根得 $(-\infty,-1]\cup(2,3]$，其中 $x=2$ 被排除。
			\par\textbf{理由：} 数轴标根取负或零区间，并除去分母零点。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{解集为 $(-\infty,-1]\cup(2,3]$。}
\end{examplebox}
