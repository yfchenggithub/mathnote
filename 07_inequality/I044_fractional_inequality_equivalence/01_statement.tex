\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（多项式非零）：}}]
			设 $P(x), Q(x)$ 为多项式，且 $Q(x) \not\equiv 0$（即分母不恒为零）。求解 $x$ 时要求 $Q(x) \neq 0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（大于0转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			分式大于零等价于分子分母乘积大于零。
			\[
				\frac{P(x)}{Q(x)} > 0 \iff P(x)Q(x) > 0.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（非负转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			分式非负等价于分子分母乘积非负，且分母不为零。
			\[
				\frac{P(x)}{Q(x)} \ge 0 \iff
				\begin{cases}
					P(x)Q(x) \ge 0, \\ Q(x) \neq 0.
			\end{cases}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（不等号推广）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对于分式小于零或非正，同样可转化为乘积符号判断。
			\[
				\frac{P(x)}{Q(x)} < 0 \iff P(x)Q(x) < 0.
			\]
			\[
				\frac{P(x)}{Q(x)} \le 0 \iff
				\begin{cases}
					P(x)Q(x) \le 0, \\ Q(x) \neq 0.
			\end{cases}
			\]
	\end{description}

\end{statementbox}
