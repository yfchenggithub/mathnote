\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略分母为零）}}]
			在解 $\frac{P(x)}{Q(x)}\ge 0$（或 $\le 0$）时，只写 $P(x)Q(x)\ge 0$（或 $\le 0$）而忘记剔除分母为零的点。
		\item[\textbf{\textcolor{CTrap}{正确理解（附加条件）：}}]
			转化后的整式不等式必须补充分母不为零的条件，即 $
			\begin{cases}
				P(x)Q(x)\ge 0,\\ Q(x)\neq 0.
		\end{cases}
			$（对于 $>0$ 或 $<0$，乘积不等号自动排除分母零点，但仍可明确写出条件）。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义域残缺）：}}]
			遗漏分母零点会导致解集多出使分母无意义的点，破坏等价性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（随意约分）}}]
			在分式不等式中，将分子分母的公因式直接约去，例如 $\frac{x-1}{(x-1)(x+2)}>0$ 约去 $x-1$，导致定义域改变。
		\item[\textbf{\textcolor{CTrap}{正确理解（保持原式）：}}]
			分式不等式不能约分；必须保持原分母的零点限制，等价转化时应直接使用原分子分母乘积 $(x-1)(x-1)(x+2)>0$ 并附加 $x\neq 1, x\neq -2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（等价性破坏）：}}]
			约分后原分母零点 $x=1$ 不再被排除，可能被错误地包含在解集中，破坏了等价性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（等号误判）}}]
			在解 $\ge 0$ 或 $\le 0$ 的不等式时，误认为等号包括分母为零的情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（等号条件）：}}]
			等号只能在分子为零且分母不为零时取得，即 $P(x)=0$ 且 $Q(x)\neq 0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（包含无效点）：}}]
			分母为零的点不满足不等式，应排除。
	\end{description}

\end{trapbox}
