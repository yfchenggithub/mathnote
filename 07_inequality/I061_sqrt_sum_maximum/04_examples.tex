\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 设 $x>0,y>0$，且 $x+y=4$，求 $\sqrt{x}+\sqrt{y}$ 的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件识别）：}]
		      由已知 $x+y=4$ 为定值，且 $x,y>0$，满足结论条件。
		      \par\textbf{理由：} 结论要求 $u,v$ 为非负实数且 $u+v$ 为定值。

		\item[\textbf{第二步（套用公式）：}]
		      由结论 $\sqrt{x}+\sqrt{y}\le\sqrt{2(x+y)}$，代入 $x+y=4$，计算右边得 $\sqrt{2\times4}=2\sqrt{2}$，故 $\sqrt{x}+\sqrt{y}\le 2\sqrt{2}$。
		      \par\textbf{理由：} 直接代入不等式可得上界。

		\item[\textbf{第三步（取等答案）：}]
		      等号成立当 $x=y$，由 $x+y=4$ 得 $x=y=2$，满足 $x>0,y>0$，此时 $\sqrt{x}+\sqrt{y}$ 取得最大值 $2\sqrt{2}$。
		      \par\textbf{理由：} 取等条件 $u=v$，且取等点在题目可行域内。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\max(\sqrt{x}+\sqrt{y})=2\sqrt{2}$，此时 $x=y=2$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知正数 $a,b$ 满足 $2a+b=8$，求 $\sqrt{2a}+\sqrt{b}$ 的最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变量代换）：}]
		      令 $u=2a$，$v=b$，则 $u+v=2a+b=8$，且 $u,v>0$。
		      \par\textbf{理由：} 转化为标准形式 $\sqrt{u}+\sqrt{v}$。

		\item[\textbf{第二步（套用结论）：}]
		      由结论 $\sqrt{u}+\sqrt{v}\le\sqrt{2(u+v)}$，代入 $u+v=8$，计算右边得 $\sqrt{2\times8}=4$，故 $\sqrt{u}+\sqrt{v}\le 4$。
		      \par\textbf{理由：} 结论给出上界。

		\item[\textbf{第三步（回代答案）：}]
		      所以 $\sqrt{2a}+\sqrt{b}\le 4$。当 $u=v$，即 $2a=b$ 时取等；结合 $2a+b=8$，解得 $2a=4$，$b=4$，即 $a=2$，$b=4$，最大值为 $4$。
		      \par\textbf{理由：} 取等条件 $u=v$，且取等点满足 $a,b>0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\max(\sqrt{2a}+\sqrt{b})=4$，此时 $a=2$，$b=4$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 求函数 $f(x)=\sqrt{2x}+\sqrt{1-2x}$ 的最大值，并指出此时的 $x$ 值。

	\textbf{\textcolor{CExample}{解题步骤：}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定定义域）：}]
		      由 $2x\ge0$ 且 $1-2x\ge0$ 得 $0\le x\le\frac12$，所以 $2x$ 与 $1-2x$ 均为非负实数。
		      \par\textbf{理由：} 根号内的数必须非负。

		\item[\textbf{第二步（构造和定）：}]
		      令 $u=2x$，$v=1-2x$，则 $u+v=1$，且 $u,v\ge0$，原函数 $f(x)=\sqrt{u}+\sqrt{v}$。
		      \par\textbf{理由：} 转化为标准形式 $u+v$ 为定值。

		\item[\textbf{第三步（套用求最值）：}]
		      由结论 $\sqrt{u}+\sqrt{v}\le\sqrt{2(u+v)}$，代入 $u+v=1$，计算右边得 $\sqrt{2\times1}=\sqrt{2}$，所以 $f(x)\le\sqrt{2}$。
		      \par\textbf{理由：} 结论直接给出上界。

		\item[\textbf{第四步（取等求解）：}]
		      等号成立当 $u=v$，即 $2x=1-2x$，解得 $x=\frac14$，在定义域内。此时 $f(x)_{\max}=\sqrt{2}$。
		      \par\textbf{理由：} 取等条件 $u=v$，且取等点在定义域内。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$f(x)_{\max}=\sqrt{2}$，此时 $x=\frac14$。}
\end{examplebox}
