\begin{statementbox}
	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（非负实数）：}}]
			$u,v$ 均为非负实数。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（根式和不等式）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个非负数的算术平方根之和不超过它们和的算术平方根的两倍。\n
			\[
				\n \sqrt{u}+\sqrt{v}\le\sqrt{2(u+v)}.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（定值最值）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $u+v$ 为定值时，$\sqrt{u}+\sqrt{v}$ 的最大值可由该定值直接算出。\n
			\[
				\n \max(\sqrt{u}+\sqrt{v})=\sqrt{2(u+v)}.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
