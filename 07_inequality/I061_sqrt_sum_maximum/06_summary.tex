\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{非负实数 $u,v$ 满足 $\sqrt{u}+\sqrt{v} \le \sqrt{2(u+v)}$；当 $u+v=S$ 为定值时，最大值为 $\sqrt{2S}$，且在 $u=v=\dfrac{S}{2}$ 时取到。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（非负实数）：} $u$ 和 $v$ 必须均为非负数，否则根号在实数范围内无意义。
		\item \textbf{条件 2（和为定值）：} 若要使用最值结论，需 $u+v$ 为定值；不等式本身恒成立，但“最大值公式”依赖和为定值。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略非负）：} 直接套用结论而未检查 $u,v$ 是否为非负。
		\item \textbf{检查项（取等验证）：} 在求最值时必须确认 $u=v$ 是否在可行域内，否则实际最大值可能小于理论上界。
	\end{itemize}
\end{summarybox}
