\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{非负实数 $u,v$ 满足 $\sqrt{u}+\sqrt{v} \le \sqrt{2(u+v)}$；当 $u+v$ 为定值时，等号成立即得到最大值。}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（非负实数）：} $u$ 和 $v$ 必须均为非负数，否则根号无意义或不等式不成立。\n
		\item \textbf{条件 2（和为定值）：} 若要使用最值结论，需 $u+v$ 为定值；不等式本身恒成立，但最值仅在和为定值时才有意义。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（忽略非负）：} 直接套用结论而未检查 $u,v$ 是否为非负。\n
		\item \textbf{检查项（取等验证）：} 在求最值时必须确认 $u=v$ 是否在可行域内，否则实际最大值可能小于理论值。\n
	\end{itemize}
	\n
\end{summarybox}
