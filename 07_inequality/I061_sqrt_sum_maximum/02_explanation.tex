\begin{explanationbox}
	\n\textbf{\textcolor{CExplanation}{一句话直觉}}\n\n\textbf{$\sqrt{u}+\sqrt{v}$ 的最大值不超过 $\sqrt{2(u+v)}$，当 $u=v$ 时取到。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{核心拆解}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{要点一（平方转化）：}]
			将待证不等式两边平方，转化为比较 $(\sqrt{u}+\sqrt{v})^2$ 与 $2(u+v)$ 的大小关系。\n
		\item[\textbf{要点二（完全平方）：}]
			作差后得到 $2(u+v) - (\sqrt{u}+\sqrt{v})^2 = (\sqrt{u} - \sqrt{v})^2 \ge 0$，从而原不等式成立。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{几何本质（向量投影）}}\n\n\textbf{该不等式是柯西不等式 $(a_1b_1 + a_2b_2)^2 \le (a_1^2+a_2^2)(b_1^2+b_2^2)$ 的特例。取 $a_1=a_2=1$, $b_1=\sqrt{u}$, $b_2=\sqrt{v}$ 即得。几何上表示向量 $(1,1)$ 与 $(\sqrt{u},\sqrt{v})$ 的内积不大于它们长度的乘积。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{代数意义（不等式链）}}\n\n\textbf{该不等式可视为 $(\sqrt{u}+\sqrt{v})^2 \le 2(u+v)$，即 $\frac{\sqrt{u}+\sqrt{v}}{\sqrt{2}} \le \sqrt{u+v}$，揭示了算术平方根之和与和的算术平方根之间的数量关系。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{考点价值（最值工具）}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{考法一（求含根式和最值）：} 当 $u+v$ 为定值时，直接使用该不等式求出 $\sqrt{u}+\sqrt{v}$ 的最大值。\n
		\item \textbf{考法二（证明含根号不等式）：} 利用该不等式放缩，将根式之和转化为一次式的平方根形式，便于进一步推导。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{顿悟点}}\n\n\textbf{不等式取等条件 $u=v$ 恰好对应 $(\sqrt{u}-\sqrt{v})^2=0$，因此平方差为 0 是等号成立的关键。}\n\n

	\medskip
	\n\textbf{\textcolor{CExplanation}{使用场景}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{场景一（已知和求根式和）：} 已知 $u+v$ 为常数，直接代入公式得 $\sqrt{u}+\sqrt{v}\le\sqrt{2(u+v)}$，最大值为 $\sqrt{2(u+v)}$。\n
		\item \textbf{场景二（含多个根式）：} 对于形如 $\sqrt{x}+\sqrt{y}$ 且 $x+y$ 为定值的最值问题，可转化为该模型。\n
	\end{itemize}
	\n
\end{explanationbox}
