	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略非负）}}]
			使用结论时未检查 $u,v$ 是否非负，直接套用。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（非负条件）：}}]
			结论要求 $u,v\ge 0$，否则 $\sqrt{u}$ 无定义或不等式不成立。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（定义域疏忽）：}}]
			$\sqrt{u}$ 只对 $u\ge0$ 有意义，且平方比较时需要两边非负；若 $u<0$ 或 $v<0$，原式无意义或符号相反。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（方向搞反）}}]
			误认为 $\sqrt{u}+\sqrt{v} \ge \sqrt{2(u+v)}$，导致最值方向错误。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（正确方向）：}}]
			正确不等式为 $\sqrt{u}+\sqrt{v} \le \sqrt{2(u+v)}$，$\sqrt{2(u+v)}$ 是上界。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（平方后误解）：}}]
			平方后得 $(\sqrt{u}+\sqrt{v})^2 = u+v+2\sqrt{uv}$，而 $2(u+v)=2u+2v$，作差后 $2(u+v)-(\sqrt{u}+\sqrt{v})^2 = (\sqrt{u}-\sqrt{v})^2 \ge 0$，故左平方小于等于右平方，开方得左小于等于右。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（取等条件缺失）}}]
			求出 $\sqrt{u}+\sqrt{v}$ 的最大值 $\sqrt{2(u+v)}$ 后，忽略验证等号能否成立。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（等号验证）：}}]
			只有当 $u=v$ 且满足题目约束时，最大值才可取到；否则该最大值只是一个不能达到的上界。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略等号条件）：}}]
			不等式取等条件与题目条件可能冲突（例如 $u=v$ 可能导致 $u+v$ 改变），必须单独检验。\n
	\end{description}
	\n
