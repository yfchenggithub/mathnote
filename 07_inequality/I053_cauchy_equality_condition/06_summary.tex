\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{柯西不等式取等当且仅当向量 $\boldsymbol{a}$ 与 $\boldsymbol{b}$ 对应分量成比例，即存在常数 $\lambda$ 使 $a_i=\lambda b_i$ 或 $b_i=\lambda a_i$（$b_i=0$ 时 $a_i$ 必为 $0$，反之亦然）。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件一（实数序列）：} 所有 $a_i,b_i$ 为实数，序列长度 $n\ge 2$（$n=1$ 时平凡成立）。
		\item \textbf{条件二（取等结构）：} 必须存在实数 $\lambda$ 满足 $a_i=\lambda b_i$ 或 $b_i=\lambda a_i$，且当某 $b_i=0$ 时 $a_i=0$，当某 $a_i=0$ 时 $b_i=0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略零系数）：} 使用比例形式 $\frac{a_i}{b_i}$ 时，必须先确认 $b_i\neq 0$，否则需单独讨论。
		\item \textbf{检查项（验证可等性）：} 利用不等式求最值后，必须代入取等条件检验是否能满足题设，防止无效最值。
	\end{itemize}
\end{summarybox}
