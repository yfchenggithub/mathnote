\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（实数序列）：}}] 对于任意实数序列 $a_i,b_i\in\mathbb{R}$ ($i=1,2,\dots,n$, $n\ge 2$).
		\item[\textbf{\textcolor{CStatement}{条件二（向量定义）：}}] 定义向量 $\boldsymbol{a}=(a_1,\dots,a_n)$, $\boldsymbol{b}=(b_1,\dots,b_n)$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（基本不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平方和之积不小于点积平方.
			\[
				\biggl(\sum_{i=1}^n a_i^2\biggr)\biggl(\sum_{i=1}^n b_i^2\biggr) \ge \biggl(\sum_{i=1}^n a_i b_i\biggr)^2.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（取等条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			等号成立当且仅当两向量对应分量成比例.

			等号成立等价于存在实数 $\lambda$ 使得
			\[
				a_i = \lambda b_i \quad \text{或} \quad b_i = \lambda a_i
				\qquad
				(i=1,\dots,n).
			\]
			这等价于向量 $\boldsymbol{a}$ 与 $\boldsymbol{b}$ 线性相关。特别地，若某些 $b_i=0$，则对应的 $a_i$ 必为 $0$；反之若某些 $a_i=0$，则对应的 $b_i$ 必为 $0$（从而保证比例关系的连贯性）.

		\item[\textbf{\textcolor{CStatement}{推广结论（向量形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			向量模积不等式，等号当向量共线.
			\[
				|\boldsymbol{a}\cdot\boldsymbol{b}| \le \|\boldsymbol{a}\|\,\|\boldsymbol{b}\|,
			\]
			等号成立当且仅当 $\boldsymbol{a}$ 与 $\boldsymbol{b}$ 线性相关.
	\end{description}

\end{statementbox}
