\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过构造平方和 $\sum (a_i - \lambda b_i)^2 \ge 0$ 并展开为关于 $\lambda$ 的二次函数，利用二次函数非负推得判别式非正，从而得到不等式；等号成立条件对应于平方和为零，进而推出分量成比例.

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（构造非负式）：}] 对任意实数 $\lambda$，有
			\[
				\sum_{i=1}^n (a_i - \lambda b_i)^2 \ge 0.
			\]
			\par\textbf{理由：} 平方和恒非负。

		\item[\textbf{步骤二（展开整理）：}] 展开得
			\[
				\sum_{i=1}^n a_i^2 - 2\lambda\sum_{i=1}^n a_i b_i + \lambda^2\sum_{i=1}^n b_i^2 \ge 0.
			\]
			\par\textbf{理由：} 完全平方公式 $(x-y)^2 = x^2 - 2xy + y^2$。

		\item[\textbf{步骤三（分类讨论一：$\sum b_i^2 = 0$）：}] 若 $\sum b_i^2 = 0$，则所有 $b_i=0$，原不等式成为 $\bigl(\sum a_i^2\bigr)\cdot 0 \ge (0)^2$，即 $0 \ge 0$，等号总成立。
			此时，取 $\lambda = 0$，有 $b_i = \lambda a_i$（即 $0 = 0 \cdot a_i$），故取等条件“存在 $\lambda$ 使 $a_i = \lambda b_i$ 或 $b_i = \lambda a_i$”成立。
			因此，无论 $a_i$ 为何值，等号与取等条件同时成立。
			\par\textbf{理由：} 直接验证。

		\item[\textbf{步骤四（分类讨论二：$\sum b_i^2 > 0$）：}] 若 $\sum b_i^2 > 0$，则步骤二的不等式是关于 $\lambda$ 的二次函数，开口向上且恒非负，故其判别式
			\[
				\Delta = 4\Bigl(\sum_{i=1}^n a_i b_i\Bigr)^2 - 4\Bigl(\sum_{i=1}^n b_i^2\Bigr)\Bigl(\sum_{i=1}^n a_i^2\Bigr) \le 0,
			\]
			整理得
			\[
				\Bigl(\sum_{i=1}^n a_i^2\Bigr)\Bigl(\sum_{i=1}^n b_i^2\Bigr) \ge \Bigl(\sum_{i=1}^n a_i b_i\Bigr)^2.
			\]
			\par\textbf{理由：} 一元二次不等式 $A\lambda^2 + B\lambda + C \ge 0$（$A>0$）恒成立 $\iff$ 判别式 $B^2-4AC \le 0$。

		\item[\textbf{步骤五（取等条件分析）：}] 当 $\sum b_i^2 > 0$ 时，等号成立等价于 $\Delta = 0$。此时二次函数有唯一零点 $\lambda_0 = \frac{\sum a_i b_i}{\sum b_i^2}$，且
			\[
				\sum_{i=1}^n (a_i - \lambda_0 b_i)^2 = 0,
			\]
			从而对每个 $i$ 有 $a_i - \lambda_0 b_i = 0$，即 $a_i = \lambda_0 b_i$。若某 $b_i=0$，则自然得 $a_i=0$。
			\par\textbf{理由：} 平方和为零推出每一项为零。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	综合步骤三和步骤五，柯西不等式取等条件为：向量 $\boldsymbol{a}$ 与 $\boldsymbol{b}$ 线性相关，即存在实数 $\lambda$ 使得 $a_i = \lambda b_i$ 或 $b_i = \lambda a_i$。
	特别地，当 $b_i$ 不全为零时，常写成 $a_i = \lambda b_i$；当 $b_i$ 全为零时，取等条件自然满足（因为 $\boldsymbol{b}=0$ 与任意 $\boldsymbol{a}$ 共线）。
\end{proofbox}
