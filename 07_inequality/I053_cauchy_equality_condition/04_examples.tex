\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知实数 $a,b$ 满足 $a^2+b^2=1$，求 $a+2b$ 的最大值.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造不等式）：}] 应用柯西不等式：
			\[
				(a^2+b^2)(1^2+2^2) \ge (a+2b)^2.
			\]
			\par\textbf{理由：}
			取 $a_1=a, a_2=b$; $b_1=1, b_2=2$，代入 $\bigl(\sum a_i^2\bigr)\bigl(\sum b_i^2\bigr) \ge \bigl(\sum a_i b_i\bigr)^2$.

		\item[\textbf{第二步（求范围）：}] 代入已知 $a^2+b^2=1$，得 $1\cdot 5 \ge (a+2b)^2$，所以 $(a+2b)^2 \le 5$，从而 $-\sqrt{5} \le a+2b \le \sqrt{5}$，故最大值为 $\sqrt{5}$.
			\par\textbf{理由：}
			算术运算及开方.

		\item[\textbf{第三步（取等验证）：}] 等号成立当且仅当 $\frac{a}{1} = \frac{b}{2}$，即 $a = \frac{b}{2}$，结合 $a^2+b^2=1$，解得 $a = \frac{\sqrt{5}}{5}, b = \frac{2\sqrt{5}}{5}$（此时 $a+2b = \sqrt{5}$），等号可取.
			\par\textbf{理由：}
			柯西取等条件 $a_i = \lambda b_i$ 解方程组.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最大值为 $\sqrt{5}$，取等时 $a=\frac{\sqrt{5}}{5}, b=\frac{2\sqrt{5}}{5}$}.

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知实数 $x,y$ 满足 $x^2+4y^2=4$，求 $x+2y$ 的取值范围.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（配凑形式）：}] 将条件改写为 $x^2+(2y)^2=4$，目标 $x+2y=1\cdot x+1\cdot(2y)$. 应用柯西不等式：
			\[
				\bigl[x^2+(2y)^2\bigr](1^2+1^2) \ge (x+2y)^2.
			\]
			\par\textbf{理由：}
			取 $a_1=x, a_2=2y$; $b_1=1, b_2=1$.

		\item[\textbf{第二步（计算范围）：}] 代入得 $4\cdot 2 \ge (x+2y)^2$，即 $(x+2y)^2 \le 8$，所以 $-2\sqrt{2} \le x+2y \le 2\sqrt{2}$.
			\par\textbf{理由：}
			代数运算.

		\item[\textbf{第三步（取等求解）：}] 等号成立当 $\frac{x}{1} = \frac{2y}{1}$，即 $x=2y$. 代入 $x^2+4y^2=4$ 得 $(2y)^2+4y^2=8y^2=4$，故 $y=\pm\frac{\sqrt{2}}{2}, x=\pm\sqrt{2}$，等号可取.
			\par\textbf{理由：}
			柯西取等条件 $a_i = \lambda b_i$ 解方程组.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x+2y$ 的取值范围是 $[-2\sqrt{2}, 2\sqrt{2}]$，取等时 $x=\sqrt{2},y=\frac{\sqrt{2}}{2}$ 或 $x=-\sqrt{2},y=-\frac{\sqrt{2}}{2}$}.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $x,y,z>0$ 且 $x+y+z=1$，求 $x^2+y^2+z^2$ 的最小值.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造三维柯西）：}] 应用三维柯西不等式：
			\[
				(1^2+1^2+1^2)(x^2+y^2+z^2) \ge (x+y+z)^2.
			\]
			\par\textbf{理由：}
			取 $a_1=a_2=a_3=1$，$b_1=x, b_2=y, b_3=z$，代入公式.

		\item[\textbf{第二步（代入求范围）：}] 由 $x+y+z=1$ 得 $3(x^2+y^2+z^2) \ge 1$，所以 $x^2+y^2+z^2 \ge \frac{1}{3}$.
			\par\textbf{理由：}
			不等式两边同除以 $3$.

		\item[\textbf{第三步（取等验证）：}] 等号成立当存在 $\lambda$ 使 $1 = \lambda x$, $1 = \lambda y$, $1 = \lambda z$，即 $x=y=z$. 结合 $x+y+z=1$ 得 $x=y=z=\frac{1}{3}$，等号可取.
			\par\textbf{理由：}
			柯西取等条件 $a_i = \lambda b_i$.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$x^2+y^2+z^2$ 的最小值为 $\dfrac{1}{3}$，取等时 $x=y=z=\dfrac{1}{3}$}.
\end{examplebox}
