\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（符号混淆）}}] 误认为$(a^2-b^2)(c^2-d^2)\ge (ac-bd)^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（方向区分）：}}] 反向柯西为$\le$，不同于标准柯西的$\ge$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}] 记混两种柯西不等式，未区分平方差与平方和。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（等号条件误判）}}] 误以为等号成立条件是$a=c$且$b=d$或$\dfrac{a}{c}=\dfrac{b}{d}$（不分情形）。
		\item[\textbf{\textcolor{CTrap}{正确理解（等号条件）：}}] 等号成立当且仅当$ad=bc$，即两内向乘积等于两外向乘积。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导不熟）：}}] 没有通过配方掌握等号来源，靠猜测记忆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（结构匹配错误）}}] 对于$(x^2-9)(4y^2-1)$等式子，不知道如何选取$a,b,c,d$。
		\item[\textbf{\textcolor{CTrap}{正确理解（对应方法）：}}] 将每个平方差写成$a^2-b^2$形式，其中$a,c$是被减数，$b,d$是减数。例如$(x^2-9)(4y^2-1)=(x^2-3^2)((2y)^2-1^2)$，则$a=x,b=3,c=2y,d=1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（观察不细）：}}] 未将常数写成平方形式，未正确识别被减数与减数。
	\end{description}
\end{trapbox}
