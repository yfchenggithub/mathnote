\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	直接作差，展开后合并，再配方成完全平方。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（移项作差）：}] 引入差式：
			\[
				D = (ac-bd)^2 - (a^2-b^2)(c^2-d^2)
			\]
			\par\textbf{理由：}
			通过比较差与0的大小来证明不等式。

		\item[\textbf{步骤二（展开计算）：}] 分别展开两项：
			\[
				\begin{aligned}
					(ac-bd)^2 &= a^2c^2 - 2abcd + b^2d^2,\\
					(a^2-b^2)(c^2-d^2) &= a^2c^2 - a^2d^2 - b^2c^2 + b^2d^2
			\end{aligned}
			\]
			\par\textbf{理由：}
			利用完全平方公式和多项式乘法展开。

		\item[\textbf{步骤三（合并化简）：}] 用前式减去后式：
			\[
				\begin{aligned}
					D &= (a^2c^2 - 2abcd + b^2d^2) - (a^2c^2 - a^2d^2 - b^2c^2 + b^2d^2)\\
					&= a^2d^2 - 2abcd + b^2c^2
			\end{aligned}
			\]
			\par\textbf{理由：}
			抵消 $a^2c^2$ 和 $b^2d^2$ 后得到交叉项。

		\item[\textbf{步骤四（配方定号）：}] 将结果写成完全平方：
			\[
				\begin{aligned}
					D &= a^2d^2 - 2abcd + b^2c^2 \\\
					&= (ad-bc)^2 \\\
					&\ge 0
			\end{aligned}
			\]
			\par\textbf{理由：}
			完全平方是非负的，因此 $D\ge0$，原不等式成立。
			等号当且仅当 $ad-bc=0$，即 $ad=bc$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{由 $D=(ad-bc)^2\ge0$ 得 $(a^2-b^2)(c^2-d^2)\leqslant(ac-bd)^2$，等号在 $ad=bc$ 时成立。}
\end{proofbox}
