\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知实数$a=2$，$b=1$，$c=3$，$d=1$，比较$P=(a^2-b^2)(c^2-d^2)$与$Q=(ac-bd)^2$的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}] 观察$P$与$Q$的形式，符合反向柯西不等式的左右两边。
			\par\textbf{理由：}
			可直接套用该不等式比较大小。
		\item[\textbf{第二步（套用结论）：}] 由反向柯西不等式得$(a^2-b^2)(c^2-d^2)\le (ac-bd)^2$，即$P\le Q$。
			\par\textbf{理由：}
			不等式直接给出大小关系。
		\item[\textbf{第三步（检验等号）：}] 计算$ad=2\times1=2$，$bc=1\times3=3$，$ad\neq bc$，故等号不成立，$P<Q$。
			\par\textbf{理由：}
			等号条件$ad=bc$不满足，所以严格不等式成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P<Q$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知实数$x,y$，求证：$(x^2-4)(9-y^2)\le (3x-2y)^2$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（辨识结构）：}] 将左边视为$(x^2-2^2)(3^2-y^2)$，右边视为$(x\cdot3-2\cdot y)^2$，对应$a=x,b=2,c=3,d=y$。
			\par\textbf{理由：}
			匹配反向柯西不等式的标准形式。
		\item[\textbf{第二步（套用不等式）：}] 由反向柯西不等式$(a^2-b^2)(c^2-d^2)\le (ac-bd)^2$得$(x^2-4)(9-y^2)\le (3x-2y)^2$。
			\par\textbf{理由：}
			直接代入$a,b,c,d$。
		\item[\textbf{第三步（说明等号）：}] 等号当且仅当$ad=bc$，即$x\cdot y=2\cdot3=6$时成立，本题只需证明不等式，无需取等。
			\par\textbf{理由：}
			等号条件来自反向柯西的充要条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{不等式恒成立}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知实数$a,b$满足$(a^2-1)(4-b^2)= (2a-b)^2$，求$a^2+b^2$的最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别等号条件）：}] 观察等式左侧为$(a^2-1^2)(2^2-b^2)$，右侧为$(a\cdot2-1\cdot b)^2$，符合反向柯西不等式的取等情况。
			\par\textbf{理由：}
			反向柯西不等式取等当且仅当$ad=bc$。
		\item[\textbf{第二步（导出关系）：}] 由$ad=bc$得$a\cdot b=1\cdot2=2$，即$ab=2$。
			\par\textbf{理由：}
			代入$a,b,c,d$对应值。
		\item[\textbf{第三步（求最值）：}] 在$ab=2$下，由基本不等式$a^2+b^2\ge 2ab=4$，当且仅当$a=b=\sqrt{2}$时取等，故最小值为$4$。
			\par\textbf{理由：}
			利用基本不等式求平方和的最值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小值为$4$}。
\end{examplebox}
