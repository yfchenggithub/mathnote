\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{反向柯西不等式等价于完全平方$(ad-bc)^2\ge0$，本质上是平方非负性的代数表达。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（实数变量）：} $a,b,c,d\in\mathbb{R}$。
		\item \textbf{条件2（对称结构）：} 式子需具有$(a^2-b^2)(c^2-d^2)$和$(ac-bd)^2$的形式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（方向易错）：} 不等号方向为$\le$，与标准柯西$\ge$相反。
		\item \textbf{检查项（等号条件）：} 等号成立条件为$ad=bc$，注意交叉检验。
	\end{itemize}
\end{summarybox}
