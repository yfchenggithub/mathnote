\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（实数变量）：}}] $a,b,c,d$ 为任意实数。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平方差型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两个平方差的乘积不超过它们交叉乘积差的平方。
			\[
				(a^2-b^2)(c^2-d^2)\leqslant(ac-bd)^2
			\]
			等号成立当且仅当 $ad=bc$。

		\item[\textbf{\textcolor{CStatement}{推广结论（等价恒等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			移项后化为完全平方，揭示不等式本质。
			\[
				(ac-bd)^2-(a^2-b^2)(c^2-d^2)=(ad-bc)^2
			\]
	\end{description}
\end{statementbox}
