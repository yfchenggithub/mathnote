\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（非负实数）：}}]
			对 $n$ 个实数 $a_1,a_2,\dots,a_n$，要求 $a_i \ge 0$，且 $n \ge 2$。
	\end{description}
	\par
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（算术几何不等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			算术平均不小于几何平均。
			\[
				\frac{a_1+a_2+\cdots+a_n}{n} \ge \sqrt[n]{a_1a_2\cdots a_n}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（两数特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两正数的算术平均不小于几何平均。
			\[
				\frac{a+b}{2} \ge \sqrt{ab},\quad a,b \ge 0.
			\]
	\end{description}
	\par
	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	当且仅当 $a_1=a_2=\cdots=a_n$ 时等号成立。
\end{statementbox}
