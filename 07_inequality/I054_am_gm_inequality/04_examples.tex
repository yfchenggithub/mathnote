\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $a,b>0$，且 $a+b=1$，求 $ab$ 的最大值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}]
			$a,b>0$，和为定值 $1$。
			\par\textbf{理由：} 均值不等式用于正数，和定求积最大。
		\item[\textbf{第二步（套用均值）：}]
			由 $\frac{a+b}{2} \ge \sqrt{ab}$，得 $\sqrt{ab} \le \frac12$，即 $ab \le \frac14$。
			\par\textbf{理由：} 二元均值不等式直接变形。
		\item[\textbf{第三步（取等验证）：}]
			等号当 $a=b$，结合 $a+b=1$ 得 $a=b=\frac12$，在条件范围内，故最大值 $\frac14$ 可取。
			\par\textbf{理由：} 等号条件必须满足且实际可达到。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$ab$ 的最大值为 $\frac14$（当 $a=b=\frac12$）。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $x,y>0$，$2x+3y=6$，求 $xy$ 的最大值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造乘积形式）：}]
			将 $2x$ 和 $3y$ 视为整体，则它们的和 $2x+3y=6$ 为定值，乘积 $(2x)(3y)=6xy$。
			\par\textbf{理由：} 通过系数配凑，使两正数的和为定值，从而可用均值不等式。
		\item[\textbf{第二步（套用均值）：}]
			对 $2x$ 和 $3y$ 使用均值不等式：$\frac{2x+3y}{2} \ge \sqrt{(2x)(3y)}$，即 $3 \ge \sqrt{6xy}$，平方得 $9 \ge 6xy$，故 $xy \le \frac{9}{6} = \frac32$。
			\par\textbf{理由：} 由 $\sqrt{uv}\le \frac{u+v}{2}$，代入 $u=2x,v=3y$。
		\item[\textbf{第三步（取等验证）：}]
			等号当 $2x=3y$ 且 $2x+3y=6$，解得 $2x=3y=3$，即 $x=\frac32, y=1$，满足正数条件，故最大值为 $\frac32$ 可达到。
			\par\textbf{理由：} 等号要求两数相等。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$xy$ 的最大值为 $\frac32$（当 $x=\frac32,y=1$）。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $a,b,c>0$，$a+b+c=3$，求 $\frac1a+\frac1b+\frac1c$ 的最小值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用均值到倒数之和）：}]
			由 $\frac1a,\frac1b,\frac1c>0$，三元均值不等式得
			\[
				\begin{aligned}
					\frac1a+\frac1b+\frac1c &\ge 3\sqrt[3]{\frac1a\cdot\frac1b\cdot\frac1c} \\ &= \frac{3}{\sqrt[3]{abc}}
			\end{aligned}
			\]
			\par\textbf{理由：} 对三个正数使用平均值不等式。
		\item[\textbf{第二步（求 $abc$ 最大值）：}]
			对 $a,b,c$ 使用均值不等式：$a+b+c \ge 3\sqrt[3]{abc}$，所以 $\sqrt[3]{abc} \le \frac{a+b+c}{3}=1$，即 $abc \le 1$，等号当 $a=b=c=1$。
			\par\textbf{理由：} 由 $n=3$ 的均值不等式直接得。
		\item[\textbf{第三步（结合求最小值）：}]
			由 $abc \le 1$ 得 $\sqrt[3]{abc} \le 1$，所以 $\frac{3}{\sqrt[3]{abc}} \ge 3$。结合第一步，得 $\frac1a+\frac1b+\frac1c \ge 3$。等号当 $a=b=c=1$ 时同时满足两步等号条件，故可取得最小值 $3$。
			\par\textbf{理由：} 分母越小分数越大，等号传递性。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\frac1a+\frac1b+\frac1c$ 的最小值为 $3$（当 $a=b=c=1$）。}
\end{examplebox}
