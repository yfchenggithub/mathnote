\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{对非负实数，算术平均不小于几何平均，等号当且仅当所有数相等。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（非负实数）：} 各数必须非负（$a_i \ge 0$）。
		\item \textbf{条件 2（至少两项）：} 个数 $n \ge 2$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（正、定、等）：} 忽视非负、未构造定和或定积、不验等号。
		\item \textbf{检查项（验证等号）：} 应用后必须检查等号能否取到，若不能则需改用其他方法。
	\end{itemize}

\end{summarybox}
