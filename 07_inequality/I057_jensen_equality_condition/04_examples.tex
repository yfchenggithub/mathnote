\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$f(x)=x^2$在整个实数集上严格凸，设$x_1,x_2\in\mathbb{R}$，取$\lambda_1=\lambda_2=\frac12$，应用琴生不等式写出不等式并指出取等条件。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定凸性）：}] $f(x)=x^2$是严格凸函数（常用结论）。
			\par\textbf{理由：} 二次函数开口向上。
		\item[\textbf{第二步（套用公式）：}] 由琴生不等式得
			\[
				f\!\left(\frac{x_1+x_2}{2}\right)\le\frac12 f(x_1)+\frac12 f(x_2),
			\]
			即$\displaystyle\Bigl(\frac{x_1+x_2}{2}\Bigr)^2\le\frac{x_1^2+x_2^2}{2}$.
			\par\textbf{理由：} 直接代入$f$和权重。
		\item[\textbf{第三步（取等条件）：}] 等号成立当且仅当$x_1=x_2$.
			\par\textbf{理由：} 严格凸函数的琴生不等式取等条件。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	整理得$x_1^2+x_2^2\ge2x_1x_2$，即均值不等式.

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$f(x)=e^x$严格凸，设$a,b\in\mathbb{R}$，利用琴生不等式比较$e^{\frac{a+b}{2}}$与$\frac{e^a+e^b}{2}$的大小，并指出取等条件.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定凸性）：}] $f(x)=e^x$是严格凸函数.
			\par\textbf{理由：} 指数函数二阶导恒正.
		\item[\textbf{第二步（套用公式）：}] 取$\lambda_1=\lambda_2=\frac12$，由琴生不等式得
			\[
				e^{\frac{a+b}{2}}\le\frac{e^a+e^b}{2}.
			\]
			\par\textbf{理由：} 直接代入.
		\item[\textbf{第三步（取等条件）：}] 等号成立当且仅当$a=b$.
			\par\textbf{理由：} 严格凸取等条件.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	得到 $e^{\frac{a+b}{2}}\le\frac{e^a+e^b}{2}$.

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$f(x)=\ln x$在$(0,+\infty)$上严格凹，设$a,b>0$，证明$\frac{a+b}{2}\ge\sqrt{ab}$，并指出取等条件.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定凹性）：}] $f(x)=\ln x$是严格凹函数.
			\par\textbf{理由：} 对数函数二阶导恒负.
		\item[\textbf{第二步（套用公式）：}] 取$\lambda_1=\lambda_2=\frac12$，由严格凹的琴生不等式（反向）得
			\[
				\begin{aligned}
					\ln\frac{a+b}{2} &\ge \frac{\ln a+\ln b}{2} \\
					&= \ln\sqrt{ab}.
			\end{aligned}
			\]
			\par\textbf{理由：} 凹函数性质，不等号反向.
		\item[\textbf{第三步（化简结论）：}] 因为$\ln x$严格递增，所以$\frac{a+b}{2}\ge\sqrt{ab}$.
			\par\textbf{理由：} 对数函数单调性.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	得到算术-几何平均不等式，等号成立当且仅当$a=b$.
\end{examplebox}
