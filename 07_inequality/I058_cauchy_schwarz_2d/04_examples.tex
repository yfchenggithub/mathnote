\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知实数 $x,y$ 满足 $x^{2}+y^{2}=1$，求 $u=3x+4y$ 的最大值.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}] 将 $3x+4y$ 视为点积 $(3,4)\cdot(x,y)$，条件为平方和 $x^{2}+y^{2}$，符合柯西不等式形式.
			\par\textbf{理由：}
			由柯西不等式 $(3^{2}+4^{2})(x^{2}+y^{2})\ge (3x+4y)^{2}$.
		\item[\textbf{第二步（代入计算）：}] $ (9+16)\times 1 \ge u^{2}$，即 $25\ge u^{2}$，故 $|u|\le 5$.
			\par\textbf{理由：}
			代入 $x^{2}+y^{2}=1$，左边为 $25$.
		\item[\textbf{第三步（确定最值）：}] 由 $|u|\le5$ 可知 $u\le5$。取 $x=\dfrac35,y=\dfrac45$ 时，条件成立且 $u=5$，故最大值为 $5$；取 $x=-\dfrac35,y=-\dfrac45$ 时，$u=-5$，故最小值为 $-5$.
			\par\textbf{理由：}
			等号条件 $ad=bc$ 对应 $3y=4x$；再根据所求的是最大值还是最小值，选择同向或反向的取等点.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
		$\max(3x+4y)=5$，在 $x=\dfrac{3}{5},y=\dfrac{4}{5}$ 时取得；$x=-\dfrac{3}{5},y=-\dfrac{4}{5}$ 对应的是最小值 $-5$.

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $a,b,c,d$ 为实数，且 $a^{2}+b^{2}=1$，$c^{2}+d^{2}=4$，求 $ac+bd$ 的最大值.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（匹配形式）：}] 题设给出两个平方和，直接应用柯西不等式 $(a^{2}+b^{2})(c^{2}+d^{2})\ge(ac+bd)^{2}$.
			\par\textbf{理由：}
			套用定理.
		\item[\textbf{第二步（代入数值）：}] 左边 $1\times4=4$，所以 $(ac+bd)^{2}\le4$，即 $|ac+bd|\le2$.
			\par\textbf{理由：}
			直接代入.
		\item[\textbf{第三步（确定最值）：}] 因此 $ac+bd$ 的最大值为 $2$，最小值为 $-2$。由 $ad=bc$ 及两个平方和条件可知，两向量取等时平行：取 $(c,d)=2(a,b)$ 得最大值 $2$，取 $(c,d)=-2(a,b)$ 得最小值 $-2$。例如 $a=1,b=0,c=2,d=0$ 时取得最大值.
			\par\textbf{理由：}
			$ad=bc$ 只保证两向量共线；还需根据比例系数的正负区分最大值与最小值.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\max(ac+bd)=2$，等号可达到.

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $x,y>0$，$x^{2}+y^{2}=2$，求 $2x+3y$ 的最大值.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（构造柯西形式）：}] 将 $2x+3y$ 看作 $(2,3)$ 与 $(x,y)$ 的点积，柯西不等式给出
			\[
				(2^{2}+3^{2})(x^{2}+y^{2})\ge(2x+3y)^{2}.
			\]
			\par\textbf{理由：}
			因为 $(2,3)$ 和 $(x,y)$ 均为实数向量.
		\item[\textbf{第二步（代入已知条件）：}] $(4+9)\times2\ge(2x+3y)^{2}$，即 $26\ge(2x+3y)^{2}$，所以 $|2x+3y|\le\sqrt{26}$.
			\par\textbf{理由：}
			代入 $x^{2}+y^{2}=2$.
		\item[\textbf{第三步（确定最大值并检查符号）：}] 由 $x,y>0$ 知 $2x+3y>0$，故最大值为 $\sqrt{26}$. 等号成立条件为 $2y=3x$，与 $x^{2}+y^{2}=2$ 联立可得 $x=\dfrac{2\sqrt{26}}{13}$, $y=\dfrac{3\sqrt{26}}{13}$（这一解满足正性）.
			\par\textbf{理由：}
			等号条件 $2\cdot y=3\cdot x$，即 $ad=bc$ 的具体体现.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\max(2x+3y)=\sqrt{26}$，在 $x=\dfrac{2\sqrt{26}}{13},y=\dfrac{3\sqrt{26}}{13}$ 时取得.
\end{examplebox}
