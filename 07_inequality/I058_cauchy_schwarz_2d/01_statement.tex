\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（变量范围）：}}] 设$a,b,c,d$为任意实数.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（二维柯西不等）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      两个平方和之积不小于内积的平方.
		      \[
			      (a^{2}+b^{2})(c^{2}+d^{2})\ge (ac+bd)^{2}
		      \]

		\item[\textbf{\textcolor{CStatement}{常见变形（向量形式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      用向量模长与点积可等价表示.
		      \[
			      |\boldsymbol{u}|^{2}|\boldsymbol{v}|^{2}\ge (\boldsymbol{u}\cdot\boldsymbol{v})^{2},
		      \]
		      其中 $\boldsymbol{u}=(a,b), \boldsymbol{v}=(c,d)$.
	\end{description}

	\textbf{\textcolor{CStatement}{等号/取等条件：}}
	当且仅当 $ad=bc$ 时等号成立.
\end{statementbox}
