\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过计算左边与右边的差，化为完全平方，利用平方非负性证明.

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（展开左边）：}]
			\[
				(a^{2}+b^{2})(c^{2}+d^{2}) = a^{2}c^{2}+a^{2}d^{2}+b^{2}c^{2}+b^{2}d^{2}.
			\]
			\par\textbf{理由：}
			按多项式乘法逐项展开.

		\item[\textbf{步骤二（展开右边）：}]
			\[
				(ac+bd)^{2} = a^{2}c^{2}+2abcd+b^{2}d^{2}.
			\]
			\par\textbf{理由：}
			完全平方公式展开.

		\item[\textbf{步骤三（作差变形）：}]
			\[
				\begin{aligned}
					(a^{2}+b^{2})(c^{2}+d^{2}) - (ac+bd)^{2}
					&= (a^{2}c^{2}+a^{2}d^{2}+b^{2}c^{2}+b^{2}d^{2}) - (a^{2}c^{2}+2abcd+b^{2}d^{2}) \\
					&= a^{2}d^{2} - 2abcd + b^{2}c^{2} \\
					&= (ad-bc)^{2}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			合并同类项，并用完全平方公式写成 $(ad-bc)^{2}$.

		\item[\textbf{步骤四（判号得证）：}]
			因为 $(ad-bc)^{2} \ge 0$，所以
			\[
				(a^{2}+b^{2})(c^{2}+d^{2}) \ge (ac+bd)^{2}.
			\]
			等号成立当且仅当 $ad-bc=0$，即 $ad=bc$.
			\par\textbf{理由：}
			平方非负，差为零时取等.
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	柯西不等式得证，等号条件是 $ad=bc$.
\end{proofbox}
