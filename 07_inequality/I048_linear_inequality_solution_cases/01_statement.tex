\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（一次含参）：}}] $a,b\in\mathbb{R}$，$x$ 为未知数，不等式 $ax>b$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（分类规则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			解集由 $a$ 的符号和 $a=0$ 时 $b$ 的正负决定。
			\[
				\begin{cases}
					\left(\frac{b}{a}, +\infty\right), & a>0,\\
					\left(-\infty, \frac{b}{a}\right), & a<0,\\
					\mathbb{R}, & a=0,\ b<0,\\
					\varnothing, & a=0,\ b\ge 0.
			\end{cases}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（不等号反向）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若不等式改为 $ax<b$，则解集规则相反。
			\[
				\begin{cases}
					\left(-\infty, \frac{b}{a}\right), & a>0,\\
					\left(\frac{b}{a}, +\infty\right), & a<0,\\
					\mathbb{R}, & a=0,\ b>0,\\
					\varnothing, & a=0,\ b\le 0.
			\end{cases}
			\]
	\end{description}

\end{statementbox}
