\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{一次不等式 $ax>b$ 的解集取决于 $a$ 的符号及 $a=0$ 时 $b$ 的值，分 $a>0$、$a<0$、$a=0$ 三类讨论。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准形式）：} 不等式能化为 $ax>b$ 且 $a,b\in\mathbb{R}$。
		\item \textbf{条件 2（参数任意）：} 对 $a,b$ 没有任何限制，必须考虑一切可能。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（零系讨论）：} 务必先检查 $a$ 是否为零，再对 $a\neq0$ 分正负。
		\item \textbf{检查项（变号确认）：} 当 $a<0$ 时，不等号方向必须反转。
	\end{itemize}

\end{summarybox}
