\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略零系）：}}] 直接写出 $x > \frac{b}{a}$ 或 $x < \frac{b}{a}$，未讨论 $a=0$ 的情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（先判零系）：}}] 必须先分 $a=0$ 与 $a\neq0$ 两类，$a\neq0$ 时再按 $a>0$、$a<0$ 讨论。
		\item[\textbf{\textcolor{CTrap}{错因分析（遗忘除零）：}}] 默认 $a\neq0$ 直接两边除以 $a$，忽略了 $a$ 可能为零导致除法无效。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（变号遗漏）：}}] 当 $a<0$ 时，仍将解集写为 $(\frac{b}{a}, +\infty)$。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号反转）：}}] $a<0$ 时除以 $a$ 必须将不等号反向，解集为 $(-\infty, \frac{b}{a})$。
		\item[\textbf{\textcolor{CTrap}{错因分析（性质模糊）：}}] 对不等式性质理解不深刻，忘记除以负数时不等号方向必须改变。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（常数混淆）：}}] 当 $a=0$ 时，误认为 $b<0$ 无解，$b\ge0$ 解集为 $\mathbb{R}$，与正确结论混淆。
		\item[\textbf{\textcolor{CTrap}{正确理解（常数判断）：}}] $a=0$ 时不等式为 $0>b$，故 $b<0$ 恒成立（解集 $\mathbb{R}$），$b\ge0$ 恒不成立（解集 $\varnothing$）。
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性思维）：}}] 习惯考虑 $a\neq0$ 的情形，对 $a=0$ 时不等式退化为常数比较缺乏认识。
	\end{description}
\end{trapbox}
