\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	解不等式 $3x > 6$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（系数识别）：}] 识别 $a=3>0$，$b=6$。
			\par\textbf{理由：} 不等式为 $ax>b$ 形式，直接读出 $a$、$b$ 的值。
		\item[\textbf{第二步（套用规则）：}] 根据 $a>0$ 时解集为 $\left(\frac{b}{a}, +\infty\right)$，得 $x > \frac{6}{3}$。
			\par\textbf{理由：} 使用分类结论中 $a>0$ 的情形。
		\item[\textbf{第三步（计算结果）：}] $\frac{6}{3}=2$，故解集为 $(2, +\infty)$。
			\par\textbf{理由：} 化简得最终集合。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{不等式 $3x>6$ 的解集为 $(2,+\infty)$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	解关于 $x$ 的不等式 $kx > 1$，其中 $k\in\mathbb{R}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（标准形式）：}] 不等式已为 $kx > 1$ 形式，$a=k$，$b=1$。
			\par\textbf{理由：} 匹配分类条件 $ax>b$。
		\item[\textbf{第二步（正系数情形）：}] 若 $k>0$，则两边除以 $k$ 得 $x > \frac{1}{k}$。
			\par\textbf{理由：} $k>0$ 时，不等号方向不变，解集为 $\left(\frac{1}{k}, +\infty\right)$。
		\item[\textbf{第三步（负系数情形）：}] 若 $k<0$，则两边除以 $k$ 得 $x < \frac{1}{k}$。
			\par\textbf{理由：} $k<0$ 时，不等号反向，解集为 $\left(-\infty, \frac{1}{k}\right)$。
		\item[\textbf{第四步（零系数情形）：}] 若 $k=0$，则不等式化为 $0 > 1$，该不等式恒不成立，故无解，解集为 $\varnothing$。
			\par\textbf{理由：} $k=0$ 时原不等式为 $0\cdot x > 1$，即 $0>1$，假，故无解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{当 $k>0$ 时解集为 $\left(\frac{1}{k}, +\infty\right)$；当 $k<0$ 时解集为 $\left(-\infty, \frac{1}{k}\right)$；当 $k=0$ 时解集为 $\varnothing$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	若关于 $x$ 的不等式 $ax > b$ 的解集为 $(-\infty, 2)$，求实数 $a,b$ 满足的条件。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（解集特征分析）：}] 已知解集为 $(-\infty,2)$，即 $x<2$ 的形式，对应分类结论中 $a<0$ 的情形。
			\par\textbf{理由：} 分类结论中，当 $a<0$ 时解集为 $\left(-\infty, \frac{b}{a}\right)$。
		\item[\textbf{第二步（等量关系建立）：}] 由 $a<0$ 可得解集为 $\left(-\infty, \frac{b}{a}\right)$，对照已知得 $\frac{b}{a}=2$。
			\par\textbf{理由：} 解集区间的右端点等于 $2$。
		\item[\textbf{第三步（条件表达）：}] 由 $\frac{b}{a}=2$ 得 $b=2a$，且结合 $a<0$，故 $a<0$ 且 $b=2a$。
			\par\textbf{理由：} 等式变形，并保留 $a<0$ 的前提。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a<0$ 且 $b=2a$}。
\end{examplebox}
