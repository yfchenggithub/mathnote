\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（外心位置）}}] 认为外心始终在三角形内部。
		\item[\textbf{\textcolor{CTrap}{正确理解（位置判定）}}] 锐角三角形的外心在形内，直角三角形外心在斜边中点，钝角三角形外心在形外。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）}}] 只记忆“外接圆圆心”，未结合三角形形状判断。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（比例系数）}}] 误记正弦定理为 $a:\sin A = R$ 或 $a/\sin A = R$。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式准确）}}] 正弦定理 $a/\sin A = 2R$，边与对角正弦的比值是直径而非半径。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）}}] 半径与直径未分清，或仅记住 $a = 2R\sin A$ 而忽略 $2R$ 整体。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（性质混淆）}}] 误认为外心到三角形三边距离相等（实为内心）。
		\item[\textbf{\textcolor{CTrap}{正确理解（性质区分）}}] 外心到三个顶点距离相等，即 $OA=OB=OC=R$；内心到三边距离相等。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）}}] 外心、内心、重心等性质记忆混乱，未区分“顶点”与“边”。
	\end{description}
\end{trapbox}
