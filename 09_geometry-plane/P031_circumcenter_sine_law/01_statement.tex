		\item[\textbf{\textcolor{CStatement}{条件1（三角形与外心）：}}] $\triangle ABC$ 为非退化三角形，$O$ 为其外心，$R$ 为外接圆半径。

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（等距关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外心到三个顶点距离相等，等于外接圆半径。
			\[
				OA = OB = OC = R.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（正弦定理形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形边长与对角正弦之比等于外接圆直径。
			\[
				\frac{a}{\sin A} = \frac{b}{\sin B} = \frac{c}{\sin C} = 2R.
			\]
	\end{description}
