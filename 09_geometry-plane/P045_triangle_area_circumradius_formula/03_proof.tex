\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用面积公式 $S=\frac12 ab\sin C$ 与正弦定理 $\sin C = \frac{c}{2R}$ 消去 $\sin C$ 即可。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（正弦定理转化）：}]
			\[
				\begin{aligned}
					\frac{c}{\sin C} &= 2R, \\
					\sin C &= \frac{c}{2R}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			正弦定理 $\frac{a}{\sin A}=2R$，对任意一边均成立；此处用于边 $c$ 与角 $C$。

		\item[\textbf{步骤二（面积公式表达）：}]
			\[
				S = \frac12 ab\sin C.
			\]
			\par\textbf{理由：}
			三角形面积等于两边与其夹角正弦乘积的一半；选择 $a,b$ 及夹角 $C$。

		\item[\textbf{步骤三（代入消元化简）：}]
			\[
				\begin{aligned}
					S &= \frac12 ab\cdot\frac{c}{2R} \\
					&= \frac{abc}{4R}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			将步骤一结果代入步骤二，完成乘法运算。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，对任意三角形 $ABC$，面积 $S$ 满足 $S=\dfrac{abc}{4R}$。}
\end{proofbox}
