\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{面积公式 $S=\dfrac{abc}{4R}$ 是连接三角形三边、外接圆半径与面积的桥梁，避免使用角度。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三边已知）：} 必须知道三边长 $a,b,c$。
		\item \textbf{条件 2（半径已知）：} 必须知道外接圆半径 $R$，或能先通过其他条件（如正弦定理）求得 $R$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（记忆分母）：} 公式分母是 $4R$，而非 $2R$ 或 $R$。
		\item \textbf{检查项（适用全体）：} 公式对任意三角形均成立，无需特殊条件。
	\end{itemize}
\end{summarybox}
