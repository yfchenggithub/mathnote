\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$a=3$, $b=4$, $c=5$, 外接圆半径 $R=2.5$，求面积 $S$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出公式）：}]
			\[
				S=\frac{abc}{4R}.
			\]
			\par\textbf{理由：}
			三角形面积与外接圆半径关系公式。

		\item[\textbf{第二步（代入数值）：}]
			\[
				S=\frac{3\times4\times5}{4\times2.5}.
			\]
			\par\textbf{理由：}
			将已知边长和半径代入公式。

		\item[\textbf{第三步（计算化简）：}]
			\[
				\frac{60}{10}=6.
			\]
			\par\textbf{理由：}
			分子 $3\times4\times5=60$，分母 $4\times2.5=10$，相除得面积。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S=6$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$a=5$, $b=5$, $c=6$，求其外接圆半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求半周长）：}]
			\[
				\begin{aligned}
					s &= \frac{a+b+c}{2} \\
					&= \frac{5+5+6}{2} \\
					&= 8.
			\end{aligned}
			\]
			\par\textbf{理由：}
			海伦公式需先计算半周长 $s$。

		\item[\textbf{第二步（求面积 $S$）：}]
			\[
				\begin{aligned}
					S &= \sqrt{s(s-a)(s-b)(s-c)} \\
					&= \sqrt{8\times3\times3\times2} \\
					&= \sqrt{144} \\
					&= 12.
			\end{aligned}
			\]
			\par\textbf{理由：}
			海伦公式 $S=\sqrt{s(s-a)(s-b)(s-c)}$。

		\item[\textbf{第三步（求外接圆半径）：}]
			\[
				\begin{aligned}
					R &= \frac{abc}{4S} \\
					&= \frac{5\times5\times6}{4\times12} \\
					&= \frac{150}{48} \\
					&= \frac{25}{8}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			由本结论变形 $R=\dfrac{abc}{4S}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=\dfrac{25}{8}$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，已知 $a=3$, $b=4$, $C=60^\circ$，求外接圆半径 $R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求面积 $S$）：}]
			\[
				\begin{aligned}
					S &= \frac12 ab\sin C \\
					&= \frac12\times3\times4\times\frac{\sqrt3}{2} \\
					&= 3\sqrt3.
			\end{aligned}
			\]
			\par\textbf{理由：}
			面积公式 $S=\dfrac12 ab\sin C$。

		\item[\textbf{第二步（求第三边 $c$）：}]
			\[
				\begin{aligned}
					c^2 &= a^2+b^2-2ab\cos C \\
					&= 3^2+4^2-2\times3\times4\times\frac12 \\
					&= 9+16-12 \\
					&= 13, \\
					c   &= \sqrt{13}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			余弦定理 $c^2=a^2+b^2-2ab\cos C$。

		\item[\textbf{第三步（求外接圆半径）：}]
			\[
				\begin{aligned}
					R &= \frac{abc}{4S} \\
					&= \frac{3\times4\times\sqrt{13}}{4\times3\sqrt3} \\
					&= \frac{\sqrt{13}}{\sqrt3} \\
					&= \frac{\sqrt{39}}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			利用 $R=\dfrac{abc}{4S}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=\dfrac{\sqrt{39}}{3}$}。
\end{examplebox}
