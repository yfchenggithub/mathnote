\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形三边长）：}}] 三角形三边长分别为 $a,b,c$。
		\item[\textbf{\textcolor{CStatement}{条件 2（外接圆半径）：}}] 三角形外接圆半径为 $R$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形面积等于三边乘积除以四倍外接圆半径。
			\[
				S = \frac{abc}{4R}.
			\]

		\item[\textbf{\textcolor{CStatement}{等价形式：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三边乘积等于四倍面积乘外接圆半径。
			\[
				abc = 4RS.
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{适用范围：}}
	任意三角形（锐角、直角、钝角均成立）。
\end{statementbox}
