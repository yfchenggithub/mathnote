\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{点到直线的距离可通过代入一般式系数并除以法向量的模计算。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（直线一般式）：} 直线方程必须为 $Ax+By+C=0$ 形式。
		\item \textbf{条件 2（点坐标）：} 点坐标 $P(x_0,y_0)$ 已知。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母开方）：} 不要忘记分母的平方根。
		\item \textbf{检查项（绝对值符号）：} 分子必须加绝对值，确保距离非负。
	\end{itemize}

\end{summarybox}
