\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l: 3x+4y-12=0$ 和点 $P(1,2)$，求点 $P$ 到直线 $l$ 的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出公式）：}]
			根据点到直线距离公式，$d = \frac{|Ax_0+By_0+C|}{\sqrt{A^2+B^2}}$。
			\par\textbf{理由：} 公式直接应用。
		\item[\textbf{第二步（代入数值）：}]
			将 $A=3, B=4, C=-12, x_0=1, y_0=2$ 代入：
			\[
				d = \frac{|3\times1+4\times2-12|}{\sqrt{3^2+4^2}}.
			\]
			\par\textbf{理由：} 代入时注意符号。
		\item[\textbf{第三步（化简得解）：}]
			计算分子 $|3+8-12| = |-1| = 1$，分母 $\sqrt{9+16}=5$，所以 $d = \frac{1}{5}$。
			\par\textbf{理由：} 绝对值与算术平方根计算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d = \frac{1}{5}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l: y = 2x + 1$ 和点 $Q(3,-1)$，求点 $Q$ 到直线 $l$ 的距离。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化一般式）：}]
			将直线方程化为一般式 $2x - y + 1 = 0$，此时 $A=2, B=-1, C=1$。
			\par\textbf{理由：} 只有一般式才能直接使用公式。
		\item[\textbf{第二步（代入公式）：}]
			代入 $x_0=3, y_0=-1$：
			\[
				d = \frac{|2\times3 + (-1)\times(-1) + 1|}{\sqrt{2^2+{(-1)}^2}}.
			\]
			\par\textbf{理由：} 公式代入，注意 $B$ 为负。
		\item[\textbf{第三步（化简得解）：}]
			计算分子 $|6+1+1|=8$，分母 $\sqrt{4+1}=\sqrt{5}$，所以 $d = \frac{8}{\sqrt{5}} = \frac{8\sqrt{5}}{5}$。
			\par\textbf{理由：} 分母有理化得到简化解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d = \frac{8\sqrt{5}}{5}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知点 $P(2,1)$ 到直线 $l: 4x+3y+C=0$ 的距离为 $1$，求常数 $C$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（建立方程）：}]
			利用距离公式，$d = \frac{|4\times2+3\times1+C|}{\sqrt{4^2+3^2}} = 1$。
			\par\textbf{理由：} 公式直接列方程。
		\item[\textbf{第二步（化简方程）：}]
			计算分母 $\sqrt{16+9}=5$，分子 $|8+3+C| = |11+C|$，得 $\frac{|11+C|}{5}=1$，即 $|11+C| = 5$。
			\par\textbf{理由：} 去分母。
		\item[\textbf{第三步（解绝对值方程）：}]
			解得 $11+C = 5$ 或 $11+C = -5$，即 $C = -6$ 或 $C = -16$。
			\par\textbf{理由：} 绝对值方程有两个解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$C = -6$ 或 $C = -16$。}
\end{examplebox}
