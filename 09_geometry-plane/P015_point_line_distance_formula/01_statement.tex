\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（直线一般式）：}}]
			给定直线方程为 $Ax+By+C=0$，其中 $A,B$ 不同时为零。
		\item[\textbf{\textcolor{CStatement}{条件 2（点坐标）：}}]
			给定点 $P(x_0,y_0)$ 为平面内任意一点。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（距离公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			点到直线的距离等于点坐标代入直线方程取绝对值后除以系数平方和的平方根。
			\[
				d = \frac{\lvert Ax_0+By_0+C \rvert}{\sqrt{A^2+B^2}}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（特殊直线情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当直线为竖直或水平时，公式退化为横纵坐标差的绝对值。
			\begin{itemize}
			\item 若 $B=0$，则直线为 $x = -\frac{C}{A}$，距离 $d = \left\lvert x_0 + \frac{C}{A} \right\rvert$；
			\item 若 $A=0$，则直线为 $y = -\frac{C}{B}$，距离 $d = \left\lvert y_0 + \frac{C}{B} \right\rvert$。
		\end{itemize}

	\end{description}

\end{statementbox}
