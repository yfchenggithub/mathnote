\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知三角形的三边长分别为 $3$, $4$, $5$，求其面积。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（计算半周长）：}]
			$p = \dfrac{3+4+5}{2} = 6$.\n
			\par\textbf{理由：} 半周长是三角形周长的一半，是海伦公式的中间量.\n
		\item[\textbf{第二步（代入公式）：}]
			\[
				\begin{aligned}
					S &= \sqrt{6 \times (6-3) \times (6-4) \times (6-5)} \\ &= \sqrt{6 \times 3 \times 2 \times 1} \\ &= \sqrt{36}
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 将 $p$ 和各边代入海伦公式.\n
		\item[\textbf{第三步（求结果）：}]
			$S = 6$.\n
			\par\textbf{理由：} 开平方得到面积数值.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{6}.\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 等腰三角形的底边长为 $6$，腰长为 $5$，求该三角形的面积。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（计算半周长）：}]
			$p = \dfrac{5+5+6}{2} = 8$.\n
			\par\textbf{理由：} 半周长等于周长一半.\n
		\item[\textbf{第二步（计算各个差值）：}]
			$p-a = 8-5 = 3$, $p-b = 3$, $p-c = 8-6 = 2$.\n
			\par\textbf{理由：} 分别将三边代入 $p-a$, $p-b$, $p-c$.\n
		\item[\textbf{第三步（代入公式）：}]
			\[
				\begin{aligned}
					S &= \sqrt{8 \times 3 \times 3 \times 2} \\ &= \sqrt{144} \\ &= 12
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 代入海伦公式并开平方.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{12}.\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 三角形的两条边长分别为 $5$ 和 $6$，第三边长 $c$ 是奇数，且三角形面积为 $6\sqrt{6}$，求 $c$ 的值。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（设半周长）：}]
			设 $p = \dfrac{5+6+c}{2} = \dfrac{11+c}{2}$.\n
			\par\textbf{理由：} 海伦公式中半周长用含 $c$ 的式子表示.\n
		\item[\textbf{第二步（代入面积平方）：}]
			由 $S=6\sqrt{6}$ 得 $S^2 = 216$. 代入海伦公式:\n
			\[
				\n 216 = p(p-5)(p-6)(p-c).\n
			\]
			\n 将 $p = \dfrac{11+c}{2}$, $p-5 = \dfrac{c+1}{2}$, $p-6 = \dfrac{c-1}{2}$, $p-c = \dfrac{11-c}{2}$ 代入,\n
			\[
				\n 216 = \dfrac{(11+c)(c+1)(c-1)(11-c)}{16}.\n
			\]
			\n
			\par\textbf{理由：} 分别计算各因子再合并分母.\n
		\item[\textbf{第三步（化简求解）：}]
			两边乘以 $16$:\n
			\[
				\n
				\begin{aligned}
					3456 &= (11+c)(11-c)(c+1)(c-1) \\ &= (121-c^2)(c^2-1).
			\end{aligned}
			\]
			\n 展开得 $-(c^2)^2 + 122c^2 - 121 = 3456$，整理为 $c^4 - 122c^2 + 3577 = 0$.\n 令 $x=c^2$，则 $x^2 - 122x + 3577 = 0$，解得 $x=49$ 或 $x=73$.\n 因此 $c=7$ 或 $c=\sqrt{73}$。由于 $c$ 为整数奇数，取 $c=7$.\n
			\par\textbf{理由：} 通过代数变换解关于 $c$ 的方程，再根据条件筛选.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$c=7$}（经验证满足三角形不等式）.\n
\end{examplebox}
