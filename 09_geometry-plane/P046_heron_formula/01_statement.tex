\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形不等式）}}]
		      三边 $a,b,c$ 为正数，且满足任意两边之和大于第三边：
		      $a+b>c$, $b+c>a$, $c+a>b$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（海伦公式）}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      用三边直接求三角形面积，避免作高。
		      \[
			      S = \sqrt{p(p-a)(p-b)(p-c)},
			      \qquad
			      p = \frac{a+b+c}{2}
		      \]

		\item[\textbf{\textcolor{CStatement}{常见变形（四次式形式）}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      利用三边平方值计算面积，无需显式写出半周长。
		      \[
			      S = \frac{1}{4}\sqrt{2(a^2b^2+b^2c^2+c^2a^2)-(a^4+b^4+c^4)}
		      \]
	\end{description}
\end{statementbox}
