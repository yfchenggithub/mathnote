	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形不等式）}}]
			三边 $a,b,c$ 满足任意两边之和大于第三边：$a+b>c$, $b+c>a$, $c+a>b$，且 $a,b,c>0$。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（海伦公式）}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			用三边直接求三角形面积，避免作高。\n
			\[
				\n S = \sqrt{p(p-a)(p-b)(p-c)},
				\qquad
				p = \frac{a+b+c}{2}\n
			\]
			。\n\n
		\item[\textbf{\textcolor{CStatement}{常见变形（平方差公式）}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			利用三边平方值计算面积，无需半周长。\n
			\[
				\n S = \frac{1}{4}\sqrt{2(a^2 b^2 + b^2 c^2 + c^2 a^2) - (a^4 + b^4 + c^4)}\n
			\]
			。\n
	\end{description}
	\n
\end{statementbox}
