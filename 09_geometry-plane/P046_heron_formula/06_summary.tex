\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{用半周长统一三边信息，直接计算三角形面积}.\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件 1（三边已知）：} 需已知三角形三条边的具体数值.\n
		\item \textbf{条件 2（三角形成立）：} 三边必须满足两边之和大于第三边.\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（半周长因子）：} 半周长 $p$ 是周长的一半；四个因子 $p$, $p-a$, $p-b$, $p-c$ 缺一不可.\n
		\item \textbf{检查项（验证不等式）：} 代入根式前先验证三角形不等式，确保平方根内非负.\n
	\end{itemize}
	\n
\end{summarybox}
