	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（三角不等式忽略）}}]
			未检查三边能否构成三角形，直接代入公式导致根号内为负。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（先验证不等式）：}}]
			计算前必须验证 $a+b>c$, $b+c>a$, $c+a>b$，否则公式无效。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视条件隐含）：}}]
			海伦公式建立在三角形存在前提下，忽略此条件将得出无意义的虚数。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（半周长误算）}}]
			将半周长 $p$ 错误地取为周长 $a+b+c$，而不是周长的一半。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（半周长定义）：}}]
			海伦公式中 $p = \dfrac{a+b+c}{2}$，是周长的一半。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（公式记忆粗心）：}}]
			学生常将半周长与周长混淆，导致后续乘积扩大而错误。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（根号因子遗漏）}}]
			在乘积 $\sqrt{p(p-a)(p-b)(p-c)}$ 中漏掉某一个因子，例如只乘三项。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（对称乘积结构）：}}]
			海伦公式是四个因子的乘积再开方，每个因子对应一条边与半周长的差。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（结构认识不足）：}}]
			未理解公式的对称性，漏乘导致面积计算错误。\n
	\end{description}
