\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（三角不等式忽略）}}]
		      未检查三边能否构成三角形，直接代入公式，可能导致根号内不为正。

		\item[\textbf{\textcolor{CTrap}{正确理解（先验证不等式）：}}]
		      计算前必须验证 $a+b>c$, $b+c>a$, $c+a>b$，否则公式不能表示正常三角形面积。

		\item[\textbf{\textcolor{CTrap}{错因分析（忽视条件隐含）：}}]
		      海伦公式建立在非退化三角形存在的前提下；若三边不成三角形，代入结果没有正常面积意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（半周长误算）}}]
		      将半周长 $p$ 错误地取为周长 $a+b+c$，而不是周长的一半。

		\item[\textbf{\textcolor{CTrap}{正确理解（半周长定义）：}}]
		      海伦公式中 $p = \dfrac{a+b+c}{2}$，是周长的一半。

		\item[\textbf{\textcolor{CTrap}{错因分析（公式记忆粗心）：}}]
		      学生常将半周长与周长混淆，导致后续乘积数值错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（根号因子遗漏）}}]
		      在乘积 $\sqrt{p(p-a)(p-b)(p-c)}$ 中漏掉某一个因子，例如只乘三项。

		\item[\textbf{\textcolor{CTrap}{正确理解（对称乘积结构）：}}]
		      海伦公式是四个因子的乘积再开方：$p$, $p-a$, $p-b$, $p-c$，缺一不可。

		\item[\textbf{\textcolor{CTrap}{错因分析（结构认识不足）：}}]
		      未理解公式的对称性，漏乘会导致面积计算错误。
	\end{description}
\end{trapbox}
