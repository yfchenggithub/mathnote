\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在四边形$ABCD$中，$\angle BAD = 110^{\circ}$，$\angle BCD = 70^{\circ}$，且$A,B,C,D$不共线。判断$A,B,C,D$是否共圆，并说明理由。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（已知角度）：}] 确定已知对角。
			\[
				\angle BAD = 110^{\circ},\quad \angle BCD = 70^{\circ}.
			\]
			\par\textbf{理由：} 题目直接给出。

		\item[\textbf{第二步（计算和）：}] 求解对角和。
			\[
				\begin{aligned}
					\angle BAD + \angle BCD &= 110^{\circ} + 70^{\circ} \\
					&= 180^{\circ}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代数和运算。

		\item[\textbf{第三步（应用判定）：}] 应用对角互补判定。
			\[
				\angle BAD + \angle BCD = 180^{\circ}.
			\]
			\par\textbf{理由：} 因为四边形一组对角互补，且四点不共线，所以$A,B,C,D$共圆。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$A,B,C,D$四点共圆。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知点$C,D$在线段$AB$同侧，且$\angle ACB = \angle ADB = 45^{\circ}$。判断$A,B,C,D$是否共圆。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等角条件）：}] 提取已知角度。
			\[
				\angle ACB = 45^{\circ},\quad \angle ADB = 45^{\circ}.
			\]
			\par\textbf{理由：} 题目条件。

		\item[\textbf{第二步（同侧等角）：}] 确认同侧且角相等等条件。
			\[
				\angle ACB = \angle ADB.
			\]
			\par\textbf{理由：} 由已知$C,D$在$AB$同侧，且角度相等。

		\item[\textbf{第三步（应用判定）：}] 应用同侧等角判定。
			\[
				\angle ACB = \angle ADB.
			\]
			\par\textbf{理由：} 若两点在线段同侧且视角相等，则四点共圆。因此$A,B,C,D$共圆。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$A,B,C,D$四点共圆。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在凸四边形$ABCD$中，对角线$AC$与$BD$交于点$P$，且$PA=4, PB=5, PC=2, PD=10$。判断$A,B,C,D$是否共圆。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算乘积）：}] 计算两线段乘积。
			\[
				\begin{aligned}
					PA\cdot PB &= 4 \times 5 \\
					&= 20, \\
					PC\cdot PD &= 2 \times 10 \\
					&= 20.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术运算。

		\item[\textbf{第二步（比较乘积）：}] 比较乘积是否相等。
			\[
				PA\cdot PB = PC\cdot PD.
			\]
			\par\textbf{理由：} 计算结果表明相等。

		\item[\textbf{第三步（应用判定）：}] 应用相交弦逆定理。
			\[
				PA\cdot PB = PC\cdot PD.
			\]
			\par\textbf{理由：} 因为点$P$在线段$AC,BD$内部（凸四边形对角线交点），满足相交弦逆定理条件，所以$A,B,C,D$四点共圆。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$A,B,C,D$四点共圆。}
\end{examplebox}
