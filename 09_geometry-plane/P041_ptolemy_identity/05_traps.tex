\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（边序混乱）}}] 将四边形边任意配对，例如用 $AB$ 与 $BC$ 作对边，得到错误乘积和。
		\item[\textbf{\textcolor{CTrap}{正确理解（顺序确定）：}}] 对边必须按圆上顺序：$AB$ 与 $CD$，$BC$ 与 $DA$。
		\item[\textbf{\textcolor{CTrap}{错因分析（配对错误）：}}] 未注意顶点顺序，导致对边识别错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略共圆）}}] 在任意四边形中直接使用定理，得到错误结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（前提检验）：}}] 必须先确认或说明四边形内接于圆。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）：}}] 忽视定理适用前提，导致误用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式混淆）}}] 记错公式形式，如 $ef=ab+cd$ 或 $ef=ad+bc$。
		\item[\textbf{\textcolor{CTrap}{正确理解（准确记忆）：}}] 公式为 $ef=ac+bd$，其中 $a,c$ 是一组对边，$b,d$ 是另一组。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号混淆）：}}] 未能正确记忆边与对角线的对应关系。
	\end{description}
\end{trapbox}
