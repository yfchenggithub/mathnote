\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	判断直线 $L_1: 2x-3y+4=0$ 与 $L_2: 4x-6y+7=0$ 的位置关系.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算交叉积）：}]
			计算 $A_1B_2$ 与 $A_2B_1$:
			\[
				A_1B_2 = 2\times(-6) = -12,
				\qquad
				A_2B_1 = 4\times(-3) = -12,
			\]
			故 $A_1B_2 = A_2B_1$.
			\par\textbf{理由：} 这是平行的必要条件.
		\item[\textbf{第二步（常数项检验）：}]
			计算 $A_1C_2$ 与 $A_2C_1$:
			\[
				A_1C_2 = 2\times7 = 14,
				\qquad
				A_2C_1 = 4\times4 = 16,
			\]
			故 $A_1C_2 \neq A_2C_1$.
			\par\textbf{理由：} 区分平行与重合的关键.
		\item[\textbf{第三步（下结论）：}]
			由平行判定定理，两直线平行（不重合）.
			\par\textbf{理由：} 条件 $A_1B_2 = A_2B_1$ 且 $A_1C_2\neq A_2C_1$ 成立.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$L_1\parallel L_2$.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两直线 $L_1: 2x+(a-1)y-3=0$ 与 $L_2: (a+1)x+4y+1=0$ 平行，求实数 $a$ 的值，并判断是否重合.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列条件）：}]
			由 $A_1B_2 = A_2B_1$ 得
			\[
				2\times 4 = (a-1)(a+1) \;\Longrightarrow\; 8 = a^2 - 1.
			\]
			\par\textbf{理由：} 平行必要条件.
		\item[\textbf{第二步（解方程）：}]
			由 $a^2 = 9$ 得 $a = 3$ 或 $a = -3$.
			\par\textbf{理由：} 解得可能平行时的参数.
		\item[\textbf{第三步（常数项检查）：}]
			分别计算检查:
			\begin{itemize}
			\item 当 $a=3$ 时，$A_1C_2 = 2\times1 = 2$, $A_2C_1 = 4\times(-3) = -12$, 不相等;
			\item 当 $a=-3$ 时，$A_1C_2 = 2\times1 = 2$, $A_2C_1 = (-2)\times(-3) = 6$, 不相等.
		\end{itemize}
			故两直线平行且不重合.
			\par\textbf{理由：} $A_1C_2\neq A_2C_1$ 说明不重合.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a=3$ 或 $a=-3$，两直线平行.

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $L_1: (m+2)x + (m^2-4)y + 2 = 0$ 与 $L_2: 3x + (m-2)y - 1 = 0$ 平行，求实数 $m$ 的值.
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列条件）：}]
			由 $A_1B_2 = A_2B_1$ 得
			\[
				(m+2)(m-2) = 3(m^2-4).
			\]
			\par\textbf{理由：} 平行必要条件.
		\item[\textbf{第二步（解方程）：}]
			整理得
			\[
				(m+2)(m-2) = 3(m+2)(m-2) \;\Longrightarrow\; (m+2)(m-2) - 3(m+2)(m-2)=0,
			\]
			即 $-2(m+2)(m-2)=0$，故 $m=2$ 或 $m=-2$.
			\par\textbf{理由：} 不能直接约去 $(m+2)(m-2)$，否则会丢解.
		\item[\textbf{第三步（检验退化）：}]
			当 $m=2$ 时，$A_1=4$, $B_1=0$, 系数非零，且 $A_1C_2=4\times(-1)=-4$, $A_2C_1=3\times2=6$ 不相等，平行；当 $m=-2$ 时，$A_1=0$, $B_1=0$, 不满足题设（系数应不同时为零），舍去.
			\par\textbf{理由：} 直线方程必须满足 $(A_1,B_1)$ 与 $(A_2,B_2)$ 不同时为零.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$m=2$，两直线平行且不重合.
\end{examplebox}
