\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（一般式形式）：}}]
			两条直线方程为 $L_1: A_1x+B_1y+C_1=0$, $L_2: A_2x+B_2y+C_2=0$, 其中 $(A_1,B_1)$ 与 $(A_2,B_2)$ 不同时为零.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平行判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对应系数成比例但常数项不成比例.
			\[
				L_1\parallel L_2 \iff A_1B_2 = A_2B_1 \;\text{且}\; \bigl( A_1C_2 \neq A_2C_1 \;\text{ 或 }\; B_1C_2 \neq B_2C_1 \bigr).
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（重合判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对应系数全部成比例.
			\[
				L_1 \text{与} L_2 \text{重合} \iff A_1B_2 = A_2B_1 \;\text{且}\; A_1C_2 = A_2C_1 \;\text{且}\; B_1C_2 = B_2C_1.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（平行直线系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			保持 $A_1,B_1$ 系数不变, 改变常数项 $C$ 且 $C\neq C_1$ 即得平行直线.
			\[
				\text{与 }L_1:A_1x+B_1y+C_1=0\text{ 平行的直线方程为 }A_1x+B_1y+C=0,\ C\neq C_1.
			\]
	\end{description}

\end{statementbox}
