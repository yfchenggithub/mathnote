\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略重合）}}]
			只验证 $A_1B_2=A_2B_1$ 就判定平行，未检查常数项不成比例（即 $A_1C_2\neq A_2C_1$ 或 $B_1C_2\neq B_2C_1$）。
		\item[\textbf{\textcolor{CTrap}{正确理解（常数项检验）：}}]
			必须同时验证常数项不成比例（即 $A_1C_2\neq A_2C_1$ 或 $B_1C_2\neq B_2C_1$）。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			未理解平行与重合的区别在于常数项比例。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略退化情形）}}]
			在求解含参直线平行时，忽略系数 $(A_1,B_1)$ 或 $(A_2,B_2)$ 可能同时为零（方程退化），未舍去无效参数。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数非零条件）：}}]
			直线方程必须满足系数不同时为零；若出现零系数需单独判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}]
			只关注代数等式，忽略了几何意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（非一般式误用）}}]
			直接对非一般式（如斜截式）使用系数条件，而未转化为一般式。
		\item[\textbf{\textcolor{CTrap}{正确理解（化为一般式）：}}]
			把直线方程化为标准一般式 $A_i x+B_i y+C_i=0$ 后再使用结论。
		\item[\textbf{\textcolor{CTrap}{错因分析（形式不匹配）：}}]
			结论只适用于一般式形式。
	\end{description}

\end{trapbox}
