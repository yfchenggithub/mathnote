\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆外一点到圆的两条切线长相等，且圆心与该点的连线平分切线的夹角并垂直平分切点弦。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（圆外一点切线）：} 点 $P$ 必须在圆 $O$ 外部，且过 $P$ 作圆的两条切线。
		\item \textbf{条件 2（已知两量求一量）：} 已知 $d$ 和 $r$ 可求切线长；已知 $l$ 和 $r$ 可求 $d$；反之亦然。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式计算顺序）：} 计算切线长时先算平方差再开方，避免跳步。
		\item \textbf{检查项（垂直平分兼备）：} 使用垂直平分性质时，必须同时检查垂直和平分两个条件。
	\end{itemize}
\end{summarybox}
