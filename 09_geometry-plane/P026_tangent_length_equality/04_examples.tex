\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $O$ 的半径为 $3$，圆外一点 $P$ 到圆心 $O$ 的距离 $OP=5$，过 $P$ 作圆 $O$ 的两条切线，切点分别为 $A$、$B$，求切线长 $PA$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别已知量）：}] 已知 $r=3,\; d=5$。
			\par\textbf{理由：} 直接给出。
		\item[\textbf{第二步（代入公式）：}] 由切线长定理
			\[
				\begin{aligned}
					PA &= \sqrt{d^{2} - r^{2}} \\
					&= \sqrt{5^{2} - 3^{2}} \\
					&= \sqrt{25 - 9} \\
					&= \sqrt{16} \\
					&= 4 .
			\end{aligned}
			\]
			\par\textbf{理由：} 切线长公式。
		\item[\textbf{第三步（写出答案）：}] $PA = 4$，且 $PB = PA = 4$。
			\par\textbf{理由：} 切线长相等。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直接套用 $l = \sqrt{d^{2} - r^{2}}$ 即可得切线长。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	从圆 $O$ 外一点 $P$ 引圆的两条切线，切点 $A$、$B$。已知切线长 $PA=4$，圆半径 $r=3$，求点 $P$ 到圆心 $O$ 的距离 $d$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定已知量）：}] 已知 $l = PA = 4$，$r=3$。
			\par\textbf{理由：} 题目给出。
		\item[\textbf{第二步（逆用公式）：}] 由 $l^{2} = d^{2} - r^{2}$ 得
			\[
				\begin{aligned}
					d^{2} &= l^{2} + r^{2} \\
					&= 4^{2} + 3^{2} \\
					&= 16 + 9 \\
					&= 25 .
			\end{aligned}
			\]
			\par\textbf{理由：} 公式变形，移项。
		\item[\textbf{第三步（求距离）：}] $d = \sqrt{25} = 5$（距离取正值）。
			\par\textbf{理由：} 开方得距离。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d = \sqrt{l^{2} + r^{2}}$，与求切线长是互逆运算。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$PA$、$PB$ 是 $\odot O$ 的两条切线，切点分别为 $A$、$B$。过圆上另一点 $C$ 作 $\odot O$ 的切线，分别交 $PA$ 于点 $D$，交 $PB$ 于点 $E$。已知 $PA=5$，求 $\triangle PDE$ 的周长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（标记相等线段）：}] 由切线长定理得：$PA = PB = 5$，$DA = DC$，$EB = EC$。
			\par\textbf{理由：} 从同一点引圆的两条切线切线长相等。
		\item[\textbf{第二步（转化周长）：}] 由第一步的等量关系，得
			\[
				\begin{aligned}
					C_{\triangle PDE} &= PD + DE + PE \\
					&= (PD+DA)+(EB+PE) \\
					&= PA + PB \\
					&= 2PA .
			\end{aligned}
			\]
			\par\textbf{理由：} 利用等线段替换。
		\item[\textbf{第三步（代入数值）：}] 周长 $= 2 \times 5 = 10$。
			\par\textbf{理由：} $PA=5$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{利用切线长定理进行等线段转化，是解决切线周长问题的通用方法。}
\end{examplebox}
