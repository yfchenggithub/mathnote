\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式混淆）：}}] 错误地将切线长记为 $l = d - r$ 或 $l = \sqrt{d^{2} + r^{2}}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（平方差再开方）：}}] 切线长 $l = \sqrt{d^{2} - r^{2}}$，必须先计算 $d^{2} - r^{2}$ 再开平方。
		\item[\textbf{\textcolor{CTrap}{错因分析（勾股位置颠倒）：}}] 误把 $OP$ 当成直角边或 $r$ 当成斜边，忘记了 $l$ 是直角边。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（角平分对象错误）：}}] 误以为 $OP$ 平分 $\angle AOB$（圆心角），而不是 $\angle APB$。
		\item[\textbf{\textcolor{CTrap}{正确理解（角平分位置）：}}] $OP$ 平分的是两条切线的夹角 $\angle APB$，即 $\angle APO = \angle BPO$。
		\item[\textbf{\textcolor{CTrap}{错因分析（全等角对应错位）：}}] 由全等得到 $\angle AOP = \angle BOP$，学生错误地将其与角平分线联系，而忽略了目标角是 $\angle APB$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（垂直平分条件缺失）：}}] 只记住 $OP \perp AB$ 而忘记 $OP$ 平分 $AB$，或反之。
		\item[\textbf{\textcolor{CTrap}{正确理解（垂直平分完整）：}}] $OP$ 既垂直于 $AB$，又平分 $AB$（交点为 $AB$ 中点）。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆碎片化）：}}] 学生对性质记忆不全，将两个条件割裂，未统一为“垂直平分”。
	\end{description}
\end{trapbox}
