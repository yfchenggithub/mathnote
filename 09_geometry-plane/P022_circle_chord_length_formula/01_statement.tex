	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（半径和弦心距）：}}] 设圆的半径为 $r$，圆心到弦的垂线距离（弦心距）为 $d$，且 $0 \le d < r$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（弦长公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知半径和弦心距，直接计算弦的长度。
			\[
				L = 2\sqrt{r^{2} - d^{2}}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（弦心距公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知半径和弦长，反求弦心距。
			\[
				d = \sqrt{r^{2} - \left(\frac{L}{2}\right)^{2}}.
			\]
	\end{description}

\end{statementbox}
