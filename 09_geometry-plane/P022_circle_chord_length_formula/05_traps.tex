\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（漏乘系数2）}}] 计算弦长时，只算得 $\sqrt{r^{2}-d^{2}}$ 即停下，忘记乘以 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（弦长公式完整）：}}] 公式是 $L=2\sqrt{r^{2}-d^{2}}$，其中 $\sqrt{r^{2}-d^{2}}$ 是半弦长。
		\item[\textbf{\textcolor{CTrap}{错因分析（半弦长概念模糊）：}}] 推导中先得到半弦长，学生误以为就是弦长。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略范围条件）}}] 当 $d\ge r$ 时，仍然使用 $L=2\sqrt{r^{2}-d^{2}}$，导致根号内为负。
		\item[\textbf{\textcolor{CTrap}{正确理解（弦心距小于半径）：}}] 公式前提是 $0\le d<r$（相交或过圆心），$d=r$ 时相切 $L=0$，$d>r$ 时无公共点。
		\item[\textbf{\textcolor{CTrap}{错因分析（未检验几何条件）：}}] 直接套代数公式，忽视公式成立的范围。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（逆用误用整弦长）}}] 已知 $r$ 和 $L$ 求 $d$ 时，错误地写为 $d=\sqrt{r^{2}-L^{2}}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（使用半弦长）：}}] 正确关系是 $d=\sqrt{r^{2}-\left(\frac{L}{2}\right)^{2}}$，因为 $r^{2}=d^{2}+\left(\frac{L}{2}\right)^{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆半弦与整弦）：}}] 未注意直角三角形中直角边是半弦长而非整弦长。
	\end{description}
\end{trapbox}
