\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件忽略）}}]
			在等边三角形中误用比例$OG:GH=1:2$，认为三点不重合，导致概念错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（退化情形）：}}]
			当三角形为等边时，$O$、$G$、$H$、$N$四点重合，欧拉线退化为一点，此时比例式$1:2$无实际几何意义，但向量式$\overrightarrow{OH}=3\overrightarrow{OG}$仍成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗忘）：}}]
			忽视定理前提“非等边”，将一般结论直接套用于特殊三角形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（比例颠倒）}}]
			误以为$\overrightarrow{OG}:\overrightarrow{OH}=1:2$或$\overrightarrow{OH}:\overrightarrow{OG}=2:1$，导致求解点坐标时系数错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确比例）：}}]
			正确关系为$\overrightarrow{OH}=3\overrightarrow{OG}$，即$OG:GH=1:2$（$OG$占$1$份，$GH$占$2$份，$OH$共$3$份）。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆混淆）：}}]
			未通过向量推导强化理解，仅死记比例导致系数张冠李戴。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（中点误判）}}]
			误将九点圆心$N$当作$OH$的三等分点或与重心$G$重合。
		\item[\textbf{\textcolor{CTrap}{正确理解（$N$为中点）：}}]
			九点圆心$N$是线段$OH$的中点，即$\overrightarrow{ON}=\frac12\overrightarrow{OH}$，$N$不同于$G$，$OG:GN:NH=1:?$但$G$并非$OH$中点。
		\item[\textbf{\textcolor{CTrap}{错因分析（知识缺陷）：}}]
			对九点圆的定义不熟悉，将$N$与其他心混淆。
	\end{description}

\end{trapbox}
