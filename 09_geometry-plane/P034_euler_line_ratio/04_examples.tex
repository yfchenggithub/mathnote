\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知$\triangle ABC$的外心$O(0,0)$，垂心$H(3,3)$，且$\triangle ABC$为非等边三角形，求重心$G$的坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（向量关系）：}]
			由欧拉线定理，$\overrightarrow{OH}=3\overrightarrow{OG}$，故$\overrightarrow{OG}=\frac13\overrightarrow{OH}$。
			\par\textbf{理由：} 直接套用定理比例式。
		\item[\textbf{第二步（坐标运算）：}]
			$\overrightarrow{OH}=(3,3)$，所以$\overrightarrow{OG}=(1,1)$，因此$G$坐标为$(1,1)$。
			\par\textbf{理由：} 向量坐标运算。
		\item[\textbf{第三步（条件确认）：}]
			题目已说明非等边，故定理适用。
			\par\textbf{理由：} 若等边则退化为一点，但本题目中$O\neq H$，故非等边。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{重心$G(1,1)$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，已知$A(-1,0),B(1,0),C(0,2)$，求$\triangle ABC$的外心$O$和重心$G$的坐标，并利用欧拉线定理求垂心$H$的坐标（假设三角形非等边）。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（重心坐标）：}]
			$G\left(\frac{-1+1+0}{3},\frac{0+0+2}{3}\right)=\left(0,\frac23\right)$。
			\par\textbf{理由：} 重心坐标公式。
		\item[\textbf{第二步（外心坐标）：}]
			设$O(0,y)$，因为$OA=OC$得$\sqrt{1+y^2}=\sqrt{(y-2)^2}$，两边平方得$1+y^2=y^2-4y+4$，解得$y=\frac34$，故$O\left(0,\frac34\right)$。
			\par\textbf{理由：} 外心到顶点距离相等，考虑对称性设坐标。
		\item[\textbf{第三步（应用欧拉线）：}]
			由欧拉线定理，$\overrightarrow{OH}=3\overrightarrow{OG}$，$\overrightarrow{OG}=\left(0,\frac23-\frac34\right)=\left(0,-\frac1{12}\right)$，则$3\overrightarrow{OG}=\left(0,-\frac14\right)$，所以$H=O+3\overrightarrow{OG}=\left(0,\frac34\right)+\left(0,-\frac14\right)=\left(0,\frac12\right)$。
			\par\textbf{理由：} 向量等式坐标运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{外心$O\left(0,\frac34\right)$，重心$G\left(0,\frac23\right)$，垂心$H\left(0,\frac12\right)$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知$\triangle ABC$的重心$G(2,1)$，垂心$H(4,3)$，且$\triangle ABC$为非等边三角形，求外心$O$的坐标以及欧拉线的方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设外心坐标）：}]
			设$O(x,y)$。
			\par\textbf{理由：} 未知数待定。
		\item[\textbf{第二步（列向量方程）：}]
			由欧拉线定理$\overrightarrow{OH}=3\overrightarrow{OG}$，即$(4-x,3-y)=3(2-x,1-y)$。
			\par\textbf{理由：} 向量比例关系。
		\item[\textbf{第三步（解方程组）：}]
			$4-x=6-3x$，$3-y=3-3y$，解得$x=1$，$y=0$，故$O(1,0)$。
			\par\textbf{理由：} 解线性方程。
		\item[\textbf{第四步（求欧拉线方程）：}]
			由$O(1,0),G(2,1)$得斜率$k=1$，直线方程$y=x-1$。
			\par\textbf{理由：} 两点确定一条直线。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{外心$O(1,0)$，欧拉线方程$y=x-1$。}
\end{examplebox}
