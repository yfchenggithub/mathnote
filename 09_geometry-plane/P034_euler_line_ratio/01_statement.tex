\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（非等边三角形）：}}]
			$\triangle ABC$为非等边三角形。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（四点共线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			重心$G$、外心$O$、垂心$H$、九点圆心$N$位于同一条直线上（称为欧拉线）。
			\[
				O,G,H,N
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（比例关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外心$O$到重心$G$的距离与重心$G$到垂心$H$的距离之比为$1:2$。
			\[
				\overrightarrow{OH}=3\overrightarrow{OG}
			\]
			\[
				OG:GH=1:2
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（中点关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			九点圆圆心$N$是线段$OH$的中点。
			\[
				\overrightarrow{ON}=\frac12\overrightarrow{OH}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{推广结论（等边退化）}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	当$\triangle ABC$为等边三角形时，点$O$、$G$、$H$、$N$重合，欧拉线退化为一点。
	\[
		O=G=H=N
	\]
\end{statementbox}
