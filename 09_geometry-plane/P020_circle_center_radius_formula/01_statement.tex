	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（方程规范）：}}]
			方程必须为 $x^2+y^2+Dx+Ey+F=0$，即 $x^2,y^2$ 系数为 $1$，且不含 $xy$ 项。
		\item[\textbf{\textcolor{CStatement}{条件2（判别式正）：}}]
			判别式满足 $D^2+E^2-4F>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{结论一（圆方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			方程表示圆心在 $(-\frac{D}{2},-\frac{E}{2})$、半径为 $\frac12\sqrt{D^2+E^2-4F}$ 的圆。
			\[
				\left(-\frac{D}{2},-\frac{E}{2}\right),\quad r=\frac12\sqrt{D^2+E^2-4F}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（配方形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过配方可得标准形式，右边符号决定图形类型。
			\[
				\left(x+\frac{D}{2}\right)^2+\left(y+\frac{E}{2}\right)^2=\frac{D^2+E^2-4F}{4}.
			\]
	\end{description}

\end{statementbox}
