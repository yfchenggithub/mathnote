\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	一般方程 $x^2+y^2+Dx+Ey+F=0$ 通过配方转化为标准式 $(x+\frac{D}{2})^2+(y+\frac{E}{2})^2 = \frac{D^2+E^2-4F}{4}$，当 $D^2+E^2-4F>0$ 时表示圆，圆心 $(-\frac{D}{2},-\frac{E}{2})$，半径 $r=\frac12\sqrt{D^2+E^2-4F}$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（系数归一）：} $x^2,y^2$ 系数必须为 $1$，且不含 $xy$ 项。
		\item \textbf{条件 2（判别式正）：} $D^2+E^2-4F > 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（系数先化简）：} 若 $x^2,y^2$ 系数不是 $1$，需先除以系数化为 $1$。
		\item \textbf{检查项（半径公式）：} 半径 $r=\frac12\sqrt{D^2+E^2-4F}$，切勿忽略 $\frac12$。
	\end{itemize}

\end{summarybox}
