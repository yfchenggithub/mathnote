\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（系数未归一）}}]
			直接对 $x^2,y^2$ 系数不为 $1$ 的方程套用圆心半径公式，结果错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（先化系数）：}}]
			应先将 $x^2,y^2$ 系数化为 $1$，再使用公式。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式前提）：}}]
			公式只适用于 $x^2,y^2$ 系数为 $1$ 且无 $xy$ 项的标准形式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（半径漏除二）}}]
			错误将半径记为 $r=\sqrt{D^2+E^2-4F}$，忘了除以 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（半径公式）：}}]
			正确半径 $r=\frac12\sqrt{D^2+E^2-4F}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（配方系数）：}}]
			配方后右边为 $\frac{D^2+E^2-4F}{4}$，开方得 $\frac12\sqrt{D^2+E^2-4F}$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（点圆混淆）}}]
			误以为 $D^2+E^2-4F\ge0$ 就是圆，包含等于 $0$ 情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（严格大于零）：}}]
			表示圆必须 $D^2+E^2-4F>0$；等于 $0$ 表示一个点（点圆）。
		\item[\textbf{\textcolor{CTrap}{错因分析（半径为零）：}}]
			半径必须大于 $0$ 才是圆，等于 $0$ 是退化情形。
	\end{description}

\end{trapbox}
