\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 判断方程 $x^2+y^2-6x+4y-12=0$ 是否表示圆，若是，求圆心和半径。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			方程是 $x^2+y^2+Dx+Ey+F=0$ 形式，其中 $D=-6$, $E=4$, $F=-12$，满足系数为1且无xy项。
			\par\textbf{理由：} 对照一般式直接确定参数。
		\item[\textbf{第二步（计算判别式）：}]
			计算
			\[
				\begin{aligned}
					D^2+E^2-4F &= (-6)^2+4^2-4(-12) \\ &= 36+16+48 \\ &= 100 > 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 判别式大于0是表示圆的充要条件。
		\item[\textbf{第三步（求圆心半径）：}]
			圆心 $(-\frac{D}{2},-\frac{E}{2}) = (3,-2)$，半径 $r=\frac12\sqrt{100}=5$。
			\par\textbf{理由：} 套用圆心半径公式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{该方程表示圆心为 $(3,-2)$、半径为 $5$ 的圆。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（变形）}}
	\textbf{\textcolor{CExample}{题目：}} 已知方程 $x^2+y^2-2x+4y+F=0$ 表示圆，求实数 $F$ 的取值范围。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}]
			$D=-2$, $E=4$, $F$ 为参数。
			\par\textbf{理由：} 对照一般式。
		\item[\textbf{第二步（写判别式）：}]
			\[
				\begin{aligned}
					D^2+E^2-4F &= (-2)^2+4^2-4F \\ &= 4+16-4F \\ &= 20-4F.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入公式。
		\item[\textbf{第三步（解不等式）：}]
			要求 $20-4F > 0$，解得 $F < 5$。
			\par\textbf{理由：} 圆的条件是判别式大于0。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{ $F$ 的取值范围是 $F < 5$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 已知圆经过点 $A(1,2)$, $B(3,4)$，且圆心在直线 $x-y=0$ 上，求圆的方程。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设一般式）：}]
			设圆的方程为 $x^2+y^2+Dx+Ey+F=0$。
			\par\textbf{理由：} 一般式含三个参数，适合待定。
		\item[\textbf{第二步（代入点坐标）：}]
			将 $A(1,2)$、$B(3,4)$ 代入得
			\[
				1^2+2^2 + D\cdot1 + E\cdot2 + F = 0,
				\qquad
				3^2+4^2 + D\cdot3 + E\cdot4 + F = 0,
			\]
			即
			\[
				5 + D + 2E + F = 0,
				\qquad
				25 + 3D + 4E + F = 0.
			\]
			\par\textbf{理由：} 点在圆上满足方程。
		\item[\textbf{第三步（圆心条件）：}]
			圆心 $(-\frac{D}{2},-\frac{E}{2})$ 在直线 $x-y=0$ 上，故 $-\frac{D}{2} - (-\frac{E}{2}) = 0$，化简得 $E = D$。
			\par\textbf{理由：} 圆心坐标代入直线方程。
		\item[\textbf{第四步（解方程组）：}]
			将 $E = D$ 代入得
			\[
				\begin{aligned}
					5 + D + 2D + F &= 0 \;\Rightarrow\; 5 + 3D + F = 0,\\ 25 + 3D + 4D + F &= 0 \;\Rightarrow\; 25 + 7D + F = 0.
			\end{aligned}
			\]
			两式相减得 $20 + 4D = 0$，所以 $D = -5$，$E = -5$，代入得 $F = -5-3D = -5-3(-5)=10$。
			\par\textbf{理由：} 解线性方程组。
		\item[\textbf{第五步（验证条件）：}]
			检验判别式：
			\[
				\begin{aligned}
					D^2+E^2-4F &= (-5)^2+(-5)^2-4\cdot10 \\ &= 25+25-40 \\ &= 10 > 0,
			\end{aligned}
			\]
			满足圆的条件。
			\par\textbf{理由：} 确保方程确实表示圆。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{所求圆的方程为 $x^2+y^2-5x-5y+10=0$。}
\end{examplebox}
