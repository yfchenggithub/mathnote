\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两直线相交）：}}]
			已知两条直线 $l_1: A_1x+B_1y+C_1=0$ 与 $l_2: A_2x+B_2y+C_2=0$ 满足 $A_1B_2 \neq A_2B_1$（即 $l_1$ 与 $l_2$ 相交）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（单参数表达）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过两直线交点的任意直线（除 $l_2$ 外）均可写成 $l_1$ 方程加上 $\lambda$ 倍的 $l_2$ 方程。
			\[
				A_1x+B_1y+C_1 + \lambda (A_2x+B_2y+C_2)=0,
				\qquad
				\lambda \in \mathbb{R}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（双参数形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			使用两个参数可表示所有过交点的直线（包括 $l_2$ 本身）。
			\[
				m(A_1x+B_1y+C_1) + n(A_2x+B_2y+C_2)=0,
				\qquad
				m,n\in\mathbb{R},\;(m,n)\neq(0,0).
			\]
	\end{description}

\end{statementbox}
