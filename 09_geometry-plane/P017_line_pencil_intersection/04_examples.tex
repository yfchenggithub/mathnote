\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l_1: x+y-1=0$，$l_2: 2x-y+4=0$，求过 $l_1$ 与 $l_2$ 交点且与直线 $x-2y+3=0$ 垂直的直线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设直线系）：}]
			设所求直线方程为 $x+y-1+\lambda(2x-y+4)=0$。
			\par\textbf{理由：} 过两相交直线交点的直线可用 $l_1+\lambda l_2=0$ 表示（$l_2$ 除外）。
		\item[\textbf{第二步（列垂直条件）：}]
			整理得 $(1+2\lambda)x+(1-\lambda)y+(-1+4\lambda)=0$，斜率 $k=-\dfrac{1+2\lambda}{1-\lambda}$（$1-\lambda\neq0$）。已知直线 $x-2y+3=0$ 的斜率为 $\dfrac12$，垂直条件要求 $k\cdot\dfrac12=-1$，即 $k=-2$。
			\par\textbf{理由：} 两直线垂直的充要条件是斜率乘积为 $-1$。
		\item[\textbf{第三步（解 $\lambda$ 并回代）：}]
			由 $-\dfrac{1+2\lambda}{1-\lambda}=-2$ 得 $\dfrac{1+2\lambda}{1-\lambda}=2$，整理得 $1+2\lambda=2-2\lambda$，解得 $\lambda=\dfrac14$。代入直线系：
			\[
				\begin{aligned}
					& x+y-1+\frac14(2x-y+4)=0 \\ & \Longrightarrow\; \frac32x+\frac34y=0 \\ & \Longrightarrow\; 2x+y=0 .
			\end{aligned}
			\]
			\par\textbf{理由：} 代回即得具体直线方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{2x+y=0}$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l_1: x+y-3=0$，$l_2: 2x-y=0$，求过 $l_1$ 与 $l_2$ 交点且与直线 $3x+y-4=0$ 平行的直线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设直线系）：}]
			设所求直线方程为 $x+y-3+\lambda(2x-y)=0$。
			\par\textbf{理由：} 利用直线系避免求交点。
		\item[\textbf{第二步（列平行条件）：}]
			整理得 $(1+2\lambda)x+(1-\lambda)y-3=0$，斜率 $k=-\dfrac{1+2\lambda}{1-\lambda}$（$1-\lambda\neq0$）。已知直线 $3x+y-4=0$ 的斜率为 $-3$，平行要求斜率相等，即 $-\dfrac{1+2\lambda}{1-\lambda} = -3$。
			\par\textbf{理由：} 两直线平行则斜率相等（当直线不垂直时）。
		\item[\textbf{第三步（解 $\lambda$ 并回代）：}]
			化简 $-\dfrac{1+2\lambda}{1-\lambda}=-3$ 得 $\dfrac{1+2\lambda}{1-\lambda}=3$，即 $1+2\lambda=3-3\lambda$，解得 $\lambda=\dfrac25$。代入：
			\[
				\begin{aligned}
					& x+y-3+\frac25(2x-y)=0 \\ & \Longrightarrow\; \frac95x+\frac35y-3=0 \\ & \Longrightarrow\; 9x+3y-15=0 \\ & \Longrightarrow\; 3x+y-5=0 .
			\end{aligned}
			\]
			\par\textbf{理由：} 代入化简即得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{3x+y-5=0}$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l_1:2x-y-2=0$ 与 $l_2:x+y-4=0$ 相交，求过它们交点且在两坐标轴上截距相等的直线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设直线系）：}]
			设所求直线方程为 $2x-y-2+\lambda(x+y-4)=0$（$\lambda\in\mathbb{R}$，表示除 $l_2$ 外的过交点直线）。
			\par\textbf{理由：} 利用直线系可避免联立求交点。
		\item[\textbf{第二步（列截距相等条件）：}]
			化为 $(2+\lambda)x+(\lambda-1)y+(-2-4\lambda)=0$。分别令 $y=0$ 得 $x$ 截距 $a_x=\dfrac{2+4\lambda}{2+\lambda}$（$2+\lambda\neq0$）；令 $x=0$ 得 $y$ 截距 $a_y=\dfrac{2+4\lambda}{\lambda-1}$（$\lambda\neq1$）。由截距相等得 $\dfrac{2+4\lambda}{2+\lambda}=\dfrac{2+4\lambda}{\lambda-1}$。
			\par\textbf{理由：} 截距即直线与坐标轴交点的坐标值。
		\item[\textbf{第三步（求解并检验缺漏）：}]
			若 $2+4\lambda=0$，即 $\lambda=-\dfrac12$，此时直线为 $x-y=0$，截距均为 $0$，满足； 若 $2+4\lambda\neq0$，则约去得 $2+\lambda=\lambda-1$，无解。 故单参数形式下得直线 $x-y=0$。但检验 $l_2:x+y-4=0$，其在 $x$ 轴、$y$ 轴截距均为 $4$，也满足条件，而单参数形式无法表示 $l_2$，需补上。因此满足条件的直线共有两条。
			\par\textbf{理由：} 当问题要求“截距相等”时，$l_2$ 也是一个解，不能遗漏。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{x-y=0}$ 或 $\boxed{x+y-4=0}$。
\end{examplebox}
