\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{过两条相交直线交点的直线，其方程都可写成 $l_1+\lambda l_2=0$（$\lambda\in\mathbb{R}$，不含 $l_2$）或 $m l_1+n l_2=0$（$m,n$ 不全为零，包含 $l_2$）的形式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（两直线相交）：} $l_1$ 与 $l_2$ 必须相交，即 $A_1B_2\neq A_2B_1$。
		\item \textbf{条件2（参数限制）：} 单参数 $\lambda$ 可取任意实数；双参数 $m,n$ 必须满足 $m^2+n^2\neq0$，否则方程无意义。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（漏除l2）：} 使用 $l_1+\lambda l_2=0$ 时，最后要检查 $l_2$ 是否满足题目要求，防止遗漏。
		\item \textbf{检查项（分母讨论）：} 利用斜率或截距列方程时，对含分母的表达式先讨论分母为零的情况。
	\end{itemize}

\end{summarybox}
