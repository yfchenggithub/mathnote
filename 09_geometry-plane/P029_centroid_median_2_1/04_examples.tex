\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 \\(\triangle ABC\\) 中，\\(AD\\) 是 \\(BC\\) 边上的中线，\\(AD=9\\)，重心为 \\(G\\)。求 \\(AG\\) 和 \\(GD\\) 的长度。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（回顾比例）：}] 重心分中线 \\(AG:GD=2:1\\)。
			\par\textbf{理由：}三角形重心性质。
		\item[\textbf{第二步（设未知数）：}] 设 \\(GD=x\\)，则 \\(AG=2x\\)。由 \\(AD=AG+GD=3x=9\\)。
			\par\textbf{理由：}线段和关系。
		\item[\textbf{第三步（求结果）：}] 解得 \\(x=3\\)，所以 \\(AG=6\\)，\\(GD=3\\)。
			\par\textbf{理由：}代入求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\\(AG=6, GD=3\\)。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 \\(\triangle ABC\\) 的三个顶点坐标分别为 \\(A(1,2),\ B(3,4),\ C(5,0)\\)。设重心为 \\(G\\)，\\(BC\\) 边中点为 \\(D\\)。求 \\(|AG|:|GD|\\)，并验证比例是否为 \\(2:1\\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求重心坐标）：}] 由重心坐标公式得
			\[
				G\left(\frac{1+3+5}{3},\frac{2+4+0}{3}\right) = (3,2).
			\]
			\par\textbf{理由：}重心坐标是顶点坐标的算术平均。
		\item[\textbf{第二步（求中点坐标）：}] \\(BC\\) 中点 \\(D\\) 的坐标为
			\[
				D\left(\frac{3+5}{2},\frac{4+0}{2}\right) = (4,2).
			\]
			\par\textbf{理由：}中点坐标公式。
		\item[\textbf{第三步（计算长度比）：}] 计算向量：
			\[
				\begin{aligned}
					\overrightarrow{AG} &= (3-1,2-2) \\
					&= (2,0),\\
					\overrightarrow{GD} &= (4-3,2-2) \\
					&= (1,0).
			\end{aligned}
			\]
			所以 \\(|AG| = 2,\ |GD| = 1\\)，即 \\(|AG|:|GD|=2:1\\)，验证成立。
			\par\textbf{理由：}共线情形下长度比等于坐标差的绝对值比。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\\(AG:GD=2:1\\) 成立。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 \\(\triangle ABC\\) 中，\\(G\\) 为重心，\\(D\\) 为 \\(BC\\) 中点。已知 \\(S_{\triangle ABC}=18\\)，求 \\(S_{\triangle GBC}\\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（中线分面积）：}] 由 \\(D\\) 为 \\(BC\\) 中点，\\(AD\\) 是中线，得
			\[
				\begin{aligned}
					S_{\triangle ABD} &= S_{\triangle ACD} \\
					&= \frac12 S_{\triangle ABC} \\
					&= 9.
			\end{aligned}
			\]
			\par\textbf{理由：}等底等高三角形面积相等。
		\item[\textbf{第二步（利用重心比例）：}] 重心 \\(G\\) 分中线 \\(AD\\) 为 \\(AG:GD=2:1\\)，故 \\(GD:AD=1:3\\)。在 \\(\triangle ABD\\) 中，\\(\triangle GBD\\) 与 \\(\triangle ABD\\) 同高（点 \\(B\\) 到 \\(AD\\) 距离不变），所以面积比等于底边比：
			\[
				\begin{aligned}
					\frac{S_{\triangle GBD}}{S_{\triangle ABD}} &= \frac{GD}{AD} \\
					&= \frac13.
			\end{aligned}
			\]
			同理，\\(\frac{S_{\triangle GCD}}{S_{\triangle ACD}} = \frac13\\)。
			\par\textbf{理由：}等高三角形面积比等于对应底边长度比。
		\item[\textbf{第三步（计算所求面积）：}]
			\[
				\begin{aligned}
					S_{\triangle GBC} &= S_{\triangle GBD} + S_{\triangle GCD} \\
					&= \frac13 S_{\triangle ABD} + \frac13 S_{\triangle ACD} \\
					&= \frac13 (S_{\triangle ABD} + S_{\triangle ACD}) \\
					&= \frac13 \times 18 \\
					&= 6.
			\end{aligned}
			\]
			\par\textbf{理由：}代入面积值求和。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\\(S_{\triangle GBC}=6\\)。
\end{examplebox}
