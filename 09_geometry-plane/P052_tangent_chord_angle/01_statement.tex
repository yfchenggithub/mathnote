	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（相切于端点）：}}] 直线 $l$ 与 $\odot O$ 相切于点 $A$，弦 $AB$ 在圆内，$T$ 是 $l$ 上异于 $A$ 的一点，点 $C$ 是 $\odot O$ 上异于 $A,B$ 且与 $T$ 在弦 $AB$ 异侧的一点。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（弦切角等于圆周角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			弦切角等于它所夹弧所对的圆周角。
			\[
				\angle BAT = \angle BCA
			\]
			\vspace{0.2em}

		\item[\textbf{\textcolor{CStatement}{推广结论（圆心角形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			弦切角等于所夹弧所对的圆心角的一半。
			\[
				\angle BAT = \frac{1}{2} \angle BOA
			\]
	\end{description}
