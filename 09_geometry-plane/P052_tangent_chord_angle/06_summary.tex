\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{弦切角等于它所夹弧所对的圆周角（$\angle BAT = \angle BCA$）。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（相切于顶点）：} 直线与圆相切于角的顶点。
		\item \textbf{条件2（过切点弦）：} 角的另一边是圆过切点的一条弦。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（弧对应）：} 务必取弦切角所夹弧所对的圆周角，即顶点在弦的另一侧弧上（与切线异侧），否则角不相等。
		\item \textbf{检查项（切点确认）：} 使用前确认顶点是切点且直线是切线。
	\end{itemize}
\end{summarybox}
