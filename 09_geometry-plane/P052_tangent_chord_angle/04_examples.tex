\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $O$ 的切线 $PA$ 切圆于点 $A$，弦 $AB$，点 $C$ 在圆上，$\angle BCA = 35^\circ$，求 $\angle PAB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别角）：}] $\angle PAB$ 是切线 $PA$ 与弦 $AB$ 构成的弦切角，$\angle BCA$ 是弧 $AB$ 所对的圆周角。
			\par\textbf{理由：} 弦切角与圆周角的定义。
		\item[\textbf{第二步（套用定理）：}] 由弦切角定理得
			\[
				\begin{aligned}
					\angle PAB &= \angle BCA \\
					&= 35^\circ
			\end{aligned}
			\]
			\par\textbf{理由：} 弦切角等于它所夹弧所对的圆周角。
		\item[\textbf{第三步（写结果）：}] 因此 $\angle PAB = 35^\circ$。
			\par\textbf{理由：} 计算得出。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle PAB = 35^\circ$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$\triangle ABC$ 内接于圆，$AB = AC$，$\angle BAC = 80^\circ$，过点 $A$ 作圆的切线 $AD$，求 $\angle DAB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等腰求角）：}] 利用等腰三角形性质，
			\[
				\begin{aligned}
					\angle ACB &= \frac{180^\circ - \angle BAC}{2} \\
					&= \frac{180^\circ-80^\circ}{2} \\
					&= 50^\circ
			\end{aligned}
			\]
			\par\textbf{理由：} 等腰三角形底角相等，三角形内角和 $180^\circ$。
		\item[\textbf{第二步（确定弦切角）：}] $\angle DAB$ 是切线 $AD$ 与弦 $AB$ 构成的弦切角，所夹弧 $AB$ 所对的圆周角是 $\angle ACB$。
			\par\textbf{理由：} 弦切角定义。
		\item[\textbf{第三步（套用定理）：}] 由弦切角定理，
			\[
				\begin{aligned}
					\angle DAB &= \angle ACB \\
					&= 50^\circ
			\end{aligned}
			\]
			\par\textbf{理由：} 直接应用定理。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle DAB = 50^\circ$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	如图，$AB$ 是圆 $O$ 的弦，$AD$ 是圆 $O$ 在点 $A$ 处的切线，$C$ 是弧 $AB$（与切线异侧）上一点，求证：$\angle DAB = \angle ACB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别弦切角）：}] $\angle DAB$ 是切线 $AD$ 与弦 $AB$ 所成的角，是弦切角，所夹弧为 $AB$。
			\par\textbf{理由：} 弦切角定义（顶点为切点 $A$，一边切线 $AD$，另一边弦 $AB$）。
		\item[\textbf{第二步（识别圆周角）：}] $\angle ACB$ 是点 $C$ 在圆上且与 $A,B$ 构成的角，是弧 $AB$ 所对的圆周角（因为 $C$ 在弧 $AB$ 上）。
			\par\textbf{理由：} 圆周角定义。
		\item[\textbf{第三步（套用定理）：}] 根据弦切角定理，
			\[
				\angle DAB = \angle ACB
			\]
			\par\textbf{理由：} 弦切角等于它所夹弧所对的圆周角。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle DAB = \angle ACB$。
\end{examplebox}
