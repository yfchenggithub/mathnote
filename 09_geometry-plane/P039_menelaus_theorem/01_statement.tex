	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（分点位置）：}}] 点$D$在$BC$所在直线上，点$E$在$CA$所在直线上，点$F$在$AB$所在直线上，且均不与顶点重合。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（充要条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$D,E,F$三点共线的充要条件是沿三角形一周的有向比值乘积为$1$。
			\[
				\frac{AF}{FB}\cdot\frac{BD}{DC}\cdot\frac{CE}{EA}=1
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（面积比值形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在三点共线的前提下，同高三角形面积比的乘积也为$1$（不可逆用于判定共线）。
			\[
				\frac{S_{\triangle ACF}}{S_{\triangle BCF}}\cdot\frac{S_{\triangle ABD}}{S_{\triangle ADC}}\cdot\frac{S_{\triangle BCE}}{S_{\triangle BAE}}=1
			\]
	\end{description}
