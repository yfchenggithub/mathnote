\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，点$D,E,F$分别在边$BC,CA,AB$上，且$BD:DC=2:1$，$CE:EA=3:2$，若$D,E,F$共线，求$AF:FB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（转化有向比）：}] 因为$D,E,F$都在边上（内分），所以有向比值$\dfrac{BD}{DC}=2$，$\dfrac{CE}{EA}=\dfrac{3}{2}$。
			\par\textbf{理由：}内分点对应的有向线段比为正数，值等于长度比。
		\item[\textbf{第二步（应用梅涅劳斯）：}] 由$D,E,F$共线，据梅涅劳斯定理得
			\[
				\frac{AF}{FB}\cdot\frac{BD}{DC}\cdot\frac{CE}{EA}=1.
			\]
			代入已知：
			\[
				\frac{AF}{FB}\cdot2\cdot\frac{3}{2}=1.
			\]
			\par\textbf{理由：}三点共线满足乘积为$1$。
		\item[\textbf{第三步（解比值）：}] 化简得 $\dfrac{AF}{FB}\cdot3=1$，所以 $\dfrac{AF}{FB}=\dfrac{1}{3}$，即$AF:FB=1:3$。
			\par\textbf{理由：}代数运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{AF:FB=1:3}$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，点$D$在$BC$的延长线上（由$B$到$C$方向为正），且$BD:DC=3:1$（长度比）；点$E$在$CA$上，$CE:EA=1:2$；点$F$在$AB$所在直线上。若$D,E,F$三点共线，求$AF:FB$（有向比值），并判断点$F$在线段$AB$上还是延长线上。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判有向符号）：}] $D$在$BC$延长线上，故$\dfrac{BD}{DC}=-3$（负号因为$BD$与$BC$同向，$DC$与$BC$反向）；$E$在$CA$内，$\dfrac{CE}{EA}=\dfrac{1}{2}$。
			\par\textbf{理由：}有向线段比与方向相关，外分点为负。
		\item[\textbf{第二步（代入梅涅劳斯）：}] 由共线得
			\[
				\frac{AF}{FB}\cdot(-3)\cdot\frac{1}{2}=1.
			\]
			\par\textbf{理由：}梅涅劳斯定理。
		\item[\textbf{第三步（解比值并解释）：}] 解得 $\dfrac{AF}{FB}=-\dfrac{2}{3}$。负号表示$F$在$AB$的延长线上，长度比$AF:FB=2:3$。
			\par\textbf{理由：}解方程，比值为负对应外分点。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$AF/FB=-\dfrac{2}{3}$，点$F$位于$AB$延长线上。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=10$，$BC=12$，$CA=15$。点$F$在$AB$上且$AF=4$，点$D$在$BC$上且$BD=4$，点$E$在$CA$上。若$D,E,F$三点共线，求$CE$与$EA$的长度。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算有向比）：}] $AF/FB=4/6=2/3$（正），$BD/DC=4/8=1/2$（正）。
			\par\textbf{理由：}内分点取正。
		\item[\textbf{第二步（应用梅涅劳斯得比例）：}] 由共线，据梅涅劳斯定理得
			\[
				\frac{AF}{FB}\cdot\frac{BD}{DC}\cdot\frac{CE}{EA}=1.
			\]
			代入已知：
			\[
				\left(\frac{2}{3}\right)\left(\frac{1}{2}\right)\frac{CE}{EA}=1.
			\]
			化简得
			\[
				\frac{1}{3}\cdot\frac{CE}{EA}=1,
			\]
			所以 $\dfrac{CE}{EA}=3$。
			\par\textbf{理由：}梅涅劳斯充要条件，代数解方程。
		\item[\textbf{第三步（求具体长度）：}] 由$CA=15=CE+EA$，且$CE=3EA$，得$3EA+EA=15$，故$EA=\dfrac{15}{4}=3.75$，$CE=11.25$。
			\par\textbf{理由：}线段和关系。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$CE=11.25$，$EA=3.75$。
\end{examplebox}
