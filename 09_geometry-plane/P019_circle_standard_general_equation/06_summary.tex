\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆的标准方程直接看出圆心半径；一般方程通过配方转化为标准方程，获得相同信息。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（标准方程适用）：}已知圆心和半径，直接使用标准方程。
		\item \textbf{条件2（一般方程适用）：}已知圆上三点或方程形式复杂时，使用一般方程待定系数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（判别式检验）：}使用一般方程前必须检验$D^2+E^2-4F>0$，否则可能不是圆。
		\item \textbf{检查项（配方平衡）：}配方时等式两边必须同加常数$(\frac{D}{2})^2$和$(\frac{E}{2})^2$。
	\end{itemize}
\end{summarybox}
