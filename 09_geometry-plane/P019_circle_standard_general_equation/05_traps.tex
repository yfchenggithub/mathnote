\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略判别条件）：}}]误以为方程$x^2+y^2+Dx+Ey+F=0$总是表示圆，未检验$D^2+E^2-4F$的正负。
		\item[\textbf{\textcolor{CTrap}{正确理解（圆的条件）：}}]只有$D^2+E^2-4F>0$时才表示圆；等于0是点圆，小于0没有实圆。
		\item[\textbf{\textcolor{CTrap}{错因分析（忘记检验）：}}]认为配方后右边自动为正，忽略了半径平方必须为正。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（配方常数漏加）：}}]配方时只给$x$和$y$补常数，但忘记在等式右边同时加上相同常数。
		\item[\textbf{\textcolor{CTrap}{正确理解（配方平衡）：}}]等式两边必须同时加$\frac{D^2}{4}+\frac{E^2}{4}$，保持平衡。
		\item[\textbf{\textcolor{CTrap}{错因分析（只补一边）：}}]只添项未保持等式平衡，导致结果常数项出错。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（圆心坐标符号）：}}]误认为圆心是$(\frac{D}{2},\frac{E}{2})$，忽略了配方后的符号。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号位置）：}}]圆心应为$(-\frac{D}{2},-\frac{E}{2})$，因为标准形式中$(x+\frac{D}{2})^2$对应 $x$ 坐标 $-\frac{D}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号理解错）：}}]完全平方展开后，未把括号内整体变号。
	\end{description}
\end{trapbox}
