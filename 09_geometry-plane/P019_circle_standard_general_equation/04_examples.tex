\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆心为$(2,-3)$，半径为$5$，求该圆的标准方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取条件）：}]从题中可知圆心$(a,b)=(2,-3)$，半径$r=5$。
			\[
				a=2,\quad b=-3,\quad r=5.
			\]
			\par\textbf{理由：}题目直接给出圆心和半径。
		\item[\textbf{第二步（套用公式）：}]代入标准方程$(x-a)^2+(y-b)^2=r^2$。
			\[
				(x-2)^2+(y+3)^2=5^2.
			\]
			\par\textbf{理由：}圆的标准方程形式。
		\item[\textbf{第三步（写出答案）：}]化简得最终方程。
			\[
				(x-2)^2+(y+3)^2=25.
			\]
			\par\textbf{理由：}计算$5^2=25$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$(x-2)^2+(y+3)^2=25$}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	判断方程$x^2+y^2-4x+6y-3=0$是否表示圆。若是，求圆心和半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取系数）：}]从方程得$D=-4,E=6,F=-3$。
			\[
				D=-4,\quad E=6,\quad F=-3.
			\]
			\par\textbf{理由：}圆的一般方程形式$x^2+y^2+Dx+Ey+F=0$。
		\item[\textbf{第二步（判别条件）：}]计算$D^2+E^2-4F$。
			\[
				D^2+E^2-4F = (-4)^2+6^2-4(-3)=16+36+12=64>0.
			\]
			\par\textbf{理由：}若$D^2+E^2-4F>0$，则方程表示圆。
		\item[\textbf{第三步（求圆心）：}]利用圆心坐标公式。
			\[
				\left(-\frac{D}{2},-\frac{E}{2}\right) = (2,-3).
			\]
			\par\textbf{理由：}圆心公式直接代入。
		\item[\textbf{第四步（求半径）：}]利用半径公式。
			\[
				r = \frac12\sqrt{D^2+E^2-4F} = \frac12\sqrt{64}=4.
			\]
			\par\textbf{理由：}半径公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{圆心$(2,-3)$，半径$4$}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	求过三点$A(1,0), B(3,0), C(0,2)$的圆的方程，并求圆心和半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设一般方程）：}]设所求圆的方程为$x^2+y^2+Dx+Ey+F=0$。
			\[
				x^2+y^2+Dx+Ey+F=0.
			\]
			\par\textbf{理由：}圆的一般方程形式，适合三点求系数。
		\item[\textbf{第二步（代入三点）：}]将$A,B,C$坐标分别代入方程，得到方程组。
			\[
				D+F = -1,
				\qquad
				3D+F = -9,
				\qquad
				2E+F = -4.
			\]
			\par\textbf{理由：}点的坐标满足方程。
		\item[\textbf{第三步（解方程组）：}]解之得$D=-4,F=3,E=-\frac{7}{2}$。
			\[
				D=-4,\quad F=3,\quad E=-\frac{7}{2}.
			\]
			\par\textbf{理由：}消元法求解线性方程组。
		\item[\textbf{第四步（写出一般方程）：}]代入系数得圆的方程。
			\[
				x^2+y^2-4x-\frac{7}{2}y+3=0.
			\]
			\par\textbf{理由：}将$D,E,F$回代。
		\item[\textbf{第五步（判别并求圆心）：}]计算判别式并利用圆心公式。
			\[
				D^2+E^2-4F = \frac{65}{4}>0,
				\qquad
				\left(-\frac{D}{2},-\frac{E}{2}\right)=\left(2,\frac{7}{4}\right).
			\]
			\par\textbf{理由：}判别式正时表示圆，圆心坐标公式。
		\item[\textbf{第六步（求半径）：}]计算半径。
			\[
				r = \frac12\sqrt{D^2+E^2-4F} = \frac12\sqrt{\frac{65}{4}} = \frac{\sqrt{65}}{4}.
			\]
			\par\textbf{理由：}半径公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{圆的方程为$x^2+y^2-4x-\frac{7}{2}y+3=0$，圆心$\left(2,\frac{7}{4}\right)$，半径$\frac{\sqrt{65}}{4}$}
\end{examplebox}
