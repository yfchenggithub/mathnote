\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（圆心与半径）：}}]在平面直角坐标系中，圆心坐标为$(a,b)$，半径$r>0$。
		\item[\textbf{\textcolor{CStatement}{条件2（方程与判别式）：}}]给定方程$x^2+y^2+Dx+Ey+F=0$，且$D^2+E^2-4F>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（标准方程）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆心在$(a,b)$、半径为$r$的圆的标准形式。
			\[
				(x-a)^2+(y-b)^2=r^2.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（一般方程配方）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			配方后显示圆心与半径。
			\[
				\Bigl(x+\frac{D}{2}\Bigr)^2+\Bigl(y+\frac{E}{2}\Bigr)^2=\frac{D^2+E^2-4F}{4}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（非圆情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若$D^2+E^2-4F=0$，方程表示点$(-\frac{D}{2},-\frac{E}{2})$；若$<0$，则不表示实圆。
			\[
				\Bigl(x+\frac{D}{2}\Bigr)^2+\Bigl(y+\frac{E}{2}\Bigr)^2=0.
			\]
	\end{description}
\end{statementbox}
