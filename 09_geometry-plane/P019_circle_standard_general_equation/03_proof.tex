\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	配方：将一般方程$x^2+y^2+Dx+Ey+F=0$通过配完全平方化为$(x+\frac{D}{2})^2+(y+\frac{E}{2})^2=\frac{D^2+E^2-4F}{4}$。右边为正时表示圆，并可直接读出圆心和半径。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（移项分组）：}]将方程按$x,y$分组，常数移到右边。
			\[
				x^2+y^2+Dx+Ey+F=0 \;\Longrightarrow\; (x^2+Dx)+(y^2+Ey) = -F.
			\]
			\par\textbf{理由：}为配方做准备，先分离含$x,y$的项。

		\item[\textbf{步骤二（加项配方）：}]两边同时加上$(\frac{D}{2})^2$和$(\frac{E}{2})^2$。
			\[
				(x^2+Dx+\frac{D^2}{4})+(y^2+Ey+\frac{E^2}{4}) = -F+\frac{D^2}{4}+\frac{E^2}{4}.
			\]
			\par\textbf{理由：}在等式两边同时加入相同常数，保持等式平衡。

		\item[\textbf{步骤三（完全平方）：}]利用完全平方公式化简。
			\[
				\Bigl(x+\frac{D}{2}\Bigr)^2+\Bigl(y+\frac{E}{2}\Bigr)^2 = \frac{D^2+E^2-4F}{4}.
			\]
			\par\textbf{理由：}$(x+\frac{D}{2})^2 = x^2+Dx+\frac{D^2}{4}$，同理$y$项。

		\item[\textbf{步骤四（条件分析）：}]分析右边符号确定是否为圆。
			\[
				\frac{D^2+E^2-4F}{4}>0 \;\iff\; D^2+E^2-4F>0.
			\]
			\par\textbf{理由：}圆的标准方程要求右边为半径平方，须为正。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	当$D^2+E^2-4F>0$时，原方程表示圆，圆心为$(-\frac{D}{2},-\frac{E}{2})$，半径为$r=\frac12\sqrt{D^2+E^2-4F}$。若$=0$则点圆，$<0$则无实圆。
\end{proofbox}
