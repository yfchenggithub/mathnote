\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，已知$AC=2$，$AB=2\sqrt{3}$，$\angle BAD=30^\circ$，$\angle CAD=60^\circ$，求$AD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(识别条件):}] 由已知得$b=AC=2$，$c=AB=2\sqrt{3}$，$\alpha=\angle BAD=30^\circ$，$\beta=\angle CAD=60^\circ$，$\alpha+\beta=90^\circ$。
			\par\textbf{理由：} 明确张角定理所需各量，注意$\alpha$对着边$b$，$\beta$对着边$c$。
		\item[\textbf{第二步(代入公式):}] 张角定理：$\displaystyle\frac{\sin\alpha}{b}+\frac{\sin\beta}{c}=\frac{\sin(\alpha+\beta)}{d}$。代入数值：
			\[
				\frac{\sin30^\circ}{2}+\frac{\sin60^\circ}{2\sqrt{3}}=\frac{\sin90^\circ}{d}.
			\]
			\par\textbf{理由：} 直接应用定理，将已知量代入等式。
		\item[\textbf{第三步(计算结果):}]
			\[
				\begin{aligned}
					\frac{\sin30^\circ}{2}+\frac{\sin60^\circ}{2\sqrt{3}}
					&= \frac{0.5}{2}+\frac{\sqrt3/2}{2\sqrt3} \\
					&= \frac14+\frac14 \\
					&= \frac12.
			\end{aligned}
			\]
			\par\textbf{理由：} 因此$\frac12=\frac{1}{d}$，解得$d=2$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$AD=2$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=5$，$AC=3$，$\angle A=60^\circ$，$AD$是$\angle A$的平分线，求$AD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(确定角度条件):}] 因为$AD$平分$\angle A=60^\circ$，所以$\alpha=\beta=30^\circ$，且$\alpha+\beta=60^\circ$。对应边$b=AC=3$，$c=AB=5$。
			\par\textbf{理由：} 角平分线将顶角平分，两小角相等。
		\item[\textbf{第二步(代入张角定理):}] 张角定理为$\displaystyle\frac{\sin\alpha}{b}+\frac{\sin\beta}{c}=\frac{\sin(\alpha+\beta)}{d}$。代入得
			\[
				\frac{\sin30^\circ}{3}+\frac{\sin30^\circ}{5}=\frac{\sin60^\circ}{d}.
			\]
			\par\textbf{理由：} 直接套用公式，注意$\sin30^\circ=\frac12$，$\sin60^\circ=\frac{\sqrt3}{2}$。
		\item[\textbf{第三步(解方程求$d$):}]
			\[
				\begin{aligned}
					\frac{\sin30^\circ}{3}+\frac{\sin30^\circ}{5}
					&= \frac{1/2}{3}+\frac{1/2}{5} \\
					&= \frac16+\frac1{10} \\
					&= \frac{5}{30}+\frac{3}{30} \\
					&= \frac{8}{30} \\
					&= \frac{4}{15}.
			\end{aligned}
			\]
			因此
			\[
				\frac{4}{15} = \frac{\sin60^\circ}{d}.
			\]
			解得
			\[
				d = \frac{15\sqrt3}{8}.
			\]
			\par\textbf{理由：} 先化简左边，再解比例式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$AD=\dfrac{15\sqrt3}{8}$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=AC=6$，$AD$是$\angle A$的平分线，且$AD=3\sqrt3$，求$\angle A$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(分析角平分条件):}] 因为$AD$平分$\angle A$，所以$\alpha=\beta$。设$\angle BAD=\alpha$，则$\angle A=2\alpha$。
			\par\textbf{理由：} 角平分线定义。
		\item[\textbf{第二步(应用张角定理化简):}] 由张角定理得：
			\[
				\frac{\sin\alpha}{6}+\frac{\sin\alpha}{6}=\frac{\sin2\alpha}{3\sqrt3}.
			\]
			化简左边得$\frac{\sin\alpha}{3}$，故
			\[
				\frac{\sin\alpha}{3} = \frac{\sin2\alpha}{3\sqrt3}.
			\]
			利用倍角公式$\sin2\alpha=2\sin\alpha\cos\alpha$代入：
			\[
				\frac{\sin\alpha}{3} = \frac{2\sin\alpha\cos\alpha}{3\sqrt3}.
			\]
			约去$\sin\alpha$（非零）：
			\[
				\frac13 = \frac{2\cos\alpha}{3\sqrt3}.
			\]
			由此得$6\cos\alpha=3\sqrt3$，即$\cos\alpha=\frac{\sqrt3}{2}$。
			\par\textbf{理由：} 每步变形依据：倍角公式、等式同除非零量、交叉相乘。
		\item[\textbf{第三步(求角):}] 由$\cos\alpha=\frac{\sqrt3}{2}$得$\alpha=30^\circ$，所以$\angle A=2\alpha=60^\circ$。
			\par\textbf{理由：} 特殊角的三角函数值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\angle A=60^\circ$。
\end{examplebox}
