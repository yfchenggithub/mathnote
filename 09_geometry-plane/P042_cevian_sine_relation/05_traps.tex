\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一(角边对应混淆):}}] 在套用张角定理时，将$\sin\alpha$对应到边$c$，或将$\sin\beta$对应到边$b$。
		\item[\textbf{\textcolor{CTrap}{正确理解(角对边匹配):}}] $\alpha=\angle BAD$对应的边是$AC=b$，$\beta=\angle CAD$对应的边是$AB=c$。
		\item[\textbf{\textcolor{CTrap}{错因分析(记忆机械):}}] 仅记忆公式左右顺序，未结合图形中角与边的位置，导致对应错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二(忽视内点条件):}}] 将点$D$取在$BC$延长线上时仍然使用张角定理。
		\item[\textbf{\textcolor{CTrap}{正确理解(内分点要求):}}] 点$D$必须在线段$BC$上，此时$\alpha,\beta$均为$\triangle ABC$的内角，才能保证面积公式成立。
		\item[\textbf{\textcolor{CTrap}{错因分析(条件遗漏):}}] 忽略定理中“点$D$在边$BC$上”的前提，对$D$的位置不加验证。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三(面积公式误用):}}] 在用面积法推导或证明时，将$\triangle ABD$的面积错写为$\frac12\cdot b\cdot d\cdot\sin\alpha$。
		\item[\textbf{\textcolor{CTrap}{正确理解(夹角选择):}}] $\triangle ABD$中，已知边$AB=c$与$AD=d$的夹角为$\alpha$，面积应为$\frac12 c d\sin\alpha$。
		\item[\textbf{\textcolor{CTrap}{错因分析(夹角混淆):}}] 未分清三角形中哪两条边夹着角$\alpha$，机械套用面积公式。
	\end{description}
\end{trapbox}
