\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一(点边上分点):}}] 在$\triangle ABC$中，点$D$在边$BC$上，记$\angle BAD=\alpha$，$\angle CAD=\beta$，$AB=c$，$AC=b$，$AD=d$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一(张角定理):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			三角形内点分对边，正弦比之和等于整体正弦比。
			\[
				\frac{\sin\alpha}{b}+\frac{\sin\beta}{c}=\frac{\sin(\alpha+\beta)}{d}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形(角平分线特例):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			当$AD$平分$\angle A$时，$\alpha=\beta$，公式简化为角平分线长公式。
			\[
				d=\frac{2bc\cos\alpha}{b+c}.
			\]
	\end{description}
\end{statementbox}
