\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{张角定理揭示了三角形中任一点分对边时，被分割出的两个小三角形面积之和等于原三角形面积，从而得到边角之间的简洁分式关系。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1(点在线段上):} 点$D$在边$BC$上，且$D$在$B,C$之间（内分点）。
		\item \textbf{条件 2(角度为内角):} $\alpha,\beta\in(0,\pi)$且$\alpha+\beta<\pi$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点(角边对应):} 公式中$\sin\alpha$对应边$b=AC$，$\sin\beta$对应边$c=AB$，切勿颠倒。
		\item \textbf{检查项(内点确认):} 应用前确认$D$在线段$BC$上，否则结论不成立。
	\end{itemize}
\end{summarybox}
