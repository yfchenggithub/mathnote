\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用圆的圆周角定理：圆上任意圆周角等于它所对弧的圆心角的一半（或直接等于所对弧的度数的一半）。因为对角所对的弧互补，所以两个圆周角之和为 $180^\circ$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（找所对弧）：}] $\angle A$ 所对的弧是 $\widehat{BCD}$，$\angle C$ 所对的弧是 $\widehat{BAD}$。
			\[
				\angle A \mapsto \widehat{BCD},\quad \angle C \mapsto \widehat{BAD}.
			\]
			\par\textbf{理由：} 圆周角顶点在圆上，两边与圆相交，所夹的弧为所对弧。

		\item[\textbf{步骤二（弧互补）：}] $\widehat{BCD}$ 与 $\widehat{BAD}$ 合起来是整个圆周，度数之和为 $360^\circ$。
			\[
				\widehat{BCD}+\widehat{BAD}=360^\circ.
			\]
			\par\textbf{理由：} 整个圆周为 $360^\circ$，$\widehat{BCD}$ 与 $\widehat{BAD}$ 合起来恰好组成整个圆周。

		\item[\textbf{步骤三（圆周角定理）：}] 圆周角的度数等于它所对弧的度数的一半。
			\[
				\angle A = \frac{1}{2}\widehat{BCD},\quad \angle C = \frac{1}{2}\widehat{BAD}.
			\]
			\par\textbf{理由：} 圆周角定理：圆周角等于对应圆心角的一半，圆心角等于所对弧的度数。

		\item[\textbf{步骤四（求和）：}] 将两式相加并代入互补关系。
			\[
				\begin{aligned}
					\angle A + \angle C &= \frac{1}{2}(\widehat{BCD}+\widehat{BAD}) \\
					&= \frac{1}{2}\times 360^\circ \\
					&= 180^\circ .
			\end{aligned}
			\]
			\par\textbf{理由：} 等量代换。

		\item[\textbf{步骤五（同理得另一组）：}] 同理可得 $\angle B+\angle D=180^\circ$。
			\[
				\angle B+\angle D=180^\circ.
			\]
			\par\textbf{理由：} 轮换对称。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，圆内接四边形的每组对角互补，即 $\angle A+\angle C=180^\circ$，$\angle B+\angle D=180^\circ$。}
\end{proofbox}
