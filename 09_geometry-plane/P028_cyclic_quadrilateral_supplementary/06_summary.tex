\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆内接四边形的对角互补；反之，一组对角互补的四边形内接于圆。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（四边形内接于圆）：} 四个顶点共圆。
		\item \textbf{条件 2（对角互补求角）：} 已知部分内角，利用互补关系求未知角。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分对角与邻角）：} 只有对角互补，邻角不一定互补。
		\item \textbf{检查项（前提条件验证）：} 应用性质前必须确认四边形确实内接于圆。
	\end{itemize}
\end{summarybox}
