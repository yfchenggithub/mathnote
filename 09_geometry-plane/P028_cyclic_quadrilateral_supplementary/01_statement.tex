	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（内接四边形）：}}] 四边形 $ABCD$ 的四个顶点在同一圆周上。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对角互补）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			四边形两组对角之和均等于 $180^\circ$。
			\[
				\angle A+\angle C=180^\circ,\quad \angle B+\angle D=180^\circ.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（四点共圆判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若四边形一组对角互补，则四个顶点共圆。
			\[
				\angle A+\angle C=180^\circ \Rightarrow A,B,C,D \text{ cyclic}.
			\]

	\end{description}

\end{statementbox}
