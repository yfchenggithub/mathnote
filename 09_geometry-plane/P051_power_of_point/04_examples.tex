\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $O$ 的半径为 $2$，点 $P$ 到圆心 $O$ 的距离为 $3$，过 $P$ 作直线交圆 $O$ 于 $A,B$（可相切），求 $PA\cdot PB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取已知量）：}] 由题意得 $R=2$，$d=3$。
			\par\textbf{理由：} 题目直接给出条件。
		\item[\textbf{第二步（应用圆幂公式）：}] 代入圆幂定理：
			\[
				\begin{aligned}
					PA\cdot PB &= |d^{2}-R^{2}| \\
					&= |9-4| \\
					&= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 过 $P$ 的任意直线与圆交于点 $A,B$，满足圆幂定理。
		\item[\textbf{第三步（得出乘积值）：}] 所以 $PA\cdot PB = 5$。
			\par\textbf{理由：} 计算无误。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA\cdot PB = 5$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	点 $P$ 在圆 $O$ 外，过 $P$ 作圆 $O$ 的切线 $PT$，切点为 $T$。已知 $PT=6$，圆半径 $R=3$，求 $OP$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判定位置关系）：}] 点 $P$ 在圆外，切线情形，适用 $PT^{2}=d^{2}-R^{2}$（由圆幂定理，此时 $A,B$ 重合于 $T$，得 $PT^{2}=PA\cdot PB$，且由 $d>R$ 知 $PA\cdot PB = d^{2}-R^{2}$）。
			\par\textbf{理由：} $d>R$，绝对值可去，直接相等。
		\item[\textbf{第二步（代入切线公式）：}] 代入 $PT=6$，$R=3$ 得 $36 = d^{2} - 9$，即 $d^{2}=45$。
			\par\textbf{理由：} 由 $PT^{2}=d^{2}-R^{2}$ 代入数值。
		\item[\textbf{第三步（开方求距离）：}] $d = \sqrt{45}$，即 $d = 3\sqrt{5}$（$OP>0$）。
			\par\textbf{理由：} 开平方取正。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$OP = 3\sqrt{5}$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $C: (x-1)^{2}+(y+2)^{2}=9$，点 $P(4,2)$，过 $P$ 作直线 $l$ 交圆 $C$ 于 $A,B$ 两点，求 $PA\cdot PB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求圆心半径）：}] 由圆方程得圆心 $C(1,-2)$，半径 $R=3$。
			\par\textbf{理由：} 标准方程 $(x-a)^{2}+(y-b)^{2}=R^{2}$。
		\item[\textbf{第二步（计算点圆心距）：}]
			\[
				\begin{aligned}
					d &= PC = \sqrt{(4-1)^{2}+(2-(-2))^{2}} \\
					&= \sqrt{9+16} \\
					&= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 两点间距离公式。
		\item[\textbf{第三步（应用定理求积）：}]
			\[
				\begin{aligned}
					PA\cdot PB &= |d^{2}-R^{2}| \\
					&= |25-9| \\
					&= 16.
			\end{aligned}
			\]
			\par\textbf{理由：} 圆幂定理对平面内任意点成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA\cdot PB = 16$}。
\end{examplebox}
