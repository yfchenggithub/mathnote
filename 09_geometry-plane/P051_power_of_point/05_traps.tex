\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忘记绝对值）}}] 误认为 $PA\cdot PB = d^{2}-R^{2}$，忽略 $P$ 在圆内时 $d^{2}-R^{2}<0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（有向与无向区别）}}] $PA,PB$ 均为长度（非有向），乘积恒非负，故公式为 $PA\cdot PB = |d^{2}-R^{2}|$。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号遗漏）}}] 未考虑 $d<R$ 情形，直接沿用了切割线定理的写法（那里 $d>R$ 自然为正）。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（局限圆外点）}}] 认为定理只适用于 $P$ 在圆外的情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（普遍适用）}}] 定理对平面内任意点（圆内、圆上、圆外）均成立，只需用绝对值统一。
		\item[\textbf{\textcolor{CTrap}{错因分析（知识片面）}}] 常见教材或题目多强调切割线定理，导致学生以为只有圆外点才有关。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽略切线退化）}}] 认为直线必须与圆有两个不同交点，相切时不适用。
		\item[\textbf{\textcolor{CTrap}{正确理解（切线包含在内）}}] 相切时 $A,B$ 重合为切点，公式化为 $PT^{2} = d^{2}-R^{2}$（$d>R$）或 $0$（$d=R$），仍满足一般形式。
		\item[\textbf{\textcolor{CTrap}{错因分析（极限视角缺失）}}] 未从“两个交点无限靠近”的角度理解，或未见过切线情形的推导。
	\end{description}
\end{trapbox}
