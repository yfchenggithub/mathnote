\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{点 $P$ 到定圆 $O$ 的幂（$|d^{2}-R^{2}|$）等于过 $P$ 任意直线截圆所得两段长度之积。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（点与圆位置已知）：} 已知半径 $R$ 和点 $P$ 到圆心距离 $d=OP$。
		\item \textbf{条件2（直线过点且与圆相交）：} 过 $P$ 的直线与圆有交点（至少一个切点）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（绝对值遗漏）：} 长度乘积恒非负，须用 $|d^{2}-R^{2}|$。
		\item \textbf{检查项（位置判定）：} 先判定 $P$ 在圆外还是圆内，确定是否需取绝对值或直接等于 $|d^{2}-R^{2}|$。
	\end{itemize}
\end{summarybox}
