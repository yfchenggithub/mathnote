\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定圆和点）：}}] 设 $\odot O$ 半径为 $R$，$P$ 为平面内任意一点，$d=OP$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（圆幂公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过 $P$ 的任意直线与圆交于 $A,B$（含切点），则 $PA\cdot PB$ 等于 $|d^{2}-R^{2}|$。
			\[
				\boxed{PA\cdot PB = |d^{2} - R^{2}|}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（切割线定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $P$ 在圆外且 $PT$ 为切线时，切线长平方等于 $d^{2}-R^{2}$。
			\[
				PT^{2} = d^{2} - R^{2}
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{特别情形：}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{点 $P$ 在圆上：} $d=R$，则 $PA\cdot PB = 0$（此时 $A$ 或 $B$ 之一与 $P$ 重合）。
		\item \textbf{点 $P$ 在圆内：} $d<R$，则 $PA\cdot PB = R^{2}-d^{2}$（常数，与直线方向无关）。
		\item \textbf{点 $P$ 在圆外：} $d>R$，则 $PA\cdot PB = d^{2}-R^{2}$（切割线时为切线长平方）。
	\end{itemize}
\end{statementbox}
