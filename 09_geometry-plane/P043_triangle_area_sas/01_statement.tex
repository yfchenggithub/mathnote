\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形与边角）：}}] 在三角形ABC中，已知两边长$a,b$及其夹角$C$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两边夹角面积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形面积等于两边长之积乘以夹角正弦的一半。
			\[
				S = \frac{1}{2}ab\sin C.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（轮换对称形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积公式可与任意两边及其夹角轮换使用。
			\[
				\begin{aligned}
					S &= \frac{1}{2}bc\sin A \\
					&= \frac{1}{2}ca\sin B
			\end{aligned}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{特殊情形（直角三角形）：}}
	当 $C = 90^\circ$ 时，$\sin C = 1$，面积退化为 $S = \frac{1}{2}ab$（直角三角形面积公式）。
\end{statementbox}
