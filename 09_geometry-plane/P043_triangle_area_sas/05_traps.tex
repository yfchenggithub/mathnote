\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（夹角辨别错误）}}] 已知两边 $a,b$ 和角 $A$，直接使用 $S=\frac12 ab\sin A$。
		\item[\textbf{\textcolor{CTrap}{正确理解（找准夹角）：}}] 面积公式中的角必须是已知两边的夹角，若给的是对角，应先求夹角再算面积。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 混淆了“两边及其夹角”与“两边及其中一边的对角”，导致公式使用错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽视两解）}}] 已知 $S,a,b$ 求角 $C$ 时，仅得出 $\sin C$ 后只取锐角。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论）：}}] $\sin C$ 确定后，$C$ 可能为锐角或钝角，需根据三角形内角条件取舍。
		\item[\textbf{\textcolor{CTrap}{错因分析（单调性忽视）：}}] 正弦函数在 $(0^\circ,180^\circ)$ 内先增后减，同一个正弦值对应两个角。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（特殊角记错）}}] 计算 $\sin30^\circ,\sin60^\circ$ 等值时混淆 $\frac12$ 和 $\frac{\sqrt3}{2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（强化记忆）：}}] 熟记 $\sin30^\circ=\frac12,\ \sin60^\circ=\frac{\sqrt3}{2},\ \sin45^\circ=\frac{\sqrt2}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（基础不牢）：}}] 特殊角三角函数值记忆不准确，导致面积计算错误。
	\end{description}
\end{trapbox}
