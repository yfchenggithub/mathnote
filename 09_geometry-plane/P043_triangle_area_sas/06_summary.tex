\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{三角形面积等于两边之积与它们夹角正弦乘积的一半。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（已知两边一夹角）：} 需要明确两边的长度及它们所夹的角。
		\item \textbf{条件 2（任意三角形）：} 该公式对锐角、直角、钝角三角形均成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（夹角确认）：} 务必确保公式中使用的是两边的夹角，而不是其他角。
		\item \textbf{检查项（面积正性）：} 由于 $\sin C > 0$（$0^\circ<C<180^\circ$），面积始终为正。
	\end{itemize}
\end{summarybox}
