\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $a=6$，$b=8$，$C=30^\circ$，求面积 $S$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入公式）：}] 直接代入两边夹角面积公式。
			\[
				\begin{aligned}
					S &= \frac12 ab\sin C \\
					&= \frac12 \times 6 \times 8 \times \sin30^\circ .
			\end{aligned}
			\]
			\par\textbf{理由：}
			已知两边及其夹角，可用 $S = \frac12 ab\sin C$。

		\item[\textbf{第二步（计算数值）：}] 计算右侧数值。
			\[
				\begin{aligned}
					S &= \frac12 \times 48 \times \frac12 \\
					&= 12 .
			\end{aligned}
			\]
			\par\textbf{理由：}
			$\sin30^\circ = \frac12$；乘法运算。

		\item[\textbf{第三步（完成作答）：}] 得出面积结果。
			\[
				S = 12.
			\]
			\par\textbf{理由：}
			面积单位略。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S = 12$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $a=6$，$b=8$，面积 $S=12\sqrt3$，求夹角 $C$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（反用公式）：}] 由面积公式 $S=\frac12 ab\sin C$ 解出 $\sin C$。
			\[
				\begin{aligned}
					\sin C &= \frac{2S}{ab} \\
					&= \frac{2\times12\sqrt3}{6\times8} \\
					&= \frac{24\sqrt3}{48} \\
					&= \frac{\sqrt3}{2} .
			\end{aligned}
			\]
			\par\textbf{理由：}
			等式两边同乘2再除以 $ab$。

		\item[\textbf{第二步（求角度）：}] 由 $\sin C = \frac{\sqrt3}{2}$ 得 $C$ 的可能值。
			\[
				\begin{aligned}
					\sin C &= \frac{\sqrt3}{2} \\
					&\Rightarrow \\
					C &= 60^\circ,\ 120^\circ .
			\end{aligned}
			\]
			\par\textbf{理由：}
			在 $(0^\circ,180^\circ)$ 内，正弦值为 $\frac{\sqrt3}{2}$ 的角有 $60^\circ$ 和 $120^\circ$。

		\item[\textbf{第三步（确认答案）：}] 两个角均符合三角形内角条件。
			\[
				C = 60^\circ,\ 120^\circ.
			\]
			\par\textbf{理由：}
			无额外限制条件时，两解成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$C=60^\circ$ 或 $C=120^\circ$}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $a=2\sqrt3$，$b=2$，面积 $S=\sqrt3$，求边 $c$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求角 $C$ 的正弦）：}] 利用面积公式求 $\sin C$。
			\[
				\begin{aligned}
					S &= \frac12 ab\sin C \\
					&\Rightarrow \\
					\sqrt3 &= \frac12 \times 2\sqrt3 \times 2 \times \sin C \\
					&= 2\sqrt3 \sin C ,
			\end{aligned}
			\]
			故 $\sin C = \frac12$。
			\par\textbf{理由：}
			面积公式变形，代入 $a,b,S$ 解出 $\sin C$。

		\item[\textbf{第二步（确定 $C$ 的可能值）：}] 根据 $\sin C = \frac12$ 得 $C$ 的两解。
			\[
				C = 30^\circ,\ 150^\circ.
			\]
			\par\textbf{理由：}
			正弦值为 $\frac12$ 的锐角和钝角各一。

		\item[\textbf{第三步（分情况使用余弦定理）：}] 分别用余弦定理求 $c$。
			当 $C = 30^\circ$ 时，
			\[
				\begin{aligned}
					c^2 &= a^2 + b^2 - 2ab\cos30^\circ \\
					&= 12 + 4 - 2\cdot2\sqrt3\cdot2\cdot\frac{\sqrt3}{2} \\
					&= 16 - 12 \\
					&= 4 ,
			\end{aligned}
			\]
			故 $c = 2$。
			当 $C = 150^\circ$ 时，
			\[
				\begin{aligned}
					c^2 &= a^2 + b^2 - 2ab\cos150^\circ \\
					&= 12 + 4 - 2\cdot2\sqrt3\cdot2\cdot\left(-\frac{\sqrt3}{2}\right) \\
					&= 16 + 12 \\
					&= 28 ,
			\end{aligned}
			\]
			故 $c = 2\sqrt7$。
			\par\textbf{理由：}
			余弦定理 $c^2 = a^2+b^2-2ab\cos C$；注意 $\cos150^\circ = -\frac{\sqrt3}{2}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$c=2$ 或 $c=2\sqrt7$}
\end{examplebox}
