\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{当两条直线斜率均存在时，平行 $\Longleftrightarrow$ 斜率相等，垂直 $\Longleftrightarrow$ 斜率乘积为 $-1$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（斜率存在前提）：} 两直线的斜率必须都存在，即直线不垂直于 $x$ 轴。
		\item \textbf{条件 2（一般式推广）：} 若使用一般式 $A_1x+B_1y+C_1=0$ 与 $A_2x+B_2y+C_2=0$，则平行条件为 $A_1B_2 = A_2B_1$ 且 $A_1C_2 \neq A_2C_1$（分母不为零），垂直条件为 $A_1A_2 + B_1B_2 = 0$，此时不需考虑斜率是否存在。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（垂直条件混淆）：} 垂直是 $k_1k_2=-1$，不是 $k_1=k_2$。
		\item \textbf{检查项（重合与平行）：} 斜率相等时，必须验证两直线是否重合（截距是否相等），只有截距不同时才平行。
	\end{itemize}
\end{summarybox}
