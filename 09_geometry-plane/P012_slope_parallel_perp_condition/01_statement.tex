\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（斜率存在前提）：}}] 两直线 $l_1,l_2$ 的斜率 $k_1,k_2$ 均存在，即直线不与 $x$ 轴垂直。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平行判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两直线斜率相等时，它们平行。
			\[
				l_1 \parallel l_2 \Longleftrightarrow k_1 = k_2.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（垂直判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两直线斜率互为负倒数时，它们垂直。
			\[
				l_1 \perp l_2 \Longleftrightarrow k_1 k_2 = -1.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（斜率不存在情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当直线垂直于 $x$ 轴时，斜率不存在，平行与垂直需按以下方法判定：
			\par
			$\bullet$ 若 $l_1$ 斜率不存在（直线 $x=a$），$l_2$ 斜率为 $k_2$，则 $l_1 \perp l_2 \Longleftrightarrow k_2 = 0$；
			\par
			$\bullet$ 若 $l_1,l_2$ 斜率均不存在且不重合，则 $l_1 \parallel l_2$。
	\end{description}

\end{statementbox}
