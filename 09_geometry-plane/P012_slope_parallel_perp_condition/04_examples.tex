\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l_1: y=2x+3$，$l_2: y=2x-5$，$l_3: y=-\dfrac{1}{3}x+2$，$l_4: y=3x+1$。判断 $l_1$ 与 $l_2$，$l_3$ 与 $l_4$ 的位置关系。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（提取斜率）：}]
			从直线方程直接写出斜率：
			\[
				k_1 = 2,\quad k_2 = 2,\quad k_3 = -\frac{1}{3},\quad k_4 = 3.
			\]
			\par\textbf{理由：} 直线 $y = kx + b$ 的斜率为 $k$。

		\item[\textbf{第二步（判定平行）：}]
			比较 $k_1$ 与 $k_2$：$k_1 = k_2 = 2$，且截距 $b_1=3 \neq b_2=-5$，故 $l_1 \parallel l_2$。
			\par\textbf{理由：} 斜率相等且截距不同时两直线平行。

		\item[\textbf{第三步（判定垂直）：}]
			计算 $k_3\cdot k_4 = \left(-\frac{1}{3}\right)\times 3 = -1$，故 $l_3 \perp l_4$。
			\par\textbf{理由：} 斜率乘积为 $-1$ 时两直线垂直。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{l_1 \parallel l_2,\quad l_3 \perp l_4}$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l_1: 2x+y-1=0$，$l_2: ax+2y+3=0$，且 $l_1 \perp l_2$，求实数 $a$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化斜截式）：}]
			将 $l_1$ 化为 $y = -2x+1$，故斜率 $k_1 = -2$；
			将 $l_2$ 化为 $y = -\dfrac{a}{2}x - \dfrac{3}{2}$，斜率 $k_2 = -\dfrac{a}{2}$。
			\par\textbf{理由：} 直线 $Ax+By+C=0$ 可化为 $y=-\dfrac{A}{B}x-\dfrac{C}{B}$（$B\neq0$）。

		\item[\textbf{第二步（列垂直方程）：}]
			由 $l_1 \perp l_2$ 得 $k_1 k_2 = -1$：
			\[
				(-2) \cdot \left(-\frac{a}{2}\right) = -1.
			\]
			\par\textbf{理由：} 垂直时斜率乘积为 $-1$。

		\item[\textbf{第三步（解参数）：}]
			化简得 $a = -1$。
			\[
				(-2)\cdot\left(-\frac{a}{2}\right) = a = -1 \quad\Longrightarrow\quad a = -1.
			\]
			\par\textbf{理由：} 解一次方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\boxed{a = -1}$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 的三个顶点 $A(0,1)$，$B(2,3)$，$C(4,-1)$。求 $AB$ 边上的高所在直线的方程。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求 AB 斜率）：}]
			$A(0,1), B(2,3)$，则 $k_{AB} = \dfrac{3-1}{2-0} = \dfrac{2}{2} = 1$。
			\par\textbf{理由：} 斜率公式 $k = \dfrac{y_2-y_1}{x_2-x_1}$。

		\item[\textbf{第二步（得高线斜率）：}]
			$AB$ 边上的高与 $AB$ 垂直，故高线斜率 $k_{\perp}$ 满足 $k_{AB}\cdot k_{\perp} = -1$，解得 $k_{\perp} = -1$。
			\par\textbf{理由：} 两直线垂直，斜率乘积为 $-1$。

		\item[\textbf{第三步（求高线方程）：}]
			高线过点 $C(4,-1)$，由点斜式得 $y - (-1) = -1 \cdot (x-4)$，即 $y = -x + 3$。
			\par\textbf{理由：} 点斜式 $y-y_0 = k(x-x_0)$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$AB$ 边上的高线方程为 $\boxed{y = -x + 3}$。
\end{examplebox}
