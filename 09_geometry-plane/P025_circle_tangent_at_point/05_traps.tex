\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记错）}}]
			误将切线方程写为 $x_0x+y_0y=r$ 或 $x_0x-y_0y=r^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（方程形式）：}}]
			正确的切线方程是 $x_0x+y_0y=r^2$，右边必须是半径的平方 $r^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆变量）：}}]
			把圆的半径 $r$ 和半径平方 $r^2$ 记混，或受到自然语言“半径”一词的影响。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽视条件）}}]
			不加验证直接用公式，而点可能不在圆上。
		\item[\textbf{\textcolor{CTrap}{正确理解（前提条件）：}}]
			公式只对切点在圆上成立，使用前必须验证 $x_0^2+y_0^2=r^2$ 是否满足。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略验证）：}}]
			审题不仔细，未判断点与圆位置关系，导致盲目套用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（默认圆心在原点）}}]
			遇到圆心不是原点的圆，仍使用 $x_0x+y_0y=r^2$ 形式。
		\item[\textbf{\textcolor{CTrap}{正确理解（平移修正）：}}]
			对于圆心 $(a,b)$，切线方程应修正为 $(x_0-a)(x-a)+(y_0-b)(y-b)=r^2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式泛化不足）：}}]
			死记硬背，没有理解公式的推导过程，没有掌握平移对应关系。
	\end{description}

\end{trapbox}
