\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $C: x^2+y^2=25$，点 $P(3,4)$ 在圆上，求过点 $P$ 的切线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证点）：}]
			计算 $3^2+4^2=9+16=25$，与 $r^2$ 相等，点 $P$ 在圆上。
			\par\textbf{理由：} 切线公式仅当点在圆上才能直接使用。
		\item[\textbf{第二步（套用公式）：}]
			根据切线方程 $x_0 x + y_0 y = r^2$，代入 $x_0=3, y_0=4, r^2=25$，得
			\[
				3x + 4y = 25
			\]
			。
			\par\textbf{理由：} 这是圆心在原点的圆的标准切线公式。
		\item[\textbf{第三步（写出结论）：}]
			所求切线方程为 $3x+4y=25$。
			\par\textbf{理由：} 方程已为最简形式，无需继续化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线方程为 $3x+4y=25$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $C: (x-2)^2+(y+1)^2=16$，点 $P(2,3)$ 在圆上，求过点 $P$ 的切线方程。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别圆心）：}]
			圆心为 $A(2,-1)$，半径 $r=4$；通过验证 $(2-2)^2+(3+1)^2=16$，知切点在圆上。
			\par\textbf{理由：} 圆心不在原点时需用平移形式。
		\item[\textbf{第二步（套用平移公式）：}]
			根据推广公式 $(x_0-a)(x-a)+(y_0-b)(y-b)=r^2$，代入 $a=2,b=-1$，$x_0=2,y_0=3$，得
			\[
				(2-2)(x-2)+(3+1)(y+1)=16
			\]
			。
			\par\textbf{理由：} 将原曲方程中的平方项替换为相应乘积。
		\item[\textbf{第三步（化简）：}]
			计算得 $0\times(x-2)+4(y+1)=16$，即 $4(y+1)=16$，解得 $y+1=4$，因此 $y=3$。
			\par\textbf{理由：} 常数项移项后即得最终切线方程。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切线方程为 $y=3$（一条水平线）。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $C: x^2+y^2=5$，过点 $M(2,4)$ 的直线与圆相切，求切点坐标。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设切点）：}]
			设切点坐标为 $P(x_0,y_0)$，则 $P$ 在圆上，故
			\[
				x_0^2 + y_0^2 = 5
			\]
			。
			\par\textbf{理由：} 切点满足圆的方程。
		\item[\textbf{第二步（写切线并代点）：}]
			过 $P$ 的切线方程为 $x_0 x + y_0 y = 5$，因为切线过 $M(2,4)$，代入得
			\[
				2x_0 + 4y_0 = 5
			\]
			。
			\par\textbf{理由：} 切线的标准公式对任意圆上点都成立。
		\item[\textbf{第三步（解方程组）：}]
			联立 $
			\begin{cases}
				x_0^2+y_0^2=5,\\ 2x_0+4y_0=5.
		\end{cases}
			$ 由第二式得 $x_0=\frac{5-4y_0}{2}$，代入第一式：
			\[
				\begin{aligned}
					\left(\frac{5-4y_0}{2}\right)^2 + y_0^2 &= 5 \\ \frac{25-40y_0+16y_0^2}{4} + y_0^2 &= 5 \\ 25-40y_0+16y_0^2+4y_0^2 &= 20 \\ 20y_0^2-40y_0+5 &= 0 \\ 4y_0^2-8y_0+1 &= 0
			\end{aligned}
			\]
			。 解此一元二次方程：
			\[
				\begin{aligned}
					y_0 &= \frac{8\pm\sqrt{64-16}}{8} \\ &= \frac{8\pm\sqrt{48}}{8} \\ &= \frac{8\pm4\sqrt{3}}{8} \\ &= 1\pm\frac{\sqrt{3}}{2}
			\end{aligned}
			\]
			。 计算对应的 $x_0$：
			\[
				\begin{aligned}
					x_0 &= \frac{5-4y_0}{2} \\ &= \frac{5-4\left(1\pm\frac{\sqrt{3}}{2}\right)}{2} \\ &= \frac{1\mp 2\sqrt{3}}{2}
			\end{aligned}
			\]
			。
			\par\textbf{理由：} 由切线方程和圆方程联立，消元后求解得到两组切点坐标。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{切点坐标为 $(\frac{1+2\sqrt{3}}{2}, 1-\frac{\sqrt{3}}{2})$ 和 $(\frac{1-2\sqrt{3}}{2}, 1+\frac{\sqrt{3}}{2})$。}
\end{examplebox}
