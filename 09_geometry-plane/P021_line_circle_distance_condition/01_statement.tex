\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（圆心与半径）：}}] 已知圆心 $P(x_0,y_0)$ 和半径 $r>0$.
		\item[\textbf{\textcolor{CStatement}{条件 2（直线方程）：}}] 直线 $l:Ax+By+C=0$, 其中 $A,B$ 不全为零.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（距离判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过圆心到直线的距离与半径比较得到位置关系.
			\[
				d<r \Leftrightarrow \text{相交},\quad d=r \Leftrightarrow \text{相切},\quad d>r \Leftrightarrow \text{相离}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（平方比较）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			避开根号，直接比较平方以简化计算.
			\[
				d^2 = \frac{(Ax_0+By_0+C)^2}{A^2+B^2},
				\qquad\text{则 } d^2 < r^2 \Leftrightarrow \text{相交},\quad d^2 = r^2 \Leftrightarrow \text{相切},\quad d^2 > r^2 \Leftrightarrow \text{相离}.
			\]
	\end{description}
\end{statementbox}
