\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础直接应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	判断圆 $(x-1)^2+(y+2)^2=9$ 与直线 $l:3x-4y+8=0$ 的位置关系.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求圆心半径）：}] 由圆的标准方程得圆心 $P(1,-2)$, 半径 $r=3$.
			\par\textbf{理由：} 标准方程 $(x-h)^2+(y-k)^2=r^2$ 直接读出.

		\item[\textbf{第二步（计算距离）：}] 由点到直线距离公式,
			\[
				\begin{aligned}
					d &= \frac{|3\cdot1-4\cdot(-2)+8|}{\sqrt{3^2+(-4)^2}} \\
					&= \frac{|3+8+8|}{5} \\
					&= \frac{19}{5} \\
					&= 3.8.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入公式 $d=\dfrac{|Ax_0+By_0+C|}{\sqrt{A^2+B^2}}$.

		\item[\textbf{第三步（比较判断）：}] 因为 $d=3.8>3=r$, 所以直线与圆相离.
			\par\textbf{理由：} 依据 $d>r\Leftrightarrow$ 相离.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直线与圆相离}.

	\textbf{\textcolor{CExample}{例 2（含参直线相交）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知直线 $l:y=kx+1$ 与圆 $C:x^2+y^2=1$ 相交, 求实数 $k$ 的取值范围.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求圆心半径）：}] 圆 $x^2+y^2=1$ 的圆心 $P(0,0)$, 半径 $r=1$.
			\par\textbf{理由：} 标准形式.

		\item[\textbf{第二步（计算距离）：}] 将直线化为一般式 $kx-y+1=0$, 代入距离公式
			\[
				\begin{aligned}
					d &= \frac{|k\cdot0-0+1|}{\sqrt{k^2+1}} \\
					&= \frac{1}{\sqrt{k^2+1}}.
			\end{aligned}
			\]
			\par\textbf{理由：} $d=\dfrac{|Ax_0+By_0+C|}{\sqrt{A^2+B^2}}$.

		\item[\textbf{第三步（解不等式）：}] 相交条件 $d<r$ 即 $\dfrac{1}{\sqrt{k^2+1}}<1$.
			\[
				\begin{aligned}
					&\frac{1}{\sqrt{k^2+1}}<1 \\
					&\Longrightarrow \sqrt{k^2+1}>1 \\
					&\Longrightarrow k^2+1>1 \\
					&\Longrightarrow k^2>0 \\
					&\Longrightarrow k\neq0.
			\end{aligned}
			\]
			\par\textbf{理由：} 不等式两边均为正, 平方后方向不变.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$k$ 的取值范围是 $k\neq0$}.

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $C$ 的圆心在直线 $y=2x$ 上, 且经过点 $A(1,1)$ 和 $B(0,0)$. 判断直线 $l:x+y-2=0$ 与圆 $C$ 的位置关系.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求圆心半径）：}] 设圆心 $P(a,2a)$, 由 $|PA|=|PB|$ 得
			\[
				(a-1)^2+(2a-1)^2 = (a-0)^2+(2a-0)^2.
			\]
			展开化简:
			\[
				\begin{aligned}
					a^2-2a+1+4a^2-4a+1 &= a^2+4a^2 \\
					\Longrightarrow 5a^2-6a+2 &= 5a^2.
			\end{aligned}
			\]
			解得 $a=\dfrac13$. 故圆心 $P\!\left(\dfrac13,\dfrac23\right)$, 半径
			\[
				\begin{aligned}
					r &= |PB| \\
					&= \sqrt{\left(\dfrac13\right)^2+\left(\dfrac23\right)^2} \\
					&= \frac{\sqrt{5}}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 圆上各点到圆心距离相等.

		\item[\textbf{第二步（计算距离）：}] 圆心 $P(\frac13,\frac23)$ 到直线 $x+y-2=0$ 的距离
			\[
				\begin{aligned}
					d &= \frac{\left|\frac13+\frac23-2\right|}{\sqrt{1^2+1^2}} \\
					&= \frac{|1-2|}{\sqrt{2}} \\
					&= \frac{1}{\sqrt{2}} \\
					&= \frac{\sqrt{2}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入距离公式.

		\item[\textbf{第三步（比较判断）：}] 比较 $d$ 与 $r$:
			\[
				d^2 = \frac12, \quad r^2 = \frac{5}{9}\approx0.5556,\quad\text{所以 }d^2<r^2.
			\]
			因此 $d<r$, 直线与圆相交.
			\par\textbf{理由：} $d<r\Leftrightarrow$ 相交.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直线与圆相交}.
\end{examplebox}
