\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三边已知）：}}]
			在$\triangle ABC$中，已知三边长$a=BC$，$b=CA$，$c=AB$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中线平方公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			边$BC$上的中线$m_a$的平方，等于邻边平方和的一半减去对边平方的四分之一。
			\[
				m_a^2 = \frac{1}{4}(2b^2 + 2c^2 - a^2)
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（其余中线公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过轮换字母可得另外两条中线$m_b$、$m_c$的表达式。
			\[
				m_b^2 = \frac{1}{4}(2a^2 + 2c^2 - b^2)
			\]
			\[
				m_c^2 = \frac{1}{4}(2a^2 + 2b^2 - c^2)
			\]
	\end{description}

\end{statementbox}
