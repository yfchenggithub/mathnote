\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆字母对应）}}]
			记错公式中$a,b,c$的位置，例如求$m_a$时误写成$m_a^2=\frac14(2b^2+2c^2+a^2)$。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数符号规律）：}}]
			公式中$m_a$对应的对边$a$的系数为负（$-\frac14 a^2$），而两邻边$b,c$的系数为正。
		\item[\textbf{\textcolor{CTrap}{错因分析（对称记混）：}}]
			公式具有对称性，但每一项中对边平方带负号是易混淆点。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（跳过三角不等式）}}]
			不验证三角形是否成立直接代三边长。
		\item[\textbf{\textcolor{CTrap}{正确理解（满足三角不等式）：}}]
			使用前确认边长满足两边之和大于第三边。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}]
			公式只对任意三角形有效，但若三边不满足三角形条件，公式无意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开平方取负）}}]
			由$m_a^2$计算$m_a$时得出$m_a=\pm\sqrt{\dots}$，忽略长度非负。
		\item[\textbf{\textcolor{CTrap}{正确理解（正数算术根）：}}]
			中线长是正数，只取算术平方根。
		\item[\textbf{\textcolor{CTrap}{错因分析（平方根概念）：}}]
			混淆平方根与算术平方根，错误地将方程$x^2=k$的解直接当作长度。
	\end{description}

\end{trapbox}
