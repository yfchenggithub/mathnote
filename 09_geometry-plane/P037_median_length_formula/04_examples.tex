\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，已知$a=6$，$b=5$，$c=5$，求$BC$边上的中线$m_a$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认条件）：}]
			三角形三边已知，可直接使用中线长公式。
			\par\textbf{理由：} 公式适用任意三角形。
		\item[\textbf{第二步（代入计算）：}]
			代入数值：
			\[
				\begin{aligned}
					m_a^2 &= \frac14(2\cdot5^2+2\cdot5^2-6^2) \\ &= \frac14(2\cdot25+2\cdot25-36) \\ &= \frac14(50+50-36) \\ &= \frac14\times64 \\ &= 16 .
			\end{aligned}
			\]
			\par\textbf{理由：} 先计算幂，再合并括号。
		\item[\textbf{第三步（开方取正）：}]
			$m_a = \sqrt{16}$，即 $m_a=4$。
			\par\textbf{理由：} 中线长是正数，只取算术平方根。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m_a=4$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$b=5$，$c=5$，$BC$边上的中线$m_a=4$，求$a$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列方程）：}]
			由中线公式，代入$b=5$，$c=5$，$m_a^2=16$，得
			\[
				\frac14(2\cdot5^2+2\cdot5^2-a^2)=16 .
			\]
			\par\textbf{理由：} $m_a=4$，所以$m_a^2=16$。
		\item[\textbf{第二步（解方程）：}]
			逐次化简：
			\[
				\begin{aligned}
					\frac14(2\cdot5^2+2\cdot5^2-a^2) &= 16 \\ \frac14(50+50-a^2) &= 16 \\ \frac14(100-a^2) &= 16 \\ 100-a^2 &= 64 \\ a^2 &= 36 .
			\end{aligned}
			\]
			\par\textbf{理由：} 先计算幂，再合并，去分母，移项。
		\item[\textbf{第三步（取正根）：}]
			$a = \sqrt{36}$，即 $a=6$。
			\par\textbf{理由：} 边长非负，只取正平方根。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=6$}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，求证三条中线长的平方和为三边平方和的$\frac34$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出中线公式）：}]
			由中线长公式得
			\[
				m_a^2 = \frac14(2b^2+2c^2-a^2)
			\]
			\[
				m_b^2 = \frac14(2a^2+2c^2-b^2)
			\]
			\[
				m_c^2 = \frac14(2a^2+2b^2-c^2)
			\]
			\par\textbf{理由：} 直接应用阿波罗尼奥斯定理。
		\item[\textbf{第二步（求和化简）：}]
			将三式相加并合并同类项：
			\[
				\begin{aligned}
					m_a^2+m_b^2+m_c^2 &= \frac14\Bigl[ (2b^2+2c^2-a^2) \\ &\quad + (2a^2+2c^2-b^2) \\ &\quad + (2a^2+2b^2-c^2) \Bigr] \\ &= \frac14\Bigl[(-a^2+2a^2+2a^2) \\ &\quad + (2b^2-b^2+2b^2) \\ &\quad + (2c^2+2c^2-c^2)\Bigr] \\ &= \frac14(3a^2+3b^2+3c^2) \\ &= \frac34(a^2+b^2+c^2).
			\end{aligned}
			\]
			\par\textbf{理由：} 括号展开后合并$a^2,b^2,c^2$系数，再提取$\frac34$。
		\item[\textbf{第三步（得出结论）：}]
			因此
			\[
				m_a^2+m_b^2+m_c^2 = \frac34(a^2+b^2+c^2).
			\]
			\par\textbf{理由：} 已由第二步证得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m_a^2+m_b^2+m_c^2 = \frac34(a^2+b^2+c^2)$}
\end{examplebox}
