\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{中线长公式直接建立了三角形三边与中线之间的平方关系，是解三角形的重要工具。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（三边求中线）：} 已知三角形三边长，可直接代入公式计算中线长。
		\item \textbf{条件2（两边中线求边）：} 已知两边及其中一条中线长，可设第三边为未知数列方程求解。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（系数符号）：} 牢记公式中对边平方系数为负，邻边系数为正。
		\item \textbf{检查项（三角不等式）：} 使用公式前应检查三边长是否满足三角形不等式。
	\end{itemize}

\end{summarybox}
