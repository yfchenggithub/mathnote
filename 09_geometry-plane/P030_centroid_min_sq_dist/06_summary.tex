\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{重心到三顶点距离平方和最小，且最小值由三边长平方和决定。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件一（三角形非退化）：} $\triangle ABC$ 必须非退化（顶点不共线）。
		\item \textbf{条件二（任意动点）：} $P$ 是平面内任意一点，不受限。
		\item \textbf{条件三（边长记号）：} 边长 $a=BC$, $b=CA$, $c=AB$ 必须对应正确。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点一（等号条件）：} 等号仅当 $P$ 与重心 $G$ 重合，不可忽略。
		\item \textbf{检查项二（公式套用）：} 计算重心距离平方和时，直接用 $\frac13(a^2+b^2+c^2)$，不必再求距离。
		\item \textbf{检查项三（适用范围）：} 该结论仅适用于三角形，不能推广到四边形或其他多边形。
	\end{itemize}
\end{summarybox}
