\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 的三边长分别为 $a=5$, $b=6$, $c=7$。求重心 $G$ 到三个顶点距离平方和 $GA^2+GB^2+GC^2$；同时求平面内一点 $P$ 使 $PA^2+PB^2+PC^2$ 最小，并求最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结论）：}] 根据定理，$GA^2+GB^2+GC^2 = \frac13(a^2+b^2+c^2)$，且 $PA^2+PB^2+PC^2$ 的最小值就等于 $GA^2+GB^2+GC^2$，此时 $P$ 与 $G$ 重合。
			\par\textbf{理由：} 重心性质。
		\item[\textbf{第二步（代入计算）：}] 计算 $a^2+b^2+c^2 = 5^2+6^2+7^2 = 25+36+49=110$。
			\par\textbf{理由：} 直接平方。
		\item[\textbf{第三步（得出答案）：}] 所以 $GA^2+GB^2+GC^2 = \frac13 \times 110 = \frac{110}{3}$。最小值为 $\frac{110}{3}$，此时 $P$ 与重心 $G$ 重合。
			\par\textbf{理由：} 公式代入。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA^2+PB^2+PC^2$ 的最小值为 $\frac{110}{3}$，取等时 $P$ 为重心 $G$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\triangle ABC$ 顶点坐标 $A(1,2)$, $B(3,4)$, $C(5,0)$。求点 $P$ 使得 $PA^2+PB^2+PC^2$ 最小，并求最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求重心坐标）：}] 重心坐标为三顶点坐标的平均：
			\[
				x_G = \frac{1+3+5}{3}=3,\quad y_G = \frac{2+4+0}{3}=2.
			\]
			\par\textbf{理由：} 重心公式。
		\item[\textbf{第二步（计算边长平方）：}] 计算三边长度平方：
			\[
				a^2 = BC^2 = (5-3)^2+(0-4)^2 = 4+16=20,
			\]
			\[
				b^2 = CA^2 = (5-1)^2+(0-2)^2 = 16+4=20,
			\]
			\[
				c^2 = AB^2 = (3-1)^2+(4-2)^2 = 4+4=8.
			\]
			\par\textbf{理由：} 两点间距离公式。
		\item[\textbf{第三步（应用结论）：}] 由定理，最小值在重心处取得，且
			\[
				\min(PA^2+PB^2+PC^2) = \frac13(a^2+b^2+c^2) = \frac13(20+20+8)=\frac13\times 48 = 16.
			\]
			因此 $P$ 点坐标为 $(3,2)$，最小值为 $16$。
			\par\textbf{理由：} 定理直接应用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$P$ 为重心 $(3,2)$，最小值为 $16$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$G$ 是重心。已知 $GA^2+GB^2+GC^2 = 2$，且点 $P$ 满足 $PA^2+PB^2+PC^2 = 10$。求 $PG$ 的长度。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（引入恒等式）：}] 由定理的向量推导，有恒等式
			\[
				PA^2+PB^2+PC^2 = 3PG^2 + (GA^2+GB^2+GC^2).
			\]
			\par\textbf{理由：} 证明中已推导。
		\item[\textbf{第二步（代入已知数据）：}] 代入 $PA^2+PB^2+PC^2=10$ 和 $GA^2+GB^2+GC^2=2$：
			\[
				10 = 3PG^2 + 2.
			\]
			\par\textbf{理由：} 直接代入。
		\item[\textbf{第三步（解出 $PG$）：}] 移项得 $3PG^2 = 8$，所以 $PG^2 = \dfrac{8}{3}$，从而 $PG = \dfrac{2\sqrt{6}}{3}$（长度取正）。
			\par\textbf{理由：} 等式运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PG = \dfrac{2\sqrt{6}}{3}$。}
\end{examplebox}
