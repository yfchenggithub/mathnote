\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{条件一（三角形非退化）：}}] $\triangle ABC$ 为非退化三角形（顶点不共线）。
		\item[\textbf{\textcolor{CStatement}{条件二（重心定义）：}}] $G$ 为 $\triangle ABC$ 的重心（三条中线的交点）。
		\item[\textbf{\textcolor{CStatement}{条件三（边长记号）：}}] $a=BC$，$b=CA$，$c=AB$。
		\item[\textbf{\textcolor{CStatement}{条件四（任意动点）：}}] $P$ 为平面内任意一点。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（距离平方最小）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任意点 $P$ 到三顶点距离平方和在重心处最小，等号仅当 $P$ 与 $G$ 重合。
			\[
				PA^2+PB^2+PC^2 \ge GA^2+GB^2+GC^2.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（重心距离公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			重心到三顶点距离平方和等于三边长平方和的三分之一。
			\[
				GA^2+GB^2+GC^2 = \frac{1}{3}(a^2+b^2+c^2).
			\]
	\end{description}
\end{statementbox}
