\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AD$平分$\angle BAC$，$D$在$BC$上。已知$AB=9$，$AC=6$，$BC=15$，求$BD$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写比例）：}] 由角平分线定理
			\[
				\begin{aligned}
					\frac{BD}{DC} &= \frac{AB}{AC} \\
					&= \frac{9}{6} \\
					&= \frac{3}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			直接应用三角形内角平分线定理。

		\item[\textbf{第二步（设参数）：}] 设$BD=3k$，$DC=2k$。
			\par\textbf{理由：}
			比例可设公共参数$k$表示两段长度。

		\item[\textbf{第三步（列方程）：}] 由$BD+DC=BC$得
			\[
				3k+2k=15 \Rightarrow k=3,\; BD=9.
			\]
			\par\textbf{理由：}
			利用线段和条件列方程求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BD=9$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AD$平分$\angle BAC$，$AB=15$，$AC=9$，且$BD-DC=4$，求$BC$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写比例）：}] 由角平分线定理
			\[
				\begin{aligned}
					\frac{BD}{DC} &= \frac{AB}{AC} \\
					&= \frac{15}{9} \\
					&= \frac{5}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			直接应用定理。

		\item[\textbf{第二步（设参数）：}] 设$BD=5t$，$DC=3t$。
			\[
				\begin{aligned}
					BD-DC &= 5t-3t \\
					&= 2t \\
					&= 4,
			\end{aligned}
				\quad\implies\quad t=2.
			\]
			\par\textbf{理由：}
			比例设参，利用已知差条件求解。

		\item[\textbf{第三步（求和）：}]
			由$t=2$得$BD=5t=10$，$DC=3t=6$，故$BC=BD+DC=16$。
			\par\textbf{理由：}
			计算各段长，相加得$BC$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BC=16$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$\angle B=90^\circ$，$AD$平分$\angle BAC$交$BC$于$D$，已知$AB=5$，$BC=12$，求$BD$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求斜边）：}] 在$Rt\triangle ABC$中，由勾股定理
			\[
				\begin{aligned}
					AC &= \sqrt{AB^2+BC^2} \\
					&= \sqrt{5^2+12^2} \\
					&= \sqrt{169} \\
					&= 13.
			\end{aligned}
			\]
			\par\textbf{理由：}
			已知两直角边求斜边。

		\item[\textbf{第二步（写比例）：}] 由角平分线定理
			\[
				\begin{aligned}
					\frac{BD}{DC} &= \frac{AB}{AC} \\
					&= \frac{5}{13}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			直接应用定理。

		\item[\textbf{第三步（设参求解）：}] 设$BD=5k$，$DC=13k$，由$BC=12$得
			\[
				\begin{aligned}
					5k+13k &= 18k \\
					&= 12,
			\end{aligned}
				\quad\implies\quad k=\frac{2}{3},
			\]
			故$BD=5\times\frac{2}{3}=\frac{10}{3}$。
			\par\textbf{理由：}
			利用$BC$长度列方程求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\displaystyle BD=\frac{10}{3}$}。
\end{examplebox}
