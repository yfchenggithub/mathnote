	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（内角平分线）：}}] 在$\triangle ABC$中，$AD$是$\angle BAC$的平分线，且点$D$在边$BC$上。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（内角平分线定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			角平分线分对边的两段与两邻边长度成比例。
			\[
				\frac{BD}{DC}=\frac{AB}{AC}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（逆定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若分点在边上且比例成立，则线段为角平分线。
			\[
				\frac{BD}{DC}=\frac{AB}{AC}\;\Longrightarrow\;\angle BAD=\angle CAD.
			\]
	\end{description}
\end{statementbox}
