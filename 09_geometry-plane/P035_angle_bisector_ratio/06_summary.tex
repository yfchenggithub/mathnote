\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{三角形内角平分线将对边分成的两段与两邻边长度成比例。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（内角平分线）：} 在$\triangle ABC$中，$AD$平分$\angle BAC$且$D$在$BC$上。
		\item \textbf{条件2（比例对应）：} 比例式$\frac{BD}{DC}=\frac{AB}{AC}$中，$AB$与$BD$、$AC$与$DC$分别对应。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（边对应混淆）：} 注意$AB$对应$BD$，$AC$对应$DC$，不可颠倒。
		\item \textbf{检查项（点位置确认）：} 使用定理前确认$D$在线段$BC$上（非延长线）。
	\end{itemize}
\end{summarybox}
