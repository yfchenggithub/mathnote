\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽视半径大小）}}]
			题目未明确给出$R\ge r$时，直接使用$R-r$比较，可能得到负数，导致内含条件错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（统一假设大减小）：}}]
			应先确保$R\ge r$，或用绝对值$|R-r|$；内含条件是$d<|R-r|$。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号疏忽）：}}]
			学生常默认$R>r$，但未说明时可能$R<r$，$R-r$为负，误以为$d<R-r$恒不成立。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（内切内含混淆）}}]
			认为$d=R-r$是内含，或$d<R-r$是内切。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分等号不等式）：}}]
			$d=R-r$是内切（一交点），$0\le d<R-r$是内含（无交点）。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆模糊）：}}]
			未理解等号对应相切（边界），严格不等才无交点。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（遗漏同心内含）}}]
			认为$d=0$且$R\neq r$时不是内含（认为$d<R-r$不成立？），实际适用。
		\item[\textbf{\textcolor{CTrap}{正确理解（$d=0$也适用）：}}]
			$d=0$满足$0\le d<R-r$（当$R>r$），属内含。
		\item[\textbf{\textcolor{CTrap}{错因分析（不全面）：}}]
			只考虑两圆分离，忽略特殊情况。
	\end{description}

\end{trapbox}
