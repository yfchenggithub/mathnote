\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两圆半径分别为 $R=5,\ r=3$，圆心距 $d=1$，判断两圆位置关系。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算和差）：}]
			$R+r=8$，$R-r=2$。
			\par\textbf{理由：} 需要比较 $d$ 与和、差的大小。
		\item[\textbf{第二步（比较大小）：}]
			$1 < 2 = R-r$，且 $1>0$，满足 $0\le d < R-r$。
			\par\textbf{理由：} 内含的条件是 $d$ 小于半径差且非负。
		\item[\textbf{第三步（下结论）：}]
			两圆内含。
			\par\textbf{理由：} 对应位置关系判定表。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{内含}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知两圆外切，圆心距 $d=8$，两圆半径之比为 $3:2$，求两圆半径。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列条件）：}]
			设 $R=3k,\ r=2k$，外切条件 $d=R+r$，即 $8 = 3k+2k = 5k$。
			\par\textbf{理由：} 用比例系数 $k$ 表示半径。
		\item[\textbf{第二步（解方程）：}]
			$k = \frac{8}{5} = 1.6$。
			\par\textbf{理由：} 等式变形。
		\item[\textbf{第三步（求半径）：}]
			$R = 3\times1.6 = 4.8$，$r = 2\times1.6 = 3.2$。
			\par\textbf{理由：} 回代。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$R=4.8,\ r=3.2$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	两圆半径 $R=5,\ r=4$，圆心距 $d=3$，求两圆公共弦长。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证相交）：}]
			$R+r=9$，$R-r=1$，$3$ 在 $(1,9)$ 内，两圆相交。
			\par\textbf{理由：} 满足相交条件 $R-r<d<R+r$。
		\item[\textbf{第二步（求弦心距）：}]
			设公共弦为 $AB$，连心线 $O_1O_2$ 垂直平分 $AB$。利用公式：
			\[
				h_1 = \frac{R^2 - r^2 + d^2}{2d} = \frac{25 - 16 + 9}{6} = \frac{18}{6}=3,
			\]
			半弦长 $l = \sqrt{R^2 - h_1^2} = \sqrt{25 - 9} = 4$。
			\par\textbf{理由：} 构造直角三角形，弦心距公式。
		\item[\textbf{第三步（得弦长）：}]
			公共弦长 $= 2l = 8$。
			\par\textbf{理由：} 弦被垂直平分。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{公共弦长为8}。
\end{examplebox}
