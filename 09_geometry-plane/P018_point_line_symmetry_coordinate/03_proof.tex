\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	设对称点$P'(x',y')$。由对称性质可得两个条件：（1）线段$PP'$的中点$M$在直线$l$上；（2）直线$PP'$与$l$垂直。利用这两个条件列方程，解出$x',y'$。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（中点条件方程）：}]
			设$M\left(\dfrac{x_0+x'}{2},\dfrac{y_0+y'}{2}\right)$在直线$l:Ax+By+C=0$上，代入得
			\[
				A\cdot\frac{x_0+x'}{2}+B\cdot\frac{y_0+y'}{2}+C=0.
			\]
			\par\textbf{理由：} 对称点与原点的中点在对称轴上。

			整理得
			\[
				A(x_0+x')+B(y_0+y')+2C=0.
			\]

		\item[\textbf{步骤二（垂直条件参数化）：}]
			直线$l$的法向量为$\boldsymbol{n}=(A,B)$，$PP'$与$l$垂直，故$\overrightarrow{PP'}=(x'-x_0,y'-y_0)$与$\boldsymbol{n}$平行。存在实数$k$使得
			\[
				\begin{cases}
					x'-x_0 = kA,\\[4pt]
					y'-y_0 = kB.
			\end{cases}
			\]
			\par\textbf{理由:} 垂直即方向平行于法向量.

		\item[\textbf{步骤三(代入消元):}]
			将参数方程代入中点方程:
			\[
				A(x_0 + x_0 + kA) + B(y_0 + y_0 + kB) + 2C = 0.
			\]
			\par\textbf{理由:} 用$x'=x_0+kA$, $y'=y_0+kB$替换.

			整理得
			\[
				2A x_0 + kA^2 + 2B y_0 + kB^2 + 2C = 0.
			\]

		\item[\textbf{步骤四(解$k$):}]
			移项合并$k$项:
			\[
				k(A^2+B^2) + 2(Ax_0+By_0+C) = 0,
			\]
			\par\textbf{理由:} 将$k$的项提出公因子$A^2+B^2$.

			所以
			\[
				k = -\frac{2(Ax_0+By_0+C)}{A^2+B^2}.
			\]

		\item[\textbf{步骤五(回代求坐标):}]
			将$k$代回参数方程得
			\[
				x' = x_0 + kA = x_0 - \frac{2A(Ax_0+By_0+C)}{A^2+B^2},
			\]
			\[
				y' = y_0 + kB = y_0 - \frac{2B(Ax_0+By_0+C)}{A^2+B^2}.
			\]
			\par\textbf{理由:} 直接代入.
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	公式可直接使用,无需每次解方程组.注意分母$A^2+B^2$不为零,因为$A,B$不同时为零.
\end{proofbox}
