\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（点坐标直线）：}}] 已知点$P(x_0,y_0)$和直线$l:Ax+By+C=0$，其中$A$与$B$不同时为零。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对称点公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直接计算点关于直线的对称点坐标，避免解方程组。
			\[
				\begin{cases}
					x' = x_0 - \dfrac{2A(Ax_0+By_0+C)}{A^2+B^2},\\[8pt]
					y' = y_0 - \dfrac{2B(Ax_0+By_0+C)}{A^2+B^2}.
			\end{cases}
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形(水平垂直直线):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			当直线水平或垂直时,公式可简化.
			\begin{itemize}[leftmargin=*, itemsep=0.3em]
			\item 若$A=0$(直线$y = -\dfrac{C}{B}$),则
				\[
					\begin{cases}
						x' = x_0,\\[4pt]
						y' = -y_0 - \dfrac{2C}{B}.
				\end{cases}
				\]
			\item 若$B=0$(直线$x = -\dfrac{C}{A}$),则
				\[
					\begin{cases}
						x' = -x_0 - \dfrac{2C}{A},\\[4pt]
						y' = y_0.
				\end{cases}
				\]
		\end{itemize}
	\end{description}

\end{statementbox}
