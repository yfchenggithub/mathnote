\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	通过倍长中线构造全等三角形，再证明四边形是矩形，从而得到对角线相等关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（作辅助线）：}] 延长$CM$至$D$，使$MD = CM$，连接$AD, BD$。
		      \[
			      MD = CM,\quad D\text{在}CM\text{的延长线上}.
		      \]
		      \par\textbf{理由：}
		      为构造与$\triangle AMC$全等的三角形，将中线倍长。

		\item[\textbf{步骤二（证全等）：}] 在$\triangle AMC$和$\triangle BMD$中，
		      \[
			      AM = BM,\quad CM = DM,\quad \angle AMC = \angle BMD,
		      \]
		      所以
		      \[
			      \triangle AMC \cong \triangle BMD \quad \text{（SAS）}.
		      \]
		      \par\textbf{理由：}
		      $M$为$AB$的中点，所以$AM=BM$；由作图得$CM=DM$；$\angle AMC$与$\angle BMD$为对顶角。

		\item[\textbf{步骤三（推平行）：}] 由$\triangle AMC\cong\triangle BMD$，可得
		      \[
			      \angle ACM = \angle BDM,\qquad AC = BD.
		      \]
		      又因为$C,M,D$三点共线，所以$\angle ACM$与$\angle BDM$是一组内错角，因此
		      \[
			      AC \parallel BD.
		      \]
		      \par\textbf{理由：}
		      全等三角形对应角相等、对应边相等；内错角相等，两直线平行。

		\item[\textbf{步骤四（证矩形）：}] 在四边形$ADBC$中，已经得到
		      \[
			      AC \parallel BD,\qquad AC = BD.
		      \]
		      因此四边形$ADBC$是平行四边形。

		      又因为
		      \[
			      \angle ACB = 90^\circ,
		      \]
		      所以平行四边形$ADBC$是矩形。
		      \[
			      \text{四边形}ADBC\text{为矩形}.
		      \]
		      \par\textbf{理由：}
		      一组对边平行且相等的四边形是平行四边形；有一个角是直角的平行四边形是矩形。

		\item[\textbf{步骤五（出结论）：}] 矩形的对角线相等，所以
		      \[
			      AB = CD.
		      \]
		      又因为$D$在$CM$的延长线上，且$MD=CM$，所以
		      \[
			      CD = CM+MD = 2CM.
		      \]
		      因此
		      \[
			      AB = 2CM,
		      \]
		      即
		      \[
			      CM = \frac12 AB.
		      \]
		      \par\textbf{理由：}
		      矩形对角线相等，再结合等量代换，得到斜边中线等于斜边的一半。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	通过倍长中线构造全等，进而证明四边形为矩形，利用矩形对角线相等，导出
	\[
		CM=\frac12 AB.
	\]
	该证明的核心技巧是“中线倍长”。
\end{proofbox}
