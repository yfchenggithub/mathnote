	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（正向与逆向混淆）：}}] 看到“中线等于一边的一半”，就直接套用结论，却没有分清是在求长度还是在证明直角。
		\item[\textbf{\textcolor{CTrap}{正确理解（正向与逆向要区分）：}}] 正向使用时，必须先确认三角形是直角三角形，且所取的是斜边上的中线；逆向使用时，若某边上的中线等于该边的一半，则可以反推出该边所对的角为直角。
		\item[\textbf{\textcolor{CTrap}{错因分析（逻辑方向不清）：}}] 把“已知直角推出中线关系”和“已知中线关系推出直角”混为一谈，导致证明或计算时条件使用不严谨。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（非直角三角形误用）：}}] 将定理直接应用于非直角三角形，认为任意三角形某边上的中线都等于该边的一半。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件需完备）：}}] 正向使用定理时，三角形必须是直角三角形，并且考察的是斜边上的中线。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略直角条件）：}}] 只记住了“中线等于一半”，却漏掉了“直角三角形”和“斜边中点”这两个关键条件。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（混淆斜边与直角边）：}}] 误以为直角边上的中线也等于该直角边的一半。
		\item[\textbf{\textcolor{CTrap}{正确理解（中线性质区别）：}}] 只有斜边上的中线才满足等于斜边的一半，直角边上的中线一般不满足这个结论。
		\item[\textbf{\textcolor{CTrap}{错因分析（未区分边的位置）：}}] 没有先判断哪一条边是斜边，就机械套用“中线等于一半”。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（忽视中点条件）：}}] 认为在直角三角形中，直角顶点到斜边上任意一点的距离都等于斜边的一半。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须是中点）：}}] 只有斜边中点到直角顶点的距离才等于斜边的一半，斜边上其他点一般不满足。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺少中点条件）：}}] 将结论误记为“直角三角形直角顶点到斜边某点的距离”，漏掉了“斜边中点”。
	\end{description}
