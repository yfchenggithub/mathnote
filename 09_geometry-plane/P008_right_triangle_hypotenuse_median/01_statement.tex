	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（直角三角形与斜边中点）：}}] 在直角三角形$\triangle ABC$中，$\angle C=90^\circ$，$M$为斜边$AB$的中点。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中线长度）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 直角三角形中，从直角顶点引向斜边中点的中线，等于斜边的一半。
		      \[
			      CM = \frac{1}{2}AB.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（逆定理）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 若某边上的中线等于该边的一半，则这条边所对的角为直角。

		      在$\triangle ABC$中，若$M$为$BC$的中点，且
		      \[
			      AM = \frac12 BC,
		      \]
		      则
		      \[
			      \angle A = 90^\circ.
		      \]

		      也就是说，点$A$在以$BC$为直径的圆上，因此$\angle A$为直角。
	\end{description}
\end{statementbox}
