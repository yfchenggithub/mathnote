\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（半周长混淆）}}] 有些学生将公式错误记成 $S=(a+b+c)r$，混淆了半周长与全长。
		\item[\textbf{\textcolor{CTrap}{正确理解（半周长定义）：}}] 公式中 $p$ 是半周长 $\frac{a+b+c}{2}$，正确形式为 $S=pr$（即 $S=\frac12(a+b+c)r$）。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆模糊）：}}] 未区分符号 $p$ 代表半周长，误当成周长使用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（漏系数1/2）}}] 由面积分割推导时，求和后忘记乘 $\frac12$，得出错误公式 $S=(a+b+c)r$。
		\item[\textbf{\textcolor{CTrap}{正确理解（推导过程）：}}] 每个小三角形面积为 $\frac12 a r$，三者和为 $\frac12(a+b+c)r$，系数 $\frac12$ 不可省略。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导过急）：}}] 在添加多个三角形面积时，遗漏公因子 $\frac12$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（错用外接圆半径）}}] 将内切圆半径 $r$ 与外接圆半径 $R$ 混淆，误将公式 $S=pr$ 用于外接圆。
		\item[\textbf{\textcolor{CTrap}{正确理解（半径区分）：}}] $S=pr$ 只适用于内切圆；外接圆半径对应的面积公式是 $S=\frac{abc}{4R}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 不同圆心的半径性质不同，不能混用。
	\end{description}
\end{trapbox}
