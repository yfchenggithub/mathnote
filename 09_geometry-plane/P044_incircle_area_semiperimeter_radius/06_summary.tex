\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{$S = pr$：三角形面积等于半周长乘以内切圆半径。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（任意三角形）：} 公式适用于所有三角形（锐角、直角、钝角）。
		\item \textbf{条件 2（变量匹配）：} 已知 $p$ 和 $r$ 可求 $S$；已知 $S$ 和 $p$ 可求 $r$；已知 $S$ 和 $r$ 可求 $p$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（周长混淆）：} 半周长 $p$ 是周长的一半，切勿直接用周长代入。
		\item \textbf{检查项（系数检验）：} 使用公式时检查是否遗漏 $\frac12$ 系数。
	\end{itemize}
\end{summarybox}
