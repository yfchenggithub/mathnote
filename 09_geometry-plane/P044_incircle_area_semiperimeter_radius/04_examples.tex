\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a=5$，$b=12$，$c=13$，求内切圆半径 $r$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求半周长）：}] $p = \frac{5+12+13}{2}$，得 $p=15$。
			\par\textbf{理由：} 半周长定义 $p=\frac{a+b+c}{2}$。
		\item[\textbf{第二步（求面积）：}] 因为 $5^2+12^2=13^2$，所以三角形是直角三角形，直角边 $5$ 和 $12$，面积 $S = \frac12 \times 5 \times 12$，得 $S=30$。
			\par\textbf{理由：} 勾股定理判定直角三角形，直角三角形面积公式。
		\item[\textbf{第三步（求半径）：}] 由 $r = \frac{S}{p}$ 得 $r = \frac{30}{15}$，即 $r=2$。
			\par\textbf{理由：} 内切圆半径公式 $r = S/p$ 直接应用。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{所求内切圆半径为 $2$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知三角形 $ABC$ 的面积 $S=24$，半周长 $p=12$，求内切圆半径 $r$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出公式）：}] $S = p r$。
			\par\textbf{理由：} 三角形内切圆面积公式。
		\item[\textbf{第二步（变形求解）：}] $r = \frac{S}{p}$。
			\par\textbf{理由：} 将公式变形，除以 $p$。
		\item[\textbf{第三步（代入数值）：}] $r = \frac{24}{12}$，即 $r=2$。
			\par\textbf{理由：} 代入 $S=24,p=12$，得 $r=2$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$r=2$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a=13$，$b=14$，$c=15$，求内切圆的面积。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求半周长）：}] $p = \frac{13+14+15}{2}$，得 $p=21$。
			\par\textbf{理由：} 半周长定义。
		\item[\textbf{第二步（求三角形面积）：}] 由海伦公式 $S=\sqrt{p(p-a)(p-b)(p-c)}$，得
			\[
				\begin{aligned}
					S &= \sqrt{p(p-a)(p-b)(p-c)} \\
					&= \sqrt{21 \times 8 \times 7 \times 6} \\
					&= \sqrt{7056} \\
					&= 84.
			\end{aligned}
			\]
			\par\textbf{理由：} 海伦公式适用于任意三角形。
		\item[\textbf{第三步（求内切圆半径与面积）：}] 由 $r = \frac{S}{p}$ 得 $r = \frac{84}{21}$，即 $r=4$；内切圆面积为
			\[
				\begin{aligned}
					\pi r^2 &= \pi \times 16 \\
					&= 16\pi.
			\end{aligned}
			\]
			\par\textbf{理由：} 内切圆半径公式 $r=S/p$ 及圆面积公式 $S=\pi r^2$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{内切圆面积为 $16\pi$}
\end{examplebox}
