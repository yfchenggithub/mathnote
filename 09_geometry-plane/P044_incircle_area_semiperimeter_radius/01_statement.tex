\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形要素）：}}] 设 $\triangle ABC$ 的三边长分别为 $a,b,c$，面积为 $S$，内切圆半径为 $r$，半周长 $p=\frac{a+b+c}{2}$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形面积等于半周长与内切圆半径之积。
			\[
				S = pr.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			内切圆半径可通过面积除以半周长求出。
			\[
				r = \frac{S}{p}.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{适用范围：}}
	适用于任意三角形（锐角、直角、钝角）。
\end{statementbox}
