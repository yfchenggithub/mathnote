	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两圆相交）：}}] 两圆 $C_1: x^2+y^2+D_1x+E_1y+F_1=0$ 与 $C_2: x^2+y^2+D_2x+E_2y+F_2=0$ 相交。
		\item[\textbf{\textcolor{CStatement}{条件2（直线与圆相交）：}}] 直线 $l: Ax+By+C=0$ 与圆 $C: x^2+y^2+Dx+Ey+F=0$ 相交。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两圆交点的圆系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过两圆所有公共点的圆（除 $C_2$ 外）可表示为两圆方程的线性组合。
			\[
				(x^2+y^2+D_1x+E_1y+F_1)+\lambda(x^2+y^2+D_2x+E_2y+F_2)=0 \quad (\lambda\in\mathbb{R},\ \lambda\neq -1).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（直线-圆系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			过直线与圆公共点的所有圆可表示为圆方程与直线方程的线性组合。
			\[
				x^2+y^2+Dx+Ey+F+\mu(Ax+By+C)=0 \quad (\mu\in\mathbb{R}).
			\]
	\end{description}

\end{statementbox}
