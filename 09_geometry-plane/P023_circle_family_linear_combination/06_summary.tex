\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆系方程用一个参数统一表达所有过两圆（或圆与直线）公共点的圆。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（相交）：} 两圆相交或直线与圆相交（有两个公共点）。
		\item \textbf{条件2（参数限制）：} 两圆系中 $\lambda\neq -1$；直线-圆系中 $\mu\in\mathbb{R}$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（参数范围）：} 务必检查 $\lambda\neq -1$，避免将公共弦混为圆。
		\item \textbf{检查项（二次项系数）：} 确认 $x^2$ 与 $y^2$ 系数相等且非零，保证是圆。
	\end{itemize}
\end{summarybox}
