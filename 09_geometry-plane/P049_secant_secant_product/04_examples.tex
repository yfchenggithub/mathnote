\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	从圆外一点$P$引两条割线$PAB$和$PCD$，已知$PA=3$，$AB=5$，$PC=4$，求$PD$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（明确线段）：}] 在割线$PAB$上，$A$为近交点，$B$为远交点，故$PB=PA+AB=3+5=8$。
			\[
				\begin{aligned}
					PB &= PA+AB \\
					&= 3+5 \\
					&= 8
			\end{aligned}
			\]
			\par\textbf{理由：} 点A在圆与P之间，因此PB=PA+AB。
		\item[\textbf{第二步（套用定理）：}] 由割线定理得$PA\cdot PB = PC\cdot PD$。
			\[
				PA\cdot PB = PC\cdot PD
			\]
			\par\textbf{理由：} 割线定理直接套用。
		\item[\textbf{第三步（代值求解）：}] 代入数值$3\times 8 = 4\times PD$，解得$PD=6$。
			\[
				3\times 8 = 4\times PD \Rightarrow PD = 6
			\]
			\par\textbf{理由：} 解一元一次方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PD=6$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	从圆外一点$P$引两条割线$PAB$和$PCD$，已知$PA=2$，$AB=4$，$PC=3$，求$CD$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}] 设$CD=x$，则$PD=PC+CD=3+x$。
			\[
				CD = x,\quad PD = 3+x
			\]
			\par\textbf{理由：} 用未知数表示待求线段。
		\item[\textbf{第二步（求PB）：}] 在割线$PAB$上，$PB=PA+AB=2+4=6$。
			\[
				\begin{aligned}
					PB &= PA+AB \\
					&= 2+4 \\
					&= 6
			\end{aligned}
			\]
			\par\textbf{理由：} 割线上近远点关系。
		\item[\textbf{第三步（列方程）：}] 由割线定理$PA\cdot PB = PC\cdot PD$，代入得$2\times 6 = 3\times (3+x)$。
			\[
				PA\cdot PB = PC\cdot PD \Rightarrow 2\times 6 = 3\times (3+x)
			\]
			\par\textbf{理由：} 代入已得数值和未知数。
		\item[\textbf{第四步（解方程）：}] 解方程$12 = 9+3x$，得$x=1$，即$CD=1$。
			\[
				12 = 9+3x \Rightarrow x = 1
			\]
			\par\textbf{理由：} 移项合并求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$CD=1$}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	从圆外一点$P$引圆的一条切线$PT$和一条割线$PAB$，已知$PT=6$，$PA=3$，求$AB$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}] 设$PB=x$，则$AB=PB-PA=x-3$。
			\[
				PB = x,
				\qquad
				AB = x-3
			\]
			\par\textbf{理由：} 便于表示未知量。
		\item[\textbf{第二步（套用推论）：}] 由切割线定理（割线定理的切线情形）得$PT^2 = PA\cdot PB$，即$36 = 3x$。
			\[
				PT^2 = PA\cdot PB \Rightarrow 36 = 3x
			\]
			\par\textbf{理由：} 圆外一点，切线长平方等于割线上两段之积。
		\item[\textbf{第三步（解出PB）：}] 解得$x=12$，即$PB=12$。
			\[
				x = 12
			\]
			\par\textbf{理由：} 解方程。
		\item[\textbf{第四步（求AB）：}] $AB = PB - PA = 12-3 = 9$。
			\[
				\begin{aligned}
					AB &= PB - PA \\
					&= 12-3 \\
					&= 9
			\end{aligned}
			\]
			\par\textbf{理由：} 割线上两段之差。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AB=9$}
\end{examplebox}
