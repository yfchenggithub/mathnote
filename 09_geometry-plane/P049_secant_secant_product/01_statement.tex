\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（圆外一点）：}}]从圆外一点$P$引两条割线，分别交圆于$A,B$和$C,D$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（割线定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同一圆外点引两割线，每条割线上两线段乘积相等。
			\[
				\boxed{PA\cdot PB = PC\cdot PD}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（切线情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若其中一条割线变成切线，则切线长的平方等于另一割线上两段乘积。
			\[
				\boxed{PT^2 = PA\cdot PB}
			\]
	\end{description}
\end{statementbox}
