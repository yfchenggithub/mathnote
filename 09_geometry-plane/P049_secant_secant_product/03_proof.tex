\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用圆内接四边形外角等于内对角构造相似三角形。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（角等构造）：}] 连接$AD$、$BC$，由圆内接四边形$ABCD$的外角等于内对角，得
			\[
				\angle PAD = \angle BCD,
				\qquad
				\angle PDA = \angle CBA
			\]
			\par\textbf{理由：} 四边形$ABCD$共圆，外角等于内对角。
		\item[\textbf{步骤二（相似证明）：}] 在$\triangle PAD$和$\triangle PCB$中，有两对角对应相等，所以
			\[
				\triangle PAD \sim \triangle PCB
			\]
			\par\textbf{理由：} 两角分别相等的两个三角形相似。
		\item[\textbf{步骤三（比例转换）：}] 由相似三角形对应边成比例，得
			\[
				\frac{PA}{PC} = \frac{PD}{PB}
			\]
			\par\textbf{理由：} 相似三角形对应边比例相等。
		\item[\textbf{步骤四（等积得出）：}] 将比例式化为等积式，得
			\[
				PA \cdot PB = PC \cdot PD
			\]
			\par\textbf{理由：} 对比例式交叉相乘即得等积式。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，对于圆外任意一点$P$和两条割线$PAB$、$PCD$，总有}$PA\cdot PB = PC\cdot PD$，与割线位置无关。
\end{proofbox}
