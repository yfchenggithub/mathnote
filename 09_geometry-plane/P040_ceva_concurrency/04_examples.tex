\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$D,E,F$ 分别在 $BC,CA,AB$ 上，且 $AD,BE,CF$ 共点。已知 $\frac{BD}{DC}=\frac{2}{3}$，$\frac{CE}{EA}=4$，求 $\frac{AF}{FB}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认条件）：}]
			点 $D,E,F$ 分别在三边上，且三线共点，满足塞瓦定理条件.
			\par\textbf{理由：} 塞瓦定理要求点共线且三线共点.
		\item[\textbf{第二步（应用定理）：}]
			由塞瓦定理：
			\[
				\frac{BD}{DC}\cdot\frac{CE}{EA}\cdot\frac{AF}{FB}=1.
			\]
			\par\textbf{理由：} 直接套用定理结论.
		\item[\textbf{第三步（代入计算）：}]
			代入已知比：
			\[
				\frac{2}{3}\times 4\times\frac{AF}{FB}=1 \;\Longrightarrow\; \frac{8}{3}\cdot\frac{AF}{FB}=1.
			\]
			\par\textbf{理由：} 分数乘法计算.
		\item[\textbf{第四步（求解比例）：}]
			\[
				\frac{AF}{FB}=\frac{3}{8}.
			\]
			\par\textbf{理由：} 方程两边同乘 $\frac{3}{8}$ 或解一次方程.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\displaystyle\frac{AF}{FB}=\frac{3}{8}$，即 $AF:FB=3:8$.}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$D$ 在 $BC$ 上，$E$ 在 $CA$ 的延长线上，$F$ 在 $AB$ 上，且 $AD,BE,CF$ 共点。已知 $\frac{BD}{DC}=2$，$\frac{CE}{EA}=-3$，求 $\frac{AF}{FB}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认条件）：}]
			点 $E$ 在延长线上，但塞瓦定理对有向线段仍成立.
			\par\textbf{理由：} 塞瓦定理使用有向比，延长线上点对应比值取负.
		\item[\textbf{第二步（应用定理）：}]
			由塞瓦定理：
			\[
				\frac{BD}{DC}\cdot\frac{CE}{EA}\cdot\frac{AF}{FB}=1.
			\]
			\par\textbf{理由：} 直接套用.
		\item[\textbf{第三步（代入计算）：}]
			代入已知：
			\[
				2\times(-3)\times\frac{AF}{FB}=1 \;\Longrightarrow\; -6\cdot\frac{AF}{FB}=1.
			\]
			\par\textbf{理由：} 乘法.
		\item[\textbf{第四步（求解比例）：}]
			\[
				\frac{AF}{FB}=-\frac{1}{6}.
			\]
			\par\textbf{理由：} 方程两边同除以 $-6$.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\displaystyle\frac{AF}{FB}=-\frac{1}{6}$，表示 $AF$ 与 $FB$ 方向相反.}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$D,E,F$ 分别在 $BC,CA,AB$ 上，且 $AD,BE,CF$ 共点。已知 $AD$ 是 $BC$ 边上的中线，故 $BD=DC$；$BE$ 是 $\angle B$ 的平分线，且 $BC=2AB$。求 $\frac{AF}{FB}$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（中线条件）：}]
			由中线得 $\frac{BD}{DC}=1$.
			\par\textbf{理由：} $D$ 为 $BC$ 中点，有向线段相等.
		\item[\textbf{第二步（角平分线条件）：}]
			由角平分线定理，$\frac{CE}{EA}=\frac{BC}{BA}$. 又 $BC=2AB$，故 $\frac{CE}{EA}=2$.
			\par\textbf{理由：} 三角形内角平分线分对边成比例.
		\item[\textbf{第三步（代入塞瓦）：}]
			代入定理：
			\[
				1\times 2\times\frac{AF}{FB}=1 \;\Longrightarrow\; \frac{AF}{FB}=\frac12.
			\]
			\par\textbf{理由：} 移项得解.
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\displaystyle\frac{AF}{FB}=\frac12$，即 $AF:FB=1:2$.}
\end{examplebox}
