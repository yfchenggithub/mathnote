	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（点位置）：}}]

		\item[\textbf{\textcolor{CStatement}{条件 2（线共点）：}}]
			直线 $AD,BE,CF$ 共点，且所涉比值均为有向线段比.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（有向比乘积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三边上的分点与顶点连线共点时，有向比例乘积恒为 $1$.
			\[
				\frac{BD}{DC}\cdot\frac{CE}{EA}\cdot\frac{AF}{FB}=1.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（面积比形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			设三线交点为 $P$，用有向面积比表示，乘积也为 $1$.
			\[
				\frac{S_{\triangle ABP}}{S_{\triangle ACP}}\cdot\frac{S_{\triangle BCP}}{S_{\triangle BAP}}\cdot\frac{S_{\triangle CAP}}{S_{\triangle CBP}}=1.
			\]
	\end{description}
