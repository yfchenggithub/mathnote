\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	利用直径所对圆周角为直角，将几何条件转化为向量数量积为零，再坐标化得到方程。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（垂直条件转化）：}] 设点 $P$ 在以 $AB$ 为直径的圆上（异于 $A,B$），则 $\angle APB=90^\circ$，故 $PA\perp PB$。
			\[
				\overrightarrow{PA}\cdot\overrightarrow{PB}=0.
			\]
			\par\textbf{理由：} 直径所对的圆周角是直角。

		\item[\textbf{步骤二（坐标化点积）：}] 设 $P(x,y)$，则 $\overrightarrow{PA}=(x_1-x,\;y_1-y)$，$\overrightarrow{PB}=(x_2-x,\;y_2-y)$。代入点积为零：
			\[
				(x_1-x)(x_2-x)+(y_1-y)(y_2-y)=0.
			\]
			注意到 $(x_1-x)(x_2-x)=(x-x_1)(x-x_2)$，同理 $y$ 项，故得
			\[
				(x-x_1)(x-x_2)+(y-y_1)(y-y_2)=0.
			\]
			\par\textbf{理由：} 向量坐标及运算。

		\item[\textbf{步骤三（反向推导）：}] 若 $P$ 满足上述方程且 $P$ 异于 $A,B$，则 $\overrightarrow{PA}\cdot\overrightarrow{PB}=0$，从而 $\angle APB=90^\circ$。由圆周角定理的逆定理，$P$ 在以 $AB$ 为直径的圆上。
			\[
				\overrightarrow{PA}\cdot\overrightarrow{PB}=0 \iff \angle APB=90^\circ.
			\]
			\par\textbf{理由：} 固定线段 $AB$，张角为直角的点的轨迹是以 $AB$ 为直径的圆。

		\item[\textbf{步骤四（边界补充）：}] 将 $A(x_1,y_1)$ 或 $B(x_2,y_2)$ 代入方程，等号成立，故 $A,B$ 也在方程表示的图形上，即该方程表示以 $AB$ 为直径的整个圆。
			\par\textbf{理由：} 验证表明端点满足方程。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{故 $(x-x_1)(x-x_2)+(y-y_1)(y-y_2)=0$ 是以 $A(x_1,y_1),B(x_2,y_2)$ 为直径端点的圆的方程。}
\end{proofbox}
