\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆形式）}}] 将直径式方程误写为 $(x-x_1)^2+(y-y_1)^2+(x-x_2)^2+(y-y_2)^2=0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（乘积和结构）：}}] 直径式方程为 $(x-x_1)(x-x_2)+(y-y_1)(y-y_2)=0$，源自向量点积。
		\item[\textbf{\textcolor{CTrap}{错因分析（受圆标形影响）：}}] 受圆的标准方程 $(x-a)^2+(y-b)^2=r^2$ 影响，误以为圆方程必须是平方和形式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（端点重合）}}] 当 $A,B$ 重合时，套用直径式方程得 $(x-x_1)^2+(y-y_1)^2=0$，错误认为仍表示圆。
		\item[\textbf{\textcolor{CTrap}{正确理解（需两点相异）：}}] $A,B$ 必须是不同的两个点，否则无法确定直径，方程退化为一点，不表示圆。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略定义基础）：}}] 忽略了“直径”的前提是两点不同，误认为公式对任意两点均成立。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（滥用公式）}}] 认为过点 $A,B$ 的所有圆的方程都可写成 $(x-x_1)(x-x_2)+(y-y_1)(y-y_2)=0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（仅限直径圆）：}}] 该方程仅表示以 $AB$ 为直径的圆，其他过 $A,B$ 的圆方程需使用圆的一般式 $x^2+y^2+Dx+Ey+F=0$，其中 $A,B$ 坐标满足方程确定参数。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}] 不明白该方程由垂直条件推导，只有直径所对圆周角为直角才成立。
	\end{description}
\end{trapbox}
