\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{已知直径两端点 $A(x_1,y_1),B(x_2,y_2)$，则以 $AB$ 为直径的圆的方程为 $(x-x_1)(x-x_2)+(y-y_1)(y-y_2)=0$，等价于 $\overrightarrow{PA}\cdot\overrightarrow{PB}=0$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（直径端点已知）：} 必须已知一条直径的两个端点坐标。
		\item \textbf{条件2（两点不重合）：} 端点 $A$ 与 $B$ 不能是同一个点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（混淆形式）：} 注意方程是乘积和而非平方和，且只适用于直径圆。
		\item \textbf{检查项（代回验证）：} 将 $A,B$ 两点坐标代入方程，左端应为 $0$，可用于快速验证。
	\end{itemize}
\end{summarybox}
