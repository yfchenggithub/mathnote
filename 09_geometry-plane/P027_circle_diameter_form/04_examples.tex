\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(0,0)$ 和 $B(4,0)$，求以 $AB$ 为直径的圆的方程，并指出圆心坐标和半径。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 以 $A(x_1,y_1),B(x_2,y_2)$ 为直径端点的圆方程为 $(x-x_1)(x-x_2)+(y-y_1)(y-y_2)=0$。
			\[
				(x-0)(x-4)+(y-0)(y-0)=0.
			\]
			\par\textbf{理由：} 直接代入 $x_1=0,y_1=0,x_2=4,y_2=0$。

		\item[\textbf{第二步（整理方程）：}] 展开乘积并合并：
			\[
				\begin{aligned}
					x(x-4)+y^2 &= x^2-4x+y^2\\
					&= 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 乘法分配律和合并同类项。

		\item[\textbf{第三步（配方得标准形）：}] 配方得：
			\[
				(x-2)^2 + y^2 = 4.
			\]
			\par\textbf{理由：} 完全平方公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	圆心为 $(2,0)$，半径为 $2$，圆方程为 $(x-2)^2+y^2=4$（或 $x^2+y^2-4x=0$）。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $A(-1,2)$ 和 $B(3,4)$，求以 $AB$ 为直径的圆的方程，并判断点 $M(1,3)$ 是否在该圆上。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 代入直径端点坐标：
			\[
				(x+1)(x-3)+(y-2)(y-4)=0.
			\]
			\par\textbf{理由：} 直径式方程。

		\item[\textbf{第二步（展开化简）：}] 展开并合并同类项：
			\[
				\begin{aligned}
					(x+1)(x-3)+(y-2)(y-4) &= (x^2-2x-3)+(y^2-6y+8)\\
					&= x^2+y^2-2x-6y+5\\
					&= 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 展开多项式并合并同类项。

		\item[\textbf{第三步（验证点）：}] 将 $M(1,3)$ 代入：
			\[
				1^2+3^2-2\cdot1-6\cdot3+5=1+9-2-18+5=-5\neq 0.
			\]
			\par\textbf{理由：} 点在圆上当且仅当坐标满足圆的方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	圆的方程为 $x^2+y^2-2x-6y+5=0$（标准形 $(x-1)^2+(y-3)^2=5$），点 $M$ 是圆心，不在圆上。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知点 $A(-1,0),B(1,0)$，点 $P$ 在圆 $C:(x-3)^2+y^2=1$ 上运动，求 $\overrightarrow{PA}\cdot\overrightarrow{PB}$ 的取值范围。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（用结论转化）：}] 设 $P(x,y)$，由直径式知 $\overrightarrow{PA}\cdot\overrightarrow{PB} = (x+1)(x-1)+y^2$，即 $x^2+y^2-1$。
			\[
				\overrightarrow{PA}\cdot\overrightarrow{PB} = x^2+y^2-1.
			\]
			\par\textbf{理由：} 代入向量点积公式或直接使用圆的直径式方程。

		\item[\textbf{第二步（几何条件分析）：}] 点 $P$ 在圆 $C$ 上，圆心 $C(3,0)$，半径 $r=1$。原点 $O$ 到圆心的距离 $|OC|=3$。点 $P$ 到原点距离 $d=\sqrt{x^2+y^2}$ 的取值范围为
			\[
				d \in [|OC|-r,\; |OC|+r] = [2,4].
			\]
			\par\textbf{理由：} 圆外一点到圆上距离最值。

		\item[\textbf{第三步（计算最值）：}] 故 $x^2+y^2 = d^2 \in [4,16]$，从而
			\[
				\overrightarrow{PA}\cdot\overrightarrow{PB} = (x^2+y^2)-1 \in [3,15].
			\]
			\par\textbf{理由：} 单调性。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\overrightarrow{PA}\cdot\overrightarrow{PB}$ 的取值范围为 $[3,15]$。
\end{examplebox}
