\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知两直线 $l_1:3x+4y-5=0$，$l_2:3x+4y+15=0$，求 $l_1$ 与 $l_2$ 之间的距离。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认系数相同）：}] 两直线 $A=3,B=4$ 相同，常数项 $C_1=-5,C_2=15$。
			\par\textbf{理由：} 应用公式前必须确认 $A,B$ 对应相等。
		\item[\textbf{第二步（代入公式计算）：}]
			\[
				\begin{aligned}
					d &= \frac{|C_1-C_2|}{\sqrt{A^{2}+B^{2}}} \\
					&= \frac{| -5 - 15|}{\sqrt{3^{2}+4^{2}}} \\
					&= \frac{20}{5} \\
					&= 4.
			\end{aligned}
			\]
			\par\textbf{理由：} 分子计算 $| -20| = 20$，分母 $\sqrt{9+16}=5$。
		\item[\textbf{第三步（写出答案）：}] 所以 $l_1$ 与 $l_2$ 之间的距离为 $4$。
			\par\textbf{理由：} 结果已化简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d=4$}.

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知两直线 $l_1:y=2x+1$，$l_2:y=2x-3$，求它们之间的距离。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化为一般式）：}] 将 $l_1,l_2$ 分别化为 $2x-y+1=0$ 和 $2x-y-3=0$。
			\par\textbf{理由：} 公式要求直线方程为 $Ax+By+C=0$ 形式，且系数对应相等。
		\item[\textbf{第二步（确定系数）：}] $A=2,B=-1$，$C_1=1,C_2=-3$。
			\par\textbf{理由：} 从一般式读出系数。
		\item[\textbf{第三步（代入公式）：}]
			\[
				\begin{aligned}
					d &= \frac{|1-(-3)|}{\sqrt{2^{2}+(-1)^{2}}} \\
					&= \frac{4}{\sqrt{5}} \\
					&= \frac{4\sqrt{5}}{5}.
			\end{aligned}
			\]
			\par\textbf{理由：} 分母 $\sqrt{4+1}=\sqrt{5}$，分子绝对值 $4$，有理化得 $\frac{4\sqrt{5}}{5}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$d = \dfrac{4\sqrt{5}}{5}$}.

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知两平行直线 $l_1:3x+4y+m=0$ 和 $l_2:3x+4y-2=0$ 的距离等于 $2$，求实数 $m$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别系数）：}] $A=3,B=4$，$C_1=m,C_2=-2$。
			\par\textbf{理由：} 两直线平行，系数 $A,B$ 相同。
		\item[\textbf{第二步（建立方程）：}] 由距离公式得 $d = \dfrac{|m-(-2)|}{\sqrt{3^{2}+4^{2}}}$，即 $\dfrac{|m+2|}{5}$。根据条件 $d=2$，故 $\dfrac{|m+2|}{5}=2$。
			\par\textbf{理由：} 代入公式列出方程。
		\item[\textbf{第三步（解绝对值方程）：}] $|m+2|=10$，所以 $m+2=10$ 或 $m+2=-10$，解得 $m=8$ 或 $m=-12$。
			\par\textbf{理由：} 绝对值方程的解有两个，均满足 $C_1\neq C_2$，直线平行不重合。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m=8$ 或 $m=-12$}.
\end{examplebox}
