\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（系数对应相同）：}}] 两直线均为一般式 $Ax+By+C=0$，且 $A,B$ 取值分别相同。
		\item[\textbf{\textcolor{CStatement}{条件 2（常数项不等）：}}] 两直线常数项 $C_1\neq C_2$，从而两直线平行但不重合。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平行线距离公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平行线间距离等于常数项差绝对值除以系数平方和的算术平方根。
			\[
				d = \frac{|C_1-C_2|}{\sqrt{A^{2}+B^{2}}}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（斜截式距离公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若两直线为 $y=kx+b_1$ 和 $y=kx+b_2$，则距离公式为
			\[
				d = \frac{|b_1-b_2|}{\sqrt{k^{2}+1}}.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{参数限制：}}
	$A$ 与 $B$ 不能同时为零，否则 $A^2+B^2=0$ 使公式分母无意义。
\end{statementbox}
