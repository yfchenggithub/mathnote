\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略平行前提）：}}] 对于 $l_1:3x+4y-5=0$，$l_2:4x-3y+2=0$，错误计算 $d=\dfrac{| -5-2|}{\sqrt{3^2+4^2}}$，得 $\dfrac{7}{5}$，但两直线不平行，公式不适用。
		\item[\textbf{\textcolor{CTrap}{正确理解（先验平行条件）：}}] 使用公式前必须验证两直线平行（系数 $A,B$ 成比例），否则不能使用。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件缺失）：}}] 学生对公式适用条件不够关注，只记形式，不顾前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（系数未化简）：}}] 对于 $l_1:3x+4y-5=0$，$l_2:6x+8y+1=0$，错误计算 $d=\dfrac{| -5-1|}{\sqrt{3^2+4^2}}$，得 $\dfrac{6}{5}$，但正确应将 $l_2$ 化简为 $3x+4y+0.5=0$，计算得 $d=\dfrac{|-5-0.5|}{5}$，即 $1.1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数化归）：}}] 若两直线系数成比例，应先化为相同 $A,B$，再读取 $C$ 计算距离。
		\item[\textbf{\textcolor{CTrap}{错因分析（比例混淆）：}}] 忽略系数倍率，未约简就代入公式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（绝对值遗漏）：}}] 计算时直接写 $d=\dfrac{C_1-C_2}{\sqrt{A^2+B^2}}$，未加绝对值。
		\item[\textbf{\textcolor{CTrap}{正确理解（非负性）：}}] 距离为非负数，分子必须带绝对值符号 $|C_1-C_2|$。
		\item[\textbf{\textcolor{CTrap}{错因分析（几何意义模糊）：}}] 未理解距离的几何意义，或粗心。
	\end{description}
\end{trapbox}
