\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式记忆混淆）}}] 容易记错成 $c^{2}m + b^{2}n = a(d^{2}+mn)$，即 $c^{2}$ 乘错了分段。
		\item[\textbf{\textcolor{CTrap}{正确理解（对应规则）：}}] 公式中 $c^{2}$ 应乘 $n$（$c$ 对的边是 $AB$，其对侧分点是 $D$ 到 $C$），$b^{2}$ 乘 $m$。
		\item[\textbf{\textcolor{CTrap}{错因分析（对称错觉）：}}] 认为 $c$ 对应 $m$ 是因为 $c$ 在 $AB$ 边靠近 $B$，但实际公式由余弦定理推导，$c^{2}$ 出现在 $\triangle ABD$ 中，乘 $n$ 才能消元。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略长度条件）}}] 使用公式时直接代入 $m,n$，但未检查 $BD+CD=BC$，导致 $n$ 值错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（验证条件）：}}] 在应用公式前必须确认 $D$ 在线段 $BC$ 上，且 $m+n=a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（惯性代入）：}}] 只关注公式形式，忘记定理前提是 $D$ 在 $BC$ 边上且 $BC$ 被分为两段。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（中线系数错误）}}] 在中点情形，错误记为 $d^{2} = \frac{b^{2}+c^{2}}{2} - \frac{a^{2}}{4}$ 或其它错误系数。
		\item[\textbf{\textcolor{CTrap}{正确理解（标准形式）：}}] 中线公式应为 $d^{2} = \frac{2(b^{2}+c^{2}) - a^{2}}{4}$，源于斯特瓦尔特定理令 $m=n=\frac{a}{2}$ 推得。
		\item[\textbf{\textcolor{CTrap}{错因分析（机械记忆）：}}] 记不住系数 $2$ 和 $4$，需要推导或加强记忆。
	\end{description}
\end{trapbox}
