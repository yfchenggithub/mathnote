	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AB=13$，$AC=15$，$BC=14$，点 $D$ 在 $BC$ 上且 $BD=5$，求 $AD$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认变量）：}] 对照斯特瓦尔特定理，记 $c=AB=13$，$b=AC=15$，$a=BC=14$，$m=BD=5$，$n=CD=BC-BD=9$。
			\par\textbf{理由：} 准确对应字母，确保符号无误。

		\item[\textbf{第二步（代入公式并计算）：}] 代入公式 $d^{2} = \frac{c^{2}n + b^{2}m}{a} - mn$，进行化简。
			\[
				\begin{aligned}
					d^{2} &= \frac{13^{2}\times 9 + 15^{2}\times 5}{14} - 5\times 9 \\
					&= \frac{1521 + 1125}{14} - 45 \\
					&= \frac{2646}{14} - 45 \\
					&= 189 - 45 \\
					&= 144.
			\end{aligned}
			\]
			\par\textbf{理由：} 先计算乘方，再乘除，最后减法。

		\item[\textbf{第三步（开平方求 $AD$）：}] 由 $d^{2}=144$，且 $d>0$，得 $d=\sqrt{144}=12$。
			\par\textbf{理由：} 线段长度取正。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$AD=12$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在等腰三角形 $ABC$ 中，$AB=AC=5$，$BC=6$，点 $D$ 在 $BC$ 上，且 $AD=\sqrt{17}$，求 $BD$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}] 设 $BD=x$，则 $CD=6-x$，且 $0<x<6$。
			\par\textbf{理由：} 用未知数表示 $m$ 和 $n$。

		\item[\textbf{第二步（代入定理公式）：}] 由斯特瓦尔特定理，代入 $c=5$，$b=5$，$a=6$，$m=x$，$n=6-x$，得
			\[
				d^{2} = \frac{25(6-x) + 25x}{6} - x(6-x).
			\]
			计算分子得 $150$，所以
			\[
				d^{2} = \frac{150}{6} - (6x - x^{2}).
			\]
			化简得
			\[
				d^{2} = 25 - 6x + x^{2}.
			\]
			\par\textbf{理由：} 代入后逐步化简。

		\item[\textbf{第三步（建立方程并求解）：}] 将已知 $d^{2}=17$ 代入，得
			\[
				x^{2} - 6x + 25 = 17 \quad \Rightarrow \quad x^{2} - 6x + 8 = 0.
			\]
			解方程得 $(x-2)(x-4)=0$，所以 $x=2$ 或 $x=4$。
			\par\textbf{理由：} 因式分解或求根公式。

		\item[\textbf{第四步（验证合理性）：}] 两个解均在 $(0,6)$ 内，故 $BD=2$ 或 $4$。
			\par\textbf{理由：} 等腰三角形对称性，两点关于中线对称。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BD$ 的长为 $2$ 或 $4$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AB=6$，$AC=8$，$AD$ 是 $BC$ 边上的中线，且 $AD=5$，求 $BC$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别中点条件）：}] 因为 $AD$ 是中线，所以 $D$ 为 $BC$ 中点。设 $BC=a$，则 $BD=CD=\frac{a}{2}$。
			\par\textbf{理由：} 中点将底边平分为两段。

		\item[\textbf{第二步（代入定理计算）：}] 代入定理公式，得
			\[
				5^{2} = \frac{6^{2}\cdot\frac{a}{2} + 8^{2}\cdot\frac{a}{2}}{a} - \frac{a}{2}\cdot\frac{a}{2}.
			\]
			化简分子：
			\[
				\begin{aligned}
					6^{2}\cdot\frac{a}{2}+8^{2}\cdot\frac{a}{2} &= \frac{a}{2}(36+64) \\
					&= 50a,
			\end{aligned}
			\]
			除以 $a$ 得 $50$，所以
			\[
				25 = 50 - \frac{a^{2}}{4}.
			\]
			\par\textbf{理由：} 先算乘方，再通分化简。

		\item[\textbf{第三步（解方程求 $a$）：}] 移项得 $\frac{a^{2}}{4} = 25$，所以 $a^{2}=100$，$a=10$（舍去负值）。
			\par\textbf{理由：} 方程求解，取正值。

		\item[\textbf{第四步（结论）：}] 因此 $BC=10$。
			\par\textbf{理由：} 结果符合题意。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BC=10$。}
