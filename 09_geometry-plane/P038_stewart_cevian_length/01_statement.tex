\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（点分线段）：}}] 在 $\triangle ABC$ 中，点 $D$ 在线段 $BC$ 上。记 $BC=a$，$CA=b$，$AB=c$，$BD=m$，$CD=n$，满足 $a=m+n$。设 $AD=d$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（斯特瓦尔特定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知三角形三边和分点位置，可求分点到对顶点的线段长度。
			\[
				c^{2}n + b^{2}m = a(d^{2}+mn),
			\]
			或等价地
			\[
				d^{2} = \frac{c^{2}n + b^{2}m}{a} - mn.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（中线公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $D$ 为 $BC$ 中点时，$m=n=\frac{a}{2}$，公式简化为中线长公式。
			\[
				d^{2} = \frac{2(b^{2}+c^{2}) - a^{2}}{4}.
			\]
	\end{description}

\end{statementbox}
