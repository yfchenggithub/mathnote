\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（边角混淆）}}] 误将斜边当作直角边代入公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（辨认方法）：}}] 在直角三角形中，直角所对的边是斜边，长度最大；公式 $a^2+b^2=c^2$ 中 $c$ 必须是斜边。
		\item[\textbf{\textcolor{CTrap}{错因分析（根源所在）：}}] 没有抓住“直角对斜边”的基本关系，对边角对应关系不熟练。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忘记开方）}}] 由 $c^2=a^2+b^2$ 直接得出 $c=a^2+b^2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（开方步骤）：}}] 由 $c^2=a^2+b^2$ 得
		      \[
			      c=\sqrt{a^2+b^2},
		      \]
		      边长必须是正数。
		\item[\textbf{\textcolor{CTrap}{错因分析（运算疏忽）：}}] 混淆了平方和与边长本身的关系。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（乱用定理）}}] 在非直角三角形中直接应用勾股定理。
		\item[\textbf{\textcolor{CTrap}{正确理解（先验直角）：}}] 使用勾股定理前，需要确认三角形是直角三角形，可由已知条件判断，也可用勾股定理的逆定理判定。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视条件）：}}] 定理的适用范围是直角三角形，不能任意推广。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（判定时没找最长边）}}] 已知三角形三边时，随便把一条边当成 $c$ 去检验。
		\item[\textbf{\textcolor{CTrap}{正确理解（最长边原则）：}}] 判定直角三角形时，必须先把最长边记为 $c$，其余两边记为 $a,b$。若
		      \[
			      a^2+b^2=c^2,
		      \]
		      则该三角形为直角三角形；否则不是直角三角形。
		\item[\textbf{\textcolor{CTrap}{错因分析（角色不清）：}}] 勾股定理中的 $c$ 不是任意边，而是斜边；在三角形三边判定中，斜边只能对应最长边。
	\end{description}
\end{trapbox}
