\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{直角三角形中，两条直角边的平方和等于斜边的平方。}
	\[
		a^2+b^2=c^2.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（直角前提）：} 三角形必须是直角三角形。
		\item \textbf{条件 2（斜边明确）：} 公式中的 $c$ 必须表示斜边，也就是直角所对的边。
		\item \textbf{条件 3（判定直角）：} 若用来判定直角三角形，必须先把最长边记为 $c$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{知二求一}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{已知两条直角边，求斜边：}
		      \[
			      c=\sqrt{a^2+b^2}.
		      \]
		\item \textbf{已知斜边和一条直角边，求另一条直角边：}
		      \[
			      a=\sqrt{c^2-b^2}
			      \quad\text{或}\quad
			      b=\sqrt{c^2-a^2}.
		      \]
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（记错公式）：} 不要把 $a^2+b^2=c^2$ 误写成 $a+b=c$ 或 $a^2+b^2=c$。
		\item \textbf{检查项（开方取值）：} 求得边长平方后必须开方，并取正值。
		\item \textbf{判定项（最长边原则）：} 已知三边判定直角时，必须让最长边对应公式中的 $c$。
	\end{itemize}
\end{summarybox}
