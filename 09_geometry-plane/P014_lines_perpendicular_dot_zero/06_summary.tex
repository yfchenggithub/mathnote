\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{两直线垂直的充要条件：$A_1A_2+B_1B_2=0$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（直线为一般式）：}$l_1:A_1x+B_1y+C_1=0$，$l_2:A_2x+B_2y+C_2=0$。
		\item \textbf{条件2（系数不全为零）：}$A_1,B_1$不全为零，$A_2,B_2$不全为零。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分平行条件）：}$A_1A_2+B_1B_2=0$是垂直；$A_1B_2-A_2B_1=0$是平行或重合，判断平行时需额外留意常数项。
		\item \textbf{检查项（斜率不存在时）：}无需化为斜截式，直接使用一般式条件更安全。
	\end{itemize}

\end{summarybox}
