	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（系数不全为零）：}}]
			两直线均采用一般式$A_1x+B_1y+C_1=0$与$A_2x+B_2y+C_2=0$，且$A_1,B_1$不全为零，$A_2,B_2$不全为零。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（垂直充要条件）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两直线垂直当且仅当它们的法向量互相垂直。
			\[
				A_1A_2+B_1B_2=0.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（斜率乘积形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当两直线的斜率都存在时，垂直条件等价于斜率乘积为$-1$。
			\[
				k_1k_2=-1.
			\]
	\end{description}

\end{statementbox}
