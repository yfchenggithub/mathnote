\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆垂直平行）}}]
			学生常将垂直条件误记为$A_1B_2-A_2B_1=0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分两组条件）：}}]
			垂直的充要条件是$A_1A_2+B_1B_2=0$；$A_1B_2-A_2B_1=0$是平行或重合的条件（严格平行还需验证常数项）。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆方式不当）：}}]
			未理解条件来源，仅靠机械记忆导致混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略斜率不存在）}}]
			直接用$k_1k_2=-1$时忽略了斜率不存在的情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论斜率）：}}]
			当$B_1=0$或$B_2=0$时，直线垂直于坐标轴，垂直条件应直接使用$A_1A_2+B_1B_2=0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
			习惯用斜截式而忽略了一般式的普适性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（系数对应错误）}}]
			计算垂直条件时，将$A_1$与$B_2$相乘等。
		\item[\textbf{\textcolor{CTrap}{正确理解（按位对应系数）：}}]
			应计算对应方程中$x$和$y$的系数乘积之和，即$A_1A_2+B_1B_2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（粗心大意）：}}]
			未养成规范书写系数的习惯。
	\end{description}

\end{trapbox}
