\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n通过连接辅助线构造圆周角，利用圆周角定理证明三角形相似，进而由相似三角形的对应边成比例推导出等积式.\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（连接辅助线）：}]
			连接$AC$和$BD$.\n
			\par\textbf{理由：}\n 构造圆周角$\angle A$、$\angle D$，以便利用同弧所对的圆周角相等.\n\n
		\item[\textbf{步骤二（角相等）：}]
			因为$\angle A$与$\angle D$均对应弧$BC$，所以$\angle A=\angle D$；又$\angle APC$与$\angle DPB$为对顶角，故$\angle APC=\angle DPB$.\n
			\par\textbf{理由：}\n 同弧所对的圆周角相等；对顶角相等.\n\n
		\item[\textbf{步骤三（三角形相似）：}]
			在$\triangle APC$和$\triangle DPB$中，$\angle A=\angle D$，$\angle APC=\angle DPB$，所以$\triangle APC\sim\triangle DPB$.\n
			\par\textbf{理由：}\n 两角对应相等，三角形相似.\n\n
		\item[\textbf{步骤四（比例乘积）：}]
			由$\triangle APC\sim\triangle DPB$，得对应边比例$\frac{PA}{PD}=\frac{PC}{PB}$，两边同乘以$PD\cdot PB$得$PA\cdot PB=PC\cdot PD$.\n
			\par\textbf{理由：}\n 相似三角形对应边成比例；等式变形.\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n由此证明了圆内两弦$AB$与$CD$交于点$P$时，恒有$PA\cdot PB=PC\cdot PD$. \textbf{即相交弦定理成立.}\n
\end{proofbox}
