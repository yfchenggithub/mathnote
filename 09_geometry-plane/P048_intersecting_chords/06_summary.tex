\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{圆内两弦相交，交点将每条弦分成两段，则两弦上的线段乘积相等：$PA\cdot PB = PC\cdot PD$.}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（圆内交点）：}两条弦的交点必须在圆内部；若在圆外，则应使用割线定理.
		\item \textbf{条件2（共点两弦）：}两条弦必须共用同一个交点，且都是圆$O$的弦.
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分内外）：}圆内相交弦乘积相等，圆外割线乘积形式相同，但定理名称和使用条件不同.
		\item \textbf{检查项（对应验证）：}写等积式时检查每条弦是否用自己的两段相乘，不要交叉错位.
	\end{itemize}
\end{summarybox}
