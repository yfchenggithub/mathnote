\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n\textbf{圆内两弦相交，交点将每条弦分成两段，则两弦上的线段乘积相等：$PA\cdot PB = PC\cdot PD$.}\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（圆内交点）：} 两条弦的交点必须在圆内部；若在圆外则不直接适用（需用割线定理）.\n
		\item \textbf{条件2（共点两弦）：} 两条弦必须共用同一个交点，且都是圆$O$的弦.\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（区分内外）：} 圆内相交弦乘积相等，圆外割线乘积形式相同，但定理名称和含义不同.\n
		\item \textbf{检查项（对应验证）：} 写等积式时检查每条弦是否用自己的两段相乘，不要交叉错位.\n
	\end{itemize}
	\n
\end{summarybox}
