	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略圆内条件）}}]
			解题时未确认点$P$是否在圆内，就直接列出$PA\cdot PB=PC\cdot PD$，导致在圆外时仍用“相交弦”的说法.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（区分圆内圆外）：}}]
			当$P$在圆外时，应称为“割线定理”，其结论形式相同但推导不同，使用时需明确条件.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（定理混淆）：}}]
			混淆圆内相交弦与圆外割线，实质上是未区分圆幂定理的不同情况.\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（对应关系错误）}}]
			将乘积关系误写为$PA\cdot PC=PB\cdot PD$或$PA\cdot PD=PB\cdot PC$.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（同弦乘积）：}}]
			正确形式是每一条弦上的两段乘积相等：$PA\cdot PB=PC\cdot PD$.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（比例意识弱）：}}]
			未从证明中理解$\triangle APC\sim\triangle DPB$的对应边关系，凭感觉写等积式.\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（负根舍去）}}]
			设未知数列方程时，解出负根未舍弃，直接作为线段长度.\n
		\item[\textbf{\textcolor{CTrap}{正确理解（取正舍负）：}}]
			线段长度为正值，凡解出负数必须舍去，只保留正数解.\n
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略检验）：}}]
			缺乏实际意义验证，仅关注代数解忽略几何意义.\n
	\end{description}
