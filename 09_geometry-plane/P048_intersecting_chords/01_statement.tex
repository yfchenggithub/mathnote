	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（圆内相交）：}}]
			圆$O$内，两条弦$AB$与$CD$相交于一点$P$，且$P$在圆$O$内部.\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（等积关系）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			圆内两弦交于一点，则交点分弦所得的两线段乘积相等.\n
			\[
				\n PA \cdot PB = PC \cdot PD.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{常见变形（圆外割线）：}}]
			\n
			\textbf{\textcolor{CStatement}{直观描述：}}
			两条割线交于圆外一点$P$，则$P$到每条割线与圆的两交点之线段乘积相等.\n
			\[
				\n PA \cdot PB = PC \cdot PD.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
