\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 在圆$O$中，弦$AB$与$CD$相交于点$P$，且$PA=3,\ PB=4,\ PC=2$，求$PD$的长.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（判断条件）：}]
			$AB$和$CD$是圆内两条弦，交于$P$，符合相交弦定理的使用条件.\n
			\par\textbf{理由：}\n 相交弦定理：圆内两弦相交，交点分弦所得线段乘积相等.\n
		\item[\textbf{第二步（列等积式）：}]
			由相交弦定理，有$PA\cdot PB = PC\cdot PD$.\n
			\par\textbf{理由：}\n 定理内容直接给出等量关系.\n
		\item[\textbf{第三步（代入求解）：}]
			将已知数值代入得$3\times 4 = 2\times PD$，即$12=2PD$，所以$PD=6$.\n
			\par\textbf{理由：}\n 代数运算解出未知量.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$PD=6$}.\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 在圆$O$中，弦$AB$与$CD$交于$P$，$PA:PB=2:3$，$PC=4,\ PD=6$，求弦$AB$的长.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（设参数）：}]
			设$PA=2k$，$PB=3k$（$k>0$）.\n
			\par\textbf{理由：}\n 利用比例引入参数$k$，方便表示未知量.\n
		\item[\textbf{第二步（列方程）：}]
			由相交弦定理，$PA\cdot PB = PC\cdot PD$，即$2k\cdot 3k = 4\times 6$，得$6k^2=24$.\n
			\par\textbf{理由：}\n 定理提供等量关系，代入后得到关于$k$的方程.\n
		\item[\textbf{第三步（求解回代）：}]
			解$6k^2=24$，得$k^2=4$，$k=2$（$k>0$），故$PA=4,\ PB=6$，所以$AB=PA+PB=10$.\n
			\par\textbf{理由：}\n 解方程后回代求线段长度.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$AB=10$}.\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 在圆$O$中，弦$AB$与$CD$交于$P$，已知$PA=4,\ PB=9$，且$CD=12$，求$PC$和$PD$的长.\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（求乘积）：}]
			由相交弦定理，$PA\cdot PB = PC\cdot PD$，得$4\times9=36$，所以$PC\cdot PD=36$.\n
			\par\textbf{理由：}\n 代入已知，得到$PC$与$PD$的乘积.\n
		\item[\textbf{第二步（列方程组）：}]
			又$CD=PC+PD=12$，故$PC$和$PD$是方程$x^2-12x+36=0$的两根.\n
			\par\textbf{理由：}\n 两数和与积，可构造一元二次方程.\n
		\item[\textbf{第三步（解方程）：}]
			解方程得$(x-6)^2=0$，所以$x=6$，即$PC=PD=6$.\n
			\par\textbf{理由：}\n 方程的根即为所求线段长.\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} \textbf{$PC=PD=6$}.\n\n
\end{examplebox}
