\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（点斜式可用）：}}] 已知直线上一点 $(x_0,y_0)$ 且斜率 $k$ 存在。
		\item[\textbf{\textcolor{CStatement}{条件二（斜截式可用）：}}] 已知斜率 $k$ 和 $y$ 截距 $b$，且斜率存在。
		\item[\textbf{\textcolor{CStatement}{条件三（两点式可用）：}}] 已知直线上两点 $(x_1,y_1)$ 和 $(x_2,y_2)$，且 $x_1\neq x_2$，$y_1\neq y_2$。
		\item[\textbf{\textcolor{CStatement}{条件四（截距式可用）：}}] 已知横截距 $a$ 和纵截距 $b$，且 $a\neq0$，$b\neq0$。
		\item[\textbf{\textcolor{CStatement}{条件五（一般式可用）：}}] 任意直线均可使用，$A,B$不同时为零。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（点斜式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用一点和斜率直接写出直线方程。
			\[
				y-y_0 = k(x-x_0)
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（斜截式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用斜率和纵截距直接写出直线方程。
			\[
				y = kx + b
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（两点式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用两点坐标列出比例关系得到方程。
			\[
				\frac{y-y_1}{y_2-y_1} = \frac{x-x_1}{x_2-x_1}
			\]

		\item[\textbf{\textcolor{CStatement}{结论四（截距式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用横、纵截距直接写出方程。
			\[
				\frac{x}{a} + \frac{y}{b} = 1
			\]

		\item[\textbf{\textcolor{CStatement}{结论五（一般式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			最通用的直线方程形式，要求$A,B$不同时为零，覆盖所有直线。
			\[
				Ax + By + C = 0
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（一般式互化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			一般式可化为斜截式（当$B\neq0$）或截距式（当$A,B,C$均非零）。
			\[
				B\neq0 \Rightarrow y = -\frac{A}{B}x - \frac{C}{B};\qquad A,B,C\neq0 \Rightarrow \frac{x}{-C/A} + \frac{y}{-C/B} = 1.
			\]
	\end{description}
\end{statementbox}
