\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	所有五种形式都是直线方程的不同写法，它们之间可以互相转化，而且都可以从最基本的“斜率定义”推导出来。下面的推导将展示从斜率先得到点斜式，再逐步得到其他形式。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（点斜式推导）：}] 设直线过定点$(x_0,y_0)$且斜率为$k$，则对直线上任意点$(x,y)$，由斜率定义$k=\frac{y-y_0}{x-x_0}$，两边乘以$x-x_0$（$x\neq x_0$）得$y-y_0=k(x-x_0)$。
			\[
				k = \frac{y-y_0}{x-x_0} \;\Rightarrow\; y-y_0 = k(x-x_0).
			\]
			\par\textbf{理由：} 斜率定义为纵坐标增量比横坐标增量，整理即得点斜式。

		\item[\textbf{步骤二（斜截式推导）：}] 在点斜式中取$x_0=0$，$y_0=b$，即得$y-b=k(x-0)$，化简为$y=kx+b$。
			\[
				y-b = k(x-0) \;\Rightarrow\; y = kx + b.
			\]
			\par\textbf{理由：} 当直线经过$(0,b)$时，点斜式自然退化为斜截式。

		\item[\textbf{步骤三（两点式推导）：}] 设直线过两点$(x_1,y_1)$和$(x_2,y_2)$（$x_1\neq x_2$，$y_1\neq y_2$），则斜率$k=\frac{y_2-y_1}{x_2-x_1}$。用点斜式写出$y-y_1=k(x-x_1)$，代入$k$得
			\[
				y-y_1 = \frac{y_2-y_1}{x_2-x_1}(x-x_1) \;\Rightarrow\; \frac{y-y_1}{y_2-y_1} = \frac{x-x_1}{x_2-x_1}.
			\]
			\par\textbf{理由：} 先求斜率，再代入点斜式，最后化为对称比例形式。

		\item[\textbf{步骤四（截距式推导）：}] 在两点式中取$x$截距点$(a,0)$和$y$截距点$(0,b)$（$a\neq0,b\neq0$），则
			\[
				\frac{y-0}{b-0} = \frac{x-a}{0-a} \;\Rightarrow\; \frac{y}{b} = \frac{x-a}{-a} \;\Rightarrow\; \frac{x}{a} + \frac{y}{b} = 1.
			\]
			\par\textbf{理由：} 将截距坐标代入两点式并整理，得到截距式。

		\item[\textbf{步骤五（一般式统一）：}] 任意一种形式都能化为$Ax+By+C=0$。例如点斜式$y-y_0=k(x-x_0)$可化为$kx - y + (y_0 - kx_0)=0$。
			\[
				y-y_0 = k(x-x_0) \;\Rightarrow\; kx - y + (y_0 - kx_0) = 0.
			\]
			\par\textbf{理由：} 一般式中$A=k$，$B=-1$，$C=y_0-kx_0$，$A,B$不同时为零。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	因此，直线方程的这五种形式相互联系，本质都是直线在直角坐标系下的代数表示。在解题时，根据已知条件选择最合适的形式，可以简化运算。
\end{proofbox}
