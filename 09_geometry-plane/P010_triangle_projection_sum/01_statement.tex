\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（三角形设定）：}}] 在任意$\triangle ABC$中，记$a=BC$，$b=CA$，$c=AB$，对应角为$\angle A,\angle B,\angle C$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（射影定理）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}每一边等于另外两边在该边方向上的有向投影的代数和。
		      \[
			      \begin{aligned}
				      a & = b\cos C + c\cos B, \\
				      b & = c\cos A + a\cos C, \\
				      c & = a\cos B + b\cos A.
			      \end{aligned}
		      \]

		\item[\textbf{\textcolor{CStatement}{特例（直角三角形）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}当$C=90^\circ$时，等式简化为
		      \[
			      a = c\cos B,\qquad b = c\cos A.
		      \]
	\end{description}
\end{statementbox}
