\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 在$\triangle ABC$中，$b=4$，$c=6$，$\cos B=\dfrac34$，$\cos C=\dfrac18$，求边长$a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（选用公式）：}] 由射影定理得 $a = b\cos C + c\cos B$。
		      \par\textbf{理由：}
		      条件给出了$b,c$以及目标边$a$两端角$B,C$的余弦，适合直接应用射影定理。
		\item[\textbf{第二步（代入数值）：}] $a = 4\times\dfrac18 + 6\times\dfrac34$。
		      \par\textbf{理由：}
		      将$b=4$，$c=6$，$\cos C=\dfrac18$，$\cos B=\dfrac34$代入。
		\item[\textbf{第三步（计算化简）：}] $a = \dfrac12 + \dfrac92 = 5$。
		      \par\textbf{理由：}
		      通分计算，$\dfrac12+\dfrac92=\dfrac{10}{2}=5$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$a = 5$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 在$\triangle ABC$中，求证：$\sin A = \sin B\cos C + \sin C\cos B$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（射影定理）：}] 由射影定理得 $a = b\cos C + c\cos B$。
		      \par\textbf{理由：}
		      三角形中该恒等式成立。
		\item[\textbf{第二步（正弦定理）：}] 由正弦定理 $\dfrac{a}{\sin A}=\dfrac{b}{\sin B}=\dfrac{c}{\sin C}=2R$（$R$为外接圆半径）。
		      \par\textbf{理由：}
		      正弦定理实现边角互化。
		\item[\textbf{第三步（代入消元）：}] 将 $a=2R\sin A$，$b=2R\sin B$，$c=2R\sin C$ 代入射影定理：
		      \[
			      2R\sin A = 2R\sin B\cos C + 2R\sin C\cos B.
		      \]
		      \par\textbf{理由：}
		      代入后等号两边同有因子$2R$。
		\item[\textbf{第四步（约简结论）：}] 两边同除以$2R$（$R>0$）即得 $\sin A = \sin B\cos C + \sin C\cos B$。
		      \par\textbf{理由：}
		      约去因子即可。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{恒等式得证。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 在$\triangle ABC$中，$a=2b\cos C$，试判断三角形的形状。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（引入射影定理）：}] 由射影定理得 $a = b\cos C + c\cos B$。
		      \par\textbf{理由：}
		      射影定理在任意三角形中均成立。
		\item[\textbf{第二步（联立条件）：}] 将已知条件$a=2b\cos C$代入上式，得 $2b\cos C = b\cos C + c\cos B$，整理得 $b\cos C = c\cos B$。
		      \par\textbf{理由：}
		      移项合并同类项。
		\item[\textbf{第三步（边化角）：}] 由正弦定理 $\dfrac{b}{\sin B}=\dfrac{c}{\sin C}=2R$，代入得 $\sin B\cos C = \sin C\cos B$，即 $\sin(B-C)=0$。
		      \par\textbf{理由：}
		      正弦定理边化角，再逆用两角差的正弦公式。
		\item[\textbf{第四步（推出角关系）：}] 在$\triangle ABC$中，$0<B,C<\pi$，所以 $-\pi<B-C<\pi$，由$\sin(B-C)=0$得 $B-C=0$或$B-C=\pm\pi$（后者不可能），故$B=C$。
		      \par\textbf{理由：}
		      三角形内角范围决定差的范围。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\triangle ABC$是等腰三角形（$B=C$）。}
\end{examplebox}
