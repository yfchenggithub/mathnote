\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{垂心到各顶点的连线垂直于对边（向量点积为零）。}
	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（非退化三角形）：} 三角形三个顶点不共线。
		\item \textbf{条件 2（垂心定义）：} $H$ 是 $\triangle ABC$ 的垂心。
	\end{itemize}
	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（钝角垂心位置）：} 钝角三角形中垂心在外部，垂直关系仍成立。
		\item \textbf{检查项（点积验证）：} 使用 $\overrightarrow{AH}\cdot\overrightarrow{BC}=0$ 等验证垂直。
	\end{itemize}
\end{summarybox}
