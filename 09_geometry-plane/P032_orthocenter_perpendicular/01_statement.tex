\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形非退化）：}}] 三角形 $ABC$ 非退化（三点不共线），$H$ 为垂心。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（垂线垂直）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			垂心与每个顶点的连线垂直于对边。
			\[
				AH\perp BC,
				\qquad
				BH\perp AC,
				\qquad
				CH\perp AB.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（向量形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用向量内积为零表示垂直关系。
			\[
				\overrightarrow{AH}\cdot\overrightarrow{BC}=0,\quad \overrightarrow{BH}\cdot\overrightarrow{AC}=0,\quad \overrightarrow{CH}\cdot\overrightarrow{AB}=0.
			\]
	\end{description}
\end{statementbox}
