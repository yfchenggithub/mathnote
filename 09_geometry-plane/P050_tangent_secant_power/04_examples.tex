\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	圆外一点 $P$，$PT$ 切圆于 $T$，割线 $PAB$ 交圆于 $A$、$B$，$PT=4$，$PA=2$，求 $PB$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别信息）：}] $PT=4$，$PA=2$，$A$ 在 $P$、$B$ 之间。
			\par\textbf{理由：}
			已知条件符合切割线定理要求。
		\item[\textbf{第二步（套用公式）：}] 由切割线定理 $PT^{2}=PA\cdot PB$，代入得
			\[
				4^{2}=2\cdot PB
			\]
			\par\textbf{理由：}
			直接应用定理。
		\item[\textbf{第三步（求解）：}] 解方程 $16=2PB$，得 $PB=8$。
			\par\textbf{理由：}
			一元一次方程求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PB=8$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	圆外一点 $P$，切线 $PT=6$，割线 $PAB$ 满足 $PA+PB=13$，且 $A$ 在 $P$、$B$ 之间，求 $PA$、$PB$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}] 设 $PA=x$，$PB=y$，则 $x+y=13$，且由切割线定理 $PT^{2}=xy$，即 $36=xy$。
			\par\textbf{理由：}
			利用定理和已知和建立方程组。
		\item[\textbf{第二步（建立方程）：}] $x$、$y$ 是方程 $t^{2}-13t+36=0$ 的两根。
			\par\textbf{理由：}
			根与系数的关系（韦达定理）。
		\item[\textbf{第三步（求解）：}] 解方程得 $t=4$ 或 $t=9$。因为 $A$ 在 $P$、$B$ 之间，所以 $PA<PB$，故 $PA=4$，$PB=9$。
			\par\textbf{理由：}
			判别大小与位置对应。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA=4$，$PB=9$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知圆 $O$ 的半径 $R=5$，$P$ 为圆外一点，$OP=13$，过 $P$ 的任意割线 $PAB$ 交圆于 $A$、$B$（$A$ 在 $P$、$B$ 之间），求 $PA\cdot PB$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求切线长）：}] 连接 $OT$、$PT$，$OT\perp PT$，由勾股定理得
			\[
				\begin{aligned}
					PT^{2} &= OP^{2} - R^{2} \\
					&= 13^{2} - 5^{2} \\
					&= 169 - 25 \\
					&= 144
			\end{aligned}
			\]
			所以 $PT=12$。
			\par\textbf{理由：}
			切线垂直于半径。
		\item[\textbf{第二步（应用定理）：}] 由切割线定理，$PT^{2}=PA\cdot PB$，所以
			\[
				PA\cdot PB = 144
			\]
			\par\textbf{理由：}
			对于任意割线，该乘积恒等于切线长的平方。
		\item[\textbf{第三步（结论）：}] 因此 $PA\cdot PB=144$。
			\par\textbf{理由：}
			代入即得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$PA\cdot PB=144$}
\end{examplebox}
