\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	圆外一点 $P$，切线 $PT$，割线 $PAB$，则有 $PT^{2}=PA\cdot PB$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（P在圆外）：} 点 $P$ 必须在圆外，才能作切线和割线。
		\item \textbf{条件 2（切线割线共点）：} 切线 $PT$ 和割线 $PAB$ 必须从同一点 $P$ 出发，且 $A$ 在 $P$、$B$ 之间。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（线段对应）：} 乘积是 $PA\cdot PB$，不是 $PA\cdot AB$。
		\item \textbf{检查项（位置顺序）：} 确认 $A$ 位于 $P$ 与 $B$ 之间，避免对应错误。
	\end{itemize}
\end{summarybox}
