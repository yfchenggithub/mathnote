\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（P在圆外引线）：}}] 点 $P$ 在圆 $O$ 外，$PT$ 为圆 $O$ 的切线（$T$ 为切点），过 $P$ 的任意割线交圆于 $A$、$B$，且 $A$ 在 $P$、$B$ 之间。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（切割线定理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			切线长是割线上两段线段长的比例中项。
			\[
				PT^{2}=PA\cdot PB
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（圆幂公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若设圆心 $O$ 到 $P$ 的距离为 $d$，圆半径为 $R$，则 $PT^{2}=d^{2}-R^{2}$，且与割线无关。
			\[
				PT^{2}=PA\cdot PB=d^{2}-R^{2}
			\]
	\end{description}
\end{statementbox}
