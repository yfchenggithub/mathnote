\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{外角平分线定理：} 在$\triangle ABC$中，若$AD$是$\angle A$的外角平分线，交$BC$延长线于$D$，则$\frac{BD}{DC} = \frac{AB}{AC}$。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三角形存在）：} $\triangle ABC$为任意三角形。
		\item \textbf{条件 2（外角平分线）：} $AD$是$\angle A$的外角平分线，且与$BC$延长线交于$D$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（内外混淆）：} 注意“外角”二字，避免与内角平分线定理混用公式。
		\item \textbf{检查项（交点位置）：} 确定$D$在$BC$延长线上，长度关系需使用$|BD-DC|=BC$（而非$BD+DC=BC$）。
	\end{itemize}

\end{summarybox}
