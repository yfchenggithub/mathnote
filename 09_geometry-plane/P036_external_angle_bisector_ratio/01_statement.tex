	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（外角平分线条件）：}}] 在$\triangle ABC$中，$AD$是$\angle A$的外角平分线，交$BC$的延长线于点$D$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（外角平分线比例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外角平分线将$BC$延长线分得的线段比等于邻边长度之比。
			\[
				\frac{BD}{DC} = \frac{AB}{AC}.
			\]
		\item[\textbf{\textcolor{CStatement}{类比定理（内角平分线）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			内角平分线将对边分得的线段比等于邻边长度之比，交点在$BC$上。
			\[
				\frac{BD}{DC} = \frac{AB}{AC}.
			\]
	\end{description}

\end{statementbox}
