	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混淆内外形式）}}] 将外角平分线比例误记为$\frac{BD}{DC}=\frac{AC}{AB}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（比例相同）}}] 外角平分线比例与内角平分线相同：$\frac{BD}{DC}=\frac{AB}{AC}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（机械记忆）}}] 只记形式，未注意条件不同；或误以为外角平分线比例是内角比例的倒数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（错误线段关系）}}] 列方程时认为$BD+DC=BC$，忽略$D$在$BC$延长线上。
		\item[\textbf{\textcolor{CTrap}{正确理解（D点位置）}}] 外角平分线交点$D$在$BC$延长线上，$BD$与$DC$之差等于$BC$（$|BD-DC|=BC$）。
		\item[\textbf{\textcolor{CTrap}{错因分析（未画图）}}] 不习惯图形中的延长线，直接套用内角平分线的和关系。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（条件误用）}}] 题目中出现“角平分线”时，未区分内外角平分线，直接套用外角平分线定理。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分内外）}}] 审题时注意是“外角平分线”还是“内角平分线”，并根据交点位置判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念模糊）}}] 对内角平分线定理和外角平分线定理的区别不清楚。
	\end{description}

\end{trapbox}
