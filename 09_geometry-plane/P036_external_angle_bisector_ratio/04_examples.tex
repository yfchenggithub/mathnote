\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=4$，$AC=6$，$AD$是$\angle A$的外角平分线，交$BC$的延长线于点$D$。已知$BD=8$，求$DC$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列比例式）：}] 由外角平分线定理得$\frac{BD}{DC}=\frac{AB}{AC}= \frac{4}{6}$，即$\frac{8}{DC}= \frac{4}{6}$。
			\par\textbf{理由：} 直接代入已知长度。
		\item[\textbf{第二步（解方程）：}] $8\times 6 = 4\times DC$，解得$DC=12$。
			\par\textbf{理由：} 交叉相乘解比例。
		\item[\textbf{第三步（验证比例）：}] $\frac{BD}{DC}=\frac{8}{12}=\frac{2}{3}$，与$\frac{AB}{AC}=\frac{4}{6}=\frac{2}{3}$一致，符合定理。
			\par\textbf{理由：} 回代验证比例式成立。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$DC=12$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=5$，$AC=3$，$BC=4$，$AD$是$\angle A$的外角平分线，交$BC$的延长线于点$D$。求$BD$的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知数）：}] 设$BD=x$，则$DC=BD-BC=x-4$。
			\par\textbf{理由：} $D$在$BC$延长线上，$BD=BC+CD$，所以$CD=x-4$。
		\item[\textbf{第二步（列方程）：}] 由外角平分线定理得$\frac{x}{x-4}=\frac{5}{3}$。
			\par\textbf{理由：} 代入比例式。
		\item[\textbf{第三步（解方程）：}] $3x=5x-20$，得$2x=20$，$x=10$，故$BD=10$。
			\par\textbf{理由：} 解一元一次方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BD=10$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$AB=4$，$AC=2$，$BC=3$。$AD$是$\angle A$的外角平分线，交$BC$延长线于$D$；$AE$是$\angle A$的内角平分线，交$BC$于$E$。求$DE$的长度。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（内角平分线求BE）：}] 由内角平分线定理$\frac{BE}{EC}=\frac{AB}{AC}= \frac{4}{2}=2$，又$BE+EC=BC=3$，所以$BE=2$，$EC=1$。
			\par\textbf{理由：} 合比性质。
		\item[\textbf{第二步（外角平分线求BD）：}] 设$BD=x$，则$DC=BD-BC=x-3$，由外角平分线定理$\frac{x}{x-3}= \frac{4}{2}=2$，解得$x=6$，故$BD=6$。
			\par\textbf{理由：} 列比例方程求解。
		\item[\textbf{第三步（求DE）：}] $DE=BD-BE=6-2=4$。
			\par\textbf{理由：} 在直线$BC$上，$B,E,C,D$依次排列，$DE=BD-BE$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$DE=4$}。

\end{examplebox}
