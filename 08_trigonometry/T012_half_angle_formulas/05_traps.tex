\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（分母非零条件）}}] 使用 $\tan\frac{\alpha}{2} = \frac{\sin\alpha}{1+\cos\alpha}$ 时未检查 $\cos\alpha$ 是否等于 $-1$，或使用另一形式时未检查 $\sin\alpha$ 是否等于 $0$，导致分母为零却仍在计算。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母条件确认）}}] 使用前必须确认分母不为零：对于 $\tan\frac{\alpha}{2} = \frac{\sin\alpha}{1+\cos\alpha}$，需 $\cos\alpha\neq -1$；对于 $\tan\frac{\alpha}{2} = \frac{1-\cos\alpha}{\sin\alpha}$，需 $\sin\alpha\neq 0$。若分母为零，该表达式无定义，应换用另一公式或考虑极限。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义域）}}] 记忆公式时忽略分数有意义的条件，导致解题中代入数值后发生除零错误或忽略无解情况。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（平方开方符号）}}] 从 $\sin^2\frac{\alpha}{2} = \frac{1-\cos\alpha}{2}$ 直接写 $\sin\frac{\alpha}{2} = \sqrt{\frac{1-\cos\alpha}{2}}$，忽略平方根的正负号选择。
		\item[\textbf{\textcolor{CTrap}{正确理解（象限决定符号）}}] 半角公式只给出平方值，欲求 $\sin\frac{\alpha}{2}$ 或 $\cos\frac{\alpha}{2}$ 的值时，需要根据 $\frac{\alpha}{2}$ 所在的象限确定正负号，不能默认取正。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆平方与一次）}}] 将 $\sin^2$ 与 $\sin$ 混淆，忘记开方需考虑符号，这是常见低级错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（角度对应错误）}}] 错误地认为 $\sin^2\frac{\alpha}{2} = \frac{1-\cos\frac{\alpha}{2}}{2}$ 或 $\sin^2\alpha = \frac{1-\cos2\alpha}{2}$ 与半角公式混淆。
		\item[\textbf{\textcolor{CTrap}{正确理解（分清倍半关系）}}] 半角公式中左边是 $\frac{\alpha}{2}$，右边是 $\alpha$ 的三角函数；二倍角变形 $\sin^2\alpha = \frac{1-\cos2\alpha}{2}$ 左边是 $\alpha$，右边是 $2\alpha$。注意区分哪个角是半角，哪个是倍角。
		\item[\textbf{\textcolor{CTrap}{错因分析（变量代换混乱）}}] 对公式中的变量关系不清晰，代换时出错，导致公式用反。
	\end{description}
\end{trapbox}
