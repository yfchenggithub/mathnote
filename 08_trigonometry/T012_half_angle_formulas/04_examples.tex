\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础直接应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\cos\alpha = \frac{1}{3}$，求 $\sin^2\frac{\alpha}{2}$ 和 $\cos^2\frac{\alpha}{2}$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用平方公式）：}]
			根据半角平方公式，$\sin^2\frac{\alpha}{2} = \frac{1-\cos\alpha}{2}$，$\cos^2\frac{\alpha}{2} = \frac{1+\cos\alpha}{2}$。
			\par\textbf{理由：} 题目只求平方，无需考虑符号，直接代入。
		\item[\textbf{第二步（代入数值）：}]
			将 $\cos\alpha = \frac{1}{3}$ 代入：
			\[
				\begin{aligned}
					\sin^2\frac{\alpha}{2} &= \frac{1-\frac{1}{3}}{2} \\ &= \frac{\frac{2}{3}}{2} \\ &= \frac{1}{3},
			\end{aligned}
				\qquad
				\begin{aligned}
					\cos^2\frac{\alpha}{2} &= \frac{1+\frac{1}{3}}{2} \\ &= \frac{\frac{4}{3}}{2} \\ &= \frac{2}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 分数四则运算。
		\item[\textbf{第三步（写出结果）：}]
			$\sin^2\frac{\alpha}{2} = \frac{1}{3}$，$\cos^2\frac{\alpha}{2} = \frac{2}{3}$。
			\par\textbf{理由：} 数值化简完毕。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{半角平方公式给出直接表达式，无需开方即可得到平方值。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\cos\alpha = -\frac{1}{2}$，且 $\alpha$ 为第二象限角，求 $\tan\frac{\alpha}{2}$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定 $\sin\alpha$）：}]
			由同角关系 $\sin^2\alpha = 1-\cos^2\alpha$，代入 $\cos\alpha=-\frac{1}{2}$ 得：
			\[
				\begin{aligned}
					\sin^2\alpha &= 1 - \cos^2\alpha \\ &= 1 - \frac{1}{4} \\ &= \frac{3}{4}.
			\end{aligned}
			\]
			因为 $\alpha$ 在第二象限，$\sin\alpha>0$，所以 $\sin\alpha = \frac{\sqrt{3}}{2}$。
			\par\textbf{理由：} 先利用同角关系求 $\sin\alpha$，注意象限确定符号。
		\item[\textbf{第二步（选正切半角公式）：}]
			选用 $\tan\frac{\alpha}{2} = \frac{\sin\alpha}{1+\cos\alpha}$，因为该形式只需已知 $\sin\alpha$ 和 $\cos\alpha$，且分母 $1+\cos\alpha = 1-\frac{1}{2}$，其值为 $\frac{1}{2}\neq 0$，有效。
			\par\textbf{理由：} 正切半角公式有两种，这里两个都可用。注意分母不为零。
		\item[\textbf{第三步（代入计算）：}]
			代入数值并化简：
			\[
				\begin{aligned}
					\tan\frac{\alpha}{2} &= \frac{\frac{\sqrt{3}}{2}}{1+(-\frac{1}{2})} \\ &= \frac{\frac{\sqrt{3}}{2}}{\frac{1}{2}} \\ &= \sqrt{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入数值并化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{正切半角公式可直接由 $\sin\alpha$ 和 $\cos\alpha$ 计算，但需根据条件选择合适的表达式并验证分母非零。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\tan\frac{\alpha}{2} = \sqrt{3}$，且 $\frac{\alpha}{2}$ 为锐角，求 $\cos\alpha$ 和 $\sin\alpha$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（使用正切半角公式建立方程）：}]
			由 $\tan\frac{\alpha}{2} = \sqrt{3}$，根据半角公式 $\tan\frac{\alpha}{2} = \frac{\sin\alpha}{1+\cos\alpha}$，得 $\frac{\sin\alpha}{1+\cos\alpha} = \sqrt{3}$，即 $\sin\alpha = \sqrt{3}(1+\cos\alpha)$。
			\par\textbf{理由：} 正切半角公式将已知与 $\sin\alpha,\cos\alpha$ 关联。
		\item[\textbf{第二步（联立平方关系求解）：}]
			将 $\sin\alpha = \sqrt{3}(1+\cos\alpha)$ 代入 $\sin^2\alpha+\cos^2\alpha=1$：
			\[
				\left[\sqrt{3}(1+\cos\alpha)\right]^2 + \cos^2\alpha = 1 \;\Longrightarrow\; 3(1+2\cos\alpha+\cos^2\alpha) + \cos^2\alpha = 1,
			\]
			整理得 $4\cos^2\alpha + 6\cos\alpha + 2 = 0$，即 $2\cos^2\alpha + 3\cos\alpha + 1 = 0$。
			\par\textbf{理由：} 代入后展开，合并同类项得到关于 $\cos\alpha$ 的一元二次方程。
		\item[\textbf{第三步（求解并检验）：}]
			解方程 $(2\cos\alpha+1)(\cos\alpha+1)=0$，得 $\cos\alpha = -\frac{1}{2}$ 或 $\cos\alpha = -1$。由 $\frac{\alpha}{2}$ 为锐角知 $\alpha\in(0,\pi)$，$\cos\alpha=-1$ 时 $\alpha=\pi$，此时 $\frac{\alpha}{2}=\frac{\pi}{2}$，$\tan\frac{\alpha}{2}$ 无定义，故舍去。因此 $\cos\alpha = -\frac{1}{2}$，代入 $\sin\alpha = \sqrt{3}(1+\cos\alpha)$ 得 $\sin\alpha = \frac{\sqrt{3}}{2}$。
			\par\textbf{理由：} 检验解是否满足原题条件，确保 $\tan\frac{\alpha}{2}$ 有意义。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{已知半角正切，可通过半角公式与平方关系联立求解一倍角的正余弦，注意检查定义域。}
\end{examplebox}
