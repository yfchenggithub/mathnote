\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	半角公式的证明基于二倍角余弦公式和倍角正弦公式。先利用 $\cos2\theta$ 公式解出 $\sin^2\theta$ 和 $\cos^2\theta$，再通过代数变形得到正切半角公式。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（写出倍角关系）：}]
			\[
				\begin{aligned}
					\cos\alpha &= 1-2\sin^2\frac{\alpha}{2},\\
					\cos\alpha &= 2\cos^2\frac{\alpha}{2}-1.
			\end{aligned}
			\]
			\par\textbf{理由：}
			将二倍角公式 $\cos2\theta = 1-2\sin^2\theta$ 和 $\cos2\theta = 2\cos^2\theta-1$ 中的 $\theta$ 替换为 $\frac{\alpha}{2}$ 即得。

		\item[\textbf{步骤二（解出半角平方）：}]
			\[
				\begin{aligned}
					\sin^2\frac{\alpha}{2} &= \frac{1-\cos\alpha}{2},\\
					\cos^2\frac{\alpha}{2} &= \frac{1+\cos\alpha}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：}
			由步骤一第一个等式移项得 $2\sin^2\frac{\alpha}{2}=1-\cos\alpha$，两边同除以 $2$ 即得 $\sin^2$ 公式；由第二个等式移项得 $2\cos^2\frac{\alpha}{2}=1+\cos\alpha$，除以 $2$ 得 $\cos^2$ 公式。

		\item[\textbf{步骤三（推正切形式一）：}]
			\[
				\tan\frac{\alpha}{2} = \frac{\sin\alpha}{1+\cos\alpha}.
			\]
			\par\textbf{理由：}
			利用 $\tan\frac{\alpha}{2} = \frac{\sin\frac{\alpha}{2}}{\cos\frac{\alpha}{2}}$，分子分母同乘 $2\cos\frac{\alpha}{2}$，得 $\frac{2\sin\frac{\alpha}{2}\cos\frac{\alpha}{2}}{2\cos^2\frac{\alpha}{2}} = \frac{\sin\alpha}{2\cos^2\frac{\alpha}{2}}$。再将分母 $2\cos^2\frac{\alpha}{2}$ 用步骤二的 $\cos^2$ 公式替换为 $1+\cos\alpha$，即得 $\tan\frac{\alpha}{2} = \frac{\sin\alpha}{1+\cos\alpha}$。要求 $1+\cos\alpha\neq 0$。

		\item[\textbf{步骤四（推正切形式二）：}]
			\[
				\tan\frac{\alpha}{2} = \frac{1-\cos\alpha}{\sin\alpha}.
			\]
			\par\textbf{理由：}
			同样从 $\tan\frac{\alpha}{2} = \frac{\sin\frac{\alpha}{2}}{\cos\frac{\alpha}{2}}$ 出发，分子分母同乘 $2\sin\frac{\alpha}{2}$，得 $\frac{2\sin^2\frac{\alpha}{2}}{2\sin\frac{\alpha}{2}\cos\frac{\alpha}{2}} = \frac{2\sin^2\frac{\alpha}{2}}{\sin\alpha}$。再将分子 $2\sin^2\frac{\alpha}{2}$ 用步骤二的 $\sin^2$ 公式替换为 $1-\cos\alpha$，即得 $\tan\frac{\alpha}{2} = \frac{1-\cos\alpha}{\sin\alpha}$。要求 $\sin\alpha\neq 0$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{以上推导全部基于已学的二倍角公式和代数恒等变形，逻辑完整，因此半角公式成立。使用时特别注意正切公式中分母不能为零。}
\end{proofbox}
