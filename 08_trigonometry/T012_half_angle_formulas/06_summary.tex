\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{半角公式将半角 $\frac{\alpha}{2}$ 的正弦平方、余弦平方、正切用 $\alpha$ 的三角函数表示，是降幂与角度统一的重要工具。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（平方公式无条件）：} $\sin^2\frac{\alpha}{2} = \frac{1-\cos\alpha}{2}$ 和 $\cos^2\frac{\alpha}{2} = \frac{1+\cos\alpha}{2}$ 对任意角 $\alpha$ 恒成立。
		\item \textbf{条件2（正切限制）：} 正切半角公式需要分母不为 $0$：$\frac{\sin\alpha}{1+\cos\alpha}$ 要求 $\cos\alpha \neq -1$；$\frac{1-\cos\alpha}{\sin\alpha}$ 要求 $\sin\alpha \neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（正切分母检查）：} 使用 $\tan\frac{\alpha}{2}$ 公式前必须验证分母是否为零，避免无意义运算。
		\item \textbf{检查项（开方定号）：} 如需从平方公式求 $\sin\frac{\alpha}{2}$ 或 $\cos\frac{\alpha}{2}$，需根据 $\frac{\alpha}{2}$ 所在象限确定正负号。
	\end{itemize}
\end{summarybox}
