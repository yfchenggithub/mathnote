\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（任意实数角）：}}] 设 $\alpha,\beta\in\mathbb{R}$，即角 $\alpha,\beta$ 可取任意实数。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正弦和化积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两正弦之和转化为乘积形式。
			\[
				\sin\alpha + \sin\beta = 2\sin\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（正弦差化积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两正弦之差转化为乘积形式。
			\[
				\sin\alpha - \sin\beta = 2\cos\frac{\alpha+\beta}{2}\sin\frac{\alpha-\beta}{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（余弦和化积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两余弦之和转化为乘积形式。
			\[
				\cos\alpha + \cos\beta = 2\cos\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论四（余弦差化积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两余弦之差转化为乘积形式，注意负号。
			\[
				\cos\alpha - \cos\beta = -2\sin\frac{\alpha+\beta}{2}\sin\frac{\alpha-\beta}{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（积化和差）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			由和差化积逆推，将乘积化为和差。
			\[
				\sin\alpha\cos\beta = \frac{1}{2}\bigl[\sin(\alpha+\beta) + \sin(\alpha-\beta)\bigr].
			\]
			其余三类公式可由类似方法得到。
	\end{description}
\end{statementbox}
