\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（角度范围）：}}] $\alpha,\beta$ 为任意实数。
		\item[\textbf{\textcolor{CStatement}{条件 2（正切约束）：}}] 当使用正切公式时，须满足 $\alpha,\beta,\alpha\pm\beta\neq\frac{\pi}{2}+k\pi\,(k\in\mathbb{Z})$ 且 $1\mp\tan\alpha\tan\beta\neq0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正弦公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两角和的正弦等于正弦余弦交叉相乘且符号与角一致。
			\[
				\sin(\alpha\pm\beta)=\sin\alpha\cos\beta\pm\cos\alpha\sin\beta
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（余弦公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两角和的余弦等于余弦乘积减去正弦乘积，符号与角相反。
			\[
				\cos(\alpha\pm\beta)=\cos\alpha\cos\beta\mp\sin\alpha\sin\beta
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（正切公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两角和的正切等于正切之和除以一减正切积，分子与分母符号一致。
			\[
				\tan(\alpha\pm\beta)=\dfrac{\tan\alpha\pm\tan\beta}{1\mp\tan\alpha\tan\beta}
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{推广结论（二倍角公式）：}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	令 $\alpha=\beta$，和角公式退化为二倍角公式。
	\[
		\begin{aligned}
			\sin 2\alpha &= 2\sin\alpha\cos\alpha,\\
			\cos 2\alpha &= \cos^2\alpha-\sin^2\alpha \\
			&= 2\cos^2\alpha-1 \\
			&= 1-2\sin^2\alpha,\\
			\tan 2\alpha &= \dfrac{2\tan\alpha}{1-\tan^2\alpha}.
		\end{aligned}
	\]
\end{statementbox}
