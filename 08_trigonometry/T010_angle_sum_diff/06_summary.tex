\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{和角公式将两角和差的三角函数转化为单角三角函数的线性组合，是三角恒等变换的基石。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（角度定义域）：} 正弦余弦公式对任意 $\alpha,\beta\in\mathbb{R}$ 成立；正切公式要求 $\alpha,\beta,\alpha\pm\beta\neq\frac{\pi}{2}+k\pi$（$k\in\mathbb{Z}$）。
		\item \textbf{条件 2（分母非零）：} 使用正切公式时，还需 $1\mp\tan\alpha\tan\beta\neq0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号规律）：} 正弦同号、余弦异号、正切分子分母符号一致。牢记结构对比。
		\item \textbf{检查项（分母检验）：} 遇到正切公式，先检验分母非零；遇到求 $\sin\alpha,\cos\alpha$，先确定象限符号。
	\end{itemize}
\end{summarybox}
