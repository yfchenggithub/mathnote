\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定义记号）：}}]
			设 $S_{+}=\sin x+\cos x$, $S_{-}=\sin x-\cos x$, $P=\sin x\cos x$, 其中 $x$ 为任意实数。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平方关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两平方和与平方差公式。
			\[
				(S_{+})^{2}+(S_{-})^{2}=2,
				\qquad
				(S_{\pm})^{2}=1\pm 2P.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（互化公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用平方差表示积。
			\[
				P=\frac{S_{+}^{2}-1}{2}=\frac{1-S_{-}^{2}}{2}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（知一求二）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			由三个量中任一个可求其余两个，并知取值范围。
			\[
				S_{+},S_{-}\in[-\sqrt{2},\sqrt{2}],\quad P\in\left[-\frac12,\frac12\right].
			\]
	\end{description}

\end{statementbox}
