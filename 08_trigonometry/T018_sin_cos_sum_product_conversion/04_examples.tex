\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin x+\cos x = \frac12$，求 $\sin x\cos x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别已知量）：}]
			已知 $S_+=\frac12$，要求 $P$。
			\par\textbf{理由：} 结论中 $S_+$ 与 $P$ 的关系。
		\item[\textbf{第二步（套用公式）：}]
			由 $P=\dfrac{S_+^2-1}{2}$，代入 $S_+=\frac12$，得 $P=\dfrac{(1/2)^2-1}{2}=\dfrac{1/4-1}{2}=-\dfrac{3}{8}$。
			\par\textbf{理由：} 直接使用互化公式 $P=\frac{S_+^2-1}{2}$。
		\item[\textbf{第三步（给出答案）：}]
			所以 $\sin x\cos x = -\dfrac38$。
			\par\textbf{理由：} 计算完成。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin x\cos x=-\dfrac38$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin x\cos x = -\dfrac14$，且 $x\in\left(\dfrac{\pi}{2},\pi\right)$，求 $\sin x-\cos x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别已知量）：}]
			已知 $P=-\frac14$，要求 $S_-$。
			\par\textbf{理由：} $S_-^2=1-2P$。
		\item[\textbf{第二步（平方计算）：}]
			$S_-^2 = 1-2\times\left(-\frac14\right)=1+\frac12=\frac32$，所以 $|S_-|=\sqrt{\frac32}=\frac{\sqrt6}{2}$。
			\par\textbf{理由：} 平方关系 $S_-^2=1-2P$。
		\item[\textbf{第三步（确定符号）：}]
			因为 $x\in\left(\frac{\pi}{2},\pi\right)$，所以 $\sin x>0,\cos x<0$，从而 $\sin x-\cos x>0$，故 $S_-=\frac{\sqrt6}{2}$。
			\par\textbf{理由：} 利用角度范围判断差的正负。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin x-\cos x = \dfrac{\sqrt6}{2}$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin x+\cos x = \dfrac15$，且 $x\in(0,\pi)$，求 $\tan x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求积）：}]
			由 $P=\dfrac{S_+^2-1}{2}$，$S_+=\dfrac15$，得 $P=\dfrac{1/25-1}{2} = -\dfrac{12}{25}$。
			\par\textbf{理由：} 互化公式。
		\item[\textbf{第二步（求差平方）：}]
			$S_-^2 = 1-2P = 1-2\times\left(-\dfrac{12}{25}\right)=1+\dfrac{24}{25}=\dfrac{49}{25}$，所以 $|S_-|=\dfrac75$。
			\par\textbf{理由：} 平方关系。
		\item[\textbf{第三步（联合求解）：}]
			因 $x\in(0,\pi)$，$P<0$，故 $\cos x<0$。又 $S_+>0$，所以 $\sin x>|\cos x|$，从而 $S_-=\sin x-\cos x>0$，取 $S_-=\dfrac75$。联立 $\sin x+\cos x=\frac15$，$\sin x-\cos x=\frac75$，解得 $\sin x=\frac45,\cos x=-\frac35$，从而 $\tan x=\dfrac{\sin x}{\cos x}=-\dfrac43$。
			\par\textbf{理由：} 解方程组得正余弦值，再求正切。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\tan x=-\dfrac43$}
\end{examplebox}
