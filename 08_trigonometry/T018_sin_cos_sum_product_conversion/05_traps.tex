\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略符号）}}]
			已知 $S_+$ 或 $S_-$ 的平方求原值时，擅自取正号而忽略负号，导致漏解或错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（符号判定）}}]
			平方关系只能确定绝对值，必须结合 $x$ 的范围（即 $\sin x,\cos x$ 的符号）才能确定 $S_+$ 或 $S_-$ 的符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（平方性质）}}]
			误以为 $\sqrt{S^2}=S$，忘记平方根的非唯一性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽视范围）}}]
			使用互化公式前未检查 $P$ 是否在 $[-\frac12,\frac12]$ 内，或 $S_+$ 是否在 $[-\sqrt2,\sqrt2]$ 内，导致代入后出现矛盾或错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（范围前提）}}]
			公式 $P=\frac{S_+^2-1}{2}$ 隐含 $S_+^2\le 2$，即 $S_+\in[-\sqrt2,\sqrt2]$；同理 $P\in[-\frac12,\frac12]$。超出此范围的数值不会出现，应重新检查条件。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略有界性）}}]
			未意识到三角函数值的固有界限，将任意实数代入公式导致无意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式混淆）}}]
			记错 $P$ 的表达式，如写成 $P=\frac{1-S_+^2}{2}$ 或 $P=\frac{S_-^2-1}{2}$，导致符号错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分符号）}}]
			由 $S_+^2=1+2P$ 推出 $P=\frac{S_+^2-1}{2}$；由 $S_-^2=1-2P$ 推出 $P=\frac{1-S_-^2}{2}$。注意 $S_+$ 对应 $+2P$，$S_-$ 对应 $-2P$。
		\item[\textbf{\textcolor{CTrap}{错因分析（推导不清）}}]
			未亲自推导公式，死记硬背导致加减号混淆。
	\end{description}

\end{trapbox}
