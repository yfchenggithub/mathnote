\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{$\sin x+\cos x$、$\sin x-\cos x$、$\sin x\cos x$ 通过平方关系 $(S_\pm)^2=1\pm 2P$ 及平方和 $S_+^2+S_-^2=2$ 相互关联，知其一可求其二，且范围受限。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（任意实数）：} $x$ 可为任意实数，恒成立。
		\item \textbf{条件2（范围前提）：} 互化公式要求 $S_+,S_-,P$ 在各自范围内；使用平方关系开方时需结合 $x$ 的范围判定符号。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（开方判号）：} 从 $S_+^2$ 求 $S_+$ 时，先开方得绝对值，再根据 $x$ 的象限确定 $S_+$ 的符号，避免漏解。
		\item \textbf{检查项（范围检验）：} 算出的 $S_+,S_-$ 必须在 $[-\sqrt2,\sqrt2]$ 内，$P$ 必须在 $[-\frac12,\frac12]$ 内，否则数据不合理。
	\end{itemize}

\end{summarybox}
