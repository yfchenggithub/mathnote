\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{万能公式将三角函数统一表示为半角正切 $t$ 的有理函数，实现三角问题代数化。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（定义域限制）：} $x \neq \pi + 2k\pi$ ($k\in\mathbb{Z}$)，保证 $\tan\frac{x}{2}$ 有意义。
		\item \textbf{条件2（正切额外）：} 使用 $\tan x$ 公式时还需 $x \neq \frac{\pi}{2} + k\pi$，确保分母 $1-t^2 \neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件检验）：} 代入公式前先检查 $x$ 是否满足定义域。
		\item \textbf{检查项（分母为零）：} 对于正切公式，注意 $t^2=1$ 时无意义。
	\end{itemize}

\end{summarybox}
