\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $t = \tan\frac{x}{2} = 2$，且 $x \neq \pi+2k\pi$，求 $\sin x$ 和 $\cos x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列出公式）：}]
			写出万能公式：$\sin x = \frac{2t}{1+t^2}$，$\cos x = \frac{1-t^2}{1+t^2}$。
			\par\textbf{理由：} 万能公式直接给出。
		\item[\textbf{第二步（代入数值）：}]
			将 $t=2$ 代入 $\sin x$ 公式：$\sin x = \frac{2\times 2}{1+2^2} = \frac{4}{5}$。
			\par\textbf{理由：} 代数运算。
		\item[\textbf{第三步（同样求余弦）：}]
			$\cos x = \frac{1-2^2}{1+2^2} = \frac{1-4}{5} = -\frac{3}{5}$。
			\par\textbf{理由：} 代入计算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin x = \frac{4}{5}$，$\cos x = -\frac{3}{5}$。}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\tan\frac{x}{2}=3$，且 $x$ 满足 $x\neq \pi+2k\pi$ 且 $x\neq \frac{\pi}{2}+k\pi$，求 $\tan x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证条件）：}]
			由于 $x\neq \frac{\pi}{2}+k\pi$，$\cos x \neq 0$，可以使用正切公式。
			\par\textbf{理由：} 万能正切公式要求分母 $1-t^2\neq 0$，即 $\cos x\neq 0$。
		\item[\textbf{第二步（代入正切公式）：}]
			$\tan x = \frac{2t}{1-t^2} = \frac{2\times 3}{1-3^2} = \frac{6}{-8}$。
			\par\textbf{理由：} 直接应用 $\tan x$ 的万能公式。
		\item[\textbf{第三步（化简结果）：}]
			$\tan x = -\frac{3}{4}$。
			\par\textbf{理由：} 约分。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\tan x = -\frac{3}{4}$。}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin x + \cos x = \frac{1}{5}$，且 $x\neq \pi+2k\pi$，求 $\tan\frac{x}{2}$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代换化归）：}]
			令 $t = \tan\frac{x}{2}$，则 $\sin x = \frac{2t}{1+t^2}$，$\cos x = \frac{1-t^2}{1+t^2}$。
			\par\textbf{理由：} 万能公式。
		\item[\textbf{第二步（列方程）：}]
			代入已知条件：$\frac{2t}{1+t^2} + \frac{1-t^2}{1+t^2} = \frac{1}{5}$，即 $\frac{2t+1-t^2}{1+t^2} = \frac{1}{5}$。
			\par\textbf{理由：} 通分合并分子。
		\item[\textbf{第三步（解方程得t）：}]
			去分母：$5(2t+1-t^2) = 1+t^2$。整理得 $10t+5-5t^2 = 1+t^2$，$6t^2 -10t -4 = 0$，两边除以2得 $3t^2 -5t -2 = 0$，解得 $t=2$ 或 $t=-\frac{1}{3}$。
			\par\textbf{理由：} 一元二次方程求解。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\tan\frac{x}{2}=2$ 或 $\tan\frac{x}{2}=-\frac{1}{3}$。}
\end{examplebox}
