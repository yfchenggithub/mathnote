\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin\alpha=\dfrac{3}{5}$ 且 $\alpha$ 为第一象限角，求 $\cos\alpha$ 和 $\tan\alpha$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（定号）：}] 因为 $\alpha$ 在第一象限，所以 $\cos\alpha>0$，$\tan\alpha>0$。
			\par\textbf{理由：} 根据三角函数在各象限的符号法则。
		\item[\textbf{第二步（平方关系）：}] 由 $\sin^2\alpha+\cos^2\alpha=1$，得 $\cos^2\alpha=1-\sin^2\alpha=1-\left(\dfrac{3}{5}\right)^2=\dfrac{16}{25}$。
			\par\textbf{理由：} 平方关系移项。
		\item[\textbf{第三步（开方与商数）：}] 开方得 $\cos\alpha=\dfrac{4}{5}$（正号），再由 $\tan\alpha=\dfrac{\sin\alpha}{\cos\alpha}=\dfrac{3/5}{4/5}=\dfrac{3}{4}$。
			\par\textbf{理由：} 开方取正符合第一象限，商数关系求正切。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\cos\alpha=\dfrac{4}{5}$，$\tan\alpha=\dfrac{3}{4}$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $\tan\alpha=2$，求 $\dfrac{\sin\alpha+2\cos\alpha}{\cos\alpha-\sin\alpha}$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（商数关系）：}] 由 $\tan\alpha=\dfrac{\sin\alpha}{\cos\alpha}=2$ 得 $\sin\alpha=2\cos\alpha$。
			\par\textbf{理由：} 商数关系移项。
		\item[\textbf{第二步（代入化简）：}] 代入原式：分子 $=2\cos\alpha+2\cos\alpha=4\cos\alpha$，分母 $=\cos\alpha-2\cos\alpha=-\cos\alpha$。
			\par\textbf{理由：} 将 $\sin\alpha$ 替换为 $2\cos\alpha$。
		\item[\textbf{第三步（约分求值）：}] 原式 $=\dfrac{4\cos\alpha}{-\cos\alpha}=-4$（$\cos\alpha\neq0$ 因为 $\tan\alpha$ 存在）。
			\par\textbf{理由：} 约去 $\cos\alpha$，分母不为零。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	原式的值为 $-4$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	化简 $\dfrac{\sqrt{1-\sin^2\theta}}{\cos\theta}$，已知 $\sin\theta=\dfrac{4}{5}$ 且 $\theta$ 在第二象限。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平方变形）：}] 由 $\sin^2\theta+\cos^2\theta=1$ 得 $1-\sin^2\theta=\cos^2\theta$。
			\par\textbf{理由：} 平方关系移项。
		\item[\textbf{第二步（开方与符号）：}] $\sqrt{1-\sin^2\theta}=\sqrt{\cos^2\theta}=|\cos\theta|$。因为 $\theta$ 在第二象限，$\cos\theta<0$，所以 $|\cos\theta|=-\cos\theta$。
			\par\textbf{理由：} 算术平方根为非负，去绝对值需结合象限符号。
		\item[\textbf{第三步（代入求值）：}] 原式 $=\dfrac{-\cos\theta}{\cos\theta}=-1$（$\cos\theta\neq0$）。
			\par\textbf{理由：} 约分，分母不为零。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	化简结果为 $-1$。
