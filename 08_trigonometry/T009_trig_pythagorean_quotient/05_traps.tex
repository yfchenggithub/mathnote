\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略符号）}}] 已知 $\sin\alpha=\frac{3}{5}$ 就直接得出 $\cos\alpha=\frac{4}{5}$，不考虑 $\alpha$ 所在象限。
		\item[\textbf{\textcolor{CTrap}{正确理解（根据象限定号）：}}] 应先判断 $\alpha$ 象限，确定 $\cos\alpha$ 正负，再开方取相应符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（开方忘符号）：}}] 平方关系给出的是 $\cos^2\alpha$，开方得 $|\cos\alpha|$，必须结合象限定号。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分母为零）}}] 直接写出 $\tan\alpha=\frac{\sin\alpha}{\cos\alpha}$ 而不注明 $\cos\alpha\neq0$，导致在 $\cos\alpha=0$ 时错误使用。
		\item[\textbf{\textcolor{CTrap}{正确理解（条件要注明）：}}] 商数关系仅在 $\cos\alpha\neq0$ 时成立，使用前必须确认分母不为零。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视定义域）：}}] $\tan\alpha$ 的定义域是 $\alpha\neq\frac{\pi}{2}+k\pi$，盲目套用公式忽略前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公式误记）}}] 将 $\sin^2\alpha+\cos^2\alpha=1$ 写成 $\sin\alpha+\cos\alpha=1$ 或 $\sin^2\alpha+\cos^2\alpha=0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（准确记忆）：}}] 正确公式是 $\sin^2\alpha+\cos^2\alpha=1$，平方和而不是和。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆平方和与和）：}}] 初学时容易丢掉平方符号，或与特殊角公式混淆，需反复默写验证。
	\end{description}
\end{trapbox}
