\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（函数形式）：}}] 函数 $y = A\sin(\omega x + \varphi)$，其中 $A\neq 0$，$\omega\neq 0$，$x\in\mathbb{R}$。
		\item[\textbf{\textcolor{CStatement}{条件 2（整数参数）：}}] $k\in\mathbb{Z}$，表示对称轴与对称中心的序号。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对称轴与中心）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正弦型曲线对称轴对应最值点，对称中心对应零点，由相位等于特定值决定。
			\[
				\begin{aligned}
					\text{对称轴:}&\quad \omega x+\varphi = k\pi+\frac{\pi}{2} \quad\Rightarrow\quad x = \frac{k\pi+\frac{\pi}{2}-\varphi}{\omega},\\[4pt]
					\text{对称中心:}&\quad \omega x+\varphi = k\pi \quad\Rightarrow\quad \left(\frac{k\pi-\varphi}{\omega},0\right).
			\end{aligned}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论(相邻间距):}}]
			\textbf{\textcolor{CStatement}{直观描述:}}
			相邻对称轴(或对称中心)之间的水平距离为$\frac{\pi}{|\omega|}$.
			\[
				\Delta x = \frac{\pi}{|\omega|}.
			\]
	\end{description}
\end{statementbox}
