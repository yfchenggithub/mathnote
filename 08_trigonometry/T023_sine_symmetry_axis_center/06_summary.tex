\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{正弦型函数的对称轴与对称中心由相位取特定值决定，通过整体代换可快速求解。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（标准函数）：}函数必须为 $y = A\sin(\omega x+\varphi)$，其中 $A\neq 0,\ \omega\neq 0$，定义域 $\mathbb{R}$。
		\item \textbf{条件 2（整数参数）：}$k\in\mathbb{Z}$ 给出所有对称轴和对称中心。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件混淆）：}区分对称轴（$\omega x+\varphi = \frac{\pi}{2}+k\pi$）和对称中心（$\omega x+\varphi = k\pi$）的相位条件。
		\item \textbf{检查项（系数处理）：}解 $x$ 时务必除以 $\omega$，注意 $\omega\neq 0$，当 $\omega<0$ 时公式仍然正确，无需取绝对值。
	\end{itemize}
\end{summarybox}
