\begin{examplebox}

	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	求 $\sin^2 15^\circ$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代入降幂公式）：}]
			使用公式 $\sin^2 \theta = \frac{1-\cos 2\theta}{2}$。
			\[
				\sin^2 15^\circ = \frac{1-\cos 30^\circ}{2}.
			\]
			\par\textbf{理由：} 将 $\theta=15^\circ$ 直接代入降幂公式。
		\item[\textbf{第二步（计算余弦值）：}]
			已知 $\cos 30^\circ = \frac{\sqrt{3}}{2}$，代入。
			\[
				\sin^2 15^\circ = \frac{1-\frac{\sqrt{3}}{2}}{2}.
			\]
			\par\textbf{理由：} 代入特殊角函数值。
		\item[\textbf{第三步（化简结果）：}]
			化简分式。
			\[
				\sin^2 15^\circ = \frac{2-\sqrt{3}}{4}.
			\]
			\par\textbf{理由：} 分子分母同乘 $2$，化简得 $\frac{2-\sqrt{3}}{4}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin^2 15^\circ = \frac{2-\sqrt{3}}{4}$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	求函数 $f(x)=\sin^2 x - \cos^2 x$ 的最小正周期和最大值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用降幂公式）：}]
			写出 $\sin^2 x$ 和 $\cos^2 x$ 的降幂形式：
			\[
				\sin^2 x = \frac{1-\cos 2x}{2},
				\qquad
				\cos^2 x = \frac{1+\cos 2x}{2}.
			\]
			\par\textbf{理由：} 标准降幂公式。
		\item[\textbf{第二步（代入化简）：}]
			\[
				\begin{aligned}
					f(x) &= \frac{1-\cos 2x}{2} - \frac{1+\cos 2x}{2} \\
					&= \frac{(1-\cos 2x)-(1+\cos 2x)}{2} \\
					&= \frac{-2\cos 2x}{2} \\
					&= -\cos 2x.
			\end{aligned}
			\]
			\par\textbf{理由：} 通分合并同类项。
		\item[\textbf{第三步（求周期与最值）：}]
			由 $f(x) = -\cos 2x$，周期 $T = \frac{2\pi}{2} = \pi$；$\cos 2x$ 值域 $[-1,1]$，故 $f(x)$ 最大值 $1$。
			\[
				T = \pi,\quad f_{\max} = 1.
			\]
			\par\textbf{理由：} 余弦函数周期公式及最值性质。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最小正周期 $\pi$，最大值 $1$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $x\in[0,\pi]$，求函数 $f(x)=\sin^2 x + \sin x\cos x$ 的最大值和最小值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（降幂与倍角）：}]
			使用降幂公式和正弦倍角公式：
			\[
				\sin^2 x = \frac{1-\cos 2x}{2},
				\qquad
				\sin x\cos x = \frac{1}{2}\sin 2x.
			\]
			\par\textbf{理由：} $\sin x\cos x = \frac{1}{2}\sin 2x$ 是正弦倍角公式。
		\item[\textbf{第二步（化简为一角一函）：}]
			\[
				\begin{aligned}
					f(x) &= \frac{1}{2} - \frac{1}{2}\cos 2x + \frac{1}{2}\sin 2x \\
					&= \frac{1}{2} + \frac{1}{2}(\sin 2x - \cos 2x) \\
					&= \frac{1}{2} + \frac{\sqrt{2}}{2}\sin\left(2x - \frac{\pi}{4}\right).
			\end{aligned}
			\]
			\par\textbf{理由：} 辅助角公式 $\sin 2x - \cos 2x = \sqrt{2}\sin(2x-\frac{\pi}{4})$。
		\item[\textbf{第三步（确定取值范围）：}]
			由 $x\in[0,\pi]$ 得 $2x-\frac{\pi}{4}\in[-\frac{\pi}{4},\frac{7\pi}{4}]$。

			在此区间内，$\sin(2x-\frac{\pi}{4})$ 在 $2x-\frac{\pi}{4}=\frac{3\pi}{2}$ 时取得最小值 $-1$；在 $2x-\frac{\pi}{4}=\frac{\pi}{2}$ 时取得最大值 $1$（端点处为 $-\frac{\sqrt{2}}{2}$，非最值）。
			\[
				f_{\min} = \frac12 - \frac{\sqrt{2}}{2},\quad f_{\max} = \frac12 + \frac{\sqrt{2}}{2}.
			\]
			\par\textbf{理由：} 结合正弦函数图像与给定区间分析极值点。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最大值 $\frac12 + \frac{\sqrt{2}}{2}$，最小值 $\frac12 - \frac{\sqrt{2}}{2}$。}
