\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式符号混淆）}}] 将 $\sin^2 x$ 误写成 $\frac{1+\cos 2x}{2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（分子符号判断）：}}] $\sin^2 x$ 分子为 $1-\cos 2x$，$\cos^2 x$ 分子为 $1+\cos 2x$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆方法缺失）：}}] 未从 $\cos 2x = 1-2\sin^2 x$ 推导，仅靠机械记忆导致混淆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（分母遗漏）}}] 写为 $\sin^2 x = 1-\cos 2x$ 或 $\cos^2 x = 1+\cos 2x$，漏掉分母 $2$。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数完整性）：}}] 公式精确形式包含分母 $2$，即 $\sin^2 x = \frac{1-\cos 2x}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（运算不严谨）：}}] 移项后忘记两边同除以 $2$，属于代数操作疏忽。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（半角代换错误）}}] 求 $\sin^2 \frac{x}{2}$ 时误写成 $\frac{1-\cos \frac{x}{2}}{2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（整体代换原则）：}}] 令 $t=\frac{x}{2}$，则 $\sin^2 \frac{x}{2} = \frac{1-\cos(2\cdot\frac{x}{2})}{2} = \frac{1-\cos x}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（倍角关系不清）：}}] 不理解公式中 $2x$ 是 $x$ 的二倍，代换后 $2$ 倍仍作用于新变量。
	\end{description}
\end{trapbox}
