\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 在 $\triangle ABC$ 中，已知 $A=30^\circ$，$B=45^\circ$，$a=10$，求 $b$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别已知条件）：}] 已知两角 $A=30^\circ, B=45^\circ$ 及一边 $a=10$，要求另一边 $b$，属于 AAS 类型，可用正弦定理直接求解。
		      \par\textbf{理由：}
		      正弦定理 $\frac{a}{\sin A}=\frac{b}{\sin B}$ 中含有未知边 $b$，可直接解出。

		\item[\textbf{第二步（代入公式）：}] 由 $\frac{a}{\sin A}=\frac{b}{\sin B}$，得 $b=\frac{a\sin B}{\sin A}$。
		      \par\textbf{理由：}
		      将等式变形，用已知量表示未知量。

		\item[\textbf{第三步（计算数值）：}] 代入 $\sin 45^\circ=\frac{\sqrt{2}}{2}$，$\sin 30^\circ=\frac12$，得
		      \[
			      b = 10\times \frac{\frac{\sqrt{2}}{2}}{\frac12}
			      = 10\times \frac{\sqrt{2}}{2}\times 2
			      = 10\sqrt{2}.
		      \]
		      \par\textbf{理由：}
		      数值计算得到最终结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$b=10\sqrt{2}$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 在 $\triangle ABC$ 中，$b=2$，$c=\sqrt{6}$，$B=45^\circ$，求角 $C$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别已知条件）：}] 已知两边 $b=2$，$c=\sqrt{6}$ 及一边对角 $B=45^\circ$，求另一对角 $C$，属于 SSA 类型，可用正弦定理，但要注意可能有多解。
		      \par\textbf{理由：}
		      正弦定理 $\frac{b}{\sin B}=\frac{c}{\sin C}$ 中含有未知角 $C$ 的正弦；SSA 类型不能默认唯一解。

		\item[\textbf{第二步（代入求 $\sin C$）：}] 由 $\frac{b}{\sin B}=\frac{c}{\sin C}$，得
		      \[
			      \sin C = \frac{c\sin B}{b}
			      = \frac{\sqrt{6}\times \sin 45^\circ}{2}.
		      \]
		      继续计算：
		      \[
			      \sin C
			      = \frac{\sqrt{6}\times \frac{\sqrt{2}}{2}}{2}
			      = \frac{\sqrt{12}}{4}
			      = \frac{2\sqrt{3}}{4}
			      = \frac{\sqrt{3}}{2}.
		      \]
		      \par\textbf{理由：}
		      代数运算得到角 $C$ 的正弦值。

		\item[\textbf{第三步（解角并判断多解）：}] 由于 $\sin C=\frac{\sqrt{3}}{2}$，且 $0^\circ<C<180^\circ$，所以
		      \[
			      C=60^\circ \quad \text{或} \quad C=120^\circ.
		      \]
		      又因为 $c>b$，所以 $C>B=45^\circ$，两个候选角都满足这一条件。
		      \par\textbf{理由：}
		      大边对大角只能作为辅助筛选条件，不能保证一定只剩一个解。

		\item[\textbf{第四步（检查内角和）：}] 若 $C=60^\circ$，则
		      \[
			      A=180^\circ-45^\circ-60^\circ=75^\circ>0.
		      \]
		      若 $C=120^\circ$，则
		      \[
			      A=180^\circ-45^\circ-120^\circ=15^\circ>0.
		      \]
		      因此两个角都能构成三角形，两个解均成立。
		      \par\textbf{理由：}
		      SSA 情形求出候选角后，必须检查内角和是否为正，满足条件的解都要保留。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$C=60^\circ$ 或 $C=120^\circ$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 在 $\triangle ABC$ 中，已知 $\sin A:\sin B:\sin C = 3:5:7$，求最大角 $C$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（化角为边）：}] 由正弦定理，$a:b:c = \sin A:\sin B:\sin C = 3:5:7$，设 $a=3k$，$b=5k$，$c=7k$（$k>0$）。
		      \par\textbf{理由：}
		      正弦定理实现边角互化，将角的正弦比直接转化为边之比。且 $3k+5k>7k$，能构成三角形。

		\item[\textbf{第二步（判断最大角）：}] 因为 $c$ 最大，所以角 $C$ 最大。用余弦定理求 $\cos C$。
		      \par\textbf{理由：}
		      三角形中大边对大角，已知三边比求角时可用余弦定理。

		\item[\textbf{第三步（余弦定理求角）：}] 由余弦定理：
		      \[
			      \cos C
			      = \frac{a^2+b^2-c^2}{2ab}
			      = \frac{(3k)^2+(5k)^2-(7k)^2}{2\cdot 3k \cdot 5k}
			      = \frac{9k^2+25k^2-49k^2}{30k^2}
			      = \frac{-15k^2}{30k^2}
			      = -\frac12.
		      \]
		      所以 $C = 120^\circ$。
		      \par\textbf{理由：}
		      $\cos C=-\frac12$，且 $0^\circ<C<180^\circ$，故 $C=120^\circ$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$C=120^\circ$。}
\end{examplebox}
