\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忘记系数2）：}}] 将正弦定理误写为 $\frac{a}{\sin A}=R$，忽略 $2R$。
		\item[\textbf{\textcolor{CTrap}{正确理解（比值是直径）：}}] 正弦定理中的比值是外接圆直径 $2R$，即 $\frac{a}{\sin A}=2R$，不是 $R$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}] 死记硬背公式，不理解 $2R$ 的来源（直径），导致系数错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（边角互化丢系数）：}}] 在边角转化时直接写 $a=\sin A$ 或 $\sin A=a$，忘记乘以或除以 $2R$。
		\item[\textbf{\textcolor{CTrap}{正确理解（完整转化式）：}}] 正确关系是 $a=2R\sin A$，$\sin A=\frac{a}{2R}$，不能漏掉 $2R$。
		\item[\textbf{\textcolor{CTrap}{错因分析（比例混淆）：}}] 记住比例相等，但忽略了中间常数 $2R$，将边与角的正弦直接等同。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（SSA 多解遗漏）：}}] 已知两边及一边对角求另一角时，只取计算器给出的锐角解，忽略钝角可能，导致丢解。
		\item[\textbf{\textcolor{CTrap}{正确理解（正弦值可能对应两角）：}}] 当 $\sin C = m$（$0<m<1$）时，在 $0^\circ<C<180^\circ$ 内，$C$ 可能是锐角，也可能是钝角。求出候选角后，要结合大边对大角和内角和一起判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（习惯性锐角）：}}] 默认三角形内角为锐角，或在利用计算器求角时只得到锐角主值，没有想到 $180^\circ-C$ 也可能满足同一个正弦值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（只用大边对大角取舍）：}}] SSA 情形中，算出两个候选角后，只用“大边对大角”判断，误以为一定能排除一个。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须检查内角和）：}}] 大边对大角只是辅助筛选条件。最终还要检查
		      \[
			      180^\circ-B-C>0.
		      \]
		      只要候选角满足边角大小关系和内角和条件，就应保留。
		\item[\textbf{\textcolor{CTrap}{错因分析（筛选条件不完整）：}}] 把必要条件当成充分条件，没有意识到多个候选角可能同时满足三角形条件。
	\end{description}
\end{trapbox}
