\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（三角形）：}}] 在非退化 $\triangle ABC$ 中，$a=BC$, $b=CA$, $c=AB$，外接圆半径为 $R$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正弦定理）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 三角形中边与对角的正弦之比相等，等于外接圆直径。
		      \[
			      \frac{a}{\sin A} = \frac{b}{\sin B} = \frac{c}{\sin C} = 2R.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（边角互化）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 每条边等于对角正弦与外接圆直径的乘积。
		      \[
			      a = 2R\sin A,\quad b = 2R\sin B,\quad c = 2R\sin C.
		      \]
	\end{description}

\end{statementbox}
