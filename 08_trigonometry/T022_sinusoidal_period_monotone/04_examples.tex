\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $y=3\sin(2x-\frac{\pi}{3})$ 的周期和单调区间。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别周期）：}]
			周期 $T=\frac{2\pi}{2}=\pi$。
			\par\textbf{理由：} 周期公式 $T=\frac{2\pi}{\omega}$。
		\item[\textbf{第二步（求增区间）：}]
			解不等式 $-\frac{\pi}{2}+2k\pi \le 2x-\frac{\pi}{3} \le \frac{\pi}{2}+2k\pi$。加 $\frac{\pi}{3}$ 得 $-\frac{\pi}{6}+2k\pi \le 2x \le \frac{5\pi}{6}+2k\pi$。除以 $2$ 得 $-\frac{\pi}{12}+k\pi \le x \le \frac{5\pi}{12}+k\pi$。所以增区间为 $[-\frac{\pi}{12}+k\pi, \frac{5\pi}{12}+k\pi]$（$k\in\mathbb{Z}$）。
			\par\textbf{理由：} 整体代换，解一元一次不等式。
		\item[\textbf{第三步（求减区间）：}]
			解不等式 $\frac{\pi}{2}+2k\pi \le 2x-\frac{\pi}{3} \le \frac{3\pi}{2}+2k\pi$。加 $\frac{\pi}{3}$ 得 $\frac{5\pi}{6}+2k\pi \le 2x \le \frac{11\pi}{6}+2k\pi$。除以 $2$ 得 $\frac{5\pi}{12}+k\pi \le x \le \frac{11\pi}{12}+k\pi$。所以减区间为 $[\frac{5\pi}{12}+k\pi, \frac{11\pi}{12}+k\pi]$（$k\in\mathbb{Z}$）。
			\par\textbf{理由：} 整体代换。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	周期为 $\pi$，增区间为 $[-\frac{\pi}{12}+k\pi, \frac{5\pi}{12}+k\pi]$，减区间为 $[\frac{5\pi}{12}+k\pi, \frac{11\pi}{12}+k\pi]$（$k\in\mathbb{Z}$）。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 求函数 $y=-2\sin(3x+\frac{\pi}{4})$ 的单调递增区间（$A<0$）。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			$A=-2$（$A<0$），$\omega=3>0$，$\varphi=\frac{\pi}{4}$。周期 $T=\frac{2\pi}{3}$。
			\par\textbf{理由：} 周期公式 $T=\frac{2\pi}{\omega}$。
		\item[\textbf{第二步（求对应区间）：}]
			因为 $A<0$，原函数的单调递增区间对应 $A>0$ 时的单调递减区间。所以求 $A>0$ 时的减区间：解 $\frac{\pi}{2}+2k\pi \le 3x+\frac{\pi}{4} \le \frac{3\pi}{2}+2k\pi$。减 $\frac{\pi}{4}$ 得 $\frac{\pi}{4}+2k\pi \le 3x \le \frac{5\pi}{4}+2k\pi$。除以 $3$ 得 $\frac{\pi}{12}+\frac{2k\pi}{3} \le x \le \frac{5\pi}{12}+\frac{2k\pi}{3}$。
			\par\textbf{理由：} $A<0$ 时单调性相反。
		\item[\textbf{第三步（得增区间）：}]
			所以原函数的单调递增区间为 $[\frac{\pi}{12}+\frac{2k\pi}{3}, \frac{5\pi}{12}+\frac{2k\pi}{3}]$（$k\in\mathbb{Z}$）。
			\par\textbf{理由：} $A<0$ 时增区间与 $A>0$ 时减区间相同。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	单调递增区间为 $[\frac{\pi}{12}+\frac{2k\pi}{3}, \frac{5\pi}{12}+\frac{2k\pi}{3}]$（$k\in\mathbb{Z}$）。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 设 $\omega>0$，函数 $f(x)=\sin(\omega x+\frac{\pi}{4})$ 在区间 $(0,\frac{\pi}{2})$ 上单调递增，求 $\omega$ 的取值范围。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写通式）：}]
			由 $A=1>0$，增区间满足 $-\frac{\pi}{2}+2k\pi \le \omega x+\frac{\pi}{4} \le \frac{\pi}{2}+2k\pi$。解得
			\[
				\frac{-\frac{3\pi}{4}+2k\pi}{\omega} \le x \le \frac{\frac{\pi}{4}+2k\pi}{\omega}.
			\]
			\par\textbf{理由：} 整体代换解不等式。
		\item[\textbf{第二步（定k值）：}]
			要使 $(0,\frac{\pi}{2})$ 包含于某个增区间，取 $k=0$ 得增区间 $[-\frac{3\pi}{4\omega}, \frac{\pi}{4\omega}]$。
			\par\textbf{理由：} 区间长度分析，$k=0$ 对应的区间原点附近。
		\item[\textbf{第三步（解不等式）：}]
			由于 $(0,\frac{\pi}{2})$ 含于该区间，需满足 $\frac{\pi}{4\omega} \ge \frac{\pi}{2}$，自动满足左端点 $\le 0$。解得 $\omega \le \frac{1}{2}$。
			\par\textbf{理由：} 区间包含条件。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\omega$ 的取值范围是 $(0, \frac{1}{2}]$。
