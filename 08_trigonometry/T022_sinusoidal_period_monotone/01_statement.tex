\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（角频率为正）：}}]
			$\omega > 0$。
		\item[\textbf{\textcolor{CStatement}{条件 2（振幅非零）：}}]
			$A \neq 0$。
		\item[\textbf{\textcolor{CStatement}{条件 3（定义域整数）：}}]
			$x \in \mathbb{R}$, $k \in \mathbb{Z}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（周期公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			周期只与 $\omega$ 有关，与 $A$, $\varphi$ 无关。
			\[
				T = \frac{2\pi}{\omega}.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（增区间求法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正弦在半个周期内递增。
			\[
				-\frac{\pi}{2} + 2k\pi \le \omega x + \varphi \le \frac{\pi}{2} + 2k\pi
			\]
			\textbf{\textcolor{CStatement}{解出 $x$ 得到增区间。}}
		\item[\textbf{\textcolor{CStatement}{结论三（减区间求法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正弦在半个周期内递减。
			\[
				\frac{\pi}{2} + 2k\pi \le \omega x + \varphi \le \frac{3\pi}{2} + 2k\pi
			\]
			\textbf{\textcolor{CStatement}{解出 $x$ 得到减区间。}}
		\item[\textbf{\textcolor{CStatement}{推广结论（符号影响单调性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $A < 0$ 时，单调区间互换。当 $A < 0$ 时：
			\[
				\frac{\pi}{2} + 2k\pi \le \omega x + \varphi \le \frac{3\pi}{2} + 2k\pi
			\]
			为增区间条件；
			\[
				-\frac{\pi}{2} + 2k\pi \le \omega x + \varphi \le \frac{\pi}{2} + 2k\pi
			\]
			为减区间条件。
	\end{description}

\end{statementbox}
