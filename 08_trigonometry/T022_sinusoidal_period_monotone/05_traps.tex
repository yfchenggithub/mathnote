\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略 $\omega$ 符号）}}]
			错误做法：直接套用区间不等式，未注意 $\omega$ 的正负。
		\item[\textbf{\textcolor{CTrap}{正确理解（使用前提）：}}]
			本结论要求 $\omega>0$；若 $\omega<0$，应先用诱导公式化为 $\omega>0$ 的形式，或通过导数单独判断。
		\item[\textbf{\textcolor{CTrap}{错因分析（误解结论适用范围）：}}]
			解不等式时系数为负会改变不等号方向，导致区间错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略 A 符号）}}]
			错误做法：不论 $A$ 正负均使用 $A>0$ 的单调区间。
		\item[\textbf{\textcolor{CTrap}{正确理解（振幅符号影响）：}}]
			$A<0$ 时单调区间与 $A>0$ 时相反。
		\item[\textbf{\textcolor{CTrap}{错因分析（误以为振幅不影响单调性）：}}]
			误认为 $A$ 只改变函数值幅度，不改变增减性。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（遗漏周期性）}}]
			错误做法：只写一个周期内的单调区间，不加 $k\in\mathbb{Z}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（整数参数必须写）：}}]
			所有单调区间应包含整数 $k$ 的通项表示。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视三角函数的周期性）：}}]
			三角函数具有周期性，单调区间每 $\frac{2\pi}{\omega}$ 重复一次。
	\end{description}

\end{trapbox}
