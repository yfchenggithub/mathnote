\[
	\boxed{\text{诱导公式：奇变偶不变，符号看象限}}
\]

\[
	k\in\mathbb{Z}
\]

\[
	\begin{aligned}
		\sin(2k\pi+\alpha) & =\sin\alpha, \\
		\cos(2k\pi+\alpha) & =\cos\alpha, \\
		\tan(k\pi+\alpha)  & =\tan\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin(-\alpha) & =-\sin\alpha, \\
		\cos(-\alpha) & =\cos\alpha,  \\
		\tan(-\alpha) & =-\tan\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin(\pi-\alpha) & =\sin\alpha,  \\
		\cos(\pi-\alpha) & =-\cos\alpha, \\
		\tan(\pi-\alpha) & =-\tan\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin(\pi+\alpha) & =-\sin\alpha, \\
		\cos(\pi+\alpha) & =-\cos\alpha, \\
		\tan(\pi+\alpha) & =\tan\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin\left(\frac{\pi}{2}-\alpha\right) & =\cos\alpha, \\
		\cos\left(\frac{\pi}{2}-\alpha\right) & =\sin\alpha, \\
		\tan\left(\frac{\pi}{2}-\alpha\right) & =\cot\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin\left(\frac{\pi}{2}+\alpha\right) & =\cos\alpha,  \\
		\cos\left(\frac{\pi}{2}+\alpha\right) & =-\sin\alpha, \\
		\tan\left(\frac{\pi}{2}+\alpha\right) & =-\cot\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin\left(\frac{3\pi}{2}-\alpha\right) & =-\cos\alpha, \\
		\cos\left(\frac{3\pi}{2}-\alpha\right) & =-\sin\alpha, \\
		\tan\left(\frac{3\pi}{2}-\alpha\right) & =\cot\alpha.
	\end{aligned}
\]

\[
	\begin{aligned}
		\sin\left(\frac{3\pi}{2}+\alpha\right) & =-\cos\alpha, \\
		\cos\left(\frac{3\pi}{2}+\alpha\right) & =\sin\alpha,  \\
		\tan\left(\frac{3\pi}{2}+\alpha\right) & =-\cot\alpha.
	\end{aligned}
\]

\[
	\begin{array}{c|ccc}
		\text{角} & \sin        & \cos & \tan \\ \hline
		-\alpha
		         & -\sin\alpha
		         & \cos\alpha
		         & -\tan\alpha               \\[4pt]

		\pi-\alpha
		         & \sin\alpha
		         & -\cos\alpha
		         & -\tan\alpha               \\[4pt]

		\pi+\alpha
		         & -\sin\alpha
		         & -\cos\alpha
		         & \tan\alpha                \\[4pt]

		\dfrac{\pi}{2}-\alpha
		         & \cos\alpha
		         & \sin\alpha
		         & \cot\alpha                \\[6pt]

		\dfrac{\pi}{2}+\alpha
		         & \cos\alpha
		         & -\sin\alpha
		         & -\cot\alpha               \\[6pt]

		\dfrac{3\pi}{2}-\alpha
		         & -\cos\alpha
		         & -\sin\alpha
		         & \cot\alpha                \\[6pt]

		\dfrac{3\pi}{2}+\alpha
		         & -\cos\alpha
		         & \sin\alpha
		         & -\cot\alpha
	\end{array}
\]

\[
	\begin{cases}
		k\text{ 为奇数，函数名改变：}\sin\leftrightarrow\cos,\quad \tan\leftrightarrow\cot, \\
		k\text{ 为偶数，函数名不变},                                                       \\
		\text{符号由原三角函数在对应象限的正负决定}.
	\end{cases}
\]

\[
	\text{注意：涉及 }\tan,\cot\text{ 的公式，要求等式两边都有意义。}
\]
