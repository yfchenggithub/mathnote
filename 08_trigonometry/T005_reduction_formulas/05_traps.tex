\begin{trapbox}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
    \item[\textbf{\textcolor{CTrap}{易错点一（把口诀条件当成公式条件）}}] 误以为诱导公式只能在 $\alpha$ 为锐角时使用，或者在 $\alpha$ 不是锐角时直接用实际 $\alpha$ 的大小判象限。
    \item[\textbf{\textcolor{CTrap}{正确理解（公式任意角，口诀参考角）：}}] 诱导公式本身对任意角 $\alpha$ 成立；使用“符号看象限”口诀时，才把 $\alpha$ 暂看作锐角或参考角，再判断整体角 $\frac{k\pi}{2}\pm\alpha$ 的象限。
    \item[\textbf{\textcolor{CTrap}{错因分析（混淆层次）：}}] 把严格公式中的任意角 $\alpha$ 与口诀判号中的参考角 $\alpha$ 混为一谈，容易造成符号判断错误。
\end{description}

\medskip
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
    \item[\textbf{\textcolor{CTrap}{易错点二（k的奇偶误判）}}] 错误认为 $\pi$ 对应“奇变”，实际 $\pi=2\cdot\frac{\pi}{2}$ 中 $k=2$ 是偶数，函数名不变。
    \item[\textbf{\textcolor{CTrap}{正确理解（数k要准确）：}}] 将角度写成 $\frac{k\pi}{2}\pm\alpha$，$k$ 为整数，偶则名不变，奇则正余弦互换、正余切互换。
    \item[\textbf{\textcolor{CTrap}{错因分析（粗心计数）：}}] 易把 $\pi$ 当成“奇数个 $\pi/2$”而误判为 $k$ 奇，实际上 $\pi$ 是 $2$ 个 $\pi/2$，$k=2$ 偶。
\end{description}

\medskip
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
    \item[\textbf{\textcolor{CTrap}{易错点三（忽略正切、余切定义域）}}] 使用 $\tan(\frac{\pi}{2}+\alpha)=-\cot\alpha$ 时，未检查两边是否有意义。
    \item[\textbf{\textcolor{CTrap}{正确理解（逐条检查分母）：}}] 出现 $\tan\theta$ 要求 $\cos\theta\neq0$，出现 $\cot\theta$ 要求 $\sin\theta\neq0$。例如 $\tan(\frac{\pi}{2}+\alpha)=-\cot\alpha$ 两边有意义的条件是 $\sin\alpha\neq0$。
    \item[\textbf{\textcolor{CTrap}{错因分析（机械套公式）：}}] 只记公式形式，忽略成立条件，可能在方程或化简中引入增根或错误结论。
\end{description}

\medskip
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
    \item[\textbf{\textcolor{CTrap}{易错点四（把 $\pi\pm\alpha$ 都理解成简单变号）}}] 认为“半周期”一定让所有三角函数都变号。
    \item[\textbf{\textcolor{CTrap}{正确理解（区分 $\pi+\alpha$ 与 $\pi-\alpha$）：}}] $\pi+\alpha$ 中，$\sin$、$\cos$ 变号，而 $\tan$、$\cot$ 不变；$\pi-\alpha$ 中，$\sin$ 不变，$\cos$、$\tan$、$\cot$ 变号。
    \item[\textbf{\textcolor{CTrap}{错因分析（只背半句）：}}] 只记“半周期改变符号”，没有结合单位圆对称关系与函数定义，容易把正切、余切的符号判断错。
\end{description}
\end{trapbox}
