\begin{examplebox}
\textbf{\textcolor{CExample}{例 1（基础）}}
\textbf{\textcolor{CExample}{题目：}} 求 $\sin 150^\circ$ 的值。
\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{第一步（拆角）：}] $150^\circ = 180^\circ-30^\circ$。
    \par\textbf{理由：} 写成 $\pi-\alpha$ 形式，其中 $30^\circ$ 是参考角。
    \item[\textbf{第二步（套公式）：}] $\sin(180^\circ-30^\circ)=\sin30^\circ$。
    \par\textbf{理由：} 诱导公式 $\sin(\pi-
    \alpha)=\sin\alpha$；也可用口诀理解为 $\pi=2\cdot\frac{\pi}{2}$，$k=2$ 为偶数，函数名不变，且第二象限正弦为正。
    \item[\textbf{第三步（求值）：}] $\sin30^\circ=\dfrac12$。
    \par\textbf{理由：} 特殊角三角函数值。
\end{description}
\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\dfrac12$}。

\medskip
\textbf{\textcolor{CExample}{例 2（稍变形）}}
\textbf{\textcolor{CExample}{题目：}} 化简 $\cos240^\circ + \tan135^\circ$。
\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{第一步（化简 $\cos240^\circ$）：}] $240^\circ = 180^\circ+60^\circ$，
    \[
    \cos(180^\circ+60^\circ)=-\cos60^\circ=-\dfrac12.
    \]
    \par\textbf{理由：} $\cos(\pi+\alpha)=-\cos\alpha$，$\pi+\alpha$ 在第三象限，余弦为负。
    \item[\textbf{第二步（化简 $\tan135^\circ$）：}] $135^\circ = 180^\circ-45^\circ$，
    \[
    \tan(180^\circ-45^\circ)=-\tan45^\circ=-1.
    \]
    \par\textbf{理由：} $\tan(\pi-\alpha)=-\tan\alpha$，且本题中各正切值均有定义。
    \item[\textbf{第三步（求和）：}] $-\dfrac12 + (-1) = -\dfrac32$。
    \par\textbf{理由：} 有理数加法。
\end{description}
\textbf{\textcolor{CExample}{关键结论：}} \textbf{$-\dfrac32$}。

\medskip
\textbf{\textcolor{CExample}{例 3（综合应用）}}
\textbf{\textcolor{CExample}{题目：}} 已知 $\sin(\pi+\alpha)=-\dfrac12$，$\alpha\in(0,\dfrac{\pi}{2})$，求 $\cos(2\pi-\alpha)$ 的值。
\textbf{\textcolor{CExample}{解题步骤：}}
\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
    \item[\textbf{第一步（由已知求 $\sin\alpha$）：}] $\sin(\pi+\alpha)=-\sin\alpha=-\dfrac12$，故 $\sin\alpha=\dfrac12$。
    \par\textbf{理由：} 诱导公式 $\sin(\pi+\alpha)=-\sin\alpha$，$\pi+\alpha$ 在第三象限，正弦为负。
    \item[\textbf{第二步（求 $\cos\alpha$）：}] 由 $\sin^2\alpha+
    \cos^2\alpha=1$ 及 $\alpha\in(0,\frac{\pi}{2})$ 得
    \[
    \cos\alpha=\sqrt{1-\left(\frac12\right)^2}=\dfrac{\sqrt3}{2}.
    \]
    \par\textbf{理由：} 同角三角函数基本关系；$\alpha$ 是第一象限角，所以 $\cos\alpha>0$。
    \item[\textbf{第三步（求目标）：}] 
    \[
    \cos(2\pi-\alpha)=\cos(-\alpha)=\cos\alpha=\dfrac{\sqrt3}{2}.
    \]
    \par\textbf{理由：} $2\pi-\alpha$ 与 $-\alpha$ 同终边，先用 $2\pi$ 周期性得到 $\cos(2\pi-\alpha)=\cos(-\alpha)$，再用余弦函数是偶函数得到 $\cos(-\alpha)=\cos\alpha$。
\end{description}
\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\dfrac{\sqrt3}{2}$}。
\end{examplebox}
