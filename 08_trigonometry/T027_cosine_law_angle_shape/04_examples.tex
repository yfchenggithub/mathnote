\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a=3$，$b=4$，$c=5$，求 $\cos C$ 并判断 $\triangle ABC$ 的形状。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认最大边）：}] 比较三边长度，得最大边为 $c=5$，其对角 $C$ 为最大角。
			\par\textbf{理由：} 三角形大边对大角，最大角的余弦符号决定形状。

		\item[\textbf{第二步（套用公式）：}] 针对最大角 $C$，使用余弦定理变形：
			\[
				\cos C = \frac{a^{2}+b^{2}-c^{2}}{2ab}.
			\]
			\par\textbf{理由：} 已知三边求余弦的直接公式。

		\item[\textbf{第三步（代入计算）：}] 代入数据：
			\[
				\begin{aligned}
					\cos C &= \frac{3^{2}+4^{2}-5^{2}}{2\cdot3\cdot4} \\
					&= \frac{9+16-25}{24} \\
					&= \frac{0}{24} \\
					&= 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 分子为零，余弦值为零。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	由 $\cos C = 0$ 得 $\angle C = 90^\circ$，所以 $\triangle ABC$ 是直角三角形。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a=7$，$b=8$，$c=9$，求最大角的余弦并判断 $\triangle ABC$ 的形状。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确认最大边）：}] 比较得最大边为 $c=9$，最大角为 $C$。
			\par\textbf{理由：} 大边对大角。

		\item[\textbf{第二步（套用公式）：}] 使用公式
			\[
				\cos C = \frac{a^{2}+b^{2}-c^{2}}{2ab}.
			\]
			\par\textbf{理由：} 余弦定理变形。

		\item[\textbf{第三步（代入计算）：}] 代入数值：
			\[
				\begin{aligned}
					\cos C &= \frac{7^{2}+8^{2}-9^{2}}{2\cdot7\cdot8} \\
					&= \frac{49+64-81}{112} \\
					&= \frac{32}{112} \\
					&= \frac{2}{7} > 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 计算结果为正。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\cos C = \frac{2}{7}>0$，最大角为锐角，所以 $\triangle ABC$ 是锐角三角形。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a:b:c=2:3:4$，求最大角的余弦并判断 $\triangle ABC$ 的形状。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设参数表示边）：}] 设 $a=2k, b=3k, c=4k\;(k>0)$，则最大边 $c=4k$，最大角为 $C$。
			\par\textbf{理由：} 比例设参可具体表示边长。

		\item[\textbf{第二步（套用公式）：}] 代入 $\cos C = \frac{a^{2}+b^{2}-c^{2}}{2ab}$，得
			\[
				\cos C = \frac{(2k)^{2}+(3k)^{2}-(4k)^{2}}{2\cdot2k\cdot3k}.
			\]
			\par\textbf{理由：} 直接代入边长表达式。

		\item[\textbf{第三步（化简求值）：}] 化简：
			\[
				\begin{aligned}
					\cos C &= \frac{4k^{2}+9k^{2}-16k^{2}}{12k^{2}} \\
					&= \frac{-3k^{2}}{12k^{2}} \\
					&= -\frac{1}{4} < 0.
			\end{aligned}
			\]
			\par\textbf{理由：} 分子合并后为负，分母恒正，结果为负。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\cos C = -\frac{1}{4}<0$，最大角为钝角，所以 $\triangle ABC$ 是钝角三角形。
\end{examplebox}
