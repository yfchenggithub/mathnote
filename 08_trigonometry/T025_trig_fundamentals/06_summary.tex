\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{三种基本三角函数的定义域、值域、周期、奇偶性、单调增区间是解决所有三角函数问题的基础。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（函数形式）：} 必须是标准 $\sin$、$\cos$、$\tan$，未经过左右平移或伸缩。
		\item \textbf{条件2（研究范围）：} 性质在实数集上成立，注意正切需排除定义域外的点。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（周期公式）：} 正弦型周期 $T=2\pi/|\omega|$，正切型 $T=\pi/|\omega|$，不要漏绝对值。
		\item \textbf{检查项（定义域）：} 每次使用正切前，必须检查定义域是否包含所求点。
	\end{itemize}

\end{summarybox}
