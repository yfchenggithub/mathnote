\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}
	三角函数的基本性质好比它们的“身份证”，记住这些才能正确使用。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（定义域限制）：}]
			正弦和余弦可对所有实数计算，正切则要避开使分母为零的点（$\frac{\pi}{2}+k\pi$）。
		\item[\textbf{要点二（周期规律）：}]
			正弦和余弦每隔$2\pi$重复，正切每隔$\pi$重复，可以利用周期将任意角转化到基本区间。
		\item[\textbf{要点三（奇偶对称）：}]
			正弦和正切是奇函数，图像关于原点对称；余弦是偶函数，图像关于$y$轴对称。
		\item[\textbf{要点四（单调区间）：}]
			单调增区间是函数值随$x$增大而上升的区间，记住它们方便求解不等式和最值。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（单位圆意义）}}
	正弦对应单位圆上点的纵坐标，余弦对应横坐标，正切对应过单位圆与$x=1$交点的竖直线段长度。定义域和值域直接由此几何意义确定：纵、横坐标范围$[-1,1]$，正切线可无限延伸。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（对称性根源）}}
	奇偶性源于诱导公式：$\sin(-x)=-\sin x$，$\cos(-x)=\cos x$，$\tan(-x)=-\tan x$。周期源于终边相同角：$\sin(x+2\pi)=\sin x$，$\tan(x+\pi)=\tan x$。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（考查形式）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接套用）：} 给出一个三角函数，判断其定义域、值域、周期、奇偶性、单调性。
		\item \textbf{考法二（复合应用）：} 将性质与不等式、图像变换、最值问题结合，例如求$y=2\sin(3x)+1$的周期和值域。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}
	\textbf{记住“正切定义域排除线”是关键易错点；奇偶性可快速判断对称性，周期可把大角度化小。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（函数性质判断）：} 已知一个三角方程或函数式，求其定义域或单调区间。
		\item \textbf{场景二（图像变换）：} 平移伸缩后，周期和值域随之改变，需要套用周期公式和振幅变换规则。
	\end{itemize}

\end{explanationbox}
