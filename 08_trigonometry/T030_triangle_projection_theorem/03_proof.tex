\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	用余弦定理将 $\cos C,\cos B$ 用边长表示，代入后通分合并，即可化简为 $a$.

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（余弦定理表达）：}] 由余弦定理，
			\[
				\cos C = \frac{a^2 + b^2 - c^2}{2ab},
				\qquad
				\cos B = \frac{a^2 + c^2 - b^2}{2ac}.
			\]
			\par\textbf{理由：} 余弦定理给出边角关系.

		\item[\textbf{步骤二（代入表达式）：}] 计算 $b\cos C + c\cos B$:
			\[
				b\cos C + c\cos B
				= b\cdot\frac{a^2 + b^2 - c^2}{2ab} + c\cdot\frac{a^2 + c^2 - b^2}{2ac}.
			\]
			\par\textbf{理由：} 代入余弦公式.

		\item[\textbf{步骤三（化简第一项）：}] 约去 $b$:
			\[
				b\cdot\frac{a^2 + b^2 - c^2}{2ab} = \frac{a^2 + b^2 - c^2}{2a}.
			\]
			\par\textbf{理由：} 分子分母中 $b$ 约分.

		\item[\textbf{步骤四（化简第二项）：}] 约去 $c$:
			\[
				c\cdot\frac{a^2 + c^2 - b^2}{2ac} = \frac{a^2 + c^2 - b^2}{2a}.
			\]
			\par\textbf{理由：} 分子分母中 $c$ 约分.

		\item[\textbf{步骤五（合并通分）：}] 两项分母相同，合并:
			\[
				\begin{aligned}
					b\cos C + c\cos B
					&= \frac{(a^2 + b^2 - c^2) + (a^2 + c^2 - b^2)}{2a} \\
					&= \frac{2a^2}{2a}.
			\end{aligned}
			\]
			\par\textbf{理由：} 同分母相加，注意 $b^2$ 与 $-c^2$ 抵消，$c^2$ 与 $-b^2$ 抵消.

		\item[\textbf{步骤六（约简得结论）：}] 约分:
			\[
				\frac{2a^2}{2a} = a.
			\]
			\par\textbf{理由：} 分子分母约去 $2a$（$a>0$）.

	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	由以上变形得到 $b\cos C + c\cos B = a$，即射影定理成立。类似可证轮换形式。
\end{proofbox}
