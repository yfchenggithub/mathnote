\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（三角形条件）：}}] 任意三角形 $ABC$ 中，内角 $A,B,C$ 所对的边分别为 $a,b,c$.
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（射影公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			边 $a$ 等于两边 $b,c$ 与它们夹角余弦的乘积之和。
			\[
				a = b\cos C + c\cos B
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（轮换对称性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			其他边同样满足投影和形式。
			\[
				b = c\cos A + a\cos C
			\]
			\[
				c = a\cos B + b\cos A
			\]
	\end{description}
\end{statementbox}
