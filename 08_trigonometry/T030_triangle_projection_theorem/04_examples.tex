\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $b = 4$, $c = 3$, $\cos C = \frac{4}{5}$, $\cos B = \frac{3}{5}$，求边 $a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出公式）：}] 由射影定理得
			\[
				a = b\cos C + c\cos B
			\]
			\par\textbf{理由：} 射影定理直接给出边 $a$ 的表达式.

		\item[\textbf{第二步（代入数值）：}] 将已知数据代入
			\[
				a = 4\cdot\frac{4}{5} + 3\cdot\frac{3}{5}
			\]
			\par\textbf{理由：} 代入已知的 $b, c, \cos C, \cos B$.

		\item[\textbf{第三步（计算结果）：}] 计算
			\[
				\begin{aligned}
					a &= \frac{16}{5} + \frac{9}{5} \\
					&= \frac{25}{5} \\
					&= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术运算.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$a = 5$.

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $2b\cos C + 2c\cos B = 10$，求边 $a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}] 注意 $b\cos C + c\cos B = a$，所以原式左边为
			\[
				\begin{aligned}
					2b\cos C + 2c\cos B
					&= 2(b\cos C + c\cos B) \\
					&= 2a.
			\end{aligned}
			\]
			\par\textbf{理由：} 射影定理及提取公因式.

		\item[\textbf{第二步（建立方程）：}] 因此
			\[
				2a = 10
			\]
			\par\textbf{理由：} 已知条件代入.

		\item[\textbf{第三步（解出 $a$）：}] 解得
			\[
				a = 5
			\]
			\par\textbf{理由：} 方程除以2.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$a = 5$.

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，角 $A,B,C$ 的对边分别为 $a,b,c$，且 $b\cos C + c\cos B = 2a\sin A$. 若 $A$ 为锐角，求角 $A$ 的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（代换左边）：}] 由射影定理，$b\cos C + c\cos B = a$，所以原式化为
			\[
				a = 2a\sin A
			\]
			\par\textbf{理由：} 射影定理将左边简化为 $a$.

		\item[\textbf{第二步（整理方程）：}] 因为 $a > 0$，两边同时除以 $a$ 得
			\[
				\begin{aligned}
					1 &= 2\sin A, \\
					\sin A &= \frac12.
			\end{aligned}
			\]
			\par\textbf{理由：} 边长 $a$ 非零，可约去.

		\item[\textbf{第三步（求角 $A$）：}] 由 $\sin A = \frac12$ 且 $A$ 为锐角，得
			\[
				A = 30^\circ
			\]
			\par\textbf{理由：} 锐角正弦值 $\frac12$ 对应 $30^\circ$.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$A = 30^\circ$.

\end{examplebox}
