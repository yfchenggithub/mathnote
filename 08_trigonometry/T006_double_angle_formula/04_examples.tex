	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $\sin x = \frac{3}{5}$，$x$ 为第一象限角，求 $\sin 2x$、$\cos 2x$、$\tan 2x$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求单角余弦）：}] 由平方关系 $\sin^2 x + \cos^2 x = 1$，得
		      \[
			      \begin{aligned}
				      \cos x & = \sqrt{1-\sin^2 x}     \\
				             & = \sqrt{1-\frac{9}{25}} \\
				             & = \sqrt{\frac{16}{25}}  \\
				             & = \frac{4}{5}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 同角平方关系，第一象限余弦取正值。

		\item[\textbf{第二步（正弦二倍角）：}]
		      \[
			      \begin{aligned}
				      \sin 2x & = 2\sin x \cos x                          \\
				              & = 2 \times \frac{3}{5} \times \frac{4}{5} \\
				              & = \frac{24}{25}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 正弦二倍角公式 $\sin 2x = 2\sin x\cos x$。

		\item[\textbf{第三步（余弦二倍角）：}]
		      \[
			      \begin{aligned}
				      \cos 2x & = \cos^2 x - \sin^2 x          \\
				              & = \frac{16}{25} - \frac{9}{25} \\
				              & = \frac{7}{25}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 余弦二倍角公式，也可用 $1-2\sin^2 x$ 验证。

		\item[\textbf{第四步（求正切值）：}]
		      因为 $\cos 2x=\frac{7}{25}\ne0$，所以
		      \[
			      \begin{aligned}
				      \tan 2x & = \frac{\sin 2x}{\cos 2x}            \\
				              & = \frac{\frac{24}{25}}{\frac{7}{25}} \\
				              & = \frac{24}{7}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 商数关系 $\tan 2x=\frac{\sin 2x}{\cos 2x}$，需保证 $\cos 2x\ne0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\sin 2x = \frac{24}{25}$, $\cos 2x = \frac{7}{25}$, $\tan 2x = \frac{24}{7}$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 已知 $\cos 2x = \frac{1}{3}$，求 $\sin^2 x$ 与 $\cos^2 x$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（正弦降幂）：}] 由降幂公式 $\sin^2 x = \frac{1 - \cos 2x}{2}$，代入 $\cos 2x = \frac{1}{3}$，得
		      \[
			      \begin{aligned}
				      \sin^2 x & = \frac{1 - \cos 2x}{2}     \\
				               & = \frac{1 - \frac{1}{3}}{2} \\
				               & = \frac{2/3}{2}             \\
				               & = \frac{1}{3}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 由余弦二倍角变形 $\cos 2x = 1 - 2\sin^2 x$ 得降幂公式。

		\item[\textbf{第二步（余弦降幂）：}]
		      \[
			      \begin{aligned}
				      \cos^2 x & = \frac{1 + \cos 2x}{2}     \\
				               & = \frac{1 + \frac{1}{3}}{2} \\
				               & = \frac{4/3}{2}             \\
				               & = \frac{2}{3}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 由 $\cos 2x = 2\cos^2 x - 1$ 得降幂公式。

		\item[\textbf{第三步（平方验证）：}]
		      \[
			      \begin{aligned}
				      \sin^2 x + \cos^2 x & = \frac{1}{3} + \frac{2}{3} \\
				                          & = 1.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 满足同角基本关系，验证结果正确。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\sin^2 x = \frac{1}{3}$, $\cos^2 x = \frac{2}{3}$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 在
	\[
		\cos x\ne0,\qquad \cos 2x\ne0
	\]
	的条件下，求证
	\[
		\frac{1+\sin 2x}{\cos 2x} = \frac{1+\tan x}{1-\tan x}.
	\]

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分子配方）：}]
		      \[
			      \begin{aligned}
				      1+\sin 2x & = \sin^2 x + \cos^2 x + 2\sin x\cos x \\
				                & = (\sin x + \cos x)^2.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 使用 $\sin^2x+\cos^2x=1$ 与 $\sin2x=2\sin x\cos x$，再配成完全平方。

		\item[\textbf{第二步（分母平方差）：}]
		      \[
			      \begin{aligned}
				      \cos 2x & = \cos^2 x - \sin^2 x                 \\
				              & = (\cos x - \sin x)(\cos x + \sin x).
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 余弦二倍角公式与平方差公式。

		\item[\textbf{第三步（约分化简）：}]
		      由于左边有意义要求 $\cos2x\ne0$，而
		      \[
			      \cos2x=(\cos x-\sin x)(\cos x+\sin x),
		      \]
		      所以
		      \[
			      \cos x+\sin x\ne0,\qquad \cos x-\sin x\ne0.
		      \]
		      因此左边可化为
		      \[
			      \begin{aligned}
				      \frac{1+\sin 2x}{\cos 2x}
				       & = \frac{(\sin x+\cos x)^2}{(\cos x-\sin x)(\cos x+\sin x)} \\
				       & = \frac{\sin x+\cos x}{\cos x-\sin x}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 在公共定义域内，分母不为零，可安全约去公因式 $\sin x+\cos x$。

		\item[\textbf{第四步（弦化切）：}]
		      由 $\cos x\ne0$，右边有意义，且
		      \[
			      \begin{aligned}
				      \frac{1+\tan x}{1-\tan x}
				       & = \frac{1+\frac{\sin x}{\cos x}}{1-\frac{\sin x}{\cos x}}           \\
				       & = \frac{\frac{\cos x+\sin x}{\cos x}}{\frac{\cos x-\sin x}{\cos x}} \\
				       & = \frac{\sin x+\cos x}{\cos x-\sin x}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 将 $\tan x = \frac{\sin x}{\cos x}$ 代入并化简，得到与左边相同的形式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{在公共定义域 $\cos x\ne0,\ \cos2x\ne0$ 内，恒等式成立。}
