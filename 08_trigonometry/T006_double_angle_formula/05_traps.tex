\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（正切定义域忽视）：}}]
		      使用
		      \[
			      \tan 2x=\frac{2\tan x}{1-\tan^2x}
		      \]
		      时，只盯着分母 $1-\tan^2x$，却忘记 $\tan x$ 本身也要有意义。

		\item[\textbf{\textcolor{CTrap}{正确理解（双重限制）：}}]
		      正切二倍角的分式公式要求
		      \[
			      \cos x\ne0,\qquad 1-\tan^2x\ne0.
		      \]
		      其中 $\cos x\ne0$ 保证 $\tan x$ 有意义，$1-\tan^2x\ne0$ 保证右端分母不为零。

		\item[\textbf{\textcolor{CTrap}{错因分析（定义域忽略）：}}]
		      将分式公式当作纯代数等式，忽视了三角函数和分式共同带来的定义域限制。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（正弦倍角误解）：}}]
		      误记 $\sin 2x = 2\sin x$，丢失 $\cos x$ 因子。

		\item[\textbf{\textcolor{CTrap}{正确理解（倍角公式）：}}]
		      正确公式为
		      \[
			      \sin 2x = 2\sin x \cos x,
		      \]
		      不能去掉 $\cos x$。

		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}]
		      简单类比 $2x$ 的系数，忽略了二倍角公式的乘积结构。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（降幂开方缺符号）：}}]
		      由
		      \[
			      \sin^2 x = \frac{1-\cos 2x}{2}
		      \]
		      直接得出
		      \[
			      \sin x = \sqrt{\frac{1-\cos 2x}{2}},
		      \]
		      未考虑正弦符号。

		\item[\textbf{\textcolor{CTrap}{正确理解（平方根符号）：}}]
		      正确应为
		      \[
			      \sin x = \pm \sqrt{\frac{1-\cos 2x}{2}},
		      \]
		      正负号由 $x$ 所在象限或题目给出的范围决定。

		\item[\textbf{\textcolor{CTrap}{错因分析（忽视平方根双重性）：}}]
		      将平方运算与开方运算等同，忘记算术平方根仅表示非负值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（只知一个函数值就求倍角）：}}]
		      仅知道 $\sin x$ 或仅知道 $\cos x$ 时，直接求 $\sin 2x$，容易忽略另一个三角函数的符号。

		\item[\textbf{\textcolor{CTrap}{正确理解（需要象限信息）：}}]
		      因为
		      \[
			      \sin2x=2\sin x\cos x,
		      \]
		      所以若只知道 $\sin x$，还需要确定 $\cos x$ 的符号；若只知道 $\cos x$，还需要确定 $\sin x$ 的符号。

		\item[\textbf{\textcolor{CTrap}{错因分析（符号判断不足）：}}]
		      平方关系只能求出绝对值，不能自动确定正负号，必须结合象限或角的范围判断。
	\end{description}

\end{trapbox}
