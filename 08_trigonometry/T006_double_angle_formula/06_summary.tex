\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{二倍角公式将角度 $x$ 的三角函数与角度 $2x$ 的三角函数联系起来，核心是乘积结构与平方关系的变形。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（角度任意性）：} 正弦、余弦二倍角公式对任意实数 $x$ 成立。
		\item \textbf{条件 2（正切公式定义域）：} 使用
		      \[
			      \tan2x=\frac{2\tan x}{1-\tan^2x}
		      \]
		      时，需保证
		      \[
			      \cos x\ne0,\qquad 1-\tan^2x\ne0.
		      \]
		      其中 $\cos x\ne0$ 保证 $\tan x$ 有意义，$1-\tan^2x\ne0$ 保证右端分母不为零。
		\item \textbf{条件 3（公式选择）：} 余弦二倍角的三种形式可根据已知条件灵活选用，以达到化简目的。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（定义域审查）：} 使用正切二倍角公式前，既要检查 $\tan x$ 是否有意义，也要检查分母 $1-\tan^2x$ 是否为零。
		\item \textbf{检查项（符号确定）：} 进行开方运算时，务必根据角 $x$ 的范围选择正确的正负号。
		\item \textbf{求值提醒（信息是否足够）：} 仅知道 $\sin x$ 或仅知道 $\cos x$ 时，$\sin2x$ 可能不能唯一确定，通常还需要象限或符号信息。
	\end{itemize}

\end{summarybox}
