\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $a=5$，$b=4$，$c=3$，且 $a$ 为角 $A$ 的对边。试判断角 $A$ 的类型，并判断三角形的形状。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定最大边）：}]
			$a=5$ 最大，为角 $A$ 的对边，因此角 $A$ 为最大角。
			\par\textbf{理由：} 大边对大角。
		\item[\textbf{第二步（平方比较）：}]
			计算 $a^2=25$，$b^2+c^2=9+16=25$。
			\par\textbf{理由：} 代入边长计算。
		\item[\textbf{第三步（应用结论）：}]
			因为 $a^2=b^2+c^2$，根据结论，$\angle A=90^\circ$，即角 $A$ 为直角，三角形为直角三角形。
			\par\textbf{理由：} 边角判据直接推出。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\triangle ABC$ 为直角三角形，$\angle A=90^\circ$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$b=3$，$c=4$，角 $A$ 为锐角。求边 $a$ 的取值范围。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用结论）：}]
			由结论，角 $A$ 为锐角等价于
			\[
				\begin{aligned}
					a^2 &< b^2+c^2 \\ &= 3^2+4^2 \\ &= 25,
			\end{aligned}
			\]
			解得 $a<5$。
			\par\textbf{理由：} 由边角判据。
		\item[\textbf{第二步（三角形存在条件）：}]
			两边之和大于第三边，两边之差小于第三边：$|b-c|<a<b+c$，即 $1<a<7$。
			\par\textbf{理由：} 三角形三边关系。
		\item[\textbf{第三步（取交集）：}]
			同时满足 $a<5$ 和 $1<a<7$，得 $a$ 的取值范围为 $1<a<5$。
			\par\textbf{理由：} 综合两个条件。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$a$ 的取值范围是 $1<a<5$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，角 $A,B,C$ 的对边分别为 $a,b,c$，且 $\sin A:\sin B:\sin C=3:4:5$。判断 $\triangle ABC$ 的形状。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（边角转化）：}]
			由正弦定理，$a:b:c=\sin A:\sin B:\sin C=3:4:5$。
			\par\textbf{理由：} 正弦定理得出边与对应角的正弦成正比。
		\item[\textbf{第二步（设比例系数）：}]
			设 $a=3k$，$b=4k$，$c=5k$（$k>0$）。
			\par\textbf{理由：} 引入参数方便计算。
		\item[\textbf{第三步（平方比较）：}]
			计算最大边 $c$ 的平方：$c^2=25k^2$；另两边平方和为
			\[
				\begin{aligned}
					a^2+b^2 &= 9k^2+16k^2 \\ &= 25k^2,
			\end{aligned}
			\]
			故 $c^2=a^2+b^2$。
			\par\textbf{理由：} 代入计算。
		\item[\textbf{第四步（应用推广结论）：}]
			由最大边判据，$\angle C=90^\circ$，三角形为直角三角形。
			\par\textbf{理由：} 边角判据的推广。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	$\triangle ABC$ 是以 $\angle C$ 为直角的直角三角形。
\end{examplebox}
