\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（锐角三角形混淆）}}]
			直接将角 $A$ 为锐角等同于三角形为锐角三角形。
		\item[\textbf{\textcolor{CTrap}{正确理解（最大角判形）：}}]
			三角形为锐角三角形必须三个角都是锐角，等价于最大边（最大角）满足平方小于另两边平方和。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视最大角）：}}]
			没有考虑 $\angle A$ 可能不是最大角，误以为一个锐角就能判定锐角三角形。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（大小关系记反）}}]
			错误记忆边平方关系与角类型对应：认为 $a^2 > b^2+c^2$ 时 $\angle A$ 为锐角。
		\item[\textbf{\textcolor{CTrap}{正确理解（大于则钝）：}}]
			由余弦定理，$a^2>b^2+c^2$ 推出 $\cos A<0$，故 $\angle A>90^\circ$ 为钝角。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号逻辑混乱）：}}]
			未理解余弦定理推导，仅机械记忆，导致不等号方向与角类型匹配错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（边角对应错位）}}]
			将边 $a$ 当作角 $B$ 或角 $C$ 的对边来应用结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（严格对应）：}}]
			结论中 $a$ 必须对应角 $A$，使用时必须明确边与角的对应关系。
		\item[\textbf{\textcolor{CTrap}{错因分析（记号疏忽）：}}]
			在复杂题目中混淆了边与角的字母对应，导致判断失误。
	\end{description}

\end{trapbox}
