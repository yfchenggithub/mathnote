	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（边角对应）：}}]
			在 $\triangle ABC$ 中，记角 $A,B,C$ 的对边分别为 $a,b,c$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（锐直钝判据）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			角 $A$ 为锐角、直角或钝角，完全由 $a^2$ 与 $b^2+c^2$ 的大小关系决定。
			\[
				\angle A<90^\circ \iff a^2<b^2+c^2,\quad \angle A=90^\circ \iff a^2=b^2+c^2,\quad \angle A>90^\circ \iff a^2>b^2+c^2.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（最大边法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若 $c$ 为最大边，则三角形形状由 $\angle C$ 的类型反映，等价于 $c^2$ 与 $a^2+b^2$ 的大小关系。
			\[
				\angle C<90^\circ \iff c^2<a^2+b^2,\quad \angle C=90^\circ \iff c^2=a^2+b^2,\quad \angle C>90^\circ \iff c^2>a^2+b^2.
			\]
	\end{description}

\end{statementbox}
