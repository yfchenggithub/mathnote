\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形边角）：}}] 在三角形 $ABC$ 中，设 $a = BC$, $b = CA$, $c = AB$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两边夹角求边）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 已知两边及其夹角，直接求出第三边。
		      \[
			      \begin{aligned}
				      a^2 & = b^2 + c^2 - 2bc\cos A, \\
				      b^2 & = c^2 + a^2 - 2ca\cos B, \\
				      c^2 & = a^2 + b^2 - 2ab\cos C.
			      \end{aligned}
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（三边求角）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}} 已知三边长，并确认三边能构成三角形后，可求任意一个内角的余弦值。
		      \[
			      \begin{aligned}
				      \cos A & = \frac{b^2 + c^2 - a^2}{2bc}, \\
				      \cos B & = \frac{c^2 + a^2 - b^2}{2ca}, \\
				      \cos C & = \frac{a^2 + b^2 - c^2}{2ab}.
			      \end{aligned}
		      \]
	\end{description}
\end{statementbox}
