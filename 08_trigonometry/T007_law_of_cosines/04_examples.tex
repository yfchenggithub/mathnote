\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}} 在 $\triangle ABC$ 中，已知 $b=5$，$c=7$，$\angle A=60^\circ$，求边长 $a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 识别已知量为两边 $b,c$ 及其夹角 $A$，应使用公式 $a^2 = b^2 + c^2 - 2bc\cos A$。
		      \par\textbf{理由：} 余弦定理直接给出已知两边及夹角求第三边的表达式。
		\item[\textbf{第二步（代入计算）：}] 代入数值：
		      \[
			      a^2 = 5^2 + 7^2 - 2 \times 5 \times 7 \times \cos 60^\circ = 25 + 49 - 70 \times 0.5 = 74 - 35 = 39.
		      \]
		      \par\textbf{理由：} 运算需注意 $\cos 60^\circ = \frac12$，乘法运算顺序。
		\item[\textbf{第三步（结果取正）：}] 开平方得 $a = \sqrt{39}$（边长取正）。
		      \par\textbf{理由：} 边长恒为正，所以只取算术平方根。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$a = \sqrt{39}$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}} 在 $\triangle ABC$ 中，$a=7$，$b=8$，$c=9$，求最大角的余弦值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（找最大角）：}] 因为 $c$ 最大，所以对角 $C$ 最大。利用余弦定理变形求 $\cos C$。
		      \par\textbf{理由：} 大边对大角，最大边的对角为最大角。
		\item[\textbf{第二步（套用公式）：}] 使用 $\cos C = \dfrac{a^2 + b^2 - c^2}{2ab}$。
		      \par\textbf{理由：} 余弦定理的角形式。
		\item[\textbf{第三步（代入计算）：}] 代入数值：
		      \[
			      \cos C = \frac{7^2 + 8^2 - 9^2}{2 \times 7 \times 8} = \frac{49 + 64 - 81}{112} = \frac{32}{112} = \frac{2}{7}.
		      \]
		      \par\textbf{理由：} 正确计算平方和差，并约分。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$\cos C = \frac{2}{7}$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}} 在 $\triangle ABC$ 中，已知 $a=3$，$b=4$，且 $a^2 + b^2 < c^2$，$c$ 为正整数，求 $c$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（三边条件）：}] 由三角形三边关系可得
		      \[
			      |a-b|<c<a+b.
		      \]
		      代入 $a=3,b=4$，得
		      \[
			      1<c<7.
		      \]
		      \par\textbf{理由：} 题目中已经说明是在三角形中，因此必须先满足三角形存在性条件。

		\item[\textbf{第二步（钝角条件）：}] 由
		      \[
			      a^2+b^2<c^2
		      \]
		      得
		      \[
			      3^2+4^2<c^2,
		      \]
		      即
		      \[
			      25<c^2.
		      \]
		      因为 $c>0$，所以
		      \[
			      c>5.
		      \]
		      \par\textbf{理由：} 边长为正数，由 $c^2>25$ 可推出 $c>5$。

		\item[\textbf{第三步（综合求整数）：}] 结合
		      \[
			      5<c<7
		      \]
		      且 $c$ 为正整数，得
		      \[
			      c=6.
		      \]
		      \par\textbf{理由：} 满足条件的正整数唯一。

		\item[\textbf{第四步（角度含义）：}] 由于
		      \[
			      a^2+b^2<c^2,
		      \]
		      等价于
		      \[
			      \cos C=\frac{a^2+b^2-c^2}{2ab}<0,
		      \]
		      所以角 $C$ 为钝角。
		      \par\textbf{理由：} 余弦定理可以把边长平方关系转化为角的锐、直、钝判断。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}} \textbf{$c=6$}。
\end{examplebox}
