\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$a=3$, $b=4$, $c=5$, 外接圆半径$R=2.5$, 求三角形的面积$S$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别已知量）：}] 题目给出了三条边长$a,b,c$和外接圆半径$R$，满足公式$S = \dfrac{abc}{4R}$的使用条件。
			\par\textbf{理由：} 公式适用于任意三角形，已知所有所需变量可直接代入。
		\item[\textbf{第二步（代入公式）：}] 将$a=3$, $b=4$, $c=5$, $R=2.5$代入面积公式：
			\[
				S = \frac{3\times4\times5}{4\times2.5}.
			\]
			\par\textbf{理由：} 直接套用二级结论。
		\item[\textbf{第三步（计算结果）：}] 计算分子分母：
			\[
				\begin{aligned}
					S &= \frac{60}{10} \\
					&= 6.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术运算即得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{三角形的面积 $S=6$}，与直角边求面积结果一致。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$a=3$, $b=4$, $S=6$, $R=2.5$, 求边长$c$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（逆用公式）：}] 由原公式变形得$c = \dfrac{4RS}{ab}$.
			\par\textbf{理由：} 将$S=abc/(4R)$两边同乘$4R$并除以$ab$即得。
		\item[\textbf{第二步（代入已知量）：}] 将$a=3$, $b=4$, $S=6$, $R=2.5$代入：
			\[
				c = \frac{4\times2.5\times6}{3\times4}.
			\]
			\par\textbf{理由：} 代入数值。
		\item[\textbf{第三步（计算求值）：}] 计算分子分母：
			\[
				\begin{aligned}
					c &= \frac{60}{12} \\
					&= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 算术运算即得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{第三边长 $c=5$}，符合勾股定理$a^2+b^2=c^2$，验证了计算的正确性。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，已知$a=4$, $b=5$, $C=60^\circ$，求外接圆半径$R$.

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（余弦求边）：}] 由余弦定理，先求第三边$c$：
			\[
				\begin{aligned}
					c^2 &= a^2+b^2-2ab\cos C \\
					&= 4^2+5^2-2\times4\times5\times\cos60^\circ \\
					&= 16+25-40\times\frac12 \\
					&= 41-20 \\
					&= 21,
			\end{aligned}
			\]
			所以$c = \sqrt{21}$.
			\par\textbf{理由：} 已知两边及其夹角，用余弦定理可求第三边。
		\item[\textbf{第二步（面积公式）：}] 利用两边及其夹角求面积：
			\[
				\begin{aligned}
					S &= \frac12 ab\sin C \\
					&= \frac12\times4\times5\times\frac{\sqrt3}{2} \\
					&= 10\times\frac{\sqrt3}{2} \\
					&= 5\sqrt3.
			\end{aligned}
			\]
			\par\textbf{理由：} 面积公式$S=\frac12 ab\sin C$是基础公式。
		\item[\textbf{第三步（半径求解）：}] 由二级结论$R = \dfrac{abc}{4S}$，代入$a,b,c,S$：
			\[
				\begin{aligned}
					R &= \frac{4\times5\times\sqrt{21}}{4\times5\sqrt3} \\
					&= \frac{20\sqrt{21}}{20\sqrt3} \\
					&= \frac{\sqrt{21}}{\sqrt3} \\
					&= \sqrt7.
			\end{aligned}
			\]
			\par\textbf{理由：} 逆用结论$R=abc/(4S)$，约分化简即得。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{外接圆半径 $R=\sqrt7$}，通过三角形边角关系推导得出。
\end{examplebox}
