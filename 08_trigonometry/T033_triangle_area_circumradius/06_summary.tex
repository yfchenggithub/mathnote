\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{任意三角形中，面积等于三边之积除以外接圆半径的$4$倍，即$S=\dfrac{abc}{4R}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（已知三边半径）：} 已知三边长$a,b,c$和外接圆半径$R$时，可直接代入求面积。
		\item \textbf{条件 2（间接求得）：} 通过正弦定理、余弦定理等先解出$R$或边长，再使用公式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分半径种类）：} 公式中$R$必为外接圆半径，不是内切圆半径$r$。
		\item \textbf{检查项（逆用公式）：} 逆求半径时公式为$R=\dfrac{abc}{4S}$，切勿遗漏分母$4$。
	\end{itemize}
\end{summarybox}
