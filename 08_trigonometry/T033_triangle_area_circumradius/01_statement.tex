\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三边半径）：}}] 三角形$ABC$的三边长分别为$a,b,c$，外接圆半径为$R$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积等于三边乘积除以四倍外接圆半径。
			\[
				S = \frac{abc}{4R}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（逆用求半径）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			外接圆半径等于三边乘积除以四倍面积。
			\[
				R = \frac{abc}{4S}.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{适用范围：}}
	任意三角形，不论直角、锐角、钝角，公式均成立。
\end{statementbox}
