\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（局限直角）：}}] 误认为$a=2R\sin A$只适用于直角三角形。
		\item[\textbf{\textcolor{CTrap}{正确理解（普遍适用）：}}] 正弦定理适用于任意三角形，该变形对任意三角形都成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}] 受直角三角形锐角三角比定义影响，忽略了一般三角形外接圆构造。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（半径混淆）：}}] 将外接圆半径$R$与内切圆半径$r$混淆，代入错误数值。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分定义）：}}] 公式中的$R$必为外接圆半径，由$2R=\dfrac{a}{\sin A}$唯一确定。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）：}}] 没有清晰区分外接圆和内切圆的定义及性质。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（变形错误）：}}] 由$a=2R\sin A$错误变形为$R=2a\sin A$或$\sin A=2Ra$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确变形）：}}] 正确变形为$R=\dfrac{a}{2\sin A}$，$\sin A=\dfrac{a}{2R}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（运算失误）：}}] 移项时混淆乘法与除法，未正确处理系数$2$。
	\end{description}
\end{trapbox}
