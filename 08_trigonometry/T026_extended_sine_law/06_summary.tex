\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{正弦定理的边角互化形式 $a=2R\sin A$ 等，可直接在边长与对角正弦之间转换，外接圆半径 $R$ 是关键桥梁。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（适用所有三角形）：} 任意三角形均有外接圆，公式普遍适用。
		\item \textbf{条件2（已知或可求R）：} 解题时需要知道或能计算出外接圆半径 $R$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（变形正确性）：} 变形时注意 $R=\dfrac{a}{2\sin A}$，勿忘分母中的 $2$。
		\item \textbf{检查项（对应关系）：} 使用公式时务必保证边与对角对应，如 $a$ 对应 $\sin A$。
	\end{itemize}
\end{summarybox}
