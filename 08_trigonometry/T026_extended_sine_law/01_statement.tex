\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（任意三角形）：}}] 在$\triangle ABC$中，设外接圆半径为$R$，$a,b,c$分别为角$A,B,C$的对边。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（边角关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			边长等于外接圆直径乘对角正弦。
			\[
				a = 2R\sin A,\quad b = 2R\sin B,\quad c = 2R\sin C.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（比例形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			各边与其对角正弦之比相等，等于外接圆直径。
			\[
				\begin{aligned}
					\frac{a}{\sin A} &= \frac{b}{\sin B} \\
					&= \frac{c}{\sin C} \\
					&= 2R.
			\end{aligned}
			\]
	\end{description}
\end{statementbox}
