\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，已知外接圆半径$R=4$，$A=30^\circ$，求边长$a$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写公式）：}] 由正弦定理变形$a=2R\sin A$。
			\par\textbf{理由：} 直接应用结论。
		\item[\textbf{第二步（代数值）：}] 代入$R=4$，$\sin30^\circ=\dfrac12$，得$a=2\times4\times\dfrac12$。
			\par\textbf{理由：} 数据代入。
		\item[\textbf{第三步（计算）：}] 计算得$a=4$。
			\par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a=4$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$a=6$，$A=60^\circ$，求外接圆半径$R$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（变形公式）：}] 由$a=2R\sin A$变形得$R=\dfrac{a}{2\sin A}$。
			\par\textbf{理由：} 等式的性质。
		\item[\textbf{第二步（代数值）：}] 代入$a=6$，$\sin60^\circ=\dfrac{\sqrt3}{2}$，得
			\[
				\begin{aligned}
					R &= \frac{6}{2\times\frac{\sqrt3}{2}} \\
					&= \frac{6}{\sqrt3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 数值代入。
		\item[\textbf{第三步（化简）：}] 化简得$R=2\sqrt3$。
			\par\textbf{理由：} 有理化分母。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=2\sqrt3$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$a=2\sqrt3$，$b=3$，$\sin A=\dfrac{\sqrt3}{2}$，求$R$和$\sin B$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求R）：}] 由$a=2R\sin A$，得
			\[
				\begin{aligned}
					R &= \frac{a}{2\sin A} \\
					&= \frac{2\sqrt3}{2\times\frac{\sqrt3}{2}} \\
					&= 2.
			\end{aligned}
			\]
			\par\textbf{理由：} 变形代入。
		\item[\textbf{第二步（求sinB）：}] 由$b=2R\sin B$，得
			\[
				\begin{aligned}
					\sin B &= \frac{b}{2R} \\
					&= \frac{3}{2\times2} \\
					&= \frac{3}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 变形代入。
		\item[\textbf{第三步（检验）：}] $\sin B=\dfrac34<1$，故结果合理。
			\par\textbf{理由：} 三角形内角正弦值在$(0,1]$内。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$R=2$，$\sin B=\dfrac34$}
\end{examplebox}
