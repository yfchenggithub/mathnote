	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（参数定义）：}}] $n$ 为非负整数，$x = \cos\theta$（$\theta\in\mathbb{R}$）。
		\item[\textbf{\textcolor{CStatement}{条件2（递推定义）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			由前两项递推生成所有多项式。
			\[
				\begin{aligned}
					T_0(x) &= 1, \\
					T_1(x) &= x, \\
					T_{n+1}(x) &= 2x\,T_n(x) - T_{n-1}(x), \quad n\ge 1.
			\end{aligned}
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论（三角恒等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			余弦倍角公式的高次推广。
			\[
				T_n(\cos\theta) = \cos(n\theta)
			\]
	\end{description}
