\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$\cos\theta = \frac12$，利用第一类切比雪夫多项式求$\cos3\theta$的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定多项式）：}] 由切比雪夫多项式的结论，$\cos3\theta = T_3(\cos\theta)$，其中$T_3(x)=4x^3-3x$。
			\par\textbf{理由：} 当$n=3$时$T_3(\cos\theta)=\cos3\theta$，且$T_3(x)=4x^3-3x$是已知表达式。
		\item[\textbf{第二步（代入数值）：}] 将$x=\frac12$代入$T_3$：
			\[
				T_3\Bigl(\frac12\Bigr)=4\times\Bigl(\frac12\Bigr)^3-3\times\frac12
			\]
			\par\textbf{理由：} 直接代换。
		\item[\textbf{第三步（结果计算）：}] 计算上式：
			\[
				T_3\Bigl(\frac12\Bigr)=4\times\frac18-\frac32=\frac12-\frac32=-1
			\]
			\par\textbf{理由：} 分数运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\cos3\theta=-1$。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$T_2(x)=2x^2-1$，$T_3(x)=4x^3-3x$，利用递推$T_{n+1}(x)=2xT_n(x)-T_{n-1}(x)$求$T_4(x)$，并计算$T_4\bigl(\frac{\sqrt2}{2}\bigr)$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用递推）：}] 取$n=3$，得$T_4(x)=2xT_3(x)-T_2(x)$。
			\par\textbf{理由：} 递推公式中令$n=3$。
		\item[\textbf{第二步（展开化简）：}] 代入$T_2,T_3$并整理：
			\[
				\begin{aligned}
					T_4(x) &= 2x(4x^3-3x)-(2x^2-1) \\
					&= 8x^4-6x^2-2x^2+1 \\
					&= 8x^4-8x^2+1
			\end{aligned}
			\]
			\par\textbf{理由：} 展开括号，合并同类项。
		\item[\textbf{第三步（计算特殊值）：}] 将$x=\frac{\sqrt2}{2}$代入：
			\[
				T_4\Bigl(\frac{\sqrt2}{2}\Bigr)=8\Bigl(\frac{\sqrt2}{2}\Bigr)^4-8\Bigl(\frac{\sqrt2}{2}\Bigr)^2+1
				=8\times\frac14-8\times\frac12+1=2-4+1=-1
			\]
			\par\textbf{理由：} 幂运算：$(\frac{\sqrt2}{2})^2=\frac12$，$(\frac{\sqrt2}{2})^4=(\frac12)^2=\frac14$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$T_4(x)=8x^4-8x^2+1$，$T_4\bigl(\frac{\sqrt2}{2}\bigr)=-1$。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	利用递推$T_{n+1}(x)=2xT_n(x)-T_{n-1}(x)$，求$T_5(x)$，并写出$\cos5\theta$的切比雪夫多项式表示。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（引用已知项）：}] 由前例知$T_3(x)=4x^3-3x$，$T_4(x)=8x^4-8x^2+1$。
			\par\textbf{理由：} 已有结论或计算结果。
		\item[\textbf{第二步（递推求五代简）：}] 由递推$n=4$得$T_5(x)=2xT_4(x)-T_3(x)$，代入并化简：
			\[
				\begin{aligned}
					T_5(x) &= 2x(8x^4-8x^2+1)-(4x^3-3x) \\
					&= 16x^5-16x^3+2x-4x^3+3x \\
					&= 16x^5-20x^3+5x
			\end{aligned}
			\]
			\par\textbf{理由：} 递推公式，展开，合并同类项。
		\item[\textbf{第三步（三角表示）：}] 由恒等式$T_n(\cos\theta)=\cos(n\theta)$，取$n=5$得
			\[
				\cos5\theta = T_5(\cos\theta) = 16\cos^5\theta -20\cos^3\theta +5\cos\theta
			\]
			\par\textbf{理由：} 代入$x=\cos\theta$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$T_5(x)=16x^5-20x^3+5x$，且$\cos5\theta = 16\cos^5\theta -20\cos^3\theta +5\cos\theta$。
\end{examplebox}
