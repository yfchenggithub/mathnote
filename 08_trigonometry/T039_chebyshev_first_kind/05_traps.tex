\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（定义域混淆）}}] 误认为切比雪夫多项式$T_n(x)$只在$x\in[-1,1]$上有定义。
		\item[\textbf{\textcolor{CTrap}{正确理解（多项式定义）：}}] $T_n(x)$是多项式，对任意实数$x$都有定义。但是恒等式$T_n(\cos\theta)=\cos(n\theta)$要求$x=\cos\theta$，即$x\in[-1,1]$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 将多项式的定义域与三角恒等式的适用条件混为一谈。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（递推序号错误）}}] 在使用递推公式时，错误地写成$T_{n+1}(x)=2xT_{n-1}(x)-T_n(x)$或其他变体。
		\item[\textbf{\textcolor{CTrap}{正确理解（递推关系）：}}] 正确的递推是$T_{n+1}(x)=2xT_n(x)-T_{n-1}(x)$（$n\ge1$）。可类比$\cos[(n+1)\theta]=2\cos\theta\cos(n\theta)-\cos[(n-1)\theta]$。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆不牢）：}}] 对三角恒等变换的和差化积公式不熟悉，导致递推关系记混。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（特殊值误判）}}] 求$T_n(1)$时，错误地认为$T_n(1)=n$或$T_n(1)=(-1)^n$等。
		\item[\textbf{\textcolor{CTrap}{正确理解（恒等式代换）：}}] 由$T_n(\cos\theta)=\cos(n\theta)$，令$\theta=0$得$T_n(1)=\cos0=1$，对所有$n\ge0$成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（直觉误导）：}}] 递推公式复杂时容易猜想特殊值，未利用三角本质简化。
	\end{description}
\end{trapbox}
