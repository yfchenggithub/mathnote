\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AD$ 平分 $\angle BAC$ 交 $BC$ 于 $D$。已知 $AB=6$，$AC=4$，$BD=3$，求 $DC$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用定理）：}]
			\[
				\frac{BD}{DC} = \frac{AB}{AC}.
			\]
			\par\textbf{理由：} 因为 $AD$ 平分 $\angle BAC$，可直接使用内角平分线定理。

		\item[\textbf{第二步（代入数据）：}]
			\[
				\frac{3}{DC} = \frac{6}{4}.
			\]
			\par\textbf{理由：} 将已知数值代入比例式。

		\item[\textbf{第三步（解比例）：}]
			\[
				\begin{aligned}
					\frac{3}{DC} &= \frac{3}{2} \\
					\Rightarrow\quad DC &= 2.
			\end{aligned}
			\]
			\par\textbf{理由：} 化简比例求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$DC=2$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$AD$ 平分 $\angle BAC$ 交 $BC$ 于 $D$。已知 $AB=8$，$AC=5$，$BC=7$，求 $BD$ 和 $DC$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设未知量）：}] 设 $BD=x$，则 $DC=BC-x=7-x$。
			\par\textbf{理由：} 用未知数表示待求线段，便于列方程。

		\item[\textbf{第二步（列比例式）：}]
			\[
				\frac{x}{7-x} = \frac{8}{5}.
			\]
			\par\textbf{理由：} 代入内角平分线定理。

		\item[\textbf{第三步（解方程）：}]
			\[
				\begin{aligned}
					\frac{x}{7-x} &= \frac{8}{5} \\
					5x &= 56 - 8x \\
					13x &= 56 \\
					x &= \frac{56}{13}.
			\end{aligned}
			\]
			\par\textbf{理由：} 交叉相乘解出 $x$，进而 $DC = 7 - x = \frac{35}{13}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BD = \dfrac{56}{13}$，$DC = \dfrac{35}{13}$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$\angle C = 90^\circ$，$AD$ 平分 $\angle BAC$ 交 $BC$ 于 $D$。已知 $AB=10$，$AC=6$，求 $BD$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（勾股求边）：}] 由勾股定理，
			\[
				\begin{aligned}
					BC &= \sqrt{AB^2 - AC^2} \\
					&= \sqrt{10^2 - 6^2} \\
					&= 8.
			\end{aligned}
			\]
			\par\textbf{理由：} 直角三角形已知斜边和一直角边，可求另一直角边。

		\item[\textbf{第二步（设未知量）：}] 设 $BD=x$，则 $DC=BC-x=8-x$。
			\par\textbf{理由：} 同例2。

		\item[\textbf{第三步（比例求解）：}] 由内角平分线定理得 $\frac{x}{8-x} = \frac{10}{6} = \frac{5}{3}$，解得
			\[
				\begin{aligned}
					\frac{x}{8-x} &= \frac{5}{3} \\
					3x &= 40 - 5x \\
					8x &= 40 \\
					x &= 5.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入定理并解方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$BD=5$}。
\end{examplebox}
