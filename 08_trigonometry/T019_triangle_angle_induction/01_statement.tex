\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（内角和条件：）}}] 在三角形 $\triangle ABC$ 中，内角满足 $A + B + C = \pi$。
		\item[\textbf{\textcolor{CStatement}{条件 2（正切定义域）：}}] 当使用正切公式时，需确保 $C \neq \frac{\pi}{2}$（即 $C$ 不是直角），否则 $\tan C$ 无定义。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{结论一（诱导恒等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两角和的正弦等于对角的正弦，余弦符号相反，正切符号相反。
			\[
				\sin(A+B) = \sin C
			\]
			\[
				\cos(A+B) = -\cos C
			\]
			\[
				\tan(A+B) = -\tan C \quad (C \neq \tfrac{\pi}{2})
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（半角关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两角和之半的正弦与余弦等于对角之半的互余函数。
			\[
				\sin\frac{A+B}{2} = \cos\frac{C}{2}
			\]
			\[
				\cos\frac{A+B}{2} = \sin\frac{C}{2}
			\]
	\end{description}
\end{statementbox}
