\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（负号遗漏）}}] 计算 $\cos(A+B)$ 时，误以为是 $\cos C$，丢掉负号。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式记忆）：}}] $\cos(A+B) = -\cos C$，不论 $C$ 为锐角、钝角，必须带负号。
		\item[\textbf{\textcolor{CTrap}{错因分析（诱导混淆）：}}] 诱导公式 $\cos(\pi - \alpha) = -\cos\alpha$，学生容易与 $\cos(\pi - \alpha) = \cos\alpha$ 记错，导致符号错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（条件缺漏）}}] 直接使用 $\tan(A+B)=-\tan C$，没有检查 $C$ 是否为直角。
		\item[\textbf{\textcolor{CTrap}{正确理解（使用前提）：}}] 正切公式仅在 $C \neq \frac{\pi}{2}$ 时成立；若 $C = \frac{\pi}{2}$，应改用正余弦形式 $\sin(A+B)=1,\ \cos(A+B)=0$。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义疏忽）：}}] 正切函数在 $\frac{\pi}{2}$ 处无定义，必须先验证 $C$ 是否等于 $\frac{\pi}{2}$，否则后续计算无意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（函数混淆）}}] 使用半角关系时，将 $\sin\frac{A+B}{2}$ 写为 $\sin\frac{C}{2}$，或将 $\cos\frac{A+B}{2}$ 写为 $\cos\frac{C}{2}$，混淆了互余关系。
		\item[\textbf{\textcolor{CTrap}{正确理解（互补互余）：}}] $\sin\frac{A+B}{2} = \cos\frac{C}{2}$，$\cos\frac{A+B}{2} = \sin\frac{C}{2}$，注意函数名互换。
		\item[\textbf{\textcolor{CTrap}{错因分析（诱导遗忘）：}}] 遗忘诱导公式 $\sin(\frac{\pi}{2} - x) = \cos x$，$\cos(\frac{\pi}{2} - x) = \sin x$ 导致函数写错。
	\end{description}
\end{trapbox}
