\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$\angle C = 90^\circ$，求 $\sin(A+B)$ 和 $\cos(A+B)$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 由三角形内角和，$A+B+C=\pi$，已知 $C=90^\circ$（即 $C=\frac{\pi}{2}$）。
			\par\textbf{理由：} 题目直接给出 $C$ 的角度。
		\item[\textbf{第二步（应用正弦公式）：}]
			\[
				\begin{aligned}
					\sin(A+B) &= \sin C \\
					&= \sin\frac{\pi}{2} \\
					&= 1
			\end{aligned}
			\]
			\par\textbf{理由：} 三角形诱导恒等 $\sin(A+B)=\sin C$，且 $\sin\frac{\pi}{2}=1$。
		\item[\textbf{第三步（应用余弦公式）：}]
			\[
				\begin{aligned}
					\cos(A+B) &= -\cos C \\
					&= -\cos\frac{\pi}{2} \\
					&= 0
			\end{aligned}
			\]
			\par\textbf{理由：} 三角形诱导恒等 $\cos(A+B)=-\cos C$，且 $\cos\frac{\pi}{2}=0$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin(A+B)=1$，$\cos(A+B)=0$}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$\cos C = -\frac12$，且 $C$ 为钝角，求 $\tan(A+B)$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求 $\tan C$）：}] 由 $\cos C = -\frac12$ 且 $C$ 为钝角，得 $C = 120^\circ$，即 $C = \frac{2\pi}{3}$。进而
			\[
				\begin{aligned}
					\tan C &= \tan\frac{2\pi}{3} \\
					&= -\sqrt{3}
			\end{aligned}
			\]
			\par\textbf{理由：} 特殊角的三角函数值。
		\item[\textbf{第二步（应用正切公式）：}]
			\[
				\begin{aligned}
					\tan(A+B) &= -\tan C \\
					&= -(-\sqrt{3}) \\
					&= \sqrt{3}
			\end{aligned}
			\]
			\par\textbf{理由：} 三角形诱导恒等 $\tan(A+B)=-\tan C$（$C\neq\frac{\pi}{2}$ 成立）。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\tan(A+B) = \sqrt{3}$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$C=60^\circ$，求 $\tan(A+B)$ 和 $\sin\frac{A+B}{2}$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求 $\tan(A+B)$）：}]
			\[
				\begin{aligned}
					\tan(A+B) &= -\tan C \\
					&= -\tan60^\circ \\
					&= -\sqrt{3}
			\end{aligned}
			\]
			\par\textbf{理由：} 三角形诱导恒等 $\tan(A+B)=-\tan C$。
		\item[\textbf{第二步（求 $\sin\frac{A+B}{2}$）：}]
			\[
				\begin{aligned}
					\sin\frac{A+B}{2} &= \cos\frac{C}{2} \\
					&= \cos30^\circ \\
					&= \frac{\sqrt{3}}{2}
			\end{aligned}
			\]
			\par\textbf{理由：} 半角关系 $\sin\frac{A+B}{2}=\cos\frac{C}{2}$。
		\item[\textbf{第三步（汇总结果）：}] $\tan(A+B)=-\sqrt{3}$，$\sin\frac{A+B}{2}=\frac{\sqrt{3}}{2}$。
			\par\textbf{理由：} 分别代入上述结果。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\tan(A+B)=-\sqrt{3}$，$\sin\frac{A+B}{2}=\frac{\sqrt{3}}{2}$}
\end{examplebox}
