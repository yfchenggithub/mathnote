\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{在三角形中，利用内角和 $A+B = \pi - C$，可将含 $A+B$ 的三角函数直接转化为 $C$ 的诱导形式，且半角 $\frac{A+B}{2}$ 与 $\frac{C}{2}$ 互余。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（内角和固定）：} 必须在三角形内角和为 $\pi$ 的前提下使用。
		\item \textbf{条件 2（正切定义域）：} 应用 $\tan(A+B)=-\tan C$ 时，需确保 $C \neq \frac{\pi}{2}$，否则正切无定义。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（符号与条件）：} $\cos(A+B)$ 结果为 $-\cos C$，不要丢掉负号；正切公式仅当 $C \neq \frac{\pi}{2}$ 成立。
		\item \textbf{检查项（半角互换）：} $\sin\frac{A+B}{2} = \cos\frac{C}{2}$，$\cos\frac{A+B}{2} = \sin\frac{C}{2}$，注意函数名互换。
	\end{itemize}
\end{summarybox}
