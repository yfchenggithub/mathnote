	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（周长半周长混淆）}}]
			计算面积时误将周长当作半周长代入公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分定义）：}}]
			半周长 $p$ 是周长的一半，公式 $S = pr$ 中的 $p$ 必须是半周长。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆偏差）：}}]
			记错公式或漏除以 $2$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（内切圆与外接圆混淆）}}]
			误将外接圆半径 $R$ 代入内切圆半径公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（概念区分）：}}]
			内切圆与三边相切，外接圆过三个顶点；$r$ 是内切圆半径，$R$ 是外接圆半径，不可混用。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念不清）：}}]
			对圆心位置和半径含义掌握不牢。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（适用范围错误）}}]
			认为该公式只适用于直角三角形或等腰三角形。
		\item[\textbf{\textcolor{CTrap}{正确理解（一般形式）：}}]
			该公式适用于任意三角形，只要存在内切圆（所有三角形都有内切圆）。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
			因部分题目中出现直角三角形较多而误以为公式有限制。
	\end{description}

\end{trapbox}
