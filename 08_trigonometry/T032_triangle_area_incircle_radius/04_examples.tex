\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}已知三角形三边长分别为 $a=6$，$b=8$，$c=10$，内切圆半径 $r=2$，求该三角形的面积。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算半周长）：}]
			半周长 $p = \frac{6+8+10}{2} = 12$。
			\par\textbf{理由：}半周长公式 $p = (a+b+c)/2$。
		\item[\textbf{第二步（代入公式）：}]
			由 $S = pr$ 得 $S = 12 \times 2 = 24$。
			\par\textbf{理由：}已知三角形面积等于半周长乘内切圆半径。
		\item[\textbf{第三步（得出结论）：}]
			该三角形面积为 $24$。
			\par\textbf{理由：}计算完毕。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{直接代入 $S=pr$ 即可，步骤简洁。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}已知 $\triangle ABC$ 的面积 $S = 24$，周长 $2p = 24$，求其内切圆半径 $r$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求半周长）：}]
			由周长 $2p = 24$ 得 $p = 12$。
			\par\textbf{理由：}半周长是周长的一半。
		\item[\textbf{第二步（应用半径公式）：}]
			由 $r = S/p$ 得 $r = 24/12 = 2$。
			\par\textbf{理由：}面积公式变形 $r = S/p$。
		\item[\textbf{第三步（得出结果）：}]
			内切圆半径 $r = 2$。
			\par\textbf{理由：}计算完成。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{已知面积和周长可反求半径。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}在 $\triangle ABC$ 中，$a=5$，$b=12$，$c=13$，求该三角形的内切圆半径 $r$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判断三角形类型）：}]
			因为 $5^2 + 12^2 = 13^2$，所以 $\angle C = 90^\circ$，$c$ 为斜边。
			\par\textbf{理由：}勾股定理的逆定理。
		\item[\textbf{第二步（计算三角形面积）：}]
			$S = \frac{1}{2} \times 5 \times 12 = 30$。
			\par\textbf{理由：}直角三角形面积等于两直角边乘积的一半。
		\item[\textbf{第三步（求内切圆半径）：}]
			半周长 $p = \frac{5+12+13}{2} = 15$，则 $r = \frac{S}{p} = \frac{30}{15} = 2$。
			\par\textbf{理由：}内切圆半径公式 $r = S/p$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{结合勾股定理与 $r=S/p$ 可快速求解。}
\end{examplebox}
