	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三边与半周长）：}}]
			设三角形 $ABC$ 的三边长分别为 $a,b,c$，半周长 $p=\frac{a+b+c}{2}$，内切圆半径为 $r$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（面积公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形面积等于半周长与内切圆半径的乘积。
			\[
				S = \frac{1}{2}(a+b+c)r = pr.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（半径公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			用面积和半周长表示内切圆半径。
			\[
				r = \frac{S}{p} = \frac{2S}{a+b+c}.
			\]
	\end{description}

\end{statementbox}
