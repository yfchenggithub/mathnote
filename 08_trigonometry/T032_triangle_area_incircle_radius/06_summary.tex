\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{三角形面积 $S$ 等于半周长 $p$ 与内切圆半径 $r$ 的乘积：$S = pr$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（已知半径）：}内切圆半径 $r$ 已知或可求。
		\item \textbf{条件 2（已知边长）：}三边长 $a,b,c$ 已知或可求半周长 $p$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（半周长定义）：}公式中的 $p$ 是半周长，不是周长。
		\item \textbf{检查项（半径类型）：}确认使用的是内切圆半径，而非外接圆半径。
	\end{itemize}

\end{summarybox}
