\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{三角形已知两边及一边对角时，解的存在性由 $A$ 是否为钝角及 $a$ 与 $h$、$a$ 与 $b$ 的大小比较唯一确定。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（SSA 已知）：} 三角形中已知两边 $a,b$ 及其中一边 $a$ 的对角 $A$。
		\item \textbf{条件 2（高线定义）：} $h = b\sin A$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（先分后比）：} 必须先判断 $A\geq90^\circ$ 还是 $A<90^\circ$，再比较对应的大小关系；锐角时先比 $a$ 与 $h$ 再比 $a$ 与 $b$。
		\item \textbf{检查项（验算边界）：} 当 $a=h$ 或 $a=b$ 时属于边界情形，解答唯一；两解仅出现在锐角且 $h<a<b$ 时。
	\end{itemize}

\end{summarybox}
