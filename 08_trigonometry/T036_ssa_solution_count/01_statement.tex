\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（已知两边及对角）：}}]
			在 $\triangle ABC$ 中，已知边长 $a,b$ 及 $a$ 的对角 $A$，记 $h = b\sin A$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（解的个数判据）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			根据 $A$ 是否为钝角以及 $a,b,h$ 的大小关系，按以下规则判断解数。
			\begin{itemize}
			\item 若 $A \geq 90^\circ$：当 $a \leq b$ 时无解；当 $a > b$ 时有一解。
			\item 若 $A < 90^\circ$：当 $a < h$ 时无解；当 $a = h$ 时有一解；当 $h < a < b$ 时有两解；当 $a \geq b$ 时有一解。
		\end{itemize}
			其中 $h = b\sin A$ 是三角形 $AB$ 边上的高。
		\item[\textbf{\textcolor{CStatement}{推广结论（几何作图法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $h$ 视为点 $C$ 到 $AB$ 的距离，解的个数等价于以 $C$ 为圆心、$a$ 为半径的圆与射线 $AB$ 的交点个数。
			\begin{itemize}
			\item 当 $A \geq 90^\circ$ 时，圆与射线最多一个交点，仅当 $a>b$ 时有解。
			\item 当 $A < 90^\circ$ 时，交点个数为 $0,1,2$ 分别对应 $a<h,\ a=h,\ h<a<b$；当 $a \geq b$ 时也有一个交点。
		\end{itemize}

	\end{description}

\end{statementbox}
