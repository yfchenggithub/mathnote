\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略角类别）}}]
			不区分类别，直接套用“$a<b$ 两解”等结论。例如 $A=120^\circ,\ a=6,\ b=5$ 时，锐角判据给出 $a>b$ 一解，虽然结果也对，但过程错误；若 $A=120^\circ,\ a=5,\ b=6$，锐角判据会误判为两解，实际无解。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论）：}}]
			必须按 $A\geq90^\circ$ 和 $A<90^\circ$ 分别选择判据。$A\geq90^\circ$ 时只看 $a$ 与 $b$；$A<90^\circ$ 时先看 $a$ 与 $h$，再看 $a$ 与 $b$。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）：}}]
			学生习惯于锐角三角形，忽略了钝角三角形中高的位置变化及正弦函数的不同表现。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽视高 h）}}]
			在 $A<90^\circ$ 时，仅比较 $a$ 与 $b$，认为 $a<b$ 即有两解。例如 $a=1,\ b=3,\ A=30^\circ$，$h=1.5$，$a=1<h$，实际无解，但若只看 $a<b$ 就会错判为两解。
		\item[\textbf{\textcolor{CTrap}{正确理解（两步比较）：}}]
			锐角情形下先比较 $a$ 与 $h$：若 $a<h$ 无解；$a=h$ 一解；$a>h$ 再比较 $a$ 与 $b$ 得最终解数。
		\item[\textbf{\textcolor{CTrap}{错因分析（知识遗漏）：}}]
			不知道 $h=b\sin A$ 是 $AB$ 边上的高，或不理解 $a<h$ 时以 $C$ 为圆心的圆与 $AB$ 不相交的几何意义。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（误读等号情形）}}]
			当 $a=h$ 时，认为无解（以为 $a$ 小于 $b$）或有两解。例如 $a=2,\ b=4,\ A=30^\circ$，$a=h$，应判一解，常被误判。
		\item[\textbf{\textcolor{CTrap}{正确理解（唯一解条件）：}}]
			$a=h$ 对应 $\sin B=1$，$B=90^\circ$，三角形唯一。这是锐角情况下的边界条件。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略几何相切）：}}]
			学生没理解 $a=h$ 时圆与射线 $AB$ 恰好相切，只有一个交点，故一解。
	\end{description}

\end{trapbox}
