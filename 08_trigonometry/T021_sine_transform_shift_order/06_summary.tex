\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{正弦曲线 $y=\sin x$ 通过两种不同顺序的平移和伸缩均能到达 $y=A\sin(\omega x+\varphi)$，唯先缩后移的平移量须用 $|\varphi/\omega|$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item **条件1（参数约束）：** $A\neq 0$，$\omega>0$，这是变换前提。
		\item **条件2（基本函数）：** 变换出发点是 $y=\sin x$。若从其他函数出发，原理相通但平移量公式可能变化。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item **易错点（平移量区别）：** 先伸缩后平移时，一定要用 $|\varphi/\omega|$ 而不是 $|\varphi|$。
		\item **检查项（结果验证）：** 可通过代入 $x=0$ 或特殊点检验变换结果是否与直接写出的一致。
	\end{itemize}
\end{summarybox}
