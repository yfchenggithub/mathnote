\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（参数条件）：}}] $A\neq 0$ 且 $\omega>0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（先移后缩）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			先水平平移 $|\varphi|$ 个单位，再将横坐标变为原来的 $\frac{1}{\omega}$，最后将纵坐标变为原来的 $|A|$ 倍。
			\[
				\begin{aligned}
					y=\sin x &\to y=\sin(x+\varphi) \\
					&\to y=\sin(\omega x+\varphi) \\
					&\to y=A\sin(\omega x+\varphi).
			\end{aligned}
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（先缩后移）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			先将横坐标变为原来的 $\frac{1}{\omega}$，再水平平移 $\left|\frac{\varphi}{\omega}\right|$ 个单位，最后将纵坐标变为原来的 $|A|$ 倍。
			\[
				\begin{aligned}
					y=\sin x &\to y=\sin(\omega x) \\
					&\to y=\sin\!\Big[\omega\Big(x+\frac{\varphi}{\omega}\Big)\Big] \\
					&\to y=A\sin(\omega x+\varphi).
			\end{aligned}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（平移量差异）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			先缩后移的平移量是 $|\varphi/\omega|$，与先移后缩的 $|\varphi|$ 不同。
			\[
				T_1 = |\varphi|,
				\qquad
				T_2 = \left|\frac{\varphi}{\omega}\right|.
			\]
	\end{description}
\end{statementbox}
