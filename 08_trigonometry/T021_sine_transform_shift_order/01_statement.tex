\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{参数条件：}}] $A\neq 0$，$\omega>0$，$\varphi\in\mathbb R$；从 $y=\sin x$ 出发。$A$ 是带符号的纵向系数，振幅为 $|A|$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（先移后缩）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			先向左平移有符号距离 $\varphi$，再将横坐标变为原来的 $\frac{1}{\omega}$ 倍，最后将纵坐标乘以 $A$。
			\[
				\begin{aligned}
					y=\sin x &\to y=\sin(x+\varphi) \\
					&\to y=\sin(\omega x+\varphi) \\
					&\to y=A\sin(\omega x+\varphi).
			\end{aligned}
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（先缩后移）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			先将横坐标变为原来的 $\frac{1}{\omega}$ 倍，再向左平移有符号距离 $\frac{\varphi}{\omega}$，最后将纵坐标乘以 $A$。
			\[
				\begin{aligned}
					y=\sin x &\to y=\sin(\omega x) \\
					&\to y=\sin\!\Big[\omega\Big(x+\frac{\varphi}{\omega}\Big)\Big] \\
					&\to y=A\sin(\omega x+\varphi).
			\end{aligned}
			\]

		\item[\textbf{\textcolor{CStatement}{平移量与方向：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$\varphi>0$ 时左移，$\varphi<0$ 时右移，$\varphi=0$ 时不平移。两条路径的平移距离分别为
			\[
				T_1 = |\varphi|,
				\qquad
				T_2 = \frac{|\varphi|}{\omega}.
			\]
			当且仅当 $\varphi\neq0$ 且 $\omega\neq1$ 时，两者不同。若 $A<0$，最后一步等价于先按 $|A|$ 倍纵向伸缩，再关于 $x$ 轴对称。
	\end{description}
\end{statementbox}
