\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（开方忘符号）}}]
			已知 $\sin\alpha = \frac{3}{5}$，不判断象限直接得到 $\cos\alpha = \pm\frac{4}{5}$ 或 $\frac{4}{5}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（结合象限定号）}}]
			必须根据 $\alpha$ 所在象限确定 $\cos\alpha$ 的正负号，再开平方取值。
		\item[\textbf{\textcolor{CTrap}{错因分析（平方关系盲区）}}]
			开平方必然产生两个值，需要附加条件约束，不能默认正值。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（除以零项）}}]
			在化简或证明中，直接除以 $\cos^2\alpha$ 或 $\sin^2\alpha$ 而不考虑它们可能为零。
		\item[\textbf{\textcolor{CTrap}{正确理解（讨论分母）}}]
			遇到除以 $\cos^2\alpha$ 等，应注明 $\cos\alpha\neq0$ 或 $\sin\alpha\neq0$，若可能为零则需分类讨论。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义域忽略）}}]
			平方关系恒成立，但派生公式如 $1+\tan^2\alpha = \sec^2\alpha$ 要求 $\cos\alpha\neq0$，学生易忽视。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（展开漏项）}}]
			错误认为 $(\sin\alpha+\cos\alpha)^2 = \sin^2\alpha+\cos^2\alpha = 1$，遗漏交叉项 $2\sin\alpha\cos\alpha$。
		\item[\textbf{\textcolor{CTrap}{正确理解（完全平方公式）}}]
			\[
				(\sin\alpha\pm\cos\alpha)^2 = \sin^2\alpha+\cos^2\alpha\pm2\sin\alpha\cos\alpha = 1\pm\sin 2\alpha.
			\]
		\item[\textbf{\textcolor{CTrap}{错因分析（习惯性简化）}}]
			受平方和恒等式影响，忘记完全平方展开规则，漏掉二倍乘积项。
	\end{description}

\end{trapbox}
