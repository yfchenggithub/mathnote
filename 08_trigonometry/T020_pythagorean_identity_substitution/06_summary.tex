\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{$\sin^2\alpha+\cos^2\alpha=1$ 是三角函数化简求值的核心平方恒等式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（同角条件）：} 仅适用于同一个角 $\alpha$。
		\item \textbf{条件2（任意实数）：} 对任意角 $\alpha$（实数）均成立。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略符号）：} 平方关系求值开方后需根据象限确定符号。
		\item \textbf{检查项（分母非零）：} 涉及除以 $\sin\alpha$ 或 $\cos\alpha$ 时需讨论是否为零。
	\end{itemize}

\end{summarybox}
