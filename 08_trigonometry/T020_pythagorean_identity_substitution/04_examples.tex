\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 已知 $\sin\alpha = \frac{3}{5}$，且 $\alpha$ 为第二象限角，求 $\cos\alpha$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（平方变形）：}]
			由 $\sin^2\alpha + \cos^2\alpha = 1$，得 $\cos^2\alpha = 1 - \sin^2\alpha$。
			\par\textbf{理由：} 利用平方恒等式移项。
		\item[\textbf{第二步（数值代入）：}]
			代入 $\sin\alpha = \frac{3}{5}$，得 $\cos^2\alpha = 1 - \left(\frac{3}{5}\right)^2 = 1 - \frac{9}{25} = \frac{16}{25}$。
			\par\textbf{理由：} 代入已知值计算。
		\item[\textbf{第三步（符号判定）：}]
			因为 $\alpha$ 是第二象限角，故 $\cos\alpha < 0$，所以 $\cos\alpha = -\sqrt{\frac{16}{25}} = -\frac{4}{5}$。
			\par\textbf{理由：} 开平方后根据象限定号。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\cos\alpha = -\frac{4}{5}$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 化简 $(\sin\alpha + \cos\alpha)^2 + (\sin\alpha - \cos\alpha)^2$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（展开平方）：}]
			\[
				(\sin\alpha + \cos\alpha)^2 = \sin^2\alpha + 2\sin\alpha\cos\alpha + \cos^2\alpha
			\]
			\[
				(\sin\alpha - \cos\alpha)^2 = \sin^2\alpha - 2\sin\alpha\cos\alpha + \cos^2\alpha.
			\]
			\par\textbf{理由：} 应用完全平方公式展开。
		\item[\textbf{第二步（合并相加）：}]
			将两式相加，得
			\[
				(\sin\alpha+\cos\alpha)^2 + (\sin\alpha-\cos\alpha)^2 = 2\sin^2\alpha + 2\cos^2\alpha = 2(\sin^2\alpha+\cos^2\alpha).
			\]
			\par\textbf{理由：} 合并同类项，提取公因子2。
		\item[\textbf{第三步（代换恒等式）：}]
			由 $\sin^2\alpha+\cos^2\alpha=1$，得原式 $=2\times1=2$。
			\par\textbf{理由：} 代入平方恒等式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{原式恒等于 $2$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 已知 $\tan\theta = 2$，求 $\sin^2\theta + \sin\theta\cos\theta$ 的值。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分母代换）：}]
			将原式写为 $\frac{\sin^2\theta + \sin\theta\cos\theta}{1}$，并将分母1代换为 $\sin^2\theta+\cos^2\theta$，得
			\[
				\frac{\sin^2\theta + \sin\theta\cos\theta}{\sin^2\theta+\cos^2\theta}.
			\]
			\par\textbf{理由：} 利用 $1 = \sin^2\theta + \cos^2\theta$ 进行“1”的代换，化为齐次分式。
		\item[\textbf{第二步（齐次化）：}]
			分子分母同除以 $\cos^2\theta$（$\cos\theta\neq0$），得
			\[
				\frac{\tan^2\theta + \tan\theta}{\tan^2\theta+1}.
			\]
			\par\textbf{理由：} $\frac{\sin^2\theta}{\cos^2\theta} = \tan^2\theta$，$\frac{\sin\theta\cos\theta}{\cos^2\theta} = \tan\theta$。
		\item[\textbf{第三步（代入求值）：}]
			代入 $\tan\theta = 2$，得
			\[
				\frac{4+2}{4+1} = \frac{6}{5}.
			\]
			\par\textbf{理由：} 数值计算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin^2\theta + \sin\theta\cos\theta = \frac{6}{5}$。}
\end{examplebox}
