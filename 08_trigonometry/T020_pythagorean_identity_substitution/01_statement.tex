\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（任意角）：}}]
			$\alpha$ 为任意实数（角度制或弧度制均可）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平方恒等）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			同角的正弦平方加余弦平方恒等于1。
			\[
				1 = \sin^2\alpha + \cos^2\alpha.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（常见变形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			平方关系可相互代换，用于消元或构造完全平方。
			\[
				\sin^2\alpha = 1 - \cos^2\alpha.
			\]
			\[
				\cos^2\alpha = 1 - \sin^2\alpha.
			\]
			\[
				(\sin\alpha \pm \cos\alpha)^2 = 1 \pm \sin 2\alpha.
			\]
	\end{description}

\end{statementbox}
