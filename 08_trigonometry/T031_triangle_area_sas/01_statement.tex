\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（两边及夹角）：}}] 在$\triangle ABC$中，边长$a>0,b>0$，夹角$C\in(0,\pi)$.
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两边夹角面积）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			面积等于两边长与夹角正弦乘积的一半.
			\[
				\boxed{S=\frac{1}{2}ab\sin C}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（其他夹角形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			选择任意两边及其夹角均可计算面积.
			\[
				\boxed{S=\frac{1}{2}bc\sin A}
				\qquad
				\boxed{S=\frac{1}{2}ca\sin B}
			\]
	\end{description}

\end{statementbox}
