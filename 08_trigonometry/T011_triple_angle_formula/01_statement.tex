\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（定义域）：}}]
			$x$ 取任意实数。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正弦三倍角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $\sin 3x$ 展开为 $\sin x$ 的三次多项式。
			\[
				\sin 3x = 3\sin x - 4\sin^3 x
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（余弦三倍角）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $\cos 3x$ 展开为 $\cos x$ 的三次多项式。
			\[
				\cos 3x = 4\cos^3 x - 3\cos x
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（降幂形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $\sin^3 x$ 和 $\cos^3 x$ 用一次幂线性表示。
			\[
				\sin^3 x = \frac{3\sin x - \sin 3x}{4},
				\qquad
				\cos^3 x = \frac{3\cos x + \cos 3x}{4}
			\]
	\end{description}

\end{statementbox}
