\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin x = \frac{1}{3}$，求 $\sin 3x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}]
			\[
				\sin 3x = 3\sin x - 4\sin^3 x
			\]
			\par\textbf{理由：} 直接应用三倍角正弦公式。
		\item[\textbf{第二步（代入计算）：}]
			\[
				\sin 3x = 3\times \frac{1}{3} - 4\times \left(\frac{1}{3}\right)^3 = 1 - 4\times \frac{1}{27} = 1 - \frac{4}{27} = \frac{23}{27}
			\]
			\par\textbf{理由：} 代入已知数值，计算幂次。
		\item[\textbf{第三步（写出答案）：}]
			所以 $\sin 3x = \dfrac{23}{27}$。
			\par\textbf{理由：} 得出最后结果。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\boxed{\frac{23}{27}}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\cos x = \frac{1}{3}$，求 $\cos 3x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}]
			\[
				\cos 3x = 4\cos^3 x - 3\cos x
			\]
			\par\textbf{理由：} 直接应用三倍角余弦公式。
		\item[\textbf{第二步（代入计算）：}]
			\[
				\cos 3x = 4\times \left(\frac{1}{3}\right)^3 - 3\times \frac{1}{3} = 4\times \frac{1}{27} - 1 = \frac{4}{27} - 1 = -\frac{23}{27}
			\]
			\par\textbf{理由：} 代入已知数值，注意符号。
		\item[\textbf{第三步（写出答案）：}]
			所以 $\cos 3x = -\dfrac{23}{27}$。
			\par\textbf{理由：} 得出最后结果。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\boxed{-\frac{23}{27}}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $\sin x = \frac{1}{3}$，且 $x$ 为锐角，求 $\sin^3 x + \cos^3 x$ 的值。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（同角求余弦）：}]
			\[
				\cos x = \sqrt{1-\sin^2 x} = \sqrt{1-\frac{1}{9}} = \frac{2\sqrt{2}}{3}
			\]
			\par\textbf{理由：} 利用同角三角函数关系，因为 $x$ 为锐角，余弦取正值。
		\item[\textbf{第二步（三倍角求值）：}]
			\[
				\begin{aligned}
					\sin 3x &= 3\sin x - 4\sin^3 x = 3\times\frac{1}{3} - 4\times\frac{1}{27} = \frac{23}{27}, \\ \cos 3x &= 4\cos^3 x - 3\cos x = 4\times\left(\frac{2\sqrt{2}}{3}\right)^3 - 3\times\frac{2\sqrt{2}}{3} \\ &= 4\times\frac{16\sqrt{2}}{27} - 2\sqrt{2} = \frac{64\sqrt{2}}{27} - 2\sqrt{2} = \frac{10\sqrt{2}}{27}.
			\end{aligned}
			\]
			\par\textbf{理由：} 分别利用正弦、余弦三倍角公式，代入 $\sin x$ 和 $\cos x$ 的值计算。
		\item[\textbf{第三步（降幂求和）：}]
			\[
				\begin{aligned}
					\sin^3 x + \cos^3 x
					&= \frac{3\sin x - \sin 3x}{4} + \frac{3\cos x + \cos 3x}{4}\\
					&= \frac{3(\sin x+\cos x) + (\cos 3x - \sin 3x)}{4}\\
					&= \frac{3\cdot\frac{1+2\sqrt{2}}{3} + \frac{10\sqrt{2}-23}{27}}{4}\\
					&= \frac{1+2\sqrt{2} + \frac{10\sqrt{2}-23}{27}}{4}\\
					&= \frac{\frac{27(1+2\sqrt{2})+10\sqrt{2}-23}{27}}{4}\\
					&= \frac{27+54\sqrt{2}+10\sqrt{2}-23}{108}\\
					&= \frac{4+64\sqrt{2}}{108}\\
					&= \frac{1+16\sqrt{2}}{27}.
			\end{aligned}
			\]
			\par\textbf{理由：} 应用降幂公式 $\sin^3 x = \frac{3\sin x - \sin 3x}{4}$，$\cos^3 x = \frac{3\cos x + \cos 3x}{4}$，代入 $\sin x,\cos x,\sin 3x,\cos 3x$ 的值并化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\boxed{\frac{1+16\sqrt{2}}{27}}
\end{examplebox}
