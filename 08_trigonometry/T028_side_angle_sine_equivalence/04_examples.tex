\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，已知$A=70^\circ$，$B=40^\circ$，比较$a$与$b$的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（比较角大小）：}] 因为$70^\circ>40^\circ$，所以$A>B$。
			\[
				A>B.
			\]
			\par\textbf{理由：} 已知角度数值。

		\item[\textbf{第二步（套用结论）：}] 根据结论$A>B\iff a>b$，直接得出$a>b$。
			\[
				a>b.
			\]
			\par\textbf{理由：} 边角等价关系。

		\item[\textbf{第三步（正弦验证）：}] 用正弦定理验证：$\dfrac{a}{b}=\dfrac{\sin A}{\sin B}>1$，所以$a>b$。
			\[
				\frac{a}{b}=\frac{\sin70^\circ}{\sin40^\circ}>1.
			\]
			\par\textbf{理由：} 计算比值确认。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a>b$。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在$\triangle ABC$中，$a=10$，$b=6$，$\sin A=\dfrac{4}{5}$，求$\sin B$，并比较$A$与$B$的大小。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（正弦求 $\sin B$）：}] 由正弦定理$\dfrac{a}{\sin A}=\dfrac{b}{\sin B}$，得
			\[
				\begin{aligned}
					\sin B &= \frac{b\sin A}{a} \\
					&= \frac{6\times\frac{4}{5}}{10} \\
					&= \frac{24}{50} \\
					&= \frac{12}{25}.
			\end{aligned}
			\]
			\par\textbf{理由：} 正弦定理公式直接代入。

		\item[\textbf{第二步（边角等价推角关系）：}] 由$a>b$，根据结论$a>b\iff A>B$，得$A>B$。
			\[
				a>b\Longrightarrow A>B.
			\]
			\par\textbf{理由：} 边大则角大。

		\item[\textbf{第三步（正弦值验证）：}] 因为$\sin A=\dfrac{4}{5}=0.8$，$\sin B=\dfrac{12}{25}=0.48$，所以$\sin A>\sin B$，这同样推出$A>B$，与第二步一致。
			\[
				\sin A>\sin B\iff A>B.
			\]
			\par\textbf{理由：} 结论的等价性。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\sin B=\dfrac{12}{25}$，$A>B$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在锐角三角形$ABC$中，$\sin A<\sin B$，且$C=60^\circ$，求角$A$的取值范围。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（正弦反推角关系）：}] 由$\sin A<\sin B$，根据结论$\sin A<\sin B\iff A<B$，得$A<B$。
			\[
				A<B.
			\]
			\par\textbf{理由：} 等价关系直接使用。

		\item[\textbf{第二步（三角和与不等式）：}] 因为$C=60^\circ$，所以$A+B=120^\circ$。结合$A<B$，得$A<60^\circ$。
			\[
				A+B=120^\circ,\quad A<B\;\Longrightarrow\;A<60^\circ.
			\]
			\par\textbf{理由：} 若$A\ge60^\circ$，则$B\le60^\circ$，与$A<B$矛盾。

		\item[\textbf{第三步（锐角条件）：}] 锐角三角形每个角都小于$90^\circ$。由$B=120^\circ-A<90^\circ$，解得$A>30^\circ$。结合$A<60^\circ$，得$30^\circ<A<60^\circ$。
			\[
				120^\circ-A<90^\circ\;\Longrightarrow\;A>30^\circ.
			\]
			\par\textbf{理由：} 锐角定义，且$C=60^\circ<90^\circ$自动满足。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$30^\circ<A<60^\circ$。}
\end{examplebox}
