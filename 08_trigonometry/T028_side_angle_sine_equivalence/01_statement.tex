	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（适用前提）：}}] 在$\triangle ABC$中，$A,B$为内角，$a,b$分别为其对边。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（边角等价链）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			正弦值越大，对应角越大，对应边越长。
			\[
				\sin A > \sin B \iff A > B \iff a > b.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（三角全序一致）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			在三角形中，角的大小顺序与对应边、对应正弦完全同步。
			\[
				A > B > C \iff a > b > c \iff \sin A > \sin B > \sin C.
			\]
	\end{description}
