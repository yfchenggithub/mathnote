\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，已知 $a\cos A = b\cos B$，且 $A = 30^{\circ}$，求角 $B$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（应用结论）：}] 由 $a\cos A = b\cos B$ 得 $A = B$ 或 $A + B = 90^{\circ}$。
			\par\textbf{理由：} 本结论的直接应用。
		\item[\textbf{第二步（代入计算）：}] 若 $A = B$，则 $B = 30^{\circ}$；若 $A + B = 90^{\circ}$，则 $B = 60^{\circ}$。
			\par\textbf{理由：} 将已知角代入两种可能情况。
		\item[\textbf{第三步（验证内角和）：}] 当 $B = 30^{\circ}$ 时，$C = 180^{\circ} - 30^{\circ} - 30^{\circ} = 120^{\circ}$；当 $B = 60^{\circ}$ 时，$C = 180^{\circ} - 30^{\circ} - 60^{\circ} = 90^{\circ}$。两者均合理。
			\par\textbf{理由：} 三角形内角和为 $180^{\circ}$，需检验各角为正。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$B = 30^{\circ}$ 或 $B = 60^{\circ}$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a\cos A = b\cos B$，且 $c = 2$，$a = \sqrt{3}$，求边 $b$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（等价转化）：}] 由 $a\cos A = b\cos B$ 得 $a = b$ 或 $a^{2} + b^{2} = c^{2}$。
			\par\textbf{理由：} 这是结论的等价边关系形式。
		\item[\textbf{第二步（分类讨论）：}] 若 $a = b$，则 $b = a = \sqrt{3}$，此时三边为 $\sqrt{3}, \sqrt{3}, 2$，可构成三角形；若 $a^{2} + b^{2} = c^{2}$，则 $3 + b^{2} = 4$，解得 $b = 1$（$b > 0$），三边为 $\sqrt{3}, 1, 2$，也可构成三角形。
			\par\textbf{理由：} 分别利用等腰和直角条件求解，并检验三角形存在性。
		\item[\textbf{第三步（结论）：}] 因此 $b = \sqrt{3}$ 或 $b = 1$。
			\par\textbf{理由：} 综合两种情形。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$b = \sqrt{3}$ 或 $b = 1$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在 $\triangle ABC$ 中，$a\cos A = b\cos B$，且 $\sin A = \dfrac{1}{2}$，求角 $C$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求 $A$）：}] 由 $\sin A = \dfrac{1}{2}$ 得 $A = 30^{\circ}$ 或 $A = 150^{\circ}$（$A\in(0,\pi)$）。
			\par\textbf{理由：} 正弦值已知，但需注意角有两个可能。
		\item[\textbf{第二步（套用结论）：}] 由 $a\cos A = b\cos B$ 得 $A = B$ 或 $A + B = 90^{\circ}$。
			\par\textbf{理由：} 直接应用结论。
		\item[\textbf{第三步（分类讨论与内角验证）：}]
			\begin{itemize}[leftmargin=*, itemsep=0.2em, topsep=0em]
			\item 若 $A = B$，当 $A = 30^{\circ}$ 时 $B = 30^{\circ}$，则 $C = 120^{\circ}$；当 $A = 150^{\circ}$ 时 $B = 150^{\circ}$，$A+B>180^{\circ}$，舍去。
			\item 若 $A + B = 90^{\circ}$，当 $A = 30^{\circ}$ 时 $B = 60^{\circ}$，$C = 90^{\circ}$；当 $A = 150^{\circ}$ 时 $B = -60^{\circ}$，舍去。
		\end{itemize}
			\par\textbf{理由：} 每种可能均需满足三角形内角和及角的正性，排除无效解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$C = 120^{\circ}$ 或 $C = 90^{\circ}$。
\end{examplebox}
