\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（边角等式）：}}] 在 $\triangle ABC$ 中，角 $A, B, C$ 所对的边分别为 $a, b, c$，且 $a\cos A = b\cos B$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（形状判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三角形可能为等腰或直角三角形。
			\[
				A = B
			\]
			或
			\[
				A + B = \frac{\pi}{2}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（等价边关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			条件等价于两条边相等或两边的平方和等于第三边平方。
			\[
				a = b
			\]
			或
			\[
				a^{2} + b^{2} = c^{2}.
			\]
	\end{description}
\end{statementbox}
