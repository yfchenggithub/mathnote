\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$\tan x=2$，求$\displaystyle\frac{3\sin x+2\cos x}{\sin x-\cos x}$的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判齐次）：}]观察分子分母每一项中$\sin x$与$\cos x$的总次数均为1，符合一次齐次分式。
			\par\textbf{理由：} 齐次性判断是使用切化弦的前提。

		\item[\textbf{第二步（同除化切）：}]因为$\cos x\neq0$，分子分母同除以$\cos x$，得$\displaystyle\frac{3\tan x+2}{\tan x-1}$。
			\par\textbf{理由：} 同除$\cos x$后原式保持不变，且$\frac{\sin x}{\cos x}=\tan x$。

		\item[\textbf{第三步（代入求值）：}]将$\tan x=2$代入，得$\displaystyle\frac{3\times2+2}{2-1}=8$。
			\par\textbf{理由：} 代入后化简即得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{原式$=8$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$\displaystyle\frac{2\sin x-\cos x}{\sin x+3\cos x}=1$，求$\tan x$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（同除化切）：}]分子分母同除以$\cos x$（$\cos x\neq0$），得$\displaystyle\frac{2\tan x-1}{\tan x+3}=1$。
			\par\textbf{理由：} 将原分式转化为关于$\tan x$的方程。

		\item[\textbf{第二步（解方程）：}]两边乘$\tan x+3$，得$2\tan x-1=\tan x+3$，解得$\tan x=4$。
			\par\textbf{理由：} 整理合并同类项。

		\item[\textbf{第三步（验证条件）：}]当$\tan x=4$时，$\cos x\neq0$且原分母$\sin x+3\cos x\neq0$，故解有效。
			\par\textbf{理由：} 需要确保原分式有意义。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\tan x=4$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$\tan x=-3$，求$\displaystyle\frac{4\sin^2 x-2\sin x\cos x+\cos^2 x}{3\sin^2 x+\sin x\cos x-2\cos^2 x}$的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判齐次）：}]分子分母各项中$\sin x$与$\cos x$的次数均为2，是二次齐次分式。
			\par\textbf{理由：} 只有齐次才能使用分子分母同除$\cos^2 x$的方法。

		\item[\textbf{第二步（同除化切）：}]分子分母同除以$\cos^2 x$（$\cos x\neq0$），得$\displaystyle\frac{4\tan^2 x-2\tan x+1}{3\tan^2 x+\tan x-2}$。
			\par\textbf{理由：} $\frac{\sin^2 x}{\cos^2 x}=\tan^2 x$，$\frac{\sin x\cos x}{\cos^2 x}=\tan x$，常数项对应$1$。

		\item[\textbf{第三步（代入求值）：}]将$\tan x=-3$代入，分子为$4\times9-2\times(-3)+1=36+6+1=43$，分母为$3\times9+(-3)-2=27-3-2=22$，故原式=$\displaystyle\frac{43}{22}$。
			\par\textbf{理由：} 直接代入计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{原式$=\displaystyle\frac{43}{22}$。}
\end{examplebox}
