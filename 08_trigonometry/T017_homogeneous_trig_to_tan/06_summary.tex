\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对于$\sin x$和$\cos x$的齐次分式，可通过同除$\cos^n x$转化为$\tan x$的有理式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（齐次要求）：}分子分母各项关于$\sin x$和$\cos x$的次数相同，即齐次。
		\item \textbf{条件 2（非零条件）：}$\cos x\neq0$且原分母$Q_n(\sin x,\cos x)\neq0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（检查前提）：}使用前验证齐次性和$\cos x\neq0$。
		\item \textbf{检查项（验证分母）：}确保原分母不为零，避免代入后无意义。
	\end{itemize}
\end{summarybox}
