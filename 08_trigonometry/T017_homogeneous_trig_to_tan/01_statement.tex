\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（齐次分式）：}}]分子分母均为$\sin x$与$\cos x$的$n$次齐次多项式，且$\cos x\neq 0$，分母$Q_n(\sin x,\cos x)\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（切化弦公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			分子分母同除$\cos^n x$，化为$\tan x$的有理式。
			\[
				\boxed{\frac{P_n(\sin x,\cos x)}{Q_n(\sin x,\cos x)} = \frac{P_n(\tan x,1)}{Q_n(\tan x,1)}}
			\]

		\item[\textbf{\textcolor{CStatement}{特例（一次与二次）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			常见的一次、二次齐次分式转化公式如下。
			\[
				\frac{a\sin x+b\cos x}{c\sin x+d\cos x}=\frac{a\tan x+b}{c\tan x+d}
			\]
			\[
				\frac{a\sin^{2}x+b\sin x\cos x+c\cos^{2}x}{d\sin^{2}x+e\sin x\cos x+f\cos^{2}x}=\frac{a\tan^{2}x+b\tan x+c}{d\tan^{2}x+e\tan x+f}
			\]
	\end{description}
\end{statementbox}
