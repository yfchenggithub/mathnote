\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	将函数 $y = 3\sin x + 4\cos x$ 合并为 $A\sin(x+\varphi)$ 的形式（$A>0$），并写出辅助角 $\varphi$ 的正切值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算振幅）：}] 计算 $A = \sqrt{a^{2}+b^{2}} = \sqrt{3^{2}+4^{2}} = 5$。
			\par\textbf{理由：} 合并后的振幅等于系数平方和开方。

		\item[\textbf{第二步（确定辅助角）：}] 由 $\cos\varphi = \dfrac{a}{A} = \dfrac{3}{5}$，$\sin\varphi = \dfrac{b}{A} = \dfrac{4}{5}$，得 $\varphi$ 为第一象限角，$\tan\varphi = \dfrac{\sin\varphi}{\cos\varphi} = \dfrac{4}{3}$。
			\par\textbf{理由：} 辅助角的正余弦直接由系数与振幅的比值定义。

		\item[\textbf{第三步（写出合并式）：}] 代入公式：
			\[
				y = 5\sin(x+\varphi),\quad \text{其中 } \tan\varphi = \frac{4}{3}
			\].
			\par\textbf{理由：} 直接应用辅助角公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y = 5\sin(x+\varphi)$，$\tan\varphi = \dfrac{4}{3}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	将函数 $y = -\sin x + \cos x$ 合并为 $A\sin(x+\varphi)$ 的形式，并指出 $\varphi$ 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算振幅）：}] $A = \sqrt{(-1)^{2}+1^{2}} = \sqrt{2}$。
			\par\textbf{理由：} 系数平方和开方。

		\item[\textbf{第二步（确定辅助角）：}] $\cos\varphi = \dfrac{-1}{\sqrt{2}} = -\dfrac{\sqrt{2}}{2}$，$\sin\varphi = \dfrac{1}{\sqrt{2}} = \dfrac{\sqrt{2}}{2}$。因为 $\cos\varphi<0$，$\sin\varphi>0$，所以 $\varphi$ 在第二象限，取 $\varphi = \dfrac{3\pi}{4}$。
			\par\textbf{理由：} 必须同时用正余弦的符号确定象限，不能只用正切。

		\item[\textbf{第三步（写出合并式）：}]
			\[
				y = \sqrt{2}\,\sin\!\left(x+\frac{3\pi}{4}\right)
			\].
			\par\textbf{理由：} 代入辅助角公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$y = \sqrt{2}\sin\!\left(x+\dfrac{3\pi}{4}\right)$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知函数 $f(x) = \sin x + \sqrt{3}\cos x$，求 $f(x)$ 的最大值、最小值及周期。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（计算振幅）：}] $A = \sqrt{1^{2}+(\sqrt{3})^{2}} = \sqrt{1+3} = 2$。
			\par\textbf{理由：} 系数平方和开方。

		\item[\textbf{第二步（确定辅助角）：}] $\cos\varphi = \dfrac{1}{2}$，$\sin\varphi = \dfrac{\sqrt{3}}{2}$，所以 $\varphi = \dfrac{\pi}{3}$（第一象限）。
			\par\textbf{理由：} 正余弦均为正，取锐角。

		\item[\textbf{第三步（求最值与周期）：}] 合并得 $f(x) = 2\sin\!\left(x+\dfrac{\pi}{3}\right)$。于是最大值为 $2$，最小值为 $-2$，最小正周期 $T = 2\pi$。
			\par\textbf{理由：} 对于 $A\sin(\omega x+\varphi)$，振幅 $|A|$ 决定最值，周期 $T = 2\pi/\omega$，这里 $\omega=1$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{最大值为 $2$，最小值为 $-2$，周期 $2\pi$。}
