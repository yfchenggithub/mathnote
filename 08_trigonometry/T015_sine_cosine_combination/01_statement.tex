\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（系数不全零）：}}] 实数 $a$ 与 $b$ 不同时为零。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（正弦合一公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			把同角的正弦和余弦合并为一个正弦函数，振幅为 $\sqrt{a^{2}+b^{2}}$。
			\[
				a\sin x + b\cos x = \sqrt{a^{2}+b^{2}}\,\sin(x+\varphi),
			\]
			其中 $\varphi$ 满足
			\[
				\cos\varphi = \frac{a}{\sqrt{a^{2}+b^{2}}},
				\qquad
				\sin\varphi = \frac{b}{\sqrt{a^{2}+b^{2}}}
			\].
			当 $a\neq 0$ 时，亦可由 $\displaystyle\tan\varphi = \frac{b}{a}$ 确定 $\varphi$（$\varphi$ 的象限由 $a,b$ 的符号决定）。

		\item[\textbf{\textcolor{CStatement}{推广结论（余弦等价形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过相位转换，也可以用余弦函数表示，辅助角不同。
			\[
				a\sin x + b\cos x = \sqrt{a^{2}+b^{2}}\,\cos(x-\varphi'),
			\]
			其中 $\varphi'$ 满足
			\[
				\cos\varphi' = \frac{b}{\sqrt{a^{2}+b^{2}}},
				\qquad
				\sin\varphi' = \frac{a}{\sqrt{a^{2}+b^{2}}}
			\].
			此时 $\varphi' = \dfrac{\pi}{2} - \varphi$。
	\end{description}

\end{statementbox}
