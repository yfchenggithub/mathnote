\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，边长 $a=7$，$b=8$，$c=9$，求 $BC$ 边上的中线 $m_a$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 中线长公式 $m_a^2 = \dfrac{2b^2+2c^2-a^2}{4}$。
			\par\textbf{理由：} 已知三角形三边，直接求中线长。
		\item[\textbf{第二步（数值计算）：}] 代入 $a=7,b=8,c=9$，得
			\[
				\begin{aligned}
					m_a^2 &= \dfrac{2\cdot8^2+2\cdot9^2-7^2}{4} \\
					&= \dfrac{128+162-49}{4} \\
					&= \dfrac{241}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 平方和及加减运算。
		\item[\textbf{第三步（开方取正）：}]
			\[
				\begin{aligned}
					m_a &= \sqrt{\dfrac{241}{4}} \\
					&= \dfrac{\sqrt{241}}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 长度为正数，开平方得中线长。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m_a = \dfrac{\sqrt{241}}{2}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$b=6$，$c=8$，$\angle A = 60^{\circ}$，求 $BC$ 边上的中线 $m_a$ 的长。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求第三边）：}] 由余弦定理
			\[
				\begin{aligned}
					a^2 &= b^2 + c^2 - 2bc\cos A \\
					&= 36 + 64 - 2 \cdot 6 \cdot 8 \cdot \frac12 \\
					&= 100 - 48 \\
					&= 52,
			\end{aligned}
			\]
			所以
			\[
				\begin{aligned}
					a &= \sqrt{52} \\
					&= 2\sqrt{13}.
			\end{aligned}
			\]
			\par\textbf{理由：} 已知两边及夹角，用余弦定理求第三边。
		\item[\textbf{第二步（套用中线公式）：}]
			\[
				\begin{aligned}
					m_a^2 &= \dfrac{2b^2+2c^2-a^2}{4} \\
					&= \dfrac{2\cdot36+2\cdot64-52}{4} \\
					&= \dfrac{72+128-52}{4} \\
					&= \dfrac{148}{4} \\
					&= 37.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入已知边和中线公式。
		\item[\textbf{第三步（开方取正）：}] $m_a = \sqrt{37}$。
			\par\textbf{理由：} 长度取正值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m_a = \sqrt{37}$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	在三角形 $ABC$ 中，$a=6$，$b=7$，$BC$ 边上的中线 $AM = 5$，求边 $c$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出中线公式）：}] $m_a^2 = \dfrac{2b^2+2c^2-a^2}{4}$。
			\par\textbf{理由：} 已知两边及中线，利用中线公式解第三边。
		\item[\textbf{第二步（代入已知值）：}]
			\[
				5^2 = \dfrac{2\cdot7^2+2c^2-6^2}{4},
			\]
			即
			\[
				\begin{aligned}
					25 &= \dfrac{98+2c^2-36}{4} \\
					&= \dfrac{62+2c^2}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入 $a=6,b=7,m_a=5$。
		\item[\textbf{第三步（解方程）：}] 两边乘以 $4$：$100 = 62+2c^2$，所以 $2c^2 = 38$，$c^2=19$，$c=\sqrt{19}$（取正）。
			\par\textbf{理由：} 代数运算，边长取正值。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$c = \sqrt{19}$。}
\end{examplebox}
