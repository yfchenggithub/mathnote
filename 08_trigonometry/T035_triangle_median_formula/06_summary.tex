\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{中线 $m_a$ 的平方满足 $m_a^2 = \dfrac{2b^2+2c^2-a^2}{4}$，其中 $a$ 是中线所对的边。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（已知三边）：} 三角形三边长度全部给出，可直接求任一边上的中线。
		\item \textbf{条件 2（间接求解）：} 已知两边及其夹角，先用余弦定理求第三边，再代入公式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（字母对应）：} 公式中 $a$ 必须是所求中线对应的对边，不可混用。
		\item \textbf{检查项（结果开方）：} 公式给出的是平方，最终中线长应取算术平方根。
	\end{itemize}
\end{summarybox}
