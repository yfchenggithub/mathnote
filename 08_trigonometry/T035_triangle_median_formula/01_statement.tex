\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三角形设定）：}}] 在任意三角形 $ABC$ 中，记边 $BC = a$，$CA = b$，$AB = c$，$M$ 为 $BC$ 中点，中线 $AM = m_a$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（中线平方公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			中线 $m_a$ 的平方可用三边 $b,c,a$ 表示。
			\[
				m_a^{2} = \frac{2b^{2}+2c^{2}-a^{2}}{4}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（对称公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			其余两边上的中线平方有类似形式。
			\[
				m_b^{2} = \frac{2a^{2}+2c^{2}-b^{2}}{4}
			\]
			\[
				m_c^{2} = \frac{2a^{2}+2b^{2}-c^{2}}{4}
			\]
	\end{description}
\end{statementbox}
