\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{子集法将充要条件判断转化为集合包含关系，小集合对应充分条件。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（集合化）：} 条件$p,q$必须能转化为明确的解集$A,B$。
		\item \textbf{条件2（可比较）：} 解集$A,B$应能比较包含关系（通常是数轴上的区间或点集）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（方向混淆）：} 记清$A\subseteq B\Leftrightarrow p\Rightarrow q$，$p$是$q$的充分条件。
		\item \textbf{检查项（等号核查）：} 判断真子集时需检验集合是否相等，防止误判。
	\end{itemize}

\end{summarybox}
