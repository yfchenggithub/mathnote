\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（解集对应）：}}]
			条件$p,q$对应的解集分别为$A=\{x\mid p(x)\}$，$B=\{x\mid q(x)\}$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（包含推充分）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若集合$A$是$B$的子集，则条件$p$是$q$的充分条件。
			\[
				\boxed{A\subseteq B \Leftrightarrow (p\Rightarrow q)}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（包含定条件类型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			不同包含关系对应不同充分/必要属性。
			\[
				\begin{aligned}
					A=B &\Leftrightarrow (p\Leftrightarrow q),\\ A\subsetneqq B &\Leftrightarrow (p\Rightarrow q)\land\lnot(q\Rightarrow p),\\ B\subseteq A &\Leftrightarrow (q\Rightarrow p).
			\end{aligned}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{适用范围（条件转化）：}}
	该结论适用于条件可以表示为明确解集（集合）的情形，如不等式解集、方程解集等。
\end{statementbox}
