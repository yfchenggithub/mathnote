\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知条件$p:x>2$，$q:x>1$，判断$p$是$q$的什么条件。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求集合）：}]
			解不等式得：
			\[
				A=(2,+\infty),\quad B=(1,+\infty).
			\]
			\par\textbf{理由：} 直接解$p$和$q$对应的不等式。
		\item[\textbf{第二步（比包含）：}]
			观察集合关系：$A\subseteq B$，且$B\not\subseteq A$（例如$x=1.5\in B$但$x\notin A$）。
			\[
				A\subseteq B,\quad B\not\subseteq A.
			\]
			\par\textbf{理由：} 集合端点比较或元素举例。
		\item[\textbf{第三步（得结论）：}]
			由子集法，$A\subseteq B$且$A\neq B$，故$p\Rightarrow q$但$q\not\Rightarrow p$，所以$p$是$q$的充分不必要条件。
			\par\textbf{理由：} 根据推广结论：$A\subsetneqq B$对应充分不必要。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$p$是$q$的充分不必要条件。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知$p:\; x^2-3x+2=0$，$q:\; x=1$，求$p$是$q$的什么条件。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求集合）：}]
			解$p$得$x=1$或$x=2$，故$A=\{1,2\}$；$B=\{1\}$。
			\[
				A=\{1,2\},\quad B=\{1\}.
			\]
			\par\textbf{理由：} 方程的解集。
		\item[\textbf{第二步（比包含）：}]
			显然$B\subseteq A$，且$B\neq A$。
			\[
				B\subseteq A,\quad B\neq A.
			\]
			\par\textbf{理由：} $A$包含$B$且有额外元素$2$。
		\item[\textbf{第三步（得结论）：}]
			由$B\subseteq A$得$q\Rightarrow p$，故$p$是$q$的必要不充分条件（$p\Leftarrow q$但$p\not\Rightarrow q$）。
			\par\textbf{理由：} $B\subseteq A$对应$q\Rightarrow p$，即$p$是$q$的必要条件，且$A\neq B$故不充分。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$p$是$q$的必要不充分条件。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知$p:\; |x-1|\le 2$，$q:\; |x|\le m$ ($m>0$)，若$p$是$q$的充分不必要条件，求实数$m$的取值范围。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求集合）：}]
			解绝对值不等式得：
			\[
				A=[-1,3],\quad B=[-m,m].
			\]
			\par\textbf{理由：} $|x-1|\le2\Rightarrow -2\le x-1\le2\Rightarrow -1\le x\le3$；$|x|\le m\Rightarrow -m\le x\le m$。
		\item[\textbf{第二步（建包含关系）：}]
			$p$是$q$的充分不必要条件$\Leftrightarrow A\subsetneqq B$，即$A\subseteq B$且$A\neq B$。故有：
			\[
				-m\le -1 \land m\ge 3.
			\]
			\par\textbf{理由：} 包含关系$[-1,3]\subseteq[-m,m]$等价于左端点$-m\le -1$且右端点$m\ge 3$；且$A\neq B$在本题自动成立（因为$m>0$，当$m\ge3$时$B$比$A$宽，不相等）。
		\item[\textbf{第三步（解不等式）：}]
			由$-m\le -1$得$m\ge 1$；由$m\ge 3$得$m\ge 3$；取交集得$m\ge 3$。
			\[
				m\ge 3.
			\]
			\par\textbf{理由：} 结合$m>0$，最终范围是$m\ge3$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{实数$m$的取值范围是$[3,+\infty)$。}
\end{examplebox}
