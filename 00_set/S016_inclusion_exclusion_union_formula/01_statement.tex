\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（集合有限）：}}]
			设 $A,B,C$ 均为有限集合。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两集合并集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两集合并集的大小等于它们各自大小之和减去重叠部分的大小。
			\[
				|A\cup B| = |A| + |B| - |A\cap B|.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（三集合并集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			三集合并集的大小等于各集大小之和减去两两交集大小之和再加上三者的公共部分大小。
			\[
				\begin{aligned}
					|A\cup B\cup C| &= |A| + |B| + |C| \\ &\quad - |A\cap B| - |B\cap C| - |C\cap A| \\ &\quad + |A\cap B\cap C|.
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
