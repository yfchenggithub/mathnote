\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（方向链颠倒）}}] 错误认为$A\Rightarrow B$，$B\Rightarrow C$可推出$C\Rightarrow A$。
		\item[\textbf{\textcolor{CTrap}{正确理解（方向不可逆）：}}] 传递性仅沿同一方向，不可反向推理。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆充分必要）：}}] 将充分条件误当作充要条件，忽略箭头指向。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（漏反向箭头）}}] 证明$A\Leftrightarrow C$时，只证到$A\Rightarrow B$和$B\Rightarrow C$就断言等价。
		\item[\textbf{\textcolor{CTrap}{正确理解（双向需齐全）：}}] 需同时验证$A\Rightarrow B$，$B\Rightarrow A$，$B\Rightarrow C$，$C\Rightarrow B$方可得出$A\Leftrightarrow C$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略反向传递）：}}] 未完成反向推导，将单向视为等价。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（中间命题不一）}}] 已知$A\Rightarrow B$和$B'\Rightarrow C$且$B\neq B'$，错误套用传递性得$A\Rightarrow C$。
		\item[\textbf{\textcolor{CTrap}{正确理解（链条需衔接）：}}] 传递要求第二步的前提与第一步的结论为同一命题。
		\item[\textbf{\textcolor{CTrap}{错因分析（未检查一致性）：}}] 忽略命题是否严格相同，造成链条断裂。
	\end{description}
\end{trapbox}
