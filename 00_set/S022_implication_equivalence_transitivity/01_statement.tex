\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（命题设定）：}}] 设$A,B,C$为任意命题。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（推出传递）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若$A$能推出$B$且$B$能推出$C$，则$A$可直接推出$C$。
			\[
				(A\Rightarrow B)\land(B\Rightarrow C)\;\Rightarrow\;(A\Rightarrow C).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（等价传递）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			充要条件也可沿链传递，等价关系具有传递性。
			\[
				(A\Leftrightarrow B)\land(B\Leftrightarrow C)\;\Rightarrow\;(A\Leftrightarrow C).
			\]
	\end{description}
\end{statementbox}
