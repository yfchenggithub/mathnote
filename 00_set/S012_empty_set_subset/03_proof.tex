\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	利用子集的定义：$B \subseteq C$ 意为“若 $x \in B$，则 $x \in C$”。对空集而言，前提 $x \in \emptyset$ 恒假，所以整个条件自动成立。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（写定义）：}]
			要证 $\emptyset \subseteq A$，按定义需证：
			\[
				\forall x (x \in \emptyset \implies x \in A)
			\]
			\par\textbf{理由：} 集合包含关系的标准定义。
		\item[\textbf{步骤二（前提分析）：}]
			注意到空集的定义：$x \in \emptyset$ 永假，即不存在任何元素属于空集。
			\par\textbf{理由：} 空集不含任何元素。
		\item[\textbf{步骤三（逻辑判定）：}]
			在逻辑中，若前提 $p$ 为假，则蕴含式 $p \rightarrow q$ 总为真（无论 $q$ 真值如何）。这里 $p$ 为 “$x \in \emptyset$”（假），$q$ 为 “$x \in A$”。因此该蕴含式对任意 $x$ 成立。
			\par\textbf{理由：} 数学逻辑的“虚真”规则。
		\item[\textbf{步骤四（得子集关系）：}]
			由步骤三，定义条件满足，故 $\emptyset \subseteq A$ 成立。
			\par\textbf{理由：} 定义与逻辑判定结合。
		\item[\textbf{步骤五（分析真子集）：}]
			现添加条件 $A \neq \emptyset$。要证 $\emptyset \subsetneqq A$，需证 $\emptyset \subseteq A$（已证）且 $\emptyset \neq A$。
			\par\textbf{理由：} 真子集定义：$B \subsetneqq C$ 当且仅当 $B \subseteq C$ 且 $B \neq C$。
		\item[\textbf{步骤六（排除相等）：}]
			若 $\emptyset = A$，则意味着 $A$ 也是空集，与 $A \neq \emptyset$ 矛盾。故 $\emptyset \neq A$。
			\par\textbf{理由：} 由相等定义，若两集合完全一致，则 $A$ 必无异于空集。
		\item[\textbf{步骤七（结论综合）：}]
			结合步骤四的 $\emptyset \subseteq A$ 与步骤六的 $\emptyset \neq A$，即得 $\emptyset \subsetneqq A$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	\textbf{因此，对任意集合 $A$ 有 $\emptyset \subseteq A$；若 $A \neq \emptyset$，则更有 $\emptyset \subsetneqq A$。}
\end{proofbox}
