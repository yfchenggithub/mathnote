\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（混用属于与包含）}}]
			认为 $\emptyset \in A$ 总是真（对非空集）。
		\item[\textbf{\textcolor{CTrap}{正确理解（子集非元素）：}}]
			空集是子集（$\subseteq$）不保证它是元素（$\in$）；除非显式列出，否则 $\emptyset \notin A$。
		\item[\textbf{\textcolor{CTrap}{错因分析（层混淆）：}}]
			日常语言“空集在集合里”模糊了集合与元素的层次。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略非空条件）}}]
			无条件说“空集是任何集合的真子集”。
		\item[\textbf{\textcolor{CTrap}{正确理解（空集自身除外）：}}]
			当集合本身是空集时，$\emptyset = A$，不是真子集。
		\item[\textbf{\textcolor{CTrap}{错因分析（记忆残缺）：}}]
			记忆结论时遗漏了“非空”前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（符号层次混淆）}}]
			书写 $\{ \emptyset \}$ 与 $\emptyset$ 不分，误认为 $\{ \emptyset \}$ 也是空集。
		\item[\textbf{\textcolor{CTrap}{正确理解（非空单元素集）：}}]
			$\{ \emptyset \}$ 是含有一个元素的集合，非空。
		\item[\textbf{\textcolor{CTrap}{错因分析（花括号误解）：}}]
			对“空集符号”与“花括号”的层次构建不清楚。
	\end{description}

\end{trapbox}
