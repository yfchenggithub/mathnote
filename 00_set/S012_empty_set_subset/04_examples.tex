\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	设集合 $A = \{ 1, 2, 3 \}$，判断下列关系式的正误，并说明理由：(1) $\emptyset \subseteq A$； (2) $\emptyset \subsetneqq A$； (3) $\emptyset \in A$； (4) $\{ \emptyset \} \subseteq A$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（运用结论判子集）：}]
			直接利用结论“空集是任何集合的子集，且对非空集合是真子集”，故(1)正确，(2)正确。
			\par\textbf{理由：} 因为 $A$ 非空，满足真子集条件。
		\item[\textbf{第二步（区分元素与子集）：}]
			检查(3)。$\emptyset \in A$ 表示空集是 $A$ 的元素，但 $A$ 的元素只有 $1,2,3$，都不等于 $\emptyset$，所以(3)错误。
			\par\textbf{理由：} $\in$ 表示元素归属，需检查集合中的具体成员。
		\item[\textbf{第三步（单元素集合与子集）：}]
			分析(4)。$\{ \emptyset \}$ 是以空集为唯一元素的集合。要使其是 $A$ 的子集，需要其元素 $\emptyset$ 属于 $A$，但由第二步知 $\emptyset \notin A$，故(4)错误。
			\par\textbf{理由：} 子集定义要求前者的每个元素都在后者中。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{空集是子集，但不一定是元素；$\{ \emptyset \}$ 与 $\emptyset$ 不同层次。}

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	设集合 $B = \{ x \mid x^2 + 1 = 0, \, x \in \mathbb{R} \}$，试回答：(1) 判断 $B$ 与 $\emptyset$ 的关系；(2) 写出 $B$ 的所有子集。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别方程解集）：}]
			方程 $x^2+1=0$ 无实数解，因此 $B = \emptyset$。
			\par\textbf{理由：} 实数范围内表达式 $x^2+1 \ge 1$，不可能为 $0$。
		\item[\textbf{第二步（判断子集关系）：}]
			根据结论，$\emptyset \subseteq B$ 成立（此时两者相等），且因为 $B = \emptyset$，所以 $\emptyset \subsetneqq B$ 不成立。
			\par\textbf{理由：} 真子集要求两集合不相等。
		\item[\textbf{第三步（列举子集）：}]
			$B$ 作为空集，其子集只有它自身，即 $\emptyset$。
			\par\textbf{理由：} 空集只有一个子集，就是它自己。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{空集本身只有自身一个子集；当集合为空时，空集不是它的真子集。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知集合 $C = \{ \emptyset, \{ 0 \} \}$，试完成下列任务：(1) 写出 $C$ 的所有子集；(2) 指出 $C$ 的真子集个数；(3) 判断 $\emptyset \in C$ 与 $\emptyset \subseteq C$ 的真假。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列举子集）：}]
			$C$ 含有两个元素，子集总数为 $2^2 = 4$ 个。逐层写出：不含元素：$\emptyset$；单元素子集：$\{ \emptyset \}$，$\{ \{ 0 \} \}$；双元素子集：$\{ \emptyset, \{ 0 \} \}$（即 $C$ 本身）。
			\par\textbf{理由：} 求子集个数公式为 $n$ 元素集合有 $2^n$ 个子集。
		\item[\textbf{第二步（辨别真子集）：}]
			真子集即除去自身外的子集，故有 $4-1=3$ 个真子集。
			\par\textbf{理由：} 真子集定义。
		\item[\textbf{第三步（区分子集与属于）：}]
			检查 $\emptyset \in C$：因为 $\emptyset$ 是 $C$ 列出的元素之一，故为真。检查 $\emptyset \subseteq C$：根据结论，空集是任何集合的子集，故为真。两者同时成立，但意义不同。
			\par\textbf{理由：} $\emptyset$ 作为元素写在花括号内，同时作为子集自动成立。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{空集可同时是某集合的元素和子集；遇到多重花括号要分层理解。}
\end{examplebox}
