\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{空集自动是任意集合的子集；若集合非空，则是真子集。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（子集定义）：} $\emptyset$ 为空集，$A$ 为任意集合。
		\item \textbf{条件 2（真子集条件）：} $A \neq \emptyset$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（区分∈与⊆）：} 牢记空集是子集未必是元素。
		\item \textbf{检查项（验证非空）：} 使用真子集符号前，必须确认集合不是空集。
	\end{itemize}

\end{summarybox}
