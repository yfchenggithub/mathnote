\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{集合运算的核心是根据元素是否满足特定条件（或、且、非）来构造新集合。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（全集明确）：}补集运算必须明确全集 \(U\)，否则补集无意义。
		\item \textbf{条件 2（集合表示确定）：}参与运算的集合应使用清晰的表示（列举法或描述法），确保元素可判定。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（补集前提）：}使用补集符号 \(\complement_U A\) 时不要漏写全集 \(U\)。
		\item \textbf{检查项（端点验证）：}区间运算后，对端点逐一代入检验，确定开闭性。
	\end{itemize}
\end{summarybox}
