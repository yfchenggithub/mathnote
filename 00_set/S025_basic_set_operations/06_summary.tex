\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{先把集合符号翻译成任意元素 \(x\) 的归属条件：并集是“或”，交集是“且”，补集是“全集内取非”，差集是“属于前者且不属于后者”。}

	\medskip
	\textbf{\textcolor{CSummary}{核心公式}}
	\[
		\begin{aligned}
			A\cup B
			 & =\{x\mid x\in A\text{ 或 }x\in B\},   \\
			A\cap B
			 & =\{x\mid x\in A\text{ 且 }x\in B\},   \\
			\complement_U A
			 & =\{x\in U\mid x\notin A\}
			=U\setminus A,                          \\
			A\setminus B
			 & =\{x\mid x\in A\text{ 且 }x\notin B\} \\
			 & =A\cap(\complement_U B).
		\end{aligned}
	\]

	\medskip
	\textbf{\textcolor{CSummary}{高频等价}}
	\[
		A\setminus B=\varnothing\iff A\subseteq B.
	\]

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{补集依赖全集：}符号下标可在上下文明确时省略，但全集必须确定。
		\item \textbf{并集的“或”允许同时成立：}公共部分也属于并集。
		\item \textbf{差集有方向：}一般 \(A\setminus B\neq B\setminus A\)。
		\item \textbf{区间检查端点：}把临界点代入元素条件，再确定开区间或闭区间。
	\end{itemize}
\end{summarybox}
