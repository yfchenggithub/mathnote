\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（全集与子集）：}}] 存在全集 \(U\)，集合 \(A, B\) 都是 \(U\) 的子集。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（并集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			取属于 \(A\) 或属于 \(B\) 的全部元素构成的新集合。
			\[
				A \cup B = \{ x \mid x \in A \text{ or } x \in B \}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（交集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			取同时属于 \(A\) 和 \(B\) 的全部元素构成的新集合。
			\[
				A \cap B = \{ x \mid x \in A \text{ and } x \in B \}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论三（补集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			取在全集 \(U\) 中但不在 \(A\) 中的全部元素构成的新集合。
			\[
				\complement_U A = \{ x \in U \mid x \notin A \}.
			\]

		\item[\textbf{\textcolor{CStatement}{结论四（差集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			取属于 \(A\) 但不属于 \(B\) 的全部元素构成的新集合。
			\[
				A \setminus B = \{ x \mid x \in A \text{ and } x \notin B \}.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{常见变形（差集补集关系）}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{常见变形一（补集为差集特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			补集是全集与集合的差集。
			\[
				\complement_U A = U \setminus A.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形二（差集用补集表示）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			差集可表示为 \(A\) 与 \(B\) 补集的交集。
			\[
				A \setminus B = A \cap (\complement_U B).
			\]
	\end{description}

\end{statementbox}
