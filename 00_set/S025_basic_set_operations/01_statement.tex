\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（全集与子集）：}}] 设 \(U\) 为全集，且 \(A,B\subseteq U\)。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（并集）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      取属于 \(A\) 或属于 \(B\) 的全部元素构成的新集合。这里的“或”是相容或，即同时属于 \(A\) 和 \(B\) 的元素也要保留。
		      \[
			      A\cup B=\{x\mid x\in A\text{ 或 }x\in B\}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（交集）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      取同时属于 \(A\) 和 \(B\) 的全部元素构成的新集合。
		      \[
			      A\cap B=\{x\mid x\in A\text{ 且 }x\in B\}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论三（补集）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      取全集 \(U\) 中不属于 \(A\) 的全部元素构成的新集合。
		      \[
			      \complement_U A=\{x\in U\mid x\notin A\}.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论四（差集）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      取属于 \(A\) 但不属于 \(B\) 的全部元素构成的新集合。
		      \[
			      A\setminus B=\{x\mid x\in A\text{ 且 }x\notin B\}.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{常见变形}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{变形一（补集写成差集）：}}]
		      \[
			      \complement_U A=U\setminus A.
		      \]

		\item[\textbf{\textcolor{CStatement}{变形二（差集写成交集与补集）：}}]
		      \[
			      A\setminus B=A\cap(\complement_U B).
		      \]

		\item[\textbf{\textcolor{CStatement}{变形三（差集为空与子集）：}}]
		      \[
			      A\setminus B=\varnothing\iff A\subseteq B.
		      \]
	\end{description}

\end{statementbox}
