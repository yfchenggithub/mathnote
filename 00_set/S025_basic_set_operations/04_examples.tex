\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 \(U=\{1,2,3,4,5\}\)，集合 \(A=\{1,2\}\)，\(B=\{2,3\}\)。求 \(A\cup B\)，\(A\cap B\)，\(\complement_U A\)，\(A\setminus B\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（并集交集）：}] 按定义写出结果。
			\[
				A\cup B = \{1,2,3\},\quad A\cap B = \{2\}.
			\]
			\par\textbf{理由：}
			并集取所有在 \(A\) 或 \(B\) 中的元素；交集取同时属于 \(A\) 和 \(B\) 的元素。
		\item[\textbf{第二步（补集计算）：}] 计算 \(\complement_U A\)。
			\[
				\complement_U A = U \setminus A = \{3,4,5\}.
			\]
			\par\textbf{理由：}
			补集是全集中不在 \(A\) 中的元素。
		\item[\textbf{第三步（差集计算）：}] 计算 \(A\setminus B\)。
			\[
				A\setminus B = \{1\}.
			\]
			\par\textbf{理由：}
			差集是 \(A\) 中不属于 \(B\) 的元素，即从 \(A\) 中去掉也在 \(B\) 中的元素。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	直接应用定义即可得到正确结果，需熟练掌握符号与定义。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 \(U=\mathbb{R}\)，集合 \(A = \{x\mid 1 \leq x \leq 3\}\)，\(B = \{x \mid 2 < x < 4\}\)。求 \(A \cup B\)，\(A \cap B\)，\(\complement_U A\)，\(A \setminus B\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（区间运算）：}] 由数轴确定结果。
			\[
				A\cap B = (2,3],\quad A\cup B = [1,4).
			\]
			\par\textbf{理由：}
			交集取公共部分，并集取全部，注意端点开闭。
		\item[\textbf{第二步（补集计算）：}] 计算 \(\complement_U A\)。
			\[
				\complement_U A = (-\infty,1) \cup (3,+\infty).
			\]
			\par\textbf{理由：}
			补集定义，全集中不在 \(A\) 内的实数。
		\item[\textbf{第三步（差集计算）：}] 计算 \(A\setminus B\)。
			\[
				A\setminus B = [1,2].
			\]
			\par\textbf{理由：}
			差集是 \(A\) 中不属于 \(B\) 的部分。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	区间运算时利用数轴，并注意端点开闭。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 \(U=\mathbb{R}\)，\(A=[1,2]\)，\(B=[a,a+1]\)，若 \(A\setminus B = \varnothing\)，求实数 \(a\) 的取值范围。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（差集条件转化）：}] 由差集定义，\(A\setminus B=\varnothing\) 表示没有元素属于 \(A\) 而不属于 \(B\)，即 \(A\) 中每个元素都属于 \(B\)，故 \(A\subseteq B\)。
			\par\textbf{理由：}
			差集空等价于子集关系。
		\item[\textbf{第二步（子集不等式）：}] 由 \(A=[1,2]\)，\(B=[a,a+1]\)，\(A\subseteq B\) 要求 \(a \leq 1\) 且 \(a+1 \geq 2\)。
			\par\textbf{理由：}
			区间包含条件。
		\item[\textbf{第三步（求解参数）：}] 解不等式组得 \(a \leq 1\) 且 \(a \geq 1\)，所以 \(a=1\)。
			\par\textbf{理由：}
			解不等式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	差集为空等价于 \(A\subseteq B\)，利用区间包含关系求参数。
\end{examplebox}
