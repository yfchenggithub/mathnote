\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础运算）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 \(U=\{1,2,3,4,5\}\)，集合 \(A=\{1,2\}\)，\(B=\{2,3\}\)。求 \(A\cup B\)、\(A\cap B\)、\(\complement_U A\)、\(A\setminus B\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（并集与交集）：}]
		      \[
			      A\cup B=\{1,2,3\},\qquad A\cap B=\{2\}.
		      \]
		      \par\textbf{理由：}
		      并集保留至少属于一个集合的元素；交集只保留同时属于两个集合的元素。

		\item[\textbf{第二步（补集）：}]
		      \[
			      \complement_U A=U\setminus A=\{3,4,5\}.
		      \]
		      \par\textbf{理由：}
		      从全集 \(U\) 中删去集合 \(A\) 的元素。

		\item[\textbf{第三步（差集）：}]
		      \[
			      A\setminus B=\{1\}.
		      \]
		      \par\textbf{理由：}
		      从 \(A\) 中删去同时属于 \(B\) 的元素 \(2\)。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	差集有方向，除本题所求外，\(B\setminus A=\{3\}\)，二者并不相同。

	\textbf{\textcolor{CExample}{例 2（区间运算）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 \(U=\mathbb{R}\)，集合 \(A=[1,3]\)，\(B=(2,4)\)。求 \(A\cup B\)、\(A\cap B\)、\(\complement_U A\)、\(A\setminus B\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（并集与交集）：}]
		      \[
			      A\cap B=(2,3],\qquad A\cup B=[1,4).
		      \]
		      \par\textbf{理由：}
		      交集取公共部分，并集取覆盖的全部部分；端点是否保留，要看该端点是否属于相应集合。

		\item[\textbf{第二步（补集）：}]
		      \[
			      \complement_U A=(-\infty,1)\cup(3,+\infty).
		      \]
		      \par\textbf{理由：}
		      \(1\) 和 \(3\) 属于 \(A\)，所以不属于补集。

		\item[\textbf{第三步（差集）：}]
		      \[
			      A\setminus B=[1,2].
		      \]
		      \par\textbf{理由：}
		      \(B=(2,4)\) 不含 \(2\)，因此 \(2\) 在 \(A\setminus B\) 中要保留。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	区间运算最容易错在端点；应把临界值逐个代回集合条件判断。

	\textbf{\textcolor{CExample}{例 3（差集为空求参数）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 \(U=\mathbb{R}\)，\(A=[1,2]\)，\(B=[a,a+1]\)。若 \(A\setminus B=\varnothing\)，求实数 \(a\) 的值。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（转化集合关系）：}]
		      由
		      \[
			      A\setminus B=\varnothing\iff A\subseteq B,
		      \]
		      可知区间 \([1,2]\) 必须完全包含在 \([a,a+1]\) 中。

		\item[\textbf{第二步（比较左右端点）：}]
		      要使 \([1,2]\subseteq[a,a+1]\)，必须满足
		      \[
			      a\leq 1,
			      \qquad
			      a+1\geq 2.
		      \]

		\item[\textbf{第三步（求参数）：}]
		      由 \(a+1\geq2\) 得 \(a\geq1\)，结合 \(a\leq1\)，得到
		      \[
			      a=1.
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	两个闭区间长度都为 \(1\)，要使一个包含另一个，只能完全重合，因此也可直接判断 \(a=1\)。
\end{examplebox}
