\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（补集忽略全集）：}}]
		      只看“不属于 \(A\)”就写补集，却没有先确定元素在哪个全集中取值。

		\item[\textbf{\textcolor{CTrap}{正确理解（全集决定补集）：}}]
		      补集必须相对于某个确定的全集。若上下文已明确全集，可以简写补集符号；但换一个全集，补集往往会改变。

		\item[\textbf{\textcolor{CTrap}{错因分析（把补集当成绝对概念）：}}]
		      忽略了补集实际表示 \(U\setminus A\)，其中 \(U\) 是定义的一部分。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（并集与交集混淆）：}}]
		      将“至少满足一个条件”误写成交集，或将“两个条件同时满足”误写成并集。

		\item[\textbf{\textcolor{CTrap}{正确理解（先读逻辑词）：}}]
		      “或、至少一个”对应并集 \(\cup\)；“且、同时、公共”对应交集 \(\cap\)。韦恩图中，并集看全部覆盖区域，交集只看重叠区域。

		\item[\textbf{\textcolor{CTrap}{错因分析（只背符号形状）：}}]
		      没有把集合运算翻译成元素条件，导致符号靠机械记忆。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（把“或”理解成二选一）：}}]
		      认为属于 \(A\cap B\) 的元素不能属于 \(A\cup B\)。

		\item[\textbf{\textcolor{CTrap}{正确理解（相容或）：}}]
		      \(x\in A\cup B\) 只要求 \(x\in A\) 与 \(x\in B\) 至少一个成立；若二者同时成立，仍然属于并集。

		\item[\textbf{\textcolor{CTrap}{错因分析（混淆相容或与排他或）：}}]
		      数学中集合并集使用的是相容或，不是“只能选一个”。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点四（差集方向写反）：}}]
		      将 \(A\setminus B\) 错算成 \(B\setminus A\)。

		\item[\textbf{\textcolor{CTrap}{正确理解（从前者中删除后者）：}}]
		      \(A\setminus B\) 表示从 \(A\) 中删掉属于 \(B\) 的元素，结果一定是 \(A\) 的子集。

		\item[\textbf{\textcolor{CTrap}{错因分析（忽略运算顺序）：}}]
		      差集通常不满足交换律，即一般有 \(A\setminus B\neq B\setminus A\)。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点五（区间端点误判）：}}]
		      在并、交、补、差运算后，凭图形直觉直接写端点开闭。

		\item[\textbf{\textcolor{CTrap}{正确理解（逐点代入）：}}]
		      对每个临界点分别判断它是否满足结果集合的元素条件，再决定使用圆括号还是方括号。

		\item[\textbf{\textcolor{CTrap}{错因分析（忽略等号条件）：}}]
		      没有检查原集合是否含端点，尤其容易在补集和差集中漏保留端点。
	\end{description}
\end{trapbox}
