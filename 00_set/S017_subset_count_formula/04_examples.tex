\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知集合 $A = \{a, b, c, d\}$。求：(1) $A$ 的所有子集个数；(2) 包含元素 $a$ 的子集个数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（读出参数）：}] 集合 $A$ 共有 $4$ 个元素，故 $n=4$；涉及包含 $a$ 时，指定元素个数 $m=1$。
			\par\textbf{理由：} 直接计数元素即可。

		\item[\textbf{第二步（计算所有子集）：}] 由公式 $2^n$ 得：
			\[
				2^4 = 16.
			\]
			\par\textbf{理由：} 套用子集总数公式。

		\item[\textbf{第三步（计算含 $a$ 的子集）：}] 将 $n=4, m=1$ 代入 $2^{n-m}$，得：
			\[
				2^{4-1} = 2^3 = 8.
			\]
			\par\textbf{理由：} 使用包含指定元素的公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{当条件只是简单的“包含某几个元素”时，直接代公式计算。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	集合 $B$ 含有 $5$ 个元素，问同时包含指定的某 $2$ 个元素的子集有多少个？

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定参数）：}] $n=5$，指定的 $2$ 个元素构成 $m=2$。
			\par\textbf{理由：} 从题中读取基本量。

		\item[\textbf{第二步（排除无关元素）：}] 由于只关心这 $2$ 个元素必须出现，其余 $n-m=3$ 个元素自由选择。
			\par\textbf{理由：} 固定元素的选择已唯一确定。

		\item[\textbf{第三步（代入公式）：}] 子集个数为：
			\[
				2^{5-2} = 2^3 = 8.
			\]
			\par\textbf{理由：} 直接应用公式 $2^{n-m}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{即使题目未直接给出“指定元素个数”，也要能根据“同时包含”识别出 $m$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	设集合 $U = \{1,2,3,4,5\}$，求满足“包含元素 $1$ 但不含元素 $2$”的子集个数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（条件转化）：}] 条件“不含 $2$”等价于子集必须从 $U \setminus \{2\}$ 中选取，故可先将 $2$ 移除，全集缩小为 $V = \{1,3,4,5\}$，其元素个数为 $4$（即新的 $n'=4$）。
			\par\textbf{理由：} 排除禁止元素，缩小选择范围。

		\item[\textbf{第二步（应用包含公式）：}] 在 $V$ 中又要求含元素 $1$，此时指定元素个数 $m=1$。
			\par\textbf{理由：} 将问题转化为基本模型。

		\item[\textbf{第三步（计算）：}] 所求个数为：
			\[
				2^{4-1} = 2^3 = 8.
			\]
			\par\textbf{理由：} 直接代入 $2^{n'-m}$；也可用分步乘法：元素 $1$ 必选（$1$ 种），元素 $2$ 必不选（$1$ 种），其余 $3$ 个元素任选（各 $2$ 种），得 $1 \times 1 \times 2^3 = 8$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{遇到“不含某元素”时，可通过缩小全集，然后在新集合上直接套用包含公式。}
\end{examplebox}
