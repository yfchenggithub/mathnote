\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（有限集合）：}}] 设 $A$ 为含有 $n$ 个元素的有限集（$n\in\mathbb{N}$）。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（所有子集）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每个元素独立决定“取”或“不取”，$n$ 次选择产生 $2^n$ 种组合。
			\[
				|\mathcal{P}(A)| = 2^n.
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（含指定元素）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			强制包含的 $m$ 个元素无选择权，其余 $n-m$ 个元素自由组合。
			\[
				N = 2^{\,n-m}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（加法乘法原理）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			互斥情形用加法串联，独立步骤用乘法串联。
			\[
				N = N_1 + N_2 + \cdots + N_k,
			\]
			\[
				N = n_1 n_2 \cdots n_m.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{参数限制：}}
	要求 $n$ 为非负整数，$m$ 为整数且满足 $0 \le m \le n$。当 $m=0$ 时，公式 $2^{n-m}$ 退化为 $2^n$；当 $m=n$ 时，结果为 $2^0=1$，即仅有全集。

\end{statementbox}
