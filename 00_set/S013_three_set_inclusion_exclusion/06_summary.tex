\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{并集计数 = 加单集 $-$ 减两两交 $+$ 加三交。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（有限集合）：} 所有集合均为有限集，元素个数明确。
		\item \textbf{条件 2（交集已知）：} 需要已知或能求出所有两两交集及三交集的大小。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（加减错乱）：} 牢记“加 $-$ 减 $-$ 加”的节奏，最后一项是加号。
		\item \textbf{检查项（三重交集）：} 计算完后核对三重交集是否被正确加回，避免多减。
	\end{itemize}
\end{summarybox}
