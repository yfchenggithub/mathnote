\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（论域非空）：}}] $M$ 为非空集合。
		\item[\textbf{\textcolor{CStatement}{条件2（谓词定义）：}}] $p(x)$ 为定义在 $M$ 上的谓词。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（全称否定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$\neg(\forall x\in M,\; p(x))$ 等价于 $\exists x\in M,\; \neg p(x)$，即“并非所有 $x$ 都满足 $p(x)$”等价于“存在 $x$ 不满足 $p(x)$”。
			\[
				\neg\big(\forall x\in M,\; p(x)\big) \;\Longleftrightarrow\; \exists x\in M,\; \neg p(x).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（对偶规则）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$\neg(\exists x\in M,\; p(x))$ 等价于 $\forall x\in M,\; \neg p(x)$，即“不存在 $x$ 满足 $p(x)$”等价于“所有 $x$ 都不满足 $p(x)$”。
			\[
				\neg\big(\exists x\in M,\; p(x)\big) \;\Longleftrightarrow\; \forall x\in M,\; \neg p(x).
			\]
	\end{description}
\end{statementbox}
