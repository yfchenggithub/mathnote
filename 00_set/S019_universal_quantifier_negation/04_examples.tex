\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	写出命题“$\forall x\in\mathbb{R},\ x^2+1>0$”的否定形式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别量词）：}] 该命题含全称量词“$\forall$”，谓词为“$x^2+1>0$”。
			\par\textbf{理由：} 判定命题类型为全称命题。
		\item[\textbf{第二步（应用规则）：}] 将全称量词改为存在量词，谓词取反：$x^2+1>0$ 的否定为 $x^2+1\leq 0$。
			\par\textbf{理由：} 全称命题的否定规则：$\neg(\forall x\in M,\ p(x)) \iff \exists x\in M,\ \neg p(x)$。
		\item[\textbf{第三步（写出否定）：}] 所以原命题的否定为 $\exists x\in\mathbb{R},\ x^2+1\leq 0$。
			\par\textbf{理由：} 组合成完整命题并规范表述。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\neg(\forall x\in\mathbb{R},\ x^2+1>0) \iff \exists x\in\mathbb{R},\ x^2+1\leq 0$。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知命题 $P:\ \exists n\in\mathbb{N},\ n^2<0$。写出 $\neg P$，并判断 $P$ 与 $\neg P$ 的真假。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别量词）：}] 命题 $P$ 含存在量词“$\exists$”，谓词为“$n^2<0$”。
			\par\textbf{理由：} 判定命题类型为存在命题。
		\item[\textbf{第二步（应用规则）：}] 将存在量词改为全称量词，谓词取反：$n^2<0$ 的否定为 $n^2\geq 0$。
			\par\textbf{理由：} 存在命题的否定规则：$\neg(\exists x\in M,\ p(x)) \iff \forall x\in M,\ \neg p(x)$。
		\item[\textbf{第三步（判断真假）：}] 不存在自然数 $n$ 使得 $n^2<0$，故 $P$ 为假；$\neg P$ 为真。
			\par\textbf{理由：} 自然数的平方非负。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$P$ 假，$\neg P$ 真；$\neg P:\ \forall n\in\mathbb{N},\ n^2\geq 0$。

	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知命题 $P:\ \forall x\in\mathbb{R},\ x^2-2x+2>0$，判断 $P$ 的真假，并写出 $\neg P$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（配方化简）：}] $x^2-2x+2=(x-1)^2+1$。
			\par\textbf{理由：} 配方法。
		\item[\textbf{第二步（判断真假）：}] 对任意 $x\in\mathbb{R}$，$(x-1)^2+1\geq 1>0$ 恒成立，故 $P$ 为真。
			\par\textbf{理由：} 平方非负性。
		\item[\textbf{第三步（应用否定）：}] 全称命题的否定：量词改为存在，谓词取反，得 $\neg P:\ \exists x\in\mathbb{R},\ x^2-2x+2\leq 0$。
			\par\textbf{理由：} 全称命题否定规则。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$P$ 为真，$\neg P$ 为假；$\neg P:\ \exists x\in\mathbb{R},\ x^2-2x+2\leq 0$。
\end{examplebox}
