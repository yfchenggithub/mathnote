\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（误写分配形式）：}}]
			错误做法：将 $A\cup(B\cap C)$ 误展开为 $(A\cup B)\cap C$，只分配了一个 $\cup$。
		\item[\textbf{\textcolor{CTrap}{正确理解（完整分配）：}}]
			正确形式应为 $(A\cup B)\cap(A\cup C)$，左右括号内都要出现 $A$。
		\item[\textbf{\textcolor{CTrap}{错因分析（类比混淆）：}}]
			受数的乘法分配律 $a\times(b+c)=a\times b+a\times c$ 影响，错把并当作“乘”，以为提出一个 $A$ 即可，未意识到集合的分配需要“两项都带 $A$”。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（乱用结合律）：}}]
			错误做法：看到 $A\cup(B\cap C)$ 就想把括号去掉，写成 $A\cup B\cap C$，或错写成 $(A\cup B)\cap C$，认为括号可随意移动。
		\item[\textbf{\textcolor{CTrap}{正确理解（限定运算）：}}]
			结合律只适用于纯并运算或纯交运算。混合运算时括号不可随意去掉，必须用分配律处理。
		\item[\textbf{\textcolor{CTrap}{错因分析（过度推广）：}}]
			受小学加法结合律 $(a+b)+c=a+(b+c)$ 的惯性影响，忽略了集合运算有不同类型，直接套用导致错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（单向包含证相等）：}}]
			错误做法：证明集合等式时，只证了 $X\subseteq Y$ 就下结论 $X=Y$，漏掉反方向。
		\item[\textbf{\textcolor{CTrap}{正确理解（双向包含）：}}]
			必须同时证明 $X\subseteq Y$ 和 $Y\subseteq X$，才能保证两集合完全相等。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义不牢）：}}]
			对集合相等定义“$A\subseteq B$ 且 $B\subseteq A$”的记忆模糊，尤其在运用运算律变形时容易只推一个方向。
	\end{description}

\end{trapbox}
