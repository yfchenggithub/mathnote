\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{全称量词不等式转化为最值不等式：全称“任意”对应最严格的最值；存在量词“存在”对应较宽松的另一端最值。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（最值存在）：} 函数 $f(x)$ 在区间 $D$ 上必须存在最大值 $M$ 和最小值 $m$（通常闭区间连续函数满足）。
		\item \textbf{条件 2（常数明确）：} $a$ 为已知常数，且不等式方向明确。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（量词混淆）：} 混淆全称与存在对应的最值，导致不等号方向错误。
		\item \textbf{检查项（最值确认）：} 使用前确认最值存在，并正确判断量词类型（全称还是存在）及不等号方向。
	\end{itemize}

\end{summarybox}
