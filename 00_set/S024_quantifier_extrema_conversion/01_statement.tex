	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（最值存在）：}}]
			函数 $f(x)$ 在区间 $D$ 上存在最大值 $M$ 和最小值 $m$。
		\item[\textbf{\textcolor{CStatement}{条件二（给定常数）：}}]
			$a$ 为常数。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（全称大于转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有函数值都不小于 $a$ 等价于最小值不小于 $a$。
			\[
				\forall x\in D, f(x)\geq a \iff m\geq a.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（全称小于转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有函数值都不大于 $a$ 等价于最大值不大于 $a$。
			\[
				\forall x\in D, f(x)\leq a \iff M\leq a.
			\]
		\item[\textbf{\textcolor{CStatement}{结论三（存在大于转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			存在函数值不小于 $a$ 等价于最大值不小于 $a$。
			\[
				\exists x\in D, f(x)\geq a \iff M\geq a.
			\]
		\item[\textbf{\textcolor{CStatement}{结论四（存在小于转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			存在函数值不大于 $a$ 等价于最小值不大于 $a$。
			\[
				\exists x\in D, f(x)\leq a \iff m\leq a.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（严格大于转化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			所有函数值都大于 $a$ 等价于最小值大于 $a$。
			\[
				\forall x\in D, f(x)>a \iff m>a.
			\]
			其余严格不等式情形（$\forall x, f(x)<a$，$\exists x, f(x)>a$，$\exists x, f(x)<a$）同理。
	\end{description}

\end{statementbox}
