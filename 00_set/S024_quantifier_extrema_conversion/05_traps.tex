\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（量词最值对应）}}]
			将全称量词不等式与存在量词不等式对应的最值颠倒。例如 $\forall f(x)\geq a$ 错误转化为 $M\geq a$，或 $\exists f(x)\leq a$ 错误转化为 $M\leq a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（记忆规则）：}}]
			全称“$\ge$”看最小值 $m$，全称“$\le$”看最大值 $M$；存在“$\ge$”看最大值 $M$，存在“$\le$”看最小值 $m$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆根源）：}}]
			语言直觉误导：“所有值 $\ge a$” 联想到最大值，实际要求整体在 $a$ 上方，必须最小值 $\ge a$；“存在”则只需一端满足。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略最值存在）}}]
			未确认函数在区间 $D$ 上是否存在最大值和最小值，直接套用结论。
		\item[\textbf{\textcolor{CTrap}{正确理解（先验最值）：}}]
			只有函数在 $D$ 上确实存在最大值 $M$ 和最小值 $m$，结论才成立。若区间为开区间或函数不连续，需单独分析最值。
		\item[\textbf{\textcolor{CTrap}{错因分析（疏忽条件）：}}]
			常见错误：在开区间 $(a,b)$ 上单调函数无最值，却仍用结论，导致参数范围错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（严格不等式遗漏）}}]
			将非严格不等式的转化结果直接用于严格不等式，忽略等号细节。例如 $\forall f(x)>a$ 错误转化为 $m\geq a$。
		\item[\textbf{\textcolor{CTrap}{正确理解（不含等号）：}}]
			严格不等式转化时不等号不含等号：$\forall f(x)>a \iff m>a$，$\forall f(x)<a \iff M<a$，$\exists f(x)>a \iff M>a$，$\exists f(x)<a \iff m<a$。
		\item[\textbf{\textcolor{CTrap}{错因分析（类比不当）：}}]
			认为严格与非严格相同，忽略当 $m=a$ 时 $f(x)>a$ 不成立，必须 $m>a$ 才保证所有值大于 $a$。
	\end{description}

\end{trapbox}
