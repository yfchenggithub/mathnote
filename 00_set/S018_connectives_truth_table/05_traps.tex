\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（合取析取混淆）}}] 误将合充当或，析充当且。
		\item[\textbf{\textcolor{CTrap}{正确理解（区分）：}}] 合取两者皆真才真，析取一真即真。
		\item[\textbf{\textcolor{CTrap}{错因分析（语义）：}}] 日常用语混淆，需通过真值表区分。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（否定分配错）}}] 误以为 $\lnot(p\land q)=\lnot p\land\lnot q$。
		\item[\textbf{\textcolor{CTrap}{正确理解（德摩根）：}}] $\lnot(p\land q)\equiv\lnot p\lor\lnot q$。
		\item[\textbf{\textcolor{CTrap}{错因分析（类比分配）：}}] 误将代数分配律用于逻辑。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（枚举遗漏）}}] 列真值表时漏掉某行或顺序乱。
		\item[\textbf{\textcolor{CTrap}{正确理解（系统枚举）：}}] 固定顺序（如二进制序）枚举所有 $2^n$ 种。
		\item[\textbf{\textcolor{CTrap}{错因分析（随意）：}}] 缺乏枚举策略，易遗漏。
	\end{description}
\end{trapbox}
