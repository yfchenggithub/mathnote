\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	设 $p=1$，$q=0$，求 $p\land q$，$p\lor q$，$\lnot p$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（合取）：}] 由真值表，$p\land q=0$。
			\par\textbf{理由：}
			合取要求两者皆真，$q=0$ 故假。
		\item[\textbf{第二步（析取）：}] $p\lor q=1$。
			\par\textbf{理由：}
			析取至少一真，$p=1$ 即真。
		\item[\textbf{第三步（否定）：}] $\lnot p=0$。
			\par\textbf{理由：}
			否定取反。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$p\land q=0$，$p\lor q=1$，$\lnot p=0$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	设 $p=0$，$q=1$，求 $( \lnot p )\lor q$ 和 $p\land( \lnot q )$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（否定 $p$）：}] $\lnot p=1$。
			\par\textbf{理由：}
			$p=0$ 否定得 $1$。
		\item[\textbf{第二步（析取）：}] $( \lnot p )\lor q = 1$。
			\par\textbf{理由：}
			$\lnot p=1$ 已使析取为真。
		\item[\textbf{第三步（否定 $q$ 与合取）：}] $\lnot q=0$，$p\land( \lnot q )=0$。
			\par\textbf{理由：}
			两者皆假，合取假。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$( \lnot p )\lor q=1$，$p\land( \lnot q )=0$。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知 $p\land q=0$，$p\lor q=1$，$\lnot p=1$，求 $p,q$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由 $\lnot p$ 得 $p$）：}] $\lnot p=1 \Rightarrow p=0$。
			\par\textbf{理由：}
			否定取反。
		\item[\textbf{第二步（由 $p\lor q=1$ 得 $q$）：}] $p=0$ 且 $p\lor q=1$，故 $q=1$。
			\par\textbf{理由：}
			析取至少一真。
		\item[\textbf{第三步（验证 $p\land q$）：}] $p\land q=0$ 符合条件。
			\par\textbf{理由：}
			$p=0$ 已使合取假。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$p=0$，$q=1$。}
\end{examplebox}
