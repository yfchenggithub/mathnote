\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{补集满足“补两次等于没补”，且全集的补是空集，空集的补是全集。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（全集给定）：} 必须明确全集 $U$，且符号 $\complement_U$ 中的下标 $U$ 不可省略。
		\item \textbf{条件 2（子集关系）：} 对象集合 $A$ 必须是 $U$ 的子集，即 $A\subseteq U$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（全集依赖）：} 全集改变，补集结果随之改变，切勿凭习惯默认全集为 $\mathbb{R}$。
		\item \textbf{检查项（条件确认）：} 每次使用双补律前，先看一眼 $A$ 是否真的包含在 $U$ 中，以及全集符号是否一致。
	\end{itemize}
\end{summarybox}
