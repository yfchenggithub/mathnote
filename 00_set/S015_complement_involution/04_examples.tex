\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 $U=\{1,2,3,4,5\}$，$A=\{1,2\}$。求 $\complement_U(\complement_U A)$，并验证其等于 $A$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求单补）：}] 计算 $\complement_U A$，即 $U$ 中不在 $A$ 的元素。
			\[
				\complement_U A = \{3,4,5\}.
			\]
			\par\textbf{理由：}
			全集去掉 $1,2$ 后剩下 $3,4,5$。

		\item[\textbf{第二步（求双补）：}] 再求 $\complement_U(\complement_U A)$，即 $U$ 中不在 $\{3,4,5\}$ 的元素。
			\[
				\complement_U(\complement_U A) = \{1,2\}.
			\]
			\par\textbf{理由：}
			相当于挖掉 $3,4,5$，回到原来的 $1,2$。

		\item[\textbf{第三步（对比验证）：}] 结果 $\{1,2\}$ 恰好等于原集合 $A$，即 $\complement_U(\complement_U A) = A$。
			\par\textbf{理由：}
			符合双重补集还原律。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{计算连续两次补集，结果就是原集合，不需要逐层硬算。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知全集 $U=\mathbb{R}$，$A=\{x \mid x>1\}$。求 $\complement_U(\complement_U A)$，并用区间表示。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求单补区间）：}] $\complement_U A = \mathbb{R} \setminus A = \{x \mid x\le 1\}$，写为区间 $(-\infty,\,1]$。
			\par\textbf{理由：}
			实数轴上挖掉大于 $1$ 的部分，剩下小于等于 $1$ 的全部。

		\item[\textbf{第二步（双补回原）：}] $\complement_U(\complement_U A) = \mathbb{R} \setminus (-\infty,\,1] = \{x \mid x>1\}$，区间 $(1,\,+\infty)$。
			\par\textbf{理由：}
			再挖掉 $(-\infty,1]$，自然剩出 $(1,+\infty)$，这与 $A$ 完全一致。

		\item[\textbf{第三步（结论确认）：}] 所以 $\complement_U(\complement_U A) = A = (1,\,+\infty)$，直接应用了双补还原律，省去重复分析。
			\par\textbf{理由：}
			无限集合情形下，结论依然成立，只需注意全集是 $\mathbb{R}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{即使集合用不等式或区间描述，双重补集还原律依然直接生效。}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	设全集 $U=\{n\in\mathbb{N}^* \mid n\le 6\}$ （即 $\{1,2,3,4,5,6\}$），已知 $\complement_U B = \{2,4,6\}$。利用补集性质求 $B$，并计算 $\complement_U(\complement_U B)$ 验证。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（反求原集）：}] 根据补集定义，$B = U \setminus (\complement_U B) = \{1,3,5\}$。
			\par\textbf{理由：}
			补集已知，原集就是全集减去补集。

		\item[\textbf{第二步（利用双补律）：}] 由双重补集还原律，$\complement_U(\complement_U B) = B$。直接得 $\complement_U(\complement_U B) = \{1,3,5\}$。
			\par\textbf{理由：}
			无需从 $\{2,4,6\}$ 再算一次补集，定理已给出结果。

		\item[\textbf{第三步（双重验证）：}] 若手动计算，$\complement_U B = \{2,4,6\}$，对其再取补得 $\{1,3,5\}$，与结论一致。这表明双补律可反向使用：看到双补立即化简。
			\par\textbf{理由：}
			无论是正向求双补还是反向利用，都能简化运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{已知一个集合的补集时，原集合就是补集的补集，这一思路可快速切换视角。}
\end{examplebox}
