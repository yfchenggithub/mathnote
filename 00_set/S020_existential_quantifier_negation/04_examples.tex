\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} \textbf{命题“存在 $x\in\mathbb{R}$，使得 $x^2+1=0$”的否定。}\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
			确定原命题为 $\exists x\in\mathbb{R}, x^2+1=0$，即 $M=\mathbb{R}$，$p(x):x^2+1=0$。
			\par\textbf{理由：} 明确命题的结构形式。
		\item[\textbf{第二步（应用规则）：}]
			将存在量词改为全称量词，谓词取反，得 $\forall x\in\mathbb{R}, x^2+1\neq 0$。
			\par\textbf{理由：} 由结论 $\neg(\exists x\in M,p(x))\iff \forall x\in M,\neg p(x)$。
		\item[\textbf{第三步（写出结果）：}]
			否定命题为“对任意实数 $x$，$x^2+1\neq 0$”。
			\par\textbf{理由：} 完成改写。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{原命题的否定是 $\forall x\in\mathbb{R}, x^2+1\neq0$。}

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} \textbf{已知命题 $p$：“存在 $x\in(0,+\infty)$，使得 $x+\dfrac{1}{x}>2$”，写出 $\neg p$。}\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别结构）：}]
			$M=(0,+\infty)$，$p(x): x+\frac{1}{x}>2$。
			\par\textbf{理由：} 确定量词和谓词。
		\item[\textbf{第二步（应用规则）：}]
			量词改为全称，谓词 $>$ 的否定是 $\le$，得 $\forall x\in(0,+\infty), x+\frac{1}{x}\le 2$。
			\par\textbf{理由：} 由否定规则，注意不等号方向变化。
		\item[\textbf{第三步（写出结果）：}]
			$\neg p$ 为“对所有正实数 $x$，$x+\frac{1}{x}\le 2$”。
			\par\textbf{理由：} 完成改写。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\neg p: \forall x\in(0,+\infty), x+\frac{1}{x}\le 2$。}

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} \textbf{若命题“存在 $x\in\mathbb{R}$，$x^2+ax+1=0$”是假命题，求实数 $a$ 的取值范围。}\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（否定为真）：}]
			原命题为假，则否定为真。即 $\neg(\exists x\in\mathbb{R}, x^2+ax+1=0)$ 为真，由结论得 $\forall x\in\mathbb{R}, x^2+ax+1\neq 0$ 为真。
			\par\textbf{理由：} 假命题的否定是真命题。
		\item[\textbf{第二步（恒不等条件）：}]
			$\forall x\in\mathbb{R}, x^2+ax+1\neq0$ 恒成立，等价于二次函数 $y=x^2+ax+1$ 的判别式 $\Delta<0$。
			\par\textbf{理由：} 二次项系数为正，恒不为零意味着图像与 $x$ 轴无交点，故判别式小于 $0$。
		\item[\textbf{第三步（解不等式）：}]
			$\Delta = a^2-4$，且 $\Delta<0$，所以 $a^2-4<0$，解得 $-2<a<2$。故 $a$ 的取值范围是 $(-2,2)$。
			\par\textbf{理由：} 解二次不等式。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a\in(-2,2)$。}
