\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{否定存在量词命题时，将存在改为全称，谓词取反，即 $\neg(\exists x\in M,p(x))\iff \forall x\in M,\neg p(x)$。}\par

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（集合非空）：} 集合 $M$ 为非空集合。
		\item \textbf{条件2（命题函数）：} $p(x)$ 是关于 $x$ 的命题函数。
	\end{itemize}
	\par

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（量词互换）：} 否定时务必同时更换量词，勿只改谓词。
		\item \textbf{检查项（谓词取反）：} 检查谓词取反是否彻底，尤其涉及不等号时。
	\end{itemize}

\end{summarybox}
