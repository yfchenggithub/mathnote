	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略非零差）}}] 直接使用$S_n$的二次形式时，忘记检查公差$d\neq0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（验证前提）：}}] 只有当公差$d\neq0$时$S_n$才是二次函数；若$d=0$，$S_n=n a_1$是一次函数。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）：}}] 学生习惯套用公式，忽略前提条件，导致结论误用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（离散当连续）}}] 求$S_n$最值时直接将二次函数顶点坐标作为答案，未考虑$n$取正整数。
		\item[\textbf{\textcolor{CTrap}{正确理解（离散定义域）：}}] $S_n$仅定义在$n\in\mathbb{N}^*$上，最值应在对称轴附近的整数点中选取。
		\item[\textbf{\textcolor{CTrap}{错因分析（函数泛化）：}}] 将离散数列与连续函数等同，忽视$n$的整数限制。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（开向定增性）}}] 由二次项系数$\frac{d}{2}>0$直接断言$S_n$一定递增。
		\item[\textbf{\textcolor{CTrap}{正确理解（转换单调性）：}}] 开口向上时$S_n$先减后增；需结合对称轴判断$n$大小时的增减性。
		\item[\textbf{\textcolor{CTrap}{错因分析（误推单调）：}}] 将二次函数开口方向与数列单调性直接对应，忽略对称轴位置。
	\end{description}
\end{trapbox}
