\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{等差数列前$n$项和$S_n$可表示为常数项为零的二次函数：$S_n = \frac{d}{2}n^2 + \left(a_1 - \frac{d}{2}\right)n$，图象是过原点的抛物线。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（等差非零差）：} 数列$\{a_n\}$是等差数列且公差$d\neq0$。
		\item \textbf{条件 2（正整数域）：} $n\in\mathbb{N}^*$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（条件验证）：} 使用二次形式前务必检查公差$d\neq0$。
		\item \textbf{检查项（离散定义域）：} 求最值时考虑$n$为正整数，勿直接使用二次函数顶点。
	\end{itemize}
\end{summarybox}
