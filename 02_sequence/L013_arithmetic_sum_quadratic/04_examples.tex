\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 中，$a_1=2$，$d=4$，求 $S_n$ 的二次函数表达式，并指出二次项系数和一次项系数。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求和平式）：}] 写出等差数列前$n$项和基本公式。
			\[
				S_n = n a_1 + \frac{n(n-1)}{2}d.
			\]
			\par\textbf{理由：} 这是等差数列求和的标准公式，适用于任何等差数列。
		\item[\textbf{第二步（代值化简）：}] 代入$a_1=2$，$d=4$并展开化简。
			\[
				\begin{aligned}
					S_n &= 2n + \frac{n(n-1)}{2}\cdot 4 \\
					&= 2n + 2n(n-1) \\
					&= 2n^2.
			\end{aligned}
			\]
			\par\textbf{理由：} 先计算乘积，再合并同类项：$2n(n-1)=2n^2-2n$，与$2n$相加后消去一次项。
		\item[\textbf{第三步（确认形式）：}] 指出二次函数形式及其系数。
			\[
				S_n = 2n^2.
			\]
			\par\textbf{理由：} 表达式是$n$的二次式，二次项系数为$2$，一次项系数为$0$，常数项为$0$，图象是过原点的抛物线。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = 2n^2$}，二次项系数$2$，一次项系数$0$，开口向上。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 的前 $n$ 项和 $S_n = n^2 - 3n$，求首项 $a_1$ 和公差 $d$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别二次）：}] 将给定表达式与标准二次形式$S_n = \frac{d}{2}n^2 + \left(a_1-\frac{d}{2}\right)n$对比。
			\[
				S_n = n^2 - 3n.
			\]
			\par\textbf{理由：} 由前$n$项和的二次形式结论，$S_n$必可表示为上述形式，因此二次项系数$\frac{d}{2}=1$，一次项系数$a_1-\frac{d}{2}=-3$。
		\item[\textbf{第二步（比较系数）：}] 比较$n^2$和$n$的系数建立方程。
			\[
				\frac{d}{2}=1,
				\qquad
				a_1-\frac{d}{2}=-3.
			\]
			解得$d=2$，$a_1=-2$。
			\par\textbf{理由：} 多项式恒等时同次幂系数相等。
		\item[\textbf{第三步（验证结果）：}] 验证比较系数得到的参数正确。
			\[
				S_n = n^2 - 3n.
			\]
			\par\textbf{理由：} 计算$S_1=1-3$得$-2$，故$a_1=-2$；计算$S_2=4-6$得$-2$，$a_2=S_2-S_1=0$，从而$d=a_2-a_1=2$，与比较系数结果一致。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_1=-2$，$d=2$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 中，$a_1=5$，$d=-2$，求 $S_n$ 的二次函数形式，并求 $S_n$ 的最大值及相应的 $n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写出形式）：}] 代入参数得到$S_n$的二次表达式。
			\[
				\begin{aligned}
					S_n &= \frac{d}{2}n^2 + \left(a_1-\frac{d}{2}\right)n \\
					&= \frac{-2}{2}n^2 + \left(5 - \frac{-2}{2}\right)n \\
					&= -n^2 + 6n.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用二次形式结论，直接将$a_1$和$d$代入标准公式计算。
		\item[\textbf{第二步（配方求最值）：}] 将$S_n$配方，求最大值。
			\[
				\begin{aligned}
					S_n &= -n^2+6n \\
					&= -(n-3)^2+9.
			\end{aligned}
			\]
			开口向下，因此$n=3$时$S_n$最大，最大值为$9$。
			\par\textbf{理由：} 二次函数在对称轴$n=3$处取最大值（$n\in\mathbb{N}^*$且对称轴为整数）。
		\item[\textbf{第三步（结论）：}] 给出最大值和对应的$n$。
			\[
				\max S_n = 9,\quad n=3.
			\]
			\par\textbf{理由：} 由配方结果可直接读出。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = -n^2+6n$，最大值为$9$，在$n=3$时取得}。
\end{examplebox}
