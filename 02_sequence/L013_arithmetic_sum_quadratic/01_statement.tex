\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（等差非零差）：}}] 数列$\{a_n\}$是等差数列，公差$d \neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（二次函数形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			$S_n$可以写成$n$的二次式，常数项为$0$。
			\[
				S_n = \frac{d}{2}n^2 + \left(a_1 - \frac{d}{2}\right)n.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（开口方向）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			公差的正负决定二次函数开口方向。当$d>0$时图象开口向上，当$d<0$时图象开口向下。
	\end{description}
\end{statementbox}
