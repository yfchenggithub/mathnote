\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（正整数条件）：}}] $n\in\mathbb{N}^*$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（裂项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将 $n\cdot n!$ 拆为相邻阶乘之差。
			\[
				n\cdot n! = (n+1)! - n!.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（求前 $n$ 项和）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			裂项后累加消去中间项，只剩末项减首项。
			\[
				\sum_{k=1}^{n} k\cdot k! = (n+1)! - 1.
			\]
	\end{description}

\end{statementbox}
