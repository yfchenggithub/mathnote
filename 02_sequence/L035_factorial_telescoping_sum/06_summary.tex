\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{裂项相消，将 $k\cdot k!$ 拆为 $(k+1)!-k!$，求和只剩 $(n+1)!-1$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（通项形式）：} 数列通项为 $k\cdot k!$（或可化为 $k\cdot k!$ 形式）。
		\item \textbf{条件 2（指标范围）：} $k$ 从 1 到 $n$，且 $n\in\mathbb{N}^*$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（抵消后项）：} 消去中间项后仅剩 $(n+1)!-1$，不是 $n!-1$ 或 $(n+1)!$。
		\item \textbf{检查项（特例检验）：} 取 $n=1$：$1\cdot1! = 1$，而 $2!-1 = 1$，正确。取 $n=2$：$1\cdot1!+2\cdot2! = 5$，而 $(2+1)!-1 = 5$，正确。
	\end{itemize}

\end{summarybox}
