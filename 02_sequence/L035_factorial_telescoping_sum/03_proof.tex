\begin{proofbox}

	\textbf{\textcolor{CProof}{思路提示}}

	利用阶乘定义 $(n+1)!=(n+1)n!$，展开后移项即得裂项公式；再通过累加消去中间项得出求和公式。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（裂项变形）：}] 计算 $(n+1)!-n!$。
			\[
				\begin{aligned}
					(n+1)!-n!
					&= (n+1)n! - n! \\
					&= n!\,[(n+1)-1] \\
					&= n\cdot n!.
			\end{aligned}
			\]
			\par\textbf{理由：} 利用 $(n+1)!=(n+1)n!$ 展开，提取公因式 $n!$，化简得 $n\cdot n!$。

		\item[\textbf{步骤二（累加消项）：}] 将通项 $k\cdot k!$ 替换为 $(k+1)!-k!$，并写出前 $n$ 项和。
			\[
				\begin{aligned}
					S_n
					&= \sum_{k=1}^{n} k\cdot k! \\
					&= \sum_{k=1}^{n} [(k+1)! - k!] \\
					&= (2!-1!)+(3!-2!)+\cdots+[(n+1)!-n!].
			\end{aligned}
			\]
			\par\textbf{理由：} 由步骤一得 $k\cdot k!=(k+1)!-k!$，代入求和式。

		\item[\textbf{步骤三（合并结果）：}] 观察并消去相邻项。
			\[
				\begin{aligned}
					S_n
					&= -1! + 2! - 2! + 3! - \cdots - n! + (n+1)! \\
					&= (n+1)! - 1! \\
					&= (n+1)! - 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 所有中间项 $2!$ 到 $n!$ 均正负抵消，仅剩 $-1!$ 和 $(n+1)!$，而 $1!=1$。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	由此得到核心裂项公式 $n\cdot n!=(n+1)!-n!$ 及其求和结论 $\displaystyle\sum_{k=1}^{n} k\cdot k! = (n+1)!-1$。

\end{proofbox}
