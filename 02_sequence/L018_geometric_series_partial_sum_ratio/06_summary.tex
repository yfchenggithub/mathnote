\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{等比数列的前 $2n$ 项和与前 $n$ 项和之比为 $1+q^n$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（数列类型）：} 数列 $\{a_n\}$ 是等比数列，公比 $q$ 已知。
		\item \textbf{条件 2（分母非零）：} 前 $n$ 项和 $S_n\neq0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母检验）：} 必须验证 $S_n\neq0$ 才能使用公式。
		\item \textbf{检查项（下标匹配）：} 确保 $S_{2n}$ 与 $S_n$ 中的 $n$ 是同一个正整数。
	\end{itemize}
\end{summarybox}
