\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	等比数列 $\{a_n\}$ 中，$a_1=3$，$q=2$，求 $\dfrac{S_4}{S_2}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定n值）：}] 由 $S_4,S_2$ 确定 $n=2$（因为 $2n=4$）。
			\par\textbf{理由：} 结论 $\dfrac{S_{2n}}{S_n}=1+q^n$ 中需要明确 $n$。
		\item[\textbf{第二步（套用公式）：}] $\dfrac{S_4}{S_2}=5$。
			\par\textbf{理由：} 直接使用结论代入 $n=2,\ q=2$。
		\item[\textbf{第三步（代入验证）：}] $S_2=9$，$S_4=45$，$\dfrac{45}{9}=5$。
			\par\textbf{理由：} 手动计算验证确保结论正确。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\dfrac{S_4}{S_2}=5$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	等比数列 $\{a_n\}$ 中，$\dfrac{S_4}{S_2}=5$，求公比 $q$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定n值）：}] 由 $S_4,S_2$ 知 $n=2$，故 $1+q^2=5$。
			\par\textbf{理由：} 结论 $\dfrac{S_{2n}}{S_n}=1+q^n$ 对应 $n=2$。
		\item[\textbf{第二步（解方程）：}] $q^2=4$，解得 $q=2$ 或 $q=-2$。
			\par\textbf{理由：} 直接解二次方程。
		\item[\textbf{第三步（检验条件）：}] 当 $q=-2$ 时，$S_2\neq0$（$a_1\neq0$），满足分母非零，故 $q=-2$ 也成立。
			\par\textbf{理由：} 结论要求 $S_n\neq0$，需验证。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$q=2$ 或 $q=-2$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	设等比数列 $\{a_n\}$ 中，$S_3=14$，$S_6=126$，求公比 $q$ 与首项 $a_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（片段比值）：}] $\dfrac{S_6}{S_3}=1+q^3$，又 $\dfrac{S_6}{S_3}=9$，故 $1+q^3=9$，即 $q^3=8$，得 $q=2$。
			\par\textbf{理由：} 利用结论 $\dfrac{S_{2n}}{S_n}=1+q^n$ 取 $n=3$。
		\item[\textbf{第二步（求和公式）：}] $S_3=a_1\dfrac{1-2^3}{1-2}$，即 $S_3=7a_1$；由 $S_3=14$ 得 $a_1=2$。
			\par\textbf{理由：} 等比数列前 $n$ 项和公式 $S_n=a_1\dfrac{1-q^n}{1-q}$（$q\neq1$）。
		\item[\textbf{第三步（结果检验）：}] $S_6=126$，符合条件。
			\par\textbf{理由：} 验证结果正确。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$q=2$，$a_1=2$。
\end{examplebox}
