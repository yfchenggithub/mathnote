\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（分母非零条件）}}] 直接套用公式 $\dfrac{S_{2n}}{S_n}=1+q^n$，但未检查 $S_n\neq0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（先检验）：}}] 使用公式前必须确认前 $n$ 项和 $S_n\neq0$，否则等式无意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（遗漏条件）：}}] 忽略等比数列片段和可能为零（如 $q=-1$ 且 $n$ 为偶数），盲目代入导致错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（下标对应不清）}}] 将 $S_6/S_2$ 直接当作 $1+q^3$ 或 $1+q^1$，混淆 $n$ 的取值。
		\item[\textbf{\textcolor{CTrap}{正确理解（下标统一）：}}] 结论中 $S_{2n}$ 和 $S_n$ 的下标 $n$ 必须相同，即 $n$ 是同一个正整数。例如 $S_6/S_3$ 对应 $n=3$，$S_4/S_2$ 对应 $n=2$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}] 未理解 $2n$ 与 $n$ 的倍数关系，错误地将任意下标代入。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（$q=1$ 特殊情形）}}] 认为 $q=1$ 时公式不成立，因为求和公式分母为零。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式统一）：}}] $q=1$ 时，$S_n=na_1$，$S_n\neq0$；$S_{2n}/S_n=2$，$2=1+1^n$，结论同样成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定式）：}}] 只记忆了 $q\neq1$ 时的推导，未理解结论的普适性。
	\end{description}
\end{trapbox}
