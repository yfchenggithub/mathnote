\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（公比定义）：}}] 等比数列 $\{a_n\}$ 的公比为 $q$，前 $n$ 项和为 $S_n$，且 $S_n\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（片段比值）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			前 $2n$ 项和与前 $n$ 项和之比等于 $1$ 加公比的 $n$ 次方。
			\[
				\frac{S_{2n}}{S_n}=1+q^{n}.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（公比负一情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			公比为 $-1$ 时公式仍适用，但须保证分母不为零。
			\[
				\frac{S_{2n}}{S_n}=1+(-1)^{n}.
			\]
	\end{description}
\end{statementbox}
