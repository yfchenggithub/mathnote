\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（定义要素）：}}]
			数列$\{a_n\}$定义在$n\in\mathbb{N}^*$上。若存在最小正整数$T$，使得对任意$n$恒有$a_{n+T}=a_n$成立。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（周期定义）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			数列的项按固定长度$T$循环重复。
			\[
				a_{n+T}=a_n
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（递推周期）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			下面递推公式对一切有意义的首项都形成周期数列，周期分别为。
			\[
				\begin{aligned}
					&\text{(1)}\ a_{n+1}=1-\frac{1}{a_n}\;(a_n\neq0,1)\ \Longrightarrow\ T=3;\\ &\text{(2)}\ a_{n+1}=\frac{1+a_n}{1-a_n}\;(a_n\neq1)\ \Longrightarrow\ T=4;\\ &\text{(3)}\ a_{n+1}=-\frac{1}{a_n+1}\;(a_n\neq-1)\ \Longrightarrow\ T=3;\\ &\text{(4)}\ a_{n+2}=a_{n+1}-a_n\ \Longrightarrow\ T=6.
			\end{aligned}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒（下标转化）：}}
	对于周期数列，可将任意下标$n$写成$n=kT+r$（$r$为$1$至$T$的正整数），则$a_n=a_r$。同理，前$n$项和与积也可转化为前$r$项和与积的倍数组合。
\end{statementbox}
