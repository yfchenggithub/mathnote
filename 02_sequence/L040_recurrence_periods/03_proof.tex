\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	周期数列的判定通常采用“连续迭代”。把递推式反复代入，若经过固定步数后回到原来的 \(a_n\)，就能得到周期关系。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（递推一验证）：}]
		      设
		      \[
			      a_{n+1}=1-\frac1{a_n}.
		      \]
		      则
		      \[
			      \begin{aligned}
				      a_{n+2}
				       & =1-\frac1{a_{n+1}}       \\[2pt]
				       & =1-\frac1{1-\frac1{a_n}} \\[2pt]
				       & =1-\frac{a_n}{a_n-1}     \\[2pt]
				       & =\frac1{1-a_n},          \\[4pt]
				      a_{n+3}
				       & =1-\frac1{a_{n+2}}       \\[2pt]
				       & =1-\frac1{\frac1{1-a_n}} \\[2pt]
				       & =1-(1-a_n)               \\[2pt]
				       & =a_n.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 连续三次使用递推式后回到 \(a_n\)，所以 \(a_{n+3}=a_n\)。在实数范围且初值避开 \(0,1\) 时，最小正周期为 \(3\)。

		\item[\textbf{步骤二（递推二验证）：}]
		      设
		      \[
			      a_{n+1}=\frac{1+a_n}{1-a_n}.
		      \]
		      则
		      \[
			      \begin{aligned}
				      a_{n+2}
				       & =\frac{1+a_{n+1}}{1-a_{n+1}}                         \\[2pt]
				       & =\frac{1+\frac{1+a_n}{1-a_n}}{1-\frac{1+a_n}{1-a_n}} \\[2pt]
				       & =-\frac1{a_n},                                       \\[4pt]
				      a_{n+3}
				       & =\frac{1+a_{n+2}}{1-a_{n+2}}                         \\[2pt]
				       & =\frac{1-\frac1{a_n}}{1+\frac1{a_n}}                 \\[2pt]
				       & =\frac{a_n-1}{a_n+1},                                \\[4pt]
				      a_{n+4}
				       & =\frac{1+a_{n+3}}{1-a_{n+3}}                         \\[2pt]
				       & =\frac{1+\frac{a_n-1}{a_n+1}}{1-\frac{a_n-1}{a_n+1}} \\[2pt]
				       & =a_n.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 四次迭代后回到 \(a_n\)，所以 \(a_{n+4}=a_n\)。为保证递推始终有意义，实数初值应避开 \(-1,0,1\)。

		\item[\textbf{步骤三（递推三验证）：}]
		      设
		      \[
			      a_{n+1}=-\frac1{a_n+1}.
		      \]
		      则
		      \[
			      \begin{aligned}
				      a_{n+2}
				       & =-\frac1{a_{n+1}+1}            \\[2pt]
				       & =-\frac1{-\frac1{a_n+1}+1}     \\[2pt]
				       & =-\frac{a_n+1}{a_n},           \\[4pt]
				      a_{n+3}
				       & =-\frac1{a_{n+2}+1}            \\[2pt]
				       & =-\frac1{-\frac{a_n+1}{a_n}+1} \\[2pt]
				       & =-\frac1{-\frac1{a_n}}         \\[2pt]
				       & =a_n.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 三次迭代后回到 \(a_n\)，所以 \(a_{n+3}=a_n\)。为保证递推始终有意义，实数初值应避开 \(-1,0\)。

		\item[\textbf{步骤四（二阶递推验证）：}]
		      设
		      \[
			      a_{n+2}=a_{n+1}-a_n.
		      \]
		      则
		      \[
			      \begin{aligned}
				      a_{n+3}
				       & =a_{n+2}-a_{n+1}          \\[2pt]
				       & =(a_{n+1}-a_n)-a_{n+1}    \\[2pt]
				       & =-a_n,                    \\[4pt]
				      a_{n+4}
				       & =a_{n+3}-a_{n+2}          \\[2pt]
				       & =-a_n-(a_{n+1}-a_n)       \\[2pt]
				       & =-a_{n+1},                \\[4pt]
				      a_{n+5}
				       & =a_{n+4}-a_{n+3}          \\[2pt]
				       & =-a_{n+1}-(-a_n)          \\[2pt]
				       & =a_n-a_{n+1},             \\[4pt]
				      a_{n+6}
				       & =a_{n+5}-a_{n+4}          \\[2pt]
				       & =(a_n-a_{n+1})-(-a_{n+1}) \\[2pt]
				       & =a_n.
			      \end{aligned}
		      \]
		      同理可得 \(a_{n+7}=a_{n+1}\)，因此整个数列满足 \(a_{n+6}=a_n\)。
		      \par\textbf{理由：} 六步后第一项和第二项的位置都回到原状态，所以该二阶递推的周期一定整除 \(6\)。若初值不是特殊情形，最小正周期通常为 \(6\)。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	以上分别验证了四类常见递推的循环结构：前三类分别满足 \(a_{n+3}=a_n\)、\(a_{n+4}=a_n\)、\(a_{n+3}=a_n\)，第四类满足 \(a_{n+6}=a_n\)。实际使用时，还要检查递推是否在定义域内始终有意义。
\end{proofbox}
