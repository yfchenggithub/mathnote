\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（递推与参数约束）：}}] 数列 $\{a_n\}$ 满足递推 $a_{n+1}=p a_n+q^n$，其中 $p,q$ 为常数且 $q\neq0$；若使用除以 $p^{n+1}$ 的变换，还需 $p\neq0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（除以 $q^{n+1}$ 型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两边同除 $q^{n+1}$，化为标准线性递推。
			\[
				\frac{a_{n+1}}{q^{n+1}}=\frac{p}{q}\cdot\frac{a_n}{q^n}+\frac{1}{q}
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（除以 $p^{n+1}$ 型）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			两边同除 $p^{n+1}$，化为可累加形式。
			\[
				\frac{a_{n+1}}{p^{n+1}}=\frac{a_n}{p^n}+\frac{1}{p}\left(\frac{q}{p}\right)^n \quad (p\neq0)
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（参数相等特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $p=q$ 时，第一个变换直接得到等差数列。
			\[
				\frac{a_{n+1}}{q^{n+1}}=\frac{a_n}{q^n}+\frac{1}{q}
			\]
			此时数列 $\{\frac{a_n}{q^n}\}$ 是以 $\frac{1}{q}$ 为公差的等差数列。
	\end{description}
\end{statementbox}
