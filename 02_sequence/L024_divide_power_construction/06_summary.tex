\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{对形如$a_{n+1}=p a_n+q^n$的递推，两边同除$q^{n+1}$或$p^{n+1}$可转化为线性递推或累加形式，是求通项的标准技巧。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（递推形式）：} 数列满足$a_{n+1}=p a_n+q^n$，其中$p,q$为常数。
		\item \textbf{条件2（分母非零）：} $q\neq0$；若使用除以$p^{n+1}$的变形，还需$p\neq0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽视参数讨论）：} 使用前检查$p,q$条件，尤其注意$p=q$与$p\neq q$对应不同解法。
		\item \textbf{检查项（变形选择）：} 优先除以绝对值较小的底数的幂，可简化后续运算；若$p=q$，任意一种都会直接得到等差数列。
	\end{itemize}

\end{summarybox}
