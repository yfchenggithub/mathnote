\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略分母为零）：}}]
			学生直接使用结论，未检查$q\neq0$或$p\neq0$，导致凭空引入分母为零的项。
		\item[\textbf{\textcolor{CTrap}{正确理解（检查底数非零）：}}]
			使用同除$q^{n+1}$必须保证$q\neq0$；使用同除$p^{n+1}$必须保证$p\neq0$。解题时先确认参数条件。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）：}}]
			递推式中$q^n$已隐含$q\neq0$，但$p$可能为零（虽然题目可能已排除），学生缺乏默认检验的意识。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（机械当等差）：}}]
			学生认为除以$q^{n+1}$后得到$b_{n+1}=\frac{p}{q}b_n+\frac{1}{q}$一定是等差数列，从而错误认为公差是$\frac{1}{q}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（等差需$p=q$）：}}]
			当$p\neq q$时，$\frac{p}{q}\neq1$，递推式是一阶线性而非等差。只有$p=q$时系数化为1才是等差。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆系数）：}}]
			学生只关注常数项而忽略一次项系数，将形式相同的式子误判为等差。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（累加项数错）：}}]
			在应用累加法时，误写成$c_n = c_1 + \sum_{k=1}^{n} f(k)$，导致多一项或少一项。
		\item[\textbf{\textcolor{CTrap}{正确理解（明确起点终点）：}}]
			由$c_{k+1}=c_k+f(k)$，从$k=1$到$k=n-1$累加得$c_n = c_1 + \sum_{k=1}^{n-1}f(k)$。可验证$n=1$时和为零。
		\item[\textbf{\textcolor{CTrap}{错因分析（下标对应混乱）：}}]
			学生不熟悉累加法的推导过程，直接套用公式忽略下标变换。
	\end{description}

\end{trapbox}
