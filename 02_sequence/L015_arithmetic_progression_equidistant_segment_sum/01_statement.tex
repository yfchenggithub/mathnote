	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（数列前提条件）：}}] 等差数列$\{a_n\}$，公差$d$，前$n$项和$S_n$，且$S_0=0$。
		\item[\textbf{\textcolor{CStatement}{条件 2（分段定义条件）：}}] 固定$m\in\mathbb{N}^*$，定义第$k$个片段和$T_k=S_{km}-S_{(k-1)m}$，其中$k=1,2,3,\ldots$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（片段等差性质）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			各片段和依次构成新的等差数列。
			\[
				T_{k+1}-T_k = m^2 d \quad (k\ge 1)
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（通项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			第$k$段的和可用首段和与公差直接表示。
			\[
				T_k = S_m + (k-1)\,m^2 d
			\]
	\end{description}

\end{statementbox}
