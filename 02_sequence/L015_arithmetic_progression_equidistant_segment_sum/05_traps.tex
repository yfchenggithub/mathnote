\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公差误判）}}] 错误地认为各片段和构成的等差数列公差仍是原公差$d$，记为$m d$。
		\item[\textbf{\textcolor{CTrap}{正确理解（公差平方因子）}}] 公差为$m^2 d$，而非$m d$或$d$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略二次项）}}] 盲目类比“每$m$项增加$m d$”，未考虑求和公式中的二次项，导致平方因子缺失。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（定义起点忽略）}}] 使用$T_k = S_{km} - S_{(k-1)m}$时，忘记$S_0=0$的约定而错误计算$T_1$；或认为$T_1 = S_m - S_1$等。
		\item[\textbf{\textcolor{CTrap}{正确理解（规范定义）}}] $T_1 = S_m - S_0$，其中$S_0=0$，故$T_1=S_m$。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件遗漏）}}] 未注意约定，或者误将$S_{(k-1)m}$当作不含$k=1$的情况。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（条件误用）}}] 在非等差数列中套用该结论，或对任意数列盲目使用。
		\item[\textbf{\textcolor{CTrap}{正确理解（等差前提）}}] 结论仅适用于等差数列，其他数列不具备此性质。
		\item[\textbf{\textcolor{CTrap}{错因分析（移植错误）}}] 将等差数列的特有性质错误推广到一般数列。
	\end{description}

\end{trapbox}
