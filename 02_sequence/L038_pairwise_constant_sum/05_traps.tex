	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点一（误用前提）}}]
			认为任何数列前$n$项和都能写成$S_n = n(a_1+a_n)/2$的形式。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（适用范围）：}}]
			只有满足$a_k + a_{n+1-k} = C$（$C$为常数）的数列才适用，且$C$必须通过验证确定。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（条件忽视）：}}]
			忽略公式推导的前提，盲目类比等差数列。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点二（项数混淆）}}]
			在等差数列求和时，混淆项数$n$与通项$a_n$中的$n$，例如求$S_n = 1+2+\cdots+n$时，认为$S_n = n(1+n)/2$已经正确，但有时概念不清以为$n$是末项不是项数。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（项数意义）：}}]
			项数$n$表示求和项数，末项$a_n$是第$n$项的值，两者数值可能相同但含义不同。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			将数列的序号$n$与第$n$项的值混为一谈。\n
	\end{description}
	\n\n

	\medskip
	\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CTrap}{易错点三（C值误认）}}]
			对于非等差数列，错误认为常数$C$一定等于首项加末项。\n
		\item[\textbf{\textcolor{CTrap}{正确理解（C值验证）：}}]
			常数$C$必须由题设条件具体给出，或通过验证所有对称项和相等得到。\n
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定式）：}}]
			等差数列的固定结论限制了思维。\n
	\end{description}
