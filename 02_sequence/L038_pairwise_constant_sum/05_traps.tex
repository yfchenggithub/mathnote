\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（误用前提）}}]
		      认为任何数列前 \(n\) 项和都能写成 \(S_n=\frac{n(a_1+a_n)}{2}\)。
		\item[\textbf{\textcolor{CTrap}{正确理解（适用范围）：}}]
		      只有前 \(n\) 项满足
		      \[
			      a_k+a_{n+1-k}=C\quad(k=1,2,\dots,n)
		      \]
		      时，才能使用 \(S_n=\frac{nC}{2}\)。等差数列只是其中一个特例。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件忽视）：}}]
		      忽略公式推导前提，盲目类比等差数列求和公式。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（项数混淆）}}]
		      混淆项数 \(n\) 与末项 \(a_n\)。例如在 \(1+2+\cdots+n\) 中，项数和末项数值恰好都为 \(n\)，但这只是特殊情况。
		\item[\textbf{\textcolor{CTrap}{正确理解（项数意义）：}}]
		      公式中的 \(n\) 表示求和项数，\(a_n\) 表示第 \(n\) 项的值。二者含义不同，不能混为一谈。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
		      把数列的下标和数列项的值混在一起，导致代入公式时变量含义不清。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（\(C\) 值误认）}}]
		      对于非等差数列，错误认为常数 \(C\) 一定等于首项加末项，且不再验证其他对称项。
		\item[\textbf{\textcolor{CTrap}{正确理解（\(C\) 值验证）：}}]
		      由于条件要求对所有 \(k=1,2,\dots,n\) 都成立，所以必须验证所有对称项和都相同。只看首末两项不够。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定式）：}}]
		      等差数列中 \(C=a_1+a_n\) 的结论很常用，但非等差数列必须重新验证对称等和关系。
	\end{description}
\end{trapbox}
