\begin{examplebox}
	\n\textbf{\textcolor{CExample}{例 1（基础）}}\n\n\textbf{\textcolor{CExample}{题目：}} 已知等差数列$\{a_n\}$中，$a_1=5$，$a_{10}=50$，求前10项和$S_{10}$。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（识别等差）：}]
			确认数列是等差数列，首项$a_1=5$，末项$a_{10}=50$，项数$n=10$。\n
			\par\textbf{理由：}\n等差数列定义及求和公式适用前提。\n
		\item[\textbf{第二步（确定常数）：}]
			由等差数列性质，常数$C$等于首末项之和，即\n
			\[
				\n
				\begin{aligned}
					C &= a_1 + a_{10} \\ &= 5 + 50 \\ &= 55.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n等差数列满足$a_k + a_{n+1-k} = a_1 + a_n$。\n
		\item[\textbf{第三步（套用公式）：}]
			\n
			\[
				\n
				\begin{aligned}
					S_{10} &= \frac{nC}{2} \\ &= \frac{10 \times 55}{2} \\ &= 275.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n倒序求和公式直接代入。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $\boxed{S_{10} = 275}$。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 2（稍变形）}}\n\n\textbf{\textcolor{CExample}{题目：}} 数列$\{a_n\}$共有9项，且满足$a_k + a_{10-k} = 20$（$k=1,2,\dots,9$），求该数列前9项和$S_9$。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（验证条件）：}]
			题设直接给出所有对称项之和为常数20，即$C=20$，且项数$n=9$。\n
			\par\textbf{理由：}\n条件明确$a_k + a_{10-k}=20$对所有$k$成立。\n
		\item[\textbf{第二步（代入公式）：}]
			由倒序求和公式，$S_9 = \frac{9 \times 20}{2}$。\n
			\par\textbf{理由：}\n无需计算每一项，直接套用结论。\n
		\item[\textbf{第三步（计算结果）：}]
			\n
			\[
				\n
				\begin{aligned}
					S_9 &= \frac{9 \times 20}{2} \\ &= 90.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n算术运算。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $\boxed{S_9 = 90}$。\n\n

	\medskip
	\n\textbf{\textcolor{CExample}{例 3（综合应用）}}\n\n\textbf{\textcolor{CExample}{题目：}} 数列$\{a_n\}$的通项公式为$a_n = \sin^2(n^\circ)$（$n=1,2,\dots,89$），求该数列的前89项和$S_{89}$。\n\n\textbf{\textcolor{CExample}{解题步骤：}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{第一步（发现对称性质）：}]
			考虑\n
			\[
				\n
				\begin{aligned}
					a_k + a_{90-k} &= \sin^2(k^\circ) + \sin^2(90^\circ - k^\circ) \\ &= \sin^2(k^\circ) + \cos^2(k^\circ) \\ &= 1.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n利用诱导公式$\sin(90^\circ - \theta) = \cos\theta$及恒等式$\sin^2\theta + \cos^2\theta = 1$。\n
		\item[\textbf{第二步（确定常数与项数）：}]
			对所有$k=1,2,\dots,89$上式成立，故$C=1$，$n=89$。\n
			\par\textbf{理由：}\n常数$C$为1。\n
		\item[\textbf{第三步（套用求和公式）：}]
			\n
			\[
				\n
				\begin{aligned}
					S_{89} &= \frac{89 \times 1}{2} \\ &= \frac{89}{2}.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n直接代入$S_n = \frac{nC}{2}$。\n
	\end{description}
	\n\n\textbf{\textcolor{CExample}{关键结论：}} $\boxed{S_{89} = \frac{89}{2}}$。\n
\end{examplebox}
