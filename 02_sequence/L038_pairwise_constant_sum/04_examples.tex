\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 \(\{a_n\}\) 中，\(a_1=5\)，\(a_{10}=50\)，求前 \(10\) 项和 \(S_{10}\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别等差）：}]
		      数列为等差数列，首项 \(a_1=5\)，末项 \(a_{10}=50\)，项数为 \(10\)。
		      \par\textbf{理由：} 等差数列满足对称等和性质。
		\item[\textbf{第二步（确定常数）：}]
		      由等差数列性质，对称两项之和均等于首末项之和，故
		      \[
			      \begin{aligned}
				      C & =a_1+a_{10} \\
				        & =5+50       \\
				        & =55.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 等差数列满足 \(a_k+a_{n+1-k}=a_1+a_n\)。
		\item[\textbf{第三步（套用公式）：}]
		      \[
			      \begin{aligned}
				      S_{10}
				       & =\frac{nC}{2}         \\
				       & =\frac{10\times55}{2} \\
				       & =275.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 直接代入倒序求和公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{S_{10}=275}.
	\]

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	数列 \(\{a_n\}\) 共有 \(9\) 项，且满足
	\[
		a_k+a_{10-k}=20\quad(k=1,2,\dots,9),
	\]
	求该数列前 \(9\) 项和 \(S_9\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（验证条件）：}]
		      题设直接给出所有对称项之和为常数 \(20\)，因此 \(C=20\)，项数 \(n=9\)。
		      \par\textbf{理由：} 对 \(9\) 项数列而言，对称下标为 \(k\) 与 \(10-k\)。
		\item[\textbf{第二步（代入公式）：}]
		      由倒序求和公式，
		      \[
			      S_9=\frac{9\times20}{2}.
		      \]
		      \par\textbf{理由：} 已满足 \(a_k+a_{n+1-k}=C\) 的形式。
		\item[\textbf{第三步（计算结果）：}]
		      \[
			      \begin{aligned}
				      S_9
				       & =\frac{9\times20}{2} \\
				       & =90.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{S_9=90}.
	\]

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	数列 \(\{a_n\}\) 的通项公式为
	\[
		a_n=\sin^2(n^\circ)\quad(n=1,2,\dots,89),
	\]
	求该数列的前 \(89\) 项和 \(S_{89}\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（发现对称性质）：}]
		      对任意 \(k=1,2,\dots,89\)，有
		      \[
			      \begin{aligned}
				      a_k+a_{90-k}
				       & =\sin^2(k^\circ)+\sin^2((90-k)^\circ) \\
				       & =\sin^2(k^\circ)+\cos^2(k^\circ)      \\
				       & =1.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 使用诱导公式 \(\sin(90^\circ-\theta)=\cos\theta\) 和恒等式 \(\sin^2\theta+\cos^2\theta=1\)。
		\item[\textbf{第二步（确定常数与项数）：}]
		      前 \(89\) 项满足对称等和，故 \(C=1\)，\(n=89\)。
		      \par\textbf{理由：} 对称下标为 \(k\) 与 \(90-k\)，正好落在 \(1\) 到 \(89\) 内。
		\item[\textbf{第三步（套用求和公式）：}]
		      \[
			      \begin{aligned}
				      S_{89}
				       & =\frac{89\times1}{2} \\
				       & =\frac{89}{2}.
			      \end{aligned}
		      \]
		      \par\textbf{理由：} 直接代入 \(S_n=\frac{nC}{2}\)。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{S_{89}=\frac{89}{2}}.
	\]
