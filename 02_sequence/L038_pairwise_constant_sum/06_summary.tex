\begin{summarybox}
	\n\textbf{\textcolor{CSummary}{一句话核心}}\n\n数列满足对称等和性时，前$n$项和等于$n$倍常数$C$的一半。\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{使用条件}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{条件1（对称等和）：} 存在常数$C$，使得$a_k + a_{n+1-k} = C$对所有$k=1,2,\dots,n$成立。\n
		\item \textbf{条件2（有限数列）：} 数列是有限项，项数为$n$。\n
	\end{itemize}
	\n\n

	\medskip
	\n\textbf{\textcolor{CSummary}{关键提醒}}\n
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\n
		\item \textbf{易错点（C值验证）：} 使用前必须验证所有对称项和是否相等，不能仅凭部分项判断。\n
		\item \textbf{检查项（逆向应用）：} 公式可逆用：已知$S_n$和$n$，可求常数$C = 2S_n/n$，也可用于检验对称性。\n
	\end{itemize}
	\n
\end{summarybox}
