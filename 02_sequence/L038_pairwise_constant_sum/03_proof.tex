\begin{proofbox}
	\n\textbf{\textcolor{CProof}{思路提示}}\n\n利用倒序相加，将$n$项和转化为$n$对常数之和，从而免去逐项累加。\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{正式推导}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\n
		\item[\textbf{步骤一（正序写出）：}]
			\n
			\[
				\nS_n = a_1 + a_2 + \dots + a_n.\n
			\]
			\n
			\par\textbf{理由：}\n前$n$项和的定义。\n\n
		\item[\textbf{步骤二（倒序写出）：}]
			\n
			\[
				\nS_n = a_n + a_{n-1} + \dots + a_1.\n
			\]
			\n
			\par\textbf{理由：}\n加法交换律，倒序排列不影响和。\n\n
		\item[\textbf{步骤三（对应相加）：}]
			\n将两式相加，对齐对称项：\n
			\[
				\n2S_n = (a_1 + a_n) + (a_2 + a_{n-1}) + \dots + (a_n + a_1).\n
			\]
			\n
			\par\textbf{理由：}\n左边为$S_n + S_n = 2S_n$；右边逐项相加。\n\n
		\item[\textbf{步骤四（代入条件）：}]
			\n由已知条件$a_k + a_{n+1-k} = C$（$k=1,2,\dots,n$），每个括号均为$C$，且共有$n$个括号：\n
			\[
				\n
				\begin{aligned}
					2S_n &= \underbrace{C + C + \dots + C}_{n} \\ &= nC.
			\end{aligned}
				\n
			\]
			\n
			\par\textbf{理由：}\n条件保证每对和都等于同一常数$C$。\n\n
		\item[\textbf{步骤五（除以系数）：}]
			\n两边同除以$2$，即得\n
			\[
				\n\boxed{S_n = \frac{nC}{2}}.\n
			\]
			\n
			\par\textbf{理由：}\n求解$S_n$。\n
	\end{description}
	\n\n

	\medskip
	\n\textbf{\textcolor{CProof}{结论回扣}}\n\n该证明显化“配对”思想：成功将$n$个不同项之和转化为$n$个相同常数之和，计算大幅简化。\n
\end{proofbox}
