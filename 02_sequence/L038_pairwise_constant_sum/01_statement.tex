	\n\textbf{\textcolor{CStatement}{条件}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\n
		\item[\textbf{\textcolor{CStatement}{条件1（倒序等和性质）：}}]
			存在常数$C$，使得对任意$k=1,2,\dots,n$都有$a_k + a_{n+1-k} = C$。\n
	\end{description}
	\n\n\textbf{\textcolor{CStatement}{结论}}\n
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\n
		\item[\textbf{\textcolor{CStatement}{结论一（倒序求和公式）：}}]
			\n\textbf{\textcolor{CStatement}{直观描述：}} 前$n$项之和等于$n$倍常数$C$的一半。\n
			\[
				\nS_n = \frac{nC}{2}.\n
			\]
			\n\n
		\item[\textbf{\textcolor{CStatement}{推广结论（等差数列特例）：}}]
			\n\textbf{\textcolor{CStatement}{直观描述：}} 等差数列中，前$n$项和等于首项与末项之和乘以项数的一半。\n
			\[
				\n\boxed{S_n = \frac{n(a_1 + a_n)}{2}}.\n
			\]
			\n
	\end{description}
	\n
\end{statementbox}
