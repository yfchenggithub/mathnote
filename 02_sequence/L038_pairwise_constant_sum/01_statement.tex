\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（前 \(n\) 项对称等和）：}}]
		      对于数列的前 \(n\) 项，存在常数 \(C\)，使得对任意
		      \(k=1,2,\dots,n\)，都有
		      \[
			      a_k+a_{n+1-k}=C.
		      \]
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（倒序求和公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      前 \(n\) 项之和等于 \(n\) 倍常数 \(C\) 的一半。
		      \[
			      S_n=\frac{nC}{2}.
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（等差数列特例）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      等差数列中，对称两项之和恒等于首项与末项之和，因此前 \(n\) 项和等于首项与末项之和乘以项数的一半。
		      \[
			      \boxed{S_n=\frac{n(a_1+a_n)}{2}}.
		      \]
	\end{description}
\end{statementbox}
