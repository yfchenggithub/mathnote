\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（指数写错）}}]
			误写通项为 $a_n = a_1 q^n$，丢失指数减 $1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（指数减一）：}}]
			从首项到第 $n$ 项需乘公比 $n-1$ 次，因此指数固定为 $n-1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆项数与乘次）：}}]
			将项数 $n$ 直接当作乘的次数，忽略了首项本身就已经是 $a_1$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略符号）}}]
			当 $q<0$ 时，项的正负会交替变化，解题时忽略符号导致计算错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（奇偶判断）：}}]
			$a_n$ 的符号由 $a_1$ 的符号和 $q^{n-1}$ 的符号共同决定；当 $q<0$ 时，$q^{n-1}$ 的符号正负交替（$n-1$ 偶正奇负），再结合 $a_1$ 的符号即得 $a_n$ 的符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（负数幂运算不熟）：}}]
			对负数的奇次幂或偶次幂符号判断不清晰，容易直接当正数计算。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（忽视零条件）}}]
			忽视等比数列中 $a_1 \neq 0$ 且 $q \neq 0$，当题目未明确时仍按公式计算，导致分母为零或无意义。
		\item[\textbf{\textcolor{CTrap}{正确理解（前提约束）：}}]
			等比数列定义要求 $a_1 \neq 0$ 且 $q \neq 0$，否则数列退化为零数列或比值无定义。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义条件遗漏）：}}]
			只记住公式形式，忽略公式成立的前提条件，导致乱用公式。
	\end{description}

\end{trapbox}
