\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（等比定义）：}}]
			数列 $\{a_n\}$ 从第 $2$ 项起，每一项与它的前一项的比等于同一个常数 $q$，即 $\dfrac{a_{n+1}}{a_n}=q\;(n\ge 1)$。
		\item[\textbf{\textcolor{CStatement}{条件二（首项非零）：}}]
			首项 $a_1 \neq 0$。
		\item[\textbf{\textcolor{CStatement}{条件三（公比非零）：}}]
			公比 $q \neq 0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（通项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			第 $n$ 项等于首项乘以公比的 $n-1$ 次幂。
			\[
				a_n = a_1 q^{\,n-1} \quad (n \in \mathbb{N}^*)
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（两项关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任意两项之间可以通过公比直接联系起来，指数等于下标差。
			\[
				a_n = a_m q^{\,n-m} \quad (n,m \in \mathbb{N}^*)
			\]
	\end{description}

\end{statementbox}
