\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{等比数列 $\{a_n\}$ 的通项公式为 $a_n = a_1 q^{\,n-1}$，它把首项、公比和项数直接联系起来。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（等比定义）：} 数列 $\{a_n\}$ 必须是等比数列，即 $a_{n+1} / a_n = q$（常数）。
		\item \textbf{条件2（首项非零）：} 首项 $a_1 \neq 0$。
		\item \textbf{条件3（公比非零）：} 公比 $q \neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（指数减一）：} 指数恒为 $n-1$，不是 $n$。
		\item \textbf{检查项（公比符号）：} 当 $q<0$ 时，注意项的正负交替。
	\end{itemize}

\end{summarybox}
