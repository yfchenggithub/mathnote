\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{不动点法：将分式递推转换为等差或等比数列求解。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（分式结构）：} 递推为 $a_{n+1}=\dfrac{Aa_n+B}{Ca_n+D}$，其中 $C\neq0$，$AD-BC\neq0$。
		\item \textbf{条件2（不动点定义）：} 解 $x=\dfrac{Ax+B}{Cx+D}$ 得不动点 $\alpha,\beta$（实数或复数）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（根的分类）：} 根据 $\alpha$ 与 $\beta$ 是否相等选择方法：不同根时构造 $\dfrac{a_n-\alpha}{a_n-\beta}$，重根时构造 $\dfrac{1}{a_n-\alpha}$。
		\item \textbf{检查项（分母非零）：} 确保递推过程中 $C a_n+D\neq0$，且不同根时 $A-\alpha C\neq0,A-\beta C\neq0$，否则公式失效。
	\end{itemize}

\end{summarybox}
