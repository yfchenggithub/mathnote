\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（递推形式）：}}]
			数列 $\{a_n\}$ 满足 $a_{n+1}=\frac{A a_n+B}{C a_n+D}$，其中 $C\neq0$，$AD-BC\neq0$。
		\item[\textbf{\textcolor{CStatement}{条件2（参数定义）：}}]
			设 $x=\frac{Ax+B}{Cx+D}$ 的不动点（即方程 $x=\frac{Ax+B}{Cx+D}$ 的根）为 $\alpha,\beta$（实数或复数）。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两不同根）：}}]
			当 $\alpha\neq\beta$ 且 $A-\alpha C\neq0$，$A-\beta C\neq0$ 时，数列 $\left\{\frac{a_n-\alpha}{a_n-\beta}\right\}$ 构成等比数列。\textbf{\textcolor{CStatement}{直观描述：}} 通过构造比例数列，将分式递推转化为等比递推。
			\[
				\frac{a_{n+1}-\alpha}{a_{n+1}-\beta}=\frac{A-\alpha C}{A-\beta C}\cdot\frac{a_n-\alpha}{a_n-\beta}.
			\]
		\item[\textbf{\textcolor{CStatement}{结论二（重根情形）：}}]
			当 $\alpha=\beta$ 时，数列 $\left\{\frac{1}{a_n-\alpha}\right\}$ 构成等差数列。\textbf{\textcolor{CStatement}{直观描述：}} 通过构造倒数数列，将分式递推转化为等差递推。
			\[
				\frac{1}{a_{n+1}-\alpha}=\frac{1}{a_n-\alpha}+\frac{2C}{A+D}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（系数退化）：}}]
			若 $A-\alpha C=0$（即 $A=\alpha C$，可推得 $B=\alpha D$），则等比公式不适用；此时递推式化为 $a_{n+1}=\alpha$（当分母 $C a_n+D\neq0$），数列从第二项起为常数列。\textbf{\textcolor{CStatement}{直观描述：}} 当系数满足特殊关系时，数列退化为常数列。
			\[
				a_{n+1}=\alpha\quad(n\geq 1).
			\]
	\end{description}

\end{statementbox}
