\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 已知数列 $\{a_n\}$ 满足 $a_{n+1}=\dfrac{3a_n+2}{a_n+4}$，$a_1=0$，求通项公式。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（不动点求解）：}]
			解不动点方程 $x=\dfrac{3x+2}{x+4}$。
			\[
				x^2+4x=3x+2\;\Longrightarrow\; x^2+x-2=0\;\Longrightarrow\; (x-1)(x+2)=0,
			\]
			得 $\alpha=1$，$\beta=-2$。
			\par\textbf{理由：} 找到两个不动点，用于构造辅助数列。
		\item[\textbf{第二步（构造等比）：}]
			计算 $A-\alpha C=3-1=2$，$A-\beta C=3-(-2)=5$，公比为 $\dfrac{2}{5}$。令 $b_n=\dfrac{a_n-1}{a_n+2}$，则
			\[
				b_{n+1}=\frac{2}{5}\,b_n,
				\qquad
				b_1=\frac{0-1}{0+2}=-\frac12.
			\]
			所以 $\displaystyle b_n = -\frac12\left(\frac{2}{5}\right)^{\!n-1}$。
			\par\textbf{理由：} 通过取比消去常数项，构造等比数列。
		\item[\textbf{第三步（反解通项）：}]
			由 $b_n=\dfrac{a_n-1}{a_n+2}$ 反解 $a_n$：
			\[
				\begin{aligned}
					b_n(a_n+2)&=a_n-1 \\ b_n a_n+2b_n &= a_n-1 \\ a_n(b_n-1) &= -1-2b_n \\ a_n &= \frac{1+2b_n}{1-b_n}.
			\end{aligned}
			\]
			代入 $b_n$ 得
			\[
				a_n = \frac{1+2\left(-\frac12\left(\frac{2}{5}\right)^{n-1}\right)}{1-\left(-\frac12\left(\frac{2}{5}\right)^{n-1}\right)} = \frac{1-\left(\frac{2}{5}\right)^{n-1}}{1+\frac12\left(\frac{2}{5}\right)^{n-1}}.
			\]
			\par\textbf{理由：} 由辅助数列返回原数列的代数变形。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n = \dfrac{1-\left(\frac{2}{5}\right)^{n-1}}{1+\frac12\left(\frac{2}{5}\right)^{n-1}}$（$n\ge 1$）}。

	\textbf{\textcolor{CExample}{例 2（重根）}}
	\textbf{\textcolor{CExample}{题目：}} 已知数列 $\{a_n\}$ 满足 $a_{n+1}=\dfrac{a_n}{a_n+1}$，$a_1=1$，求通项公式。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（不动点求解）：}]
			解 $x=\dfrac{x}{x+1}$，得
			\[
				x^2+x=x\;\Longrightarrow\; x^2=0\;\Longrightarrow\; \alpha=0\ (\text{重根}).
			\]
			此时 $A=1,B=0,C=1,D=1$，$A+D=2\neq0$。
			\par\textbf{理由：} 确认属于重根情形，且条件满足。
		\item[\textbf{第二步（构造等差）：}]
			计算公差 $d=\dfrac{2C}{A+D}=\dfrac{2}{2}=1$。令 $c_n=\dfrac{1}{a_n-\alpha}=\dfrac{1}{a_n}$，则
			\[
				c_{n+1}=c_n+1,
				\qquad
				c_1=\frac{1}{1}=1.
			\]
			所以 $c_n=1+(n-1)\cdot1=n$。
			\par\textbf{理由：} 重根时构造倒数得到等差数列。
		\item[\textbf{第三步（反解通项）：}]
			由 $c_n=n$ 直接得 $a_n=\dfrac{1}{n}$。
			\par\textbf{理由：} 倒数关系的逆运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n = \dfrac{1}{n}$（$n\ge 1$）}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 已知数列 $\{a_n\}$ 满足 $a_{n+1}=\dfrac{2}{3-a_n}$，$a_1=0$，求通项公式。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（不动点求解）：}]
			解 $x=\dfrac{2}{3-x}$，得
			\[
				x(3-x)=2\;\Longrightarrow\; -x^2+3x-2=0\;\Longrightarrow\; x^2-3x+2=0\;\Longrightarrow\; (x-1)(x-2)=0,
			\]
			得 $\alpha=1$，$\beta=2$。
			\par\textbf{理由：} 找到两个不动点。
		\item[\textbf{第二步（构造等比）：}]
			将递推写为标准形式 $a_{n+1}=\dfrac{0\cdot a_n+2}{-1\cdot a_n+3}$，即 $A=0,B=2,C=-1,D=3$。计算
			\[
				A-\alpha C = 0-1\cdot(-1)=1,
				\qquad
				A-\beta C = 0-2\cdot(-1)=2,
				\qquad\text{公比 } \frac12.
			\]
			令 $b_n=\dfrac{a_n-1}{a_n-2}$，则
			\[
				b_{n+1}=\frac12\,b_n,
				\qquad
				b_1=\frac{0-1}{0-2}=\frac12.
			\]
			所以 $\displaystyle b_n = \frac12\left(\frac12\right)^{\!n-1} = \left(\frac12\right)^{\!n} = 2^{-n}$。
			\par\textbf{理由：} 构造等比数列。
		\item[\textbf{第三步（反解通项）：}]
			由 $b_n=\dfrac{a_n-1}{a_n-2}$ 解 $a_n$：
			\[
				\begin{aligned}
					b_n(a_n-2) &= a_n-1 \\ b_n a_n -2b_n &= a_n-1 \\ (b_n-1)a_n &= 2b_n-1 \\ a_n &= \frac{2b_n-1}{b_n-1} = \frac{1-2b_n}{1-b_n}.
			\end{aligned}
			\]
			代入 $b_n=2^{-n}$：
			\[
				a_n = \frac{1-2\cdot2^{-n}}{1-2^{-n}} = \frac{1-2^{1-n}}{1-2^{-n}}.
			\]
			分子分母同乘 $2^n$ 得 $a_n = \dfrac{2^n-2}{2^n-1}$。
			\par\textbf{理由：} 从辅助数列返回原数列，并化简。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n = \dfrac{2^n-2}{2^n-1}$（$n\ge 1$）}。
\end{examplebox}
