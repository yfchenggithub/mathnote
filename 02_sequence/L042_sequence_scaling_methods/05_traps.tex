\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（裂项方向混淆）：}}]
		      使用裂项放缩时，不等号方向用反。
		\item[\textbf{\textcolor{CTrap}{正确理解（注意目标方向）：}}]
		      要证 $\sum a_n<C$，通常要把 $a_n$ 放大为 $b_n$，即 $a_n\le b_n$，再证明 $\sum b_n<C$；要证 $\sum a_n>C$，通常要把 $a_n$ 缩小。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略目标）：}}]
		      学生常不检查目标方向，只看到能裂项就套用，导致放缩方向错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（等比条件遗忘）：}}]
		      忽略 $0<q<1$ 的条件，或把有限等比和直接写成无穷等比和。
		\item[\textbf{\textcolor{CTrap}{正确理解（公比与常数）：}}]
		      等比放缩必须确保 $0<q<1$，且存在常数 $C>0$ 使 $a_n\le Cq^n$。求和时先写有限和
		      \[
			      \sum_{k=1}^n q^k=\frac{q(1-q^n)}{1-q},
		      \]
		      再根据需要放大为 $\frac{q}{1-q}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件缺失）：}}]
		      学生忘记检验 $q$ 的范围，或者直接使用无穷项和，忽略了有限和中的 $q^n$ 项。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（糖水连乘条件遗漏）：}}]
		      在连乘中，有的因子不满足分子小于分母，却仍然直接套用糖水放缩。
		\item[\textbf{\textcolor{CTrap}{正确理解（每项条件一致）：}}]
		      使用糖水放缩累乘时，必须保证每一项都满足 $0<a_k<b_k$，且添加的是正数。这样每一项的不等号方向一致，累乘后方向才不会错。
		\item[\textbf{\textcolor{CTrap}{错因分析（盲目累乘）：}}]
		      学生未逐项检查条件，只凭形式相似就套用，容易造成方向错误或结论不成立。
	\end{description}
\end{trapbox}
