\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{a_n\}$ 满足 $a_1=2$，$a_{n+1}=2a_n+3$，求 $\{a_n\}$ 的通项公式。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（确定$\lambda$值）：}]
			由递推 $a_{n+1}=2a_n+3$ 知 $p=2$，$q=3$。代入公式 $\lambda=\dfrac{q}{1-p}$ 得 $\lambda=\dfrac{3}{1-2}$，即 $\lambda=-3$。
			\par\textbf{理由：} $\lambda$ 是使新数列等比的平移常数，由待定系数法确定。
		\item[\textbf{第二步（构造新数列）：}]
			令 $b_n=a_n+3$，则 $a_{n+1}+3=2(a_n+3)$，即 $b_{n+1}=2b_n$。
			\par\textbf{理由：} 代入验证可得新数列 $\{b_n\}$ 是公比为 $2$ 的等比数列。
		\item[\textbf{第三步（回代求通项）：}]
			由 $b_n$ 等比得
			\[
				\begin{aligned}
					b_n &= b_1\cdot2^{n-1} \\ &= (a_1+3)\cdot2^{n-1} \\ &= 5\cdot2^{n-1}.
			\end{aligned}
			\]
			于是
			\[
				\begin{aligned}
					a_n &= b_n-3 \\ &= 5\cdot2^{n-1}-3.
			\end{aligned}
			\]
			\par\textbf{理由：} 先求 $b_n$ 再减 $\lambda$ 得 $a_n$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n=5\cdot2^{n-1}-3$}。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{b_n\}$ 满足 $b_1=1$，$b_{n+1}=\dfrac12 b_n+2$，求通项公式。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（计算$\lambda$）：}]
			$p=\dfrac12$，$q=2$。计算 $\lambda=\dfrac{q}{1-p}$，得 $\lambda=\dfrac{2}{1-\frac12}$，即 $\lambda=4$。
			\par\textbf{理由：} 注意分母为 $1-p$，分数运算要准确。
		\item[\textbf{第二步（构造等比）：}]
			令 $c_n=b_n-4$，则
			\[
				\begin{aligned}
					c_{n+1} &= b_{n+1}-4 \\ &= \frac12 b_n+2-4 \\ &= \frac12 b_n-2 \\ &= \frac12(b_n-4) \\ &= \frac12 c_n.
			\end{aligned}
			\]
			\par\textbf{理由：} 新数列 $\{c_n\}$ 是公比为 $\dfrac12$ 的等比数列。
		\item[\textbf{第三步（求通项）：}]
			$c_n$ 等比得
			\[
				\begin{aligned}
					c_n &= c_1\left(\dfrac12\right)^{n-1} \\ &= (1-4)\cdot2^{1-n} \\ &= -3\cdot2^{1-n}.
			\end{aligned}
			\]
			故
			\[
				\begin{aligned}
					b_n &= c_n+4 \\ &= 4-3\cdot2^{1-n}.
			\end{aligned}
			\]
			\par\textbf{理由：} 先算 $c_n$ 再加回 $\lambda$ 得 $b_n$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$b_n=4-3\cdot2^{1-n}$}。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{a_n\}$ 满足 $a_1=2$，$a_{n+1}=3a_n^2$（$a_n>0$），求通项公式。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{第一步（取对数降次）：}]
			对 $a_{n+1}=3a_n^2$ 两边取自然对数，得 $\ln a_{n+1} = \ln3 + 2\ln a_n$。
			\par\textbf{理由：} 利用对数性质将指数 $2$ 化为系数，使递推线性化。
		\item[\textbf{第二步（变量代换线性化）：}]
			令 $b_n = \ln a_n$，则 $b_{n+1} = 2b_n + \ln3$。
			\par\textbf{理由：} 得到形如 $b_{n+1}=p b_n+q$ 的一阶线性递推。
		\item[\textbf{第三步（构造等比求通项）：}]
			对线性递推 $b_{n+1}=2b_n+\ln3$，采用构造法。取 $\mu = \dfrac{\ln3}{1-2} = -\ln3$，则 $b_{n+1}+\ln3 = 2(b_n+\ln3)$。故 $\{b_n+\ln3\}$ 是公比为 $2$ 的等比数列，首项 $b_1+\ln3 = \ln6$。于是
			\[
				\begin{aligned}
					b_n+\ln3 &= \ln6\cdot2^{n-1} \\ b_n &= \ln6\cdot2^{n-1} - \ln3 \\ &= \ln\frac{6^{2^{n-1}}}{3}.
			\end{aligned}
			\]
			从而
			\[
				\begin{aligned}
					a_n &= e^{b_n} \\ &= \frac{6^{2^{n-1}}}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 先通过等比求 $b_n$，再取指数还原为 $a_n$。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n = \dfrac{6^{2^{n-1}}}{3}$}。
\end{examplebox}
