\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{构造辅助数列法：通过平移（线性型）或取对数（幂指型）将递推关系转化为等比数列，进而求得通项。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*,itemsep=0.5em,topsep=0.4em]
		\item \textbf{条件 1（线性递推识别）：} 递推形式为 $a_{n+1}=p a_n+q$，且 $p\neq1$。
		\item \textbf{条件 2（幂指递推识别）：} 递推形式为 $a_{n+1}=c a_n^k$，且 $c>0$，$a_n>0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*,itemsep=0.5em,topsep=0.4em]
		\item \textbf{易错点（检查p=1）：} 使用线性构造法前必须确认 $p\neq1$，否则直接使用等差数列。
		\item \textbf{检查项（验证定义域）：} 对幂指递推，使用对数前必须确认 $a_n>0$。
	\end{itemize}

\end{summarybox}
