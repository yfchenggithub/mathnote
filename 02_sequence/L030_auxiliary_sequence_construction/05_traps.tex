\begin{trapbox}

	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略p=1情形）：}}]
			当 $p=1$ 时仍使用 $\lambda=\frac{q}{1-p}$ 公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（p=1为等差）：}}]
			$p=1$ 时 $\{a_n\}$ 是公差为 $q$ 的等差数列，直接求通项。
		\item[\textbf{\textcolor{CTrap}{错因分析（未检查条件）：}}]
			机械套用公式，未检验 $p\neq1$ 的适用前提。
	\end{description}

	\medskip
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（符号记反）：}}]
			将 $\lambda$ 错误记为 $\lambda=\frac{q}{p-1}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（分母1-p）：}}]
			$\lambda=\frac{q}{1-p}$，由 $\lambda(1-p)=q$ 解出。
		\item[\textbf{\textcolor{CTrap}{错因分析（解方程符号错）：}}]
			从 $a_{n+1}=p a_n+q$ 与 $a_{n+1}=p a_n+\lambda(1-p)$ 对比时，$\lambda(1-p)=q$ 移项未注意符号。
	\end{description}

	\medskip
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（对数定义域遗漏）：}}]
			对 $a_{n+1}=c a_n^k$ 直接取对数，未验证 $a_n>0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正数前提）：}}]
			使用对数转化必须 $a_n>0$，通常题目会明确条件。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视对数要求）：}}]
			忽略对数函数的定义域，或未注意 $c>0,a_n>0$ 的前提。
	\end{description}

\end{trapbox}
