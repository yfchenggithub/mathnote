\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（线性递推）：}}]
			数列 $\{a_n\}$ 满足 $a_{n+1}=p a_n+q$，其中 $p\neq 1$。
		\item[\textbf{\textcolor{CStatement}{条件 2（幂指递推）：}}]
			数列 $\{a_n\}$ 满足 $a_{n+1}=c a_n^k$，其中 $c>0$，$a_n>0$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（线性构造法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通过平移消去常数项，化为等比数列。
			\[
				\begin{aligned}
					\lambda &= \frac{q}{1-p}, \\ a_{n+1}-\lambda &= p(a_n-\lambda).
			\end{aligned}
			\]
			从而数列 $\{a_n-\lambda\}$ 是首项为 $a_1-\lambda$、公比为 $p$ 的等比数列。
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{推广结论（对数转化法）：}}
	\textbf{\textcolor{CStatement}{直观描述：}}
	幂指递推取对数后转化为线性递推，再用构造法。
	\[
		\begin{aligned}
			a_{n+1} &= c a_n^k, \\ \ln a_{n+1} &= \ln c + k\ln a_n.
		\end{aligned}
	\]
	令 $b_n=\ln a_n$，得 $b_{n+1}=k b_n+\ln c$。当 $k\neq 1$ 时，取 $\mu=\dfrac{\ln c}{1-k}$，则 $\{b_n-\mu\}$ 是公比为 $k$ 的等比数列。
\end{statementbox}
