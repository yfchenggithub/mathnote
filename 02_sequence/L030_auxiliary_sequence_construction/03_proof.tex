\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}
	构造法的核心是待定系数与对数化归。先通过假设等比形式确定$\lambda$，再验证；幂指型通过对数转化为线性，复用同一方法。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline,leftmargin=2.8em,labelsep=0.5em,itemsep=0.55em]
		\item[\textbf{步骤一（设定目标形式）：}]
			假设存在常数$\lambda$使得$a_{n+1}-\lambda=p(a_n-\lambda)$。这是构造等比数列的基本形式。
			\[
				a_{n+1}-\lambda=p(a_n-\lambda).
			\]
			\par\textbf{理由：} 若$\{a_n-\lambda\}$是等比，则一定满足此式；反过来，若此式成立，则新数列等比。
		\item[\textbf{步骤二（展开对比系数）：}]
			将假设式右边展开，并与原递推比较。
			\[
				\begin{aligned}
					a_{n+1}-\lambda &= p a_n - p\lambda, \\ a_{n+1} &= p a_n + \lambda(1-p).
			\end{aligned}
			\]
			与原递推$a_{n+1}=p a_n+q$对比得$\lambda(1-p)=q$。
			\par\textbf{理由：} 比较系数得到关于$\lambda$的方程，解出$\lambda=\dfrac{q}{1-p}\ (p\neq1)$。
		\item[\textbf{步骤三（验证等比成立）：}]
			将$\lambda$代回原递推，验证恒成立。
			\[
				\begin{aligned}
					a_{n+1}-\lambda &= p a_n+q-\lambda \\ &= p a_n+q-\frac{q}{1-p} \\ &= p a_n-\frac{pq}{1-p} \\ &= p\left(a_n-\frac{q}{1-p}\right) \\ &= p(a_n-\lambda).
			\end{aligned}
			\]
			\par\textbf{理由：} 上式表明$a_{n+1}-\lambda$与$a_n-\lambda$确为等比关系，故$\{a_n-\lambda\}$是首项$a_1-\lambda$、公比$p$的等比数列。
		\item[\textbf{步骤四（幂指递推取对数）：}]
			对$a_{n+1}=c a_n^k\ (c>0,a_n>0)$两边取自然对数。
			\[
				\begin{aligned}
					\ln a_{n+1} &= \ln(c a_n^k) \\ &= \ln c + k\ln a_n.
			\end{aligned}
			\]
			\par\textbf{理由：} 对数运算将乘法化为加法，指数化为系数，使幂指形式线性化。
		\item[\textbf{步骤五（变量代换线性化）：}]
			令$b_n=\ln a_n$，则递推变为$b_{n+1}=k b_n+\ln c$。
			\[
				b_{n+1}=k b_n+\ln c.
			\]
			\par\textbf{理由：} 这是典型的一阶线性递推，可套用线性构造法。
		\item[\textbf{步骤六（构造等比求通项）：}]
			当$k\neq1$时，对线性递推$b_{n+1}=k b_n+\ln c$重复步骤一至三：取$\mu=\dfrac{\ln c}{1-k}$，则$b_{n+1}-\mu=k(b_n-\mu)$。
			\[
				\begin{aligned}
					\mu &= \frac{\ln c}{1-k}, \\ b_{n+1}-\mu &= k(b_n-\mu).
			\end{aligned}
			\]
			\par\textbf{理由：} 从而$\{b_n-\mu\}$是公比为$k$的等比数列，求得$b_n$后回代$a_n=\mathrm{e}^{b_n}$即得$a_n$通项。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}
	通过待定系数直接构造等比（线性型）或取对数后构造（幂指型），核心思路都是将陌生的递推转化为标准的等比数列，从而求得通项。
\end{proofbox}
