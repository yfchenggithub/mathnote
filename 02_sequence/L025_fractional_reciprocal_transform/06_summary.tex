\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{对于分子无常数项的分式线性递推，取倒数即得一阶线性递推，可继续求通项。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（递推形式匹配）：} 递推必须为 $a_{n+1}=\dfrac{f(n)a_n}{g(n)a_n+h(n)}$，分子只有 $a_n$ 的一次项，没有常数项。
		\item \textbf{条件2（非零条件保证）：} 对任意 $n$，$a_n\neq 0$，$f(n)\neq 0$，$h(n)\neq 0$，且分母 $g(n)a_n+h(n)\neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分子带常数不要套）：} 若递推分子含常数项（如 $a_{n+1}=\frac{p a_n+q}{r a_n+s}$），则不能直接使用此变换，需要先做其他处理。
		\item \textbf{检查项（验证每一步非零）：} 使用前务必验证 $a_n$ 全非零，否则需要分段处理；化简时确认分母不为零。
	\end{itemize}
\end{summarybox}
