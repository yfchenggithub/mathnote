\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（分式线性递推）：}}] 数列 $\{a_n\}$ 满足递推关系
			\[
				a_{n+1} = \frac{f(n)\,a_n}{g(n)\,a_n + h(n)},
			\]
			其中 $f(n),g(n),h(n)$ 是关于 $n$ 的已知函数，且对任意 $n$ 有 $f(n)\neq 0$, $h(n)\neq 0$, $a_n\neq 0$，分母 $g(n)a_n+h(n)\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（倒数变换化线性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			令 $b_n = 1/a_n$，则 $\{b_n\}$ 满足一阶线性递推。
			\[
				b_{n+1} = \frac{h(n)}{f(n)}\,b_n + \frac{g(n)}{f(n)}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（常系数特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $f(n),g(n),h(n)$ 均为常数 $p,q,r$ 时，递推化为
			\[
				b_{n+1} = \frac{r}{p}\,b_n + \frac{q}{p}.
			\]
	\end{description}

\end{statementbox}
