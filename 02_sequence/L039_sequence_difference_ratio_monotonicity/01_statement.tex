\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（数列与指标）：}}]
			设$\{a_n\}$是数列，$n\in\mathbb{N}^*$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（差比法判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			相邻两项之差的正负直接决定单调性。
			\[
				a_{n+1}-a_n\
				\begin{cases}
					>0&\iff\{a_n\}\text{严格递增},\\=0&\iff\{a_n\}\text{为常数列},\\<0&\iff\{a_n\}\text{严格递减}.
			\end{cases}
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（商比法判定）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当各项为正时，相邻两项之比与$1$比较等价于单调性。
			\[
				\frac{a_{n+1}}{a_n}\
				\begin{cases}
					>1&\iff\{a_n\}\text{严格递增},\\=1&\iff\{a_n\}\text{为常数列},\\<1&\iff\{a_n\}\text{严格递减}.
			\end{cases}
			\]
	\end{description}

	\medskip
	\textbf{\textcolor{CStatement}{使用提醒（条件限制）：}}
	商比法仅当所有$a_n>0$时才能使用；若出现$a_n\leq 0$，必须改用差比法。
\end{statementbox}
