\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}已知数列$a_n = 3n+2$，判断其单调性。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作差计算）：}]
			计算差：
			\[
				\begin{aligned}
					a_{n+1}-a_n &= [3(n+1)+2] - (3n+2) \\ &= 3.
			\end{aligned}
			\]
			\par\textbf{理由：}直接代入通项公式。
		\item[\textbf{第二步（符号判断）：}]
			对任意$n\in\mathbb{N}^*$，$a_{n+1}-a_n = 3 > 0$。
			\par\textbf{理由：}差为正常数。
		\item[\textbf{第三步（得出结论）：}]
			由差比法，$a_{n+1}-a_n>0$等价于数列严格递增，故$\{a_n\}$严格递增。
			\par\textbf{理由：}差比法结论（条件满足）。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	差比法直接得到$a_{n+1}-a_n>0$从而判定严格递增。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}已知数列$a_n = 2n^2+1$，试判断其单调性。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作商构造）：}]
			显然$a_n>0$，作商$\frac{a_{n+1}}{a_n} = \frac{2(n+1)^2+1}{2n^2+1}$。
			\par\textbf{理由：}正项数列可用商比法。
		\item[\textbf{第二步（化简比较）：}]
			化简：
			\[
				\begin{aligned}
					\frac{a_{n+1}}{a_n}&= \frac{2n^2+4n+3}{2n^2+1} \\ &= 1 + \frac{4n+2}{2n^2+1} \\ &> 1.
			\end{aligned}
			\]
			\par\textbf{理由：}分子分母展开，分离出$1$，剩余分式恒正。
		\item[\textbf{第三步（得出结论）：}]
			由商比法，$\frac{a_{n+1}}{a_n}>1$等价于严格递增，故$\{a_n\}$严格递增。
			\par\textbf{理由：}商比法结论成立。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	商比法避免了减法，分式化简后直接比较与1的大小。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}已知数列$a_n = n^2 + \lambda n$，若$\{a_n\}$对一切$n\in\mathbb{N}^*$严格递增，求实数$\lambda$的取值范围。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（作差化简）：}]
			计算差：
			\[
				\begin{aligned}
					a_{n+1}-a_n &= [(n+1)^2+\lambda(n+1)] - (n^2+\lambda n) \\ &= 2n+1+\lambda.
			\end{aligned}
			\]
			\par\textbf{理由：}展开并合并同类项。
		\item[\textbf{第二步（建立条件）：}]
			严格递增要求$a_{n+1}-a_n > 0$对所有$n\in\mathbb{N}^*$成立，即$2n+1+\lambda > 0$恒成立。
			\par\textbf{理由：}差比法结论。
		\item[\textbf{第三步（求参数范围）：}]
			$2n+1+\lambda > 0$对$n\ge 1$恒成立，只需$n=1$时成立（因函数递增），即$2\times 1+1+\lambda >0$，解得$\lambda > -3$。
			\par\textbf{理由：}求一次函数最小值在$n=1$处。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	差比法将单调性条件转化为关于$n$的一次不等式，进一步化为参数不等式求范围。
\end{examplebox}
