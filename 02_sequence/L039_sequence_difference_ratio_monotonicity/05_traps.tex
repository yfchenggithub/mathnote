\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略正项条件）}}]
			直接对所有数列使用商比法$\frac{a_{n+1}}{a_n}>1$判定递增，而不检查各项是否为正。
		\item[\textbf{\textcolor{CTrap}{正确理解（商比前提）：}}]
			商比法仅当所有$a_n>0$时才等价；若$a_n<0$，$\frac{a_{n+1}}{a_n}>1$反而对应递减。
		\item[\textbf{\textcolor{CTrap}{错因分析（正数保号）：}}]
			忽略了除法保号性依赖于分母的正负，负分母会反转不等号方向。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（严格与不严格）}}]
			认为$a_{n+1}-a_n\ge 0$就是严格递增，或者$a_{n+1}-a_n>0$是“递增”（可能非严格）。
		\item[\textbf{\textcolor{CTrap}{正确理解（严格定义）：}}]
			严格递增必须$a_{n+1}-a_n>0$；若出现等于0，则不是严格递增（可能为常数列或摆动）。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义混淆）：}}]
			混淆了“递增（非严格）”与“严格递增”的区别，或只记住定义大意。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（局部代替整体）}}]
			只验证前几项$a_{n+1}-a_n>0$就断言数列严格递增。
		\item[\textbf{\textcolor{CTrap}{正确理解（必须全部）：}}]
			必须对所有$n\in\mathbb{N}^*$都成立$a_{n+1}-a_n>0$，才能判定严格递增；前几项成立不足以推出全部。
		\item[\textbf{\textcolor{CTrap}{错因分析（归纳错误）：}}]
			错误使用不完全归纳，或误以为数列具有“传递性”前项成立后项自动成立。
	\end{description}

\end{trapbox}
