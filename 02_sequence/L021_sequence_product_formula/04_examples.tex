\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$a_1=2$，$a_{n+1}=3a_n$（$n\in\mathbb{N}^*$），求通项$a_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别类型）：}] 递推式即$a_{n+1}=a_n\cdot3$，故$f(n)=3$。
			\par\textbf{理由：} 符合累乘法的标准形式$a_{n+1}=a_n f(n)$。
		\item[\textbf{第二步（套用公式）：}] 由累乘公式得
			\[
				\begin{aligned}
					a_n &= a_1 \prod_{k=1}^{n-1} f(k) \\
					&= 2 \cdot \prod_{k=1}^{n-1} 3 \\
					&= 2 \cdot 3^{\,n-1} \quad (n\ge 2) .
			\end{aligned}
			\]
			\par\textbf{理由：} 代入累乘公式。
		\item[\textbf{第三步（验证首项）：}] 当$n=1$时，$2= a_1$，公式对$n=1$也成立。
			\par\textbf{理由：} 验证初始项。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n=2\cdot3^{\,n-1}$（$n\in\mathbb{N}^*$）}。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$a_1=1$，$a_{n+1}=2^n a_n$（$n\in\mathbb{N}^*$），求通项$a_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别形式）：}] $f(n)=2^n$。
			\par\textbf{理由：} 对照$a_{n+1}=a_n f(n)$。
		\item[\textbf{第二步（套用公式）：}] 由累乘公式得
			\[
				\begin{aligned}
					a_n &= 1 \cdot \prod_{k=1}^{n-1} 2^{\,k} \\
					&= 2^{\sum_{k=1}^{n-1} k} .
			\end{aligned}
			\]
			\par\textbf{理由：} 代入累乘公式并合并指数。
		\item[\textbf{第三步（求和化简）：}] $\sum_{k=1}^{n-1}k=\frac{n(n-1)}{2}$，故$a_n=2^{\frac{n(n-1)}{2}}$（$n\ge2$），$n=1$时亦成立。
			\par\textbf{理由：} 等差数列求和。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n=2^{\frac{n(n-1)}{2}}$（$n\in\mathbb{N}^*$）}。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知$a_1=1$，$a_{n+1}=\frac{n+1}{n}a_n$（$n\in\mathbb{N}^*$），求通项$a_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别形式）：}] $f(n)=\frac{n+1}{n}$。
			\par\textbf{理由：} 写成$a_{n+1}=a_n\cdot\frac{n+1}{n}$。
		\item[\textbf{第二步（套用公式）：}] $a_n=1\cdot\prod_{k=1}^{n-1}\frac{k+1}{k}$。
			\par\textbf{理由：} 代入累乘公式。
		\item[\textbf{第三步（化简求值）：}] $\prod_{k=1}^{n-1}\frac{k+1}{k}=n$，故$a_n=n$（$n\ge2$），$n=1$时$a_1=1$也满足。
			\par\textbf{理由：} 相邻项分子分母约简。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n=n$（$n\in\mathbb{N}^*$）}。

\end{examplebox}
