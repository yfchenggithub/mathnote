\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	写出比值再连乘。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（比值式）：}] 由$a_{n+1}=a_n f(n)$得$\frac{a_{n+1}}{a_n}=f(n)$。
			\par\textbf{理由：} 两边除以$a_n\neq0$。
		\item[\textbf{步骤二（列等式）：}] 取$n=1,\dots,n-1$得$\frac{a_2}{a_1}=f(1),\dots,\frac{a_n}{a_{n-1}}=f(n-1)$。
			\par\textbf{理由：} 代入$n$值。
		\item[\textbf{步骤三（连乘消元）：}] 连乘得$\frac{a_2}{a_1}\cdots\frac{a_n}{a_{n-1}}=f(1)\cdots f(n-1)$，约简得$\frac{a_n}{a_1}=\prod_{k=1}^{n-1}f(k)$。
			\par\textbf{理由：} 相邻项抵消。
		\item[\textbf{步骤四（得公式）：}] 两边乘$a_1$得$a_n=a_1\prod_{k=1}^{n-1}f(k)$（$n\ge2$）。
			\par\textbf{理由：} 乘法。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{得到累乘公式。}
\end{proofbox}
