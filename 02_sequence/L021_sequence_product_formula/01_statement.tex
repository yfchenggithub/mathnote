\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（递推与定义）：}}] 数列$\{a_n\}$满足递推关系$a_{n+1}=a_n\cdot f(n)$（$n\in\mathbb{N}^*$），其中$f(n)$是定义在$\mathbb{N}^*$上的函数。
		\item[\textbf{\textcolor{CStatement}{条件2（非零条件）：}}] 对所有$n\in\mathbb{N}^*$，有$a_n\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（累乘公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			通项等于首项乘以相邻项之比的累乘积。
			\[
				a_n = a_1\prod_{k=1}^{n-1}f(k),
				\qquad
				n\geq 2.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（特例等比）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$f(n)$恒为常数$q$时，累乘法退化为等比数列公式。
			\[
				a_n = a_1 q^{\,n-1}.
			\]
	\end{description}
\end{statementbox}
