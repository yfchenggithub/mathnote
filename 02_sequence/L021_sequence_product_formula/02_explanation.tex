\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}

	累乘法是连乘相邻项比值求通项。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（递推比值）：}] 得$\frac{a_{n+1}}{a_n}=f(n)$。
		\item[\textbf{要点二（累乘消元）：}] 连乘消去中间项得$\frac{a_n}{a_1}=\prod_{k=1}^{n-1}f(k)$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（推广等比）}}

	等比数列推广，公比变为$f(n)$。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（连乘表示）}}

	通项用连乘积表示。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（命题角度）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（直接套用）：} 已知$a_1$和$f(n)$，直接计算$a_1\prod f(k)$。
		\item \textbf{考法二（变形转化）：} 形如$a_{n+1}=g(n)a_n$可直接用。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}

	\textbf{关键是乘积下标范围。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（显式比值）：} 已知$\frac{a_{n+1}}{a_n}=f(n)$。
		\item \textbf{场景二（隐含比值）：} 已知$a_{n+1}=g(n)a_n$。
	\end{itemize}
\end{explanationbox}
