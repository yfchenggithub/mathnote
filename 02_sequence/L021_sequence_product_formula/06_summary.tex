\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{累乘法是将递推式转化为比值连乘形式，直接求出通项公式的方法，使用前提是各项均不为零。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（递推形式）：} $a_{n+1}=a_n\cdot f(n)$，其中 $f(n)$ 是 $\mathbb{N}^*$ 上的函数。
		\item \textbf{条件2（非零条件）：} 对所有 $n\in\mathbb{N}^*$，有 $a_n\neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（连乘范围）：} 连乘积从 $k=1$ 到 $n-1$，共 $n-1$ 个因子，切勿多乘或少乘。
		\item \textbf{检查项（初始验证）：} 公式只对 $n\ge 2$ 保证成立，$n=1$ 时需单独验证或规定空积为 $1$。
	\end{itemize}
\end{summarybox}
