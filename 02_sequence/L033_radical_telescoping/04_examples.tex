\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	计算 $\dfrac{1}{\sqrt{5}+\sqrt{3}}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 认出分母为 $\sqrt{5}+\sqrt{3}$，对应公式中 $n=3$，$k=2$，代入得 $\dfrac{\sqrt{5}-\sqrt{3}}{2}$。
			\par\textbf{理由：} 分母有理化公式 $\dfrac{1}{\sqrt{n+k}+\sqrt{n}}=\dfrac{\sqrt{n+k}-\sqrt{n}}{k}$。
		\item[\textbf{第二步（化简结果）：}] 结果已为最简，无需进一步化简。
			\par\textbf{理由：} 分子分母无公因式。
		\item[\textbf{第三步（写出答案）：}] 最终答案为 $\dfrac{\sqrt{5}-\sqrt{3}}{2}$。
			\par\textbf{理由：} 公式直接给出。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$\displaystyle\frac{1}{\sqrt{5}+\sqrt{3}}=\frac{\sqrt{5}-\sqrt{3}}{2}$}

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{a_n\}$ 的通项 $a_n=\dfrac{1}{\sqrt{n+1}+\sqrt{n}}$，求其前10项的和 $S_{10}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（裂项）：}] 对 $a_n$ 使用 $k=1$ 的公式：$a_n = \sqrt{n+1}-\sqrt{n}$。
			\par\textbf{理由：} 直接代入 $\dfrac{1}{\sqrt{n+1}+\sqrt{n}}=\sqrt{n+1}-\sqrt{n}$。
		\item[\textbf{第二步（展开求和）：}] $S_{10}=(\sqrt{2}-\sqrt{1})+(\sqrt{3}-\sqrt{2})+\cdots+(\sqrt{11}-\sqrt{10})$。
			\par\textbf{理由：} 将 $n=1$ 至 $10$ 的表达式依次写出。
		\item[\textbf{第三步（逐项相消）：}] 相邻项正负相消，得 $S_{10}=\sqrt{11}-1$。
			\par\textbf{理由：} 除首尾项外，其余全部抵消。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_{10}=\sqrt{11}-1$}

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	证明：
	\[
		\frac{1}{\sqrt{2}+1}+\frac{1}{\sqrt{3}+\sqrt{2}}+\cdots+\frac{1}{\sqrt{n+1}+\sqrt{n}}=\sqrt{n+1}-1.
	\]

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（裂每一项）：}] 由公式（$k=1$），$\dfrac{1}{\sqrt{k+1}+\sqrt{k}}=\sqrt{k+1}-\sqrt{k}$，$k=1,2,\cdots,n$。
			\par\textbf{理由：} 恒等变形。
		\item[\textbf{第二步（求和相消）：}] 左式 $=(\sqrt{2}-\sqrt{1})+(\sqrt{3}-\sqrt{2})+\cdots+(\sqrt{n+1}-\sqrt{n})$。
			\par\textbf{理由：} 代入裂项后表达式。
		\item[\textbf{第三步（化简结果）：}] 所有中间项抵消，左式 $=\sqrt{n+1}-1$，右式相同，故等式成立。
			\par\textbf{理由：} 逐项相消，仅剩首项 $-\sqrt{1}$ 与末项 $\sqrt{n+1}$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{等式 $\displaystyle\sum_{k=1}^{n}\frac{1}{\sqrt{k+1}+\sqrt{k}}=\sqrt{n+1}-1$ 成立}
\end{examplebox}
