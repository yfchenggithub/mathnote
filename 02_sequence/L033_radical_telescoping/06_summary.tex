\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{$\dfrac{1}{\sqrt{n+k}+\sqrt{n}}=\dfrac{\sqrt{n+k}-\sqrt{n}}{k}$，将根式分母有理化，为裂项求和提供基础。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（变量范围）：} $k$为正整数，$n\in\mathbb{N}^{*}$。
		\item \textbf{条件2（根式意义）：} $\sqrt{n+k}$与$\sqrt{n}$均有意义，即 $n+k\geq 0$ 与 $n\geq 0$（自然数自动满足）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（漏系数k）：} 公式右侧分母有$k$，不可写成$\sqrt{n+k}-\sqrt{n}$。
		\item \textbf{检查项（共轭选择）：} 分子分母同乘的必须是$\sqrt{n+k}-\sqrt{n}$，而不是加号。
	\end{itemize}
\end{summarybox}
