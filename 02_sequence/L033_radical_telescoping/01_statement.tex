\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（变量范围）：}}] $k$为正整数，$n\in\mathbb{N}^*$，且$\sqrt{n+k}$与$\sqrt{n}$均有意义。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（分母有理化）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将根号分母转化为差形式，便于裂项。
			\[
				\frac{1}{\sqrt{n+k}+\sqrt{n}} = \frac{\sqrt{n+k}-\sqrt{n}}{k}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（相邻根式差）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$k=1$时，公式简化为相邻根式差，常用于数列求和。
			\[
				\frac{1}{\sqrt{n+1}+\sqrt{n}} = \sqrt{n+1}-\sqrt{n}.
			\]
	\end{description}
\end{statementbox}
