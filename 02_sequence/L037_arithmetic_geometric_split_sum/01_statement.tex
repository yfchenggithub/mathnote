\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（通项可拆分）：}}]
			数列$\{c_n\}$的通项可表示为有限个可独立求和数列的代数和。
		\item[\textbf{\textcolor{CStatement}{条件2（等差和等比）：}}]
			设$\{a_n\}$为等差数列，$a_n=An+B$（公差$A$，首项$A+B$）；$\{b_n\}$为等比数列，$b_n=C\cdot q^{\,n-1}$（首项$C$，公比$q\neq1$）；且$c_n=a_n+b_n$。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（分组求和法）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将复杂数列拆解为简单数列分别求和再相加。
			\[
				\begin{aligned}
					S_n &= \sum_{k=1}^{n}c_k \\ &= \sum_{k=1}^{n}a_k + \sum_{k=1}^{n}b_k.
			\end{aligned}
			\]
			特别地，当$a_n=An+B$, $b_n=C q^{\,n-1}$时，
			\[
				\boxed{S_n = \frac{n}{2}\bigl[(A+B)+(An+B)\bigr] + \frac{C(1-q^{\,n})}{1-q}}.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（线性组合）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			多项拆分后仍可分别求和再相加。 对于$c_n = a_n + b_n + d$（$d$为常数），有
			\[
				\begin{aligned}
					S_n &= \frac{n}{2}\bigl[(A+B)+(An+B)\bigr] + \frac{C(1-q^{\,n})}{1-q} \\ &\quad + n d.
			\end{aligned}
			\]
	\end{description}

\end{statementbox}
