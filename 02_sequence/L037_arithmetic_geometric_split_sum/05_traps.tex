\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（忽略公比分类）}}]
			等比公比$q=1$时仍使用$\frac{a_1(1-q^n)}{1-q}$求和，导致分母为零错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类讨论原则）：}}]
			当$q=1$时，数列为常数数列，前$n$项和为$n a_1$；当$q\neq1$时使用公式。
		\item[\textbf{\textcolor{CTrap}{错因分析（公式适用忽略）：}}]
			公式分母$1-q$隐含$q\neq1$前提，未分类讨论直接套用。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（首项确认错误）}}]
			错误地将通项表达式中的系数直接当作首项代入求和公式。如$b_n=2\cdot4^n$，误用首项$2$，实际首项为$2\cdot4=8$。
		\item[\textbf{\textcolor{CTrap}{正确理解（首项代入原则）：}}]
			首项一定是$n=1$时的通项值，必须将$n=1$代入$b_n$计算。
		\item[\textbf{\textcolor{CTrap}{错因分析（形式模仿错误）：}}]
			仅关注通项形式上的系数，忽略了指数部分的初始贡献。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（漏加常数项）}}]
			对于含常数项的数列（如$c_n = a_n + b_n + d$），只计算等差和等比部分，未加常数项和$n d$。
		\item[\textbf{\textcolor{CTrap}{正确理解（常数项求和）：}}]
			常数项$d$可视为公差为$0$的等差数列，其前$n$项和为$n d$。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维遗漏）：}}]
			容易被明显的等差、等比部分吸引全部注意力，简单常数项被疏忽。
	\end{description}

\end{trapbox}
