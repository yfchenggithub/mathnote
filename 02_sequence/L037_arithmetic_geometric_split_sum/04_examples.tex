\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列$\{c_n\}$的通项$c_n=2n+3^n$（$n\geq1$），求前$n$项和$S_n$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（通项拆分）：}]
			将$c_n$拆分为等差数列$a_n=2n$和等比数列$b_n=3^n$。
			\par\textbf{理由：} $2n$是线性形式，$3^n$是指数形式，符合等差+等比拆分。
		\item[\textbf{第二步（等差求和）：}]
			等差数列$a_n=2n$的首项$a_1=2$，末项$a_n=2n$，项数为$n$，因此
			\[
				\begin{aligned}
					\sum_{k=1}^{n} 2k &= \frac{n}{2}(2+2n) \\ &= n(n+1).
			\end{aligned}
			\]
			\par\textbf{理由：} 等差数列求和公式$S_n=\frac{n}{2}(a_1+a_n)$。
		\item[\textbf{第三步（等比求和）：}]
			等比数列$b_n=3^n$的首项$b_1=3$，公比$q=3$，项数为$n$，因此
			\[
				\begin{aligned}
					\sum_{k=1}^{n} 3^k &= \frac{3(3^n-1)}{3-1} \\ &= \frac{3^{n+1}-3}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 等比数列求和公式$S_n=\frac{a_1(1-q^n)}{1-q}$（$q\neq1$）。
		\item[\textbf{第四步（合并结果）：}]
			将第二步和第三步的结果相加，得
			\[
				S_n = n(n+1) + \frac{3^{n+1}-3}{2}.
			\]
			\par\textbf{理由：} 分组求和法。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = n(n+1) + \frac{3^{n+1}-3}{2}$}.

	\textbf{\textcolor{CExample}{例2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列$\{c_n\}$的通项$c_n=3n-1+2\cdot4^n$（$n\geq1$），求$S_n$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（通项拆分）：}]
			将$c_n$拆分为等差数列$a_n=3n-1$和等比数列$b_n=2\cdot4^n$。
			\par\textbf{理由：} $3n-1$为一次函数，$2\cdot4^n$为指数形式。
		\item[\textbf{第二步（等差求和）：}]
			等差数列$a_n=3n-1$的首项$a_1=2$，末项$a_n=3n-1$，项数$n$，因此
			\[
				\begin{aligned}
					\sum_{k=1}^{n} (3k-1) &= \frac{n}{2}(2+3n-1) \\ &= \frac{n(3n+1)}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 等差数列求和公式。
		\item[\textbf{第三步（等比求和）：}]
			等比数列$b_n=2\cdot4^n$的首项$b_1=8$，公比$q=4$，项数$n$，因此
			\[
				\begin{aligned}
					\sum_{k=1}^{n} 2\cdot4^k &= \frac{8(4^n-1)}{4-1} \\ &= \frac{8(4^n-1)}{3}.
			\end{aligned}
			\]
			\par\textbf{理由：} 等比数列求和公式（$q\neq1$）。
		\item[\textbf{第四步（合并结果）：}]
			将第二步和第三步的结果相加，得
			\[
				S_n = \frac{n(3n+1)}{2} + \frac{8(4^n-1)}{3}.
			\]
			\par\textbf{理由：} 分组求和。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = \frac{n(3n+1)}{2} + \frac{8(4^n-1)}{3}$}.

	\textbf{\textcolor{CExample}{例3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列$\{c_n\}$的通项$c_n=2n+(-3)^{n}+5$（$n\geq1$），求$S_n$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（通项拆分）：}]
			将$c_n$拆分为等差数列$a_n=2n$、等比数列$b_n=(-3)^{n}$和常数项$d=5$。
			\par\textbf{理由：} 符合等差、等比、常数三类结构。
		\item[\textbf{第二步（等差求和）：}]
			等差数列$a_n=2n$的前$n$项和为
			\[
				\sum_{k=1}^{n} 2k = n(n+1).
			\]
			\par\textbf{理由：} 等差数列求和公式（首项$2$，末项$2n$）。
		\item[\textbf{第三步（等比求和）：}]
			等比数列$b_n=(-3)^{n}$的首项$b_1=-3$，公比$q=-3$（$q\neq1$），项数$n$，因此
			\[
				\begin{aligned}
					\sum_{k=1}^{n} (-3)^{k} &= \frac{(-3)\bigl(1-(-3)^{n}\bigr)}{1-(-3)} \\ &= \frac{-3\bigl(1-(-3)^{n}\bigr)}{4} \\ &= \frac{3(-3)^{n} - 3}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 等比数列求和公式$S_n=\frac{a_1(1-q^n)}{1-q}$。
		\item[\textbf{第四步（常数求和）：}]
			常数项$d=5$的前$n$项和为$\sum_{k=1}^{n} 5 = 5n$。
			\par\textbf{理由：} 常数数列求和。
		\item[\textbf{第五步（合并结果）：}]
			将第二步、第三步、第四步的结果相加，得
			\[
				\begin{aligned}
					S_n &= n(n+1) + 5n + \frac{3(-3)^{n} - 3}{4} \\ &= n^2 + 6n + \frac{3(-3)^{n} - 3}{4}.
			\end{aligned}
			\]
			\par\textbf{理由：} 分组求和法。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = n^2 + 6n + \dfrac{3(-3)^{n} - 3}{4}$}.
\end{examplebox}
