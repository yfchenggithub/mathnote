\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\par\textbf{三个求和公式（等差、等比、自然数幂和）将累加转化为$n$的多项式或分式，直接代入参数即可。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（参数已知）：}已知数列类型及首项、公差（比）、项数。
		\item \textbf{条件2（幂和范围）：}自然数幂和公式中$n$为正整数。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（等比分类）：}慎用分母为零情况，牢记$q=1$单独处理。
		\item \textbf{检查项（公式符号）：}代入前核对项数及符号，避免$n-1$与$n+1$混淆。
	\end{itemize}

\end{summarybox}
