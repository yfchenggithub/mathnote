\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（等差参数）：}}]
		      等差数列 \(\{a_n\}\) 的首项 \(a_1\)、公差 \(d\)、项数 \(n\) 已知。
		\item[\textbf{\textcolor{CStatement}{条件二（等比参数）：}}]
		      等比数列 \(\{a_n\}\) 的首项 \(a_1\)、公比 \(q\)、项数 \(n\) 已知。
		\item[\textbf{\textcolor{CStatement}{条件三（幂和范围）：}}]
		      自然数幂和中，\(n\in\mathbb N^*\)，求和范围为 \(1\) 到 \(n\)。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（等差求和）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      前 \(n\) 项和等于首末项平均值乘以项数。
		      \[
			      \begin{aligned}
				      S_n & = \frac{n(a_1+a_n)}{2},      \\
				      S_n & = n a_1 + \frac{n(n-1)}{2}d.
			      \end{aligned}
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（等比求和）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      公比为 \(1\) 时和为 \(n\) 倍首项；公比不为 \(1\) 时使用错位相减公式。
		      \[
			      S_n =
			      \begin{cases}
				      n a_1,                   & q=1,    \\[4pt]
				      \dfrac{a_1(1-q^n)}{1-q}, & q\neq1.
			      \end{cases}
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（自然幂和）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      平方和与立方和分别可用关于 \(n\) 的三次、四次多项式表示。
		      \[
			      \begin{aligned}
				      \sum_{k=1}^{n} k^2 & = \frac{n(n+1)(2n+1)}{6},          \\
				      \sum_{k=1}^{n} k^3 & = \left[\frac{n(n+1)}{2}\right]^2.
			      \end{aligned}
		      \]
	\end{description}
\end{statementbox}
