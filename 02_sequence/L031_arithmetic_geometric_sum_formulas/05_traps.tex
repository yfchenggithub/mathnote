\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（项数常数误记）}}]
			用等差数列求和公式$S_n = na_1 + \frac{n(n-1)}{2}d$时，将第二项$\frac{n(n-1)}{2}d$误写为$\frac{n(n+1)}{2}d$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确公式形式）：}}]
			正确形式确为$\frac{n(n-1)}{2}$，其中$n-1$来源于$a_n = a_1+(n-1)d$中的$(n-1)$。
		\item[\textbf{\textcolor{CTrap}{错因分析（负迁移干扰）：}}]
			受平方和公式$\sum k^2=\frac{n(n+1)(2n+1)}{6}$中$n(n+1)$的干扰，记忆混淆导致符号出错。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（忽略公比分类）}}]
			遇到等比数列求和时，未检查公比$q$是否等于$1$，盲目使用$\frac{a_1(1-q^n)}{1-q}$这一公式。
		\item[\textbf{\textcolor{CTrap}{正确理解（分类必要性）：}}]
			必须分情况：$q=1$时$S_n = na_1$；$q\neq1$时$S_n = \frac{a_1(1-q^n)}{1-q}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（定义域疏忽）：}}]
			原公式推导中使用了$(1-q)$做分母，分母为零时无意义，故必须单独处理$q=1$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（平方和公式记错）}}]
			将$\sum_{k=1}^n k^2 = \frac{n(n+1)(2n+1)}{6}$误记成$\frac{n(n+1)(n+2)}{6}$或$\frac{n(n+1)}{2}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（标准公式记忆）：}}]
			平方和公式为$\frac{n(n+1)(2n+1)}{6}$；可通过$n=1$验算：$\frac{1\cdot2\cdot3}{6}=1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（缺乏验证习惯）：}}]
			死记硬背不验证，混淆了自然数和与平方和的关联。
	\end{description}

\end{trapbox}
