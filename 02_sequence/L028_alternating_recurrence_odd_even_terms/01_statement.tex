\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（递推与初值）：}}] 数列$\{a_n\}$满足隔项线性递推$a_{n+2}=p a_n+q$，其中$p,q$为常数，且已知首两项$a_1,a_2$。
		\item[\textbf{\textcolor{CStatement}{条件2（参数限制）：}}] $p\neq0$（保证递推结构有效）。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（公比非1情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$p\neq1$时，奇数项和偶数项分别构成以$p$为公比的等比型数列。
			\[
				a_{2k-1}=p^{\,k-1}a_1+q\frac{p^{\,k-1}-1}{p-1},
				\qquad
				a_{2k}=p^{\,k-1}a_2+q\frac{p^{\,k-1}-1}{p-1}\quad(k=1,2,\dots).
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（公比为1情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$p=1$时，奇数项和偶数项分别构成公差为$q$的等差数列。
			\[
				a_{2k-1}=a_1+(k-1)q,
				\qquad
				a_{2k}=a_2+(k-1)q\quad(k=1,2,\dots).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（常数项为零）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$q=0$时，奇数项和偶数项分别成为等比数列，公比为$p$。
			\[
				a_{2k-1}=p^{\,k-1}a_1,
				\qquad
				a_{2k}=p^{\,k-1}a_2.
			\]
	\end{description}
\end{statementbox}
