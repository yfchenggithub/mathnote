\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{隔项递推拆成奇偶子列，各自化归为等比或等差数列，直接写出通项公式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（递推形式）：} 递推式必须为$a_{n+2}=p a_n+q$（$p,q$常数，$p\neq0$）。
		\item \textbf{条件2（初值给定）：} 已知首两项$a_1,a_2$的值。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分类讨论）：} 先判断$p=1$还是$p\neq1$，错误混用公式会导致分母为零或形式不对。
		\item \textbf{检查项（k的确定）：} 求$a_n$时，先判断$n$奇偶，再正确计算子列索引$k$。
	\end{itemize}
\end{summarybox}
