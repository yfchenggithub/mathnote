\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（数列类型）：}}] 等差数列$\{a_n\}$，首项$a_1>0$，公差$d<0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（最值位置）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			前$n$项和$S_n$在最后一个非负项处取得最大值。
			\[
				S_n\text{最大} \iff a_n\ge 0 \text{且} a_{n+1}\le 0.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（对偶情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若数列递增且首项为负，则$S_n$在最后一个非正项处最小。
			\[
				a_1<0,\ d>0 \;\Longrightarrow\; S_n\text{最小} \iff a_n\le 0 \text{且} a_{n+1}\ge 0.
			\]
	\end{description}

\end{statementbox}
