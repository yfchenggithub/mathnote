	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	等差数列 $\{a_n\}$ 中，$a_1=20$，$d=-3$，求 $S_n$ 的最大值及相应的 $n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求通项）：}] $a_n = a_1+(n-1)d = 20-3(n-1)=23-3n$。
			\par\textbf{理由：}
			等差数列通项公式。
		\item[\textbf{第二步（解不等式）：}] 令 $a_n\ge 0$ 得 $23-3n \ge 0$，解得 $n\le \frac{23}{3}\approx 7.667$。因此满足 $a_n\ge 0$ 的最大整数 $n=7$。
			\par\textbf{理由：}
			由结论，$S_n$ 最大当 $n$ 是最后一个非负项的序号。
		\item[\textbf{第三步（求最大值）：}] $S_7 = 7a_1 + \frac{7\times 6}{2}d = 7\times 20 + 21\times (-3) = 140 - 63 = 77$。
			\par\textbf{理由：}
			等差数列前 $n$ 项和公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	当 $n=7$ 时 $S_n$ 取得最大值，最大值为 $\boldsymbol{77}$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	等差数列 $\{a_n\}$ 中，$a_1=12$，$a_4=0$，求 $S_n$ 的最大值及相应的 $n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求公差）：}] 由 $a_4 = a_1+3d$ 得 $12+3d = 0$，解得 $d=-4$。
			\par\textbf{理由：}
			等差数列通项公式。
		\item[\textbf{第二步（求通项）：}] $a_n = a_1+(n-1)d = 12-4(n-1)=16-4n$。
			\par\textbf{理由：}
			通项公式代入。
		\item[\textbf{第三步（确定 $n$）：}] 令 $a_n\ge 0$ 得 $16-4n\ge 0$，解得 $n\le 4$，故满足 $a_n\ge 0$ 的最大整数 $n=4$。
			\par\textbf{理由：}
			由结论，$S_n$ 最大当 $n$ 为最后一个非负项。
		\item[\textbf{第四步（求最大值）：}] $S_4 = 4a_1 + \frac{4\times 3}{2}d = 4\times 12 + 6\times (-4) = 48-24=24$。
			\par\textbf{理由：}
			前 $n$ 项和公式。注意 $a_4=0$，故 $S_3=S_4=24$，最大值也在 $n=3$ 取得。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	当 $n=4$ 时 $S_n$ 取得最大值，最大值为 $\boldsymbol{24}$；同时 $S_3$ 也等于该最大值。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	等差数列 $\{a_n\}$ 中，$a_1=15$，且 $S_2=S_8$，求 $S_n$ 的最大值及相应的 $n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（由条件求公差）：}] $S_2=2a_1+d=30+d$，$S_8=8a_1+28d=120+28d$。由 $S_2=S_8$ 得 $30+d=120+28d$，解得 $d=-\frac{10}{3}$。
			\par\textbf{理由：}
			前 $n$ 项和公式。
		\item[\textbf{第二步（求通项）：}] $a_n = a_1+(n-1)d = 15 - \frac{10}{3}(n-1) = \frac{55-10n}{3}$。
			\par\textbf{理由：}
			通项公式。
		\item[\textbf{第三步（确定 $n$）：}] 令 $a_n\ge 0$ 得 $\frac{55-10n}{3}\ge 0$，即 $55-10n\ge 0$，$n\le 5.5$。故满足 $a_n\ge 0$ 的最大整数 $n=5$（检验 $a_5>0$，$a_6<0$）。
			\par\textbf{理由：}
			由结论，$S_n$ 最大当 $n$ 是最后一个非负项。
		\item[\textbf{第四步（求最大值）：}] $S_5 = 5a_1 + \frac{5\times 4}{2}d = 5\times 15 + 10\times \left(-\frac{10}{3}\right) = 75 - \frac{100}{3} = \frac{125}{3}$。
			\par\textbf{理由：}
			前 $n$ 项和公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	当 $n=5$ 时 $S_n$ 取得最大值，最大值为 $\boldsymbol{\frac{125}{3}}$。
