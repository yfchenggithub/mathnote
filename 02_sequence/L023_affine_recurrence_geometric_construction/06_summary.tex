\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{通过设 $b_n=a_n+An+B$ 消去一次项，将递推化为等比数列，直接写出通项。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（递推类型）：} $a_{n+1}=p a_n+q n+r$，$p,q,r$ 为常数。
		\item \textbf{条件 2（参数限制）：} $p\neq0,1$（否则需用其他方法）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（分母非零）：} 计算 $A,B$ 时分母 $p-1$ 不能为零。
		\item \textbf{检查项（回代完整）：} 求 $a_n$ 时要减去 $An+B$，不可遗漏。
	\end{itemize}
\end{summarybox}
