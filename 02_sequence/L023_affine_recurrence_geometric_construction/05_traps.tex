\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（条件检查）：}}] 当 $p=1$ 时，直接使用公式 $A=\dfrac{q}{p-1}$，导致分母为零，无法计算。
		\item[\textbf{\textcolor{CTrap}{正确理解（范围判断）：}}] 该构造法仅适用于 $p\neq0,1$；$p=1$ 时属于等差数列型，应采用累加法。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）：}}] 学生未先判断 $p$ 的值，盲目套用公式，造成数学错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公式遗漏）：}}] 计算 $B$ 时只写了 $\dfrac{r}{p-1}$，漏掉了 $\dfrac{q}{(p-1)^2}$，导致通项出错。
		\item[\textbf{\textcolor{CTrap}{正确理解（完整公式）：}}] $B = \dfrac{r}{p-1} + \dfrac{q}{(p-1)^2}$，需由方程组 $A=\frac{q}{p-1}$ 和 $-pB+r+A+B=0$ 解出。
		\item[\textbf{\textcolor{CTrap}{错因分析（不求甚解）：}}] 学生记忆公式不全，没有理解 $B$ 的来源，仅凭记忆成分出错。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（回代遗忘）：}}] 求得 $b_n = b_1 p^{\,n-1}$ 后直接写 $a_n = b_1 p^{\,n-1}$，忘记减去 $An+B$。
		\item[\textbf{\textcolor{CTrap}{正确理解（回代关系）：}}] $a_n = b_n - An - B$，必须减去构造的一次式。
		\item[\textbf{\textcolor{CTrap}{错因分析（关系混淆）：}}] 混淆了 $a_n$ 与 $b_n$ 的关系，忽略构造的代换。
	\end{description}
\end{trapbox}
