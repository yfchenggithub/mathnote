\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{等比数列中，任意两项通过公比的整数次幂联系：$a_n = a_m q^{\,n-m}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（等比数列）：} 数列必须为等比数列。
		\item \textbf{条件 2（公比非零）：} 公比 $q \neq 0$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}

	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（指数方向）：} 指数 $n-m$，切勿写成 $m-n$。
		\item \textbf{检查项（公比符号）：} 当 $q<0$ 时，注意幂的正负交替。
	\end{itemize}

\end{summarybox}
