\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（公比非零）：}}] 等比数列 $\{a_n\}$，公比 $q \neq 0$，$m,n$ 为任意正整数。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（两项关系）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			任意两项可通过公比的幂次相互表示。
			\[
				a_n = a_m \cdot q^{\,n-m}.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（通项特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $m=1$ 时，得到首项表达的通项公式。
			\[
				a_n = a_1 \cdot q^{\,n-1}.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{参数限制：}}
	公比 $q \neq 0$，确保指数运算有意义。

\end{statementbox}
