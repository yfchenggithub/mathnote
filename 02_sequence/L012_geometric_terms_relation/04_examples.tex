	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等比数列 $\{a_n\}$ 中，$a_2=6$，$q=2$，求 $a_5$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（确定下标差）：}] $n=5,\,m=2$，下标差 $n-m=3$。
			\par\textbf{理由：}
			由两点关系公式 $a_n = a_m q^{n-m}$ 可知需先确定指数。
		\item[\textbf{第二步（代入公式）：}] $a_5 = 6 \times 2^{3}$。
			\par\textbf{理由：}
			直接代入已知数值。
		\item[\textbf{第三步（计算结果）：}] $6 \times 8 = 48$。
			\par\textbf{理由：}
			先计算 $2^3=8$，再与 $6$ 相乘。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_5 = 48$}。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	等比数列 $\{a_n\}$ 中，$a_3=12$，$a_6=96$，求公比 $q$，并求 $a_9$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（建立方程）：}] $a_6 = a_3 q^{6-3} = 12 q^3 = 96$。
			\par\textbf{理由：}
			应用两项关系，用已知项表达未知项。
		\item[\textbf{第二步（解方程）：}] $q^3 = 8$，故 $q=2$（取正实根）。
			\par\textbf{理由：}
			开立方得 $q=2$（实数唯一解）。
		\item[\textbf{第三步（求 $a_9$）：}] $a_9 = a_3 q^{9-3} = 12 \times 2^6 = 12 \times 64 = 768$。
			\par\textbf{理由：}
			再次套用公式，用 $a_3$ 和 $q$ 计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$q=2$，$a_9=768$}。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等比数列 $\{a_n\}$ 中，$a_2=3$，$a_5=81$，求首项 $a_1$ 和通项公式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求公比）：}] $a_5 = a_2 q^{5-2} = 3q^3 = 81$，得 $q^3=27$，$q=3$。
			\par\textbf{理由：}
			利用两项关系列方程。
		\item[\textbf{第二步（求首项）：}] $a_2 = a_1 q$，所以 $a_1 = \dfrac{a_2}{q} = \dfrac{3}{3}=1$。
			\par\textbf{理由：}
			由通项特例 $a_n = a_1 q^{n-1}$ 取 $n=2$ 得 $a_2 = a_1 q$。
		\item[\textbf{第三步（写通项）：}] $a_n = a_1 q^{n-1} = 1 \cdot 3^{n-1} = 3^{n-1}$。
			\par\textbf{理由：}
			代入首项和公比得到通项公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_1=1$，$a_n=3^{n-1}$}。
