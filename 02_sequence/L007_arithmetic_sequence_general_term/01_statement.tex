\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（等差定义）：}}] 数列 $\{a_n\}$ 是等差数列，首项为 $a_1$，公差为 $d$，其中 $n\in\mathbb{N}^*$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（通项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			第 $n$ 项等于首项加上 $(n-1)$ 倍的公差。
			\[
				\begin{aligned}
					a_n &= a_1 + (n-1)d \\
					&= d\,n + (a_1 - d)
			\end{aligned}
			\]
			其中 $n\in\mathbb{N}^*$。
		\item[\textbf{\textcolor{CStatement}{推广结论（通项变形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			已知任意一项 $a_m$ 和公差 $d$，可求其他项 $a_n$。
			\[
				a_n = a_m + (n-m)d
			\]
			其中 $m,n\in\mathbb{N}^*$。
	\end{description}
\end{statementbox}
