\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（项数混淆）}}] 将通项误写为 $a_n = a_1 + n d$，忘记减 $1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（增量次数）：}}] 第 $n$ 项比首项多 $(n-1)$ 个公差。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略减一）：}}] 混淆了项数与增加次数的关系。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（下标起始）}}] 默认 $n$ 从 $0$ 开始，认为存在 $a_0$。
		\item[\textbf{\textcolor{CTrap}{正确理解（正整数集）：}}] 等差数列的项数 $n\in\mathbb{N}^*$，最小为 $1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（起始混淆）：}}] 有时在其他数列中遇到 $a_0$，直接套用导致错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（公差符号）}}] 当 $d<0$ 时，代入公式漏掉负号或误加绝对值。
		\item[\textbf{\textcolor{CTrap}{正确理解（代数处理）：}}] 公差 $d$ 可正可负，代入公式时保留原符号。
		\item[\textbf{\textcolor{CTrap}{错因分析（负号遗漏）：}}] 习惯性认为等差是递增，忽略递减情形。
	\end{description}
\end{trapbox}
