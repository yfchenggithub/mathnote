\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	在等差数列 $\{a_n\}$ 中，$a_1=2$，$d=3$，求 $a_5$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 已知 $a_1=2$，$d=3$，$n=5$。
			\par\textbf{理由：} 确认已知量。
		\item[\textbf{第二步（代入公式）：}] 套用通项公式 $a_n = a_1+(n-1)d$，得
			\[
				\begin{aligned}
					a_5 &= 2 + (5-1)\times 3 \\
					&= 2 + 12 \\
					&= 14.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接代入数值计算。
		\item[\textbf{第三步（验证结论）：}] 所以 $a_5 = 14$。
			\par\textbf{理由：} 计算无误。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\mathbf{a_5 = 14}$。

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	在等差数列 $\{a_n\}$ 中，$a_3=10$，$d=2$，求首项 $a_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别条件）：}] 已知 $a_3=10$，$d=2$，$n=3$。
			\par\textbf{理由：} 确认已知量。
		\item[\textbf{第二步（代入公式）：}] 由 $a_n = a_1+(n-1)d$ 得
			\[
				\begin{aligned}
					10 &= a_1 + (3-1)\times 2 \\
					&= a_1 + 4.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入已知项。
		\item[\textbf{第三步（解方程）：}] $a_1 = 10 - 4$，所以 $a_1 = 6$。
			\par\textbf{理由：} 移项求解。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\mathbf{a_1 = 6}$。

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 满足 $a_2=5$，$a_5=14$，求通项公式 $a_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列方程组）：}] 设首项 $a_1$，公差 $d$，由通项公式得
			\[
				\begin{cases}
					a_1 + d = 5,\\
					a_1 + 4d = 14.
			\end{cases}
			\]
			\par\textbf{理由：} 将已知条件转化为关于 $a_1,d$ 的方程组。
		\item[\textbf{第二步（解方程组）：}] 两式相减得 $3d = 9$，所以 $d=3$；代入第一式得 $a_1 + 3 = 5$，所以 $a_1 = 2$。
			\par\textbf{理由：} 消元求解。
		\item[\textbf{第三步（写出通项）：}] 代入公式
			\[
				\begin{aligned}
					a_n &= a_1 + (n-1)d \\
					&= 2 + (n-1)\times 3 \\
					&= 3n - 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 化简为一次形式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	通项公式为 $\mathbf{a_n = 3n - 1}$。
\end{examplebox}
