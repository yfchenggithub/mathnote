	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（前n项和条件）：}}] 已知数列 $\{a_n\}$ 的前 $n$ 项和 $S_n$（$n\in\mathbb{N}^*$），且对 $n\ge 2$ 均有定义，并约定 $S_0=0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（分段通项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当 $n=1$ 时等于 $S_1$，当 $n\ge2$ 时等于相邻两项和之差。
			\[
				a_n=\begin{cases}
				S_1, & n=1,\\
				S_n-S_{n-1}, & n\ge 2.
		\end{cases}
		\]

	\item[\textbf{\textcolor{CStatement}{推广结论（统一条件判断）：}}]
		\textbf{\textcolor{CStatement}{直观描述：}}
		若 $n\ge2$ 时由 $S_n-S_{n-1}$ 化简得到的式子代入 $n=1$ 后等于 $S_1$，则该式对 $n=1$ 也成立，通项可统一写为该式，无需分段。
\end{description}

\end{statementbox}
