\begin{summarybox}

	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{已知 $S_n$ 求 $a_n$：$n=1$ 时取 $S_1$，$n\ge2$ 时取差，验证一致才可合并。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（前 $n$ 项和已知）：} 数列 $\{a_n\}$ 的前 $n$ 项和 $S_n$ 必须给出具体表达式或可计算的值。
		\item \textbf{条件 2（约定 $S_0$）：} $S_0$ 通常约定为 $0$，但这不影响推导，只需按定义分段处理。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（忽略 $n=1$）：} 切忌直接套用 $a_n=S_n-S_{n-1}$ 而不验证 $n=1$。
		\item \textbf{检查项（验证回代）：} 得到 $n\ge2$ 表达式后，必须代入 $n=1$ 看是否等于 $a_1$；若不等，立刻分段。
	\end{itemize}
\end{summarybox}
