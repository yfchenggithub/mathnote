\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{a_n\}$ 的前 $n$ 项和 $S_n = n^2$，求通项 $a_n$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求首项）：}]
			由 $n=1$ 得 $a_1 = S_1 = 1^2 = 1$。
			\par\textbf{理由：} 通项公式中 $n=1$ 时 $a_1=S_1$。
		\item[\textbf{第二步（求 $n\ge2$ 通项）：}]
			当 $n\ge2$，$a_n = S_n - S_{n-1} = n^2 - (n-1)^2 = 2n-1$。
			\par\textbf{理由：} 应用公式 $a_n=S_n-S_{n-1}$。
		\item[\textbf{第三步（验证 $n=1$）：}]
			将 $n=1$ 代入 $a_n = 2n-1$ 得 $2\times1-1=1$，与 $a_1=1$ 一致。
			\par\textbf{理由：} 若一致则通项可统一，无需分段。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n = 2n-1$（$n\in\mathbb{N}^*$）。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $S_n = 2^n + 1$，求 $a_n$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求首项）：}]
			$a_1 = S_1 = 2^1 + 1 = 3$。
			\par\textbf{理由：} $n=1$ 时直接取 $S_1$。
		\item[\textbf{第二步（求 $n\ge2$ 通项）：}]
			$a_n = S_n - S_{n-1} = (2^n+1) - (2^{n-1}+1) = 2^n - 2^{n-1} = 2^{n-1}$。
			\par\textbf{理由：} 使用差公式化简。
		\item[\textbf{第三步（验证 $n=1$ 并分段）：}]
			将 $n=1$ 代入 $a_n=2^{n-1}$ 得 $2^{0}=1$，不等于 $a_1=3$。
			\par\textbf{理由：} 不一致，故通项必须分段。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n =
	\begin{cases}
		3, & n=1,\\ 2^{n-1}, & n\ge2.
	\end{cases}
	$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}}
	已知 $S_n = 3n^2 - 2n + 1$，求 $a_n$。
	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求首项）：}]
			$a_1 = S_1 = 3\times1^2 -2\times1+1 = 2$。
			\par\textbf{理由：} $n=1$ 时 $a_1=S_1$。
		\item[\textbf{第二步（求 $n\ge2$ 通项）：}]
			\[
				\begin{aligned}
					a_n &= S_n - S_{n-1} \\ &= (3n^2-2n+1) - [3(n-1)^2-2(n-1)+1] \\ &= (3n^2-2n+1) - [3(n^2-2n+1)-2n+2+1] \\ &= 3n^2-2n+1 - (3n^2-6n+3-2n+3) \\ &= 3n^2-2n+1 - (3n^2-8n+6) \\ &= 6n-5.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入 $S_n$ 表达式化简。
		\item[\textbf{第三步（验证 $n=1$ 并输出）：}]
			将 $n=1$ 代入 $a_n=6n-5$ 得 $1$，与 $a_1=2$ 不相等。
			\par\textbf{理由：} 不一致，故通项需分段。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n =
	\begin{cases}
		2, & n=1,\\ 6n-5, & n\ge2.
	\end{cases}
	$}
\end{examplebox}
