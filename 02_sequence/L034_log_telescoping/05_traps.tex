\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（底数条件）}}]
			忽略$a>0$且$a\neq 1$，错误地在$a\leq0$或$a=1$时使用该恒等式。
		\item[\textbf{\textcolor{CTrap}{正确理解（对数前提）：}}]
			恒等式成立的条件是$a>0$且$a\neq 1$，否则对数无定义或恒等式不成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（对数定义）：}}]
			对底数范围理解不清，误以为$a$可为任意实数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（法则混淆）}}]
			误将$\log_a(1+\frac{1}{n})$拆成$\log_a 1 + \log_a \frac{1}{n}$，即错误使用对数加法法则。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确裂项）：}}]
			应为$\log_a(1+\frac{1}{n}) = \log_a(n+1) - \log_a n$，不能拆成和。
		\item[\textbf{\textcolor{CTrap}{错因分析（法则混淆）：}}]
			将对数加法与对数减法记反，或混淆$\log(M+N)$与$\log M + \log N$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（求和遗漏）}}]
			在数列求和时，裂项后展开不完整，导致中间项抵消出错，如忘记$\log_a 1=0$而多出项。
		\item[\textbf{\textcolor{CTrap}{正确理解（完全展开）：}}]
			必须逐项写出并观察抵消规律，确保首末项正确。
		\item[\textbf{\textcolor{CTrap}{错因分析（过程跳步）：}}]
			为了节省步骤，直接给出结果，可能导致中间项抵消错误。
	\end{description}

\end{trapbox}
