\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（底数条件）}}]
		      忽略 \(a>0\) 且 \(a\neq1\)，错误地在 \(a\leq0\) 或 \(a=1\) 时使用该恒等式。
		\item[\textbf{\textcolor{CTrap}{正确理解（对数前提）：}}]
		      恒等式成立的底数条件是 \(a>0\) 且 \(a\neq1\)，否则对数没有意义。
		\item[\textbf{\textcolor{CTrap}{错因分析（对数定义）：}}]
		      对底数范围理解不清，误以为 \(a\) 可以为任意实数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（法则混淆）}}]
		      误将 \(\log_a\!\left(1+\frac{1}{n}\right)\) 拆成
		      \[
			      \log_a1+\log_a\frac{1}{n},
		      \]
		      即错误使用对数乘法法则。
		\item[\textbf{\textcolor{CTrap}{正确理解（正确裂项）：}}]
		      正确裂项为
		      \[
			      \log_a\!\left(1+\frac{1}{n}\right)=\log_a(n+1)-\log_a n,
		      \]
		      不能把 \(\log(M+N)\) 拆成 \(\log M+\log N\)。
		\item[\textbf{\textcolor{CTrap}{错因分析（法则混淆）：}}]
		      对数公式中，只有
		      \[
			      \log_a(MN)=\log_aM+\log_aN,
			      \qquad
			      \log_a\frac{M}{N}=\log_aM-\log_aN,
		      \]
		      不存在 \(\log_a(M+N)=\log_aM+\log_aN\)。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（求和遗漏）}}]
		      在数列求和时，裂项后展开不完整，导致中间项抵消出错，或忘记 \(\log_a1=0\)。
		\item[\textbf{\textcolor{CTrap}{正确理解（完全展开）：}}]
		      必须逐项写出并观察抵消规律，确保首末项正确。
		\item[\textbf{\textcolor{CTrap}{错因分析（过程跳步）：}}]
		      为了节省步骤，直接给出结果，容易导致首项、末项或抵消方向出错。
	\end{description}
\end{trapbox}
