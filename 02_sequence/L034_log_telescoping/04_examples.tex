\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础裂项求和）}}

	\textbf{\textcolor{CExample}{题目：}}
	计算
	\[
		S=\sum_{k=1}^{99}\log_3\!\left(1+\frac{1}{k}\right).
	\]

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（逐项裂项）：}]
		      \[
			      \log_3\!\left(1+\frac{1}{k}\right)
			      =\log_3(k+1)-\log_3 k.
		      \]

		\item[\textbf{第二步（展开观察相消）：}]
		      \[
			      \begin{aligned}
			      S
			      &=(\log_3 2-\log_3 1)
			      +(\log_3 3-\log_3 2)\\
			      &\quad +\cdots
			      +(\log_3 100-\log_3 99)\\
			      &=\log_3 100-\log_3 1.
			      \end{aligned}
		      \]

		\item[\textbf{第三步（利用 \(\log_3 1=0\)）：}]
		      \[
			      S=\log_3 100.
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{S=\log_3 100}.
	\]

	\textbf{\textcolor{CExample}{例 2（两个相邻裂项式合并）}}

	\textbf{\textcolor{CExample}{题目：}}
	设 \(n\in\mathbb N^*\)，化简
	\[
		A_n=
		\log_2\!\left(1+\frac{1}{n}\right)
		+\log_2\!\left(1+\frac{1}{n+1}\right),
	\]
	并求 \(A_{100}\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分别裂项）：}]
		      \[
			      \log_2\!\left(1+\frac{1}{n}\right)
			      =\log_2(n+1)-\log_2 n,
		      \]
		      \[
			      \log_2\!\left(1+\frac{1}{n+1}\right)
			      =\log_2(n+2)-\log_2(n+1).
		      \]

		\item[\textbf{第二步（消去中间项）：}]
		      \[
			      \begin{aligned}
			      A_n
			      &=[\log_2(n+1)-\log_2 n]\\
			      &\quad +[\log_2(n+2)-\log_2(n+1)]\\
			      &=\log_2(n+2)-\log_2 n\\
			      &=\log_2\!\left(\frac{n+2}{n}\right)\\
			      &=\log_2\!\left(1+\frac{2}{n}\right).
			      \end{aligned}
		      \]

		\item[\textbf{第三步（代入 \(n=100\)）：}]
		      \[
			      \begin{aligned}
			      A_{100}
			      &=\log_2\!\left(\frac{102}{100}\right)\\
			      &=\log_2\!\left(\frac{51}{50}\right).
			      \end{aligned}
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{
		A_n=\log_2\!\left(\frac{n+2}{n}\right)
		}.
	\]
	\[
		\boxed{
		A_{100}=\log_2\!\left(\frac{51}{50}\right)
		}.
	\]

	\textbf{\textcolor{CExample}{例 3（隔项结构转化为连续相消）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 \(\{c_n\}\) 满足
	\[
		c_n=
		\log_2\!\left(1+\frac{1}{2n-1}\right)
		+\log_2\!\left(1+\frac{1}{2n}\right),
		\qquad n\in\mathbb N^*,
	\]
	求其前 \(n\) 项和 \(S_n\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（分别裂项）：}]
		      \[
			      \log_2\!\left(1+\frac{1}{2n-1}\right)
			      =\log_2(2n)-\log_2(2n-1),
		      \]
		      \[
			      \log_2\!\left(1+\frac{1}{2n}\right)
			      =\log_2(2n+1)-\log_2(2n).
		      \]

		\item[\textbf{第二步（先化简通项）：}]
		      \[
			      \begin{aligned}
			      c_n
			      &=[\log_2(2n)-\log_2(2n-1)]\\
			      &\quad +[\log_2(2n+1)-\log_2(2n)]\\
			      &=\log_2(2n+1)-\log_2(2n-1).
			      \end{aligned}
		      \]

		\item[\textbf{第三步（对通项再次裂项相消）：}]
		      \[
			      \begin{aligned}
			      S_n
			      &=\sum_{k=1}^{n}
			      [\log_2(2k+1)-\log_2(2k-1)]\\
			      &=(\log_2 3-\log_2 1)
			      +(\log_2 5-\log_2 3)\\
			      &\quad +\cdots
			      +(\log_2(2n+1)-\log_2(2n-1))\\
			      &=\log_2(2n+1)-\log_2 1\\
			      &=\log_2(2n+1).
			      \end{aligned}
		      \]
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{S_n=\log_2(2n+1)}.
	\]
\end{examplebox}
