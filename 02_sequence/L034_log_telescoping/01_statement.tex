\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（底数范围）：}}]
		      \(a>0\) 且 \(a\neq 1\)。

		\item[\textbf{\textcolor{CStatement}{条件二（恒等式定义域）：}}]
		      \(n>0\)。此时 \(n\)、\(n+1\) 以及
		      \(1+\frac{1}{n}=\frac{n+1}{n}\) 均为正数，相关对数都有意义。

		\item[\textbf{\textcolor{CStatement}{条件三（数列场景）：}}]
		      在数列问题中，通常进一步取 \(n\in\mathbb N^*\)。
	\end{description}

	\textbf{\textcolor{CStatement}{核心结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（对数裂项恒等式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      将相邻两数之比的对数，拆成相邻两项对数之差：
		      \[
			      \boxed{
				      \log_a\!\left(1+\frac{1}{n}\right)
				      =
				      \log_a(n+1)-\log_a n
			      }.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论二（倒数形式）：}}]
		      将真数取倒数后，裂项次序相反：
		      \[
			      \boxed{
				      \log_a\!\left(\frac{n}{n+1}\right)
				      =
				      \log_a n-\log_a(n+1)
			      }.
		      \]

		\item[\textbf{\textcolor{CStatement}{结论三（连续求和）：}}]
		      若 \(p,q\in\mathbb N^*\) 且 \(p\leq q\)，则
		      \[
			      \boxed{
				      \sum_{k=p}^{q}
				      \log_a\!\left(1+\frac{1}{k}\right)
				      =
				      \log_a\!\left(\frac{q+1}{p}\right)
			      }.
		      \]
		      特别地，当 \(p=1\) 时，
		      \[
			      \sum_{k=1}^{q}
			      \log_a\!\left(1+\frac{1}{k}\right)
			      =\log_a(q+1).
		      \]
	\end{description}
\end{statementbox}
