\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件一（底数范围）：}}]
			$a>0$且$a\neq 1$。
		\item[\textbf{\textcolor{CStatement}{条件二（变量限制）：}}]
			$n\in\mathbb{N}^*$（正整数）。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（裂项恒等式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将对数分式拆成相邻整数对数之差，累加可相互抵消。
			\[
				\log_a\!\left(1+\frac{1}{n}\right)=\log_a(n+1)-\log_a n.
			\]
		\item[\textbf{\textcolor{CStatement}{推广结论（倒数形式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			对数内取倒数后，变成反向减法，符号相反。
			\[
				\log_a\!\left(\frac{n}{n+1}\right)=\log_a n-\log_a(n+1).
			\]
	\end{description}

\end{statementbox}
