\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（各项非零）：}}] 等比数列 $\{a_n\}$ 中所有项均不为零。
		\item[\textbf{\textcolor{CStatement}{条件2（下标限制）：}}] 要求 $n\ge 2$，即 $a_{n-1}$ 存在。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（等比中项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			等比数列中相邻三项满足中间项的平方等于两端项之积。
			\[
				a_n^2 = a_{n-1} a_{n+1}, \quad n\ge 2.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（符号取舍）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			等比中项可以取正负两个值，解题时需根据题意确定符号。
			\[
				a_n = \pm \sqrt{a_{n-1} a_{n+1}}.
			\]
	\end{description}

\end{statementbox}
