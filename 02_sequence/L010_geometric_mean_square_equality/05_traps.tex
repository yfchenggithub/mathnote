\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（符号遗漏）}}] 在求等比中项时，经常只取正值，忽略负值。
		\item[\textbf{\textcolor{CTrap}{正确理解（正负双解）}}] 由 $x^2 = ab$ 应得 $x = \pm \sqrt{ab}$，必须考虑两个值。
		\item[\textbf{\textcolor{CTrap}{错因分析（思维定势）}}] 学生习惯了算术平方根表示非负数，忘记方程 $x^2 = ab$ 有两个根。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（零项失范）}}] 在等比数列中忽视“各项非零”条件，将等比中项公式用于含有零项的数列。
		\item[\textbf{\textcolor{CTrap}{正确理解（非零约束）}}] 等比中项公式要求 $a_{n-1}$ 与 $a_{n+1}$ 均不为零，否则等比中项无定义。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视前提）}}] 等比数列定义中公比不为零，但若首项为零则全项为零，此时不构成等比数列（通常等比数列排除零项）。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（混用等差）}}] 误将等差中项公式 $2a_n = a_{n-1}+a_{n+1}$ 用于等比数列。
		\item[\textbf{\textcolor{CTrap}{正确理解（区别对比）}}] 等比中项是乘积关系 $a_n^2 = a_{n-1}a_{n+1}$，等差中项是和的关系，两者完全不同。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）}}] 没有区分等差与等比中项，在解决等比问题时错误使用加法公式。
	\end{description}
\end{trapbox}
