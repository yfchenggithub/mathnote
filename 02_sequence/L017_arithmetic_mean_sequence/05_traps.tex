\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（前项和等差）}}] 误以为数列 $\{S_n\}$ 是等差数列。
		\item[\textbf{\textcolor{CTrap}{正确理解（平均与和区别）：}}] 数列 $\{S_n\}$ 是二次型，不是等差数列；但 $\{\frac{S_n}{n}\}$ 是等差数列。
		\item[\textbf{\textcolor{CTrap}{错因分析（混淆概念）：}}] 混淆了前 $n$ 项和序列与前 $n$ 项平均值序列，二者等差性质不同。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（公差系数错）}}] 认为数列 $\{\frac{S_n}{n}\}$ 的公差等于 $d$。
		\item[\textbf{\textcolor{CTrap}{正确理解（公差一半）：}}] 正确公差为 $\frac{d}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忘记除二）：}}] 从 $\frac{S_n}{n} = a_1 + \frac{n-1}{2}d$ 可知公差是 $\frac{d}{2}$，忽视了系数 $\frac{1}{2}$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（首项误认）}}] 误认为数列 $\{\frac{S_n}{n}\}$ 的首项不是 $a_1$。
		\item[\textbf{\textcolor{CTrap}{正确理解（首项确认）：}}] 由 $n=1$ 时 $S_1 = a_1$，得 $\frac{S_1}{1}=a_1$，所以首项就是 $a_1$。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略第一项）：}}] 未考虑 $n=1$ 的情形，或错误地认为首项与 $a_1$ 不同。
	\end{description}
\end{trapbox}
