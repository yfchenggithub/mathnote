\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{等差数列前 $n$ 项的平均值 $\frac{S_n}{n}$ 构成以 $a_1$ 为首项、$\frac{d}{2}$ 为公差的等差数列。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（原数列等差）：} 原数列 $\{a_n\}$ 必须是等差数列。
		\item \textbf{条件 2（已知求和）：} 已知前 $n$ 项和 $S_n$ 或可求 $S_n$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（混淆等差）：} 注意区分 $\{S_n\}$ 与 $\{S_n/n\}$，只有后者是等差数列。
		\item \textbf{检查项（公差系数）：} 计算公差时记得除以 $2$，即公差为 $\frac{d}{2}$。
	\end{itemize}
\end{summarybox}
