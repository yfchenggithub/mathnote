\begin{proofbox}
	\textbf{\textcolor{CProof}{思路提示}}

	要证明数列 $\{\frac{S_n}{n}\}$ 是等差数列，只需证明 $\frac{S_{n}}{n} - \frac{S_{n-1}}{n-1}$ 为常数，或直接证明 $\frac{S_n}{n}$ 是 $n$ 的一次函数。

	\medskip
	\textbf{\textcolor{CProof}{正式推导}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.55em]
		\item[\textbf{步骤一（求和公式）：}] 由等差数列前 $n$ 项和公式得：
			\[
				S_n = n a_1 + \frac{n(n-1)}{2}d.
			\]
			\par\textbf{理由：} 等差数列求和公式，其中首项 $a_1$，公差 $d$。

		\item[\textbf{步骤二（平均表达式）：}] 两边除以 $n$（$n\ge 1$）得：
			\[
				\frac{S_n}{n} = a_1 + \frac{n-1}{2}d.
			\]
			\par\textbf{理由：} 代数恒等变形。

		\item[\textbf{步骤三（差值验证）：}] 考虑相邻两项之差：
			\[
				\begin{aligned}
					\frac{S_{n+1}}{n+1} - \frac{S_n}{n}
					&= \left( a_1 + \frac{(n+1)-1}{2}d \right) - \left( a_1 + \frac{n-1}{2}d \right) \\
					&= \left( a_1 + \frac{n}{2}d \right) - \left( a_1 + \frac{n-1}{2}d \right) \\
					&= \frac{d}{2}.
			\end{aligned}
			\]
			\par\textbf{理由：} 代入步骤二的表达式，合并化简。
	\end{description}

	\medskip
	\textbf{\textcolor{CProof}{结论回扣}}

	\textbf{因此，$\frac{S_n}{n}$ 与 $n$ 有关，且相邻两项差为常数 $\frac{d}{2}$，故数列 $\{\frac{S_n}{n}\}$ 是首项为 $a_1$、公差为 $\frac{d}{2}$ 的等差数列。}
\end{proofbox}
