\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}

	等差数列前 $n$ 项的平均值 $\frac{S_n}{n}$ 等于首项 $a_1$ 与 $\frac{n-1}{2}d$ 之和，即关于 $n$ 的一次函数。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（公式变形）：}] 由等差数列前 $n$ 项和公式 $S_n = n a_1 + \frac{n(n-1)}{2}d$，两边除以 $n$ 得 $\frac{S_n}{n} = a_1 + \frac{n-1}{2}d$。
		\item[\textbf{要点二（线性关系）：}] $\frac{S_n}{n}$ 是 $n$ 的一次函数 $f(n) = a_1 + \frac{d}{2}(n-1)$，因此构成等差数列，公差为 $\frac{d}{2}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（线性图像）}}

	将 $n$ 视为横坐标，$\frac{S_n}{n}$ 视为纵坐标，则所有点 $(n, \frac{S_n}{n})$ 落在斜率为 $\frac{d}{2}$ 的直线上。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（一次函数）}}

	$\frac{S_n}{n}$ 可写为 $a_1 - \frac{d}{2} + \frac{d}{2}n$，是关于 $n$ 的一次函数，一次项系数为 $\frac{d}{2}$。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（解题应用）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（求公差）：} 若已知 $\frac{S_n}{n}$ 的相邻两项差，可直接得到 $\frac{d}{2}$，从而求出 $d$。
		\item \textbf{考法二（求总和）：} 利用 $\frac{S_n}{n}$ 表达式，乘以 $n$ 快速得到 $S_n$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}

	\textbf{为什么公差是 $\frac{d}{2}$？因为 $S_n$ 是 $n$ 的二次函数，除以 $n$ 后变成一次函数，其斜率恰好等于二次项系数 $d/2$。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（由和求公差）：} 已知 $S_n = n^2 + n$，可先求 $\frac{S_n}{n} = n+1$，得到公差 $d=2$。
		\item \textbf{场景二（利用等差求公差）：} 已知数列 $\{\frac{S_n}{n}\}$ 是等差数列，利用其相邻项差可直接求得 $\frac{d}{2}$，进而得到 $d$。
	\end{itemize}
\end{explanationbox}
