\begin{statementbox}

	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（等差数列）：}}] 设数列 $\{a_n\}$ 为等差数列，公差为 $d$，前 $n$ 项和为 $S_n$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（平均等差）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			前 $n$ 项的平均值构成等差数列，公差为 $d/2$。
			\[
				\frac{S_n}{n} = a_1 + \frac{n-1}{2}d.
			\]

		\item[\textbf{\textcolor{CStatement}{常见变形（求和公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			由上式两边乘以 $n$ 即得前 $n$ 项和公式。
			\[
				S_n = n a_1 + \frac{n(n-1)}{2}d.
			\]
	\end{description}

\end{statementbox}
