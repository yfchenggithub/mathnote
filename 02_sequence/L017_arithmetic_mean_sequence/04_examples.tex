\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 中，$a_1 = 1$，$d = 2$，求 $\frac{S_{10}}{10}$ 的值，并写出数列 $\{\frac{S_n}{n}\}$ 的首项与公差。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（套用公式）：}] 由结论 $\frac{S_n}{n} = a_1 + \frac{n-1}{2}d$，代入 $a_1=1, d=2$ 得
			\[
				\begin{aligned}
					\frac{S_n}{n} &= 1 + \frac{n-1}{2}\times 2 \\
					&= 1 + (n-1) \\
					&= n.
			\end{aligned}
			\]
			\par\textbf{理由：} 直接使用平均数列的表达式。
		\item[\textbf{第二步（代入求值）：}] 令 $n=10$，得 $\frac{S_{10}}{10} = 10$。
			\par\textbf{理由：} 代入具体 $n$ 值计算。
		\item[\textbf{第三步（判断性质）：}] 由 $\frac{S_n}{n}=n$ 可知数列 $\{\frac{S_n}{n}\}$ 是首项为 $1$、公差为 $1$ 的等差数列。
			\par\textbf{理由：} 表达式为 $n$，是一次函数，公差为系数 $1$。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	$\frac{S_{10}}{10}=10$；数列 $\{\frac{S_n}{n}\}$ 的首项为 $1$，公差为 $1$。

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 的前 $n$ 项和 $S_n$ 满足 $\frac{S_n}{n} = 3n - 2$，求公差 $d$ 和首项 $a_1$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（形式对照）：}] 由结论
			\[
				\begin{aligned}
					\frac{S_n}{n} &= a_1 + \frac{n-1}{2}d \\
					&= \frac{d}{2}n + (a_1 - \frac{d}{2}).
			\end{aligned}
			\]
			与给定式 $3n-2$ 比较。
			\par\textbf{理由：} 将平均表达式写成一次函数形式便于比较系数。
		\item[\textbf{第二步（系数比较）：}] 对比一次项系数与常数项：$\frac{d}{2} = 3$，$a_1 - \frac{d}{2} = -2$。
			\par\textbf{理由：} 两个一次多项式相等，对应系数相等。
		\item[\textbf{第三步（求解）：}] 由 $\frac{d}{2}=3$ 得 $d=6$；代入 $a_1 - \frac{d}{2} = -2$ 得 $a_1 = -2 + 3$，即 $a_1 = 1$。
			\par\textbf{理由：} 解一元一次方程。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	公差 $d=6$，首项 $a_1=1$。

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 $\{a_n\}$ 中，$\frac{S_5}{5} = 7$，$\frac{S_{10}}{10} = 12$，求数列 $\{a_n\}$ 的公差 $d$ 和通项公式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（求平均公差）：}] 数列 $\{\frac{S_n}{n}\}$ 的公差
			\[
				\begin{aligned}
					d_{\text{avg}} &= \frac{\frac{S_{10}}{10} - \frac{S_5}{5}}{10-5} \\
					&= \frac{12-7}{5} \\
					&= 1.
			\end{aligned}
			\]
			由 $d_{\text{avg}} = \frac{d}{2}$ 得 $d=2$。
			\par\textbf{理由：} 等差数列公差定义：两项差除以项数差。
		\item[\textbf{第二步（求首项）：}] 由
			\[
				\begin{aligned}
					\frac{S_5}{5} &= a_1 + \frac{5-1}{2}d \\
					&= a_1 + 2d,
			\end{aligned}
			\]
			代入 $d=2$ 得 $a_1 + 4 = 7$，所以 $a_1=3$。
			\par\textbf{理由：} 代入平均表达式解方程。
		\item[\textbf{第三步（写通项）：}] 通项公式
			\[
				\begin{aligned}
					a_n &= a_1 + (n-1)d \\
					&= 3 + 2(n-1) \\
					&= 2n+1.
			\end{aligned}
			\]
			\par\textbf{理由：} 等差数列通项公式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	公差 $d=2$，通项 $a_n = 2n+1$。
\end{examplebox}
