\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{累加法：将递推差累加为求和，直接写出通项公式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（递推形式）：} 数列满足 $a_{n+1}=a_n+f(n)$（或 $a_{n+1}-a_n=f(n)$）。
		\item \textbf{条件2（求和能力）：} $f(n)$ 可求和，包括等差、等比、裂项、多项式等常见形式。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（下标范围）：} 求和为 $\sum_{k=1}^{n-1} f(k)$，不是 $k=1$ 到 $n$。
		\item \textbf{检查项（首项检验）：} 代入 $n=1$ 检验公式是否自动成立（空和为0），若不成立需单独说明。
	\end{itemize}
\end{summarybox}
