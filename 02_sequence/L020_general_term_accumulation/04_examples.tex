\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{a_n\}$ 满足 $a_1=1$，$a_{n+1}=a_n+2n+1$（$n\in\mathbb{N}^*$），求 $a_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别递推类型）：}] 由 $a_{n+1}=a_n+2n+1$ 得 $a_{n+1}-a_n=2n+1$，是 $f(n)=2n+1$ 的累加型。
			\par\textbf{理由：} 符合结论中 $a_{n+1}=a_n+f(n)$ 的形式。
		\item[\textbf{第二步（套用累加公式）：}] 由结论直接写出
			\[
				a_n = a_1 + \sum_{k=1}^{n-1} (2k+1)
			\]
			\par\textbf{理由：} 累加法结论 $a_n=a_1+\sum_{k=1}^{n-1}f(k)$。
		\item[\textbf{第三步（计算求和并化简）：}] 计算
			\[
				\begin{aligned}
					\sum_{k=1}^{n-1} (2k+1) &= 2\cdot\frac{(n-1)n}{2} + (n-1) \\
					&= n(n-1)+(n-1) \\
					&= n^2-1
			\end{aligned}
			\]
			因此 $a_n=1+(n^2-1)=n^2$。
			\par\textbf{理由：} 等差数列求和公式与常数项求和。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_n=n^2$}。

	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 $\{a_n\}$ 满足 $a_1=2$，$a_{n+1}=a_n+3^n$（$n\in\mathbb{N}^*$），求 $a_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别递推类型）：}] 由 $a_{n+1}=a_n+3^n$ 得 $a_{n+1}-a_n=3^n$，是 $f(n)=3^n$ 的累加型。
			\par\textbf{理由：} 符合 $a_{n+1}=a_n+f(n)$ 的形式。
		\item[\textbf{第二步（套用累加公式）：}] 由结论并代入$a_1=2$得
			\[
				a_n = 2 + \sum_{k=1}^{n-1} 3^k
			\]
			\par\textbf{理由：} 累加法结论。
		\item[\textbf{第三步（计算等比求和并化简）：}] 求等比数列和并代入：
			\[
				\begin{aligned}
					\sum_{k=1}^{n-1} 3^k &= \frac{3^n-3}{2} \\[4pt]
					a_n &= 2 + \frac{3^n-3}{2} \\
					&= \frac{3^n+1}{2}
			\end{aligned}
			\]
			\par\textbf{理由:} 等比数列前 $m$ 项和公式,并化简代数式.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论:}}
	\textbf{$a_n=\dfrac{3^n+1}{2}$}.

	\medskip
	\textbf{\textcolor{CExample}{例3(综合应用)}}

	\textbf{\textcolor{CExample}{题目:}}
	已知数列 $\{a_n\}$ 满足 $a_1=1$,$a_{n+1}=a_n+\dfrac{1}{n(n+1)}$($n\in\mathbb{N}^*$),求 $a_n$.

	\textbf{\textcolor{CExample}{解题步骤:}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步(裂项变形):}] 将 $f(n)=\dfrac{1}{n(n+1)}$ 裂项为 $\dfrac{1}{n}-\dfrac{1}{n+1}$,即
			\[
				a_{n+1}-a_n = \frac{1}{n} - \frac{1}{n+1}
			\]
			\par\textbf{理由:} 裂项公式 $\dfrac{1}{n(n+1)}=\dfrac{1}{n}-\dfrac{1}{n+1}$,便于求和.
		\item[\textbf{第二步(累加求和):}] 由累加法结论,对 $k=1$ 到 $n-1$ 求和:
			\[
				a_n = a_1 + \sum_{k=1}^{n-1} \left(\frac{1}{k}-\frac{1}{k+1}\right)
			\]
			右边求和:
			\[
				\sum_{k=1}^{n-1} \left(\frac{1}{k}-\frac{1}{k+1}\right) = 1-\frac{1}{n}
			\]
			\par\textbf{理由:} 裂项相消,中间项全部抵消.
		\item[\textbf{第三步(得到通项):}] 因此
			\[
				\begin{aligned}
					a_n &= 1 + \left(1-\frac{1}{n}\right) \\
					&= 2-\frac{1}{n}
			\end{aligned}
			\]
			\par\textbf{理由:} 代数化简.
	\end{description}

	\textbf{\textcolor{CExample}{关键结论:}}
	\textbf{$a_n=2-\dfrac{1}{n}$}.
\end{examplebox}
