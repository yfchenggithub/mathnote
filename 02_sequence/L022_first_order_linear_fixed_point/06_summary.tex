\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{一阶线性递推减去不动点，构造等比、快速求通项。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（递推形式）：} $a_{n+1}=p a_n+q$，$p,q$ 为常数。
		\item \textbf{条件 2（系数非1）：} $p\neq 1$（否则为等差数列）。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（p值检查）：} 先确认 $p\neq 1$，再计算 $\lambda$。
		\item \textbf{检查项（首项核对）：} 新数列首项是 $a_1-\lambda$，不是 $a_1$。
	\end{itemize}
\end{summarybox}
