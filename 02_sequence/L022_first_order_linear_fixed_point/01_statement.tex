\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（线性递推）：}}] 数列 $\{a_n\}$ 满足 $a_{n+1}=p a_n+q$，其中 $p,q$ 为常数，且 $p\neq 1$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（构造等比）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			从每一项减去不动点，新数列成等比。
			\[
				b_n = a_n - \lambda,
				\qquad
				b_{n+1} = p b_n
			\]

		\item[\textbf{\textcolor{CStatement}{结论二（通项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			等比型公式加回不动点。
			\[
				a_n = p^{\,n-1}\bigl(a_1-\frac{q}{1-p}\bigr)+\frac{q}{1-p}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（$p=1$ 情况）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			此时递推变为等差数列，直接求通项。
			\[
				a_n = a_1 + (n-1)q
			\]
	\end{description}
\end{statementbox}
