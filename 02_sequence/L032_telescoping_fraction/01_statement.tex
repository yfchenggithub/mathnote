\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（分母结构）：}}] $n\in\mathbb{N}^*$，常数 $k\neq0$（通常为正整数），且分母为两个关于 $n$ 的一次因式 $n$ 与 $n+k$ 的乘积，分子为常数（通常为 $1$）。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（裂项公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			将分式拆成两个简单分式之差，系数为 $\frac{1}{k}$。
			\[
				\frac{1}{n(n+k)} = \frac{1}{k}\left(\frac{1}{n} - \frac{1}{n+k}\right).
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（分子系数）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			若分子为常数 $A$，则裂项结果整体乘以 $A$。
			\[
				\frac{A}{n(n+k)} = \frac{A}{k}\left(\frac{1}{n} - \frac{1}{n+k}\right).
			\]
	\end{description}
\end{statementbox}
