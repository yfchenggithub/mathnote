\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（遗漏系数）}}] 错误地将 $\frac{1}{n(n+k)}$ 直接裂项为 $\frac{1}{n} - \frac{1}{n+k}$，丢弃了系数 $\frac{1}{k}$。
		\item[\textbf{\textcolor{CTrap}{正确理解（系数乘倒）：}}] 正确裂项为 $\frac{1}{k}\bigl(\frac{1}{n} - \frac{1}{n+k}\bigr)$，系数 $\frac{1}{k}$ 不可省略，因为通分后分子差为 $k$。
		\item[\textbf{\textcolor{CTrap}{错因分析（特例泛化）：}}] 学生记牢了 $k=1$ 时 $\frac{1}{n(n+1)}=\frac{1}{n}-\frac{1}{n+1}$ 的形式，误以为 $k\neq1$ 时形式相同，忽略了通分带来的系数。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（求和剩余项）}}] 当 $k>1$ 时，误认为求和只会剩下首项和末项，例如对 $\sum_{i=1}^n\frac{1}{i(i+2)}$ 直接得到 $\frac12(1-\frac{1}{n+2})$，实际上剩余前 $2$ 项和后 $2$ 项的部分。
		\item[\textbf{\textcolor{CTrap}{正确理解（展开验证）：}}] 必须展开至少前几项和末几项，观察抵消规律。对于一般的 $k$，剩余项为前 $k$ 个正项与后 $k$ 个负项。
		\item[\textbf{\textcolor{CTrap}{错因分析（经验主义）：}}] 将 $k=1$ 时仅剩首尾的简化情况错误地推广到一般 $k$，没有通过具体展开验证。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（分子系数遗漏）}}] 当分子为 $A$（$A\neq1$）时，裂项时未将 $A$ 包含在结果中，或错误地将 $A$ 放入分母。
		\item[\textbf{\textcolor{CTrap}{正确理解（整体相乘）：}}] 正确结果为 $A\cdot\frac{1}{k}\bigl(\frac{1}{n} - \frac{1}{n+k}\bigr) = \frac{A}{k}\bigl(\frac{1}{n} - \frac{1}{n+k}\bigr)$，分子 $A$ 整体乘到裂项结果前。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽视常数）：}}] 由于标准公式分子为 $1$，学生常默认为分子是 $1$，遇到分子为常数时忽略将其乘入。
	\end{description}
\end{trapbox}
