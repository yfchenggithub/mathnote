\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（公式系数遗忘）}}]
			误以为等差中项公式为 $a_n = a_{n-1}+a_{n+1}$，而漏写系数2。
		\item[\textbf{\textcolor{CTrap}{正确理解（公式正确形式）：}}]
			正确公式为 $2a_n = a_{n-1}+a_{n+1}$，即 $a_n = \frac{a_{n-1}+a_{n+1}}{2}$。
		\item[\textbf{\textcolor{CTrap}{错因分析（概念混淆）：}}]
			对等差中项概念理解不深，误将等差中项与通项公式类比，造成系数遗漏。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（局部误推整体）}}]
			由某三项满足 $2a_n = a_{n-1}+a_{n+1}$ 就判断整个数列为等差数列。
		\item[\textbf{\textcolor{CTrap}{正确理解（整体条件）：}}]
			该等式对单组三项成立只能说明那三项成等差；要推出数列等差，必须对所有 $n\ge 2$ 都成立。
		\item[\textbf{\textcolor{CTrap}{错因分析（条件泛化）：}}]
			忽略了等差中项法判定等差需要“任意 $n$”的条件，将局部性质当作整体性质。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（下标范围忽视）}}]
			在 $n=1$ 时使用公式，导致涉及 $a_0$ 或 $a_n$ 未定义。
		\item[\textbf{\textcolor{CTrap}{正确理解（下标限制）：}}]
			公式 $2a_n = a_{n-1}+a_{n+1}$ 仅在 $n\ge 2$ 时有意义，因为需保证 $a_{n-1}$ 和 $a_{n+1}$ 存在。
		\item[\textbf{\textcolor{CTrap}{错因分析（机械套用）：}}]
			没有注意下标范围，随意使用公式，导致下标越界错误。
	\end{description}

\end{trapbox}
