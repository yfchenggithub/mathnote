\begin{explanationbox}
	\textbf{\textcolor{CExplanation}{一句话直觉}}
	连续三项成等差，则中间项等于前后两项的算术平均数。

	\medskip
	\textbf{\textcolor{CExplanation}{核心拆解}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{要点一（等差定义）：}]
			由 $a_{n-1},a_n,a_{n+1}$ 成等差数列，得 $a_n - a_{n-1} = a_{n+1} - a_n$。
		\item[\textbf{要点二（移项化简）：}]
			移项合并得 $2a_n = a_{n-1}+a_{n+1}$，即 $a_n = \frac{a_{n-1}+a_{n+1}}{2}$。
	\end{description}

	\medskip
	\textbf{\textcolor{CExplanation}{几何本质（数轴等距）}}
	在数轴上，若连续三项成等差，则对应点等距，$a_n$ 恰好处于 $a_{n-1}$ 与 $a_{n+1}$ 的中点：$a_n = \frac{a_{n-1}+a_{n+1}}{2}$。

	\medskip
	\textbf{\textcolor{CExplanation}{代数意义（算术平均）}}
	等差中项 $a_n$ 等于前后两项 $a_{n-1}$ 与 $a_{n+1}$ 的算术平均数。这是一条纯代数性质。

	\medskip
	\textbf{\textcolor{CExplanation}{考点价值（高考常考）}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{考法一（等差中项求值）：} 已知等差数列中连续三项的首尾两项（如 $a_{n-1}$ 和 $a_{n+1}$），可直接利用等差中项公式求出中间项。
		\item \textbf{考法二（证明等差数列）：} 在已知数列中，若能证明 $2a_n = a_{n-1}+a_{n+1}$ 对所有 $n\ge 2$ 成立，即可判定为等差数列。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CExplanation}{顿悟点}}
	\textbf{等差中项是等差数列的“局部缩影”：只要相邻三项满足该关系，就具有等差性质；若全局满足，则整个数列等差。}

	\medskip
	\textbf{\textcolor{CExplanation}{使用场景}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{场景一（求中间项）：} 已知 $a_{n-1}$ 和 $a_{n+1}$，直接使用 $a_n = \frac{a_{n-1}+a_{n+1}}{2}$。
		\item \textbf{场景二（判定等差）：} 要证明数列等差，设法验证 $2a_n = a_{n-1}+a_{n+1}$ 对所有 $n\ge 2$ 成立。
	\end{itemize}

\end{explanationbox}
