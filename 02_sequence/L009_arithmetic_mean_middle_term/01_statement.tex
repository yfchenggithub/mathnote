\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（三项成等差）：}}]
			存在 $n\ge 2$，$n\in\mathbb{N}^{*}$，使得数列 $\{a_n\}$ 的三项 $a_{n-1},a_n,a_{n+1}$ 成等差数列。
	\end{description}
	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（等差中项）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			中间项等于前项加后项的一半，即算术平均数。
			\[
				2a_n = a_{n-1} + a_{n+1}.
			\]
		\item[\textbf{\textcolor{CStatement}{常见变形（等差充要）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			每项都满足等差中项则数列为等差数列。
			\[
				\forall n\ge 2,\; 2a_n = a_{n-1}+a_{n+1}.
			\]
	\end{description}

\end{statementbox}
