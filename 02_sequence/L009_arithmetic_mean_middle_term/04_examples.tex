\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}
	\textbf{\textcolor{CExample}{题目：}} 已知等差数列 $\{a_n\}$ 中，$a_1=3$，$a_3=9$，求 $a_2$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（判断等差）：}]
			由 $\{a_n\}$ 为等差数列，知 $a_1,a_2,a_3$ 成等差数列。
			\par\textbf{理由：} 等差数列任意连续三项成等差。
		\item[\textbf{第二步（套用公式）：}]
			由等差中项公式得 $2a_2 = a_1 + a_3$。
			\par\textbf{理由：} $2a_n = a_{n-1}+a_{n+1}$。
		\item[\textbf{第三步（计算求值）：}]
			代入数值 $2a_2 = 3 + 9 = 12$，故 $a_2=6$。
			\par\textbf{理由：} 代数运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$a_2=6$}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}
	\textbf{\textcolor{CExample}{题目：}} 三个数 $2-\sqrt{3}$，$m$，$2+\sqrt{3}$ 成等差数列，求 $m$。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（设定条件）：}]
			由三数成等差，得 $2-\sqrt{3},m,2+\sqrt{3}$ 满足等差中项关系。
			\par\textbf{理由：} 等差中项条件。
		\item[\textbf{第二步（建立方程）：}]
			由等差中项公式，$2m = (2-\sqrt{3}) + (2+\sqrt{3})$。
			\par\textbf{理由：} $2a_n = a_{n-1}+a_{n+1}$。
		\item[\textbf{第三步（化简求解）：}]
			$2m = 4$，解得 $m = 2$。
			\par\textbf{理由：} 代数运算。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$m=2$}

	\medskip
	\textbf{\textcolor{CExample}{例 3（综合应用）}}
	\textbf{\textcolor{CExample}{题目：}} 已知数列 $\{a_n\}$ 满足 $2a_{n+1}=a_n+a_{n+2}$ 对任意 $n\ge 1$ 成立，求证：$\{a_n\}$ 为等差数列。\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（改写条件）：}]
			条件即为 $2a_{n+1}=a_n+a_{n+2}$，与等差中项公式 $2a_n = a_{n-1}+a_{n+1}$ 形式一致。
			\par\textbf{理由：} 等差中项公式。
		\item[\textbf{第二步（变量替换）：}]
			令 $k=n+1$，则 $n=k-1$，条件化为 $2a_k = a_{k-1}+a_{k+1}$，其中 $k\ge 2$。
			\par\textbf{理由：} 将 $n$ 替换为 $k-1$，所有 $n\ge 1$ 对应 $k\ge 2$。
		\item[\textbf{第三步（判定等差）：}]
			由 $2a_k = a_{k-1}+a_{k+1}$ 对所有 $k\ge 2$ 成立，根据等差中项法，数列 $\{a_n\}$ 为等差数列。
			\par\textbf{理由：} 等差中项法是等差数列的判定定理：若 $\forall n\ge 2,\;2a_n=a_{n-1}+a_{n+1}$，则 $\{a_n\}$ 为等差数列。
	\end{description}
	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{数列 $\{a_n\}$ 为等差数列。}
\end{examplebox}
