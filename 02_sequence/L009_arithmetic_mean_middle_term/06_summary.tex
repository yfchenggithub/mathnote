\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{若数列中连续三项成等差，则中间项等于前后项的算术平均数：$2a_n = a_{n-1}+a_{n+1}$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（三项等差）：} 数列 $\{a_n\}$ 中存在 $n\ge 2$ 使 $a_{n-1},a_n,a_{n+1}$ 成等差数列。
		\item \textbf{条件 2（下标范围）：} $n\ge 2$ 且 $n\in\mathbb{N}^{*}$，保证前后项存在。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公式系数）：} 公式是 $2a_n = a_{n-1}+a_{n+1}$，不是 $a_n = a_{n-1}+a_{n+1}$，容易漏系数 $2$。
		\item \textbf{检查项（整体等差）：} 仅由某三项满足该等式不能推出全部等差；只有当对所有 $n\ge 2$ 成立时，数列才为等差数列。
	\end{itemize}

\end{summarybox}
