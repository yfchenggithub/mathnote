\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	\textbf{等比数列的间隔$m$项片段和（差）$S_m, S_{2m}-S_m, S_{3m}-S_{2m},\ldots$成等比数列，公比为$q^m$。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（公比限制）：} 公比$q\neq 1$，否则求和公式形式不同。
		\item \textbf{条件2（非零片段和）：} 第一个片段$S_m\neq 0$，否则分式无意义。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（混淆项）：} 注意是$S_m, S_{2m}-S_m, S_{3m}-S_{2m}$成等比，而不是$S_m, S_{2m}, S_{3m}$。
		\item \textbf{检查项（分母非零）：} 每次应用比值$\frac{S_{2m}-S_m}{S_m}$前，先确认$S_m\neq 0$。
	\end{itemize}
\end{summarybox}
