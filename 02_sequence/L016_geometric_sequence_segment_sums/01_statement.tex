\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（公比限制）：}}] 等比数列$\{a_n\}$满足公比$q\neq 1$。
		\item[\textbf{\textcolor{CStatement}{条件2（非零前m项和）：}}] 存在正整数$m$使得前$m$项和$S_m\neq 0$。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（片段等比性）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			间隔$m$项的片段和构成新等比数列，公比为$q^m$。
			\[
				S_m, S_{2m}-S_m, S_{3m}-S_{2m},\ldots
			\]
			成等比数列，即
			\[
				\frac{S_{2m}-S_m}{S_m}=q^m,
			\]
			\[
				\frac{S_{3m}-S_{2m}}{S_{2m}-S_m}=q^m.
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（退化情形）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当$m=1$时，片段和序列退化为原数列，结论回到等比定义。
			\[
				S_1, S_2-S_1, S_3-S_2,\ldots
			\]
			即$a_1, a_2, a_3,\ldots$，且
			\[
				\frac{a_{k+1}}{a_k}=q.
			\]
	\end{description}

	\textbf{\textcolor{CStatement}{适用范围：}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item 要求$q\neq 1$，否则求和公式$S_n=\frac{a_1(1-q^n)}{1-q}$失效。
		\item 要求$S_m\neq 0$，否则分式分母为零，结论无意义。
	\end{itemize}
\end{statementbox}
