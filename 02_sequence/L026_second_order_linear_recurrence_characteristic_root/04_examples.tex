\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 \(\{a_n\}\) 满足 \(a_1=1,a_2=3\)，且对任意 \(n\in\mathbb N^*\) 有
	\[
		a_{n+2}=3a_{n+1}-2a_n,
	\]
	求通项公式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写特征方程）：}]
		      由递推得到特征方程
		      \[
			      r^2-3r+2=0.
		      \]
		      \par\textbf{理由：} 对照标准形式 \(a_{n+2}=p a_{n+1}+q a_n\)，此处 \(p=3,q=-2\)，所以特征方程为 \(r^2-pr-q=0\)。

		\item[\textbf{第二步（求特征根）：}]
		      分解得
		      \[
			      (r-1)(r-2)=0,
		      \]
		      所以 \(r_1=1,\;r_2=2\)。
		      \par\textbf{理由：} 二次方程因式分解即可求根。

		\item[\textbf{第三步（通项并定系数）：}]
		      两根不等，设
		      \[
			      a_n=A\cdot1^n+B\cdot2^n=A+B\cdot2^n.
		      \]
		      代入 \(a_1=1,a_2=3\)，得
		      \[
			      \begin{cases}
				      A+2B=1, \\
				      A+4B=3.
			      \end{cases}
		      \]
		      解得 \(B=1,A=-1\)，所以
		      \[
			      a_n=2^n-1.
		      \]
		      \par\textbf{理由：} 利用两个初值建立二元一次方程组，解出两个待定常数。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{a_n=2^n-1}.
	\]

	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 \(\{F_n\}\) 满足 \(F_1=1,F_2=1\)，且
	\[
		F_{n+2}=F_{n+1}+F_n\quad(n\in\mathbb N^*),
	\]
	求通项公式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写特征方程）：}]
		      递推对应特征方程
		      \[
			      r^2-r-1=0.
		      \]
		      \par\textbf{理由：} 此处 \(p=1,q=1\)，代入 \(r^2-pr-q=0\) 即可。

		\item[\textbf{第二步（求特征根）：}]
		      由求根公式得
		      \[
			      r_{1,2}=\frac{1\pm\sqrt5}{2}.
		      \]
		      记
		      \[
			      \varphi=\frac{1+\sqrt5}{2},\qquad \psi=\frac{1-\sqrt5}{2}.
		      \]
		      \par\textbf{理由：} 判别式 \(\Delta=1+4=5\)。

		\item[\textbf{第三步（通项并定系数）：}]
		      设
		      \[
			      F_n=A\varphi^n+B\psi^n.
		      \]
		      代入 \(F_1=1,F_2=1\)，得
		      \[
			      \begin{cases}
				      A\varphi+B\psi=1, \\[4pt]
				      A\varphi^2+B\psi^2=1.
			      \end{cases}
		      \]
		      解得
		      \[
			      A=\frac1{\sqrt5},\qquad B=-\frac1{\sqrt5}.
		      \]
		      因此
		      \[
			      F_n=\frac1{\sqrt5}\left[\left(\frac{1+\sqrt5}{2}\right)^n-
				      \left(\frac{1-\sqrt5}{2}\right)^n\right].
		      \]
		      \par\textbf{理由：} 两个不同特征根对应两个指数项，两个初值确定两个系数。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{F_n=\frac1{\sqrt5}\left[\left(\frac{1+\sqrt5}{2}\right)^n-
				\left(\frac{1-\sqrt5}{2}\right)^n\right]}.
	\]

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知数列 \(\{a_n\}\) 满足 \(a_1=2,a_2=4\)，且
	\[
		a_{n+2}=2a_{n+1}-2a_n\quad(n\in\mathbb N^*),
	\]
	求通项公式。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（写特征方程）：}]
		      由递推得特征方程
		      \[
			      r^2-2r+2=0.
		      \]
		      \par\textbf{理由：} 此处 \(p=2,q=-2\)，代入 \(r^2-pr-q=0\) 得 \(r^2-2r+2=0\)。

		\item[\textbf{第二步（求根并化三角形式）：}]
		      解方程得
		      \[
			      r=1\pm i.
		      \]
		      又
		      \[
			      1+i=\sqrt2\,e^{i\frac{\pi}{4}},
			      \qquad
			      1-i=\sqrt2\,e^{-i\frac{\pi}{4}}.
		      \]
		      所以模长为 \(\sqrt2\)，辐角为 \(\frac{\pi}{4}\)。
		      \par\textbf{理由：} 复数 \(1\pm i\) 的模为 \(\sqrt2\)，辐角分别为 \(\pm\frac{\pi}{4}\)。

		\item[\textbf{第三步（通项并定系数）：}]
		      通项可写为
		      \[
			      a_n=(\sqrt2)^n\left(C_1\cos\frac{n\pi}{4}+C_2\sin\frac{n\pi}{4}\right).
		      \]
		      代入初值：
		      \[
			      \begin{aligned}
				      n=1: & \quad \sqrt2\left(C_1\frac{\sqrt2}{2}+C_2\frac{\sqrt2}{2}\right)=C_1+C_2=2, \\
				      n=2: & \quad 2\left(C_1\cos\frac{\pi}{2}+C_2\sin\frac{\pi}{2}\right)=2C_2=4.
			      \end{aligned}
		      \]
		      故 \(C_2=2,C_1=0\)，因此
		      \[
			      a_n=2(\sqrt2)^n\sin\frac{n\pi}{4}.
		      \]
		      \par\textbf{理由：} 利用 \(\cos\frac{\pi}{4}=\sin\frac{\pi}{4}=\frac{\sqrt2}{2}\)，\(\cos\frac{\pi}{2}=0\)，\(\sin\frac{\pi}{2}=1\) 简化计算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\[
		\boxed{a_n=2(\sqrt2)^n\sin\frac{n\pi}{4}}.
	\]
