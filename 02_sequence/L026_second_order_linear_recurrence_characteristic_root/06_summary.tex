\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}
	\textbf{二阶线性递推的通项公式可由特征方程的两个根直接写出，包括指数组合、多项式乘指数或三角形式。}

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件1（递推形式）：} 数列$\{a_n\}$满足$a_{n+2}=p a_{n+1}+q a_n$，其中$p,q$为常数且$q\neq0$。
		\item \textbf{条件2（初始值）：} 必须已知两个连续项（例如$a_1,a_2$），才能唯一确定常数$C_1,C_2$。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（特征方程符号）：} 特征方程是$r^2-p r-q=0$，不是$r^2-p r+q=0$，注意常数项为$-q$。
		\item \textbf{检查项（重根含线性项）：} 当特征根相等时，通项必须是$(C_1+nC_2)\lambda^n$，不可漏掉$n$乘项。
	\end{itemize}

\end{summarybox}
