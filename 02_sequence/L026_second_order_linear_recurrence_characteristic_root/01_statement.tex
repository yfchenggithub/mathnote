\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（递推关系）：}}]
		      数列 \(\{a_n\}\) 满足
		      \[
			      a_{n+2}=p a_{n+1}+q a_n\quad (n\in\mathbb N^*),
		      \]
		      其中 \(p,q\) 为实常数，且 \(q\neq0\)，初值 \(a_1,a_2\) 已知。
		\item[\textbf{\textcolor{CStatement}{条件2（特征方程）：}}]
		      称方程
		      \[
			      r^2-pr-q=0
		      \]
		      为特征方程，其根 \(\lambda_1,\lambda_2\) 称为特征根。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（通项公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      通项由特征根的取值情况决定，通常分为不等实根、相等实根、共轭复根三种情形。
		      \[
			      \boxed{
				      a_n=
				      \begin{cases}
					      C_1\lambda_1^{\,n}+C_2\lambda_2^{\,n},
					       & \lambda_1\neq\lambda_2\text{，且二者为实根},        \\[4pt]
					      (C_1+nC_2)\lambda^{\,n},
					       & \lambda_1=\lambda_2=\lambda,                 \\[4pt]
					      r^{\,n}\bigl(C_1\cos n\theta+C_2\sin n\theta\bigr),
					       & \lambda_{1,2}=re^{\pm i\theta}\text{，且为非实根}.
				      \end{cases}}
		      \]
		      其中 \(C_1,C_2\) 由初值 \(a_1,a_2\) 解线性方程组唯一确定。

		\item[\textbf{\textcolor{CStatement}{推广结论（根与系数关系）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      特征方程的两根之和等于 \(p\)，两根之积等于 \(-q\)，可快速检验或计算系数。
		      \[
			      \lambda_1+\lambda_2=p,
			      \qquad
			      \lambda_1\lambda_2=-q.
		      \]
	\end{description}
\end{statementbox}
