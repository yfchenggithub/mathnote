\begin{trapbox}

	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（特征方程符号）}}]
			写特征方程时，误写为$r^2-pr+q=0$或$r^2+pr-q=0$，导致根错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（标准形式）：}}]
			由递推$a_{n+2}=p a_{n+1}+q a_n$，代入$r^n$约去$r^n$后得$r^2 = p r + q$，移项即$r^2-p r-q=0$，所以常数项为$-q$。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号混淆）：}}]
			对公式记忆不牢，或移项时没有变号，导致符号错误。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（重根漏多项式）}}]
			当特征根相等时，通项漏写$n$倍项，只写为$a_n = C\lambda^n$。
		\item[\textbf{\textcolor{CTrap}{正确理解（等根通项结构）：}}]
			两根相等时，通项必须是$(C_1 + n C_2)\lambda^n$，否则缺少一个线性无关解。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略线性独立）：}}]
			学生认为重根只需一个解，实际递推是二阶，需要两个独立常数，必须引入$n\lambda^n$。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（复根未化实）}}]
			特征根为共轭复数时，通项写成$C_1\lambda_1^n+C_2\lambda_2^n$（含虚数$i$），没有转化为实数三角形式，导致后续计算困难。
		\item[\textbf{\textcolor{CTrap}{正确理解（化实三角形式）：}}]
			当$\lambda_{1,2}=re^{\pm i\theta}$，通项可转化为$a_n = r^n(C_1\cos n\theta + C_2\sin n\theta)$，其中$C_1,C_2$为实数，便于运算。
		\item[\textbf{\textcolor{CTrap}{错因分析（复数处理不当）：}}]
			不熟悉欧拉公式或不清楚$C_1,C_2$的共轭关系，直接保留复数指数形式，无法应用初始条件确定常数。
	\end{description}

\end{trapbox}
