\begin{trapbox}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点一（特征方程符号）}}]
		      写特征方程时，误写为 \(r^2-pr+q=0\) 或 \(r^2+pr-q=0\)，导致特征根错误。
		\item[\textbf{\textcolor{CTrap}{正确理解（标准形式）：}}]
		      由递推 \(a_{n+2}=p a_{n+1}+q a_n\)，代入 \(a_n=r^n\) 后得
		      \[
			      r^2=pr+q,
		      \]
		      移项即
		      \[
			      r^2-pr-q=0.
		      \]
		      所以常数项是 \(-q\)。
		\item[\textbf{\textcolor{CTrap}{错因分析（符号混淆）：}}]
		      对递推式移项时没有变号，或者把递推中的 \(+q a_n\) 机械地写成特征方程中的 \(+q\)。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点二（重根漏多项式）}}]
		      当特征根相等时，通项漏写 \(n\) 倍项，只写成 \(a_n=C\lambda^n\)。
		\item[\textbf{\textcolor{CTrap}{正确理解（等根通项结构）：}}]
		      两根相等时，通项必须写成
		      \[
			      a_n=(C_1+nC_2)\lambda^n,
		      \]
		      否则缺少一个线性无关解，无法匹配两个初值。
		\item[\textbf{\textcolor{CTrap}{错因分析（忽略线性独立）：}}]
		      二阶递推需要两个独立常数。重根虽然只有一个特征根，但仍要通过 \(n\lambda^n\) 补出第二个独立解。
	\end{description}

	\medskip
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CTrap}{易错点三（复根未化实）}}]
		      特征根为共轭复数时，通项只写成 \(C_1\lambda_1^n+C_2\lambda_2^n\)，没有化为实数三角形式，后续代入初值容易出错。
		\item[\textbf{\textcolor{CTrap}{正确理解（化实三角形式）：}}]
		      当 \(\lambda_{1,2}=re^{\pm i\theta}\) 时，实数数列的通项可写成
		      \[
			      a_n=r^n\bigl(C_1\cos n\theta+C_2\sin n\theta\bigr),
		      \]
		      其中 \(C_1,C_2\) 为实数。
		\item[\textbf{\textcolor{CTrap}{错因分析（复数处理不当）：}}]
		      不熟悉欧拉公式，或不清楚共轭复根最终可以合并成实数的正弦、余弦组合。
	\end{description}
\end{trapbox}
