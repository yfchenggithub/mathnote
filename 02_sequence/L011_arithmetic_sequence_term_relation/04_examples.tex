\begin{examplebox}
	\textbf{\textcolor{CExample}{例1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 \(\{a_n\}\) 中，\(a_3=7\)，公差 \(d=4\)，求 \(a_8\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（认定条件）：}] 数列是等差数列，已知 \(a_3=7\)，\(d=4\)，\(m=3\)，\(n=8\)。
		      \par\textbf{理由：} 题目直接给出了这些条件。
		\item[\textbf{第二步（套用公式）：}] 由公式 \(a_n = a_m + (n-m)d\) 得
		      \[
			      a_8 = 7 + (8-3)\times 4 .
		      \]
		      \par\textbf{理由：} 公式适用于任意等差数列的两项关系，这里 \(m=3,n=8\)。
		\item[\textbf{第三步（计算答案）：}] 化简得 \(a_8 = 7 + 5\times 4 = 27\)。
		      \par\textbf{理由：} 算术运算。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{\(a_8 = 27\)}。

	\medskip
	\textbf{\textcolor{CExample}{例2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 \(\{a_n\}\) 中，\(a_5=12\)，\(a_9=28\)，求 \(a_1\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（建立方程）：}] 设公差为 \(d\)，由公式得 \(a_9 = a_5 + (9-5)d\)，即 \(28 = 12 + 4d\)。
		      \par\textbf{理由：} 利用已知两项的关系建立方程。
		\item[\textbf{第二步（求解公差）：}] 解方程 \(4d = 16\)，得 \(d = 4\)。
		      \par\textbf{理由：} 等式两边同除以4。
		\item[\textbf{第三步（代回求首项）：}] 利用公式 \(a_5 = a_1 + (5-1)d\) 得 \(12 = a_1 + 4\times 4\)，所以 \(a_1 = 12 - 16 = -4\)。
		      \par\textbf{理由：} 将 \(a_1\) 视为起点项，\(a_5\) 视为目标项，令 \(m=1,n=5\) 再次使用结论。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{\(a_1 = -4\)}，通项公式为 \(a_n = -4 + (n-1)\times 4\)。

	\medskip
	\textbf{\textcolor{CExample}{例3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	已知等差数列 \(\{a_n\}\) 中，\(a_m = n\)，\(a_n = m\)（\(m\neq n\)），求 \(a_{m+n}\)。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（列出关系）：}] 由公式 \(a_n = a_m + (n-m)d\) 得 \(m = n + (n-m)d\)。
		      \par\textbf{理由：} 将已知值代入结论。
		\item[\textbf{第二步（化简求解）：}] 整理得 \((n-m)d = m - n = -(n-m)\)，因为 \(m\neq n\)，两边同除以 \(n-m\) 得 \(d = -1\)。
		      \par\textbf{理由：} 等式变形，分母非零。
		\item[\textbf{第三步（计算目标）：}] 再次使用公式 \(a_{m+n} = a_m + (m+n - m)d = n + n\cdot(-1) = 0\)。
		      \par\textbf{理由：} 代入已知 \(a_m\) 和已求公差。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{\(a_{m+n} = 0\)}。
\end{examplebox}
