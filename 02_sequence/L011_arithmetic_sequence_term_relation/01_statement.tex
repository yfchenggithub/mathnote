\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件1（数列定义）：}}] 数列 \(\{a_n\}\) 是等差数列，公差为 \(d\)。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（任意项公式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      任何一项可由任意另一项加上差倍得到。
		      \[
			      a_n = a_m + (n-m)d .
		      \]

		\item[\textbf{\textcolor{CStatement}{推广结论（差的形式）：}}]
		      \textbf{\textcolor{CStatement}{直观描述：}}
		      任意两项之差等于公差乘以项数差。
		      \[
			      a_n - a_m = (n-m)d .
		      \]
	\end{description}
\end{statementbox}
