\begin{examplebox}
	\textbf{\textcolor{CExample}{例 1（基础）}}

	\textbf{\textcolor{CExample}{题目：}}
	求数列 $\{n\cdot 2^{n-1}\}$ 的前 $n$ 项和 $S_n$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			\[
				a_1 = 1,\ d = 1;\quad b_1 = 1,\ q = 2.
			\]
			\par\textbf{理由：} 由通项 $a_n = n$, $b_n = 2^{n-1}$ 直接读出。

		\item[\textbf{第二步（套用公式）：}]
			公式：
			\[
				S_n = \frac{a_1 b_1 - a_n b_n q}{1-q} + \frac{d\,b_1 q\,(1-q^{\,n-1})}{(1-q)^2}.
			\]
			代入 $a_1=1$, $b_1=1$, $a_n=n$, $b_n=2^{n-1}$, $q=2$, $d=1$，得
			\[
				S_n = \frac{1\cdot1 - n\cdot2^{n-1}\cdot2}{1-2} + \frac{1\cdot1\cdot2\,(1-2^{\,n-1})}{(1-2)^2}.
			\]
			\par\textbf{理由：} 代入参数到结论公式。

		\item[\textbf{第三步（化简求值）：}]
			\[
				\begin{aligned}
					S_n &= \frac{1 - n\cdot2^{n}}{-1} + \frac{2\,(1-2^{\,n-1})}{1} \\
					&= (n\cdot2^{n} - 2^{n}) + 1 \\
					&= (n-1)2^{n} + 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 分母为 $-1$ 和 $1$，直接运算，合并同类项。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = (n-1)2^{n} + 1$。}

	\medskip
	\textbf{\textcolor{CExample}{例 2（稍变形）}}

	\textbf{\textcolor{CExample}{题目：}}
	求 $S_n = \sum_{k=1}^{n} (2k-1)3^{k-1}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			\[
				a_n = 2n-1,\ a_1 = 1,\ d = 2;\quad b_n = 3^{n-1},\ b_1 = 1,\ q = 3.
			\]
			\par\textbf{理由：} 通项对照。

		\item[\textbf{第二步（套用公式）：}]
			\[
				S_n = \frac{1\cdot1 - (2n-1)\cdot3^{n-1}\cdot3}{1-3} + \frac{2\cdot1\cdot3\,(1-3^{\,n-1})}{(1-3)^2}.
			\]
			\par\textbf{理由：} 代入公式。

		\item[\textbf{第三步（化简求值）：}]
			\[
				\begin{aligned}
					S_n &= \frac{1 - (2n-1)3^{n}}{-2} + \frac{6\,(1-3^{\,n-1})}{4} \\
					&= \frac{(2n-1)3^{n} - 1}{2} + \frac{3\,(1-3^{\,n-1})}{2} \\
					&= \frac{(2n-1)3^{n} - 1}{2} + \frac{3 - 3^{n}}{2} \\
					&= \frac{(2n-2)3^{n} + 2}{2} \\
					&= (n-1)3^{n} + 1.
			\end{aligned}
			\]
			\par\textbf{理由：} 分母计算，通分，合并。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = (n-1)3^{n} + 1$。}

	\textbf{\textcolor{CExample}{例 3（综合应用）}}

	\textbf{\textcolor{CExample}{题目：}}
	求 $S_n = \sum_{k=1}^{n} (3k-2)(-2)^{k-1}$。

	\textbf{\textcolor{CExample}{解题步骤：}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{第一步（识别参数）：}]
			\[
				a_n = 3n-2,\ a_1 = 1,\ d = 3;\quad b_n = (-2)^{n-1},\ b_1 = 1,\ q = -2.
			\]
			\par\textbf{理由：} 通项匹配。

		\item[\textbf{第二步（套用公式）：}]
			\[
				S_n = \frac{1\cdot1 - (3n-2)\cdot(-2)^{n-1}\cdot(-2)}{1-(-2)} + \frac{3\cdot1\cdot(-2)\,(1-(-2)^{\,n-1})}{(1-(-2))^2}.
			\]
			\par\textbf{理由：} 代入参数。

		\item[\textbf{第三步（化简求值）：}]
			\[
				\begin{aligned}
					S_n &= \frac{1 - (3n-2)(-2)^{n}}{3} + \frac{-6\,(1-(-2)^{\,n-1})}{9} \\
					&= \frac{1}{3} - \frac{(3n-2)(-2)^{n}}{3} \\
					&\quad - \frac{2}{3}\bigl(1-(-2)^{\,n-1}\bigr) \\
					&= -\frac{1}{3} - \frac{(3n-2)(-2)^{n}}{3} \\
					&\quad + \frac{2}{3}(-2)^{\,n-1} \\
					&= -\frac{1}{3} + \frac{(-2)^{\,n-1}}{3}\bigl(-(3n-2)(-2) + 2\bigr) \\
					&= -\frac{1}{3} + \frac{2(3n-1)}{3}(-2)^{\,n-1}.
			\end{aligned}
			\]
			\par\textbf{理由：} 逐项化简，提取公因式。
	\end{description}

	\textbf{\textcolor{CExample}{关键结论：}}
	\textbf{$S_n = -\frac{1}{3} + \frac{2(3n-1)}{3}(-2)^{\,n-1}$。}
\end{examplebox}
