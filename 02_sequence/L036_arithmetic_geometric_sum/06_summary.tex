\begin{summarybox}
	\textbf{\textcolor{CSummary}{一句话核心}}

	等差乘等比前 $n$ 项和可直接用错位相减公式 $S_n = \frac{a_1 b_1 - a_n b_n q}{1-q} + \frac{d\,b_1 q\,(1-q^{\,n-1})}{(1-q)^2}$ 计算，规避逐项推算。

	\medskip
	\textbf{\textcolor{CSummary}{使用条件}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{条件 1（等差等比存在）：} 数列 $\{a_n\}$ 是等差数列，$\{b_n\}$ 是等比数列。
		\item \textbf{条件 2（公比非一）：} 公比 $q \neq 1$，否则分母为 $0$，需要另行处理。
	\end{itemize}

	\medskip
	\textbf{\textcolor{CSummary}{关键提醒}}
	\begin{itemize}[leftmargin=*, itemsep=0.5em, topsep=0.4em]
		\item \textbf{易错点（公比检验）：} 每次使用前先验证 $q \neq 1$，避免无意义运算。
		\item \textbf{检查项（指数与符号）：} 确认第二项指数为 $n-1$，且 $q$ 为负数时整体加括号。
	\end{itemize}
\end{summarybox}
