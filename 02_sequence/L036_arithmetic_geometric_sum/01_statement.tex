\begin{statementbox}
	\textbf{\textcolor{CStatement}{条件}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.35em]
		\item[\textbf{\textcolor{CStatement}{条件 1（等差等比存在）：}}] 存在等差数列 $\{a_n\}$，首项 $a_1$，公差 $d$；等比数列 $\{b_n\}$，首项 $b_1$，公比 $q$（$q \neq 1$）。
		\item[\textbf{\textcolor{CStatement}{条件 2（公式适用前提）：}}] 公比 $q \neq 1$，否则分母为零，公式无效。
	\end{description}

	\textbf{\textcolor{CStatement}{结论}}
	\begin{description}[style=nextline, leftmargin=2.8em, labelsep=0.5em, itemsep=0.45em]
		\item[\textbf{\textcolor{CStatement}{结论一（错位相减公式）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			直接计算等差乘等比前 $n$ 项和，无需逐步错位相减。
			\[
				\begin{aligned}
					S_n &= \sum_{k=1}^{n} a_k b_k \\
					&= \frac{a_1 b_1 - a_n b_n q}{1-q} + \frac{d\,b_1 q\,(1-q^{\,n-1})}{(1-q)^2}.
			\end{aligned}
			\]

		\item[\textbf{\textcolor{CStatement}{推广结论（常数列特例）：}}]
			\textbf{\textcolor{CStatement}{直观描述：}}
			当等差数列公差为零时，公式退化为等比数列求和。
			\[
				d = 0,\quad S_n = a_1 b_1 \cdot \frac{1-q^{\,n}}{1-q}.
			\]
	\end{description}
\end{statementbox}
